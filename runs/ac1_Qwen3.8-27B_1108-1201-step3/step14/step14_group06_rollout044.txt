    # Build constraint matrix: for each active lag k, the constraint is
    # sum_j a[k-j] * b[j] <= M (where a[k-j] is valid, i.e., 0 <= k-j < n)
    n_constraints = len(active_lags)
    # Use sparse representation
    row_idx = []
    col_idx = []
    data = []
    for ci, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_idx.append(ci)
            col_idx.append(j)
            data.append(a[k - j])
    
    from scipy.sparse import csc_matrix
    T = csc_matrix((data, (row_idx, col_idx)), shape=(n_constraints, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_lags
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_lags
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, active_lags
    b_val = np.maximum(b_val, 0.0)
    
    # Check if violated constraints exist
    conv_ab = np.zeros(2 * n - 1)
    # Compute full conv(a, b) to check
    fft_size = 1 << (2 * n - 1).bit_length()
    A_fft = rfft(a, fft_size)
    B_fft = rfft(b_val, fft_size)
    conv_ab = irfft(A_fft * B_fft, fft_size)[:2*n-1]
    
    violated = np.where(conv_ab > M * 1.001)[0]
    
    if len(violated) > 0 and len(violated) < 50:
        # Add violated constraints and re-solve
        new_active = sorted(set(list(active_lags) + [int(v) for v in violated]))
        if len(new_active) <= max_lags + 50:
            return lp_best_response_lazy(a, M, solver, new_active, max_lags)
    
    s = b_val.sum()
    if s < 1e-15:
        return None, active_lags
    return b_val, active_lags

def multiplicative_weight_update(a, eta, n_steps=100):
    """Multiplicative weight update to reduce peak autocorrelation."""
    n = len(a)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * norm_val / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for step in range(n_steps):
        conv_a = fast_autocorr(a)
        k_star = int(np.argmax(conv_a))
        
        # Subgradient direction: 2 * a[k_star - i] (valid i)
        # Multiplicative update: a[i] *= exp(-eta * a[k_star - i] * indicator)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        
        # Compute the "gradient" for multiplicative update
        grad = np.zeros(n)
        for i in range(js, je + 1):
            grad[i] = a[k_star - i]
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        a = a * norm_val / s
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a

def improved_coord_descent(a0, time_limit, seed=42):
    """Improved coordinate descent with pair moves and line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_a = fast_autocorr(a)
        k_peak = int(np.argmax(conv_a))
        
        # Find top contributing elements to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]
        
        # Find low-value elements (good transfer targets)
        # Elements that don't contribute much to any peak
        peak_mask = np.zeros(n, dtype=bool)
        top_peaks = np.argsort(conv_a)[::-1][:20]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        low_val_targets = non_peak[np.argsort(a[non_peak])][:20]
        
        # Also try: elements that are at the "edges" of the sequence
        edge_targets = np.concatenate([np.arange(min(20, n//4)), np.arange(max(0, n-20), n)])
        edge_targets = edge_targets[~np.isin(edge_targets, top_contrib)]
        
        improved = False
        
        # Try single moves
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in list(low_val_targets) + list(edge_targets):
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    if amt > a[src]:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        # Try pair moves if single moves stalled
        if not improved:
            for _ in range(50):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                k2 = int(rng.integers(n))
                l = int(rng.integers(n))
                if len(set([i, j, k2, l])) < 3:
                    continue
                if a[i] < 1e-10 or a[k2] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.1)
                amt1 = a[i] * frac
                amt2 = a[k2] * frac
                a_trial = a.copy()
                a_trial[i] -= amt1
                a_trial[j] += amt1
                a_trial[k2] -= amt2
                a_trial[l] += amt2
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            # Random perturbation to escape local minimum
            for _ in range(100):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score

def lse_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=4):
    """LSE smoothed optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)
        
        def lse_obj_and_grad(x, T_val=T):
            af = np.maximum(x, 0.0)
            s = af.sum()
            if s < 1e-10:
                return 1e10, np.zeros(n)
            conv = fast_autocorr(af)
            max_conv = conv.max()
            if max_conv < 1e-15:
                return 1e10, np.zeros(n)
            shifted = (conv - max_conv) * T_val
            exp_vals = np.exp(shifted)
            exp_sum = exp_vals.sum()
            log_sum = max_conv + np.log(exp_sum) / T_val
            w = exp_vals / exp_sum
            
            # Gradient: d/dx_j of log_sum_exp(conv) = sum_k w[k] * d(conv[k])/dx[j]
            # d(conv[k])/dx[j] = 2*x[k-j] (if valid)
            pdf_size = 1 << (2 * n - 1).bit_length()
            w_full = np.zeros(pdf_size)
            w_full[:2*n-1] = w
            x_full = np.zeros(pdf_size)
            x_full[:n] = x
            
            W_fft = rfft(w_full, pdf_size)
            X_fft = rfft(x_full, pdf_size)
            # cross correlation: sum_k w[k] * x[k-j]
            cross = irfft(W_fft * np.conj(X_fft), pdf_size)[:n]
            
            obj_val = 2.0 * n * log_sum / (s * s)
            grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * af / s
            return obj_val, grad
        
        def obj_func(x):
            val, _ = lse_obj_and_grad(x)
            return val
        
        def grad_func(x):
            _, g = lse_obj_and_grad(x)
            return g
        
        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-9, 'ftol': 1e-11,
                                           'maxls': 40})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_val / s_c
                    score_c = fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand.copy()
        except Exception:
            pass
    
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_tight_M=True, lazy_constraints=True, max_lags=150):
    """LP best-response with Nesterov acceleration and lazy constraints."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    
    active_lags_cache = None
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break
        
        if use_tight_M:
            # Adaptive eps: start aggressive, decrease over time
            eps = min(0.05, 0.01 * np.sqrt(t + 1) / max(1, np.sqrt(t/100)))
            eps = min(0.05, max(0.005, 0.02 * (1.0 + 50.0/(t+1))))
            M_used = M_curr * (1.0 - eps)
        else:
            M_used = M_curr
        
        if lazy_constraints:
            b, active_lags_cache = lp_best_response_lazy(y, M_used, solver, active_lags_cache, max_lags)
        else:
            # Full constraint LP
            size = 2 * n - 1
            rows = []
            cols = []
            data = []
            for k in range(size):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    rows.append(k)
                    cols.append(j)
                    data.append(y[k - j])
            from scipy.sparse import csc_matrix
            T = csc_matrix((data, (rows, cols)), shape=(size, n))
            b_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_used])
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                b = None
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b = np.asarray(b_var.value, dtype=np.float64)
                b = np.maximum(b, 0.0)
            else:
                b = None
        
        if b is None:
            b, _ = lp_best_response_lazy(y, M_curr, solver, None, max_lags)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall_count = 0
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
            stall_count = 0
        
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
                    stall_count = 0
        
        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 8:
                break
        else:
            stall_count = 0
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def mul_weight_optimize(a0, time_limit, n_phase=5, verbose=True):
    """Multiplicative weight optimization with adaptive eta."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for phase in range(n_phase):
        if time.monotonic() - t0 > time_limit:
            break
        # Adaptive eta: start small, increase
        eta_start = 0.1 / (phase + 1)
        eta_end = 0.5 / (phase + 1)
        n_steps = 200
        
        a_phase = a.copy()
        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            eta = eta_start + (eta_end - eta_start) * step / n_steps
            
            conv_a = fast_autocorr(a_phase)
            k_star = int(np.argmax(conv_a))
            
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            
            grad = np.zeros(n)
            grad[js:je+1] = a_phase[k_star - np.arange(js, je+1)]
            
            a_phase = a_phase * np.exp(-eta * grad)
            a_phase = np.maximum(a_phase, 1e-30)
            s = a_phase.sum()
            a_phase = a_phase * norm_val / s
            
            score = fast_autocorr(a_phase).max()
            if score < best_score:
                best_score = score
                best_a = a_phase.copy()
        
        a = a_phase
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Alternating peaks
        a = np.ones(n)
        a[::3] = 1.5
        a[1::3] = 0.5
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(10):
            center = int(rng.integers(n // 4, 3 * n // 4))
            width = int(rng.integers(5, n // 4))
            amp = rng.uniform(0.3, 1.0)
            x = np.arange(n) - center
            a += amp * np.exp(-0.5 * (x / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
