    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def optimize_lp(a0, t_limit, solver, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    best_a = a.copy()
    best_sc = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    b_warm = None
    stall = 0
    k_frac = 0.20

    for t in range(1, 500000):
        if time.monotonic() - t0 > t_limit:
            break
        cy = fast_autocorr(y)
        M = cy.max()
        if M < 1e-10:
            break
        n_con = max(8, int(k_frac * (2 * n - 1)))
        top_idx = np.argsort(cy)[::-1][:n_con]
        b = lp_solve(y, M, top_idx, solver, b_warm)
        if b is None:
            b = lp_solve(y, M, np.arange(2 * n - 1), solver, b_warm)
            if b is None:
                continue
        b_warm = b
        bs = b.sum()
        if bs < 1e-15:
            continue
        g = b * S / bs
        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1 - alpha) * y + alpha * g, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new * S / sy
        if t == 1:
            a_new = g.copy()
        else:
            mom = min((t - 1) / (t + 2), 0.85)
            a_new = np.maximum(y_new + mom * (y_new - y), 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new * S / sa
            else:
                a_new = y_new.copy()
        y = y_new
        a = a_new
        sc_y = fast_autocorr(y).max()
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        sc_a = fast_autocorr(a).max()
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a.copy()
            stall = 0
        if stall > 60:
            y = best_a.copy()
            a = best_a.copy()
            b_warm = None
            k_frac = min(k_frac * 1.25, 0.5)
            stall = 0
            if verbose:
                print(f"    LP restart t={t}, k_frac={k_frac:.2f}, best={best_sc:.8f}")
        if verbose and t % 200 == 0:
            print(f"    LP t={t} ({time.monotonic() - t0:.0f}s): best={best_sc:.10f}")

    return best_a, best_sc


def pair_cd(a0, t_limit, rng, verbose=False):
    """Vectorized pair coordinate descent with exact line search."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * S / s

    conv = fast_autocorr(a)
    best_sc = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_imp = 0

    while time.monotonic() - t0 < t_limit:
        improved = False
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > t_limit:
                break
            i = int(perm[idx])
            ai = a[i]
            if ai < 1e-12:
                continue
            # Vectorized L computation for moving from i to j
            # L_i[k] = -(a[k-i]_v + a[i-k]_v) : contribution of removing from i
            Li = np.zeros(size)
            lo1 = max(0, i)
            hi1 = min(size, i + n)
            if hi1 > lo1:
                Li[lo1:hi1] -= a[0:hi1 - lo1]
            lo2 = max(0, i - n + 1)
            hi2 = min(size, i + 1)
            if hi2 > lo2:
                Li[lo2:hi2] -= a[i - (hi2 - 1):i - lo2 + 1][::-1]

            # Q_i: quadratic from removing at i
            Qi = np.zeros(size)
            if 2 * i < size:
                Qi[2 * i] = 1.0

            for _ in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                # L_j[k] = +(a[k-j]_v + a[j-k]_v)
                Lj = np.zeros(size)
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    Lj[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    Lj[lo4:hi4] += a[j - (hi4 - 1):j - lo4 + 1][::-1]

                Qj = np.zeros(size)
                if 2 * j < size:
                    Qj[2 * j] = 1.0
                ij = i + j
                if ij < size:
                    Qj[ij] = -1.0

                L = Li + Lj
                Q = Qi + Qj

                # Quick rejection check
                mid_d = ai * 0.5
                if (conv + mid_d * L + mid_d * mid_d * Q).max() >= best_sc:
                    # Check full delta
                    if (conv + ai * L + ai * ai * Q).max() >= best_sc:
                        continue

                # Ternary search for optimal delta in [0, ai]
                lo_d = 0.0
                hi_d = ai
                for _ in range(18):
                    m1 = lo_d + (hi_d - lo_d) / 3.0
                    m2 = hi_d - (hi_d - lo_d) / 3.0
                    f1 = (conv + m1 * L + m1 * m1 * Q).max()
                    f2 = (conv + m2 * L + m2 * m2 * Q).max()
                    if f1 < f2:
                        hi_d = m2
                    else:
                        lo_d = m1
                best_d = (lo_d + hi_d) / 2.0
                if best_d < 1e-10:
                    continue

                new_conv = conv + best_d * L + best_d * best_d * Q
                nm = new_conv.max()
                if nm < best_sc - 1e-12:
                    a[i] -= best_d
                    a[j] += best_d
                    sn = a.sum()
                    if sn < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * S / sn
                    conv = fast_autocorr(a)
                    cs = conv.max()
                    if cs < best_sc - 1e-12:
                        best_sc = cs
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += best_d
                        a[j] -= best_d
                        conv = fast_autocorr(a)
                if improved:
                    break  # Restart partner search with updated a

        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        if verbose:
            print(f"    CD ({time.monotonic() - t0:.0f}s): best={best_sc:.10f}, no_imp={no_imp}")

    return best_a, best_sc


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 4
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(8.0 * np.pi * x)
    elif choice == 5:
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = 1.0 / (1.0 + 20.0 * x * (1 - x))
    elif choice == 7:
        a = np.cos(np.pi * x / 2.0) ** 2 + 0.3 * np.sin(np.pi * x) ** 2
    elif choice == 8:
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.5
    else:
        a = (1 + np.cos(2 * np.pi * x)) ** 1.5
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    si = fi.sum()
    if si < 1e-15:
        return np.ones(new_n) / new_n
    return fi / si


def perturb(a, rng, scale=0.04):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    global _best_seq, _best_score
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    budget = 1740.0

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_global
            print(f"  *** [{label}]: {sc:.8f} (n={len(a)}, t={el:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_cur = len(best_seq)
    remaining = budget - (time.monotonic() - t_global)

    # Phase 1: LP at base length
    print(f"\n=== Phase 1: LP n={n_cur} ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = optimize_lp(a_norm, t1, solver, seed=0, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")
        n_cur = len(best_seq)

    # Phase 2: Pair CD
    remaining = budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Pair CD n={n_cur} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t2, rng, verbose=False)
        try_update(a_out, f"pcd1_n{n_cur}")
        n_cur = len(best_seq)

    # Phase 3: Perturbation + LP restarts
    remaining = budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Perturbation restarts ===")
    t3_total = min(remaining * 0.22, 300.0)
    n_restarts = 4
    for r in range(n_restarts):
        remaining = budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t3 = t3_total / n_restarts
        if t3 < 30:
            break
        n_c = len(best_seq)
        scales = [0.05, 0.03, 0.08, 0.02]
        pert = perturb(best_seq, rng, scale=scales[r % len(scales)])
        a_norm = pert * np.sqrt(2.0 * n_c) / pert.sum()
        a_out, _ = optimize_lp(a_norm, min(t3, remaining - 60), solver, seed=100 + r, verbose=False)
        try_update(a_out, f"pert{r}_n{n_c}")
        n_cur = len(best_seq)

    # Phase 4: Try longer length
    remaining = budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 4: LP n={n_long} ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = optimize_lp(a_norm, t4, solver, seed=7, verbose=True)
        try_update(a_out, f"lp_n{n_long}")
        n_cur = len(best_seq)

    # Phase 5: Pair CD on best
    remaining = budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_cur} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t5, rng, verbose=False)
        try_update(a_out, f"pcd2_n{n_cur}")
        n_cur = len(best_seq)

    # Phase 6: Final LP polish
    remaining = budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LP final n={n_cur} ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = optimize_lp(a_norm, t6, solver, seed=42, verbose=True)
        try_update(a_out, f"lp_fin_n{n_cur}")
        n_cur = len(best_seq)

    # Phase 7: Final CD
    remaining = budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_cur} ===")
    t7 = min(remaining * 0.35, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t7, rng, verbose=False)
        try_update(a_out, f"pcd_fin_n{n_cur}")

    # Phase 8: Try shorter length
    remaining = budget - (time.monotonic() - t_global)
    n_short = max(800, len(best_seq) - 400)
    if remaining > 80 and n_short != len(best_seq):
        print(f"\n=== Phase 8: LP n={n_short} ===")
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 20:
            init_short = interpolate_seq(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = optimize_lp(a_norm, t8, solver, seed=99, verbose=False)
            try_update(a_out, f"lp_n{n_short}")

    # Final evaluation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
