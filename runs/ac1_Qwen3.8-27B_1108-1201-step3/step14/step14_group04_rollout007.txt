import numpy as np
import cvxpy as cp

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1:n - 1 + n]
    return grad

def subgrad_max_autocorr(a):
    """Compute subgradient of max_k conv(a,a)[k] with respect to a.
    Returns the subgradient at the maximum lag."""
    n = len(a)
    ac = fast_autocorr(a)
    # Find the argmax lag
    k_star = np.argmax(ac)
    # Gradient of conv(a,a)[k] w.r.t. a[i] = a[k-i] + a[i-k] (valid terms)
    grad = np.zeros(n)
    # Term 1: a[k-i] for i in [max(0, k-n+1), min(k, n-1)]
    i_lo = max(0, k_star - n + 1)
    i_hi = min(k_star, n - 1)
    for i in range(i_lo, i_hi + 1):
        grad[i] += a[k_star - i]
    # Term 2: a[i-k] for i in [max(0, k), min(n-1, n-1+k)]  = [k, n-1]
    if k_star < n:
        for i in range(k_star, n):
            grad[i] += a[i - k_star]
    return grad, k_star

def subgrad_fista(a0, time_limit, verbose=True):
    """FISTA with subgradient of max autocorrelation + simplex projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t_fista = 1
    t0 = time.monotonic()
    
    # Initial step size
    step = 1.0 / n  # Conservative
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, k_star = subgrad_max_autocorr(y)
        
        # Gradient step
        z = y - step * grad
        
        # Project onto simplex (with the norm constraint)
        a_new = project_simplex(z, 1.0)
        a_new = a_new * norm  # Scale back
        
        # FISTA update
        if t_fista == 1:
            y_new = a_new.copy()
        else:
            mom = (t_fista - 1.0) / (t_fista + 2.0)
            y_new = a_new + mom * (a_new - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new * norm / s_y
        
        staging = a_new.copy()
        y = y_new
        a = a_new
        t_fista += 1
        
        # Track best
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adaptive step size: increase if we're making progress
        if it % 100 == 0:
            step *= 1.01
            if verbose and it % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Subgrad-FISTA iter={it} ({elapsed:.0f}s): best={best_score:.8f}, step={step:.6f}")
    
    return best_a, best_score

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_crosscorr(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1:n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 10.0), n / (T * 2.0))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y < 1e-15:
                    y_new = a_new.copy()
                else:
                    y_new = a_new  # keep projected
                    y_new = (a_new + mom * (a_new - staging))
                    y_new = np.maximum(y_new, 0.0)
                    s_y = y_new.sum()
                    if s_y > 1e-15:
                        y_new = y_new / s_y
                    else:
                        y_new = a_new.copy()
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc

        sc = fast_autocorr(a).max()

        if verbose:

def heuristic_best_response(a, M, warm_start=None, max_iters=200, tol=1e-9):
    """Fast heuristic for LP best response: maximize sum(b) s.t. conv(a,b)<=M, b>=0.
    Uses iterative constraint satisfaction with gradient-based pressure."""
    n = len(a)
    if warm_start is not None and warm_start.sum() > 1e-15:
        b = np.maximum(warm_start.copy(), 0.0)
        b = b / b.sum()
    else:
        b = np.ones(n) / n

    for it in range(max_iters):
        c = fast_crosscorr(a, b)
        c_max = c.max()
        if c_max <= M * (1.0 + tol):
            break

        # Find all lags violating the constraint
        violating = np.where(c > M * (1.0 + tol))[0]
        if len(violating) == 0:
            break

        # Compute "pressure" on each b[j] from violating lags
        pressure = np.zeros(n)
        for k in violating:
            # c[k] = sum_j a[k-j]*b[j] (valid j)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            excess = (c[k] - M) / max(M, 1e-15)
            for j in range(i_lo, i_hi + 1):
                pressure[j] += a[k - j] * excess

        # Reduce b where pressure is high
        max_pressure = pressure.max()
        if max_pressure < 1e-15:
            break

        eta = 0.5
        b = b - eta * pressure / max_pressure
        b = np.maximum(b, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / b_sum

    return b


def lp_best_response(a, M, solver, warm_start=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix sparsely
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for j in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    T_mat = cp.csr_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(2 * n - 1, n))
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, use_heuristic, verbose=True):
    """LP Nesterov with optional heuristic best response for speed."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    ema_a = a.copy()
    ema_alpha = 0.05
    t0 = time.monotonic()
    n_iters = 0
    last_b = None

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Choose best response method
        if use_heuristic:
            b = heuristic_best_response(y, M_curr, warm_start=last_b, max_iters=50)
        else:
            b = lp_best_response(y, M_curr, solver, warm_start=last_b)
            if b is None:
                b = heuristic_best_response(y, M_curr, warm_start=last_b, max_iters=50)

        if b is None or b.sum() < 1e-15:
            continue
        last_b = b.copy()
        b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov-style update with adaptive alpha
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best via multiple strategies
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Ergodic average
        avg_a += a
        n_avg += 1
        ema_a = (1.0 - ema_alpha) * ema_a + ema_alpha * a

        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

            score_ema = fast_autocorr(ema_a * norm / max(ema_a.sum(), 1e-15)).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema_a * norm / max(ema_a.sum(), 1e-15)

        if n_avg % 1000 == 0:
            # Decay EMA alpha
            ema_alpha = max(0.01, ema_alpha * 0.95)

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest({use_heuristic and 'heur' or 'LP'}) iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Peak-aware coordinate descent: move mass from high-peak-contributors to low contributors."""
    rng = np.random.default_rng(2024)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:8]

        improved = False
        for k in top_k:
            if ac[k] <= 0:
                continue
            # Contributors to lag k: a[j]*a[k-j] for valid j
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo > i_hi:
                continue
            js = np.arange(i_lo, i_hi + 1)
            contrib = a[js] * a[k - js]
            total_contrib = contrib.sum()
            if total_contrib < 1e-18:
                continue

            # Find the position with highest contribution
            src_j = js[np.argmax(contrib)]
            src_val = a[src_j]
            if src_val < 1e-15:
                continue

            # Find destination: positions that DON'T contribute to any top peak
            peak_positions = set()
            for k2 in top_k:
                lo2 = max(0, k2 - n + 1)
                hi2 = min(k2, n - 1)
                for jj in range(lo2, hi2 + 1):
                    peak_positions.add(jj)
                    if k2 - jj >= 0 and k2 - jj < n:
                        peak_positions.add(k2 - jj)

            # Candidates: positions not in peak_positions, with small values
            all_pos = np.arange(n)
            non_peak = np.array([p for p in all_pos if p not in peak_positions and p != src_j])
            if len(non_peak) == 0:
                non_peak = np.array([p for p in all_pos if p != src_j])

            # Pick a few candidates
            num_cand = min(10, len(non_peak))
            # Prefer positions with small values (to minimize disturbance)
            cand_idx = non_peak[np.argsort(a[non_peak])][:num_cand]

            for dst in cand_idx:
                for frac in [0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-18 or amt > a[src_j]:
                        continue
                    a_trial = a.copy()
                    a_trial[src_j] -= amt
                    a_trial[dst] += amt
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape local minima
            stall += 1
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                dir_sign = 1 if a[i] > a[j] else -1
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac * dir_sign * (1 if dir_sign > 0 else 1)
                if dir_sign > 0:
                    amt = a[i] * frac
                else:
                    amt = a[j] * frac
                a_trial = a.copy()
                if dir_sign > 0:
                    a_trial[i] -= amt
                    a_trial[j] += amt
                else:
                    a_trial[j] -= amt
                    a_trial[i] += amt
                if a_trial[i] < 0 or a_trial[j] < 0:
                    continue
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if stall > 2000:
                break

        if verbose and (n_iters_check := n_avg) // 50 == 0:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 600:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization combining LP, heuristic, and coordinate descent."""
    global _best_overall, _best_overall_score
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            _best_overall = best_seq.copy()
            _best_overall_score = best_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            _best_overall = best_seq.copy()
            _best_overall_score = best_score
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 60.0)
    if t1 > 15 and best_score > 1.503:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [15.0, 4.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP Nesterov (full LP, main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov (full LP) n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, use_heuristic=False, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Heuristic best response Nesterov (faster) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Heuristic-Nesterov n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t3, solver, use_heuristic=True, verbose=True)
        try_update(a_out, f"heur1_n{n_best}")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Coord Descent n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t4, verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Targeted peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Targeted Peak Shave n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t5, seed=777)
        try_update(a_out, f"targeted_n{n_best}")

    # === Phase 6: More LP (if time remains) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP polish n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t6, solver, use_heuristic=False, verbose=False)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for delta in [1, -1, 2, -2]:
            n2 = len(best_seq) + delta
            if n2 < 100 or n2 > 3000:
                continue
            t7 = min(remaining * 0.15, 90.0)
            if t7 < 20:
                break
            print(f"\n=== Phase 7: Heuristic at n={n2} (budget={t7:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, n2)
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, use_heuristic=True, verbose=False)
            try_update(a_out, f"len{n2}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.4, 100.0)
        if t8 > 15:
            print(f"\n=== Phase 8: Final Coord Descent n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, t8, verbose=False)
            try_update(a_out, "final_coord")

    # === Phase 9: Final heuristic Nesterov squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.5, 120.0)
        if t9 > 15:
            print(f"\n=== Phase 9: Final Heuristic Nesterov n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t9, solver, use_heuristic=True, verbose=False)
            try_update(a_out, "final_heur")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
