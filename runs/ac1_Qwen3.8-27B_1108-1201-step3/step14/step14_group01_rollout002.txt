def fast_conv_self(a):

def build_T_rows(a, rows):

def lp_best_response(a, M, active_idx, solver, max_iters=3):

def lp_nesterov_ftrl(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    Z = np.zeros(n)
    L2_base = 0.01
    ftrl_iter = 0

    top_k = 80

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Select active constraints: top-k by value
        k_sel = min(top_k + t // 50, size)
        active_idx = np.argsort(conv_y)[-k_sel:]
        active_idx = np.sort(active_idx)

        # Solve LP with constraint generation
        b, violated = lp_best_response_full_check(y, M, active_idx, solver)
        if b is None:
            continue

        if len(violated) > 0:
            # Add violated constraints and re-solve
            new_active = np.unique(np.concatenate([active_idx, violated]))
            b2, violated2 = lp_best_response_full_check(y, M, new_active, solver)
            if b2 is not None:
                b = b2
                if violated2 is not None and len(violated2) > 0:
                    # One more round
                    new_active2 = np.unique(np.concatenate([new_active, violated2]))
                    b3, _ = lp_best_response_full_check(y, M, new_active2, solver)
                    if b3 is not None:
                        b = b3

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # FTRL update
        ftrl_iter += 1
        L2 = L2_base * ftrl_iter

        Z_new = Z + g / L2
        a_new = np.maximum(Z_new, 0.0)
        # Project: scale to norm
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new * norm / s_new
        Z = Z_new

        # Update y with Nesterov-like step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new

        # Check scores
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a_new).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_new.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FTRL t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.4f}")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):

    n_stages = 8
    temps = np.geomspace(30.0, 0.2, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)

            break

            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n) * (1.0 + 0.05 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Candidate j selection: random + peak-informed
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            for kl in peak_lags[:4]:
                j_struct = (kl - i) % n
                j_candidates.add(j_struct)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i or j >= n:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 20
                deltas = np.linspace(0.0, ai, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d == 0:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def multi_gaussian_init(n, rng, n_peaks=8):
    x = np.linspace(0.0, 1.0, n)
    centers = rng.uniform(0.1, 0.9, n_peaks)
    amplitudes = rng.uniform(0.5, 1.5, n_peaks)
    widths = rng.uniform(0.03, 0.12, n_peaks)
    a = 0.1 * np.ones(n)
    for c, amp, w in zip(centers, amplitudes, widths):
        a += amp * np.exp(-((x - c) ** 2) / (2 * w * w))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def sinusoidal_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    k1 = rng.integers(2, 8)
    k2 = rng.integers(k1 + 1, k1 + 6)
    phi = rng.uniform(0, 2 * np.pi)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * k1 * x + phi) + 0.2 * np.sin(2 * np.pi * k2 * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = multi_gaussian_init(n_base, rng, n_peaks=10)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE to sharpen
    remaining = time_budget - (time.monotonic() - t0_global())
    print(f"\n=== Phase 1: LSE GD n={n_base} ===")
    t1 = min(remaining * 0.12, 120.0)
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # Phase 2: LP FTRL (main workhorse)
    remaining = time_budget - (time.monotonic() - t0_global())
    print(f"\n=== Phase 2: LP FTRL n={n_base} (budget={remaining*0.35:.0f}s) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 20:
        a_out, _ = lp_nesterov_ftrl(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lpftrl1_n{n_base}")

    # Phase 3: LSE refinement
    remaining = time_budget - (time.monotonic() - t0_global())
    print(f"\n=== Phase 3: LSE refinement n={n_base} ===")
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t3, verbose=True)
        try_update(a_out, f"lse2_n{n_base}")

    # Phase 4: LP FTRL round 2
    remaining = time_budget - (time.monotonic() - t0_global())
    print(f"\n=== Phase 4: LP FTRL round 2 n={n_base} ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 15:
        a_out, _ = lp_nesterov_ftrl(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lpftrl2_n{n_base}")

    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t0_global())
    print(f"\n=== Phase 5: Pair CD n={n_base} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_base}")

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t0_global())
    n_long = n_base + 600
    print(f"\n=== Phase 6: Longer n={n_long} ===")
    if remaining > 100:
        t6 = min(remaining * 0.30, 180.0)
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = lp_nesterov_ftrl(init_long, t6, solver, verbose=True)
        try_update(a_out, f"lpftrl_long_{n_long}")

    # Phase 7: Alternative initialization and quick LP
    remaining = time_budget - (time.monotonic() - t0_global())
    if remaining > 60:
        print(f"\n=== Phase 7: Alternative init ===")
        n_alt = n_base
        alt_inits = [
            multi_gaussian_init(n_alt, rng, n_peaks=14),
            sinusoidal_init(n_alt, rng),
        ]
        for ai_idx, init in enumerate(alt_inits):
            if remaining < 30:
                break
            t_alt = min(remaining * 0.3, 80.0)
            a_out, _ = lp_nesterov_ftrl(init, t_alt, solver, verbose=False)
            try_update(a_out, f"alt{ai_idx}_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t0_global())

    # Phase 8: Final LP + CD
    remaining = time_budget - (time.monotonic() - t0_global())
    print(f"\n=== Phase 8: Final polish n={len(best_seq)} ===")
    if remaining > 30:
        t_final_lp = min(remaining * 0.5, 60.0)
        a_out, _ = lp_nesterov_ftrl(best_seq, t_final_lp, solver, verbose=False)
        try_update(a_out, "final_lp")
        remaining = time_budget - (time.monotonic() - t0_global())
        if remaining > 15:
            t_final_cd = min(remaining * 0.6, 40.0)
            a_out, _ = pair_cd_optimize(best_seq, t_final_cd, rng, verbose=False)
            try_update(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
