So: `∂f/∂a[i] = 2 * sum_k w[k] * a[k-i]`

This is a cross-correlation of `w` and `a`! Specifically:
`∂f/∂a[i] = 2 * (w ⋆ a)[i]` where `⋆` denotes cross-correlation.

Using FFT: `cross_corr(w, a)[i] = ifft(conj(FFT(w)) * FFT(a))[i]`

This is O(n log n) per step. With n=1600, each step is very fast. We can do hundreds of thousands of steps.

Let me now write the complete code:

1. LSE-smoothed gradient with Nesterov acceleration (fast, O(n log n) per step)
2. LP best-response with Nesterov (precise refinement)
3. Peak-directed coordinate descent (targeted improvements)
4. Multi-length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_nesterov(a0, time_limit, verbose=True):
    """LSE-smoothed gradient with Nesterov acceleration and projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Temperature schedule
    n_stages = 15
    temps = np.geomspace(30.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Number of steps for this temperature
        n_steps = max(50, int(remaining * 30))
        
        # Initialize Nesterov
        y = a.copy()  # Nesterov's y
        x_prev = a.copy()
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute LSE gradient at y
            conv_y = fast_conv_self(y)
            max_c = conv_y.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv_y - max_c) / T_val
            shifted = np.clip(shifted, -100.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: 2 * cross_corr(w, y)
            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            grad = 2.0 * irfft(np.conj(W) * Y, fft_size)[:n]
            
            # Nesterov update
            # Step size: eta = 1/L where L is the Lipschitz constant
            # L ≈ 2 * max|y| * (2/T_val) for the LSE smooth approximation
            # Use a safe step size
            eta = T_val / (2.0 * np.max(np.abs(grad)) + 1e-10)
            eta = min(eta, 0.1 * norm / (np.linalg.norm(grad) + 1e-10))
            
            if it == 0:
                x_new = y - eta * grad
            else:
                # Nesterov momentum
                beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
                momentum = (beta - 1.0) / beta
                x_new = y - eta * grad
                x_new = x_new + momentum * (x_new - x_prev)
            
            # Project to {a >= 0, sum(a) = norm}
            x_new = np.maximum(x_new, 0.0)
            s_new = x_new.sum()
            if s_new < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new * norm / s_new
            
            # Update y for next iteration
            y = x_new
            x_prev = x_new.copy()
            
            # Track best (use the iterate and the average)
            sc = fast_conv_self(x_new).max()
            if sc < best_score:
                best_score = sc
                best_a = x_new.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Nesterov stage {si} T={T_val:.2f} ({elapsed:.0f}s): peak={best_score:.8f}")
    
    return best_a, best_score

def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Constraint screening with hysteresis
        threshold_high = M * 0.90
        threshold_low = M * 0.70
        active = np.where(conv_y >= threshold_high)[0]
        if len(active) < 50:
            active = np.where(conv_y >= threshold_low)[0]
        if len(active) < 20:
            active = None  # Use all constraints
        
        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify dropped constraints
        if active is not None:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 1e-3):
                b = lp_solve(y, M, solver, None)
                if b is not None:
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        for candidate in [y, a]:
            sc = fast_conv_self(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        # Average tracking
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov t={t} ({elapsed:.0f}s): peak={best_score:.8f}")
    
    return best_a, best_score

def peak_cd(a0, time_limit, rng, verbose=True):
    """Peak-directed coordinate descent: focus on reducing the max peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        k_star = conv.argmax()
        
        # The value at k_star: b[k_star] = sum_j a[j] * a[k_star - j]
        # To reduce this, transfer mass from positions that contribute to the peak
        # to positions that don't
        
        perm = rng.permutation(n)
        pass_improved = False
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            ai = a[i]
            if ai < 1e-12 * norm:
                continue
            
            # Contribution of a[i] to b[k_star]:
            # a[i] * a[k_star - i] (if valid) + a[k_star - i] * a[i] (same if 2i != k_star)
            # = a[i] * (a[k_star-i] if 0<=k_star-i<n else 0) * (2 if 2i != k_star else 1)
            
            # Actually, the gradient of b[k_star] w.r.t. a[i] is 2*a[k_star-i] (if valid)
            k_off = k_star - i
            if 0 <= k_off < n:
                grad_ik = 2.0 * a[k_off]
            else:
                grad_ik = 0.0
            
            if grad_ik <= 0:
                # This position doesn't contribute to the peak (or reduces other peaks)
                continue
            
            # Try transferring mass from i to a random j
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for _ in range(20):
                j = int(rng.integers(n))
                if j == i:
                    continue
                aj = a[j]
                
                # Gradient of b[k_star] w.r.t. a[j]
                k_off_j = k_star - j
                if 0 <= k_off_j < n:
                    grad_jk = 2.0 * a[k_off_j]
                else:
                    grad_jk = 0.0
                
                # Net gradient for transfer i -> j: grad_ik - grad_jk
                net_grad = grad_ik - grad_jk
                if net_grad <= 0:
                    continue
                
                # Try several fractions
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    d = ai * frac
                    a[i] = ai - d
                    a[j] = aj + d
                    a = a * norm / a.sum()
                    
                    conv_new = fast_conv_self(a)
                    v = conv_new.max()
                    
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j
                        best_frac = frac
                    elif v < best_val:
                        pass
                    
                    a[i] = ai
                    a[j] = aj
                    a = a * norm / a.sum()
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 4:

                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        # S-shaped
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    else:
        # Double-peak with different widths
        a = np.exp(-12.0 * (x - 0.25) ** 2) + np.exp(-6.0 * (x - 0.75) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

- This gives a "direction" to move `a`
- The intuition is: if `b` puts a lot of mass where conv(a,.) is close to M, then moving `a` toward `b` will reduce the peak

This is related to the "multiplicative weights" or "exponentiated gradient" method for matrix games.

For our specific problem (not a true game), the LP approach still works well empirically (as shown by the previous code achieving 1.504).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient with Nesterov acceleration (fast O(n log n) per step)
2. LP best-response with Nesterov and hysteresis-based constraint screening
3. Peak-directed coordinate descent targeting the dominant autocorrelation lag
4. Multi-phase alternating optimization with adaptive time allocation

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_nesterov(a0, time_limit, verbose=True):
    """LSE-smoothed gradient with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_peak = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 20
    temps = np.geomspace(40.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        n_steps = max(100, int(remaining * 40))

        y = a.copy()
        x_prev = a.copy()
        avg = np.zeros(n)
        n_avg = 0

        for it in range(1, n_steps + 1):
            if time.monotonic() - t0 > time_limit:
                break

            conv_y = fast_conv_self(y)
            max_c = conv_y.max()
            if max_c < 1e-15:
                break

            shifted = (conv_y - max_c) / T_val
            shifted = np.clip(shifted, -80.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            grad = 2.0 * irfft(np.conj(W) * Y, fft_size)[:n]

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            eta = 1.0 / (grad_norm + 1e-10)
            eta = min(eta, 0.5 * norm / (grad_norm + 1e-10))

            if it == 1:
                x_new = y - eta * grad
            else:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
                momentum = (beta - 1.0) / beta
                x_new = y - eta * grad + momentum * (y - x_prev)

            x_new = np.maximum(x_new, 0.0)
            s_new = x_new.sum()
            if s_new < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new * norm / s_new

            y = x_new
            x_prev = x_new.copy()

            avg += x_new
            n_avg += 1

            if it % 50 == 0:
                sc = fast_conv_self(x_new).max()
                if sc < best_peak:
                    best_peak = sc
                    best_a = x_new.copy()

                if n_avg >= 100 and it % 500 == 0:
                    avg_check = avg / n_avg
                    s_avg = avg_check.sum()
                    if s_avg > 1e-15:
                        avg_check = avg_check * norm / s_avg
                        sc_avg = fast_conv_self(avg_check).max()
                        if sc_avg < best_peak:
                            best_peak = sc_avg
                            best_a = avg_check.copy()

        sc_final = fast_conv_self(y).max()
        if sc_final < best_peak:
            best_peak = sc_final
            best_a = y.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Nest stage {si} T={T_val:.2f} ({elapsed:.0f}s): peak={best_peak:.8f}")

    return best_a, best_peak

def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and hysteresis screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_peak = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    active_set = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        if active_set is None:
            active_set = np.where(conv_y >= M * 0.85)[0]
            if len(active_set) < 30:
                active_set = np.where(conv_y >= M * 0.6)[0]
            if len(active_set) < 10:
                active_set = None

        use_active = active_set if (active_set is not None and len(active_set) < size) else None
        b = lp_solve(y, M, solver, use_active)

        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if use_active is not None:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 5e-4):
                b = lp_solve(y, M, solver, None)

                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_peak:
            best_peak = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_peak:
            best_peak = sc_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_peak:

        if t % 500 == 0:

        if verbose and t % 500 == 0:

            n_act = len(active_set) if active_set is not None else size
            print(f"    LP-Nest t={t} ({elapsed:.0f}s, n_act={n_act}): peak={best_peak:.8f}")

    return best_a, best_peak

def peak_cd(a0, time_limit, rng, verbose=True):
    """Peak-directed coordinate descent targeting the dominant lag."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_peak = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve = 0
    total_improves = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        k_star = int(conv.argmax())

        perm = rng.permutation(n)
        pass_improved = False

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            k_off = k_star - i
            if 0 <= k_off < n:
                contrib = a[k_off]
            else:
                contrib = 0.0

            if contrib < 1e-10:
                continue

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            n_try = 15
            for _ in range(n_try):
                j = int(rng.integers(n))
                if j == i:
                    continue

                k_off_j = k_star - j
                if 0 <= k_off_j < n:
                    contrib_j = a[k_off_j]
                else:
                    contrib_j = 0.0

                if contrib_j >= contrib * 0.9:
                    continue

                aj = a[j]
                for frac in [0.2, 0.4, 0.6, 0.8]:
                    d = ai * frac
                    if d <= 0:
                        continue
                    a[i] = ai - d
                    a[j] = aj + d
                    a = a * norm / a.sum()
                    v = fast_conv_self(a).max()
                    if v < best_val - 1e-13:
                        best_val = v
                        best_delta = d
                        best_j = j
                    a[i] = ai
                    a[j] = aj
                    a = a * norm / a.sum()

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                new_peak = fast_conv_self(a).max()
                if new_peak < best_peak - 1e-13:
                    best_peak = new_peak
                    best_a = a.copy()
                    pass_improved = True
                    total_improves += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve == 5:
                n_pert = max(5, n // 30)
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improves % 200 == 0 and total_improves > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD improves={total_improves} ({elapsed:.0f}s): peak={best_peak:.8f}")

    return best_a, best_peak

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
