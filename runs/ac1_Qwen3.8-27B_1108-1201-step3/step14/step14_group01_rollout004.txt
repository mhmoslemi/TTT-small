def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def build_T_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        T = build_T_matrix(a)
        constraints = [T[active, :] @ b_var <= M]
    else:
        T = build_T_matrix(a)
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):

    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(80.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(300, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= max_c + 0.05 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:

            else:

                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Quadratic correction at k=2i and k=2j
                q_2i = -2.0 * ai + 1.0  # d^2 term coefficient for k=2i
                q_2j = 2.0 * aj + 1.0    # d^2 term coefficient for k=2j

                # Sample delta values
                n_samples = 25
                deltas = np.linspace(0.01 * ai, 0.99 * ai, n_samples)

                best_d_local = 0.0
                best_v_local = cur_max

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] += q_2i * d
                    if 2 * j < size:
                        new_conv[2 * j] += q_2j * d
                    v = new_conv.max()
                    if v < best_v_local:
                        best_v_local = v
                        best_d_local = d

                if best_d_local > 0 and best_v_local < best_val - 1e-12:
                    best_val = best_v_local
                    best_delta = best_d_local
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve >= 3:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def global_cd_optimize(a0, time_limit, rng, verbose=True):
    """Full coordinate descent trying all (i,j) pairs for the most promising i."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        conv = fast_conv_self(a)
        cur_max = conv.max()
        improved = False

        # Focus on coordinates that contribute most to the peak
        # Contribution of a[i] to conv[k] is a[i]*a[k-i] and a[i]*a[i-k]
        # Total contribution: sum_k conv_contribution_from_i
        # For simplicity, just process in order of a[i] (largest first)
        order = np.argsort(-a)
        n_try = min(n, max(50, n // 3))

        for idx in range(n_try):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            conv = fast_conv_self(a)
            cur_max = conv.max()

            # Try all j (vectorized)
            # For each j, compute the max of conv + delta*(g_j) - delta*(g_i)
            # where g_j adds the a[j] contribution and g_i removes the a[i] contribution
            # This is O(n) per i, so total O(n^2) per pass - too slow for large n
            # Instead, use top-k j candidates
            peak_lags = np.where(conv >= cur_max * 0.85)[0]

            # Sort j by how much they'd help reduce the peak
            j_scores = np.zeros(n)
            for kl in peak_lags[:5]:
                # j that maximize a[kl-j] + a[j-kl] (the "contribution" at lag kl)
                contrib = np.zeros(n)
                lo = max(0, kl)
                hi = min(n, kl + n)
                if hi > lo:
                    contrib[0:hi-lo] += a[0:hi-lo]
                lo2 = max(0, kl - n + 1)
                hi2 = min(n, kl + 1)
                if hi2 > lo2:
                    contrib[lo2:hi2] += a[kl - np.arange(lo2, hi2)]
                j_scores += contrib
            j_scores[i] = -np.inf

            top_j = np.argsort(-j_scores)[:min(20, n - 1)]

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in top_j:
                j = int(j)
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                q_2i = -2.0 * ai + 1.0
                q_2j = 2.0 * aj + 1.0

                n_samples = 30
                deltas = np.linspace(0.02 * ai, 0.98 * ai, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] += q_2i * d
                    if 2 * j < size:
                        new_conv[2 * j] += q_2j * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        improved = True

        if not improved:
            break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GCD pass {n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) ** 0.5
    else:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * x) ** 2 + 0.2 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast, improves structure)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True, rng=rng)
        try_update(a_out, f"lp_n{len(best_seq)}")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")

    # Phase 4: Second LP round from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, rng=rng)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Global CD (focus on top coordinates)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Global CD (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = global_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"gcd_n{len(best_seq)}")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length exploration (budget={remaining:.0f}s) ===")
    candidate_lengths = [n_base + 200, n_base - 200, n_base + 400]
    for new_n in candidate_lengths:
        if time.monotonic() - t_global > time_budget - 100:
            break
        if new_n < 500:
            continue
        remaining_here = time_budget - (time.monotonic() - t_global)
        if remaining_here < 60:
            break
        t_here = min(remaining_here * 0.4, 120.0)
        print(f"  Trying n={new_n} (budget={t_here:.0f}s)")
        if t_here > 30:
            init_long = interpolate_seq(best_seq, new_n)
            # Quick LSE then LP
            t_half = t_here * 0.4
            a_tmp, _ = lse_gradient_descent(init_long, t_half, verbose=False)
            # Quick LP
            t_lp = t_here - t_half - 5
            if t_lp > 20:
                a_out, _ = nesterov_lp_optimize(a_tmp, t_lp, solver, verbose=False, rng=rng)
                try_update(a_out, f"lp_n{new_n}")
            try_update(a_tmp, f"lse_n{new_n}")

    # Phase 7: Final polish - alternate LSE and CD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        n_rounds = 3
        for r in range(n_rounds):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_half = remaining * 0.45 / (n_rounds - r)
            t_half = min(t_half, 80.0)
            if t_half > 15:
                a_out, _ = lse_gradient_descent(best_seq, t_half, verbose=False)
                try_update(a_out, f"lse_polish_{r}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_cd = min(remaining * 0.45 / (n_rounds - r), 80.0)
            if t_cd > 10:
                a_out, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
                try_update(a_out, f"pcd_polish_{r}")

    # Phase 8: One final LP burst with whatever time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP burst (budget={remaining:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, remaining - 10, solver, verbose=False, rng=rng)
        try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
