    return best_a * norm / best_a.sum(), best_M


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        p_star = int(np.argmax(c))
        
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-25:
            stall += 1
            if stall > 500:
                break
            continue
        
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        # Find good destinations (low impact on peaks)
        top_peaks = np.argsort(c)[::-1][:min(5, 2*n-1)]
        
        # Impact score for each position
        impact = np.zeros(n)
        for k in top_peaks:
            js_k = max(0, k - n + 1)
            je_k = min(n - 1, k)
            if js_k <= je_k:
                impact[js_k:je_k+1] += a[k - np.arange(js_k, je_k+1)]
        
        impact[src] = 1e30
        candidates = np.argsort(impact)[:min(10, n)]
        
        best_trial = best_M
        best_move = None
        
        for dst in candidates:
            if dst == src:
                continue
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                a_t = a_t / a_t.sum()
                sc = fast_autocorr(a_t).max()
                if sc < best_trial:
                    best_trial = sc
                    best_move = (dst, frac)
        
        if best_move is not None:
            dst, frac = best_move
            a[src] -= a[src] * frac  # note: src_val already captured
            # Actually need to be careful
            a_t = a.copy()
            a_t[src] -= src_val * frac
            a_t[dst] += src_val * frac
            a_t = np.maximum(a_t, 0)
            a_t = a_t / a_t.sum()
            a = a_t
            if best_trial < best_M:
                best_M = best_trial
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Random exploration when stalled
        if stall > 30:
            n_pert = max(3, n // 100)
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a_t = a.copy()
            for i in idx_p:
                if a_t[i] > 0:
                    j = rng.integers(n)
                    if j != i:
                        amt = a_t[i] * rng.uniform(0.1, 0.3)
                        a_t[i] -= amt
                        a_t[j] += amt
            a_t = np.maximum(a_t, 0)
            a_t = a_t / a_t.sum()
            sc = fast_autocorr(a_t).max()
            if sc < best_M * 1.05:
                a = a_t
                if sc < best_M:
                    best_M = sc
                    best_a = a.copy()
            stall = 0
        
        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"  PeakShave ({elapsed:.0f}s) best={best_M:.8f} stall={stall}")
    
    return best_a * norm / best_a.sum(), best_M


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    return f_new
