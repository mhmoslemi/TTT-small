import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _lp_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _lp_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), 10.0)
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            if time.monotonic() - t0 > time_limit:
                break
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 300 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            print(f"    LSE stage={stage_i} T={T:.2f} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = 2.0 * n * fast_autocorr(y).max() / (y.sum() ** 2)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max() / (avg_check.sum() ** 2)
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 40 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_directed_search(a0, time_limit, seed=42):
    """Exploit gradient structure: g[i]=2*a[k*-i] for peak lag k*.
    Do 2-coordinate moves with line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    stall = 0

    def line_search_2move(base, i, j, base_score):
        """Find best delta for moving mass from i to j."""
        lo, hi = 0.0, base[i] * 0.5
        if hi < 1e-15:
            return base.copy(), base_score
        for _ in range(30):
            mid = (lo + hi) / 2.0
            trial = base.copy()
            trial[i] -= mid
            trial[j] += mid
            if trial[i] < -1e-15:
                hi = mid
                continue
            sc = 2.0 * n * fast_autocorr(trial).max() / (trial.sum() ** 2)
            if sc < base_score:
                lo = mid
            else:
                hi = mid
            if hi - lo < 1e-10 * base[i]:
                break
        best_delta = lo
        if best_delta < 1e-15:
            return base.copy(), base_score
        best_trial = base.copy()
        best_trial[i] -= best_delta
        best_trial[j] += best_delta
        best_sc = 2.0 * n * fast_autocorr(best_trial).max() / (best_trial.sum() ** 2)
        return best_trial, best_sc

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_idx = np.argmax(b)
        peak_val = b[peak_idx]
        curr_score = 2.0 * n * peak_val / (a.sum() ** 2)

        # Gradient direction: g[i] = 2*a[peak_idx - i] if valid
        # We want to decrease a at high-g positions and increase at low-g positions
        # g[i] = 2*a[peak_idx - i] for 0 <= peak_idx - i < n
        valid = np.arange(n)
        gi_vals = np.zeros(n)
        for i in valid:
            ji = peak_idx - i
            if 0 <= ji < n:
                gi_vals[i] = 2.0 * a[ji]

        # Find top source (high g) and best destination (low g)
        # Source: high a[peak_idx - i] means high a at reflected position
        # Destination: low a[peak_idx - i]
        # But we also need destination not to be a source for other peaks

        top_sources = np.argsort(gi_vals)[::-1][:10]
        top_dests = np.argsort(gi_vals)[:10]

        improved = False
        for si in top_sources:
            if a[si] < 1e-10:
                continue
            for di in top_dests:
                if di == si:
                    continue
                trial, sc = line_search_2move(a, si, di, curr_score)
                if sc < best_score - 1e-12:
                    a = trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try moving from peak-contributing positions to random positions
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contrib = a[idx] * a[peak_idx - idx]
                src = idx[np.argmax(contrib)]
                if a[src] > 1e-10:
                    candidates = np.where(np.arange(n) != src)[0]
                    dst = candidates[rng.integers(len(candidates))]
                    trial, sc = line_search_2move(a, src, dst, curr_score)
                    if sc < best_score - 1e-12:
                        a = trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                if trial[i] >= 0:
                    sc = 2.0 * n * fast_autocorr(trial).max() / (trial.sum() ** 2)
                    if sc < best_score:
                        a = trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            stall += 1
            if stall > 500:
                break

    return best_a, best_score

def fine_tune(a0, time_limit, seed=777):
    """Intensive local refinement: try many 2-moves and 3-moves around current best."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top 5 peaks
        top_peaks = np.argsort(b)[::-1][:5]
        improved = False

        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Try multiple destinations
            # Prefer destinations that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            safe_dests = np.where(~peak_mask)[0]
            if len(safe_dests) == 0:
                safe_dests = np.where(np.arange(n) != src)[0]

            # Try several fractions
            for frac in [0.0003, 0.001, 0.003, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                # Pick destination with lowest value among safe ones
                dst = safe_dests[np.argmin(a[safe_dests])]
                trial = a.copy()
                trial[src] -= amt
                trial[dst] += amt
                if trial[src] < -1e-15:
                    continue
                sc = 2.0 * n * fast_autocorr(trial).max() / (trial.sum() ** 2)
                if sc < best_score - 1e-13:
                    a = trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.0005, 0.02)
                amt = a[i] * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                if trial[i] < -1e-15:
                    continue
                sc = 2.0 * n * fast_autocorr(trial).max() / (trial.sum() ** 2)
                if sc < best_score:
                    a = trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
                if stall > 600:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalized_init(n, dtype='gaussian', seed=42):
    rng = np.random.default_rng(seed)
    if dtype == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif dtype == 'sine':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.2
    elif dtype == 'flat':
        a = np.ones(n)
    elif dtype == 'cosine_mod':
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x)
    elif dtype == 'ramp':
        a = np.linspace(0.1, 1.0, n)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with peak-directed moves and LP."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        # Multi-start: try several initializations
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for dtype, s in [('gaussian', 1), ('sine', 2), ('cosine_mod', 3), ('ramp', 4)]:
            init = normalized_init(n_base, dtype, seed=s)
            sc = evaluate_safe(init)
            print(f"  Init {dtype}: score={sc:.6f}")
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 75.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} ({t1:.0f}s) ===")
        T_sched = [30.0, 10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: Multi-peak LP Nesterov (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 60:
        print(f"\n=== Phase 2: Multi-peak LP n={n_best} ({t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"mlp1_n{n_best}")

    # === Phase 3: Peak-directed search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Peak-directed n={n_best} ({t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_directed_search(a_norm, t3, seed=42)
        try_update(a_out, f"pd1_n{n_best}")

    # === Phase 4: LP again on improved sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP polish n={n_best} ({t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_current = len(best_seq)
        for n_try in [n_current + 200, n_current + 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            t5 = min(remaining * 0.3, 120.0)
            if t5 < 30:
                continue
            print(f"\n=== Phase 5: LP at n={n_try} ({t5:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, n_try)
            a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
            a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # === Phase 6: Fine-tuning ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t6 = min(remaining * 0.3, 150.0)
        if t6 > 20:
            print(f"\n=== Phase 6: Fine-tune n={n_best} ({t6:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = fine_tune(a_norm, t6, seed=99999)
            try_update(a_out, "fine1")

    # === Phase 7: Final LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 30:
            print(f"\n=== Phase 7: Final LP n={n_best} ({t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=False)
            try_update(a_out, "lp_final")

    # === Phase 8: Final fine-tune ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 15:
            print(f"\n=== Phase 8: Final tune n={n_best} ({t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = fine_tune(a_norm, t8, seed=123456)
            try_update(a_out, "fine_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
