    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8
    b_warm = None
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        
        if M_curr < 1e-15:
            break
        
        # LP solve (fast version)
        b = lp_fast(y, M_curr, max_active=max_active, warm_start=b_warm)
        if b is None:
            b = lp_subgradient(y, M_curr, n_iters=30, max_active=max_active)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        b_warm = g / norm  # Save for warm start next iteration
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_new.copy()
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        # Adaptive gamma
        if stall > 200:
            gamma = max(1.0, gamma - 0.1)
        elif stall == 0:
            gamma = min(2.0, gamma + 0.02)
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}")
        
        # Restart
        if stall > 500:
            rng = np.random.default_rng(n_iters * 7 + 3)
            pert = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.5
            b_warm = None
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=True, n_peaks=3):
    """
    Direct projected subgradient on max autocorrelation.
    Works in the normalized space (sum=1).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_size = 0.5 / n
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        ac = fast_autocorr(a)
        max_ac = ac.max()
        
        if max_ac < best_score:
            best_score = max_ac
            best_a = a.copy()
            stall = 0
            step_size = min(step_size * 1.3, 5.0 / n)
        else:
            stall += 1
            step_size *= 0.95
        
        # Multi-peak subgradient
        sorted_idx = np.argsort(ac)[::-1]
        peaks = sorted_idx[:n_peaks]
        peak_vals = ac[peaks]
        
        # Weight by how close to max
        T_w = max(0.01 * max_ac, 1e-10)
        w = np.exp((peak_vals - max_ac) / T_w)
        w = w / w.sum()
        
        # Combined subgradient
        grad = np.zeros(n)
        for m, p in enumerate(peaks):
            if w[m] < 1e-6:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * w[m] * a[p - idx]
        
        # Project onto tangent space of sum=1
        grad -= np.mean(grad)
        
        # Polyak step
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        polyak = (max_ac - best_score * 1.0) / grad_norm_sq
        step = min(step_size, max(polyak, 0))
        step = max(step, 1e-8 / n)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a
        a = a_new
        steps += 1
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}, ss={step:.2e}")
        
        # Escape
        if stall > 200:
            rng = np.random.default_rng(steps)
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_new = a.copy()
            for idx in idx_pert:
                dest = rng.integers(n)
                amt = a[idx] * rng.uniform(0.1, 0.3)
                a_new[idx] -= amt
                a_new[dest] += amt
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = fast_autocorr(a_new).max()
                if sc < best_score * 1.05:
                    a = a_new
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            step_size = 0.5 / n
            stall = 0
    
    return best_a, best_score

def lse_adam_polish(a0, time_limit, verbose=True, T_init=None, T_final=None):
    """
    Log-domain Adam with LSE-smoothed max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    eps = 1e-14
    b = np.log(a + eps)
    
    if T_init is None:
        T_init = fast_autocorr(a).max() * 0.05
    if T_final is None:
        T_final = max(fast_autocorr(a).max() * 0.002, 1e-8)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.75
    beta2 = 0.999
    eps_adam = 1e-10
    lr = 0.05
    
    t0 = time.monotonic()
    steps = 0
    stall = 0
    last_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** elapsed_frac
        
        # Recover a from b (softmax)
        b_shifted = b - b.max()
        exp_b = np.exp(b_shifted)
        a = exp_b / exp_b.sum()
        
        ac = fast_autocorr(a)
        max_ac = ac.max()
        
        if max_ac < best_score - 1e-12:
            best_score = max_ac
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # LSE weights
        w = np.exp((ac - max_ac) / T)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(ac))
            w[np.argmax(ac)] = 1.0
        
        # Gradient w.r.t. a: grad_j = 2 * sum_k w_k * a_{k-j} for valid k
        # = 2 * (w cross-correlate a)[j]
        a_rev = a[::-1]
        conv_size = 2 * n - 1
        fft_sz = 1 << (conv_size - 1).bit_length()
        W = rfft(w, fft_sz)
        AR = rfft(a_rev, fft_sz)
        conv_wa = irfft(W * AR, fft_sz)
        grad_a = 2.0 * conv_wa[n - 1:2 * n - 1]
        
        # Gradient w.r.t. b (log-domain)
        sum_a_grad = np.dot(a, grad_a)
        grad_b = a * (grad_a - sum_a_grad)
        
        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        lr_eff = lr * (1.0 - 0.3 * elapsed_frac)
        b = b - lr_eff * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Periodic restart
        if stall > 300:
            rng = np.random.default_rng(steps)
            b = b + rng.normal(0, lr_eff * 0.5, n)
            stall = 0
            m = np.zeros(n)
            v = np.zeros(n)
        
        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam step={steps} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.4e}")
    
    return best_a, best_score

def parametric_init(n, shape='cosine', rng=None):
    """Generate parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif shape == 'raised_cos':
        a = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
    elif shape == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif shape == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif shape == 'bump_edges':
        a = np.exp(-8 * (x - 0.5) ** 2) + 0.5 * (np.exp(-20 * x**2) + np.exp(-20 * (x-1)**2))
    elif shape == 'plateau':
        a = np.maximum(0, 1 - 4 * (x - 0.5) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining LP Nesterov, subgradient polishing, and LSE-Adam.
    Multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)
    
    print(f"=== Starting optimization (budget={time_budget:.0f}s) ===")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        # Try several parametric inits
        n_try = 1600
        shapes = ['cosine', 'raised_cos', 'gaussian', 'sin2', 'bump_edges', 'plateau']
        for shape in shapes:
            a_try = parametric_init(n_try, shape)
            sc = evaluate_safe(a_try)
            if sc < best_score:
                best_score = sc
                best_seq = a_try.copy()
                print(f"  Parametric init '{shape}': score={sc:.6f}")
        if best_seq is None:
            best_seq = parametric_init(n_try, 'cosine')
            best_score = evaluate_safe(best_seq)
    
    n_base = len(best_seq)
    print(f"Initial best: n={n_base}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, verbose=True, max_active=60)
        try_update(a_out, f"lp1_n{n_cur}")
    
    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: SubGrad polish n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_out, _ = subgradient_polish(best_seq, t2, verbose=True, n_peaks=3)
        try_update(a_out, f"subgrad_n{n_cur}")
    
    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, verbose=True, max_active=50)
        try_update(a_out, f"lp2_n{n_cur}")
    
    # === Phase 4: LSE-Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LSE-Adam n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_out, _ = lse_adam_polish(best_seq, t4, verbose=True)
        try_update(a_out, f"lseadam_n{n_cur}")
    
    # === Phase 5: Try longer length ===
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        init_long = interpolate_seq(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t5, verbose=True, max_active=50)
        try_update(a_out, f"lp_n{n2}")
    
    # === Phase 6: Final subgradient polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final SubGrad n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_out, _ = subgradient_polish(best_seq, t6, verbose=True, n_peaks=5)
        try_update(a_out, f"subgrad_final_n{n_cur}")
    
    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, verbose=True, max_active=40)
        try_update(a_out, f"lp_final_n{n_cur}")
    
    # === Phase 8: Tiny LSE-Adam finish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15 and best_seq is not None:
        print(f"\n=== Phase 8: LSE-Adam finish n={n_cur} (budget={remaining:.0f}s) ===")
        a_out, _ = lse_adam_polish(best_seq, min(remaining * 0.15, 60.0), verbose=True)
        try_update(a_out, f"lse_final_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s ===")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(100)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in best_seq]
