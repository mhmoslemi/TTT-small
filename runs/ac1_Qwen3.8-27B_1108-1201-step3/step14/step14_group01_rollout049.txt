import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)

                _solver = s
                print(f"  Using solver: {s}")

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):

            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify no constraint violations for screened LP
        if not do_full and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * (1 + 5e-3):
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum
                last_full_lp = t

        # Update
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_a

        a = a_new
        sc_a = fast_conv_self(a).max()

        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Restart logic
        if no_improve_count > max_no_improve:
            y = a.copy()
            no_improve_count = 0
            max_no_improve = min(int(max_no_improve * 1.4), 300)
            # Small random perturbation scaled to current magnitude
            perturb_size = 0.02 * norm / n
            a = a + perturb_size * rng_local.standard_normal(n)
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a

        if verbose and (t <= 20 or t % 100 == 0):
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, max_no_imp={max_no_improve}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Find peak lags for candidate j selection
            peak_threshold = cur_max * 0.93
            peak_lags = np.where(conv >= peak_threshold)[0]

            # Build candidate set
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:3]:
                if kl < 2 * n - 1:
                    j_struct = int(kl - i)
                    if 0 <= j_struct < n:
                        j_candidates.add(j_struct)
                    j_struct2 = int(kl + i)
                    if 0 <= j_struct2 < n:
                        j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector g[k] = conv change per unit delta transferred from i to j
                # delta conv[k] = a[k-j] + a[j-k] - a[k-i] - a[i-k] (with zero-padding)
                # But we also have second-order terms at k=2i and k=2j
                g = np.zeros(size)

                # a[k-j]: k-j in [0,n-1] => k in [j, j+n-1]
                lo = j
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]

                # a[j-k]: j-k in [0,n-1] => k in [j-n+1, j]
                lo2 = max(0, j - n + 1)
                hi2 = j + 1
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]

                # -a[k-i]: k-i in [0,n-1] => k in [i, i+n-1]
                lo3 = i
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]

                # -a[i-k]: i-k in [0,n-1] => k in [i-n+1, i]
                lo4 = max(0, i - n + 1)
                hi4 = i + 1
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                # conv_new = conv + d*g + second_order
                # second_order: at k=2i: -2*ai*d + d^2; at k=2j: +2*aj*d + d^2
                # For small d, the linear term dominates. Use coarse search.
                n_samples = 20
                deltas = np.linspace(0.01 * ai, 0.95 * ai, n_samples)

                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] += -2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                else:
                    a = a * norm / s_new
                    conv = fast_conv_self(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_perturb = max(3, n // 60)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.03 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Initialize
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast initial improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n1} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, f"lse1_n{n1}")

    # Phase 2: LP Nesterov (main workhorse - allocate most time here)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{n3}")

    # Phase 4: Second LP round from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n4}")

    # Phase 5: Try longer sequence (n+400)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t5 = min(remaining * 0.40, 300.0)
    print(f"\n=== Phase 5: Longer length n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 80:
        init_long = interpolate_seq(best_seq, n_long)
        # Quick LSE on longer sequence
        t5a = min(remaining * 0.15, 80.0)
        if t5a > 20:
            a_out, _ = lse_gradient_descent(init_long, t5a, verbose=False)
            try_update(a_out, f"lse_long_{n_long}")
        # LP on longer
        remaining5 = time_budget - (time.monotonic() - t_global)
        current_best_long = best_seq
        if best_seq is not None and len(best_seq) == n_long:
            current_best_long = best_seq
        else:
            current_best_long = init_long
        t5b = min(remaining5 * 0.40, 180.0)
        if t5b > 30:
            a_out, _ = nesterov_lp_optimize(current_best_long, t5b, solver, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")

    # Phase 6: Final polish on best found
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Quick LSE
        t6a = min(remaining * 0.4, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t6a, verbose=False)
        try_update(a_out, "lse_final")
        # CD
        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 20:
            t6b = min(remaining6 * 0.6, 60.0)
            a_out, _ = pair_cd_optimize(best_seq, t6b, rng, verbose=False)
            try_update(a_out, "pcd_final")
        # One more short LP
        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 40:
            t6c = min(remaining6 * 0.7, 80.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t6c, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
