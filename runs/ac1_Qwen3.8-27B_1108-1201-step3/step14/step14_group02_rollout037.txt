def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - 1)
    rho = np.max(rng[cond]) if np.any(cond) else 0

def coordinate_push(a0, time_limit, verbose=True):
    """
    Systematic coordinate push: move small amount of mass from one coordinate
    to another, choosing the pair that most reduces the max convolution.
    Uses a score-based approach for efficiency.
    """
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_autocorr(a)
    best_m = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    peak_count = min(20, len(conv))
    
    iters = 0
    while time.monotonic() - t0 < time_limit and iters < 500000:
        iters += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find current peaks
        peak_idx = np.argpartition(conv, -peak_count)[-peak_count:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        
        # For each peak, find the coordinate with maximum contribution
        # and try to redistribute
        improved = False
        
        for k in peak_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n, k_int + 1)
            idx = np.arange(js, je)
            j_vals = k_int - idx
            contrib = a[idx] * a[j_vals]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination: minimize increase in peak values
            # Score for destination d: how much would adding mass there increase peaks?
            # Adding to d increases conv[k2] by 2*a[|k2-d|] for each peak k2
            # (approximately, ignoring the removal effect from src)
            
            # Sample candidate destinations
            n_cand = min(n, 300)
            # Bias towards low-value positions
            weights = 1.0 / (a + 0.01)
            weights /= weights.sum()
            candidates = rng.choice(n, size=n_cand, replace=False, p=weights)
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in candidates:
                if dst == src:
                    continue
                # Estimate peak increase
                sens = 0.0
                for k2 in peak_idx[:10]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    if 0 <= d1 < n:
                        sens += a[d1]
                    d2 = dst - k2i
                    if 0 <= d2 < n:
                        sens += a[d2]
                # Remove effect from src
                rem = 0.0
                for k2 in peak_idx[:10]:
                    k2i = int(k2)
                    d1 = k2i - src
                    if 0 <= d1 < n:
                        rem += a[d1]
                    d2 = src - k2i
                    if 0 <= d2 < n:
                        rem += a[d2]
                score = (sens - rem) / max(a[src], 1e-10)
                if score < best_score:
                    best_score = score
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            # Try different fractions
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                new_conv = fast_autocorr(a_trial)
                new_m = new_conv.max()
                if new_m < best_m - 1e-14:
                    a = a_trial
                    conv = new_conv
                    best_m = new_m
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and iters % 100 == 0:
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                new_conv = fast_autocorr(a)
                new_m = new_conv.max()
                if new_m < best_m:
                    best_m = new_m
                    best_a = a.copy()
                    conv = new_conv
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, verbose=True, max_iters=None):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    if not _cvxpy_available:
        return a0, fast_autocorr(a0).max()
    
    solver = get_solver()
    if solver is None:
        return a0, fast_autocorr(a0).max()
    
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_autocorr(a)
    best_m = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    iter_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if max_iters is not None and iter_count >= max_iters:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(np.atleast_1d(b_var.value), dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence([float(x) for x in parent]))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            sc = float(evaluate_sequence([float(x) for x in a_norm]))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        best_seq = make_initialization(n_base, 'sine', rng)
        best_score = 2.0 * n_base * fast_autocorr(best_seq).max()
        print(f"Default init: n={n_base}, est_score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-start (budget {remaining:.0f}s) ===")
    inits = ['sine', 'cosine', 'gaussian', 'power', 'bimodal', 'uniform']
    t_per_init = min(remaining * 0.08 / len(inits), 20.0)
    for init_name in inits:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 10:
            break
        n_test = n0
        a_init = make_initialization(n_test, init_name, rng)
        a_out, _ = subgradient_optimize(a_init, min(t_per_init, remaining * 0.15),
                                         top_k=8, beta=2.5, verbose=False)
        try_update(a_out, f"init_{init_name}")
    
    # Blend parent with best alternative
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        blend = 0.6 * best_seq + 0.4 * make_initialization(len(best_seq), 'gaussian', rng)
        blend = norm_sum1(blend)
        a_out, _ = subgradient_optimize(blend, min(30, remaining * 0.1),
                                         top_k=10, beta=3.0, verbose=False)
        try_update(a_out, "blend")
    
    # === Phase 2: Intensive subgradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Subgradient n={n_best} (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t2, top_k=10, beta=3.0, verbose=True)
        try_update(a_out, "sg_phase2")
    
    # === Phase 3: Coordinate push ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord push n={n_best} (budget {remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_push(a_in, t3, verbose=True)
        try_update(a_out, "coord_push")
    
    # === Phase 4: LP Frank-Wolfe (if n is manageable) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP n={n_best} (budget {remaining:.0f}s) ===")
    if n_best <= 2500:
        t4 = min(remaining * 0.20, 300.0)
        if t4 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t4, verbose=True)
            try_update(a_out, "lp_phase4")
    
    # === Phase 5: More coordinate push ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord push 2 n={n_best} (budget {remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_push(a_in, t5, verbose=True)
        try_update(a_out, "coord_push2")
    
    # === Phase 6: Subgradient with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SG alt params n={n_best} (budget {remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t6, top_k=15, beta=2.0, verbose=True)
        try_update(a_out, "sg_alt")
    
    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length (budget {remaining:.0f}s) ===")
    target_lengths = [1200, 2000, 2400, 2800, 3200, 4000]
    target_lengths = [tl for tl in target_lengths if tl != n_best]
    target_lengths.sort()
    t_per_len = min(remaining * 0.6 / max(len(target_lengths), 1), 80.0)
    
    for n_target in target_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        if t_per_len < 10:
            continue
        print(f"  Trying n={n_target}...")
        a_interp = interpolate_to(best_seq, n_target)
        a_out, _ = subgradient_optimize(a_interp, min(t_per_len, remaining * 0.3),
                                         top_k=10, beta=3.0, verbose=False)
        try_update(a_out, f"multilen_{n_target}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > t_per_len * 0.4:
            a_out2, _ = coordinate_push(a_out, min(t_per_len * 0.4, remaining * 0.3), verbose=False)
            try_update(a_out2, f"coord_{n_target}")
    
    # === Phase 8: LP at best length if feasible ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} (budget {remaining:.0f}s) ===")
    if n_best <= 2500 and remaining > 30:
        t8 = min(remaining * 0.35, 250.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, verbose=True)
            try_update(a_out, "lp_final")
    
    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord polish n={n_best} (budget {remaining:.0f}s) ===")
    if remaining > 15:
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_push(a_in, t9, verbose=True)
            try_update(a_out, "final_coord")
    
    # === Phase 10: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 10: Final SG n={n_best} (budget {remaining:.0f}s) ===")
    if remaining > 15:
        t10 = min(remaining * 0.5, 120.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_optimize(a_in, t10, top_k=12, beta=2.5, verbose=True)
            try_update(a_out, "final_sg")
    
    # Final validation
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
