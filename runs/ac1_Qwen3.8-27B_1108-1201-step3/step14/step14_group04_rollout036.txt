def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.full(n, target_sum / n)
    return target_sum * v / s


def lse_gradient(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):

        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 4.0), 1.0)
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_gradient(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = project_simplex(a_new + mom * (a_new - staging), 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 2000 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:

                    best_a = a.copy()

def mirror_descent(a0, time_limit, T_init=5.0, T_min=0.1, verbose=True):
    """Multiplicative (mirror) descent with LSE gradient - respects non-negativity naturally."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    T = T_init
    eta = 0.5
    stall = 0

    while time.monotonic() - t0 < time_limit:
        grad = lse_gradient(a, T)
        # Multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i])
        # Then normalize to sum = 1
        log_ratio = -eta * grad
        log_ratio -= log_ratio.max()  # numerical stability
        ratios = np.exp(log_ratio)
        a_new = a * ratios
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-12:
            a = a_new
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive step size and temperature
        if stall > 20:
            eta *= 0.7
            T = max(T * 0.95, T_min)
            if eta < 0.001:
                break
        elif stall == 0:
            eta = min(eta * 1.1, 2.0)

        if verbose and (t0 and (time.monotonic() - t0) % 10 < 0.1):
            print(f"    MirrorDescent: best={best_score:.8f}, T={T:.3f}, eta={eta:.4f}")

    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        a = a * norm / s0

    y = a.copy()

    n_filled = 0

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()

        a = a_new

            best_score = score_y
            best_a = y.copy()
        if score_y < best_m:
            best_m = score_y

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_a < best_m:
            best_m = score_a

        # Sliding window average

        window_idx = (window_idx + 1) % window_size
        n_filled = min(n_filled + 1, window_size)

        if verbose and n_iters % 30 == 0:

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        # Find all lags that are within 1% of peak
        threshold = peak_val * 0.99
        peak_lags = np.where(b >= threshold)[0]

        # Compute gradient contribution from peak lags
        grad = np.zeros(n)
        for k in peak_lags:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * a[ki]

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Take a step in negative gradient direction
        for step_size in [0.001, 0.005, 0.01, 0.02, 0.05]:
            a_trial = a - step_size * grad * peak_val
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
                break
        else:
            # No improvement from gradient step, try random perturbation
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break

    return best_a, best_score


def targeted_peak_shave(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution from peak-contributing terms."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find destinations: positions not contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            # Sort by current value (move to low-value positions)
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(3, len(candidates_sorted) // 10)
            dst_pool = candidates_sorted[:n_try]

            for _ in range(min(5, len(dst_pool))):
                dst = dst_pool[rng.integers(len(dst_pool))]
                for frac in [0.0003, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random restart perturbation
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
            if stall > 600:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def make_initial_sequence(n, variant=0):
    """Create various initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if variant == 0:
        # sinc^2 based
        t = (x - 0.5) * np.pi
        a = np.sinc(x - 0.5) ** 2
    elif variant == 1:
        # Gaussian
        a = np.exp(-10 * (x - 0.5) ** 2)
    elif variant == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 3:
        # Bump (compact support in middle)
        a = np.exp(-50 * (x - 0.5) ** 4)
    elif variant == 4:
        # Two bumps
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif variant == 5:
        # Uniform
        a = np.ones(n)
    elif variant == 6:
        # Sine squared
        a = np.sin(np.pi * x) ** 2
    elif variant == 7:
        # Inverse linear
        a = 1.0 / (x + 0.1)
    elif variant == 8:
        # Random smooth
        rng = np.random.default_rng(variant * 1000)
        a = np.abs(np.fft.irfft(np.exp(-rng.uniform(0, 3, n // 2 + 1) * np.arange(n // 2 + 1))))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization phases."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        # Try multiple initializations and pick the best
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for v in range(9):
            init = make_initial_sequence(n_base, v)
            sc = evaluate_safe(init)
            if np.isfinite(sc) and sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = best_init_score
        n_base = 1600
        print(f"Best init: score={best_score:.6f}, n={n_base}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [30.0, 10.0, 3.0, 1.0, 0.3]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP-Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.45, 750.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Mirror descent (novel) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Mirror Descent n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, t3, T_init=5.0, T_min=0.05, verbose=True)
        try_update(a_out, f"mirror_n{n_best}")

    # === Phase 4: Targeted peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 170.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Targeted peak shave n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t4, seed=777)
        try_update(a_out, f"targeted_n{n_best}")

    # === Phase 5: LP polish after mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 40:
        print(f"\n=== Phase 5: LP polish n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Gradient peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 130.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Gradient peak shave n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_peak_shave(a_norm, t6, seed=555)
        try_update(a_out, f"gradshave_n{n_best}")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t7 = min(remaining * 0.12, 200.0)
    if t7 > 40:
        print(f"\n=== Phase 7: LP at n={n2} (budget={t7:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t7, solver, verbose=True)
        try_update(a_out, f"lp3_n{n2}")

    # === Phase 8: Final targeted peak shave on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.08, 120.0)
    if t8 > 15:
        print(f"\n=== Phase 8: Final peak shave n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t8, seed=99999)
        try_update(a_out, "final_shave")

    # === Phase 9: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 30:
            print(f"\n=== Phase 9: Final LP n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov_improved(a_norm, t9, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
