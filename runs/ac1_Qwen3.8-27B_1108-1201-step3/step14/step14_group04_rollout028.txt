def fast_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    elif kind == 'bilateral_exp':
        center = n // 2
        dist = np.abs(np.arange(n) - center) / center
        a = np.exp(-dist * 1.5)
    elif kind == 'random_smooth':
        # Generate random smooth signal via low-pass filtering
        raw = rng.random(n)
        # Simple moving average smoothing
        kernel_size = max(1, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(raw, kernel, mode='same')
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            a = np.ones(n)
    elif kind == 'quartic':
        a = 1.0 - (2 * x - 1) ** 4
        a = np.maximum(a, 0.0)
    else:
        a = np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix: T[k, j] = a[j] if valid
    # (a*b)[k] = sum_j a[j] * b[k-j] where 0 <= k-j < n
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        # b contributes to (a*b)[k] at positions k = j, j+1, ..., j+n-1
        # T[k, j] = a[k-j] for k = j..j+n-1
        T_mat[j:j + n, j] = a[n - 1: -1: -1]  # a reversed: a[n-1], a[n-2], ..., a[0]
    # Wait, let me redo this. (a*b)[k] = sum_j a[j]*b[k-j].
    # Fix j: b[k-j] appears when k-j is in [0, n-1], i.e., k in [j, j+n-1].
    # At position k, the coefficient of b[k-j] is a[j]. So T_mat[k, j] = a[j] when k in [j, j+n-1].
    # Let me rebuild:
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a[j]
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_minimax(a0, time_limit, solver, verbose=True):
    """LP-based minimax with Nesterov acceleration and ergodic averaging."""
    n = len(a0)
    a = project_sum1(a0)
    best_a = a.copy()
    best_score = eval_score_normalized(a)
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_print = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_conv(y).max()
        if M_curr < 1e-15:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b = project_sum1(b)
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b

        if t == 1:
            a_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        # Track best
        sc = eval_score_normalized(y)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
        sc = eval_score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Ergodic averaging
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            sc_avg = eval_score_normalized(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()

        if verbose and n_iters - last_print >= 30:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}")
            last_print = n_iters

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_passes_max=20, verbose=True):
    """
    Coordinate descent: for each coordinate i, find optimal a[i] given others fixed.
    Handles the normalization properly by optimizing J(t) = 2n*max_k(base[k]+g_i(k,t))/(S'+t)^2.
    """
    n = len(a0)
    a = project_sum1(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = eval_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for pass_num in range(n_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        pass_improved = 0

        for i in order:
            if time.monotonic() - t0 > time_limit * 0.95:
                break

            old_val = a[i]
            if old_val < 1e-15 and pass_num > 0:
                # Skip zero coordinates in later passes unless random
                if rng.random() > 0.01:
                    continue

            S_prime = a.sum() - old_val  # sum of all other elements
            if S_prime < 1e-15:
                continue

            # Compute base: convolution without a[i]'s contribution
            conv_full = fast_conv(a)
            # Subtract contribution of a[i]
            # a[i] contributes to conv_full[k] as:
            #   2*a[i]*a[k-i] if 0 <= k-i < n and k != 2*i
            #   a[i]^2 if k == 2*i
            conv_base = conv_full.copy()
            # Valid k for a[i]*a[k-i]: k-i in [0, n-1], so k in [i, i+n-1]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            valid_k = np.arange(k_start, k_end + 1)
            for k in valid_k:
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        conv_base[k] -= old_val * old_val
                    else:
                        conv_base[k] -= 2.0 * old_val * a[j]

            # Now optimize over t = a[i]
            # J(t) = 2n * max_k(conv_base[k] + g_i(k,t)) / (S_prime + t)^2
            # where g_i(k,t) = 2*t*a[k-i] for valid k != 2i, t^2 for k=2i, 0 otherwise

            # Build the coefficients for the linear part
            # For each valid k (k in [i, i+n-1], k != 2i): contribution is 2*t*a[k-i]
            # For k = 2i: contribution is t^2
            # For invalid k: contribution is 0

            # The function to minimize: f(t) = max(all_vals) / (S_prime + t)^2
            # where all_vals[k] = conv_base[k] + g_i(k, t)

            # Use grid search + refinement
            # Range: [0, 3 * old_val] or [0, 3 * mean]
            a_mean = a.sum() / n
            t_max = max(3.0 * old_val, 3.0 * a_mean, 1e-6)

            # Coarse grid
            n_grid = 40
            t_vals = np.linspace(0.0, t_max, n_grid)
            best_t = t_vals[0]
            best_f = float('inf')

            # Precompute for each k: the linear coefficient (2*a[k-i] if valid)
            # and whether k is the quadratic one
            lin_coeff = np.zeros(2 * n - 1)
            quad_k = 2 * i  # the k where t^2 appears
            for k in range(k_start, k_end + 1):
                j = k - i
                if 0 <= j < n and k != quad_k:
                    lin_coeff[k] = 2.0 * a[j]

            # Precompute the k's that are valid for linear contribution
            valid_lin_ks = np.where(lin_coeff > 0)[0]

            for t in t_vals:
                # Compute max of conv_base[k] + lin_coeff[k]*t for linear part
                if len(valid_lin_ks) > 0:
                    lin_part = conv_base[valid_lin_ks] + lin_coeff[valid_lin_ks] * t
                    max_lin = lin_part.max()
                else:
                    max_lin = 0.0

                # Quadratic part at k = quad_k
                if 0 <= quad_k < 2 * n - 1:
                    quad_val = conv_base[quad_k] + t * t
                else:
                    quad_val = 0.0

                # Invariant part (k's not affected by t)
                invariant_ks = np.where(lin_coeff == 0)[0]
                if 0 <= quad_k < 2 * n - 1:
                    invariant_ks = invariant_ks[invariant_ks != quad_k]
                if len(invariant_ks) > 0:
                    max_inv = conv_base[invariant_ks].max()
                else:
                    max_inv = 0.0

                overall_max = max(max_lin, quad_val, max_inv)
                denom = (S_prime + t) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * overall_max / denom

                if f_val < best_f:
                    best_f = f_val
                    best_t = t

            # Refine around best_t
            if best_t > 0 and best_t < t_max:
                t_range = t_max / n_grid
                fine_vals = np.linspace(max(0, best_t - t_range), min(t_max, best_t + t_range), 20)
                for t in fine_vals:
                    if t < 0:
                        continue
                    if len(valid_lin_ks) > 0:
                        lin_part = conv_base[valid_lin_ks] + lin_coeff[valid_lin_ks] * t
                        max_lin = lin_part.max()
                    else:
                        max_lin = 0.0
                    quad_val = conv_base[quad_k] + t * t if 0 <= quad_k < 2 * n - 1 else 0.0
                    max_inv = conv_base[invariant_ks].max() if len(invariant_ks) > 0 else 0.0
                    overall_max = max(max_lin, quad_val, max_inv)
                    denom = (S_prime + t) ** 2
                    if denom < 1e-30:
                        continue
                    f_val = 2.0 * n * overall_max / denom
                    if f_val < best_f:
                        best_f = f_val
                        best_t = t
            elif best_t == 0:
                # Check small values
                for t in np.linspace(0, t_max / n_grid, 10):
                    if len(valid_lin_ks) > 0:
                        lin_part = conv_base[valid_lin_ks] + lin_coeff[valid_lin_ks] * t
                        max_lin = lin_part.max()
                    else:
                        max_lin = 0.0
                    quad_val = conv_base[quad_k] + t * t if 0 <= quad_k < 2 * n - 1 else 0.0
                    max_inv = conv_base[invariant_ks].max() if len(invariant_ks) > 0 else 0.0
                    overall_max = max(max_lin, quad_val, max_inv)
                    denom = (S_prime + t) ** 2
                    if denom < 1e-30:
                        continue
                    f_val = 2.0 * n * overall_max / denom
                    if f_val < best_f:
                        best_f = f_val
                        best_t = t

            # Update
            if best_f < best_score - 1e-14:
                a[i] = best_t
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    sc = eval_score_normalized(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        pass_improved += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_num} ({elapsed:.0f}s): best={best_score:.8f}, improved={pass_improved}")

        if pass_improved == 0:
            # Try random perturbation before giving up
            if pass_num < n_passes_max - 2:
                for _ in range(5):
                    idx = rng.integers(n)
                    amt = a[idx] * 0.05
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[idx] -= amt
                    if a_trial[idx] < 0:
                        continue
                    jdx = rng.integers(n)
                    a_trial[jdx] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    sc = eval_score_normalized(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break
            else:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to length new_n, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with LP minimax and coordinate descent."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create initial
    if parent is None:
        n_base = 1600
        best_seq = make_initializer(n_base, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Try multiple initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 0: Multi-init exploration ===")
        n_base = len(best_seq)
        inits = [
            ('sine', 42), ('raised_cos', 43), ('quartic', 44),
            ('random_smooth', 100), ('bilateral_exp', 45), ('triangular', 46)
        ]
        if parent is not None:
            inits.append(('parent', 0))

        for kind, s in inits:
            if time.monotonic() - t_global > time_budget - 100:
                break
            if kind == 'parent':
                a_init = parent.copy()
            else:
                if len(best_seq) != n_base and kind != 'parent':
                    a_init = interpolate_sequence(parent if parent is not None else best_seq, n_base)
                else:
                    a_init = make_initializer(n_base, kind, seed=s)
            sc = eval_score_normalized(a_init)
            if sc < best_score:
                best_score = 2.0 * n_base * sc  # Convert to full score
                # Actually just use evaluate_safe
                real_sc = evaluate_safe(a_init)
                if real_sc < best_score:
                    best_score = real_sc
                    best_seq = a_init.copy()
                    print(f"  Better init [{kind}]: {real_sc:.6f}")
            else:
                print(f"  Init [{kind}]: score={evaluate_safe(a_init):.6f}")

    # === Phase 1: LP minimax (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP-Nesterov n={n_best} (budget={t1:.0f}s) ===")
        a_out, _ = lp_minimax(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Coordinate descent n={n_best} (budget={t2:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t2, n_passes_max=15, verbose=True)
        try_update(a_out, f"cd1_n{n_best}")

    # === Phase 3: LP again after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP polish n={n_best} (budget={t3:.0f}s) ===")
        a_out, _ = lp_minimax(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 4: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 30:
        print(f"\n=== Phase 4: CD polish n={n_best} (budget={t4:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t4, n_passes_max=10, verbose=True)
        try_update(a_out, f"cd2_n{n_best}")

    # === Phase 5: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_best + 300
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 60:
        print(f"\n=== Phase 5: LP at n={n2} (budget={t5:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = lp_minimax(init_long, t5, solver, verbose=True)
        try_update(a_out, f"lp_long_n{n2}")

    # === Phase 6: CD on longer ===
    remaining = time_budget - (time.monotonic() - t_global)
    if len(best_seq) == n2:
        t6 = min(remaining * 0.10, 100.0)
        if t6 > 20:
            print(f"\n=== Phase 6: CD at n={n2} (budget={t6:.0f}s) ===")
            a_out, _ = coordinate_descent(best_seq, t6, n_passes_max=8, verbose=True)
            try_update(a_out, f"cd_long_n{n2}")

    # === Phase 7: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_out, _ = lp_minimax(best_seq, t7, solver, verbose=False)
            try_update(a_out, "lp_final")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final CD n={len(best_seq)} (budget={t8:.0f}s) ===")
            a_out, _ = coordinate_descent(best_seq, t8, n_passes_max=5, verbose=False)
            try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
