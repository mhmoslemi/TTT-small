    """Build the (2n-1) x n matrix T where T @ b gives conv(a, b)."""
    n = len(a)

    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = min(size, j + n)
        T[lo:hi, j] = a[:hi - lo]
    return T

def gauss_newton_refine(a0, time_limit, verbose=True):
    """Gauss-Newton refinement using active set identification."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for iteration in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        cur_score = M  # This is the score before the 2n/(sum)^2 factor

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Identify active set: bins within 1% of max
        threshold = M * 0.995
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            break

        # Build the Gauss-Newton system
        # Variables: da (n-dim) and dM (scalar)
        # Equations: conv(a+da, a+da)[k] - (M + dM) = 0 for k in active
        #           sum(a + da) - norm = 0
        # Linearized: J @ da - dM * 1 = -(conv[k] - M) for k in active
        #             sum(da) = 0

        m = len(active)
        # Jacobian: J[k, i] = 2 * a[k - i] if 0 <= k-i < n, else 0
        # This is a Hankel-like matrix
        J = np.zeros((m, n))
        for idx, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n, k + 1)
            # J[idx, i] = 2 * a[k - i] for i in [lo, hi)
            # a[k-i] for i in [lo, hi) is a[k-hi+1 : k-lo+1] (reversed)
            a_slice = a[k - hi + 1 : k - lo + 1][::-1]
            J[idx, lo:hi] = 2.0 * a_slice

        # System: [J  -1] [da]   [M - conv[active]]
        #         [1'  0] [dM] = [0]
        
        # Augmented system
        A_sys = np.zeros((m + 1, n + 1))
        A_sys[:m, :n] = J
        A_sys[:m, n] = -1.0
        A_sys[m, :n] = 1.0
        A_sys[m, n] = 0.0

        b_sys = np.zeros(m + 1)
        b_sys[:m] = M - conv[active]
        b_sys[m] = 0.0

        # Solve the linear system
        try:
            sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            try:
                sol, _, _, _ = np.linalg.lstsq(A_sys, b_sys, rcond=None)
            except Exception:
                break

        da = sol[:n]
        dM = sol[n]

        # Damping: limit the step size
        # Ensure a + da >= 0
        if np.any(a + da < 0):
            # Find the maximum step
           _neg = da[da < 0]
            if len(_neg) > 0:
                max_step = -a[da < 0].min() / (-_neg).max() if (-_neg).max() > 0 else 1.0
                max_step = max_step * 0.9
            else:
                max_step = 1.0
            da = da * min(1.0, max_step)
            dM = dM * min(1.0, max_step)

        a_new = a + da
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        new_conv = fast_autocorr(a_new)
        new_M = new_conv.max()

        if new_M < M * (1 - 1e-12):
            a = a_new
        else:
            # Try a smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = a + da * frac
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                new_conv_try = fast_autocorr(a_try)
                if new_conv_try.max() < M:
                    a = a_try
                    break

        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GN iter={iteration} ({elapsed:.0f}s): M={new_M:.8f}, active={m}")

    return best_a, best_score

                    # delta_conv[k] = -2*d*a[k-i] + 2*d*a[k-j] (linearized, ignoring second order)
                    # For the actual change including second order:
                    # conv_new[k] = conv[k] + d*(2*a[k-j] - 2*a[k-i]) + d^2*(delta terms)
                    
                    # Compute shifted vectors
                    v_i = np.zeros(size)
                    v_j = np.zeros(size)
                    
                    # v_i[k] = a[k-i] if 0 <= k-i < n
                    lo = max(0, i)
                    hi = min(size, i + n)
                    if hi > lo:
                        v_i[lo:hi] = a[:hi - lo]
                    
                    # v_j[k] = a[k-j] if 0 <= k-j < n
                    lo = max(0, j)
                    hi = min(size, j + n)
                    if hi > lo:
                        v_j[lo:hi] = a[:hi - lo]
                    
                    # Linearized delta: 2*d*(v_j - v_i)
                    # Second order: d^2 for k=2i (lose a[i]^2 term) and k=2j (gain)
                    # Actually for moving d from i to j:
                    # a_new[i] = a[i] - d, a_new[j] = a[j] + d
                    # conv_new[k] = conv[k] + (-d)*2*a[k-i] + d*2*a[k-j] + d^2*(correction)
                    # The correction is: if k = 2i, we have -2*d*a[i] + d^2 (from (a[i]-d)^2 vs a[i]^2)
                    # if k = 2j, we have +2*d*a[j] + d^2 (from (a[j]+d)^2 vs a[j]^2)
                    
                    delta_linear = 2.0 * (v_j - v_i)
                    
                    # Find optimal d using a few samples
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 20
                    d_max = a[i] * 0.99
                    deltas = np.linspace(0.0, d_max, n_samples)
                    
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_linear
                        # Second order corrections
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_d = d
                    
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-14:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            # Revert
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
