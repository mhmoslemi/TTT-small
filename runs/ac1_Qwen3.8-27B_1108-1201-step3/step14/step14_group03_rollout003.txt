def compute_lse_gradient(a, w):
    """Compute gradient of sum_k w_k * c_k where c = conv(a,a).
    Uses FFT for O(n log n) complexity.
    Returns gradient of shape (n,).
    """
    n = len(a)
    size = 2 * n - 1

    # grad_j = sum_k w_k * [a_{k-j} * 1_{valid} + a_{j-k} * 1_{valid}]
    # cross2_j = sum_m a[m] * w[j-m] = (a * w)_j  [standard convolution]
    # cross1_j = sum_m a[m] * w[m+j] = (a * w_rev)_{size-1-j}  [conv with reversed w]

    w_rev = w[::-1].copy()

    # Convolution of a (length n) with w (length size=2n-1) has length 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    W_rev = rfft(w_rev, fft_size)

    conv_aw = irfft(A * W, fft_size)
    conv_awrev = irfft(A * W_rev, fft_size)

    # cross2_j = conv_aw[j] for j=0..n-1
    cross2 = conv_aw[:n]

    # cross1_j = conv_awrev[size-1-j] for j=0..n-1
    # size-1 = 2n-2, so indices: 2n-2, 2n-3, ..., 2n-2-(n-1) = n-1
    cross1 = conv_awrev[n - 1:2 * n - 1][::-1]

    return cross1 + cross2


def lse_mirror_descent(a0, time_limit, T_start=5.0, T_end=0.1, n_temps=10,
                       verbose=True, label="LSE"):
    """Mirror descent on LSE-smoothed max autocorrelation.
    Uses FFT-based gradient and multiplicative updates.

    norm = np.sqrt(2.0 * n)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_temps)
    time_per_temp = time_limit / n_temps

    total_iters = 0
    for ti, T in enumerate(temps):
        t_start = time.monotonic()
        n_iters = 0
        eta = 0.8
        no_improve = 0

        while time.monotonic() - t_start < time_per_temp:
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            M = conv.max()

            # Softmax weights
            shifted = (conv - M) / T
            shifted -= shifted.max()
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(conv))] = 1.0
            else:
                w /= w_sum

            # FFT-based gradient
            grad = compute_lse_gradient(a, w)

            # Multiplicative update with clipping
            max_g = np.abs(grad).max()
            if max_g > 1e-15:
                eta_eff = eta / max_g
            else:
                eta_eff = eta

            log_mult = -np.clip(eta_eff * grad, -30.0, 30.0)
            a_mult = a * np.exp(log_mult)
            a_mult = np.maximum(a_mult, 1e-30)

            # Small linear component for stability
            alpha = 0.05
            lin_step = 0.3 / n

            s_new = a_new.sum()
            if s_new < 1e-15:

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-11:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        break
                    if a[i] < norm * 1e-11:
                        break

                    # Compute delta vector for moving mass from i to j
                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] += a[hi1 - n + lo1 - lo1:hi1 - lo1] if False else a[0:n]
                        # More carefully:
                        delta_vec[lo1:hi1] = 0  # reset
                    # Redo properly
                    delta_vec = np.zeros(size)
                    # Terms where a_i appears in conv: positions i..i+n-1 (forward) and (n-1-i)..(n-1+i) (reflected)
                    for pos in range(max(0, i), min(size, i + n)):
                        idx_a = pos - i
                        if 0 <= idx_a < n:
                            delta_vec[pos] += a[idx_a]
                    for pos in range(max(0, n - 1 - i), min(size, n + i)):
                        idx_a = n - 1 - (pos - (n - 1 - i))
                        if 0 <= idx_a < n:
                            delta_vec[pos] += a[idx_a]

                    # Subtract terms where a_j appears
                    for pos in range(max(0, j), min(size, j + n)):
                        idx_a = pos - j
                        if 0 <= idx_a < n:
                            delta_vec[pos] -= a[idx_a]
                    for pos in range(max(0, n - 1 - j), min(size, n + j)):
                        idx_a = n - 1 - (pos - (n - 1 - j))
                        if 0 <= idx_a < n:
                            delta_vec[pos] -= a[idx_a]

                    delta_vec *= 2.0

                    # Line search: find best d in [0, a[i]]
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 12
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-14:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ===== Phase 1: LSE Mirror Descent (40% time) =====
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    n1 = n_base
    print(f"\n=== Phase 1: LSE Mirror Descent n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = lse_mirror_descent(a_norm, t1, T_start=5.0, T_end=0.15,
                                      n_temps=10, verbose=True, label="LSE1")
        do_try_update(a_out, f"lse1_n{n1}")

    # ===== Phase 2: LP Nesterov (20% time) =====
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, label="LP1")
        do_try_update(a_out, f"lp1_n{n_best}")

    # ===== Phase 3: Refined LSE (20% time) =====
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Refined LSE n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_mirror_descent(a_norm, t3, T_start=0.5, T_end=0.05,
                                      n_temps=8, verbose=True, label="LSE2")
        do_try_update(a_out, f"lse2_n{n_best}")

    # ===== Phase 4: Pair CD (12% time) =====
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.12
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t4, rng, window=35, verbose=True, label="PCD1")
        do_try_update(a_out, f"pcd1_n{n_best}")

    # ===== Phase 5: Final LP Nesterov (8% time) =====
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.08
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final LP Nesterov n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, label="LP2")
        do_try_update(a_out, f"lp2_n{n_best}")

    # ===== Phase 6: Alternate length exploration (remaining time) =====
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for n_alt in [n_base - 50, n_base + 50]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t6 = min(remaining * 0.45, 120.0)
            if t6 < 30:
                break
            print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out_alt, _ = lse_mirror_descent(a_norm_alt, t6 * 0.5, T_start=3.0,
                                              T_end=0.1, n_temps=5, verbose=False, label="LSE_alt")
            do_try_update(a_out_alt, f"lse_alt_n{n_alt}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30 and best_seq is not None and len(best_seq) == n_alt:
                a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                a_out_alt2, _ = pair_cd_window(a_norm_alt2, min(remaining * 0.4, 40.0),
                                               rng, window=25, verbose=False, label="PCD_alt")
                do_try_update(a_out_alt2, f"pcd_alt_n{n_alt}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
