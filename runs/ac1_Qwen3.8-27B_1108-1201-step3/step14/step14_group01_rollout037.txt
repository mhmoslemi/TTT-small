def fast_conv_self(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute cross-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def proj_simplex(v, c):

    n = len(v)

def fast_conv_self(a):

def proj_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    if c <= 0:
        return np.zeros(n)
    v_clipped = np.maximum(v, 0.0)
    sv = v_clipped.sum()
    if sv < 1e-15:
        return np.full(n, c / n)
    if abs(sv - c) < 1e-10 * c:
        return v_clipped
    # Binary search for threshold
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.maximum(v - t, 0.0)

def subgradient_optimize(a0, time_limit, verbose=True):
    """Projected subgradient method for max conv(a,a) with sum(a)=norm, a>=0."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = make_symmetric(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        a = make_symmetric(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # AdaGrad-like step size
    sq_sum = np.zeros(n)
    eta0 = norm * 2.0 / np.sqrt(n)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Find active set (top 20% or threshold)
        threshold = max_c * 0.8
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.where(conv == max_c)[0]
        n_active = len(active)

        # Compute subgradient
        # g[i] = (1/|K|) * sum_{k in K} (a[k-i] + a[i-k]) with a[j]=0 outside [0,n-1]
        g = np.zeros(n)
        for k in active:
            # a[k-i] valid when 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi > lo:
                # a[k-i] for i in [lo, hi): a[k-lo], a[k-lo-1], ..., a[k-hi+1]
                idx = k - np.arange(lo, hi)
                g[lo:hi] += a[idx]
            # a[i-k] valid when 0 <= i-k <= n-1, i.e., k <= i <= k+n-1
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if hi2 > lo2:
                idx2 = np.arange(lo2, hi2) - k
                g[lo2:hi2] += a[idx2]
        g /= n_active

        # Step size
        sq_sum += g * g
        step = eta0 / np.sqrt(sq_sum + 1e-8)

        a_new = a - step * g
        a_new = proj_simplex(a_new, norm)
        a_new = make_symmetric(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def build_conv_matrix(a):
    """Build T such that (T @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_best_response(a, M, solver, active=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = make_symmetric(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        a = make_symmetric(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Running average for convergence
    avg_a = np.zeros(n)
    n_avg = 0
    avg_best_score = best_score
    avg_best_a = a.copy()

    # Adaptive active set
    active = None
    solve_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Constraint screening with adaptation
        threshold_frac = max(0.5, 0.95 - 0.01 * min(t, 50) / 50.0)
        if active is None or t % 100 == 0:
            thr = M * threshold_frac
            active = np.where(conv_y >= thr)[0]
            if len(active) < 20:
                active = np.where(conv_y >= M * 0.5)[0]

        b = lp_best_response(y, M, solver, active)
        solve_count += 1
        if b is None:
            b = lp_best_response(y, M, solver, None)
            solve_count += 1
        if b is None:
            continue

        # Validate: check if dropped constraints are violated
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, b)
            violated = np.where(check_conv > M * (1 + 1e-5))[0]
            if len(violated) > 0:
                new_active = np.sort(np.concatenate([active, violated]))
                b = lp_best_response(y, M, solver, new_active)
                solve_count += 1
                if b is None:
                    b = lp_best_response(y, M, solver, None)
                    solve_count += 1
                if b is None:
                    continue
                active = new_active

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = make_symmetric(b * norm / b_sum)

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = make_symmetric(np.maximum(y_new, 0.0))
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = make_symmetric(np.maximum(a_new, 0.0))
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best (current and Nesterov iterates)
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = make_symmetric(avg_check * norm / s_avg)
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < avg_best_score:
                    avg_best_score = sc_avg
                    avg_best_a = avg_check.copy()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): y={sc_y:.8f} a={sc_a:.8f} best={best_score:.8f}")

    # Compare average vs best
    if avg_best_score < best_score:
        best_score = avg_best_score
        best_a = avg_best_a

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with vectorized 1D minimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = make_symmetric(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        a = make_symmetric(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Candidate j's: random + structured
            j_candidates = list(set(int(rng.integers(n)) for _ in range(6)))
            # Add symmetry partner
            j_sym = n - 1 - i
            if j_sym != i:
                j_candidates.append(j_sym)
            j_candidates = [j for j in j_candidates if j != i]

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates[:8]:
                aj = a[j]

                # Gain vector: conv change when transferring delta from i to j
                # conv_new[k] = conv[k] + delta*(a[k-j]+a[j-k] - a[k-i]-a[i-k])
                # + delta^2*(delta_{k,2j} - delta_{k,2i})  (second order at self-terms)
                g = np.zeros(size)
                # +a[k-j]: k-j in [0,n-1] -> k in [j, j+n-1]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                # +a[j-k]: j-k in [0,n-1] -> k in [j-n+1, j]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # -a[k-i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                # -a[i-k]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta via sampling + refinement
                n_samples = 20
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    # Second-order corrections
                    if 2 * i < size:
                        new_conv[2 * i] -= 2 * ai * d - d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = make_symmetric(np.maximum(a, 0.0))
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        # Revert
                        a = best_a.copy()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                # Perturb
                n_perturb = max(3, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = make_symmetric(np.maximum(a, 0.0))
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = make_symmetric(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        a = make_symmetric(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(80.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # g[i] = 2 * (sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k])
            # First term: cross-correlation
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            # Second term: convolution
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Adaptive step
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.05 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = make_symmetric(a_new)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            sc = fast_conv_self(a_new).max()
            if sc <= max_c + 0.01 * T_val:
                a = a_new
            else:
                # Halve step and retry
                step *= 0.5
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = make_symmetric(a_new)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                    a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    f_interp = make_symmetric(f_interp)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Subgradient (fast exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Subgradient n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = subgradient_optimize(best_seq, t2, verbose=True)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 600.0)
    n3 = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n3}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 300.0)
    n4 = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n4}")

    # Phase 5: Second LP round (refined)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 250.0)
    n5 = len(best_seq)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{n5}")

    # Phase 6: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 200
    t6 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
    if t6 > 40 and remaining > 200:
        init_alt = interpolate_seq(best_seq, n_alt)
        a
        a_out, _ = nesterov_lp_optimize(init_alt, min(t6 * 0.6, 120.0), solver, verbose=False)
        try_update(a_out, f"lp_alt_{n_alt}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.4, 80.0)
            a_out2, _ = pair_cd_optimize(best_seq, t6b, rng, verbose=False)
            try_update(a_out2, f"pcd_alt_{n_alt}")

    # Phase 7: Deep polish - alternate LSE and CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Deep polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        # Quick subgradient
        t7a = min(remaining * 0.25, 60.0)
        a_out, _ = subgradient_optimize(best_seq, t7a, verbose=False)
        try_update(a_out, "subgrad_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # Pair CD
            t7b = min(remaining * 0.4, 100.0)
            a_out, _ = pair_cd_optimize(best_seq, t7b, rng, verbose=False)
            try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # Final LP
            t7c = min(remaining * 0.5, 80.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t7c, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
