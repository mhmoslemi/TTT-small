def fft_size_for(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fast_conv_self(a):
    n = len(a)
    fs = fft_size_for(n)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:2 * n - 1]

def fast_conv_cross(a, b):
    n = len(a)
    fs = fft_size_for(n)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:2 * n - 1]

def compute_local_score(a):

def build_T_rows(a, active):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                T[idx, i] = a[j]
    return T

def lp_solve_fast(a, M, solver, active):
    n = len(a)
    T = build_T_rows(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

            break

        elapsed = time.monotonic() - t0
        # Decay momentum

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        # Phase 1 (early): top 8%, Phase 2 (later): top 20%

                    b = b2
                    g = b * S / b.sum()

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:

def peak_shaving(a0, time_limit, verbose=True):
    """Gradient descent on the active peaks of the convolution."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * S / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fs = fft_size_for(n)

    n_peaks = 5  # number of peaks to simultaneously reduce
    lr_init = S / (10.0 * n ** 0.5)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < 1e-12:
            break

        # Find top n_peaks
        peak_lags = np.argsort(conv)[::-1][:n_peaks]

        # Compute gradient of max of top peaks
        # grad[i] = sum over active peaks k of 2*a[k-i] (if valid)
        # Use FFT for efficiency: cross-correlation of a with the peak indicator
        peak_weight = np.zeros(size)
        for k in peak_lags:
            peak_weight[k] = 1.0 / n_peaks

        PW = rfft(peak_weight, fs)
        A = rfft(a, fs)
        # grad[i] proportional to sum_k peak_weight[k] * a[k-i]
        # This is the correlation of peak_weight with a
        grad = irfft(PW * A, fs)[:n]
        # Actually: (peak_weight * a as convolution) at position i
        # conv(PW, a)[i] = sum_k PW[k] * a[i-k] ... hmm need to be careful

        # Let me compute directly: for each active peak k, grad contribution at i is a[k-i]
        # Total grad[i] = sum_k active a[k-i] / n_peaks
        # This equals (a_flipped * peak_weight)[size-1-i] or similar
        # Simpler: just compute via the cross-conv
        # cross[k] = sum_i a[i] * peak_weight[k-i]... no

        # Direct: grad[i] = (1/n_peaks) * sum_{k in peaks} a[k-i] (if 0<=k-i<n)
        # = (1/n_peaks) * sum_{k in peaks} a[max(0, min(n-1, k-i))]... no, just a[k-i] if valid

        # Using the convolution: if w is the peak indicator (length size),
        # then (a * w)[k] = sum_i a[i] * w[k-i]
        # We want for each i: sum_k w[k] * a[k-i]
        # = sum_k w[k] * a[k-i] = (w * a_flipped)[n-1+i]... this is getting confusing

        # Let me just do it directly for the top peaks (only n_peaks of them)
        grad = np.zeros(n)
        for k in peak_lags:
            # grad contribution: a[k-i] for valid i
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= n_peaks

        # Projected gradient step: a_new = a - lr * grad, project to {a>=0, sum=a*S}
        lr = lr_init / (1.0 + 0.01 * it)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * S / s_new

        new_max = fast_conv_self(a_new).max()
        if new_max < cur_max - 1e-14:
            a = a_new
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                if verbose and int((best_score * 1e8)) % 100000 == 0:
                    print(f"    PS it={it}: best={best_score:.10f}")
        else:
            # Backtrack
            lr *= 0.5
            lr_init = lr
            if lr < 1e-15:
                break

    return best_a, best_score

    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def pair_cd_polish(a0, time_limit, rng, verbose=False):
    """Lightweight pair coordinate descent for final polish."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * S / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    no_improve = 0
    max_no_improve = 8

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_lags = np.argsort(conv)[::-1][:10]
        improved = False

        n_tries = min(200, n)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break

            i = int(rng.integers(n))
            # Choose j to reduce a peak
            k = int(peak_lags[int(rng.integers(len(peak_lags)))])
            # Best j: one where a[k-j] is small (transferring to j won't increase conv[k] much)
            # j should be such that k-j is out of range or a[k-j] is small
            candidates = []
            for dj in [1, -1, 2, -2, 5, -5, 10, -10, 50, -50, 100, -100]:
                j = (i + dj) % n
                if j != i:
                    candidates.append(j)
            candidates.append(int(rng.integers(n)))

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in candidates[:8]:
                if j == i:
                    continue
                ai = a[i]
                if ai < 1e-15:
                    continue
                # Try transferring delta from i to j
                # New conv at peak k changes by: -delta*a[k-i] + delta*a[k-j] (approximately)
                # (ignoring second-order terms)
                # Best delta minimizes the new max

                # Quick estimate: transfer delta = ai * 0.1
                deltas = [ai * 0.05, ai * 0.1, ai * 0.2, ai * 0.3, ai * 0.5]
                for d in deltas:
                    a[i] -= d
                    a[j] += d
                    a = a * S / a.sum()
                    v = fast_conv_self(a).max()
                    a[j] -= d
                    a[i] += d
                    a = a * S / a.sum()

                    if v < best_val - 1e-14:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * S / a.sum()
                v = fast_conv_self(a).max()
                if v < best_score - 1e-14:
                    best_score = v
                    best_a = a.copy()
                    improved = True
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * S / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                n_pert = max(10, n // 100)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.05 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a * S / s
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * 3 * x) + 0.05 * np.cos(2 * np.pi * 7 * x)
    elif choice == 1:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 2 * x) ** 2
    elif choice == 2:
        a = 1.0 + 0.15 * np.cos(4 * np.pi * x) * np.sin(2 * np.pi * x)
    elif choice == 3:
        a = np.abs(np.cos(np.pi * x)) ** 0.5
    elif choice == 4:
        a = 1.0 + 0.1 * rng.standard_normal(n)
    else:
        a = np.ones(n) * (1.0 + 0.05 * np.cos(2 * np.pi * 5 * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Pool of good solutions for convex combination
    solution_pool = [best_seq.copy()]

    # === Phase 1: LP Nesterov (main workhorse) - 55% of time ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t1, solver, verbose=True)
        try_update(a_out, "lp1")
        if a_out is not None:
            solution_pool.append(a_out.copy())

    # === Phase 2: Peak shaving - 15% of time ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: Peak shaving (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = peak_shaving(best_seq, t2, verbose=True)
        try_update(a_out, "peak_shave")
        if a_out is not None:
            solution_pool.append(a_out.copy())

    # === Phase 3: LP again from improved point - 15% ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LP round 2 (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, "lp2")
        if a_out is not None:
            solution_pool.append(a_out.copy())

    # === Phase 4: Pair CD polish - 20% ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: Pair CD polish (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_polish(best_seq, t4, rng, verbose=True)
        try_update(a_out, "pcd")
        if a_out is not None:
            solution_pool.append(a_out.copy())

    # === Phase 5: Multi-length exploration - if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length (remaining={remaining:.0f}s) ===")
    if remaining > 80:
        t5 = remaining * 0.5
        best_seq, best_score = multi_length_explore(best_seq, best_score, t5, solver, rng)

    # === Phase 6: Convex combination ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and len(solution_pool) > 1:
        print(f"\n=== Phase 6: Convex combination ===")
        combo_a, combo_sc = convex_combine(solution_pool, 30.0)
        if combo_a is not None:
            try_update(combo_a, "convex_combo")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP polish (budget={remaining:.0f}s) ===")
        a_out, _ = lp_nesterov_optimize(best_seq, min(remaining * 0.7, 120.0), solver, verbose=False)
        try_update(a_out, "final_lp")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, _ = peak_shaving(best_seq, min(remaining * 0.6, 60.0), verbose=False)
        try_update(a_out, "final_ps")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
