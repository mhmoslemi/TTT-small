from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize_to_sum(a, target_sum):
    """Normalize a to have the target sum, keeping non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s


def build_lp_matrix(a):
    """Build the convolution matrix T where (T @ b) = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lse_gradient(a, w, fft_size=None):
    """
    Compute gradient of LSE approximation.
    grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    Computed via FFT as conv(w, a[::-1])[j+n-1].
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_grad = irfft(W * AR, fft_size)
    grad = 2.0 * conv_grad[n - 1:2 * n - 1]
    return grad


def nesterov_lse_optimize(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed objective.
    Minimizes T * log(sum exp(conv(a,a)/T)) subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # Adaptive temperature
    initial_max = fast_autocorr(a).max()
    T = initial_max * 0.1
    T_min = initial_max * 0.005

    y = a.copy()  # Nesterov momentum variable
    prev_y = a.copy()
    t0 = time.monotonic()
    steps = 0
    fft_size = 1 << (3 * n - 2).bit_length()

    # Learning rate
    lr = 1.0 / (n * 0.1)
    patience = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)

        # Compute autocorrelation of y
        b = fast_autocorr(y)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # LSE weights (softmax of b / T)
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Gradient
        grad = lse_gradient(y, w, fft_size)

        # Project gradient to tangent space (sum = 0)
        grad = grad - np.mean(grad)

        # Nesterov step
        t = steps + 1
        alpha = 2.0 / (t + 1.0)
        y_new = (1.0 - alpha) * y + alpha * (y - lr * grad)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 1.0)
        a_new = y_new + momentum * (y_new - prev_y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new
        steps += 1

        # Evaluate
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
            lr = min(lr * 1.02, 5.0 / n)
        else:
            patience += 1

        # Decrease temperature
        T = max(T_min, T * (1.0 - 0.002 * progress))

        # Adaptive learning rate
        if patience > 200:
            lr = max(lr * 0.8, 0.01 / n)
            patience = 0

        # Restart if stuck
        if patience > 1000:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
            a = normalize_to_sum(a, norm)
            y = a.copy()
            prev_y = a.copy()
            lr = 1.0 / (n * 0.1)
            patience = 0
            T = fast_autocorr(a).max() * 0.05

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Nest step={steps} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.4f}, lr={lr:.2e}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """
    Subgradient method with top-peak weighting and Polyak step.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    lower_bound = best_score * 0.99  # Crude lower bound

    t0 = time.monotonic()
    steps = 0
    lr = 0.5 / n
    patience = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        # Top K peaks with softmax weighting
        K = min(10, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:K]
        peak_vals = b[top_peaks]

        T_w = max(0.01 * (max_b - b[sorted_idx[min(K-1, len(sorted_idx)-1)] + 1e-15), 1e-10)
        w = np.exp((peak_vals - max_b) / T_w)
        w = w / w.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for m, p in enumerate(top_peaks):
            if w[m] < 1e-8:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[js:je+1] += 2.0 * w[m] * a[p - idx]

        # Project to tangent space
        grad = grad - np.mean(grad)

        # Polyak step
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        polyak = (max_b - lower_bound) / grad_norm_sq
        step = min(lr, polyak) if polyak > 0 else lr
        step = max(step, 1e-10 / n)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            a = a_new
            best_score = score
            best_a = a.copy()
            lower_bound = max(lower_bound, best_score * 0.999)
            patience = 0
        else:
            patience += 1
            if score < max_b * 1.01:
                a = a_new

        steps += 1

        # Adapt step size
        if patience % 500 == 0:
            if patience > 2000:
                lr = max(lr * 0.5, 1e-8 / n)
            else:
                lr = min(lr * 1.01, 2.0 / n)

        # Restart
        if patience > 5000:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.01 * a.mean(), n), 0)
            a = normalize_to_sum(a, norm)
            lr = 0.5 / n
            patience = 0

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.2e}")

    return best_a, best_score


def lp_best_response(a, M):
    """
    Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the optimal b or None.
    """
    n = len(a)
    import cvxpy as cp

    b_var = cp.Variable(n, nonneg=True)
    T = build_lp_matrix(a)
    constraints = [T @ b_var <= M, cp.sum(b_var) <= 100.0]  # Bound for numerical stability

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                if b_val.sum() > 1e-15:
                    return b_val
        except Exception:
            continue

    # Fallback: use subgradient step as pseudo best response
    b_corr = fast_autocorr(a)
    peak_idx = int(np.argmax(b_corr))
    p = peak_idx
    js = max(0, p - n + 1)
    je = min(n - 1, p)
    idx = np.arange(js, je + 1)
    b_val = a[p - idx] / a[p - idx].max()
    return b_val


def fessler_parks_optimize(a0, time_limit, verbose=True, max_iters=500):
    """
    Fessler-Parks accelerated LP Nesterov method.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    iterations = 0
    gamma = 1.8

    while time.monotonic() - t0 < time_limit and iterations < max_iters:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Solve LP best response
        b = lp_best_response(y, M_curr)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        iterations += 1

        # Fessler-Parks step
        t = iterations
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Adaptive gamma
        if iterations % 50 == 0:
            recent_scores = [fast_autocorr(y).max() for _ in range(1)]
            if iterations > 20:
                osc = abs(M_curr - fast_autocorr(prev_y).max()) / max(M_curr, 1e-15)
                if osc > 0.03:
                    gamma = max(1.2, gamma - 0.1)
                elif osc < 0.005:
                    gamma = min(2.0, gamma + 0.05)

        if verbose and iterations % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP-LP iter={iterations} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}")

    return best_a, best_score


def create_initializations(n, seed=42):
    """Create diverse one-sided initializations."""
    inits = {}

    # Linear ramp (increasing)
    x = np.arange(n, dtype=np.float64)
    ramp_up = x + 1.0
    inits['ramp_up'] = ramp_up / ramp_up.sum()

    # Linear ramp (decreasing)
    ramp_down = n - x
    inits['ramp_down'] = ramp_down / ramp_down.sum()

    # Quadratic ramp
    quad = (x + 1.0) ** 2
    inits['quad_up'] = quad / quad.sum()

    # Exponential decay
    exp_dec = np.exp(-0.5 * x / n * 3)
    inits['exp_dec'] = exp_dec / exp_dec.sum()

    # Exponential growth
    exp_grow = np.exp(0.5 * x / n * 3)
    inits['exp_grow'] = exp_grow / exp_grow.sum()

    # Cosine taper (one-sided)
    cos_taper = np.cos(np.pi * x / (2 * n)) ** 2
    cos_taper = np.maximum(cos_taper, 0.01)
    inits['cos_taper'] = cos_taper / cos_taper.sum()

    # Power law
    power = (x + 1.0) ** 0.5
    inits['sqrt_ramp'] = power / power.sum()

    # Cubic
    cubic = (x + 1.0) ** 3
    inits['cubic'] = cubic / cubic.sum()

    # Uniform (baseline)
    inits['uniform'] = np.ones(n) / n

    # Two-sided: concentrated in middle-left
    center = n * 0.4
    two_side = np.exp(-0.5 * ((x - center) / (n * 0.15)) ** 2)
    two_side = np.maximum(two_side, 1e-6)
    inits['two_side'] = two_side / two_side.sum()

    return inits


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """
    Targeted peak shaving: move mass from high-contributing positions to low ones.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = normalize_to_sum(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stalls = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing indices
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            stalls += 1
            if stalls > 200:
                break
            continue

        src = idx[np.argmax(contrib)]

        # Find non-contributing destination
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True
        non_contrib_idx = np.where(non_contrib)[0]

        if len(non_contrib_idx) == 0:
            stalls += 1
            continue

        # Random destination from non-contributing, weighted by low value
        n_sel = min(5, len(non_contrib_idx))
        dst_candidates = non_contrib_idx[rng.choice(len(non_contrib_idx), size=n_sel, replace=False)]
        dst = dst_candidates[rng.integers(n_sel)]

        # Try several fractions
        best_trial = max_b
        best_frac = 0
        for frac in [0.05, 0.1, 0.2, 0.3, 0.4]:
            amt = a[src] * frac
            if amt < 1e-18:
                break
            trial = a.copy()
            trial[src] -= amt
            trial[dst] += amt
            trial = np.maximum(trial, 0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial * norm / s_t
            sc = fast_autocorr(trial).max()
            if sc < best_trial:
                best_trial = sc
                best_frac = frac

        if best_frac > 0 and best_trial < best_score:
            a[src] -= a[src] * best_frac
            a[dst] += a[src] / (1 - best_frac) * best_frac if best_frac < 1 else 0
            # Redo properly
            a = a.copy()
            amt = best_frac * (a0[src] if False else 0)  # need to recompute
            # Actually just redo from best_a
            a = best_a.copy()
            amt = a[src] * best_frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            s_t = a.sum()
            a = a * norm / s_t
            best_score = best_trial
            best_a = a.copy()
            stalls = 0
        else:
            stalls += 1
            # Try random move
            if stalls % 10 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    amt = a[i] * rng.uniform(0.05, 0.2)
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0)
                    s_t = trial.sum()
                    if s_t > 1e-15:
                        trial = trial * norm / s_t
                        sc = fast_autocorr(trial).max()
                        if sc < best_score:
                            a = trial
                            best_score = sc
                            best_a = a.copy()
                            stalls = 0

        if stalls > 500:
            # Large perturbation
            n_pert = max(10, n // 20)
            a = np.maximum(a + rng.normal(0, 0.05 * a.mean(), n), 0)
            a = normalize_to_sum(a, norm)
            stalls = 0

        if verbose and best_score < 1.51 and (best_score * 10000) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.8f}, stalls={stalls}")

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n + 1)[:-1]
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses one-sided initializations to avoid the
    symmetric local optimum (score >= 2), then refines with LP and gradient methods.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = a - a.min()  # Shift to ensure non-negative
        s = a.sum()
        if s < 0.01:
            a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
            s = a.sum()
            if s < 0.01:
                return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to use parent as a starting point for size reference
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 100:
            n_base = len(p)
            # Normalize and evaluate
            p_clip = np.clip(p, 0, 1000)
            sc = evaluate_safe(p_clip / p_clip.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = p_clip / p_clip.sum()
                print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    print(f"Time budget: {time_budget:.0f}s, base size: {n_base}")
    if best_seq is not None:
        print(f"Initial best: {best_score:.6f}")

    # === Phase 1: Generate and evaluate diverse initializations ===
    print("\n=== Phase 1: Initialization exploration ===")
    init_scores = {}
    for n_try in [n_base]:
        inits = create_initializations(n_try, seed)
        for name, p in inits.items():
            norm_p = p * np.sqrt(2.0 * n_try) / p.sum()
            sc = evaluate_safe(norm_p / norm_p.sum())
            init_scores[name] = (sc, p, n_try)
            if np.isfinite(sc):
                print(f"  {name}: score={sc:.6f}")

    # Find best initialization
    best_init_name = min(init_scores, key=lambda k: init_scores[k][0] if np.isfinite(init_scores[k][0]) else float('inf'))
    best_init_score = init_scores[best_init_name][0]
    best_init = init_scores[best_init_name][1]
    print(f"  Best init: {best_init_name} with score={best_init_score:.6f}")

    if np.isfinite(best_init_score) and best_init_score < best_score:
        n_cur = len(best_init)
        a_norm = best_init * np.sqrt(2.0 * n_cur) / best_init.sum()
        do_try_update(a_norm / a_norm.sum(), f"init_{best_init_name}")

    # === Phase 2: LSE Nesterov on best initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is None:
        # Fallback: use linear ramp
        n_cur = n_base
        x = np.arange(n_cur, dtype=np.float64)
        best_seq = (x + 1.0) / (x + 1.0).sum()
        best_score = evaluate_safe(best_seq)
        print(f"Fallback init: n={n_cur}, score={best_score:.6f}")

    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE-Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # === Phase 3: Fessler-Parks LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: FP-LP n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = fessler_parks_optimize(a_norm, t3, verbose=True, max_iters=300)
        do_try_update(a_out, f"fp_n{n_cur}")

    # === Phase 4: Subgradient refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: SubGrad n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t4, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 5: LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LSE polish n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_optimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"lse2_n{n_cur}")

    # === Phase 6: Try nearby sizes ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Size exploration (budget={remaining:.0f}s) ===")

    for n_try in [n_cur - 100, n_cur + 100, n_cur + 200]:
        if n_try < 200 or remaining < 60:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        print(f"  Trying n={n_try} ...")
        init_long = interpolate_to_length(best_seq, n_try)
        # Also try a fresh one-sided init at this size
        x = np.arange(n_try, dtype=np.float64)
        ramp = (x + 1.0) ** 1.5
        ramp = ramp / ramp.sum()
        # Combine: interpolate best + ramp
        combined = 0.5 * init_long + 0.5 * ramp
        combined = combined / combined.sum()

        a_norm = combined * np.sqrt(2.0 * n_try) / combined.sum()
        # Quick LSE optimization
        a_out, _ = nesterov_lse_optimize(a_norm, min(remaining * 0.3, 120.0), verbose=False)
        do_try_update(a_out, f"size_n{n_try}")

        # Also try the ramp directly
        a_norm2 = ramp * np.sqrt(2.0 * n_try) / ramp.sum()
        a_out2, _ = nesterov_lse_optimize(a_norm2, min(remaining * 0.2, 80.0), verbose=False)
        do_try_update(a_out2, f"ramp_n{n_try}")

    # === Phase 7: Final FP-LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final FP-LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.3, 300.0)
    if t7 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = fessler_parks_optimize(a_norm, t7, verbose=True, max_iters=200)
        do_try_update(a_out, f"fp_final_n{n_cur}")

    # === Phase 8: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.3, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t8, seed=12345, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 9: Final LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final LSE n={n_cur} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.4, 200.0)
    if t9 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_optimize(a_norm, t9, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback: linear ramp
        n_fb = 1600
        x = np.arange(n_fb, dtype=np.float64)
        fallback = (x + 1.0) / (x + 1.0).sum()
        print(f"Fallback ramp score: {evaluate_safe(fallback):.6f}")
        return [float(v) for v in fallback]

    return [float(x) for x in best_seq]
