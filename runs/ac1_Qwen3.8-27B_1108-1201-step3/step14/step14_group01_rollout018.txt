            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            # sum_k w[k] * a[k-i] = cross-correlation
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            
            # sum_k w[k] * a[i-k] = convolution of a and w
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            
            grad = 2.0 * (g1 + g2)
            
            # Adaptive step size with backtracking
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n))
            for _ in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_max:
                best_max = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best_max={best_max:.8f} (t={time.monotonic()-t0:.0f}s)")
    
    return best_a, best_max


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive slack."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    
    no_improve = 0
    patience = 80
    
    for t in range(1, 500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive slack: starts at 0.15, decreases to 0
        progress = min(1.0, t / 500.0)
        slack = 0.15 * (1.0 - progress)
        
        # Upper bound to encourage uniformity
        S_y = y.sum()
        U = 3.0 * M / S_y
        
        # Constraint screening: adaptive
        if t < 50:
            screen_frac = 0.7
        elif t < 200:
            screen_frac = 0.8
        else:
            screen_frac = 0.85
        
        b = lp_solve(y, M, solver, slack=slack, upper_bound=U, screen_frac=screen_frac)
        if b is None or b.sum() < 1e-15:
            no_improve += 1
            if no_improve > patience:
                # Restart with perturbation
                perturb = 0.05 * np.abs(y) + 0.01
                y = y + perturb * np.random.standard_normal(n)
                y = np.maximum(y, 0.0)
                y = y * norm / y.sum()
                no_improve = 0
            continue
        
        b_sum = b.sum()
        g = b * norm / b_sum
        
        # Verify constraint satisfaction (check if screening was too aggressive)
        if screen_frac < 0.85:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * (1 + slack + 0.01):
                # Re-solve with tighter screening
                b2 = lp_solve(y, M, solver, slack=slack, upper_bound=U, screen_frac=0.95)
                if b2 is not None and b2.sum() > 1e-15:
                    g = b2 * norm / b2.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()
            no_improve = 0
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()
            no_improve = 0
        
        if no_improve > patience:
            # Perturb and restart
            n_pert = max(10, n // 20)
            idx = np.random.choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.1 * np.random.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            y = y * norm / y.sum()
            a = y.copy()
            no_improve = 0
        
        if verbose and t % 50 == 0:
            score = 2.0 * n * best_max / norm**2
            print(f"    LP t={t} ({elapsed:.0f}s): best_max={best_max:.8f}, score={score:.6f}")
    
    return best_a, best_max
