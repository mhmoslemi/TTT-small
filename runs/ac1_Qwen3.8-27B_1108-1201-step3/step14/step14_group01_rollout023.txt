def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def normalize_sum(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a) * target_sum

def build_toeplitz_rows(a, indices):
    """Build rows of the cross-convolution matrix for given indices."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros((len(indices), n))
    for idx_i, k in enumerate(indices):
        # conv(y,b)[k] = sum_i y[i] * b[k-i] where 0<=i<n, 0<=k-i<n
        # So coefficient of b[j] is y[k-j] where j = k-i, i.e., b[j] gets y[k-j]
        # Valid j: max(0, k-(n-1)) <= j <= min(k, n-1)
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            i_vals = k - np.arange(j_lo, j_hi + 1)
            rows[idx_i, j_lo:j_hi + 1] = a[i_vals]
    return rows


def lp_best_response(y, M, solver, n_constraints=200):
    """Solve LP: max sum(b) s.t. conv(y,b) <= M, b >= 0.
    Uses constraint generation for efficiency."""
    n = len(y)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with a subset of constraints (the most "active" ones)
    conv_y = fast_conv_self(y)
    # Pick constraints near the peak
    sorted_idx = np.argsort(-conv_y)
    active = sorted_idx[:n_constraints]
    
    for gen_round in range(3):
        T = build_toeplitz_rows(y, active)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        
        if gen_round < 2:
            # Check for violated constraints
            conv_yb = fast_conv_cross(y, b_val)
            violated = np.where(conv_yb > M * 1.001)[0]
            if len(violated) == 0:
                break
            # Add violated constraints
            new_active = np.unique(np.concatenate([active, violated]))
            if len(new_active) > size:
                new_active = sorted_idx[:min(size, 2 * n_constraints)]
            active = new_active
    
    return b_val


def lse_smoothed_objective_and_grad(a, T_val):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    conv = fast_conv_self(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T_val
    shifted = np.maximum(shifted, -100.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    
    # Gradient: d/d a[i] of sum_k w[k] * conv[k]
    # conv[k] = sum_j a[j] * a[k-j]
    # d conv[k] / d a[i] = a[k-i] + a[i-k] (with bounds)
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    # First term: sum_k w[k] * a[k-i]
    # = cross-correlation: (w wrapped) * a
    # Using FFT: ifft(FFT(w) * conj(FFT(a))) gives circular cross-correlation
    # corr[m] = sum_k w[k+m] * a[k] (circular)
    # We want: for each i, sum_k w[k] * a[k-i]
    # Let j = k-i: sum_j w[j+i] * a[j] = corr_w_a[i] where corr_w_a[i] = sum_j w[j+i]*a[j]
    
    ca = irfft(np.conj(A) * W, fft_size)[:n]
    # ca[i] = sum_j a[j] * w[j+i] (circular, but fft_size > 2n so linear for i < n)
    
    # Second term: sum_k w[k] * a[i-k]
    # Let j = i-k: sum_j w[i-j] * a[j] = conv(w, a_flipped)[i]
    # = sum_j a[j] * w[i-j]
    # This is the linear convolution of a and w at position i
    aw = irfft(A * W, fft_size)[:n]
    # aw[i] = sum_j a[j] * w[i-j] (circular, linear for i < n when fft_size >= 2n)
    
    grad = 2.0 * ca + 2.0 * aw
    obj_val = max_c + T_val * np.log(exp_c.sum())
    return obj_val, grad, conv, max_c


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive temperature and step size."""
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = normalize_sum(a0, norm)
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Adaptive temperature schedule
    n_stages = 12
    temps = np.geomspace(80.0, 0.5, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, int(remaining * 8))
        
        # Initial step size
        step = norm / (T_val * math.sqrt(n))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj_val, grad, conv, max_c = lse_smoothed_objective_and_grad(a, T_val)
            grad_norm = np.linalg.norm(grad)
            
            if grad_norm < 1e-15:
                break
            
            # Backtracking line search
            a_trial = a
            step_trial = step
            accepted = False
            for bt in range(15):
                a_trial = a - step_trial * grad
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    step_trial *= 0.5
                    continue
                a_trial = a_trial * norm / s
                new_max = fast_conv_self(a_trial).max()
                # Accept if the actual max decreased (not just the smoothed one)
                if new_max <= max_c + 0.05 * T_val / n:
                    accepted = True
                    break
                step_trial *= 0.4
            
            if accepted:
                a = a_trial
                step = step_trial * 1.1  # Slightly increase step if accepted
                new_max = fast_conv_self(a).max()
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
            else:
                step *= 0.5
                if step < 1e-10:
                    break
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si} T={T_val:.2f} ({elapsed:.0f}s): best_max={best_max:.8f}")
    
    return best_a, best_max

                # Verify improvement
                new_conv = fast_conv_self(a)
                new_max = new_conv.max()
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_a = a.copy()
                    pass_improved = True
                    conv = new_conv
                    cur_max = new_max
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = normalize_sum(a, norm)
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
            if no_improve_passes == 2:
                # Perturb
                n_pert = max(5, n // 20)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.05 * rng.standard_normal(n_pert))
                a = normalize_sum(a, norm)
                best_max = fast_conv_self(a).max()
        else:
            no_improve_passes = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass ({elapsed:.0f}s): best_max={best_max:.8f}")
    
    return best_a, best_max


def make_cosine_init(n, freq=1.0, rng=None):
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.5 * np.cos(np.pi * freq * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian_init(n, n_peaks=3, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    positions = np.linspace(0.1, 0.9, n_peaks) + rng.uniform(-0.05, 0.05, n_peaks)
    positions = np.sort(positions)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.1
    for p in positions:
        width = 0.15 / n_peaks
        a += np.exp(-((x - p) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_msequence_init(n, rng=None):
    """Create initialization inspired by m-sequence autocorrelation properties."""
    if rng is None:
        rng = np.random.default_rng(42)
    # Find largest m-sequence length <= n
    m = 1
    while (2 ** (m + 1) - 1) <= n:
        m += 1
    p = 2 ** m - 1
    
    # Generate m-sequence using LFSR
    # Use a primitive polynomial of degree m
    # For small m, use known primitive polynomials
    primitive_polys = {
        1: 0b11, 2: 0b111, 3: 0b11011, 4: 0b110111, 5: 0b11010111,
        6: 0b111000111, 7: 0b1100001000111, 8: 0b11000101000111,
        9: 0b11000100110000111, 10: 0b110000010011000111,
        11: 0b11001000010001000011, 12: 0b1100110000011001000111,
        13: 0b11001100101010100000011, 14: 0b110000000000110010100011,
        15: 0b110010101000101110000100001,
    }
    
    if m in primitive_polys:
        poly = primitive_polys[m]
        # Generate m-sequence
        state = 1
        seq = np.zeros(p, dtype=np.float64)
        for i in range(p):
            seq[i] = 1.0 - state  # Map to {0, 1}
            # LFSR step
            bit = state & 1
            # Feedback: XOR of bits specified by polynomial
            feedback = 0
            for bit_pos in range(1, m + 1):
                if (poly >> bit_pos) & 1:
                    feedback ^= (state >> (bit_pos - 1)) & 1
            state = (state >> 1) | (feedback << (m - 1))
    else:
        # Fallback: use a smooth random sequence
        x = np.linspace(0, 1, n)
        seq_base = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 5 * x + 0.5)
        seq = np.maximum(seq_base, 0.0)
    
    # Reshape to length n
    if p >= n:
        a = seq[:n]
    else:
        # Tile and truncate
        tiles = int(np.ceil(n / p))
        a = np.tile(seq, tiles)[:n]
    
    # Smooth it a bit
    if n > 10:
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main entry point: search for best sequence within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = make_cosine_init(n_base, freq=1.0)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)
    
    # === Phase 1: Quick LSE GD to smooth out (60s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 1: LSE GD (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")
    
    # === Phase 2: LP-Nesterov main optimization (largest chunk) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP-Nesterov (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")
    
    # === Phase 3: LSE GD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 3: LSE polish (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t3, verbose=True)
        try_update(a_out, "lse2")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Coordinate Descent (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = fast_coordinate_descent(best_seq, t4, rng, verbose=True)
        try_update(a_out, "cd1")
    
    # === Phase 5: Second LP-Nesterov from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.30, 300.0)
    print(f"\n=== Phase 5: LP-Nesterov round 2 (budget={t5:.0f}s) ===")
    if t5 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")
    
    # === Phase 6: Try alternative initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        t6 = min(remaining * 0.4, 200.0)
        print(f"\n=== Phase 6: Alt init + LP (budget={t6:.0f}s) ===")
        
        # Try m-sequence inspired init
        alt_init = make_msequence_init(n_base, rng)
        # Blend with current best
        blend = 0.3 * alt_init + 0.7 * best_seq
        blend = normalize_sum(blend, 1.0)
        
        a_out, _ = nesterov_lp_optimize(blend, t6, solver, verbose=False)
        try_update(a_out, "lp_alt")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t7 = min(remaining * 0.5, 80.0)
        print(f"\n=== Phase 7: Final LSE+CD (budget={remaining:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, min(t7, 40.0), verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = fast_coordinate_descent(best_seq, remaining * 0.6, rng, verbose=False)
            try_update(a_out, "cd_final")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
