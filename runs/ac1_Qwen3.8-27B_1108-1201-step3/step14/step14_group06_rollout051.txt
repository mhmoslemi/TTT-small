    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None

        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:

    b_val = np.maximum(b_val, 0.0)
    return b_val

def smooth_vector(a, sigma=0.5):
    """Apply a simple Gaussian-like smoothing."""
    n = len(a)
    # Simple box filter as approximation
    kernel_size = max(3, int(2 * sigma * 3) | 1)  # odd
    kernel = np.ones(kernel_size) / kernel_size
    a_smooth = np.convolve(a, kernel, mode='same')
    return np.maximum(a_smooth, 0.0)

def nesterov_lp_optimize_v2(a0, time_limit, solver, verbose=True):
    """
    Improved LP Nesterov with:
    - Adaptive epsilon schedule
    - Periodic smoothing
    - Multi-peak constraints (top-k)
    - Better stall detection
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    smooth_every = 20  # smooth every 20 iterations
    smooth_amount = 0.3

    # Precompute T (it changes as y changes, so we rebuild periodically)
    T = build_constraint_matrix(y, n)
    T_rebuild_interval = 10

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive epsilon: start at 0.05, decrease to 0.002
        progress = min(1.0, t / 500.0)
        eps = 0.05 * (1.0 - progress) + 0.002 * progress
        
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        M_used = M_curr * (1.0 - eps)

        # Periodically rebuild T
        if t % T_rebuild_interval == 0:
            T = build_constraint_matrix(y, n)

        # Use top-k constraints for speed (top 5 peaks)
        topk = np.argsort(b_y)[::-1][:5]
        
        b = lp_best_response_topk(y, M_used, topk, solver)
        if b is None:
            b = lp_best_response_topk(y, M_curr, topk, solver)
            if b is None:
                b = lp_best_response_full(y, M_curr, solver)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        # Periodic smoothing
        if t % smooth_every == 0 and progress < 0.5:
            y_new = (1.0 - smooth_amount) * y_new + smooth_amount * smooth_vector(y_new, sigma=1.0)
            s_y = y_new.sum()
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Check for improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 10:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest2 iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}")

    return best_a, best_score


def gradient_descent_lse(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=4):
    """LSE-smoothed gradient descent with L-BFGS-B."""
    from scipy.optimize import minimize as scipy_minimize
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute FFT sizes
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    def lse_obj_and_grad(a_flat, T_val):
        a_local = np.maximum(a_flat, 0.0)
        s = a_local.sum()
        if s < 1e-10:
            return 1e10, np.zeros_like(a_local)
        b = fast_autocorr(a_local)
        max_b = b.max()
        if max_b < 1e-15:
            return 1e10, np.zeros_like(a_local)
        
        shifted = T_val * (b - max_b)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        log_sum = max_b + np.log(exp_vals.sum()) / T_val
        w = exp_vals / exp_vals.sum()

        # Gradient of log-sum-exp w.r.t. a[i]
        # d/d a[i] log(sum_k exp(T*(b[k]-max_b))) / T
        # = sum_k w[k] * T * db[k]/da[i] / T = sum_k w[k] * db[k]/da[i]
        # db[k]/da[i] = a[k-i] if 0 <= k-i < n
        # Use FFT: this is the cross-correlation of w and a
        w_full = np.zeros(fft_size)
        w_full[:size] = w
        a_full = np.zeros(fft_size)
        a_full[:n] = a_local
        
        W_fft = rfft(w_full, fft_size)
        A_fft = rfft(a_full, fft_size)
        # Cross-correlation: sum_k w[k] * a[k-i] 
        # = ifft(conj(W) * A) ... need to be careful
        # Actually: (w * a_conv)[i] = sum_k w[k] a[k-i]
        # In FFT: ifft(FFT(w_full) * conj(FFT(a_full))) ... hmm
        # Let's think: corr_w_a[i] = sum_k w[k] * a[k-i]
        # This is the cross-correlation of w and a at lag i
        # = ifft(conj(FFT(w)) * FFT(a)) ... no
        # Cross-correlation: (w ⋆ a)[i] = sum_k w[k+i] * a[k] = ifft(conj(FFT(w)) * FFT(a))
        # But we want sum_k w[k] * a[k-i] = sum_k w[k] * a[k-i]
        # Let j = k-i: sum_j w[j+i] * a[j] = (w ⋆ a)[-i]... 
        # Actually, let's just use the definition directly with FFT
        # sum_k w[k] * a[k-i] for k=0..2n-2
        # = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
        # This is a convolution-like operation
        # (w * a_rev)[i + n - 1] where a_rev is reversed... 
        # Let me just compute it directly for the gradient
        # Actually, the gradient of b[k] w.r.t. a[i] is a[k-i] (if 0<=k-i<n)
        # So grad[i] = sum_k w[k] * a[k-i] (for k where 0<=k-i<n)
        # = sum_{k=i}^{i+n-1} w[k] * a[k-i] (clipped to valid range)
        # This is the cross-correlation of w and a
        
        # Using FFT: cross_corr[i] = sum_k w[k] a[k-i]
        # Let's pad and use: ifft(FFT(w_padded) * conj(FFT(a_padded)))
        # Actually the standard cross-correlation formula:
        # corr[i] = ifft(conj(FFT(w)) * FFT(a))[i] gives sum_k w[k] a[k+i]
        # We want sum_k w[k] a[k-i] = corr[-i]
        
        W_fft = rfft(w_full, fft_size)
        A_fft = rfft(a_full, fft_size)
        corr = irfft(np.conj(W_fft) * A_fft, fft_size)
        # corr[i] = sum_k w_full[k] * a_full[k-i] (circular)
        # For our purposes with proper padding, this should work
        grad_b = corr[:n]  # This gives sum_k w[k] * a[k-i] for i=0..n-1
        
        obj_val = 2.0 * n * log_sum / (s * s)
        grad = 2.0 * n * grad_b / (s * s) - 2.0 * obj_val * a_local / s
        return obj_val, grad

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)

        def obj_func(x, T_val=T):
            val, _ = lse_obj_and_grad(x, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_obj_and_grad(x, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 800,
                                           'gtol': 1e-8, 'ftol': 1e-10,
                                           'maxls': 40, 'maxfun': 800})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score


def coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Find elements contributing most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find low-value elements to receive mass
        non_zero = np.where(a > 1e-10)[0]
        low_val = non_zero[np.argsort(a[non_zero])][:20]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random moves
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    """Try to update best state with candidate."""
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()

def build_full_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def build_topk_constraint_matrix(a, n, topk_indices):
    T = np.zeros((len(topk_indices), n))
    for row, k in enumerate(topk_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            T[row, i] = a[k - i]
    return T


def lp_best_response_full(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_full_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_topk(a, M, topk_indices, solver):

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive epsilon: start wider, get tighter
        progress = min(1.0, float(t) / 300.0)
        eps = 0.04 * (1.0 - progress) + 0.003 * progress
        M_used = M_curr * (1.0 - eps)

        # Use full constraints for first 20 iterations, then top-k
        if t <= 20:
            T = build_full_constraint_matrix(y, n)
            b = lp_best_response_full(y, M_used, solver, T)
            if b is None:
                b = lp_best_response_full(y, M_curr, solver, T)
            if b is None:
                break
        else:
            topk = np.argsort(b_y)[::-1][:7]
            b = lp_best_response_topk(y, M_used, topk, solver)
            if b is None:
                b = lp_best_response_topk(y, M_curr, topk, solver)
            if b is None:
                T_full = build_full_constraint_matrix(y, n)
                b = lp_best_response_full(y, M_curr, solver, T_full)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov step with adaptive alpha
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.03), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        # Light smoothing in early iterations to avoid spikiness
        if t <= 100 and t % 10 == 0:
            y_new = 0.85 * y_new + 0.15 * smooth_vector(y_new, 5)
            s_y = y_new.sum()
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score - 1e-10:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 50.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    Stall at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_optimize(a0, time_limit, T_start=60.0, T_end=2.0, n_stages=5):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    def compute_obj_grad(a_flat, T_val):
        a_local = np.maximum(a_flat, 0.0)
        s = a_local.sum()
        if s < 1e-10:
            return 1e10, np.zeros(n)
        b = fast_autocorr(a_local)
        max_b = b.max()
        if max_b < 1e-15:
            return 1e10, np.zeros(n)
        shifted = np.clip(T_val * (b - max_b), -400, 400)
        exp_vals = np.exp(shifted)
        exp_sum = exp_vals.sum()
        if exp_sum < 1e-300:
            return 1e10, np.zeros(n)
        log_sum = max_b + np.log(exp_sum) / T_val
        w = exp_vals / exp_sum

        # Gradient: sum_k w[k] * a[k-i] for valid k,i
        w_full = np.zeros(fft_size)
        w_full[:size] = w
        a_full = np.zeros(fft_size)
        a_full[:n] = a_local
        W_fft = rfft(w_full, fft_size)
        A_fft = rfft(a_full, fft_size)
        cross = irfft(np.conj(W_fft) * A_fft, fft_size)[:n]

        obj_val = 2.0 * n * log_sum / (s * s)
        grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a_local / s
        return obj_val, grad

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def obj_func(x, Tv=T):
            val, _ = compute_obj_grad(x, Tv)
            return val

        def grad_func(x, Tv=T):
            _, g = compute_obj_grad(x, Tv)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 250, 'maxfun': 1000,
                                           'gtol': 1e-8, 'ftol': 1e-10,
                                           'maxls': 50})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score - 1e-10:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score


def coord_descent(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        non_zero = np.where(a > 1e-10)[0]
        if len(non_zero) == 0:
            break
        low_val = non_zero[np.argsort(a[non_zero])][:20]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            a_norm2 = best_seq * np.sqrt(2.0 * n_c2) / best_seq.sum()
            a_out, _ = lse_optimize(a_norm2, min(remaining2, third), T_start=30.0, T_end=1.0, n_stages=3)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "lse_final", t_global)

        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            n_c3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_c3) / best_seq.sum()
            a_out, _ = coord_descent(a_norm3, remaining3, seed=77777)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "cd_final2", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
