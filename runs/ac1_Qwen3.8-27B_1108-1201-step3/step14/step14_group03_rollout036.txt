def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Compute cross-convolution a * b using FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize_seq(a, target_sum=None):

def compute_grad(a, w):

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_seq(a0, target_sum)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    stuck = 0
    size = 2 * n - 1

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Build constraint matrix
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            stuck += 1
            if stuck > 30:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stuck += 1
            if stuck > 30:
                break
            continue

        stuck = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue

        g = b_val * target_sum / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * target_sum / s_y

        if t == 1:
            a_new = g.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * target_sum / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * target_sum / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 50 == 0:
            print(f"    LP iter={t}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    if verbose:
        print(f"  LP Nesterov done: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def pair_cd(a0, time_limit, rng, window=25, verbose=True):
    """Pair coordinate descent with local window."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_seq(a0, target_sum)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < target_sum * 1e-10:
                continue

            for offset in range(1, min(window + 1, n)):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[j] < target_sum * 1e-12:
                        continue

                    # Try moving mass from i to j
                    conv = fast_autocorr(a)
                    cur_max = conv.max()

                    # Compute delta for moving d from i to j
                    # conv changes: -d*a[k-i] - d*a[i-k] + d*a[k-j] + d*a[j-k] + diagonal terms
                    best_d = 0.0
                    best_val = cur_max
                    n_samples = 12
                    d_max = a[i] * 0.95
                    for d in np.linspace(0, d_max, n_samples):
                        if d < 1e-10:
                            continue
                        a[i] -= d
                        a[j] += d
                        s = a.sum()
                        if s > 1e-15:
                            a_scaled = a * target_sum / s
                            new_conv = fast_autocorr(a_scaled)
                            new_max = new_conv.max()
                            if new_max < best_val - 1e-12:
                                best_val = new_max
                                best_d = d
                        a[i] += d
                        a[j] -= d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a * target_sum / a.sum()
                        conv = fast_autocorr(a)
                        cur = conv.max()
                        if cur < best_score - 1e-12:
                            best_score = cur
                            best_a = a.copy()
                            pass_improved = True

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

        if verbose:
            print(f"    PCD: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    if verbose:
        print(f"  PCD done: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def perturbation_search(a_best, time_limit, rng, verbose=True):
    """Try random perturbations and re-optimize briefly."""
    n = len(a_best)
    target_sum = np.sqrt(2.0 * n)
    best_a = a_best.copy()
    best_score = fast_autocorr(a_best).max()
    t0 = time.monotonic()
    mean_val = target_sum / n

    # Try different perturbation scales
    scales = [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    n_tries_per_scale = 3
    total_tries = len(scales) * n_tries_per_scale
    time_per_try = max(0.5, time_limit / total_tries)

    for si, scale in enumerate(scales):
        for trial in range(n_tries_per_scale):
            if time.monotonic() - t0 > time_limit:
                break
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 1:
                break

            # Generate perturbation
            noise = rng.normal(0, scale * mean_val, n)
            a_pert = a_best + noise
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * target_sum / a_pert.sum()

            # Quick optimization: short SGD
            t_budget = min(time_per_try, remaining * 0.4)
            a_out, sc = softmax_sgd(a_pert, t_budget, verbose=False,
                                    T_start=1.0, T_end=0.1, n_stages=5)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_out.copy()
                if verbose:
                    print(f"    Pert scale={scale}: NEW BEST {best_score:.8f}")

    if verbose:
        print(f"  Perturbation done: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def fft_flatten(a0, time_limit, verbose=True):
    """FFT amplitude flattening (Gerchberg-Saxton variant)."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_seq(a0, target_sum)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for alpha in [0.3, 0.5, 0.7, 0.9]:
        if time.monotonic() - t0 > time_limit:
            break
        n_iters = max(50, int((time_limit - (time.monotonic() - t0)) * 8))

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            A = rfft(a, fft_size)
            power = np.abs(A) ** 2
            target_power = np.mean(power)
            target_amp = np.sqrt(target_power) + 1e-10

            amp = np.abs(A)
            phase = np.angle(A)
            new_amp = alpha * target_amp + (1 - alpha) * amp
            A_new = new_amp * np.exp(1j * phase)
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new * target_sum / s

            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new

        if verbose:
            print(f"    FFT-alpha={alpha}: best={best_score:.8f}")

    if verbose:
        print(f"  FFT-flatten done: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Softmax SGD (30% time)
    Phase 2: Active-set SGD (15% time)
    Phase 3: LP Nesterov (20% time)
    Phase 4: Perturbation search (25% time)
    Phase 5: Final polish (10% time)
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        parent = None

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
        parent = a_init

    n_base = len(best_seq)
    target_sum = np.sqrt(2.0 * n_base)
    a_norm = parent * target_sum / parent.sum()

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Softmax SGD (30%)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Softmax SGD n={n_base} ({remaining*0.30:.0f}s) ===")
    t1 = remaining * 0.30
    if t1 > 15:
        a_out, _ = softmax_sgd(a_norm, t1, verbose=True)
        try_update(a_out, "sgd1")

    # Phase 2: Active-set SGD (12%)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Active-Set SGD n={n_base} ({remaining*0.12:.0f}s) ===")
    t2 = remaining * 0.12
    if t2 > 10:
        a_out, _ = active_set_sgd(a_norm, t2, verbose=True)
        try_update(a_out, "active1")

    # Phase 3: LP Nesterov (20%)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n_base} ({remaining*0.20:.0f}s) ===")
    t3 = remaining * 0.20
    if t3 > 20:
        solver = _get_solver()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 4: Perturbation search (28%)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Perturbation Search ({remaining*0.28:.0f}s) ===")
    t4 = remaining * 0.28
    if t4 > 20:
        a_out, _ = perturbation_search(best_seq, t4, rng, verbose=True)
        try_update(a_out, "perturb")

    # Phase 5: Second round of refinement on best found
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Refinement n={n_best} ({remaining*0.20:.0f}s) ===")
    t5 = remaining * 0.20
    if t5 > 15:
        a_norm2 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = softmax_sgd(a_norm2, t5, verbose=True, T_start=1.5, T_end=0.05, n_stages=10)
        try_update(a_out, "sgd2")

    # Phase 6: LP on best found
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP Polish n={n_best} ({remaining*0.15:.0f}s) ===")
    t6 = remaining * 0.15
    if t6 > 15:
        solver = _get_solver()
        a_norm3 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm3, t6, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 7: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Pair CD n={n_best} ({remaining*0.15:.0f}s) ===")
    t7 = remaining * 0.15
    if t7 > 10:
        a_norm4 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd(a_norm4, t7, rng, window=20, verbose=True)
        try_update(a_out, "pcd")

    # Phase 8: Final perturbation round
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Perturbations ({remaining*0.30:.0f}s) ===")
    t8 = remaining * 0.30
    if t8 > 15:
        a_out, _ = perturbation_search(best_seq, t8, rng, verbose=True)
        try_update(a_out, "perturb2")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
