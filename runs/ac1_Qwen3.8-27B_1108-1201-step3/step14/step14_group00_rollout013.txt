"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive active-set screening
2. Subgradient polishing at the active peaks
3. Exact coordinate descent for final refinement
4. Symmetrization and spectral shaping ideas
Key improvements over previous: tighter active-set management, dual-guided
constraint selection, subgradient fine-tuning, and better time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_constraint_matrix(a, k_indices, n):
    """
    Build the constraint matrix for (a * b)[k] <= M.
    For fixed a, (a*b)[k] = sum_j a[k-j] * b[j] where 0 <= k-j < n.
    So the row for lag k has entries: row[j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat


def lp_best_response(a_fixed, M, active_set, solver):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0.
    Returns (b, duals) or (None, None).
    """
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def verify_and_expand(a_fixed, b, M, active_set, solver, n):
    """Check for constraint violations and expand active set if needed."""
    size = 2 * n - 1
    all_k = np.arange(size)
    complement = np.setdiff1d(all_k, active_set)
    if len(complement) == 0:
        return active_set, b
    # Check violations on complement
    T_comp = build_constraint_matrix(a_fixed, complement, n)
    violations = T_comp @ b
    viol_idx = np.where(violations > M * (1 + 1e-6))[0]
    if len(viol_idx) > 0:
        new_active = np.union1d(active_set, complement[viol_idx])
        if len(new_active) > 0.9 * size:
            new_active = all_k
        b_new, _ = lp_best_response(a_fixed, M, new_active, solver)
        if b_new is not None:
            return new_active, b_new
    return active_set, b


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response iteration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 40
    n_iters = 0

    # Ergodic average for stability
    erg_a = np.zeros(n)
    erg_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set selection
        if active_set is None or t % 50 == 0:
            # Use a tolerance that tightens over time
            tol = max(0.01, 0.05 * np.exp(-t / 300.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            # Ensure we always include the top few peaks
            top_k = np.argsort(conv_y)[-max(3, n // 200):]
            active_set = np.unique(np.concatenate([active_set, top_k]))

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            if t % 10 == 0:
                active_set = np.arange(2 * n - 1)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        # Expand active set if needed
        active_set, b = verify_and_expand(y, b, M_curr, active_set, solver, n)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.98)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        erg_a += a
        erg_count += 1
        if erg_count % 20 == 0:
            erg_check = erg_a / erg_count
            s_erg = erg_check.sum()
            if s_erg > 1e-15:
                erg_check = erg_check * norm / s_erg
                score_erg = fast_autocorr(erg_check).max()
                if score_erg < best_score:
                    best_score = score_erg
                    best_a = erg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Subgradient method polishing: minimize max_k (a*a)[k] subject to sum(a)=S, a>=0.
    Uses the subgradient at the active peak(s).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive step size
    step = 1.0 / (n ** 0.5)
    patience = 0
    max_patience = 500

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        k_star = int(np.argmax(conv))

        # Compute subgradient at k_star
        # d/d a[j] of (a*a)[k_star] = a[k_star - j] + a[j - k_star] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            j1 = k_star - j
            if 0 <= j1 < n:
                v += a[j1]
            j2 = j - k_star
            if 0 <= j2 < n:
                v += a[j2]
            grad[j] = v

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Projected gradient step
        # We want to decrease M, so move in direction -grad
        # But we need to maintain sum(a) = norm and a >= 0
        # Projected step: a_new = proj_simplex(a - step * grad)
        a_trial = a - step * grad
        # Project onto {x >= 0, sum(x) = norm}
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            # Slowly increase step
            step = min(step * 1.01, 5.0)
        else:
            patience += 1
            step *= 0.95
            if step < 1e-10:
                break

        # Try a few random k_star for diversity
        if t % 50 == 0:
            conv = fast_autocorr(a)
            # Pick a random high-value lag
            threshold = conv.max() * 0.95
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 1:
                k_rand = candidates[np.random.randint(len(candidates))]
                grad_r = np.zeros(n)
                for j in range(n):
                    v = 0.0
                    j1 = k_rand - j
                    if 0 <= j1 < n:
                        v += a[j1]
                    j2 = j - k_rand
                    if 0 <= j2 < n:
                        v += a[j2]
                    grad_r[j] = v
                grad_r_norm = np.linalg.norm(grad_r)
                if grad_r_norm > 1e-15:
                    grad_r = grad_r / grad_r_norm
                    a_trial2 = a - step * 0.5 * grad_r
                    a_trial2 = np.maximum(a_trial2, 0.0)
                    s2 = a_trial2.sum()
                    if s2 > 1e-15:
                        a_trial2 = a_trial2 * norm / s2
                        score2 = fast_autocorr(a_trial2).max()
                        if score2 < best_score - 1e-12:
                            a = a_trial2
                            best_score = score2
                            best_a = a_trial2.copy()
                            patience = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=False):
    """
    Systematic coordinate descent with exact 1D line search.
    For each coordinate j, find optimal a[j] given all others fixed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]

            # Compute the "background" autocorrelation (contributions not involving a[j])
            # (a*a)[k] = bg[k] + 2*a[j]*contrib[k] + a[j]^2 * delta
            # where contrib[k] = a[k-j] (if valid) + a[j-k] (if valid) [excluding a[j] itself]
            # Actually: (a*a)[k] includes terms a[i]*a[k-i]. Terms involving a[j]:
            #   a[j]*a[k-j] (when k-j is valid, i.e., 0 <= k-j < n) -> gives a[k-j]
            #   a[i]*a[j] (when i = k-j, same as above)
            #   a[j]*a[j] (when j = k-j, i.e., k = 2j) -> gives a[j]^2
            # So: (a*a)[k] = bg[k] + 2*a[j]*c[k] + (a[j]^2 if k==2j else 0)
            # where c[k] = a[k-j] (if 0<=k-j<n and k-j != j) + a[j-k] (if 0<=j-k<n and j-k != j)

            # Precompute c[k] for all k
            c = np.zeros(size)
            for k in range(size):
                val = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    val += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    val += a[i2]
                c[k] = val

            # bg[k] = (a*a)[k] - 2*aj*c[k] - (aj^2 if k==2j else 0)
            conv = fast_autocorr(a)
            bg = conv - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Now minimize max_k (bg[k] + 2*x*c[k] + (x^2 if k==2j else 0)) over x in [0, norm]
            # Subject to: sum(a) - aj + x = norm, so x = norm - (sum(a) - aj)
            # Wait, we're changing a[j] from aj to x, so new sum = sum(a) - aj + x
            # We need new sum = norm, so x = norm - (sum(a) - aj) = aj (no change in sum!)
            # 
            # Actually, since we normalize after each step, let's think of it differently:
            # We fix all a[i] for i != j, and optimize a[j] = x, then renormalize.
            # After renormalization, the function is homogeneous, so we can just minimize
            # max_k (bg[k] + 2*x*c[k] + x^2*delta_{k,2j}) / (sum others + x)^2 * (norm^2)
            # But this is complicated. Let's just do a simple grid search.

            # Simple approach: try a few values of x
            s_others = a.sum() - aj
            candidates_x = [0.0, aj * 0.5, aj * 0.75, aj * 1.0, aj * 1.25, aj * 1.5,
                           s_others * 0.1, s_others * 0.5, s_others]
            candidates_x = [x for x in candidates_x if 0 <= x <= norm]

            best_x = aj
            best_obj = float('inf')

            for x in candidates_x:
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # Compute new autocorrelation
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                obj = new_conv.max() / (new_sum * new_sum) * (2 * n)
                # Actually the score is 2*n*max_conv/sum^2, and we want to minimize this
                # But since we'll renormalize, the score is 2*n*max_conv_at_norm
                # Let's just compute: after setting a[j]=x and normalizing to sum=norm:
                # a_scaled = a * (norm / new_sum)
                # conv_scaled = conv * (norm/new_sum)^2
                # score = 2*n*max(conv_scaled)/norm^2 = 2*n*max_conv/(new_sum^2)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            # Also try bisection between best candidates
            if best_x != aj and abs(best_x - aj) > 1e-10:
                lo = min(aj, best_x)
                hi = max(aj, best_x)
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    new_sum = s_others + mid
                    if new_sum < 1e-15:
                        break
                    new_conv = bg + 2.0 * mid * c
                    if 2 * j < size:
                        new_conv[2 * j] += mid * mid
                    obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                    aj_score = 2.0 * n * conv.max() / (a.sum() ** 2)
                    if obj < aj_score:
                        lo = mid
                    else:
                        hi = mid
                best_x = (lo + hi) / 2.0

            if best_x != aj and abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def projected_subgradient(a0, time_limit, verbose=False):
    """
    More aggressive subgradient method with multiple active peaks.
    Uses the top few peaks to compute a combined gradient.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passed = 0
    max_patience = 2000
    step = 0.5 / n ** 0.3

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if n_passed > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks
        n_peaks = max(3, min(10, n // 100))
        top_indices = np.argsort(conv)[-n_peaks:]
        # Weight by how close to maximum
        weights = np.exp(-5.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for j in range(n):
                v = 0.0
                j1 = k - j
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = j - k
                if 0 <= j2 < n:
                    v += a[j2]
                grad[j] += w * v

        # Projected step
        a_trial = a - step * grad / (n ** 0.5)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            n_passed = 0
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            step = min(step * 1.02, 2.0)
        else:
            n_passed += 1
            step *= 0.98
            if step < 1e-8:
                break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ProjSubgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def symmetrize(a):
    """Symmetrize the sequence: a = (a + reverse(a)) / 2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def evaluate_symmetrized(a):
    """Check if symmetrization helps."""
    a_s = symmetrize(a)
    s = a_s.sum()
    if s < 1e-15:
        return None
    a_s_norm = a_s * np.sqrt(2.0 * len(a)) / s
    return fast_autocorr(a_s_norm).max()


def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate():
    """
    Main search function. Uses LP best-response with Nesterov acceleration,
    subgradient polishing, and coordinate descent to minimize the score.
    """
    rng = np.random.default_rng(999)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Try symmetrization first (quick check)
    a_sym = symmetrize(best_seq)
    s_sym = a_sym.sum()
    if s_sym > 1e-15:
        a_sym_norm = a_sym * np.sqrt(2.0 * len(a_sym)) / s_sym
        sc_sym = evaluate_safe(a_sym_norm)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym_norm.copy()
            print(f"  *** Symmetrization improved: {sc_sym:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Subgradient polish n={n_cur}, remaining={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 150.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        try_update(a_out, f"subg1_n{n_cur}")

    # === Phase 3: Projected subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Projected subgradient n={n_cur}, remaining={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 130.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = projected_subgradient(a_norm, t3, verbose=True)
        try_update(a_out, f"projsubg_n{n_cur}")

    # === Phase 4: LP Nesterov again (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_cur}, remaining={remaining:.0f}s ===")
    t4 = min(remaining * 0.20, 200.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=123)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent n={n_cur}, remaining={remaining:.0f}s ===")
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t5, rng, verbose=True)
        try_update(a_out, f"cd1_n{n_cur}")

    # === Phase 6: Perturbation + LP restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Perturbation restarts, remaining={remaining:.0f}s ===")
    n_restarts = 2
    t6_total = min(remaining * 0.25, 200.0)
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t6 = min(t6_each, remaining - 30)
        if t6 < 20:
            break
        pert = perturb_sequence(best_seq, rng, scale=0.02 + 0.01 * r)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False, seed=200 + r)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_cur}, remaining={remaining:.0f}s ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t7, rng, verbose=True)
        try_update(a_out, f"cd_final_n{n_cur}")

    # === Phase 8: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final subgrad n={n_cur} ===")
        t8 = min(remaining * 0.4, 60.0)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t8, verbose=False)
        try_update(a_out, f"subg_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
