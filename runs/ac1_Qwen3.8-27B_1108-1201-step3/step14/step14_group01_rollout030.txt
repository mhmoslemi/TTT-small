def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    best_a = a.copy()
    best_max = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(80.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 8)))

        step_base = 1.0 / (T_val * np.sqrt(n))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]

            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = step_base / (1.0 + 0.05 * it)

            for trial in range(12):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new / s_new
                sc = compute_max_conv(a_new)
                if sc <= max_c + 0.005 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new

            sc = compute_max_conv(a)
            if sc < best_max:
                best_max = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_max:.10f}")

    return best_a, best_max

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, patience=5000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s

    best_a = a.copy()
    best_max = compute_max_conv(a)
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    stagnant = 0
    best_at_stagnation = best_max

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        # Adaptive constraint screening
        frac = 0.30 if t < 500 else 0.25
        n_active = max(50, int(size * frac))
        thresh_idx = np.argsort(conv_y)[::-1][:n_active]
        active = thresh_idx

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        # Verify dropped constraints
        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.any(check_conv > M * (1 + 1e-6))
            if violated:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b / b_sum

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = y.sum()
        y_new = y_new / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        sc_y = compute_max_conv(y)
        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()
            stagnant = 0
            best_at_stagnation = best_max
        else:
            stagnant += 1

        sc_a = compute_max_conv(a)
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()
            stagnant = 0
            best_at_stagnation = best_max

        # Reset if stagnant for too long
        if stagnant > patience:
            stagnant = 0
            # Slightly perturb to escape
            rng_local = np.random.default_rng(t)
            n_pert = max(10, n // 20)
            idx_pert = rng_local.choice(n, size=n_pert, replace=False)
            y[idx_pert] *= (1.0 + 0.05 * rng_local.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            a = y.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_max:.10f}")

    return best_a, best_max

def mirror_descent_phase(a0, time_limit, verbose=True):
    """Mirror descent with KL projection for the max-conv problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s

    best_a = a.copy()
    best_max = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    eta_base = 0.1
    stagnant = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < 1e-14:
            break

        # Find the argmax position
        k_star = np.argmax(conv)

        # Gradient of conv(a,a) at position k_star w.r.t. a[i]
        # = a[k_star - i] + a[i - k_star] (for valid indices)
        grad = np.zeros(n)
        # a[k_star - i] valid when 0 <= k_star - i < n => k_star - n + 1 <= i <= k_star
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n, k_star + 1)
        if hi1 > lo1:
            grad[lo1:hi1] += a[k_star - np.arange(lo1, hi1)]
        # a[i - k_star] valid when 0 <= i - k_star < n => k_star <= i <= k_star + n - 1
        lo2 = max(0, k_star)
        hi2 = min(n, k_star + n)
        if hi2 > lo2:
            grad[lo2:hi2] += a[np.arange(lo2, hi2) - k_star]

        grad *= 2.0  # from the quadratic

        # Mirror descent step: a_new[i] proportional to a[i] * exp(-eta * grad[i])
        eta = eta_base / (1.0 + 0.01 * t)
        log_step = -eta * grad
        log_step = np.maximum(log_step, -50.0)  # clip for stability
        log_step = np.minimum(log_step, 50.0)
        log_step -= log_step.max()  # for numerical stability
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        sc = compute_max_conv(a_new)
        if sc < cur_max + 0.01 * eta:
            a = a_new
            if sc < best_max:
                best_max = sc
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            # Reduce step and try again
            a_mid = a * np.exp(0.5 * log_step)
            a_mid = np.maximum(a_mid, 1e-30)
            a_mid = a_mid / a_mid.sum()
            sc_mid = compute_max_conv(a_mid)
            if sc_mid < cur_max + 0.005 * eta:
                a = a_mid
            stagnant += 1

        if stagnant > 2000:
            # Adaptive: reduce eta_base
            eta_base *= 0.5
            stagnant = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): best={best_max:.10f}")

    return best_a, best_max

def pair_cd_optimize(a0, time_limit, rng, verbose=True, max_j_candidates=16):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    best_a = a.copy()
    best_max = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Find peak lags for structured candidates
        peak_threshold = cur_max * 0.90
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            # Build candidate set for j
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_struct = (kl - i) % n
                j_candidates.add(j_struct)
                j_struct2 = (kl + i) % n
                j_candidates.add(j_struct2)
            # Also try neighbors of i
            for dy in [-3, -2, -1, 1, 2, 3]:
                jj = (i + dy) % n
                if jj != i:
                    j_candidates.add(jj)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 40
                deltas = np.linspace(0.01, 0.99, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a / s_new
                    new_max = compute_max_conv(a)
                    if new_max < best_max - 1e-13:
                        best_max = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 4:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_max:.10f}")

    return best_a, best_max

def peak_targeted_refinement(a0, time_limit, rng, verbose=True):
    """Specifically target the peak positions in the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    best_a = a.copy()
    best_max = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Get top 5 peak positions
        top_k = 5
        peak_positions = np.argsort(conv)[::-1][:top_k]

        improved_any = False
        for k_pos in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break

            # The contribution to conv[k_pos] is: sum_m a[m] * a[k_pos - m]
            # To reduce this, we want to transfer mass from large a[m] to small a[j]
            # such that the new contribution at k_pos is reduced.
            # For a transfer from i to j: change in conv[k_pos] =
            #   a[k_pos-j]*delta + a[j-k_pos]*delta - a[k_pos-i]*delta - a[i-k_pos]*delta
            #   + second order: -delta^2 if k_pos = 2i, +delta^2 if k_pos = 2j

            # Find i with largest contribution at k_pos
            contrib = np.zeros(n)
            for m in range(n):
                other = k_pos - m
                if 0 <= other < n:
                    contrib[m] = a[m] * a[other]
            # Also the "reverse" contribution
            for m in range(n):
                other = k_pos - m
                if 0 <= other < n:
                    contrib[m] += a[m] * a[other]  # this is the same, conv is symmetric at center

            # Actually for conv(a,a)[k] = sum_m a[m]*a[k-m], the contribution of a[i] is:
            # a[i]*a[k-i] (if k-i valid) + a[k-i]*a[i] (same thing, counted once in the sum)
            # Wait, conv(a,a)[k] = sum_{m: 0<=m<n, 0<=k-m<n} a[m]*a[k-m]
            # The term involving a[i] appears when m=i (gives a[i]*a[k-i]) and when k-m=i i.e. m=k-i (gives a[k-i]*a[i])
            # So total contribution of a[i] to conv[k] is: a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] (if both valid)
            # Unless i = k-i, in which case it's a[i]^2.

            # Find the i with the largest "transferable" contribution
            # We want to reduce a[i] and increase a[j] where the net effect at k_pos is negative.
            # Delta of conv[k_pos] when transferring delta from i to j:
            # = delta*(a[k_pos-j] + a[j-k_pos]) - delta*(a[k_pos-i] + a[i-k_pos])
            #   + second order terms
            # We want this to be negative, i.e.:
            # a[k_pos-j] + a[j-k_pos] < a[k_pos-i] + a[i-k_pos]

            # So: pick i with large (a[k_pos-i] + a[i-k_pos]) and j with small (a[k_pos-j] + a[j-k_pos])
            coeff = np.zeros(n)
            for m in range(n):
                c = 0.0
                if 0 <= k_pos - m < n:
                    c += a[k_pos - m]
                if 0 <= m - k_pos < n:
                    c += a[m - k_pos]
                coeff[m] = c

            # Sort by coeff: transfer from high coeff to low coeff
            sorted_idx = np.argsort(coeff)
            # Try top few transfers
            n_tried = 0
            for rank_i in range(n - 1, max(n - 20, -1), -1):
                if time.monotonic() - t0 > time_limit:
                    break
                i = sorted_idx[rank_i]
                if a[i] < 1e-12:
                    continue
                for rank_j in range(min(20, n), -1, -1):
                    j = sorted_idx[rank_j]
                    if j == i:
                        continue
                    if rank_j >= rank_i:
                        break

                    ai = a[i]
                    aj = a[j]

                    # Compute the gain
                    g = np.zeros(size)
                    lo = max(0, j)
                    hi = min(size, j + n)
                    if hi > lo:
                        g[lo:hi] += a[0:hi - lo]
                    lo2 = max(0, j - n + 1)
                    hi2 = min(size, j + 1)
                    if hi2 > lo2:
                        g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                    lo3 = max(0, i)
                    hi3 = min(size, i + n)
                    if hi3 > lo3:
                        g[lo3:hi3] -= a[0:hi3 - lo3]
                    lo4 = max(0, i - n + 1)
                    hi4 = min(size, i + 1)
                    if hi4 > lo4:
                        g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                    # Find best delta
                    n_samples = 30
                    deltas = np.linspace(0.01, 0.99, n_samples) * ai
                    best_d = 0.0
                    best_v = cur_max
                    for d in deltas:
                        new_conv = conv + d * g
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                        v = new_conv.max()
                        if v < best_v:
                            best_v = v
                            best_d = d

                    if best_d > 0 and best_v < best_max - 1e-13:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a / a.sum()
                        new_max = compute_max_conv(a)
                        if new_max < best_max - 1e-13:
                            best_max = new_max
                            best_a = a.copy()
                            improved_any = True
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a / a.sum()
                        break

                    n_tried += 1
                    if n_tried > 100:
                        break

        if not improved_any:
            # Perturb
            n_perturb = max(5, n // 50)
            idx_perturb = rng.choice(n, size=n_perturb, replace=False)
            a[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            # If still stuck, break
            break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakRef ({elapsed:.0f}s): best={best_max:.10f}")

    return best_a, best_max

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) ** 2 + 0.1 * np.sin(2 * np.pi * x)
    else:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{len(best_seq)}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 400.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{len(best_seq)}")

    # Phase 3: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 3: Mirror Descent n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = mirror_descent_phase(best_seq, t3, verbose=True)
        try_update(a_out, f"mirror_n{len(best_seq)}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 6: Peak targeted refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 120.0)
    print(f"\n=== Phase 6: Peak Refinement n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = peak_targeted_refinement(best_seq, t6, rng, verbose=True)
        try_update(a_out, f"peak_n{len(best_seq)}")

    # Phase 7: Try longer length n=2000
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_base + 400, 2200)
    t7 = min(remaining * 0.25, 200.0)
    print(f"\n=== Phase 7: Longer n={n_long} (budget={t7:.0f}s) ===")
    if t7 > 40 and remaining > 150:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = lse_gradient_descent(init_long, min(t7 * 0.3, 40.0), verbose=False)
        try_update(a_out, f"lse_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_out, _ = nesterov_lp_optimize(best_seq if len(best_seq) == n_long else init_long,
                                            min(remaining * 0.5, 80.0), solver, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.6, 60.0), rng, verbose=False)
            try_update(a_out, f"pcd_long_{n_long}")

    # Phase 8: Final pair CD on best
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.5, 100.0)
    print(f"\n=== Phase 8: Final CD n={len(best_seq)} (budget={t8:.0f}s) ===")
    if t8 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t8, rng, verbose=True)
        try_update(a_out, "pcd_final")

    # Phase 9: Final quick LSE + LP
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 9: Final polish (budget={remaining:.0f}s) ===")
        t_half = remaining * 0.5
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 50.0), verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, min(remaining * 0.8, 40.0), solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
