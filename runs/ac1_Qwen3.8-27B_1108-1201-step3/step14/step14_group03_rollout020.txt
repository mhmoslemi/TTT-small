def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1:
        # Already in the simplex? No, we need sum = 1.
        # Just normalize
        s = v.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return np.maximum(v, 0) / np.maximum(v, 0).sum()
    
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def smoothed_gradient_step(p, T, fft_size):
    """
    Compute gradient of F_T(p) = (1/T) log(sum exp(T * conv(p,p)))
    Returns the gradient vector (length n).
    """
    n = len(p)
    conv = fast_autocorr(p)
    max_conv = conv.max()
    logits = T * (conv - max_conv)
    w = np.exp(logits)
    Z = w.sum()
    w = w / Z  # normalized weights, sum to 1
    
    # grad[i] = 2 * sum_k w[i+k] * p[k] for k = 0..n-1
    # This is cross-correlation of w and p
    L = 1 << (3 * n - 2).bit_length()
    W_fft = rfft(w, L)
    P_fft = rfft(p, L)
    cross = irfft(W_fft * np.conj(P_fft), L)
    grad = 2.0 * cross[:n]
    
    return grad, conv, max_conv

def multiplicative_update(p, grad, eta):
    """
    Multiplicative weight update: p_new[i] ∝ p[i] * exp(-eta * grad[i])
    """
    log_p = np.where(p > 1e-300, np.log(p), -700)
    log_p_new = log_p - eta * grad
    log_p_new -= log_p_new.max()
    p_new = np.exp(log_p_new)
    s = p_new.sum()
    if s < 1e-300:
        return p.copy()
    return p_new / s

def smoothed_gradient_optimize(p0, time_limit, verbose=True):
    """
    Smoothed gradient descent with multiplicative updates and T annealing.
    p0 should be a probability vector (sum = 1, non-negative).
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_score = fast_autocorr(p).max()
    t0 = time.monotonic()
    
    # T annealing schedule: from low to high
    n_stages = 8
    T_values = np.geomspace(5.0, 200.0, n_stages)
    
    for stage, T in enumerate(T_values):
        stage_time = time_limit / n_stages
        t_stage = time.monotonic()
        
        # Step size: larger for small T, smaller for large T
        base_eta = 4.0 / T
        momentum = 0.0
        p_prev = p.copy()
        
        n_steps = 0
        while time.monotonic() - t_stage < stage_time:
            grad, conv, max_conv = smoothed_gradient_step(p, T)
            n_steps += 1
            
            # Multiplicative update
            eta = base_eta / (1.0 + 0.01 * n_steps)
            p_new = multiplicative_update(p, grad, eta)
            
            # Check improvement
            score = fast_autocorr(p_new).max()
            if score < best_score:
                best_score = score
                best_p = p_new.copy()
            
            p = p_new
            
            if verbose and n_steps % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SGD stage={stage} T={T:.0f} step={n_steps} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SGD stage={stage} T={T:.0f} done ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_p, best_score
