In general, the gradient of (a*a)_k with respect to a is:
grad_i = 2 * a_{k-i} if 0 <= k-i < n (and i is valid)
But when k-i = i (k = 2i), it's still 2*a_i (the factor of 2 is correct because both terms in the product involve a_i).

Wait, I showed above that for k=2, i=1 (so k-i=1=i): grad_1 = 2*a_1. Yes, the factor of 2 is correct.

So in general: grad_i = 2 * a_{k-i} if 0 <= k-i < n, else 0.

This is simply 2 times the "reversed and shifted" version of a.

So the subgradient of max_k (a*a)_k at the unique maximum k* is:
2 * a_{k* - i} for 0 <= k* - i < n

This is just 2 times a reversed and shifted! 

For the max of multiple functions, the subdifferential is the convex hull of the individual gradients at the active points.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fessler-Parks LP Nesterov with adaptive working-set constraint management
2. LogSumExp-smoothed gradient descent with backtracking line search
3. Systematic peak shaving with exact line search

5. Diversification-based restarts to escape local minima

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_mask=None, max_constraints=400):
    """Solve the LP best response: max sum(g) s.t. T@g <= M, g >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    if active_mask is not None and active_mask.sum() < size:
        idx = np.where(active_mask)[0]
        if len(idx) == 0:
            active_mask = np.ones(size, dtype=bool)
            idx = np.arange(size)
        constraints = [T[idx, :] @ b_var <= M[idx]]
    else:
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b_var.value is None:
        return None, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

    # Identify active constraints
    circ = T @ b_val
    tol = 1e-6 * M
    new_active = circ >= M - tol

    return b_val, new_active


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_constraints=300,
                         gamma_init=2.0, gamma_min=1.4, stall_threshold=200):
    """Fessler-Parks accelerated LP Nesterov with adaptive working set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    size = 2 * n - 1

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = gamma_init
    M_hist = []
    active_mask = np.ones(size, dtype=bool)
    last_lp_time = 0.0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2.0:
            break

        t = n_iters + 1

        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        # Solve LP with working set
        lp_t0 = time.monotonic()
        b, new_active = lp_best_response(y, M_curr, solver, active_mask=active_mask,
                                         max_constraints=max_constraints)
        last_lp_time = time.monotonic() - lp_t0

        if b is None:
            # Fallback: use all constraints
            b, new_active = lp_best_response(y, M_curr, solver, active_mask=None)
            if b is None:
                continue
            active_mask = np.ones(size, dtype=bool)

        # Verify solution satisfies all constraints
        T_full = build_conv_matrix(y)
        circ_check = T_full @ b
        if circ_check.max() > M_curr * (1 + 1e-4):
            # Solution infeasible, use all constraints
            b, new_active = lp_best_response(y, M_curr, solver, active_mask=None)
            if b is None:
                continue
            active_mask = np.ones(size, dtype=bool)

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Update working set: keep active + some margin
        if new_active is not None:
            n_active = int(new_active.sum())
            if n_active > max_constraints:
                circ_vals = T_full @ b
                top_idx = np.argsort(circ_vals)[::-1][:max_constraints]
                active_mask = np.zeros(size, dtype=bool)
                active_mask[top_idx] = True
            else:
                # Add a buffer of near-active constraints
                circ_vals = T_full @ b
                threshold = M_curr * (1.0 - 0.05)
                buffer = circ_vals >= threshold
                combined = new_active | buffer
                if combined.sum() > max_constraints:
                    top_idx = np.argsort(circ_vals * combined)[::-1][:max_constraints]
                    active_mask = np.zeros(size, dtype=bool)
                    active_mask[top_idx] = True
                else:
                    active_mask = combined

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        if alpha > 0.5:
            alpha = 0.5

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            if momentum > 1.0:
                momentum = 1.0
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if len(M_hist) > 50:
            recent = M_hist[-30:]
            osc = np.sum(np.abs(np.diff(recent))) / (30 * max(recent[0], 1e-15))
            if osc > 0.03:
                gamma = max(gamma_min, gamma - 0.05)
            elif osc < 0.005:
                gamma = min(2.0, gamma + 0.03)

        if verbose and n_iters % 500 == 0:
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, "
                  f"M={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}, lp_t={last_lp_time*1000:.0f}ms")

        # Adaptive restart with diversification
        if stall > stall_threshold:
            b_curr = fast_autocorr(y)
            p_peak = int(np.argmax(b_curr))
            # Diversification: perturb positions that contribute to the peak
            js = max(0, p_peak - n + 1)
            je = min(n - 1, p_peak)
            contrib_positions = np.arange(js, je + 1)
            rng = np.random.default_rng(n_iters * 7 + 13)

            # Reduce mass at contributing positions, add to non-contributing
            n_pert = max(10, n // 20)
            pert_idx = rng.choice(contrib_positions, size=min(n_pert, len(contrib_positions)),
                                  replace=False)
            non_contrib = np.setdiff1d(np.arange(n), contrib_positions)
            if len(non_contrib) == 0:
                non_contrib = np.arange(n)

            y_trial = y.copy()
            a_trial = a.copy()
            for idx in pert_idx:
                amt = y[idx] * rng.uniform(0.1, 0.4)
                dst = non_contrib[rng.integers(len(non_contrib))]
                y_trial[idx] -= amt
                y_trial[dst] += amt
                a_trial[idx] -= amt
                a_trial[dst] += amt

            y_trial = np.maximum(y_trial, 0)
            a_trial = np.maximum(a_trial, 0)
            if y_trial.sum() > 1e-15:
                y = y_trial * norm / y_trial.sum()
            if a_trial.sum() > 1e-15:
                a = a_trial * norm / a_trial.sum()

            prev_y = y.copy()
            stall = 0
            gamma = gamma_init
            M_hist = []
            active_mask = np.ones(size, dtype=bool)
            if verbose:
                print(f"    [Diversification restart at iter {n_iters}]")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LogSumExp-smoothed gradient descent with backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.1, best_score * 0.05)
    T_end = max(0.005, best_score * 0.002)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient via FFT: grad = 2 * (w * a_reversed) convolved appropriately
        # (a*a)_k = sum_j a_j a_{k-j}, d/da_i = 2 * sum_k w_k a_{k-i}
        # This is 2 * (w convolved with reversed a) at appropriate indices
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a[::-1], fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]

        # Backtracking line search
        step = 1.0 / (n * 0.5)
        current_score = max_b
        best_trial = a.copy()
        best_trial_score = current_score

        for ls in range(10):
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_trial.copy()
            if sc < current_score * (1 - 0.001 * (1 + ls)):
                break
            step *= 0.5

        a = best_trial
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """Systematic peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Find top 3 peaks
        sorted_peaks = np.argsort(b)[::-1][:3]

        for p_star in sorted_peaks:
            p_star = int(p_star)
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[p_star - idx]

            if contrib.max() < 1e-18:
                continue

            # Top contributing sources
            n_top = min(5, len(idx))
            top_src = idx[np.argsort(contrib)[::-1][:n_top]]

            # Non-contributing positions
            non_contrib = np.zeros(n, dtype=bool)
            if js > 0:
                non_contrib[:js] = True
            if je < n - 1:
                non_contrib[je + 1:] = True

            for src in top_src:
                non_contrib[src] = True
                non_contrib_idx = np.where(non_contrib)[0]
                if len(non_contrib_idx) == 0:
                    continue

                # Pick destination: lowest value non-contributing
                n_select = min(8, len(non_contrib_idx))
                sorted_dst = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_select]

                for dst in sorted_dst[:3]:
                    # Line search
                    src_val = a[src]
                    if src_val < 1e-15:
                        continue
                    best_frac = 0
                    best_sc = max_b
                    for frac in [0.03, 0.07, 0.12, 0.2, 0.35]:
                        amt = src_val * frac
                        if amt < 1e-18:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            break
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_sc:
                            best_sc = sc
                            best_frac = frac

                    if best_frac > 0:
                        amt = src_val * best_frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial * norm / a_trial.sum()
                        sc = fast_autocorr(a_trial).max()

                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
                        else:
                            # Annealing
                            T_ann = best_score * 0.0005 * (1.0 - elapsed_frac)
                            if rng.random() < np.exp(-(sc - best_score) / max(T_ann, 1e-10)):
                                a = a_trial
                            n_stall += 1

        # Random exploration when stalled
        if n_stall > 30:
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                if a[idx] > 1e-15:
                    amt = a[idx] * rng.uniform(0.05, 0.3)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    n_improve += 1
                n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def multi_peak_subgradient(a0, time_limit, verbose=True):
    """Multi-peak subgradient with adaptive step size and projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_size = 0.05 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Top K peaks with softmax weighting
        K = min(7, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:K]
        peak_vals = b[sorted_idx]

        T_w = max(0.005, 0.05 * (max_b - peak_vals.min() + 1e-10) * (1.0 - elapsed_frac))
        w = np.exp((peak_vals - max_b) / T_w)
        w = w / w.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for m, p in enumerate(sorted_idx):
            if w[m] < 1e-6:
                continue
            p = int(p)
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            grad[js:je + 1] += 2.0 * w[m] * a[p - np.arange(js, je + 1)]

        # Project onto tangent space
        grad -= np.mean(grad)

        # Adaptive step with backtracking
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break

        step = step_size
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()

        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            step_size = min(step_size * 1.1, 0.5 / n)
        elif score_new < max_b:
            # Accept small worsening
            if np.random.random() < 0.3 * (1.0 - elapsed_frac):
                a = a_new
            step_size *= 0.7
        else:
            step_size *= 0.5

        steps += 1

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeak-SG step={steps} ({elapsed:.0f}s): best={best_score:.8f}, step={step:.2e}")

        # Periodic restart
        if steps % 2000 == 0 and step_size < 1e-8 / n:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.005 * a.mean(), n), 0)
            a = a * norm / a.sum()
            step_size = 0.05 / n

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining LP Nesterov, LSE gradient, multi-peak subgradient,
    and peak shaving with adaptive time allocation and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 450.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                        max_constraints=250, gamma_init=2.0,
                                        gamma_min=1.4, stall_threshold=150)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: LSE gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE-GD n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True,
                                        max_constraints=200, gamma_init=1.8,
                                        gamma_min=1.3, stall_threshold=120)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Multi-peak subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: MultiPeak-SG n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = multi_peak_subgradient(a_norm, t4, verbose=True)
        do_try_update(a_out, f"mpsg_n{n_cur}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t5, seed=777, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-n exploration (budget={remaining:.0f}s) ===")
    n_alternatives = [n_base + 100, n_base - 100, n_base + 200]
    n_alternatives = [nn for nn in n_alternatives if 500 <= nn <= 3000]
    per_n_time = min(remaining * 0.12, 120.0) / max(len(n_alternatives), 1)

    for n_alt in n_alternatives:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_alt = min(per_n_time, remaining * 0.4)
        if t_alt < 15:
            continue
        print(f"  Trying n={n_alt} (budget={t_alt:.0f}s)")
        init_long = interpolated_sequence(best_seq, n_alt)
        a_norm = init_long * np.sqrt(2.0 * n_alt) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t_alt, solver, verbose=False,
                                        max_constraints=150, gamma_init=1.8,
                                        gamma_min=1.3, stall_threshold=100)
        do_try_update(a_out, f"lp_n{n_alt}")

    # === Phase 7: Final LP polish at best n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP polish n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True,
                                        max_constraints=150, gamma_init=1.6,
                                        gamma_min=1.2, stall_threshold=100)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 120.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t8, seed=27182, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # === Phase 9: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            print(f"\n=== Phase 9: Final LSE n={n_cur} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_gradient_descent(a_norm, t9, verbose=True)
            do_try_update(a_out, f"lse_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
