```python
def lse_gradient(a, T):
    n = len(a)
    fft_size = 1 << (3*n-2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    A = rfft(a_pad, fft_size)
    conv = irfft(A * A, fft_size)[:2*n-1]
    
    # Softmax weights
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(T * conv_shifted)
    w = exp_vals / exp_vals.sum()
    
    # Gradient: sum_k w[k] * (a[k-m] + a[m-k]) for each m
    # = (w conv a)[m] + (w conv flip(a))[m]
    # Using FFT:
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    W = rfft(w_pad, fft_size)
    
    # (w * a)[m] = ifft(W * A)[m] for m = 0,...,n-1
    # This is the linear convolution of w (len 2n-1) with a (len n), evaluated at positions 0..n-1
    w_a = irfft(W * A, fft_size)[:n]
    
    # (w * flip(a))[m]: flip(a) has length n, (flip(a))[j] = a[n-1-j]
    # (w * flip(a))[m] = sum_k w[k] * flip(a)[m-k] = sum_k w[k] * a[n-1-m+k]
    fa_pad = np.zeros(fft_size)
    fa_pad[:n] = a[::-1]
    FA = rfft(fa_pad, fft_size)
    w_fa = irfft(W * FA, fft_size)[:n]
    
    grad = T * (w_a + w_fa)
    return grad
```

```python
    for stage in range(n_stages):
        T = T_start * (T_end / T_start) ** (stage / max(1, n_stages - 1))
        stage_t0 = time.monotonic()
        if verbose:
            print(f"    Adam stage {stage+1}/{n_stages}: T={T:.1f}")

        while time.monotonic() - t0 < time_limit:
            t_count += 1

            lse_val, grad, conv_max = lse_obj_and_grad(a, T)

            lr = base_lr / (1.0 + 0.1 * t_count / 1000.0)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_count)
            v_hat = v / (1.0 - beta2 ** t_count)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            a = a_new

            if lp_recenter_interval > 0 and solver is not None and t_count % lp_recenter_interval == 0:
                M_curr = fast_autocorr(a).max()
                b = lp_best_response_fast(a, M_curr, solver)
                if b is not None:
                    b_sum = b.sum()
                    if b_sum > 1e-12:
                        g = b * norm / b_sum
                        blend = 0.3
                        a_mixed = (1.0 - blend) * a + blend * g
                        a_mixed = np.maximum(a_mixed, 0.0)
                        s_mix = a_mixed.sum()
                        if s_mix > 1e-12:
                            a_mixed = a_mixed * norm / s_mix
                            score_mix = fast_autocorr(a_mixed).max()
                            if score_mix < best_score:
                                best_score = score_mix
                                best_a = a_mixed.copy()
                            if score_mix < score:
                                a = a_mixed
                                m = 0.5 * m
                                v = 0.5 * v

    return best_a, best_score


def lp_best_response_fast(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_iters=500):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    stall_count = 0
    last_best = best_score

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_fast(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if best_score < last_best - 1e-10:
            stall_count = 0
            last_best = best_score
        else:
            stall_count += 1

        if stall_count > 100:
            break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        improved = False

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-12:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]

                for dst in candidates_sorted[:8]:
                    for frac in [0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            if stall % 20 == 0:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.5, 1.5)
                try_update(a_trial)

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    # Phase 1: Quick LP to get a good starting point
    print(f"\n=== Phase 1: Quick LP n={n_base} (budget={min(45, time_budget*0.03):.0f}s) ===")
    t1 = min(45.0, time_budget * 0.03)
    if t1 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t1, solver, verbose=False, max_iters=60)
        do_try_update(a_out, f"lp_quick_n{n_base}")

    # Phase 2: Adam LSE optimization (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 2: Adam LSE n={n_best} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.5
    a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, sc_out = adam_lse_optimize(a_init, t2, T_start=60.0, T_end=3.0,
                                       n_stages=8, base_lr=0.08,
                                       lp_recenter_interval=2000, solver=solver, verbose=True)
    do_try_update(a_out, f"adam_lse_n{n_best}")

    # Phase 3: LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 3: LP polish n={n_best} (budget={min(remaining*0.2, 200):.0f}s) ===")
    t3 = min(remaining * 0.2, 200.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, max_iters=300)
        do_try_update(a_out, f"lp_polish_n{n_best}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 4: Coord descent n={n_best} (budget={min(remaining*0.25, 300):.0f}s) ===")
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t4, seed=12345)
        do_try_update(a_out, "coord_d1")

    # Phase 5: Second Adam LSE round (lower temperature)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 5: Adam LSE round 2 n={n_best} (budget={min(remaining*0.2, 180):.0f}s) ===")
    t5 = min(remaining * 0.2, 180.0)
    if t5 > 10:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adam_lse_optimize(a_init, t5, T_start=20.0, T_end=1.5,
                                       n_stages=5, base_lr=0.05,
                                       lp_recenter_interval=0, solver=None, verbose=True)
        do_try_update(a_out, f"adam_lse2_n{n_best}")

    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 6: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, remaining, seed=77777)
        do_try_update(a_out, "coord_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
