- Else: we're close to δ*

After 30-40 iterations, we'll have δ* to high precision.

But wait, there's a subtlety: the argmax set K(δ) can change as δ varies (when two linear/quadratic functions cross). This means the subgradient can jump. The bisection should still work because F is convex (the subgradient is monotonically non-decreasing).

Let me code this up.

Actually, I realize there might be an even faster approach. Since d_k = a[k-i] - a[k-j], the vector d is the difference of two shifted versions of a. The convolution conv + 2δ*d can be computed as:
conv + 2δ*(shifted_a_i - shifted_a_j)

where shifted_a_i[k] = a[k-i] (zero-padded) and shifted_a_j[k] = a[k-j] (zero-padded).

We can pre-compute the "affected" range of d and only update those positions.

But for simplicity, let me just do the O(n) computation each time.

Let me now write the full code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with adaptive active set
2. Pair coordinate descent with subgradient line search
3. Subgradient "water-filling" method
4. Multi-length exploration with interpolation
5. Perturbation-based restarts
"""

import numpy as np

import time
from scipy.fft import rfft, irfft

_GLOBAL_SOLVER = None

def get_solver():

    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _GLOBAL_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _GLOBAL_SOLVER

def fast_autocorr(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific k indices.
    Row k: T[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    T = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def build_conv_matrix_full(a):
    """Build full convolution matrix."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver, active_set=None, warm_start=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M * np.ones(len(active_set))]
    else:
        active_set = np.arange(size)
        T = build_conv_matrix_full(a)
        constraints = [T @ b_var <= M * np.ones(size)]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set
    
    return b_val, active_set

def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    """Nesterov-accelerated LP best-response."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max() / (a.sum()**2)  # max(conv)/(sum)^2
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_set = None
    stall_count = 0
    stall_threshold = 30
    momentum_history = []
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-15:
            break
        
        # Adaptive active set
        if active_set is None or t % 50 == 0:
            tol = 0.05 + 0.02 * np.exp(-t / 300.0)
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            if len(active_set) < 3:
                active_set = np.where(conv_y >= M_curr * 0.7)[0]
            # Cap active set size
            if len(active_set) > 0.6 * (2 * n - 1):
                # Keep top 60%
                idx = np.argsort(conv_y)[::-1][:int(0.6 * (2 * n - 1))]
                active_set = idx
        
        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        
        # Check for violations
        if b is not None and len(active_set) < 2 * n - 1:
            conv_b = np.zeros(2 * n - 1)
            # Quick check using FFT
            conv_b = fast_autocorr(y) # This is wrong, we need conv(y, b)
            # Actually compute conv(y, b) properly
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            Y = rfft(y, fft_size)
            B = rfft(b, fft_size)
            conv_yb = irfft(Y * B, fft_size)[:2 * n - 1]
            if conv_yb.max() > M_curr * (1 + 1e-6):
                # Expand active set
                active_set = np.arange(2 * n - 1)
                b, active_set = lp_best_response(y, M_curr, solver, None)
        
        if b is None:
            if t % 50 == 0:
                active_set = None
                b, active_set = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.9)  # Cap momentum
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0
        
        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={n_iters} ({elapsed:.0f}s): best_norm={best_score:.10f}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with subgradient line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:

    best_score = conv.max()
    best_a = a.copy()

    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        n_improvements = 0
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

                # a[k-i] for k in [i, i+n-1]

                # -a[k-j] for k in [j, j+n-1]
