def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def get_solver():

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve_lucky(a, M, solver, max_constraints=500, warm_start=None):
    """Solve LP with lazy constraint generation.
    
    Maximize sum(b) subject to:
    - b >= 0
    - (T @ b)_k <= M for all k (but only enforce a subset)
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    b_var = cp.Variable(n, nonneg=True)
    
    if warm_start is not None:
        b_var.value = warm_start
    
    # Start with top K constraints
    test_b = np.ones(n) / n
    circ = T @ test_b
    top_idx = np.argsort(circ)[::-1][:max_constraints]
    
    constraints = [T[top_idx, :] @ b_var <= M[top_idx]]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Check all constraints, add violated ones
    for _ in range(10):
        circ = T @ b_val
        violated = circ > M + 1e-10
        n_violated = int(violated.sum())
        
        if n_violated == 0:
            break
        
        # Add the most violated constraints
        if n_violated <= 200:
            add_idx = np.where(violated)[0]
        else:
            add_idx = np.where(violated)[0]
            viol_amounts = circ[add_idx]
            top_add = add_idx[np.argsort(viol_amounts)[::-1][:200]]
            add_idx = top_add
        
        # Rebuild with existing active + new violated
        active_mask = (circ >= M - 1e-8 * max(M.max(), 1e-15)) | violated
        if active_mask.sum() > max_constraints * 2:
            # Prioritize most violated
            active_idx = np.where(active_mask)[0]
            circ_active = circ[active_idx]
            top = active_idx[np.argsort(circ_active)[::-1][:max_constraints * 2]]
            active_mask = np.zeros(size, dtype=bool)
            active_mask[top] = True
        
        constraints = [T[active_mask, :] @ b_var <= M[active_mask]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, max_constraints=300):
    """Fessler-Parks accelerated LP iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.9
    warm_start = None

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_solve_lucky(y, M_curr, solver, max_constraints=max_constraints, warm_start=warm_start)
        if b is None:
            # Fallback: use y itself
            b = y.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        warm_start = b_val = b.copy()

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if stall > 200 and stall % 50 == 0:
            gamma = max(1.0, gamma - 0.15)
        
        if stall > 300:
            rng = np.random.default_rng(n_iters * 31 + 17)
            scale = 0.05 * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.9
            warm_start = None
            if verbose:
                print(f"    [LP Restart at iter {n_iters}]")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def adam_lse_gd(a0, time_limit, verbose=True, lr_base=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.001, best_score * 0.001)
    
    a_min = len(a0)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** (elapsed_frac ** 0.5)
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # LSE weights
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient via FFT
        # grad_j = 2 * sum_k w_k * a_{k-j} (with appropriate bounds)
        # This is 2 * (w * a_reversed) full convolution, sliced
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        
        # Project gradient onto tangent space (maintain sum)
        grad -= np.mean(grad)
        
        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        # Adaptive learning rate
        lr = lr_base / (1.0 + steps * 0.01)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = a - lr * update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-LSE step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
        
        # Periodic restart if stuck
        if steps % 2000 == 0 and steps > 0:
            # Check if we've made progress in last 2000 steps
            # (simplified: just do a small perturbation)
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.01 * a.mean(), n), 0)
            a = a * norm / a.sum()
            m *= 0.5
            v *= 0.5

    return best_a, best_score


def coordinate_descent_peak(a0, time_limit, verbose=True):
    """Coordinate descent that targets the peak of the autocorrelation.
    
    At each step, identify the peak lag and the coordinates that most contribute
    to it, then perform a 1D optimization on one of those coordinates.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing coordinates
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        if js > je:
            # Peak is outside valid range (shouldn't happen for linear conv)
            break
        
        idx = np.arange(js, je + 1)
        # Contribution of a_j to b_{p_star}: a_j * a_{p_star - j}
        contrib = a[idx] * a[p_star - idx]
        
        # Pick the coordinate with max contribution
        j_star = idx[np.argmax(contrib)]
        
        # 1D optimization: find optimal value of a_{j_star}
        # b_{p_star} as function of a_{j_star} = x:
        # = x * a_{p_star - j_star} + (contribution from other pairs)
        # But changing a_{j_star} also affects other b_k values!
        # For simplicity, just try a few values
        
        a_save = a[j_star]
        base_contrib = contrib.max()  # a[j_star] * a[p_star - j_star]
        other_b = b.copy()
        other_b[p_star] -= base_contrib
        
        best_x = a_save
        best_local = max_b
        
        # Try reducing a[j_star]
        for frac in [0.95, 0.9, 0.85, 0.8, 0.7, 0.5, 0.3]:
            x = a_save * frac
            # New b_{p_star} = x * a[p_star - j_star] + (other_b)[p_star]
            # But we also need to check other peaks...
            # Approximation: just check b_{p_star}
            new_peak = x * a[p_star - j_star] + other_b[p_star]
            if new_peak < best_local:
                best_local = new_peak
                best_x = x
        
        # Also try increasing (if the peak is elsewhere due to normalization)
        # Actually, let's do a proper mini line search
        
        if best_x < a_save:
            a_trial = a.copy()
            # Move mass from j_star to a position far from the peak
            moved = a_save - best_x
            # Find position that minimizes addition to all peaks
            # Approximate: position with smallest a value that doesn't contribute to p_star
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je+1] = False  # positions contributing to p_star
            non_contrib[j_star] = True  # can't move to self
            
            if non_contrib.sum() > 0:
                dst_candidates = np.where(non_contrib)[0]
                # Pick the one with smallest current value
                dst = dst_candidates[np.argmin(a[dst_candidates])]
                a_trial[j_star] = best_x
                a_trial[dst] += moved
            else:
                # Move to the position that contributes least
                dst = idx[np.argmin(contrib)]
                if dst == j_star:
                    dst = (j_star + 1) % n
                a_trial[j_star] = best_x
                a_trial[dst] += moved
            
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    iters += 1
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 200:
            # Random perturbation
            rng = np.random.default_rng(iters * 7 + stall)
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a = np.maximum(a * (1 + rng.normal(0, 0.1, n_pert))[:, :n_pert] if False else a * np.ones(n), 0)
            # Simpler perturbation
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * 0.1
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            stall = 0
        
        if verbose and iters % 500 == 0 and iters > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent iter={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True, eta0=1.0):
    """Multiplicative (mirror) descent on LSE-smoothed objective.
    
    Update: a_j <- a_j * exp(-eta * grad_j) / normalization
    This naturally keeps values positive.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = eta0 / n
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.001, best_score * 0.002)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** (elapsed_frac ** 0.7)
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        grad -= np.mean(grad)
        
        # Mirror update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)  # Keep strictly positive
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Decay step size
        eta *= 0.999
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent step={steps} ({elapsed:.0f}s): eta={eta:.2e}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def diverse_restart(a, rng, mode=0):
    """Generate diverse restarts from current best."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    
    if mode == 0:
        # Gaussian perturbation
        a_new = np.maximum(a + rng.normal(0, 0.05 * a.mean(), n), 0)
    elif mode == 1:
        # Block move
        n_blocks = rng.integers(3, 10)
        a_new = a.copy()
        for _ in range(n_blocks):
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = a[i] * rng.uniform(0.05, 0.3)
                a_new[i] -= amt
                a_new[j] += amt
        a_new = np.maximum(a_new, 0)
    elif mode == 2:
        # Blend with Gaussian shape
        x = np.linspace(0, 1, n)
        gauss = np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2))
        alpha = rng.uniform(0.1, 0.4)
        a_new = (1 - alpha) * a + alpha * gauss / gauss.sum() * a.sum()
    elif mode == 3:
        # Blend with triangular shape
        x = np.linspace(0, 1, n)
        tri = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        alpha = rng.uniform(0.1, 0.4)
        a_new = (1 - alpha) * a + alpha * tri / tri.sum() * a.sum()
    else:
        # Random reshuffle (permutation of sorted values)
        a_sorted = np.sort(a)[::-1]
        perm = rng.permutation(n)
        a_new = a_sorted[perm]
        # This changes the autocorrelation significantly
    
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return np.ones(n) / n
    return a_new / s_new
