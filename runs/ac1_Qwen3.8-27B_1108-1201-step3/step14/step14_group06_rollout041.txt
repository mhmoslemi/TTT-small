def fast_autocorr(a):
    """Compute linear autocorrelation using FFT. Input: 1D array. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad(a, beta):
    """
    Compute LSE-smoothed max-convolution objective and gradient.
    f = (1/beta)*log(sum_k exp(beta*b_k)) where b = autocorr(a)
    grad_i = 2 * sum_k p_k * a_{k-i} where p = softmax(beta*b)
    Returns (f, grad) where grad has same length as a.
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    shifted = beta * (b - max_b)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    w = exp_vals / exp_sum
    obj = max_b + np.log(exp_sum) / beta

    # Gradient: h_i = sum_j a_j * w[i+j] for j=0..n-1
    # Use FFT cross-correlation: h = ifft(FFT(w_padded) * conj(FFT(a_padded)))[:n]
    L = 1 << (3 * n - 1).bit_length()
    w_pad = np.zeros(L)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(L)
    a_pad[:n] = a
    W_fft = rfft(w_pad, L)
    A_fft = rfft(a_pad, L)
    h = irfft(W_fft * np.conj(A_fft), L)[:n]
    grad = 2.0 * h
    return obj, grad


def momentum_lse_optimize(a0, time_limit, beta_start=60.0, beta_end=4.0, n_stages=10, lr_scale=0.5):
    """
    Momentum-based SGD on the LSE-smoothed objective.
    a is kept on the simplex {a>=0, sum(a)=1}.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    betas = np.geomspace(beta_start, beta_end, n_stages)
    v = np.zeros(n)
    gamma = 0.9
    total_iters = 0

    for si, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        stage_time_left = time_limit - (time.monotonic() - t0)
        if stage_time_left < 5:
            break
        stage_budget = stage_time_left / (n_stages - si)

        iters_in_stage = 0
        while time.monotonic() - t0 < t0 + stage_budget * 1.2:
            obj, grad = lse_obj_grad(a, beta)
            # Adaptive lr
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            lr = lr_scale / (n ** 0.5) / (1.0 + 0.1 * total_iters / 1000.0)
            lr = min(lr, 0.05 / max(grad_norm, 1e-10))

            v = gamma * v + (1.0 - gamma) * grad
            a_new = a - lr * v
            # Project
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            a = a_new
            total_iters += 1
            iters_in_stage += 1

            # Check improvement
            if iters_in_stage % 20 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        # End-of-stage check
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def build_constraint_matrix(a, n):
    """Build the (2n-1) x n constraint matrix T where (T @ b)_k = sum_j a_{|k-j|} b_j approximately."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T @ b <= M, b >= 0."""
    if not _have_cvxpy or solver is None:
        return None
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_optimize(a0, time_limit, solver, use_multi_M=True):

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall = 0

    for t in range(1, 200000):

            # Adaptively choose based on iteration count
            min_eps = min(0.01, 0.002 * np.sqrt(t))
            max_eps = min(0.06, 0.004 * np.sqrt(t))
            eps_values = [e for e in eps_values if min_eps <= e <= max_eps]

            eps_values = [0.03]

        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:

        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = g.copy()
        else:

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        # Track best

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        else:

def pair_local_search(a0, time_limit, seed=42, n_peaks=20, n_pairs=20):
    """
    Targeted local search: identify pairs (i, k-i) contributing to top peaks,
    then try redistributing mass to reduce peak contributions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top peaks
        top_peak_indices = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peak_indices:
            # Valid range for i in (a_i * a_{k-i})
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            # Contribution of each i to peak k
            contrib = a[idx] * a[k - idx]
            # Find top contributors
            order = np.argsort(contrib)[::-1]
            top_contrib_idx = idx[order[:n_pairs]]

            # For each top contributor, try moving mass to a "quieter" position
            for src_i in top_contrib_idx:
                if a[src_i] < 1e-12:
                    continue
                # Need to reduce a[src_i] * a[k-src_i]
                # Option 1: reduce a[src_i] (move to a different position)
                # Option 2: reduce a[k-src_i] (move to a different position)
                partner = k - src_i
                if partner < 0 or partner >= n:
                    continue

                # Move some mass from src_i to a random "low" position
                # or to a position that doesn't contribute to this peak
                for _ in range(3):
                    # Choose destination that minimizes new peak contribution
                    # Simple: random position with low value
                    candidates = np.where(a < 0.7 * a.mean())[0]
                    if len(candidates) == 0:
                        candidates = np.arange(n)
                    dst = rng.choice(candidates)
                    if dst == src_i:
                        continue
                    for frac in [0.03, 0.08, 0.15]:
                        amt = a[src_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-12:
                            continue
                        a_trial = a_trial / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape local minima
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-12:
                    continue
                a_trial = a_trial / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Larger random kick
            kick = rng.standard_normal(n) * 0.02 * a.mean()
            a_trial = a + kick
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-12:
                a_trial = a_trial / s_t
                score = fast_autocorr(a_trial).max()
                # Accept with some probability (simulated annealing-like)
                if score < best_score * 1.02:
                    a = a_trial.copy()
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        improved = True

        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate diverse structured initializations (normalized to sum=1)."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        # Linearly increasing (asymmetric)
        x = np.linspace(0.0, 1.0, n)
        a = x + 0.1
    elif choice == 5:
        # V-shape (high at ends)
        x = np.linspace(0.0, 1.0, n)
        a = 2.0 * np.abs(x - 0.5) + 0.1
    elif choice == 6:
        # Triangular peak
        x = np.linspace(0.0, 1.0, n)
        a = np.maximum(0, 1.0 - 2.0 * np.abs(x - 0.3))
    else:
        # Sum of cosines with random phases
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for k in range(1, 6):
            a += 0.3 / k * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_normalized(a):
    """Evaluate the objective for a normalized sequence (sum=1)."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()


def attempt_update(best_seq, best_score, candidate, label, t_global):
    """Try to update best with candidate. Both are normalized (sum=1)."""
    if candidate is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-12:
        return best_seq, best_score
    a = a / s
    score = evaluate_normalized(a)
    if score < best_score:
        best_score = score
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and lengths to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_normalized(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_normalized(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Momentum LSE (fast global convergence) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: Momentum LSE n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = momentum_lse_optimize(best_seq, t1, beta_start=80.0, beta_end=5.0, n_stages=8, lr_scale=0.8)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"mom_lse_n{n_base}", t_global)

    # === Phase 2: LP Nesterov (main refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30 and solver is not None:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, use_multi_M=True)
        best_seq, best_score = attempt_update(best_seq, best_score, f"lp_r1_n{n_base}", t_global)

    # === Phase 3: Momentum LSE round 2 (refine after LP) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 3: Momentum LSE round 2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = momentum_lse_optimize(best_seq, t3, beta_start=40.0, beta_end=2.0, n_stages=6, lr_scale=0.5)
        best_seq, best_score = attempt_update(best_seq, best_score, f"mom_lse2_n{n_cur}", t_global)

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 30 and solver is not None:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, use_multi_M=True)
        best_seq, best_score = attempt_update(best_seq, best_score, f"lp_r2_n{n_cur}", t_global)

    # === Phase 5: Pair local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 5: Pair local search n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_local_search(best_seq, t5, seed=42, n_peaks=20, n_pairs=15)
        best_seq, best_score = attempt_update(best_seq, best_score, f"pair_ls_n{n_cur}", t_global)

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    t6 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick LSE on long sequence
        t6a = t6 * 0.4
        if t6a > 20:
            a_out, _ = momentum_lse_optimize(init_long, t6a, beta_start=50.0, beta_end=5.0, n_stages=6, lr_scale=0.6)
            best_seq, best_score = attempt_update(best_seq, best_score, f"lse_long_n{n_long}", t_global)
        remaining6 = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining6, t6 * 0.5)
        if t6b > 30 and solver is not None:
            a_cur = interpolate_sequence(best_seq, n_long)
            a_out, _ = lp_nesterov_optimize(a_cur, t6b, solver, use_multi_M=True)
            best_seq, best_score = attempt_update(best_seq, best_score, f"lp_long_n{n_long}", t_global)

    # === Phase 7: Try shorter sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(800, len(best_seq) - 400)
    if n_short != len(best_seq):
        t7 = min(remaining * 0.15, 200.0)
        print(f"\n=== Phase 7: Shorter n={n_short} (budget={t7:.0f}s) ===")
        if t7 > 30:
            init_short = interpolate_sequence(best_seq, n_short)
            a_out, _ = momentum_lse_optimize(init_short, t7 * 0.5, beta_start=50.0, beta_end=5.0, n_stages=5, lr_scale=0.6)
            best_seq, best_score = attempt_update(best_seq, best_score, f"lse_short_n{n_short}", t_global)
            remaining7 = time_budget - (time.monotonic() - t_global)
            if remaining7 > 20 and solver is not None:
                a_cur = interpolate_sequence(best_seq, n_short)
                a_out, _ = lp_nesterov_optimize(a_cur, min(remaining7, 100.0), solver, use_multi_M=True)
                best_seq, best_score = attempt_update(best_seq, best_score, f"lp_short_n{n_short}", t_global)

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        # Quick LSE with very low temperature
        t8a = min(remaining * 0.4, 100.0)
        if t8a > 15:
            a_out, _ = momentum_lse_optimize(best_seq, t8a, beta_start=20.0, beta_end=1.0, n_stages=4, lr_scale=0.3)
            best_seq, best_score = attempt_update(best_seq, best_score, "lse_polish", t_global)

        # Pair search with different seed
        remaining8 = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining8 * 0.5, 80.0)
        if t8b > 10:
            a_out, _ = pair_local_search(best_seq, t8b, seed=99999, n_peaks=15, n_pairs=10)
            best_seq, best_score = attempt_update(best_seq, best_score, "pair_polish", t_global)

    # Final evaluation
    if best_seq is not None:
        final_score = evaluate_normalized(best_seq)
        if final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Return as list of floats
    return [float(x) for x in best_seq]
