def fast_autocorr(a):
    """Compute linear autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def mirror_descent_nesterov(a0, time_limit, top_k=50, beta=3.0, gamma0=1.0, verbose=True):
    """Mirror descent with Nesterov acceleration and soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac0 = fast_autocorr(a)
    best_m = ac0.max()
    t0 = time.monotonic()
    
    # Nesterov state
    y = a.copy()  # Nesterov look-ahead point
    x_prev = a.copy()
    
    # Running average
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = gamma0 / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Compute soft subgradient
        k_max = min(top_k, len(ac))
        # Use all constraints within 95% of max as "active"
        threshold = 0.95 * M_curr
        active = np.where(ac >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        
        # Weight by ac^beta
        w = ac[active] ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j = k_int - idx
            grad[idx] += 2.0 * wi * y[j]
        
        # Mirror descent step on log scale
        log_y = np.log(y + 1e-30)
        log_y_new = log_y - step * grad
        max_log = log_y_new.max()
        y_new = np.exp(log_y_new - max_log)
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        # Nesterov update
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_n = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_n * (y_new - x_prev)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        x_prev = y_new
        y = x_new
        
        # Check improvements
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        avg_a += y
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, |A|={len(active)}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_constraints=200):
    """LP Frank-Wolfe with constraint generation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Track active constraints
    active_set = set()
    
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Select constraints: top max_constraints by value
        k_sel = min(max_constraints, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        
        # Build constraint matrix for selected lags
        # T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        b_var = cp.Variable(n, nonneg=True)
        
        constraints = []
        for k in top_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            # conv_k = sum_{j=i_start}^{i_end} y[k_int - j] * b[j]
            coeffs = y[k_int - np.arange(i_start, i_end + 1)]
            # Build sparse row
            row = np.zeros(n)
            row[i_start:i_end + 1] = coeffs
            constraints.append(row @ b_var <= M_curr + 1e-10)
        
        if not constraints:
            constraints = [b_var >= 0]
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Verify: check if g violates any constraint not in our set
        # (optional, for correctness)
        
        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta_n = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_n * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

This is not a standard TOEPLITZ matrix (where `T[k,j]` depends only on `k-j`). It IS a TOEPLITZ matrix! Because `T[k,j] = a[k-j]` depends only on `k-j`.

Wait, but only if `0 <= k-j < n`. Otherwise it's 0. So it's a TOEPLITZ matrix

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start asymmetric initializations exploiting problem non-convexity
2. Nesterov-accelerated mirror descent with adaptive soft subgradient
3. LP Frank-Wolfe with structure-exploiting constraint generation
4. Coordinate descent polish with targeted moves
5. Multi-length scaling (1600 -> 2048 -> 2800)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, threshold_frac=0.92, beta=3.0, top_k=80):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    thresh = threshold_frac * M
    active = np.where(ac >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(ac)])
    if len(active) > top_k:
        top_idx = np.argsort(ac)[::-1][:top_k]
        active = top_idx
    w = np.maximum(ac[active], 0) ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_vals = k_int - idx
        grad[idx] += wi * a[j_vals]
    return 2.0 * grad

def mirror_descent_nesterov(a0, time_limit, gamma0=1.0, threshold_frac=0.92, beta=3.0, top_k=80, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x_prev = a.copy()
    avg_a = a.copy()
    n_avg = 1

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        step = gamma0 / (np.sqrt(float(t)) + 1.0)
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, threshold_frac=threshold_frac, beta=beta, top_k=top_k)
        log_y = np.log(y + 1e-30)
        log_y_new = log_y - step * g
        max_log = log_y_new.max()
        y_new = np.exp(log_y_new - max_log)
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_n = (float(t) - 1.0) / (float(t) + 2.0)
            x_new = y_new + beta_n * (y_new - x_prev)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        x_prev = y_new.copy()
        y = x_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        avg_a += y
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, max_constraints=120):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-15:
            break
        k_sel = min(max_constraints, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in top_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            coeffs = y[k_int - np.arange(i_start, i_end + 1)]
            row = np.zeros(n)
            row[i_start:i_end + 1] = coeffs
            constraints.append(row @ b_var <= M_curr + 1e-10)
        if not constraints:
            continue
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_pt = b_val / b_sum
        alpha = 2.0 / (float(t) + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_pt
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        if t == 1:
            x_new = g_pt.copy()
        else:
            beta_n = (float(t) - 1.0) / (float(t) + 2.0)
            x_new = y_new + beta_n * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        y = y_new
        x = x_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    stall = 0
    max_stall = 400
    iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iters += 1
        ac = fast_autocorr(a)
        improved = False
        top_lags = np.argsort(ac)[::-1][:min(8, len(ac))]
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-25:
                continue
            src = idx[np.argmax(products)]
            for _ in range(15):
                dst = rng.integers(n)
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-17:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            stall += 1
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord it={iters} ({elapsed:.1f}s): J={J:.6f}")
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.1f}s): J={J:.6f}, iters={iters}")
    return best_a, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def gen_asymmetric_init(n, rng, variant=0):
    x = np.arange(n, dtype=np.float64)
    if variant == 0:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n)
    elif variant == 1:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 2 * x / n) + 0.1 * np.sin(2 * np.pi * x / n)
    elif variant == 2:
        t = x / (n - 1)
        a = 1.0 + 0.5 * (2 * t - 1)
        a = np.maximum(a, 0.01)
    elif variant == 3:
        t = x / (n - 1)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.sin(2 * np.pi * t)
    elif variant == 4:
        t = x / (n - 1)
        a = np.exp(-4.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (t - 0.7) ** 2) + 0.05
    else:
        a = 1.0 + 0.4 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
