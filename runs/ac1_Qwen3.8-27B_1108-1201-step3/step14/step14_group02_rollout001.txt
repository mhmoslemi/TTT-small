def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_lp_solver():
    if not HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lp_best_response(y, M):
    """Solve the LP: max sum(b) s.t. T(y) @ b <= M, b >= 0"""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    solver = get_lp_solver()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

    n = len(a0)

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        g = lp_best_response(y, M_curr)
        if g is None:
            break
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}")

def lp_smoothing_optimize(a0, time_limit, p=50, lr_init=0.1, verbose=True):
    """Smooth L_p optimization with natural gradient (mirror descent on simplex)."""
    n = len(a0)
    w = np.log(np.maximum(a0, 1e-30)).copy()
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    
    lr = lr_init
    momentum = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current a from w
        w_shifted = w - w.max()
        a = np.exp(w_shifted)
        a = a / a.sum()
        
        # Compute convolution
        ac = fast_autocorr(a)
        M = ac.max()
        
        # L_p "gradient": (ac - M)^(p-1) for ac > M*0.9, else 0
        # Actually, use the full gradient of sum max(0, ac[k] - M/2)^p
        # Simpler: gradient of sum ac[k]^p / p = ac[k]^(p-1) * dac/da
        # But this doesn't focus on the peak. Let's use:
        # f = sum_k (ac[k])^p / p  (this is smooth and its minimizer approximately minimizes the max)
        
        # For large p, the gradient is dominated by the largest elements
        # grad_f[i] = sum_k (ac[k])^(p-1) * (a[k-i] + a[i-k])
        
        # Compute in O(n * num_active) where num_active is small for large p
        # For efficiency, only use elements where ac[k] > 0.5 * M
        active = ac > 0.5 * M
        if not active.any():
            active = ac > 0
        
        # For very large p, only top few matter
        if p > 20:
            threshold = M * (0.5 + 0.5 * (1.0 / float(p)))
            active = ac > threshold
        
        weights = np.zeros(len(ac))
        weights[active] = ac[active] ** (p - 1)
        
        # Compute gradient
        grad = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += weights[k] * a[j]
        
        # Natural gradient step for softmax parameterization
        # dw = -lr * a * (grad - a^T * grad)
        a_dot_g = a @ grad
        nat_grad = a * (grad - a_dot_g)
        
        # Adam-like update
        momentum = beta1 * momentum + (1 - beta1) * nat_grad
        m_hat = momentum / (1 - beta1 ** t)
        v = (beta2 * (np.zeros(n))) + (1 - beta2) * nat_grad ** 2
        # Simplified: just use momentum
        w = w - lr * m_hat
        
        # Adaptive learning rate
        lr = lr_init / (np.sqrt(t) + 1.0)
        
        # Track best
        w_shifted = w - w.max()
        a_new = np.exp(w_shifted)
        a_new = a_new / a_new.sum()
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-smooth t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish_v2(a0, time_limit, verbose=False):
    """Enhanced coordinate polish with diverse moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top contributing positions (sources to move FROM)
        # A position i contributes to conv[k] through a[i]*a[k-i]
        # Total contribution of position i: sum_k a[i]*a[k-i] = a[i] * (sum of valid a[k-i])
        # = a[i] * sum_j a[j] = a[i] (since sum a = 1)
        # So all positions contribute proportionally to their value!
        # The key is WHERE they contribute: positions near the center contribute
        # to the peak, while positions at the edges contribute to the tails.
        
        # Better metric: for each position i, compute how much it contributes
        # to the TOP peaks specifically
        top_k_idx = np.argsort(ac)[::-1][:min(8, len(ac))]
        
        # For each source candidate, compute its "peak contribution"
        peak_contrib = np.zeros(n)
        for k in top_k_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            peak_contrib[idx] += a[idx] * a[j]
        
        # Source: highest peak contribution
        src_candidates = np.argsort(peak_contrib)[::-1][:20]
        
        # Destination: lowest peak contribution, prefer edges
        edge_bonus = np.linspace(1.0, 0.0, n)  # more bonus for edges
        dst_score = peak_contrib - 0.1 * edge_bonus
        dst_candidates = np.argsort(dst_score)[:20]
        
        improved = False
        
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
                
            for dst in dst_candidates:
                if dst == src:
                    continue
                
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_perturb = rng.integers(1, 5)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish v2 ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def spectral_equalization(a0, time_limit, verbose=False):
    """Try to flatten the convolution by adjusting in frequency domain."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length() - 1
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution
        ac = fast_autocorr(a)
        M = ac.max()
        target = 1.0 / (2 * n - 1)  # ideal flat value
        
        # Compute spectrum
        A = rfft(a, fft_size)
        C = np.abs(A) ** 2
        
        # We want ifft(C) to be flatter
        # Strategy: reduce the high-frequency components of C to flatten the time-domain
        # But we need to maintain that ifft(C) >= 0 (which is guaranteed if C = |A|^2 for real A)
        # And we need the time-domain signal to be non-negative
        
        # Alternative: directly modify a in frequency domain
        # Phase of A is determined by a (since a is real)
        # We can modify |A| to make C flatter, then take ifft and clip to positive
        
        # Simple approach: reduce |A[k]| for high k (where a has high-frequency content)
        # This makes a smoother, which... actually makes the convolution MORE peaked
        
        # Better: increase |A[k]| for intermediate k to spread the convolution
        # But this might make a negative
        
        # Let's try: a_new = a + epsilon * (target_pattern - a)
        # where target_pattern is chosen to flatten the convolution
        
        # Actually, let's use a different approach:
        # Compute the "ideal" a that would give flat convolution
        # If conv(a,a) = c_uniform = 1/(2n-1) for all k
        # Then |A|^2 = FFT(c_uniform) = delta_0 (only DC component)
        # This means A[k] = 0 for k > 0, so a is constant = uniform
        # But uniform gives conv peak = 1/n, not 1/(2n-1)
        
        # So flat convolution is unachievable. The best we can do is
        # minimize the deviation from flat, subject to a >= 0.
        
        # Practical approach: use the gradient of the "peak excess"
        # excess_k = max(0, ac[k] - target)
        # gradient of sum excess_k^2 with respect to a[i]
        
        excess = np.maximum(ac - target, 0)
        if excess.max() < 1e-15:
            break
        
        # Gradient
        grad = np.zeros(n)
        active = excess > 0
        for k in np.where(active)[0]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += excess[k] * a[j]
        
        # Step
        for lr in [0.01, 0.005, 0.001, 0.0005]:
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Spectral eq ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def soft_subgrad_mirrors_descent(a0, time_limit, top_k=30, beta=5.0, verbose=True):
    """Mirror descent with soft subgradient (improved version)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    w = np.log(a + 1e-30)
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        
        # Soft subgradient: weighted average of active constraint gradients
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 0)
        w_active = top_vals ** beta
        w_sum = w_active.sum()
        if w_sum < 1e-30:
            w_active = np.ones(top_k) / top_k
        else:
            w_active = w_active / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w_active, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]
        
        # Mirror descent step
        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * grad
        w_shifted = w - w.max()
        a_new = np.exp(w_shifted)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

    return best_a, best_m
