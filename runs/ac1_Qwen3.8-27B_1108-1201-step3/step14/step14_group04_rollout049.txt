def lse_fista(a0, time_limit, verbose=True):
    """LSE-FISTA with adaptive temperature schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    norm = np.sqrt(2.0 * n)
    a = a * norm / norm  # keep sum = 1 for now, scale at end
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    T_list = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    L_base = n / 20.0
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()
        stage_time = min((time_limit - (time.monotonic() - t0)) * 0.12, 60.0)
        
        L = max(L_base / (T * 5.0), 0.1)
        
        y = a.copy()
        staging = a.copy()
        t_fista = 1
        count = 0
        
        while time.monotonic() - stage_start < stage_time:
            ac = fast_autocorr(y)
            ac_max = ac.max()
            if ac_max < 1e-15:
                break
            shifted = np.clip((ac - ac_max) / T, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            
            # Gradient of LSE approximation
            # d/dy_j [sum_k w_k * (y*y)_k] = 2 * sum_k w_k * (P_k y)_j
            # where (P_k y)_j = y[k-j]
            # = 2 * (w * y_reversed)_j ... complex
            # Simpler: compute via FFT
            N = 1 << (3 * n - 2).bit_length()
            W = np.fft.rfft(w, N)
            Y = np.fft.rfft(y, N)
            # Cross-correlation: ifft(conj(W) * Y) gives (w * y) correlation
            cross = np.fft.irfft(W * Y, N)
            grad = 2.0 * cross[n-1:n-1+n]
            
            step = 1.0 / L
            z = y - step * grad
            a_new = project_nonneg_sum(z, 1.0)
            
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_nonneg_sum(y_new, 1.0)
            
            staging = a_new
            a = a_new
            y = y_new
            t_fista += 1
            count += 1
            
            if count % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose:
            print(f"    LSE T={T:.2f} ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    return best_a * norm / best_a.sum()

def lp_nesterov_improved(a0, time_limit, solver, verbose=True):
    """LP-Nesterov with adaptive restart and better averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    y = a.copy()
    t = 0
    stall_count = 0
    last_best_t = 0
    
    # Precompute convolution structure
    # T_mat[k, j] = a[k-j] if 0 <= k-j < n
    # We'll rebuild this each iteration since a changes
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        T_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_mat[j:j+n, j] = y
        
        constraints = [T_mat @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check score
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            last_best_t = t
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            last_best_t = t
        
        # Ergodic averaging
        avg_a += a
        n_avg += 1
        
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score - 1e-14:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0
                    last_best_t = t
        
        # Adaptive restart: if stuck for a while, reset momentum
        if stall_count > 200 and (t - last_best_t) > 200:
            # Try a random perturbation
            rng = np.random.default_rng(t)
            a_perturbed = a + rng.normal(0, 0.01 * norm / n, n)
            a_perturbed = project_nonneg_sum(a_perturbed, norm)
            
            sc_pert = fast_autocorr(a_perturbed).max()
            if sc_pert < best_score:
                a = a_perturbed
                y = a_perturbed.copy()
                t = 0  # restart Nesterov
                stall_count = 0
                last_best_t = t
                avg_a = np.zeros(n)
                n_avg = 0
            else:
                # Just reset momentum
                y = a.copy()
                stall_count = 0
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a

        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    staging = a.copy()
    t = 1
    prev_a = a.copy()
    prev_grad = None
    eta = 1.0 / n  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient (subgradient at the peak)
        ac = fast_autocorr(y)
        peak_k = int(np.argmax(ac))
        
        # Subgradient of (a*a)_{peak_k}
        lo = max(0, peak_k - n + 1)
        hi = min(peak_k, n - 1)
        grad = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * y[peak_k - idx]
        
        # BB step size
        if prev_grad is not None and t > 1:
            s_vec = y - prev_a
            y_vec = grad - prev_grad
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if s_dot_s > 1e-15 and abs(s_dot_y) > 1e-15:
                eta_new = s_dot_s / abs(s_dot_y)
                eta = 0.5 * eta + 0.5 * np.clip(eta_new, 0.01 * eta, 10.0 * eta)
        
        # Nesterov step
        z = y - eta * grad
        a_new = project_nonneg_sum(z, norm)
        
        if t == 1:
            y_new = a_new.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            y_new = a_new + mom * (a_new - staging)
            y_new = project_nonneg_sum(y_new, norm)
        
        staging = a_new
        prev_a = y.copy()
        prev_grad = grad.copy()
        y = y_new
        t += 1
        
        if t % 100 == 0:
            sc = fast_autocorr(y).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = y.copy()
    
    sc = fast_autocorr(y).max()
    if sc < best_score:
        best_score = sc
        best_a = y.copy()
    
    return best_a

def targeted_shave(a0, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find low-contribution destinations
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 5:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick a few random candidates
            n_try = min(5, len(candidates))
            dsts = rng.choice(candidates, size=n_try, replace=False)
            
            for dst in dsts:
                for frac in [0.001, 0.003, 0.01, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18 or a[src] < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break
    
    return best_a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
