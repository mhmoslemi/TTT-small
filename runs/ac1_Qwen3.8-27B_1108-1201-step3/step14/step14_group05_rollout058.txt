        # conv1[k] = sum_j w[j] * a_rev[k-j] where a_rev[t] = a[n-1-t]
        # = sum_j w[j] * a[n-1-k+j]
        # For k = i + n - 1: = sum_j w[j] * a[n-1-(i+n-1)+j] = sum_j w[j] * a[j-i]
        # Hmm, that gives sum_j w[j] * a[j-i] which is valid when j-i in [0,n-1]
        # = sum_m w[m+i] * a[m] (m = j-i) ... wait that's the first sum! (with m as the a-index)
        
        # So first_sum_i = conv1[i + n - 1] ... let me double check
        # conv1[k] = sum_j w[j] * a[n-1-k+j]
        # k = i + n - 1: conv1[i+n-1] = sum_j w[j] * a[n-1-i-n+1+j] = sum_j w[j] * a[j-i]
        # For this to be non-zero: 0 <= j-i < n, so j in [i, i+n-1]
        # = sum_{j=i}^{i+n-1} w[j] * a[j-i] = sum_m w[m+i] * a[m]  ✓
        
        # Now for the second sum: sum_m a[m] * w[i-m]
        # = sum_m a[m] * w[i-m] (m from max(1,i-(2n-2)) to min(n-1, i))
        # Let's compute conv2 = conv(w, a) (not reversed)
        # conv2[k] = sum_j w[j] * a[k-j] (valid: 0 <= k-j < n)
        # For k = i: conv2[i] = sum_j w[j] * a[i-j] = sum_m w[i-m] * a[m] (m = i-j) ✓
        
        conv2_w = irfft(rfft(w, fft_sz) * rfft(a, fft_sz), fft_sz)
        # conv2_w[k] = sum_j w[j] * a[k-j]
        # second_sum_i = conv2_w[i]
        
        grad = np.zeros(n)
        grad[:] = conv1[n - 1:n + n - 1] + conv2_w[0:n]
        # grad[i] = first_sum_i + second_sum_i
        
        # Project gradient onto tangent space (sum constraint)
        grad -= np.mean(grad)
        
        # L2 normalize for step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        grad_unit = grad / grad_norm
        
        # Backtracking line search
        best_trial = a.copy()
        best_trial_score = fast_autocorr(a).max()
        step = base_lr
        accepted = False
        
        for ls_iter in range(8):
            s = step
            a_trial = a - s * grad_unit * grad_norm  # = a - s * grad (scaled)
            # Actually let's use: a_trial = a - s * grad (where grad is the actual gradient)
            # And project
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_sum(a_trial, norm)
            
            sc = fast_autocorr(a_trial).max()
            
            # Armijo condition: f(x + s*d) <= f(x) - c * s * ||grad||^2
            # Simplified: accept if sc < max_b * (1 - 0.01)
            if sc < max_b * 0.999:
                a = a_trial
                accepted = True
                step = min(step * 1.5, 2.0 / n)
                break
            else:
                step *= 0.5
        
        if not accepted:
            # Try a very small step
            a_trial = a - (step * 0.01) * grad
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_sum(a_trial, norm)
            sc = fast_autocorr(a_trial).max()
            if sc < max_b:
                a = a_trial
            step = max(step * 0.5, 1e-6 / n)
        
        # Track best
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        steps += 1
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-GD step={steps} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.4f}, ls={step:.2e}")
        
        if max_steps and steps >= max_steps:
            break
    
    return best_a, best_score


def build_conv_matrix_rows(a, indices):
    """Build rows of the convolution matrix for given constraint indices."""
    n = len(a)
    rows = []
    for k in indices:
        row = np.zeros(n)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            row[m_start:m_end + 1] = a[k - np.arange(m_start, m_end + 1)]
        rows.append(row)
    return np.array(rows)

def lp_solve_cvxpy(a, M, solver=None, max_constraints=100):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0
    Using only the top max_constraints active constraints.
    """
    n = len(a)
    size = 2 * n - 1
    
    if solver is None:
        solver = cp.PDLP
    
    # Find which constraints are most likely active
    # Start with b = ones (gives conv values)
    b_init = np.ones(n)
    conv_init = fast_conv(a, b_init)
    
    # Select top constraints (those with highest conv_init / M ratio)
    ratios = conv_init / max(M, 1e-15)
    n_sel = min(max_constraints, size)
    top_idx = np.argsort(ratios)[::-1][:n_sel]
    
    # Build constraint matrix for selected constraints
    T_sel = build_conv_matrix_rows(a, top_idx)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sel @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return np.ones(n) / n
    
    if prob.status not in ('optimal', 'optimal_inaccurate') or b_var.value is None:
        return np.ones(n) / n
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Verify and fix if needed
    conv_check = fast_conv(a, b_val)
    if conv_check.max() > M * 1.01:
        b_val = b_val * M / conv_check.max()
    
    if b_val.sum() < 1e-15:
        return np.ones(n) / n
    
    return b_val


def nesterov_lp(a0, time_limit, solver=None, verbose=True, max_constraints=80):
    """Fessler-Parks accelerated LP-based optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_sum(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Solve LP
        b = lp_solve_cvxpy(y, M_curr, solver=solver, max_constraints=max_constraints)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, norm)
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, stall={stall}")
        
        # Adaptive restart
        if stall > 200:
            rng = np.random.default_rng(n_iters * 31 + 7)
            perturb = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb, n), 0)
            y = normalize_sum(y, norm)
            a = np.maximum(a + rng.normal(0, perturb, n), 0)
            a = normalize_sum(a, norm)
            prev_y = y.copy()
            stall = 0
            gamma = max(1.0, gamma - 0.1)
        
        # Adaptive gamma
        if stall > 100 and gamma > 1.2:
            gamma -= 0.05
    
    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Aggressive peak shaving with multiple move types and simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize_sum(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_it = 0
    n_improve = 0
    T_anneal = best_score * 0.005
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T_anneal = best_score * 0.005 * (1.0 - elapsed_frac ** 2)
        
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Move type 1: Shift mass from peak contributor to non-contributor
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() > 1e-18:
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            
            # Find good destination
            non_contrib = np.ones(n, dtype=bool)
            if js > 0:
                non_contrib[:js] = True
            if je < n - 1:
                non_contrib[je + 1:] = True
            else:
                non_contrib[:js+1] = False
            non_contrib[src] = False
            
            if non_contrib.any():
                # Pick from non-contributing positions with low values
                nc_idx = np.where(non_contrib)[0]
                # Weight by (1 - a[nc_idx]) to prefer empty positions
                weights = 1.0 / (a[nc_idx] + a.mean() * 0.1)
                weights /= weights.sum()
                dst = nc_idx[rng.choice(len(nc_idx), p=weights)]
                
                # Try multiple fractions
                best_sc = max_b
                best_frac = 0
                for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    amt = src_val * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0)
                    a_t = normalize_sum(a_t, norm)
                    sc = fast_autocorr(a_t).max()
                    if sc < best_sc:
                        best_sc = sc
                        best_frac = frac
                
                if best_frac > 0:
                    a_t = a.copy()
                    amt = src_val * best_frac
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0)
                    a_t = normalize_sum(a_t, norm)
                    sc = fast_autocorr(a_t).max()
                    
                    if sc < best_score or (T_anneal > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T_anneal)):
                        a = a_t
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
            n_it += 1
        else:
            # Move type 2: Random pair swap
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.1, 0.4)
                amt = a[i1] * frac
                a_t = a.copy()
                a_t[i1] -= amt
                a_t[i2] += amt
                a_t = np.maximum(a_t, 0)
                a_t = normalize_sum(a_t, norm)
                sc = fast_autocorr(a_t).max()
                if sc < best_score or (T_anneal > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T_anneal)):
                    a = a_t
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
            n_it += 1
        
        # Move type 3 (every 50 steps): Multi-point perturbation
        if n_it % 50 == 0:
            n_pert = max(3, n // 100)
            pert_idx = rng.choice(n, size=n_pert * 2, replace=False)
            a_t = a.copy()
            for pp in range(n_pert):
                src = pert_idx[pp]
                dst = pert_idx[pp + n_pert]
                if src != dst and a_t[src] > 1e-10:
                    amt = a_t[src] * rng.uniform(0.1, 0.3)
                    a_t[src] -= amt
                    a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            a_t = normalize_sum(a_t, norm)
            sc = fast_autocorr(a_t).max()
            if sc < best_score * 1.01:
                a = a_t
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
        
        # Periodic large restart
        if n_it % 2000 == 0:
            rng2 = np.random.default_rng(n_it)
            n_large = max(10, n // 20)
            a_t = a.copy()
            for _ in range(n_large):
                i1 = rng2.integers(n)
                i2 = rng2.integers(n)
                if i1 != i2 and a_t[i1] > 1e-10:
                    amt = a_t[i1] * rng2.uniform(0.2, 0.5)
                    a_t[i1] -= amt
                    a_t[i2] += amt
            a_t = np.maximum(a_t, 0)
            a_t = normalize_sum(a_t, norm)
            sc = fast_autocorr(a_t).max()
            if sc < best_score * 1.05:
                a = a_t
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def get_solver():
    """Get a working LP solver."""
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.PDLP


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining LSE gradient descent, LP-Nesterov,
    and peak shaving for minimizing the autocorrelation ratio.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    
    solver = get_solver()
    print(f"Solver: {solver}, Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 2.0
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    # === Phase 1: LSE Gradient Descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 1: LSE-GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_cur}")
    
    # === Phase 2: LP-Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver=solver, verbose=True, max_constraints=80)
        try_update(a_out, f"lp1_n{n_cur}")
    
    # === Phase 3: Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: PeakShave n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t3, seed=12345, verbose=True)
        try_update(a_out, f"peak1_n{n_cur}")
    
    # === Phase 4: LSE-GD round 2 (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LSE-GD r2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        T_init = fast_autocorr(a_norm).max() * 0.03
        T_min = fast_autocorr(a_norm).max() * 0.001
        a_out, _ = lse_gradient_descent(a_norm, t4, verbose=True, T_init=T_init, T_min=T_min)
        try_update(a_out, f"lse2_n{n_cur}")
    
    # === Phase 5: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    for n_try in [n_base + 50, n_base - 50, n_base + 100]:
        if n_try < 100 or n_try > 3000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        t5 = min(remaining * 0.10, 150.0)
        if t5 < 30:
            break
        print(f"\n=== Phase 5: LSE-GD n={n_try} (budget={t5:.0f}s) ===")
        init_interp = interpolate_sequence(best_seq, n_try)
        a_norm = init_interp * np.sqrt(2.0 * n_try) / init_interp.sum()
        a_out, _ = lse_gradient_descent(a_norm, t5, verbose=False)
        try_update(a_out, f"lse_n{n_try}")
    
    # === Phase 6: LP-Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LP-Nesterov r2 n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver=solver, verbose=True, max_constraints=60)
        try_update(a_out, f"lp2_n{n_cur}")
    
    # === Phase 7: Final Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.15, 180.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final PeakShave n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t7, seed=99999, verbose=True)
        try_update(a_out, f"peak_final_n{n_cur}")
    
    # === Phase 8: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = max(remaining * 0.8, 10.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LSE polish n={n_cur} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        M = fast_autocorr(a_norm).max()
        T_init = M * 0.01
        T_min = M * 0.0005
        a_out, _ = lse_gradient_descent(a_norm, t8, verbose=True, T_init=T_init, T_min=T_min)
        try_update(a_out, f"lse_final_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
