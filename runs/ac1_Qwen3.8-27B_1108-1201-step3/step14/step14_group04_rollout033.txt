    """Solve the soft max problem: max sum(b) - (1/(2*eta))*sum_k max(0, (Tb)_k - M)^2
    using projected gradient descent with FFT-accelerated gradient."""
    n = len(a)
    b = np.ones(n)  # initial guess
    
    # Precompute FFT size
    fft_size = 1 << (2*n-1).bit_length()
    
    for iter in range(max_iters):
        # Compute r = max(0, T@b - M)
        Tb = fast_conv(a, b)  # O(n log n)
        r = np.maximum(Tb - M, 0)
        
        # Compute gradient: 1 - (1/eta) * T^T @ r
        # T^T @ r: (T^T r)_i = sum_k r_k * a[k-i] for valid indices
        # This is the cross-correlation of r and a
        Tt_r = cross_corr(r, a, n)  # O(n log n)
        
        grad = np.ones(n) - (1.0/eta) * Tt_r
        
        # Gradient step
        step = eta / (1.0 + np.linalg.norm(Tt_r))  # adaptive step
        b_new = b + step * grad
        b_new = np.maximum(b_new, 0)  # project to non-negative
        
        # Check convergence
        if np.max(np.abs(b_new - b)) < tol:
            b = b_new
            break
        b = b_new
    
    return b

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Compute linear convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def project_nonneg_sum(v, target_sum):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return v * target_sum / s


def lse_grad(y, T, n_accel=1):
    """Compute gradient of LSE approximation of max(conv(y,y))/sum(y)^2."""
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    # Gradient of sum_k w_k * ac_k w.r.t. y
    # ac_k = sum_i y[i]*y[k-i], so d(ac_k)/dy[j] = sum_i [delta(j,i)*y[k-i] + delta(j,k-i)*y[i]]
    # = y[k-j] + y[j-k] (for valid indices)
    # Total grad[j] = sum_k w_k * (y[k-j] + y[j-k] (if j-k in range))
    # This is: grad = 2 * cross_corr(w, y) (approximately, for the valid range)
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y = rfft(y, N)
    # cross: sum_k w[k] * y[k-j] for k-j in [0,n-1]
    cross1 = irfft(W * Y, N)  # conv(w, y)
    # cross2: sum_k w[k] * y[j-k] = sum_k w[k] * y[|k-j|]... 
    # Actually for symmetric handling: conv(w, y)[j] = sum_k w[k]*y[j-k] (one direction)
    # And conv(y, w)[j] = sum_k y[k]*w[j-k] = sum_k w[k]*y[j-k]... 
    # Since conv is commutative: conv(w,y) = conv(y,w)
    # So the gradient is: grad[j] = sum_k w[k] * (y[k-j] + y[j-k])
    # The first term: sum_k w[k]*y[k-j] = (conv of w with y, reversed) at position j
    # = conv(w, y)[2n-2-j] ... let me just use the full convolution
    # Actually, let's use a simpler approach:
    # grad[j] = 2 * sum_{i: j <= k <= j+n-1} w[k] * y[k-j] (for the k direction)
    # + 2 * sum_{i: j-n+1 <= k <= j} w[k] * y[j-k] (for the -k direction)
    # But since w is symmetric-ish (it's based on autocorrelation which is symmetric),
    # and y is what we're differentiating w.r.t., we can approximate:

    grad = 2.0 * cross_corr(w, y, n)
    return grad

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    # Get dual variables
    dual_vals = None
    try:
        dual_vals = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, dual_vals

            avg_check = project_nonneg_sum(avg_check, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score_unnorm - 1e-10:
                best_score_unnorm = score_avg
                best_a = avg_check.copy()
                stall_counter = 0
                last_improvement = best_score_unnorm

        # Adaptive restart: if stalled too long, perturb
        if stall_counter > patience:
            # Large perturbation
            perturbation = rng.random(n) * 0.1
            y_pert = y + perturbation * norm
            y_pert = project_nonneg_sum(y_pert, norm)
            y = y_pert
            a = y_pert.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall_counter = 0
            if verbose:
                print(f"    [restart] iter={n_iters}, perturb applied")

        if verbose and n_iters % 100 == 0:
            elapsed = time.time() - t0
            cur_score = 2 * n * best_score_unnorm / norm**2
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): score={cur_score:.6f}")

    return best_a, best_score_unnorm


def freq_domain_refine(a0, time_limit, seed=42):
    """Novel frequency-domain refinement: reduce peak lags by modifying spectrum.
    
    Key idea: the peak autocorrelation lag k gets contributions from all frequency
    bins. By reducing the magnitude of bins that constructively interfere at the
    peak lag, we can reduce the peak while preserving the DC component (sum).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.time()
    stall = 0

    # FFT size
    pad_len = 1 << (2 * n - 1).bit_length()

    while time.time() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_lag = np.argmax(b)
        peak_val = b[peak_lag]
        
        # Compute spectrum
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = rfft(a_pad)
        power = np.abs(A)**2
        
        # For the peak lag, the contribution at frequency f is power[f] * exp(2*pi*i*f*peak_lag/pad_len)
        # To reduce the peak, we want to reduce power at frequencies where the phasor
        # aligns with the peak direction
        # Phase at frequency k for lag peak_lag: 2*pi*k*peak_lag/pad_len
        freqs = np.arange(len(A))
        phases = 2.0 * np.pi * freqs * peak_lag / pad_len
        # Contribution to peak from each bin: power[k] * cos(phases[k]) (real part for real signal)
        # We want to reduce bins where cos(phases[k]) > 0 (constructive for peak)
        
        # Find top contributing bins (excluding DC)
        contrib = power * 0.5 * (1 + np.cos(phases))  # weighted by constructive alignment
        contrib[0] = 0  # don't touch DC
        
        # Select top k bins to reduce
        top_k = min(8, len(A) // 4)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        
        reduced = False
        for reduce_frac in [0.01, 0.005, 0.002]:
            A_mod = A.copy()
            for ki in top_idx:
                if ki == 0:
                    continue
                A_mod[ki] *= (1.0 - reduce_frac)
            
            a_new_full = irfft(A_mod, pad_len)
            a_new = a_new_full[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new
            score = fast_autocorr(a_new).max()
            if score < best_score - 1e-12:
                a = a_new
                best_score = score
                best_a = a.copy()
                stall = 0
                reduced = True
                break
        
        if not reduced:
            # Try random small perturbations in frequency domain
            ki = rng.integers(1, len(A) // 2)
            for frac in [0.005, 0.01, 0.02]:
                A_mod = A.copy()
                A_mod[ki] *= (1.0 - frac)
                a_new_full = irfft(A_mod, pad_len)
                a_new = a_new_full[:n]
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new = a_new * norm / s_new
                score = fast_autocorr(a_new).max()
                if score < best_score - 1e-12:
                    a = a_new
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    reduced = True
                    break
        
        if not reduced:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score


def local_search_refine(a0, time_limit, seed=777):
    """Targeted local search: move mass from positions that contribute to peaks
    to positions that don't. Uses autocorrelation structure to guide moves."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.time()
    stall = 0

    while time.time() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:4]
        improved = False

        for k in top_peaks:
            # Positions contributing to this peak: i and k-i for valid i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find good destination: position not contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            # Prefer low-mass positions
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(3, len(candidates_sorted) // 10)
            dst_pool = candidates_sorted[:n_try]

            for dst in dst_pool:
                for frac in [0.001, 0.003, 0.01, 0.03, 0.07]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if not improved:
                stall += 1
                if stall > 2000:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization for minimum autocorrelation peak."""
    rng = np.random.default_rng(seed)
    t_global = time.time()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.time() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE-FISTA warm-up on parent length ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    t1 = min(remaining * 0.05, 45.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_best} (budget={t1:.0f}s) ===")
        T_sched = [30.0, 10.0, 3.0, 1.0, 0.3]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_best}")

    # === Phase 2: Multi-n LP exploration ===
    remaining = time_budget - (time.time() - t_global)
    n_values = [len(best_seq)]
    # Add nearby lengths
    for delta in [-200, -100, 100, 200, 300, 400]:
        nn = len(best_seq) + delta
        if 800 <= nn <= 2400 and nn not in n_values:
            n_values.append(nn)
    n_values.sort()
    
    t_per_n = min(remaining * 0.25 / max(1, len(n_values)), 40.0)
    print(f"\n=== Phase 2: Multi-n LP (budget per n={t_per_n:.0f}s, n_values={n_values}) ===")
    
    for nn in n_values:
        if time.time() - t_global >= time_budget * 0.30:
            break
        init = interpolate_sequence(best_seq, nn)
        a_norm = init * np.sqrt(2.0 * nn) / init.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t_per_n, solver, verbose=False, patent=100)
        try_update(a_out, f"lp_explore_n{nn}")

    # === Phase 3: Main LP-Nesterov (long run) ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.40, 550.0)
    if t3 > 60:
        print(f"\n=== Phase 3: Main LP-Nesterov n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t3, solver, verbose=True, patience=300)
        try_update(a_out, f"lp_main_n{n_best}")

    # === Phase 4: Frequency-domain refinement ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Freq-domain refine n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = freq_domain_refine(a_norm, t4, seed=42)
        try_update(a_out, f"freq_n{n_best}")

    # === Phase 5: Local search ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Local search n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_search_refine(a_norm, t5, seed=777)
        try_update(a_out, f"local_n{n_best}")

    # === Phase 6: Second LP round on best ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP round 2 n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t6, solver, verbose=True, patience=200)
        try_update(a_out, f"lp_r2_n{n_best}")

    # === Phase 7: Try a different n (longer) ===
    remaining = time_budget - (time.time() - t_global)
    n2 = len(best_seq) + 200
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 30:
        print(f"\n=== Phase 7: LP at n={n2} (budget={t7:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t7, solver, verbose=False, patience=150)
        try_update(a_out, f"lp_long_n{n2}")

    # === Phase 8: Final local search ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.08, 60.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final local search n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_search_refine(a_norm, t8, seed=99999)
        try_update(a_out, "final_local")

    # === Phase 9: Final LP if time remains ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.6, 90.0)
        if t9 > 15:
            print(f"\n=== Phase 9: Final LP n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov_enhanced(a_norm, t9, solver, verbose=False, patience=100)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.time()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
