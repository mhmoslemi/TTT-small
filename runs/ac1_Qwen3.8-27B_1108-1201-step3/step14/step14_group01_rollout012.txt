        cross = fast_conv_cross(a, val)
        violated = np.where(cross > M * (1 + 1e-6))[0]
        if len(violated) == 0:
            return val

        # Add most violated constraints (up to 50 at a time)
        viol_idx = violated[np.argsort(cross[violated])[::-1]]
        top_viol = viol_idx[:50]
        active = list(set(active + list(top_viol)))
        if len(active) >= size:
            active = list(range(size))
            T_full = build_T_rows(a, active)
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_full @ b_var <= M])
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                return val
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                val2 = np.asarray(b_var.value, dtype=np.float64)
                return np.maximum(val2, 0.0)
            return val

    return val

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(80.0, 0.2, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 6)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for trial in range(8):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= conv.max() + 0.05 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    a_prev = a.copy()
    size = 2 * n - 1

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 30:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint selection: use top constraints
        threshold = M * 0.90
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 30:
            threshold = M * 0.70
            active = np.where(conv_y >= threshold)[0]
        if len(active) < 10:
            active = np.argsort(conv_y)[::-1][:50]

        b = lp_solve_constraint_gen(y, M, solver, max_iter=4, init_active=active)
        if b is None:
            b = lp_solve_constraint_gen(y, M, solver, max_iter=2, init_active=None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a_prev = a
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Gradient-informed partner selection
            # For coordinate i, the gradient contribution to conv[k] is a[k-i]+a[i-k]
            # We want to find j that, when we transfer from i to j, reduces the max most
            grad_i = np.zeros(size)
            lo = max(0, i)
            hi = min(size, i + n)
            if hi > lo:
                grad_i[lo:hi] = a[0:hi - lo]
            lo2 = max(0, i - n + 1)
            hi2 = min(size, i + 1)
            if hi2 > lo2:
                grad_i[lo2:hi2] += a[i - np.arange(lo2, hi2)]

            # Find j that best reduces peaks
            peak_lags = np.where(conv >= cur_max * 0.98)[0]
            j_candidates = set()
            for _ in range(12):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                # j that maximally reduces conv[kl]: we want a[kl-j]+a[j-kl] to be large
                # while a[kl-i]+a[i-kl] is large (to subtract)
                # Approximate: j near kl (since a[0] + a[kl] would be added)
                j_struct = kl % n
                j_candidates.add(j_struct)
                j_struct2 = (kl + i) % n
                j_candidates.add(j_struct2)
                j_struct3 = abs(kl - i) % n
                j_candidates.add(j_struct3)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for transferring delta from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta via sampling + refinement
                n_samples = 25
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d < 1e-15:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at i+j and j-i
                    if i + j < size:
                        new_conv[i + j] = conv[i + j] + d * (a[0] + a[0])  # approximation
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def peak_equalization(a0, time_limit, verbose=True):
    """Try to equalize the top autocorrelation peaks by directly adjusting the sequence."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < 1e-15:
            break

        # Find top peaks
        top_k = min(20, size // 10)
        peak_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[peak_idx]
        mean_peak = peak_vals.mean()
        spread = peak_vals.std()

        if spread < 1e-8 * cur_max:
            break

        # Compute subgradient: weighted average of gradients at active peaks
        weights = np.exp((peak_vals - cur_max) / (0.1 * cur_max + 1e-15))
        weights /= weights.sum()

        grad = np.zeros(n)
        for w_k, k in zip(weights, peak_idx):
            for i in range(n):
                # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with bounds)
                v1 = a[k - i] if 0 <= k - i < n else 0.0
                v2 = a[i - k] if 0 <= i - k < n else 0.0
                grad[i] += w_k * (v1 + v2)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search
        lr = norm / (gnorm * np.sqrt(n))
        found = False
        for trial in range(15):
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                lr *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < cur_max - 1e-14:
                found = True
                break
            lr *= 0.5

        if found:
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and outer % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq iter={outer} ({elapsed:.0f}s): best={best_score:.8f}, spread={spread:.6f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.exp(-5.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-5.0 * (x - 0.8) ** 2)
    else:
        a = (1.0 + np.cos(2 * np.pi * x)) ** 0.8
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.6f}")

    # Phase 1: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_base}")

    # Phase 3: Peak equalization
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 3: Peak Eq n={n_base} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_equalization(best_seq, t3, verbose=True)
        try_update(a_out, f"peq_n{n_base}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 4: Pair CD n={n_base} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd1_n{n_base}")

    # Phase 5: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n_base} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_base}")

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.35, 200.0)
    print(f"\n=== Phase 6: Longer length n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 40 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t6a = min(t6 * 0.5, 80.0)
        a_out, _ = lse_gradient_descent(init_long, t6a, verbose=False)
        try_update(a_out, f"lse_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6b = min(remaining * 0.6, 100.0)
            a_out2, _ = nesterov_lp_optimize(best_seq, t6b, solver, verbose=False)
            try_update(a_out2, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out3, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 60.0), rng, verbose=False)
            try_update(a_out3, f"pcd_long_{n_long}")

    # Phase 7: Final LSE + peak eq polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t7a = min(remaining * 0.4, 60.0)
        a_out, _ = lse_gradient_descent(best_seq, t7a, verbose=False)
        try_update(a_out, f"lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.4, 60.0)
            a_out2, _ = peak_equalization(best_seq, t7b, verbose=False)
            try_update(a_out2, f"peq_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out3, _ = pair_cd_optimize(best_seq, min(remaining * 0.8, 50.0), rng, verbose=False)
            try_update(a_out3, f"pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
