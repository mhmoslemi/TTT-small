def peak_gradient_descent(a, time_limit, step_size=None):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute: for each k, the vector 2*a[k-i] for i=0..n-1
    # This is just 2*a reversed and padded
    
    lr = 0.01  # initial learning rate
    
    for iteration in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        k_max = c.argmax()
        
        # Descent direction (unconstrained): d[i] = -2*a[k_max - i] (with a[j]=0 outside [0,n-1])
        # a[k_max - i] is valid when 0 <= k_max - i < n, i.e., k_max - n + 1 <= i <= k_max
        d = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        d[lo:hi+1] = -2.0 * a[k_max - hi : k_max - lo - 1 : -1]
        # Wait, let me be more careful.
        # d[i] = -2 * a[k_max - i] for i where 0 <= k_max - i < n
        # Let j = k_max - i, so i = k_max - j, j in [0, n-1]
        # i ranges from k_max - (n-1) to k_max, intersected with [0, n-1]
        # d[k_max - j] = -2 * a[j] for j = 0, ..., n-1
        # So d is -2*a reversed and placed at positions [k_max-n+1, ..., k_max] (clipped to [0,n-1])
        
        # Simpler:
        d = np.zeros(n)
        for j in range(n):
            i = k_max - j
            if 0 <= i < n:
                d[i] = -2.0 * a[j]
        
        # Project d onto the tangent space of the simplex: d - mean(d)
        d = d - d.mean()
        
        # Line search
        best_step = lr
        best_val = best_peak
        for step in [lr, lr*0.5, lr*0.25, lr*2, lr*4]:
            a_trial = a + step * d
            # Project onto simplex
            a_trial = project_simplex(a_trial)
            c_trial = fast_autocorr(a_trial)
            peak_trial = c_trial.max()
            if peak_trial < best_val:
                best_val = peak_trial
                best_step = step
        
        a = a + best_step * d
        a = project_simplex(a)
        
        peak = fast_autocorr(a).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
    
    return best_a, best_peak

def fast_autocorr(a):

def pnorm_autocorr(a, p):
    """Compute F_p(a) = (sum(c_k^p)/(2n-1))^(1/p) and its gradient."""
    n = len(a)
    c = fast_autocorr(a)
    L = len(c)  # 2n-1
    if c.max() < 1e-15:
        return 0.0, np.zeros(n)
    cp = c ** (p - 1)
    Fp = (c ** p).sum() ** (1.0 / p) / L ** (1.0 / p)
    # Actually: Fp = (sum c_k^p / L)^(1/p)
    Fp = (np.sum(c ** p) / L) ** (1.0 / p)
    
    # Gradient: dFp/da[i] = (2 * Fp^(1-p) / L) * sum_k c_k^(p-1) * a[k-i]
    # Compute sum_k c_k^(p-1) * a[k-i] via FFT
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(cp, N)
    A = rfft(a, N)
    cross = irfft(W * np.conj(A), N)
    grad = (2.0 * Fp ** (1 - p) / L) * cross[:n]
    return Fp, grad

def peak_gradient_direction(a, c, k_max, n_peaks=1):
    """
    Compute the gradient direction for the peak at k_max.
    If n_peaks > 1, use weighted average of top peaks.
    Returns the descent direction (to be subtracted from a).
    """
    n = len(a)
    # For the peak at k_max: grad[i] = 2*a[k_max - i] (with a[j]=0 outside [0,n-1])
    # We want the descent direction, so: d = -grad (projected onto simplex tangent)
    
    if n_peaks == 1:
        d = np.zeros(n)
        for j in range(n):
            i = k_max - j
            if 0 <= i < n:
                d[i] = -2.0 * a[j]
    else:
        # Use top n_peaks peaks with weights based on their values
        top_k = np.argsort(c)[::-1][:n_peaks]
        weights = c[top_k]
        w_sum = weights.sum()
        if w_sum < 1e-15:
            return np.zeros(n)
        weights = weights / w_sum
        d = np.zeros(n)
        for w, k in zip(weights, top_k):
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    d[i] -= 2.0 * w * a[j]
    
    # Project onto tangent space of simplex (subtract mean)
    d = d - d.mean()
    return d

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def phase2_pnorm_lbfgs(a0, time_limit, p=30):

    n = len(a0)
    a_init = project_simplex(np.asarray(a0, dtype=np.float64))
    
    # Work in reduced space: x = a[:n-1], a[n-1] = 1 - sum(x)
    x0 = a_init[:n-1].copy()
    
    def obj_func(x):
        a = np.empty(n)
        a[:n-1] = x
        a[n-1] = 1.0 - x.sum()
        if a[n-1] < -1e-10:
            return 1e10
        a = np.maximum(a, 0)
        a = a / a.sum()  # renormalize for safety
        Fp, _ = pnorm_autocorr(a, p)
        return Fp
    
    def grad_func(x):
        a = np.empty(n)
        a[:n-1] = x
        a[n-1] = 1.0 - x.sum()
        if a[n-1] < -1e-10:
            return np.zeros(n-1)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.zeros(n-1)
        a = a / s
        Fp, grad_a = pnorm_autocorr(a, p)
        # Chain rule: dF/dx_i = grad_a[i] * 1 + grad_a[n-1] * (-1) = grad_a[i] - grad_a[n-1]
        grad_x = grad_a[:n-1] - grad_a[n-1]
        return grad_x
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    t0 = time.monotonic()

        x0 = best_a[:n-1].copy()
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, 1)] * (n-1),
                                  options={'maxiter': 500, 'maxfun': 2000,
                                           'time_limit': stage_time,
                                           'gtol': 1e-8})
            if res.x is not None:
                a_cand = np.empty(n)
                a_cand[:n-1] = np.maximum(res.x, 0)
                a_cand[n-1] = max(0.0, 1.0 - a_cand[:n-1].sum())
                s_c = a_cand.sum()

            print(f"    L-BFGS p={p_val} failed: {e}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
