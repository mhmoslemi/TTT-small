def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix (2n-1) x n where row k is the k-th row of the convolution operator
    # conv(a,b)[k] = sum_j a[j] * b[k-j]
    # This is a lower triangular Toeplitz-like structure
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:

    return b_val

                    improved = True
                    break

        elif strategy == 2:
            # Targeted: reduce peak lag contribution
            ac = fast_autocorr(a)
            peak_k = np.argmax(ac)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contrib = a[idx] * a[peak_k - idx]
                if contrib.max() > 1e-20:
                    src = idx[np.argmax(contrib)]
                    dst = rng.integers(n)
                    if dst != src:
                        for frac in [0.001, 0.003, 0.01, 0.03]:
                            amt = a[src] * frac
                            if a[src] - amt < 0:
                                continue
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            s_t = a_trial.sum()
                            if s_t < 1e-15:
                                continue
                            a_trial = a_trial * norm / s_t
                            sc = fast_autocorr(a_trial).max()
                            if sc < best_score - 1e-16:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break

        else:
            # Multi-peak: try reducing top 3 peaks simultaneously
            ac = fast_autocorr(a)
            top_peaks = np.argsort(ac)[::-1][:3]
            for pk in top_peaks:
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                if len(idx) == 0:
                    continue
                contrib = a[idx] * a[pk - idx]
                if contrib.max() < 1e-20:
                    continue
                src = idx[np.argmax(contrib)]
                # Move to a position that doesn't contribute to top peaks
                in_peak = set()
                for pk2 in top_peaks:
                    js2 = max(0, pk2 - n + 1)
                    je2 = min(n - 1, pk2)
                    in_peak.update(range(js2, je2 + 1))
                candidates = np.array([x for x in range(n) if x not in in_peak])
                if len(candidates) == 0:
                    candidates = np.array([x for x in range(n) if x != src])
                if len(candidates) == 0:
                    continue
                dst = candidates[rng.integers(len(candidates))]
                amt = a[src] * 0.01
                if a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score - 1e-16:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def try_update_best(best_state, a, label, t_global):
    """Update best state if a is better. best_state is a dict with 'seq' and 'score'."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: LP-Nesterov with LSE-FISTA warm-up, local polish,
    and multi-length exploration. Optimized for maximizing time spent on
    the most effective method (LP-Nesterov).
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_state = {'seq': None, 'score': float('inf')}
    parent = None

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        init_seq = np.sin(np.pi * x) ** 1.5
        init_seq = np.maximum(init_seq, 0.0)
        init_seq = init_seq / init_seq.sum()
        best_state['seq'] = init_seq
        best_state['score'] = evaluate_safe(init_seq)
        print(f"Default init: n={n_base}, score={best_state['score']:.6f}")
    else:
        n_base = len(best_state['seq'])

    # === Phase 1: Short LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 25.0)
    if t1 > 8:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [5.0, 1.5, 0.5]
        a_out, _ = lse_fista_run(best_state['seq'], T_sched, t1, True)
        try_update_best(best_state, a_out, f"lse_n{n_base}", t_global)

    # === Phase 2: Main LP-Nesterov (the bulk) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    t2 = min(remaining * 0.55, 1000.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_state['seq'] * np.sqrt(2.0 * n_best) / best_state['seq'].sum()
        a_out, _ = lp_nesterov_run(a_norm, t2, solver, True, 5)
        try_update_best(best_state, a_out, f"lp_main_n{n_best}", t_global)

    # === Phase 3: Local polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Local polish n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_state['seq'] * np.sqrt(2.0 * n_best) / best_state['seq'].sum()
        a_out, _ = local_polish_run(a_norm, t3, 12345)
        try_update_best(best_state, a_out, f"polish1_n{n_best}", t_global)

    # === Phase 4: LP polish (second pass) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP polish n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_state['seq'] * np.sqrt(2.0 * n_best) / best_state['seq'].sum()
        a_out, _ = lp_nesterov_run(a_norm, t4, solver, True, 3)
        try_update_best(best_state, a_out, f"lp_polish_n{n_best}", t_global)

    # === Phase 5: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_state['seq']) + 100
    t5 = min(remaining * 0.10, 130.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP at n={n2} (budget={t5:.0f}s) ===")
        init_long = interpolate_sequence(best_state['seq'], n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_run(a_norm, t5, solver, True, 5)
        try_update_best(best_state, a_out, f"lp_n{n2}", t_global)

    # === Phase 6: Final local polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    t6 = min(remaining * 0.10, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final polish n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_state['seq'] * np.sqrt(2.0 * n_best) / best_state['seq'].sum()
        a_out, _ = local_polish_run(a_norm, t6, 99999)
        try_update_best(best_state, a_out, "final_polish", t_global)

    # === Phase 7: Remaining time -> LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    if remaining > 30:
        t7 = min(remaining * 0.85, 250.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_state['seq'] * np.sqrt(2.0 * n_best) / best_state['seq'].sum()
            a_out, _ = lp_nesterov_run(a_norm, t7, solver, False, 3)
            try_update_best(best_state, a_out, "lp_final", t_global)

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_state['seq']]
