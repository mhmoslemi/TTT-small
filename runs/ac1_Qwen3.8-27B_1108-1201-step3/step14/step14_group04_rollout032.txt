def fast_autocorr(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):

    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_fista(a0, T_schedule, time_limit, verbose=False):

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 30.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 8.0), n / (T * 1.5))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            val, grad = lse_objective_and_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 200 == 0:

        if verbose:

def lp_best_response(a_fixed, M, solver, b_prev=None):
    """Solve: max sum(b) s.t. (a_fixed * b)_k <= M for all k, b >= 0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True, value=b_prev)
    # Build the convolution constraint matrix
    # (a_fixed * b)_k = sum_j a_fixed[j] * b[k-j]
    # For each k in [0, 2n-2]: sum_j a_fixed[j] * b[k-j] <= M
    # where k-j is in [0, n-1]
    # This can be written as T @ b <= M where T is (2n-1) x n
    # T[k, j] = a_fixed[j] if k-j is in [0, n-1], else 0
    # i.e., T[k, j] = a_fixed[j] if j is in [max(0, k-n+1), min(n-1, k)]
    
    # For efficiency, use the structure: T is a "banded" Toeplitz-like matrix
    # T[k] = [0...0, a[k-(n-1)], ..., a[k], ..., 0...0] (shifted version of a)
    
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a_fixed
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    step = 1.0 / n
    patience = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 500 == 0:
            T = max(0.01, T * 0.95)  # Anneal temperature
        
        val, grad = lse_objective_and_grad(a, T)
        
        # Backtracking line search
        step_trial = step
        improved = False
        for ls in range(20):
            a_new = a - step_trial * grad
            a_new = project_simplex(a_new, 1.0)
            val_new, _ = lse_objective_and_grad(a_new, T)
            if val_new < val - 1e-14:
                a = a_new
                improved = True
                break
            step_trial *= 0.5
        
        if improved:
            step = min(step * 1.1, 1.0 / n * 10)
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        else:
            step *= 0.5
            patience += 1
        
        if patience > 200:
            break
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            sc = fast_autocorr(a).max()
            print(f"    LSE polish it={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
