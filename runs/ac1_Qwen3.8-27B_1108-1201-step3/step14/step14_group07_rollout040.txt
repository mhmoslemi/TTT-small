"""

1. Nesterov-accelerated mirror descent (entropic) with softmax-smoothed subgradient
2. LP best-response for initial basin exploration
3. Adaptive temperature for subgradient smoothing
4. Coordinate descent polish targeting peak lags

Key innovation: FFT-based smoothed subgradient enables 50x faster per-iteration

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smoothed_subgrad_fft(a, w):
    """
    Compute h[j] = sum_k w[k] * a[k-j] for j=0..n-1.
    w has length 2*n-1, a has length n.
    Uses FFT for efficiency.
    """
    n = len(a)
    size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, size)
    W = rfft(w, size)
    h = irfft(np.conj(W) * A, size)[:n]
    return h


def lp_best_response(a, M, solver):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    """Nesterov LP-based best response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_conv(y)
        M_curr = c_y.max()
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stall += 1
            if stall > 100:
                break
            continue

        b = b / b.sum()
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        S = y_new.sum()
        if S < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / S

        if t > 2:
            momentum = (t - 1.0) / (t + 2.0)
            a_nest = y_new + momentum * (y_new - y)
            a_nest = np.maximum(a_nest, 0.0)
            S_a = a_nest.sum()
            if S_a > 1e-15:
                a_nest = a_nest / S_a
        else:
            a_nest = y_new.copy()

        y = y_new

        M_y = fast_conv(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        M_a = fast_conv(a_nest).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a_nest.copy()
            stall = 0

        if verbose and t % 100 == 0:
            print(f"    LP iter={t}, M={best_M:.8f}, t={time.monotonic()-t0:.1f}s")

        if stall > 200:
            break

    return best_a, best_M


def nesterov_mirror_descent(a0, time_limit, T_start=50.0, T_end=1.0,
                            n_stages=8, verbose=False, use_nesterov=True):
    """
    Nesterov-accelerated mirror descent with softmax-smoothed subgradient.
    Works on simplex: sum(a)=1, a>=0.
    Objective: minimize max(conv(a,a)) (equivalent to minimizing 2n*max_b since sum=1).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    global_t = 0
    y = a.copy()
    y_prev = a.copy()
    stall = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        stage_limit = min(time_limit, time_limit / max(1, n_stages - si))

        for t in range(1, 2000000):
            elapsed = time.monotonic() - t0
            if elapsed > stage_limit:
                break

            c = fast_conv(y)
            max_b = c.max()
            if max_b < 1e-18:
                break

            # Softmax weights over lags
            w = np.exp(T * (c - max_b))
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(c))
                w[np.argmax(c)] = 1.0
            else:
                w = w / w_sum

            # Compute smoothed subgradient via FFT
            h = smoothed_subgrad_fft(y, w)

            # Mirror descent update (exponential weighting):
            # a_new[j] ∝ a[j] * exp(-alpha * h[j])
            # with adaptive step size
            base_alpha = 4.0 / (t + 2.0)
            best_trial = None
            best_trial_obj = np.inf

            for mult in [0.5, 1.0, 2.0, 4.0]:
                alpha = base_alpha * mult
                log_a = np.log(np.maximum(y, 1e-300)) - alpha * h
                # Softmax to normalize
                log_a_shifted = log_a - log_a.max()
                a_trial = np.exp(log_a_shifted)
                a_trial = a_trial / a_trial.sum()

                obj = fast_conv(a_trial).max()
                if obj < best_trial_obj:
                    best_trial_obj = obj
                    best_trial = a_trial

            if best_trial is None:
                break

            y_new = best_trial

            # Nesterov extrapolation
            if use_nesterov and global_t > 2:
                momentum = (global_t) / (global_t + 2.0)
                a_nest = y_new + momentum * (y_new - y)
                a_nest = np.maximum(a_nest, 0.0)
                S_a = a_nest.sum()
                if S_a > 1e-15:
                    a_nest = a_nest / S_a
                # Check Nesterov point
                M_nest = fast_conv(a_nest).max()
                if M_nest < best_M:
                    best_M = M_nest
                    best_a = a_nest.copy()
                    stall = 0

            y_prev = y
            y = y_new
            global_t += 1

            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
                stall = 0
            else:
                stall += 1

            if verbose and t % 50000 == 0:
                print(f"      MD stage={si} T={T:.1f} iter={t} M={best_M:.8f} t={elapsed:.1f}s")

            if stall > 50000:
                break

    return best_a, best_M


def nesterov_additive_subgrad(a0, time_limit, T_start=20.0, T_end=0.5,

        for t in range(1, 3000000):
            elapsed = time.monotonic() - t0
            if elapsed > stage_limit:
                break

            max_b = c.max()
            if max_b < 1e-18:

            if global_t > 2:
                momentum = global_t / (global_t + 2.0)
                a_nest = y_new + momentum * (y_new - y)
                a_nest = np.maximum(a_nest, 0.0)
                S_a = a_nest.sum()
                if S_a > 1e-15:
                    a_nest = a_nest / S_a
                    M_nest = fast_conv(a_nest).max()
                    if M_nest < best_M:
                        best_M = M_nest
                        best_a = a_nest.copy()
                        stall = 0

            y = y_new
            global_t += 1

            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
                stall = 0
            else:
                stall += 1

            if verbose and t % 50000 == 0:
                print(f"      ASG stage={si} T={T:.1f} iter={t} M={best_M:.8f} t={elapsed:.1f}s")

            if stall > 100000:
                break

    return best_a, best_M


def coordinate_descent_polish(a0, time_limit, n_trials_per_coord=5, seed=42):
    """
    Coordinate descent: move mass from high-contributing coordinates to low ones
    to reduce the peak convolution value.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_M, stall
        S_t = trial.sum()
        if S_t < 1e-15:
            return False
        trial = trial / S_t
        M = fast_conv(trial).max()
        if M < best_M - 1e-15:
            a = trial
            best_a = trial.copy()
            best_M = M
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        # Find top peaks
        n_peaks = min(10, len(c))
        top_peaks = np.argpartition(c, -n_peaks)[-n_peaks:]

        improved = False
        for k_peak in top_peaks:
            # Which coordinates contribute to this peak?
            j_min = max(0, k_peak - n + 1)
            j_max = min(k_peak, n - 1)
            if j_min > j_max:
                continue
            idx = np.arange(j_min, j_max + 1)
            contrib = a[idx] * a[k_peak - idx]
            # Top contributors
            top_idx = idx[np.argsort(contrib)[::-1][:min(8, len(idx))]]

            # Mask of coordinates involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                jm2 = max(0, k2 - n + 1)
                je2 = min(k2, n - 1)
                peak_mask[jm2:je2 + 1] = True

            # Targets: coordinates NOT in peak regions, with small values
            free = np.where(~peak_mask)[0]
            if len(free) == 0:
                free = np.where(np.arange(n) != top_idx[0])[0]
            free_sorted = free[np.argsort(a[free])][:max(1, len(free) // 3)]

            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                for dst in free_sorted[:8]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbations
            if stall % 10 == 0:
                for _ in range(20):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    a_trial = a.copy()
                    a_trial[i] *= (1 - frac)
                    a_trial[j] += a[i] * frac
                    if try_update(a_trial):
                        break

            # Larger perturbation
            if stall % 50 == 0:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.0)
                try_update(a_trial)

            if stall > 500:
                break

    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    S = f.sum()
    if S < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / S
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    S_new = f_interp.sum()
    if S_new < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / S_new


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        a = np.ones(n) / n
    elif choice == 8:
        # Two-sided triangular
        center = n // 2
        idx = np.abs(np.arange(n) - center)
        a = np.maximum(0, 1.0 - idx / (n * 0.4))
    else:
        # Random
        a = rng.random(n) + 0.1
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    return a


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        S = a.sum()
        if S < 0.01:
            return
        a = a / S
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Multi-length LP scan ===
    print("\n=== Phase 0: Multi-length LP scan ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)

    scan_lengths = sorted(set([
        n_base,
        max(800, n_base - 200),
        n_base + 200,
        n_base + 400
    ]))

    for n_scan in scan_lengths:
        if time.monotonic() - t_global > time_budget * 0.10:
            break
        remaining = time_budget * 0.10 - (time.monotonic() - t_global)
        if remaining < 15:
            break

        if best_seq is not None:
            init = interpolate_sequence(best_seq, n_scan)
        else:
            init = structured_init(n_scan, rng)

        a_out, _ = nesterov_lp_optimize(init, remaining, solver, verbose=False)
        do_try_update(a_out, f"lp_scan_n{n_scan}")

    n_best = len(best_seq)
    print(f"  After Phase 0: n={n_best}, score={best_score:.6f}")

    # === Phase 1: Mirror Descent (main new method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror Descent n={n_best} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 30:
        a_init = best_seq / best_seq.sum()
        a_out, _ = nesterov_mirror_descent(a_init, t1, T_start=80.0, T_end=1.0,
                                           n_stages=10, verbose=True)
        do_try_update(a_out, "mirror_d1")

    # === Phase 2: Additive Subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Additive Subgrad n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_init = best_seq / best_seq.sum()
        a_out, _ = nesterov_additive_subgrad(a_init, t2, T_start=30.0, T_end=0.5,
                                             n_stages=6, verbose=True)
        do_try_update(a_out, "add_sg1")

    # === Phase 3: LP refinement at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP refine n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_init = best_seq / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_init, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_refine_n{n_best}")

    # === Phase 4: Mirror Descent with different temperature schedule ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Mirror Descent v2 n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 20:
        # Add small noise to break symmetry
        a_noisy = best_seq + rng.normal(0, 0.01 * best_seq.mean(), n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy / a_noisy.sum()
        a_out, _ = nesterov_mirror_descent(a_noisy, t4, T_start=30.0, T_end=0.3,
                                           n_stages=8, verbose=True)
        do_try_update(a_out, "mirror_d2")

    # === Phase 5: Coordinate Descent Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 400.0)
    if t5 > 15:
        a_init = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t5, seed=12345)
        do_try_update(a_out, "cd_polish1")

    # === Phase 6: Final Mirror Descent polish (low temperature) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Mirror polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 10:
        a_init = best_seq / best_seq.sum()
        a_out, _ = nesterov_mirror_descent(a_init, t6, T_start=10.0, T_end=0.1,
                                           n_stages=5, verbose=False)
        do_try_update(a_out, "mirror_polish")

    # === Phase 7: Final Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Coord Desc n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.6
    if t7 > 10:
        a_init = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t7, seed=77777)
        do_try_update(a_out, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
