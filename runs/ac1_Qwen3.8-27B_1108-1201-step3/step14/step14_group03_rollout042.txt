            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return cp.ECOS


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_grad_at_lag(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a. grad[j] = 2*a[k-j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if hi > lo:
        grad[lo:hi] = 2.0 * np.flip(a[k - hi + 1:k - lo + 1])
    return grad


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_stall=30):
    """Nesterov-accelerated LP best-response with adaptive restart."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    best_score_at_restart = best_score

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stall_count += 1
            if stall_count > max_stall:
                break
            continue

        stall_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Adaptive restart: if no improvement in many iterations, restart Nesterov
        if n_iters % 500 == 0 and n_iters > 0:
            if best_score >= best_score_at_restart - 1e-10:
                stall_count += 5
                # Perturb and restart
                rng_r = np.random.default_rng(n_iters)
                perturbation = rng_r.normal(0, 0.01 * norm_val / n, n)
                y = np.maximum(best_a + perturbation, 0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y * norm_val / s_y
                t = max(1, t // 3)  # reduce momentum
                best_score_at_restart = best_score
            else:
                best_score_at_restart = best_score

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """Minimize peak autocorrelation using Nesterov-accelerated subgradient method."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()  # Nesterov lookahead
    momentum_prev = 0.0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        max_val = conv_y.max()
        if max_val < 1e-14:
            break
        k_star = int(np.argmax(conv_y))

        # Compute subgradient at y for lag k_star
        grad = compute_grad_at_lag(y, k_star)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step size: eta = c / sqrt(t)
        eta = 1.0 / np.sqrt(t)

        # Gradient step
        a_cand = a - eta * grad / (n * max_val)
        a_cand = np.maximum(a_cand, 0.0)
        s = a_cand.sum()
        if s < 1e-15:
            a_cand = a.copy()
        else:
            a_cand = a_cand / s

        # Nesterov update
        momentum_new = (1.0 + np.sqrt(1.0 + 4.0 * momentum_prev ** 2)) / 2.0
        y_new = a_cand + ((momentum_new - 1.0) / momentum_new) * (a_cand - a)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_cand.copy()
        else:
            y_new = y_new / s_y

        momentum_prev = momentum_new
        a = a_cand
        y = y_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_golden(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent with golden-section search for optimal transfer."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm_val * 1e-10:
                continue

            # Determine which offsets to try (adaptive based on current peak)
            peak_lag = int(np.argmax(conv))
            # Prioritize offsets related to the peak
            offsets = []
            # Offsets that reduce the peak
            for d in range(1, window + 1):
                offsets.append((i + d) % n)
                offsets.append((i - d) % n)
            # Remove duplicates and self
            seen = set()
            unique_offsets = []
            for j in offsets:
                if j != i and j not in seen:
                    seen.add(j)
                    unique_offsets.append(j)

            for j in unique_offsets[:20]:  # Limit candidates per i
                if a[i] < norm_val * 1e-10:
                    break

                # Compute delta_vec: change in autocorrelation per unit transfer from i to j
                delta_vec = np.zeros(size)
                # Removing mass from i: -a at positions related to i
                lo1 = i
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    delta_vec[lo1:hi1] -= a[0:hi1 - lo1]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n - 1 + i + 1)
                if hi2 > lo2:
                    delta_vec[lo2:hi2] -= np.flip(a[max(0, lo2 - (n - 1)):max(0, lo2 - (n - 1)) + (hi2 - lo2)])

                # Adding mass at j: +a at positions related to j
                lo3 = j
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    delta_vec[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n - 1 + j + 1)
                if hi4 > lo4:
                    src_start = max(0, lo4 - (n - 1))
                    delta_vec[lo4:hi4] += np.flip(a[src_start:src_start + (hi4 - lo4)])

                delta_vec *= 2.0

                # Self-interaction terms (d^2 contributions)
                # +d^2 at 2j, -2d^2 at i+j, +d^2 at 2i
                self_vec = np.zeros(size)
                if 2 * j < size:
                    self_vec[2 * j] += 1.0
                if i + j < size:
                    self_vec[i + j] -= 2.0
                if 2 * i < size:
                    self_vec[2 * i] += 1.0

                # Golden section search for optimal d in [0, a[i]]
                d_max = a[i] * 0.999
                if d_max < 1e-15:
                    continue

                golden = (np.sqrt(5.0) - 1.0) / 2.0

                def obj_d(d):
                    new_conv = conv + d * delta_vec + d * d * self_vec
                    return new_conv.max()

                # Coarse search first
                n_samples = 20
                ds = np.linspace(0.0, d_max, n_samples)
                vals = np.array([obj_d(d) for d in ds])
                best_idx_coarse = np.argmin(vals)
                best_d_coarse = ds[best_idx_coarse]
                best_val_coarse = vals[best_idx_coarse]

                if best_val_coarse >= conv.max() - 1e-13:
                    continue

                # Fine golden section around the coarse optimum
                lo_d = max(0.0, ds[max(0, best_idx_coarse - 1)])
                hi_d = min(d_max, ds[min(n_samples - 1, best_idx_coarse + 1)])

                c_d = hi_d - golden * (hi_d - lo_d)
                d_d = lo_d + golden * (hi_d - lo_d)
                f_c = obj_d(c_d)
                f_d = obj_d(d_d)

                for _ in range(30):
                    if hi_d - lo_d < 1e-10:
                        break
                    if f_c < f_d:
                        hi_d = d_d
                        d_d = c_d
                        f_d = f_c
                        c_d = hi_d - golden * (hi_d - lo_d)
                        f_c = obj_d(c_d)
                    else:
                        lo_d = c_d
                        c_d = d_d
                        f_c = f_d
                        d_d = lo_d + golden * (hi_d - lo_d)
                        f_d = obj_d(d_d)

                best_d = (lo_d + hi_d) / 2.0
                best_val = obj_d(best_d)

                if best_val < conv.max() - 1e-12:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm_val / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_polish(a0, time_limit, verbose=True):
    """Log-sum-exp based smoothing polish."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_stages = 5
    temps = np.geomspace(5.0, 0.5, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(10, min(100, int(remaining * 2)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            max_conv = conv.max()
            # Soft-max weighting
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Gradient of sum(w * conv) w.r.t. a
            # = w-weighted sum of gradients at each lag
            grad = np.zeros(n)
            for k in range(len(w)):
                if w[k] < 1e-10:
                    continue
                g_k = compute_grad_at_lag(a, k)
                grad += w[k] * g_k

            step = 1.0 / (T * (1.0 + 0.05 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    n1 = n_base
    norm_val = np.sqrt(2.0 * n1)

    # Phase 1: LP Nesterov (55% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * norm_val / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Subgradient polish (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    norm_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 2: Subgradient polish n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 10:
        a_norm = best_seq * norm_best / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # Phase 3: Pair CD golden section (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    norm_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 3: Pair CD golden n={n_best} (budget={remaining * 0.20:.0f}s) ===")
    t3 = remaining * 0.20
    if t3 > 10:
        a_norm = best_seq * norm_best / best_seq.sum()
        a_out, _ = pair_cd_golden(a_norm, t3, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcdg_n{n_best}")

    # Phase 4: LSE polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    norm_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 4: LSE polish n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t4 = remaining * 0.10
    if t4 > 10:
        a_norm = best_seq * norm_best / best_seq.sum()
        a_out, _ = lse_polish(a_norm, t4, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 5: Final LP Nesterov round (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    norm_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n_best} (budget={remaining * 0.50:.0f}s) ===")
    t5 = remaining * 0.50
    if t5 > 20:
        a_norm = best_seq * norm_best / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 6: Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    norm_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 6: Final Pair CD n={n_best} (budget={remaining * 0.50:.0f}s) ===")
    t6 = remaining * 0.50
    if t6 > 10:
        a_norm = best_seq * norm_best / best_seq.sum()
        a_out, _ = pair_cd_golden(a_norm, t6, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcdg2_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
