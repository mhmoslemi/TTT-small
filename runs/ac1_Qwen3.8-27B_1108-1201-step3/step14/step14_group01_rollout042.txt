"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def normalize(a, target=1.0):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / s)


def subgradient_conv(a, k):
    """Compute gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    # d(ck)/da_i = a[k-i] + a[i-k] (with a[m]=0 for m<0 or m>=n)
    for i in range(n):
        val = 0.0
        ki = k - i
        if 0 <= ki < n:
            val += a[ki]
        ik = i - k
        if 0 <= ik < n:
            val += a[ik]
        g[i] = val
    return g


def subgradient_conv_fast(a, k):
    """Vectorized subgradient of conv(a,a)[k]."""
    n = len(a)
    g = np.zeros(n)
    # a[k-i] for i in range: k-i in [0,n-1] => i in [max(0,k-n+1), min(n-1,k)]
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if hi1 > lo1:
        g[lo1:hi1] += a[k - np.arange(lo1, hi1)]
    # a[i-k] for i in range: i-k in [0,n-1] => i in [k, k+n-1]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if hi2 > lo2:
        g[lo2:hi2] += a[np.arange(lo2, hi2) - k]
    return g


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = np.geomspace(100.0, 0.1, 15)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 4)))
        lr = 1.0 / (T_val * np.sqrt(n)) * 5.0

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -50.0, 50.0)
            w = np.exp(shifted)
            w = w / w.sum()

            # Gradient: 2 * (cross_corr(w,a) + conv(w,a))
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Backtracking line search
            step = lr
            a_old = a.copy()
            sc_old = max_c
            for trial in range(15):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new / s_new
                sc_new = fast_conv_self(a_new).max()
                if sc_new <= sc_old + 0.1 * step * np.linalg.norm(grad):
                    break
                step *= 0.5
            else:
                a_new = a_old

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE T={T_val:.2f}: best={best_score:.8f}, score={compute_score(best_a):.6f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient method with Polyak step size and averaging."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    # Lower bound: 2n/(2n-1) for the normalized objective (max of conv)
    lb = 1.0 / (2.0 * n - 1.0)  # lower bound on max_conv when sum=1
    f_lb = 2.0 * n * lb  # lower bound on score

    avg_a = np.zeros(n)
    n_avg = 0
    gamma = best_score * 0.01  # step size parameter
    patience = 0
    best_patience_score = best_score

    t_iter = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # since sum=1

        # Find active set (top few lags)
        top_k = max(1, min(20, int(0.05 * len(conv))))
        top_indices = np.argsort(conv)[-top_k:]
        # Weight by how close they are to the max
        weights = np.maximum(conv[top_indices] - cur_max * 0.8, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(top_indices)) / len(top_indices)
        else:
            weights = weights / weights.sum()

        # Average subgradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_indices):
            grad += w_k * subgradient_conv_fast(a, k)

        # Polyak-like step size
        gap = cur_score - f_lb
        if gap < 1e-8:
            break
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break
        step = min(gap / grad_norm_sq, gamma / (1.0 + 0.01 * t_iter))

        # Projected gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new = a_new / s_new

        a = a_new
        t_iter += 1

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            patience = 0
            # Shrink gamma on improvement
            gamma = max(gamma * 0.8, 1e-6)
        else:
            patience += 1
            if patience % 500 == 0:
                gamma *= 1.05  # gently increase step if stuck

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = normalize(avg_check)
            sc_avg = fast_conv_self(avg_check).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()

        if patience > 5000 and best_patience_score - best_score < 1e-6:
            # Perturb to escape
            n_pert = max(10, n // 50)
            idx = np.random.default_rng(t_iter).choice(n, size=n_pert, replace=False)
            a[idx] *= (1.0 + 0.05 * np.random.default_rng(t_iter + 1).standard_normal(n_pert))
            a = normalize(a)
            patience = 0
            gamma = best_score * 0.005

        if verbose and t_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t_iter} ({elapsed:.0f}s): best={best_score:.8f}, score={compute_score(best_a):.6f}")

    return best_a, best_score

    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    temp = 50.0

    while time.monotonic() - t0 < time_limit:
        # Phase A: LSE GD (a few steps)
        n_gd_steps = max(5, int((time_limit - (time.monotonic() - t0)) * 0.3))
        for _ in range(min(n_gd_steps, 50)):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            shifted = (conv - max_c) / temp
            shifted = np.clip(shifted, -50, 50)
            w = np.exp(shifted)
            w = w / w.sum()

            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            step = 2.0 / (temp * np.sqrt(n))
            a_new = np.maximum(a - step * grad, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                sc = fast_conv_self(a_new).max()
                if sc < max_c:
                    a = a_new
                else:
                    step *= 0.5
                    a_new = np.maximum(a - step * grad, 0.0)
                    a_new = a_new / a_new.sum()
                    a = a_new
            temp *= 0.98

        # Phase B: Pair CD
        a, sc = pair_cd_simple(a, min(5.0, time_limit - (time.monotonic() - t0)), rng, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Reset temp occasionally
        if temp < 1.0:
            temp = 10.0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE+CD ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        # Multi-peak
        k = rng.integers(3, 7)
        centers = np.sort(rng.uniform(0.1, 0.9, k))
        a = np.ones(n) * 0.3
        for c in centers:
            a += 0.8 * np.exp(-20.0 * (x - c) ** 2)
    else:
        # Flat with noise
        a = np.ones(n) * 0.5 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE GD (fast, 15% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: Subgradient (25% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: Subgradient n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = subgradient_optimize(best_seq, t2, verbose=True)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    # === Phase 3: Pair CD (25% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_simple(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")

    # === Phase 4: LSE+CD Hybrid (20% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LSE+CD Hybrid n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lse_cd_hybrid(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"hybrid_n{len(best_seq)}")

    # === Phase 5: Multi-restart with perturbations (10% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Multi-restart (budget={t5:.0f}s) ===")
    if t5 > 60:
        n_restarts = max(2, int(t5 / 40))
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            # Perturb the best solution
            a_pert = best_seq.copy()
            n_pert = max(20, len(a_pert) // 20)
            idx_p = rng.choice(len(a_pert), size=n_pert, replace=False)
            a_pert[idx_p] *= np.exp(0.2 * rng.standard_normal(n_pert))
            a_pert = normalize(a_pert)

            t_sub = min(t5 / n_restarts - 5, 30.0)
            a_out, _ = lse_gradient_descent(a_pert, max(t_sub, 10.0), verbose=False)
            try_update(a_out, f"restart{r}")

    # === Phase 6: Try different length (10% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Different length n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 60 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = lse_gradient_descent(init_long, min(t6 * 0.4, 80.0), verbose=False)
        try_update(a_out, f"lse_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out2, _ = pair_cd_simple(init_long, min(remaining * 0.3, 60.0), rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # === Phase 7: Final polish with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_half = remaining * 0.5
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 60.0), verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_simple(best_seq, min(remaining * 0.7, 80.0), rng, verbose=False)
            try_update(a_out, "pcd_final")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
