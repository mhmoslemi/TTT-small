Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with tight LP oracle (using actual autocorrelation bounds)

import numpy as np

import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_T(y, n):
    """Build the cross-correlation matrix T where (T @ b)[k] = sum_j y[j] * b[|k-j|]... 
    Actually, (T @ b)[k] = conv(y, b)[k] = sum_j y[j] * b[k-j]"""
    # For k in 0..2n-2, (conv(y,b))[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} y[j] * b[k-j]
    # So T[k, j] = y[j] if j is in valid range for lag k, i.e., 0 <= k-j < n
    # T[k, j] is nonzero iff 0 <= k-j < n, i.e., k-n+1 <= j <= k
    T = np.zeros((2*n-1, n))
    for j in range(n):
        # j contributes to T[k, j] for k in j..j+n-1
        T[j:j+n, j] = y
    return T

def solve_lp_oracle(y, ac, slack_mult=1.0, solver='CLARABEL'):
    """Solve the LP oracle: max sum(b) s.t. conv(y,b) <= ac * slack_mult, b >= 0"""
    n = len(y)
    T = build_T(y, n)
    bounds = ac * slack_mult
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= bounds, cp.sum(b_var) <= 1.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def line_search(f, x, y, max_iter=30):
    """Golden section line search to find alpha minimizing f((1-alpha)*x + alpha*y)."""
    a, b = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    f_a = f(x)
    f_b = f(y)
    
    # If endpoint is already better, return it
    if f_a <= f_b:
        # Maybe the minimum is near 0
        # Check if moving a tiny bit helps
        c = x + 0.01 * (y - x)
        f_c = f(np.maximum(c, 0))
        if f_c >= f_a:
            return 0.0
    
    c = a + gr * (b - a)
    d = b - gr * (b - a)
    f_c = f(np.maximum(x + c * (y - x), 0))
    f_d = f(np.maximum(x + d * (y - x), 0))
    
    for _ in range(max_iter):
        if f_c < f_d:
            b = d
            d = c
            f_d = f_c
            c = a + gr * (b - a)
            f_c = f(np.maximum(x + c * (y - x), 0))
        else:
            a = c
            c = d
            f_c = f_d
            d = b - gr * (b - a)
            f_d = f(np.maximum(x + d * (y - x), 0))
    
    best_alpha = (a + b) / 2
    return best_alpha

    t0 = time.monotonic()

            break

        M_curr = ac.max()

            g = solve_lp_oracle(a, ac, slack_mult=slack, solver=solver)
            if g is not None:
                break
        
        if g is None:
            break
        
        # Line search
        def obj(alpha_val):
            trial = np.maximum(a + alpha_val * (g - a), 0.0)
            s = trial.sum()
            if s < 1e-15:
                return float('inf')
            trial = trial / s
            return 2.0 * n * fast_autocorr(trial).max()
        
        alpha = line_search(obj, a, g, max_iter=20)
        
        if alpha < 1e-6:
            # Try away step
            if len(vertices) > 2:
                # Find the vertex most "opposite" to g
                best_away = None
                best_away_score = float('inf')
                for v in vertices:
                    # Direction away from v towards g

                        continue
                    away_dir = g - v
                    trial = a + 0.1 * away_dir
                    trial = np.maximum(trial, 0.0)
                    s = trial.sum()
                    if s < 1e-15:

                    trial = trial / s
                    m = fast_autocorr(trial).max()
                    if m < best_away_score:

                            best_m = m_new

        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
            alpha = 0.0
        a_new = a_new / s
        a = a_new

def golden_line_search(a, direction, max_iter=25):
    """Find alpha in [0,1] minimizing objective(a + alpha*(direction - a))."""
    a = np.asarray(a, dtype=np.float64)

            d = c
            fd = fc
            d_new = lo + gr * (hi - lo)

            fc = f_alpha(c)

            c = d
            fc = fd

def mirror_descent(a0, time_limit, top_k=20, beta=4.0, verbose=False):
    """Mirror descent with soft subgradient for finer polishing."""

    t0 = time.monotonic()
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = 0.5 / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        
        # Compute soft subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0.0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:

            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:

            J = 2.0 * n * best_m

            if time.monotonic() - t0 > time_limit:
                break
            
            k = int(top_k[rng.integers(len(top_k))])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Score candidates: how much would moving to dst affect top peaks?
            # Precompute removal effect
            rem_vals = np.zeros(len(top_k))
            for ii, k2 in enumerate(top_k):
                k2i = int(k2)
                d = k2i - src
                rem_vals[ii] = a[d] if 0 <= d < n else 0.0
            
            # Sample candidates
            n_cand = min(n, 400)
            if n_cand < n:
                # Prefer low-contribution positions
                peak_sens = np.zeros(n)
                for k2 in top_k:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    idx2 = np.arange(s2, e2 + 1)
                    peak_sens[idx2] += a[k2i - idx2]
                order = np.argsort(peak_sens)
                candidates = order[:n_cand]
            else:
                candidates = np.arange(n)
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in candidates:
                if dst == src:
                    continue
                add_vals = np.zeros(len(top_k))
                for ii, k2 in enumerate(top_k):
                    k2i = int(k2)
                    d = k2i - dst
                    add_vals[ii] = a[d] if 0 <= d < n else 0.0
                
                net = add_vals - rem_vals
                score = (ac[top_k] + 2.0 * net).max()
                
                if score < best_score:
                    best_score = score
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            for frac in [0.02, 0.05, 0.1, 0.2, 0.35]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-12:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: FW with line search at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FW+LS n={n0} ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fw_with_linesearch(a_in, t1, solver, verbose=True)
        try_update(a_out, f"fw_ls_n{n0}")
    
    # === Phase 2: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t2, verbose=True)
        try_update(a_out, f"coord_n{len(best_seq)}")
    
    # === Phase 3: MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD n={len(best_seq)} ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t3, top_k=20, beta=4.0, verbose=True)
        try_update(a_out, f"md_n{len(best_seq)}")
    
    # === Phase 4: Scale to 2x, FW + polish ===
    n1 = n0 + 800  # e.g., 1600 -> 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Scale to n={n1} ===")
    if remaining > 60:
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        t4 = min(remaining * 0.25, 350.0)
        if t4 > 30:
            a_out, _ = fw_with_linesearch(a_in, t4 * 0.6, solver, verbose=True)
            try_update(a_out, f"fw_ls_n{n1}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, min(remaining * 0.3, 100.0), verbose=True)
                try_update(a_out, f"coord_n{len(best_seq)}")
    
    # === Phase 5: Second FW pass at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: FW2 n={n_best} ===")
    if n_best <= 2500 and remaining > 40:
        t5 = min(remaining * 0.25, 300.0)
        if t5 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = fw_with_linesearch(a_in, t5, solver, verbose=True)
            try_update(a_out, f"fw2_n{n_best}")
    
    # === Phase 6: Final coord + MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} ===")
    if remaining > 20:
        t6a = min(remaining * 0.5, 120.0)
        if t6a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t6a, verbose=True)
            try_update(a_out, f"coord_final_{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent(a_in, min(remaining * 0.6, 80.0), top_k=15, beta=3.0, verbose=True)
            try_update(a_out, f"md_final_{n_best}")
    
    # === Phase 7: Try alternative initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Alt init n={n_best} ===")
        # Exponential-ish init
        x = np.linspace(0, 1, n_best)
        alt1 = np.exp(-2.0 * x)
        # Blend with current best
        blend = 0.3 * norm_sum1(best_seq) + 0.7 * norm_sum1(alt1)
        blend = norm_sum1(blend)
        
        t7 = min(remaining * 0.4, 100.0)
        if t7 > 10:
            if n_best <= 2000:
                a_out, _ = fw_with_linesearch(blend, t7, solver, verbose=False)
            else:
                a_out, _ = mirror_descent(blend, t7, verbose=False)
            try_update(a_out, "alt_fw")
    
    # === Phase 8: Final short coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Last coord n={n_best} ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, min(remaining * 0.7, 120.0), verbose=True)
        try_update(a_out, "coord_last")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
