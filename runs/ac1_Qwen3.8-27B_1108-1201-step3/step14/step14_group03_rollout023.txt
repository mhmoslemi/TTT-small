Actually, one issue with the subgradient: the subgradient g[i] = 2*a[k*-i] is a shifted version of a. If k* = n-1, then g[i] = 2*a[n-1-i] which is the reversed a. The step a - eta*g would be a - eta*2*flip(a), which could create negative values. After projection (clipping to 0 and renormalizing), this should be fine.

But the question is: does this actually reduce the max? Let's think intuitively:
- If the max is at lag k*, we want to reduce (a*a)[k*]
- (a*a)[k*] = sum_j a[j]*a[k*-j]
- Reducing a[j] for j that has a large a[k*-j] will help
- The subgradient points in the direction of increasing (a*a)[k*], so we step in the opposite direction
- This means we decrease a[i] where a[k*-i] is large, i.e., where the "partner" element is large

This makes sense! We're reducing the elements that are "responsible" for the peak correlation.

However, there's a subtlety: reducing a[i] also affects other correlations (k != k*). We might reduce the peak at k* but increase peaks at other lags. The subgradient method handles this by tracking the running maximum.

Let me also add a "diversity" element: sometimes, instead of using the current argmax, use a random lag near the max. This can help escape local minima.

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()

    y = a.copy()
    running_avg = np.zeros(n)
    n_avg = 0
    t0 = time.time()
    n_iters = 0
    stuck = 0

    while time.time() - t0 < time_limit:
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()

        b = lp_solve_subset(y, M_curr, top_idx, solver)

            continue

        g = b * norm / b_sum
        n_iters += 1

        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if n_iters == 1:
            a_new = g.copy()
        else:
            momentum = (n_iters - 1.0) / (n_iters + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()

def subgradient_polish(a0, time_limit, verbose=True):

    n_avg = 0

def lse_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.time()
    fft_size = 1 << (2 * n - 2).bit_length()

    n_stages = 5
    temps = np.geomspace(8.0, 0.5, n_stages)

    for si, T in enumerate(temps):
        if time.time() - t0 > time_limit:
            break
        remaining = time_limit - (time.time() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 3)))

        for it in range(n_steps):
            if time.time() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            max_conv = conv.max()
            shifted = conv - max_conv
            max_shift = np.max(shifted)
            if max_shift > 50 * T:
                shifted = np.clip(shifted, -50 * T, 0)
            exp_conv = np.exp(shifted / T)
            w = exp_conv / (exp_conv.sum() + 1e-30)

            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)
            _ = irfft(W_fft * np.conj(A_fft), fft_size)

            grad = np.zeros(n)
            for i in range(n):
                k_lo = max(0, i)
                k_hi = min(n, i + n)
                if k_hi > k_lo:
                    grad[i] += np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(n, i + 1)
                if k_hi2 > k_lo2:
                    grad[i] += np.dot(w[k_lo2:k_hi2], a[i - np.arange(k_lo2, k_hi2)])
            grad *= 2.0

            step = norm / (T * n) / (1.0 + 0.05 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.time() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
