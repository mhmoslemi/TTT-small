def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))

def lp_best_response_his(a, M):
    """Solve LP best-response using scipy HiGHS solver."""
    n = len(a)
    size = 2 * n - 1
    
    # Build sparse constraint matrix T where T[k, j] = a[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for k in range(size):
        # j must satisfy: 0 <= j < n and 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    # LP: maximize sum(b) subject to T@b <= M, b >= 0
    # scipy linprog minimizes, so we negate the objective
    c = -np.ones(n)
    A_ub = T
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None

        if b_sum < 1e-15:
            continue

            if cs < 1e-15:
                continue

            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        
        y_new = np.maximum((1-best_alpha)*y + best_alpha*g, 0.0)

            y_new = y_new * norm / s_y

            a_new = g.copy()
        else:

            stall = 0
            best_stall_reset = best_maxb
        
        score_a = fast_autocorr(a).max()
        if score_a < best_maxb:
            best_maxb = score_a
            best_a = a.copy()
            last_improve_time = time.monotonic()
            stall = 0

                    best_a = avg.copy()

    # So w[m-i] = w[2n-2-(m-i)] = w[2n-2-m+i]... hmm this isn't the same as w[i+m].
    
    # OK let me just do it numerically. For efficiency, I'll use the fact that:
    # sum_i a[i]*w[i+m] is a "sliding dot product" which can be computed via convolution.
    
    # Define: extend a to length 2n-1 (pad with zeros): a_ext = [a, 0, 0, ..., 0] (length 2n-1)
    # Then sum_i a[i]*w[i+m] for i=0..n-1 = sum_i a_ext[i]*w[i+m]
    # This is the cross-correlation of a_ext and w at lag m.
    
    # Cross-correlation via FFT:
    # corr[m] = sum_k x[k]*y[k+m] = (x ⊛ y)[m]
    # In frequency domain: FFT(corr) = conj(FFT(x)) * FFT(y)
    # But this gives circular correlation. For linear, we need to zero-pad.
    
    fft_size2 = 1 << ((3*n - 1)).bit_length()
    a_ext = np.zeros(fft_size2)
    a_ext[:n] = a
    w_ext = np.zeros(fft_size2)
    w_ext[:2*n-1] = w
    
    # corr[m] = sum_k a_ext[k] * w_ext[k+m]
    # = sum_k a_ext[k] * w_ext[k+m]
    # This is the cross-correlation. In FFT:
    # corr = irfft(conj(rfft(a_ext)) * rfft(w_ext))
    # But this gives the CIRCULAR cross-correlation, which for zero-padded signals equals the linear one.
    
    A_ext = rfft(a_ext, fft_size2)
    W_ext = rfft(w_ext, fft_size2)
    corr = irfft(np.conj(A_ext) * W_ext, fft_size2)
    
    # corr[m] = sum_k a_ext[k] * w_ext[k+m] (circular, but since padded, equals linear for m in [0, 2n-2])
    # Actually I need to double-check the sign convention.
    # corr[m] = sum_k x[k] * y[(k+m) mod N]
    # With N = fft_size2 and zero padding, for m in [0, N-1-n], this equals sum_k x[k]*y[k+m]
    # which is what we want.
    
    # So grad_lse[m] = 2 * corr[m] for m = 0..n-1
    # Wait, I need to verify the factor of 2.
    # 
    # The full gradient is: sum_k w[k]*(a[k-m] + a[m-k])
    # First term: sum_k w[k]*a[k-m] = sum_i a[i]*w[i+m] = corr[m] ✓
    # Second term: sum_k w[k]*a[m-k] = sum_i a[i]*w[m-i]
    #   Let j = m-i: = sum_j a[m-j]*w[j] = sum_j w[j]*a[m-j]
    #   This is the cross-correlation at lag -m, or equivalently:
    #   = sum_j w[j]*a[(m)-j] = (w ⊛ a)[m]... 
    #   
    #   By symmetry of w (w[k] = w[2n-2-k]):
    #   sum_i a[i]*w[m-i] = sum_i a[i]*w[2n-2-(m-i)] = sum_i a[i]*w[2n-2-m+i]
    #   Let j = 2n-2-m+i, so i = j-2n+2+m:
    #   = sum_j a[j-2n+2+m]*w[j]
    #   This is getting messy. Let me just compute both terms.
    
    # Second term: sum_i a[i]*w[m-i] for i=0..n-1 (w[m-i] valid when 0<=m-i<2n-1)
    # = sum_i a[i]*w[m-i]
    # Let's compute this via FFT too:
    # sum_i a[i]*w[m-i] = (a * w)[m] where * is convolution
    # In FFT: = irfft(A_ext * W_ext) evaluated at m... but w is length 2n-1, not n.
    
    # Actually, the second term sum_k w[k]*a[m-k] with k in [0,2n-2] and m-k in [0,n-1]:
    # = sum_{k=0}^{m} w[k]*a[m-k] (for m < n, the constraint m-k < n is automatic for k >= 0)
    # Wait, we also need m-k >= 0, so k <= m. And k >= 0. And m-k < n means k > m-n.
    # For m in [0, n-1]: k in [max(0, m-n+1), m] = [0, m] (since m < n)
    # So second term = sum_{k=0}^{m} w[k]*a[m-k]
    
    # And first term = sum_{k=m}^{m+n-1} w[k]*a[k-m] (for m in [0,n-1])
    # Let i = k-m: sum_{i=0}^{n-1} w[i+m]*a[i]
    
    # So total = sum_{i=0}^{n-1} w[i+m]*a[i] + sum_{k=0}^{m} w[k]*a[m-k]
    
    # The first is what I called corr[m] above.
    # The second is the convolution (w * a)[m] = sum_k w[k]*a[m-k] for k=0..m.
    
    # Both can be computed via FFT:
    # First: corr[m] = sum_i a[i]*w[i+m] → computed above as corr[m]
    # Second: conv[m] = sum_k w[k]*a[m-k] (k from 0 to min(m, 2n-2), a[m-k] valid when 0<=m-k<n)
    #   For m in [0, n-1]: k from 0 to m (since m-k >= 0 means k <= m, and m-k < n is automatic)
    #   = sum_{k=0}^{m} w[k]*a[m-k]
    #   This is the convolution of w (first m+1 elements) with a, evaluated at m.
    #   Via FFT: (w_full * a)[m] where w_full is padded to length 2n-1.
    
    # Let me compute conv via FFT:
    conv_val = irfft(A_ext * W_ext, fft_size2)
    # conv_val[m] = sum_k a_ext[k]*w_ext[m-k] = sum_k a[k]*w[m-k] (for appropriate k)
    # This is the standard convolution.
    # conv_val[m] = sum_{k=0}^{min(m, n-1)} a[k]*w[m-k] (since a is zero for k>=n)
    # For m in [0, n-1]: = sum_{k=0}^{m} a[k]*w[m-k] ✓ (this is the second term!)
    
    # So: grad_lse[m] = corr[m] + conv_val[m]
    # where corr[m] = sum_i a[i]*w[i+m] and conv_val[m] = sum_k a[k]*w[m-k]
    
    grad_conv = corr[:n] + conv_val[:n]
    
    # Now the full objective gradient:
    # f(a) = 2*n*LSE / s^2 where s = sum(a)
    # df/da[m] = 2*n * (grad_conv[m] * s - 2*LSE*a[m]) / s^3
    #          = 2*n*grad_conv[m]/s^2 - 4*n*LSE*a[m]/s^3
    
    obj_val = 2.0 * n * lse / (s * s)
    grad = 2.0 * n * grad_conv / (s * s) - 2.0 * obj_val * a / s
    
    return obj_val, grad


def lse_optimize(a
0, time_limit, T_schedule=None, verbose=True):
    """Optimize using LSE-smoothed objective with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Temperature schedule
    if T_schedule is None:
        T_schedule = [50.0, 30.0, 15.0, 8.0, 4.0, 2.0]
    
    n_stages = len(T_schedule)
    base_time = time_limit / n_stages
    
    for stage, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        stage_time = min(base_time, time_limit - (time.monotonic() - t0))
        if stage_time < 5:
            break
        
        x0 = a.copy()
        n_iter = 0
        
        def obj_func(x):
            nonlocal n_iter
            n_iter += 1
            val, _ = lse_objective_and_grad(x, n, T)
            return val
        
        def grad_func(x):
            nonlocal n_iter
            n_iter += 1
            _, g = lse_objective_and_grad(x, n, T)
            return g
        
        try:
            if verbose:
                print(f"    LSE stage {stage+1}/{n_stages} (T={T}): starting")
            
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxls': 50, 'gtol': 1e-8})
            
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = compute_score(a_candidate)
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
                        if verbose:
                            print(f"      New best: {score_c:.8f}")
                    
                    a = a_candidate  # always update a for next stage even if not best
        
        except Exception as e:
            if verbose:
                print(f"      L-BFGS-B exception: {e}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42, verbose=False):
    """Gradient-guided coordinate descent with mass redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        b = fast_autocorr(a)
        # Find peak lag
        peak_lag = np.argmax(b)
        
        # Contributions to the peak
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_lag - idx]
        
        # Top contributors
        top_k = min(20, len(idx))
        top_contrib = idx[np.argsort(contrib)[::-1][:top_k]]
        
        # Try moving mass from high contributors to low-contributing positions
        improved = False
        
        # Identify "safe" positions (those that don't contribute much to any top peak)
        top_peaks = np.argsort(b)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2+1] = True
        
        safe_positions = np.where(~peak_mask)[0]
        if len(safe_positions) == 0:
            safe_positions = np.where(a < a.mean())[0]
        
        for src in top_contrib[:10]:
            if a[src] < 1e-12:
                continue
            # Try moving to safe positions
            dst_candidates = safe_positions[np.argsort(a[safe_positions])][:min(10, len(safe_positions))]
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if try_update(a_trial):
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbations
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                # Move from higher to lower or random
                if a[i] > a[j]:
                    src, dst = i, j
                else:
                    src, dst = j, i
                if a[src] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if try_update(a_trial):
                    improved = True
                    break
            
            if not improved:
                stall += 1
                if stall > 200:
                    break
    
    return best_a, best_score
