def lse_gradient(a, c, T):
    """Compute gradient of T*log(sum exp(c[k]/T)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(2 * n - 1)
        p[c.argmax()] = 1.0
    else:
        p = p / p_sum
    # g[i] = 2 * sum_m a[m] * p[i+m]
    # = 2 * conv(a[::-1], p)[n-1:2n-1]
    a_flip = a[::-1]
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a_flip, fft_size)
    P = rfft(p, fft_size)
    conv = irfft(A * P, fft_size)[:3 * n - 1]
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g


def lse_mirror_descent(a0, time_limit, rng, verbose=True):
    """LSE mirror descent with Nesterov momentum."""
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    c = fast_autocorr(a)
    best_max_c = c.max()

    t_start = time.monotonic()
    momentum = 0.7
    v = np.zeros(n)

    alphas = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03]
    n_levels = len(alphas)
    time_per_level = time_limit / n_levels
    total_iters = 0

    for level, alpha in enumerate(alphas):
        if time.monotonic() - t_start > time_limit:
            break
        level_start = time.monotonic()
        eta = 0.5
        v = np.zeros(n)
        local_best = best_max_c
        n_stuck = 0
        n_iter = 0

        while time.monotonic() - level_start < time_per_level:
            if time.monotonic() - t_start > time_limit:
                break

            c = fast_autocorr(a)
            c_max = c.max()
            T = alpha * c_max
            if T < 1e-14:
                T = 1e-14

            g = lse_gradient(a, c, T)
            g_norm = np.abs(g).max()
            if g_norm < 1e-15:
                break

            scale = min(5.0, eta * g_norm)
            log_step = -np.sign(g) * np.abs(g) / g_norm * scale
            log_step = np.clip(log_step, -6.0, 6.0)

            v = momentum * v + log_step
            v = np.clip(v, -10.0, 10.0)

            a_new = a * np.exp(v)
            a_new = np.maximum(a_new, 1e-18)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            a = a_new
            n_iter += 1
            total_iters += 1

            c_new = fast_autocorr(a)
            max_c_new = c_new.max()

            if max_c_new < local_best - 1e-12:
                local_best = max_c_new
                n_stuck = 0
            else:
                n_stuck += 1
                if n_stuck > 200:
                    eta *= 0.8
                    if eta < 0.01:
                        break

            if max_c_new < best_max_c:
                best_max_c = max_c_new
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"    LSE level {level} (alpha={alpha:.2f}): {n_iter} iters, best={best_max_c:.10f}, t={elapsed:.1f}s")

    return best_a


def pair_cd_optimize(a0, time_limit, rng, window=50, verbose=True):
    """Pair coordinate descent with optimal 1D line search."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_autocorr(a)
    best_max_c = c.max()
    best_a = a.copy()
    t_start = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t_start < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() - t_start > time_limit:
                break
            i = perm[pos]
            ai = a[i]
            if ai < 1e-12:
                continue

            best_improve = 0.0
            best_j = -1
            best_delta = 0.0

            for offset in range(1, window + 1):
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i:
                        continue

                    # Compute delta vector: d[k] = 2*(a[k-j] - a[k-i])
                    # c_new[k] = c[k] + delta * d[k] (first order)
                    # Also quadratic corrections at k=2i and k=2j

                    # Build d vector efficiently
                    d = np.zeros(size)
                    # a[k-i] for k where 0<=k-i<n: k in [i, i+n-1]
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        d[lo1:hi1] -= 2.0 * a[:hi1 - lo1]
                    # a[k-j] for k where 0<=k-j<n: k in [j, j+n-1]
                    lo2 = j
                    hi2 = min(size, j + n)
                    if hi2 > lo2:
                        d[lo2:hi2] += 2.0 * a[:hi2 - lo2]

                    # Find best delta in [0, ai] that minimizes max(c + delta*d)
                    # Use grid search
                    n_samples = 20
                    max_delta = ai * 0.99
                    deltas = np.linspace(0, max_delta, n_samples)
                    best_local = c.max()
                    best_local_d = 0.0

                    for delta in deltas[1:]:
                        new_c = c + delta * d
                        if 2 * i < size:
                            new_c[2 * i] = c[2 * i] - 2 * ai * delta + delta * delta
                        if 2 * j < size:
                            new_c[2 * j] = c[2 * j] + 2 * a[j] * delta + delta * delta
                        # Cross term at k=i+j (if i != j): the a[i]*a[j] term changes
                        k_cross = i + j
                        if k_cross < size and i != j:
                            new_c[k_cross] = c[k_cross] + 2 * delta * (a[j] - a[i]) - 2 * delta * delta
                        # Actually for i+j: the term a[i]*a[j] becomes (a[i]-d)(a[j]+d)
                        # change = d*(a[j]-a[i]) - d^2, and it appears once in c[i+j]
                        # But we already counted it in the linear term. Let's just use the linear approx.
                        nm = new_c.max()
                        if nm < best_local - 1e-13:
                            best_local = nm
                            best_local_d = delta

                    if best_local_d > 0 and best_local < best_max_c - best_improve:
                        best_improve = best_max_c - best_local
                        best_j = j
                        best_delta = best_local_d

                if best_j >= 0:
                    break

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    continue
                a = a / s2
                c = fast_autocorr(a)
                cur_max = c.max()
                if cur_max < best_max_c - 1e-13:
                    best_max_c = cur_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_max_c:.10f}")

    return best_a


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Build constraint matrix (sparse would be better but dense works for n~1600)
        size = 2 * n - 1
        T_mat = np.zeros((size, n))
        for j in range(n):
            T_mat[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            if solver != 'none':
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            stuck += 1
            if stuck > 50:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stuck += 1
            if stuck > 50:
                break
            continue

        stuck = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue

        g = b_val * norm / b_sum
        n_iters += 1

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a


def try_candidate(a, current_best_score):
    """Evaluate and return (a, score) if improved."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return None, float('inf')
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < current_best_score:
        return a_norm, sc
    return None, current_best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_a = None
    best_score = float('inf')

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            sc = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = parent.copy()
    except Exception:
        parent = None

    n_base = 1600
    if parent is not None:
        n_base = len(parent)

    # Generate diverse starting points
    starts = []
    starts.append(('ramp_up', make_ramp(n_base, 1)))
    starts.append(('ramp_down', make_ramp(n_base, -1)))
    starts.append(('quad_up', make_quadratic(n_base, 1)))
    starts.append(('quad_down', make_quadratic(n_base, -1)))
    starts.append(('bell_left', make_bell(n_base, 0.3, 0.25)))
    starts.append(('bell_right', make_bell(n_base, 0.7, 0.25)))
    starts.append(('bell_narrow', make_bell(n_base, 0.4, 0.15)))
    starts.append(('uniform', np.ones(n_base) / n_base))

    if parent is not None:
        starts.append(('parent', parent.copy()))
        starts.append(('parent_pert', make_perturbed(rng, n_base, parent, 0.05)))

    # Add some random asymmetric starts
    for _ in range(3):
        a_rand = np.abs(rng.normal(0, 1, n_base))
        a_rand = a_rand / a_rand.sum()
        starts.append((f'rand_{_}', a_rand))

    # Evaluate all starting points
    print("\n=== Evaluating starting points ===")
    for name, a_start in starts:
        sc = evaluate_safe(a_start)
        print(f"  {name}: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_start.copy()

    if best_a is None:
        best_a = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_a)

    # Sort starts by score for optimization priority
    start_scores = [(evaluate_safe(a), name, a) for name, a in starts]
    start_scores.sort(key=lambda x: x[0])
    print(f"\nBest start: {start_scores[0][1]} with score={start_scores[0][0]:.6f}")

    # Optimize top starts
    n_top = min(5, len(start_scores))
    time_per_start = time_budget / (n_top * 2)  # leave time for final polish

    for rank in range(n_top):

        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
