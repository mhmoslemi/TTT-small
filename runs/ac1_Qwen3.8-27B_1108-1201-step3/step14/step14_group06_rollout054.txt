def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0. Returns (b, dual_vars)."""
    n = len(a)

        return None, None

        return None, None

def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)

        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    # Phase 1: LP Nesterov (main workhorse) - 45% of budget
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.45
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, eps_base=0.02,
                                     use_multi_M=True, avg_decay=0.995)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp1_n{n_best}", t_global)

    # Phase 2: Local LP optimization - 20% of budget
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Local LP n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_lp_optimize(a_norm, t2, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"localLP_n{n_best}", t_global)

    # Phase 3: LP Nesterov round 2 from improved point - 10%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-Nest r2 n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 180.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=False, eps_base=0.015,
                                         use_multi_M=True, avg_decay=0.997)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_r2_n{n_best}", t_global)

    # Phase 4: LSE gradient optimization - 10%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t4, T_start=40.0, T_end=0.5, n_stages=4)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_n{n_best}", t_global)

    # Phase 5: Coordinate descent - 7%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.07, 120.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t5, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"coord_n{n_best}", t_global)

    # Phase 6: Try slightly different length - 5%
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 200
    print(f"\n=== Phase 6: Longer n={n_long} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.05, 80.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False, eps_base=0.02)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"long_n{n_long}", t_global)

    # Phase 7: Final polish - remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        # Local LP again
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        t7a = min(remaining * 0.4, 100.0)
        if t7a > 15:
            a_out, _ = local_lp_optimize(a_norm, t7a, solver, verbose=True)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "localLP_final", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        # Coordinate descent
        if remaining2 > 15:
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out, _ = targeted_coord_descent(a_norm2, min(remaining2 * 0.5, 80.0), seed=999)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final", t_global)

        remaining3 = time_budget - (time.monotonic() - t_global)
        # LSE with very low temperature
        if remaining3 > 15:
            n_best3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_best3) / best_seq.sum()
            a_out, _ = lse_scipy_optimize(a_norm3, min(remaining3, 60.0),
                                           T_start=10.0, T_end=0.2, n_stages=3)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "lse_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
