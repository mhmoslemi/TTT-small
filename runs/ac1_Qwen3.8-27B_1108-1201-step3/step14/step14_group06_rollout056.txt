def peak_gradient(a, k):
    """Compute gradient of (a*a)[k] with respect to a.
    Result: grad[j] = 2 * a[k-j] (with zero padding for out of bounds)."""
    n = len(a)
    grad = np.zeros(n)
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        # grad[j] = 2 * a[k-j] for j in [j_start, j_end]
        js = np.arange(j_start, j_end + 1)
        grad[js] = 2.0 * a[k - js]
    return grad

def top_k_peak_gradient(a, b, K=5):
    """Compute weighted gradient from top K peaks."""
    n = len(a)
    # Get top K peaks
    top_idx = np.argsort(b)[::-1][:K]
    # Weight by peak value (exponential weighting for emphasis on top peak)
    weights = np.exp(-0.5 * (b[top_idx] - b[top_idx[0]]) / max(b[top_idx[0]], 1e-15))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * peak_gradient(a, k)
    return grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_conv(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def peak_gradient_single(a, k):
    """Gradient of (a*a)[k] w.r.t. a: grad[j] = 2*a[k-j] (zero-padded)."""
    n = len(a)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    grad = np.zeros(n)
    if j_start <= j_end:
        js = np.arange(j_start, j_end + 1)
        grad[js] = 2.0 * a[k - js]
    return grad


def multi_peak_gradient(a, b, K=5, temperature=0.1):
    """Weighted gradient from top-K peaks with softmax weighting."""
    n = len(a)
    top_idx = np.argsort(b)[::-1][:K]
    peak_vals = b[top_idx]
    max_peak = peak_vals[0]
    if max_peak < 1e-15:
        return np.zeros(n)
    # Exponential weighting (higher weight for higher peaks)
    weights = np.exp(-temperature * (max_peak - peak_vals) / max_peak)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * peak_gradient_single(a, int(k))
    return grad


def build_constraint_matrix(a, n):
    """Build T matrix where (T @ b)[k] = sum_j a[j]*b[k-j] (for fixed a)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                score = fast_conv(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def simulated_annealing_peak(a0, time_limit, seed=123):
    """Simulated annealing that specifically targets peak reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    T_start = 0.1
    T_end = 0.001
    n_steps = 50000

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_start * (T_end / T_start) ** (step / max(n_steps, 1))

        b = fast_conv(a)
        k_peak = int(np.argmax(b))

        # Propose move: transfer mass from high-contrib to low-contrib at peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        # Sample source proportional to contribution
        total_c = contrib.sum()
        if total_c < 1e-15:
            src_pos = idx[rng.integers(len(idx))]
        else:
            probs = contrib / total_c
            src_pos = idx[rng.choice(len(idx), p=probs)]

        # Sample target uniformly from non-peak regions
        target_candidates = np.where(a < np.median(a))[0]
        if len(target_candidates) == 0:
            target_candidates = np.arange(n)
        dst_pos = target_candidates[rng.integers(len(target_candidates))]

        if src_pos == dst_pos or a[src_pos] < 1e-14:
            continue

        frac = rng.uniform(0.02, 0.15)
        amt = min(a[src_pos] * frac, a[src_pos] - 1e-14)
        if amt < 1e-14:
            continue

        a_trial = a.copy()
        a_trial[src_pos] -= amt
        a_trial[dst_pos] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            continue
        a_trial = a_trial / s_t

        score_trial = fast_conv(a_trial).max()
        delta = score_trial - b.max()

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    old_vals = np.concatenate([f_norm, [0.0]])
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    new_vals = np.interp(new_x, old_x, old_vals)
    f_interp = new_vals[:-1]
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def antisymmetric_init(n, rng):
    """Create an initialization with anti-symmetric structure around center."""
    x = np.linspace(0.0, 1.0, n)
    # Base: smooth bump
    a = np.exp(-3.0 * (x - 0.5) ** 2)
    # Anti-symmetric perturbation (large at edges, small at center or vice versa)
    antisym = np.sin(np.pi * x)
    # Or: linear tilt
    tilt = 2.0 * x - 1.0
    # Combine
    beta = rng.uniform(0.1, 0.4)
    a = a * (1.0 + beta * tilt)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def oscillatory_init(n, rng):
    """Create an oscillatory initialization."""
    choice = rng.integers(0, 4)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 4) * x) + 0.2 * np.cos(2 * np.pi * rng.uniform(5, 10) * x)
    elif choice == 1:
        freq = rng.uniform(2, 6)
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2) + 0.1 * np.cos(2 * np.pi * freq * x)
    elif choice == 2:
        a = np.sin(np.pi * x) ** rng.uniform(0.5, 2.0)
    else:
        a = np.exp(-4 * (x - rng.uniform(0.2, 0.8)) ** 2) + 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search combining gradient, LP, coordinate descent, and SA."""
    rng = np.random.default_rng(seed)

    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

            best_score = p_score

    if best_seq is None:
        n_base = 2000
        best_seq = antisymmetric_init(n_base, rng)

        print(f"Default init: n={n_base}, score={best_score:.6f}")

    return [float(x) for x in best_seq]
