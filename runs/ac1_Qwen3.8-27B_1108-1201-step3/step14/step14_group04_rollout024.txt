    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    """Project onto simplex {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:

        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    return 2.0 * cross[n - 1: n - 1 + n]


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 30.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 8.0), n / (T * 1.5))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix: T_mat[i, j] = a[j] if i-j is in [0, n-1]
    # i.e., (conv(a,b))[i] = sum_j a[j] * b[i-j] for valid j
    # We need: sum_j a[j] * b[i-j] <= M for all i
    # For a fixed i, the valid j are: max(0, i-n+1) <= j <= min(n-1, i)
    # So T_mat[i, j] = a[j] if max(0, i-n+1) <= j <= min(n-1, i), else 0
    
    # More efficient: use the structure
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True, use_geometric_avg=False, decay=0.999):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    # For geometric averaging
    geo_avg_a = np.zeros(n)
    geo_weight = 0.0
    t0 = time.monotonic()
    n_iters = 0
    last_print = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Arithmetic ergodic average
        avg_a += a
        n_avg += 1
        
        # Geometric ergodic average
        if use_geometric_avg:
            geo_avg_a = decay * geo_avg_a + (1 - decay) * a
            geo_weight = decay * geo_weight + (1 - decay)
        
        check_interval = 10
        if n_avg % check_interval == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
            
            if use_geometric_avg and geo_weight > 1e-10:
                geo_check = geo_avg_a / geo_weight
                s_geo = geo_check.sum()
                if s_geo > 1e-15:
                    geo_check = geo_check * norm / s_geo
                    score_geo = fast_autocorr(geo_check).max()
                    if score_geo < best_score:
                        best_score = score_geo
                        best_a = geo_check.copy()

        if verbose and n_iters - last_print >= 50:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
            last_print = n_iters

    return best_a, best_score


def subgradient_heavy_ball(a0, time_limit, verbose=True):
    """Direct subgradient method with heavy ball momentum and periodic restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Heavy ball parameters
    momentum = 0.9
    # Step size: use Polyak-like step
    # f*(a) >= 1 (theoretical lower bound), so step = (f(a) - f*) / ||g||^2
    f_lower = 1.0  # conservative lower bound
    
    v = np.zeros(n)  # velocity for heavy ball
    iter_count = 0
    restart_interval = 500
    best_at_restart = best_score
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        
        # Compute subgradient
        k = int(np.argmax(b))
        g = np.zeros(n)
        m1 = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
        if len(m1) > 0:
            g[m1] += a[k - m1]
        m2 = np.arange(max(0, k), min(k + n, n))
        if len(m2) > 0:
            g[m2] += a[m2 - k]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        step = (M_curr / (norm**2) - f_lower / (norm**2)) / (g_norm**2 / norm**2 + 1e-10)
        step = max(1e-6, min(step, 1.0))  # clip step
        
        # Heavy ball update
        v = momentum * v - step * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        iter_count += 1
        a = a_new
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Periodic restart
        if iter_count % restart_interval == 0:
            if best_score < best_at_restart - 1e-8:
                best_at_restart = best_score
                v = np.zeros(n)  # Reset momentum
            else:
                # Restart from best
                a = best_a.copy()
                v = np.zeros(n)
                # Reduce step size
                momentum = max(0.5, momentum * 0.95)
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-HB iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, mom={momentum:.3f}")
    
    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weight updates) for the non-smooth max(b) problem."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Mirror descent with KL divergence
    # a_new[j] = a[j] * exp(-eta * g[j]) / Z
    # where g is the subgradient, eta is the step size
    # Step size: eta_t = sqrt(2 * log(n) / (t * D)) where D is the diameter
    # Use adaptive step size
    
    eta0 = 0.1
    eta = eta0
    avg_a = np.zeros(n)
    n_avg = 0
    iter_count = 0
    best_eta = eta0
    best_score_history = best_score
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        k = int(np.argmax(b))
        
        # Subgradient
        g = np.zeros(n)
        m1 = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
        if len(m1) > 0:
            g[m1] += a[k - m1]
        m2 = np.arange(max(0, k), min(k + n, n))
        if len(m2) > 0:
            g[m2] += a[m2 - k]
        
        # Normalize gradient
        g_max = np.max(np.abs(g))
        if g_max < 1e-15:
            break
        g = g / g_max
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        iter_count += 1
        a = a_new
        avg_a += a
        n_avg += 1
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_eta = eta
        
        # Check ergodic average periodically
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        # Adaptive step size: reduce if not improving
        if iter_count % 100 == 0:
            if sc > best_score_history:
                eta *= 0.9
                best_score_history = best_score
            else:
                best_score_history = best_score
                eta = min(eta * 1.05, 1.0)  # Try increasing slightly
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")
    
    return best_a, best_score


def targeted_peak_shave(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find low-contribution positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]
            
            for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 800:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    return 2.0 * cross[n - 1: n - 1 + n]

def compute_subgradient(a):
    """Compute subgradient of max(conv(a,a)) at point a."""
    b = fast_autocorr(a)
    k = int(np.argmax(b))
    n = len(a)
    g = np.zeros(n)
    m1 = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
    if len(m1) > 0:
        g[m1] += a[k - m1]
    m2 = np.arange(max(0, k), min(k + n, n))
    if len(m2) > 0:
        g[m2] += a[m2 - k]
    return g


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def subgradient_heavy_ball(a0, time_limit, verbose=True):
    """Subgradient method with heavy ball momentum and periodic restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    momentum = 0.85
    v = np.zeros(n)
    iter_count = 0
    restart_interval = 300
    best_at_restart = best_score
    step_scale = 0.5

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        g = compute_subgradient(a)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Normalize gradient direction
        g_dir = g / g_norm

        # Adaptive step size
        step = step_scale / (np.sqrt(iter_count + 1.0))
        step = max(1e-5, min(step, 1.0))

        v = momentum * v - step * g_dir
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        iter_count += 1
        a = a_new

        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_at_restart = best_score

        if iter_count % restart_interval == 0:
            # Restart from best with reduced momentum
            a = best_a.copy()
            v = np.zeros(n)
            momentum = max(0.5, momentum * 0.92)
            step_scale = max(0.1, step_scale * 0.85)

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-HB iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, mom={momentum:.3f}, step={step_scale:.4f}")

    return best_a, best_score


def mirror_descent_mwu(a0, time_limit, verbose=True):

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
