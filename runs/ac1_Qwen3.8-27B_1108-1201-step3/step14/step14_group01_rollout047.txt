import cvxpy as cp
import time
from scipy.fft import rfft, irfft

        try:

            prob.solve(solver=s)

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T_rows(a, rows):
    """Build constraint matrix rows for given row indices. T[k,i] = a[i] if 0<=k-i<n."""
    n = len(a)
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        T[idx, lo:hi] = a[lo:hi]
    return T

def lp_best_response(a, M, active_rows, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_rows(a, active_rows)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lp_best_response_cutting_plane(a, M, solver, max_rounds=10, init_frac=0.5):
    """LP with cutting-plane constraint management."""
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_conv_self(a)
    
    # Start with top fraction of constraints
    n_init = max(50, int(size * init_frac))
    top_idx = np.argsort(conv_a)[-n_init:]
    active = np.sort(top_idx)
    
    b = None
    for round_i in range(max_rounds):
        b = lp_best_response(a, M, active, solver)
        if b is None:
            return None
        b_sum = b.sum()
        if b_sum < 1e-15:
            return None
        
        # Check violations
        if len(active) < size:
            check = fast_conv_cross(a, b)
            violated = np.where(check > M * (1 + 1e-6))[0]
            # Only check inactive constraints
            active_set = set(active.tolist())
            violated_inactive = [v for v in violated if v not in active_set]
            if violated_inactive:
                # Add most violated constraints
                violated_arr = np.array(violated_inactive)
                violation_amounts = check[violated_arr] - M
                worst = violated_arr[np.argsort(violation_amounts)[-min(50, len(violated_arr)):]]
                active = np.sort(np.concatenate([active, worst]))
                continue
        
        break
    
    return b

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for the non-smooth max-conv objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(80.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE gradient
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient computation via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # grad_i = 2 * (sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k])
            # First term: cross-correlation
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            # Second term: convolution
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Adaptive step with backtracking
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                continue
            step = norm / (T_val * np.sqrt(n)) * 0.5
            
            a_new = None
            for trial in range(8):
                candidate = a - step * grad
                candidate = np.maximum(candidate, 0.0)
                s_new = candidate.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                candidate = candidate * norm / s_new
                sc = fast_conv_self(candidate).max()
                if sc <= max_c + 0.05 * T_val:
                    a_new = candidate
                    break
                step *= 0.5

            if a_new is not None:
                a = a_new
                sc = fast_conv_self(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si} T={T_val:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and cutting-plane."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP with cutting plane
        b = lp_best_response_cutting_plane(y, M, solver, max_rounds=5, init_frac=0.4)
        if b is None:
            # Fallback: full constraints
            b = lp_best_response(y, M, np.arange(size), solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov momentum update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Periodic average
        if t % 20 == 0:
            # Simple running average check every 20 iterations
            avg_candidate = y.copy()  # y already incorporates averaging through Nesterov
            sc_avg = fast_conv_self(avg_candidate).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_candidate.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov t={t} ({elapsed:.0f}s): M={M:.8f}, best={best_score:.8f}")

    return best_a, best_score

def subgradient_cd(a0, time_limit, rng, verbose=True):
    """Subgradient-guided coordinate descent with smart pair selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 1e9
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Identify peak positions
        peak_threshold = cur_max * 0.98
        peak_positions = np.where(conv >= peak_threshold)[0]
        
        perm = rng.permutation(n)
        round_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Select target j: prefer positions that reduce peak contributions
            # The contribution of a[i] to conv[k] is a[k-i] (for valid k)
            # We want to move mass to j where it contributes least to peaks
            n_candidates = min(12, n)
            j_candidates = rng.integers(0, n, size=n_candidates)
            # Also add structured candidates based on peaks
            for pk in peak_positions[:3]:
                # Moving mass from i to j adds a[pk-j] to conv[pk]
                # We want j where a[pk-j] is small
                candidates_struct = np.argmin(a)[:5]
                for c in candidates_struct:
                    if c != i and c not in j_candidates:
                        j_candidates = np.append(j_candidates, c)
                        break

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                j = int(j)
                if j == i:
                    continue
                aj = a[j]

                # Gain vector: how conv changes when we transfer delta from i to j
                # conv_new[k] = conv[k] + delta*(a[k-j] + a[j-k] - a[k-i] - a[i-k])
                # (ignoring second-order terms at k=2i and k=2j)
                g = np.zeros(size)
                # +a[k-j]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                # +a[j-k]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # -a[k-i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                # -a[i-k]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find best delta in (0, ai)
                # Binary search for the delta that minimizes max(conv + delta*g)
                lo_d, hi_d = 0.0, ai
                for _ in range(12):
                    mid_d = (lo_d + hi_d) / 2.0
                    new_max = (conv + mid_d * g).max()
                    # Second order correction
                    if 2 * i < size:
                        v = conv[2*i] - 2*ai*mid_d + mid_d*mid_d
                        if v > new_max:
                            new_max = v
                    if 2 * j < size:
                        v = conv[2*j] + 2*aj*mid_d + mid_d*mid_d
                        if v > new_max:
                            new_max = v
                    # Also check: for small delta, the max is at the current peak
                    # if g at peak is negative, increasing delta helps
                    # Binary search: if new_max is decreasing, go higher
                    peak_g_max = g[peak_positions].max() if len(peak_positions) > 0 else 0
                    if new_max < best_val:
                        best_val = new_max
                        best_delta = mid_d
                        best_j = j
                    if peak_g_max < 0:
                        lo_d = mid_d
                    else:
                        hi_d = mid_d

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        round_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not round_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 8:
                break
            if no_improve_rounds >= 3:
                # Perturb
                n_perturb = max(10, n // 40)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve_rounds = 0

        if verbose and (idx + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_wrapper(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_wrapper(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_wrapper(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast initial improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 2: LP-Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 2: LP-Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n2}")

    # Phase 3: Subgradient CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Subgradient CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = subgradient_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"cd1_n{n3}")

    # Phase 4: Another LP-Nesterov round
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: LP-Nesterov round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n4}")

    # Phase 5: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.25, 250.0)
    print(f"\n=== Phase 5: Final CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = subgradient_cd(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"cd2_n{len(best_seq)}")

    # Phase 6: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.40, 200.0)
    print(f"\n=== Phase 6: Longer length n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t6_lp = min(t6 * 0.6, 120.0)
        a_out, _ = lp_nesterov_optimize(init_long, t6_lp, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6_cd = min(remaining * 0.5, 80.0)
            a_out2, _ = subgradient_cd(best_seq, t6_cd, rng, verbose=False)
            try_update(a_out2, f"cd_long_{n_long}")

    # Phase 7: Final LSE + CD alternating polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_half = remaining * 0.4
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 60.0), verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = subgradient_cd(best_seq, min(remaining * 0.7, 60.0), rng, verbose=False)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_wrapper(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
