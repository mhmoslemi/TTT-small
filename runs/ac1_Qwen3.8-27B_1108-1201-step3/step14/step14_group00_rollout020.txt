import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_fast_autocorr_cache = {}

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def spectral_flatten(a, n_steps=100, alpha_start=0.7, alpha_end=0.95):
    """Flatten power spectrum via iterative projection."""
    n = len(a)
    a_work = np.maximum(a.astype(np.float64), 0.0)
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    prev_score = fast_autocorr(a_work).max()
    best_a = a_work.copy()
    best_score = prev_score
    
    for it in range(n_steps):
        # Animate alpha from alpha_start to alpha_end
        frac = it / max(n_steps - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        A = rfft(a_work, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Flatten
        new_mag = mag ** alpha
        # Also try: new_mag = mag / (1 + mag) or mag * (1 - alpha) + alpha * mag.mean()

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Blend to maintain continuity
        blend = 0.5
        a_work = (1 - blend) * a_work + blend * a_new
        
    return best_a

        active_set = np.arange(size)

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_n = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a_n)
    f = np.maximum(f, 0)
    s2 = f.sum()
    if s2 < 1e-15:
        return np.ones(new_n)
    return f / s2

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def spectral_flatten(a, n_steps=200, alpha_start=0.6, alpha_end=0.92, verbose=False):
    """Flatten power spectrum via iterative projection (Gerchberg-Saxton)."""
    n = len(a)
    a_work = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a_work.copy()
    best_score = fast_autocorr(a_work).max()
    
    for it in range(n_steps):
        frac = it / max(n_steps - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        A = rfft(a_work, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Flatten: reduce dynamic range
        new_mag = mag ** alpha
        A_new = new_mag * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Soft blend
        a_work = 0.4 * a_work + 0.6 * a_new
        
        if verbose and it % 50 == 0:
            print(f"      spec it={it}, alpha={alpha:.3f}, score={best_score:.10f}")
    
    return best_a

def build_conv_matrix(a, k_indices=None):
    """Build convolution matrix rows for given k indices."""
    n = len(a)
    size = 2 * n - 1
    if k_indices is None:
        k_indices = np.arange(size)
    T = np.zeros((len(k_indices), n))
    for row_idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row_idx, j] = a[idx]
    return T
