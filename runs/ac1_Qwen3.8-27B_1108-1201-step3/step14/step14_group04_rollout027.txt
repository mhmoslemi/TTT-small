import numpy as np

import time
from scipy.fft import rfft, irfft

_solver = None

def get_lp_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.CLARABEL
    return _solver

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 30.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 8.0), n / (T * 1.5))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 300 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()

    return best_a, best_score

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov_enhanced(a0, time_limit, solver, verbose=True):

    s0 = a.sum()

    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    
    # Multi-window averaging
    windows = [50, 100, 200, 500]
    avg_buffers = {w: [] for w in windows}
    
    t0 = time.monotonic()
    n_iters = 0
    stagnation = 0
    best_print_score = best_score
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check scores
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stagnation = 0
        
        # Multi-window averaging
        for w in windows:
            avg_buffers[w].append(a.copy())
            if len(avg_buffers[w]) > w:
                avg_buffers[w].pop(0)
            if len(avg_buffers[w]) == w:
                avg = np.mean(avg_buffers[w], axis=0)
                s_avg = avg.sum()
                if s_avg > 1e-15:
                    avg = avg * norm / s_avg
                    score_avg = fast_autocorr(avg).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg.copy()
                        stagnation = 0
        
        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
            
    return best_a, best_score

def local_search(a0, time_limit, verbose=True):
    """Greedy coordinate descent: try small perturbations of individual elements."""
    rng = np.random.default_rng(42)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Compute autocorrelation for reference
    b = fast_autocorr(a)
    
    # Try perturbing elements in order of their contribution to peaks
    peak_idx = np.argmax(b)
    
    # Build a priority list: elements that contribute to the peak
    js = max(0, peak_idx - n + 1)
    je = min(n - 1, peak_idx)
    peak_elements = list(range(js, je + 1))
    
    # Also add random elements
    all_elements = list(range(n))
    rng.shuffle(all_elements)
    priority = peak_elements + all_elements
    
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for idx in priority:
            if time.monotonic() - t0 > time_limit:
                break
            if a[idx] < 1e-10:
                continue
            # Try moving mass from idx to other positions
            # Find best destination
            best_frac = 0
            best_dst = -1
            best_new_score = best_score
            
            # Try a few fractions
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[idx] * frac
                if amt < 1e-15:
                    continue
                # Try transferring to a few candidates
                # Pick candidates that are far from peak-contributing positions
                candidates = []
                for d in rng.choice(n, size=min(20, n), replace=False):
                    if d != idx:
                        candidates.append(d)
                
                for dst in candidates:
                    a_trial = a.copy()
                    a_trial[idx] -= amt
                    a_trial[dst] += amt
                    if a_trial[idx] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_new_score:
                        best_new_score = sc
                        best_frac = frac
                        best_dst = dst
            
            if best_dst >= 0 and best_new_score < best_score - 1e-14:
                amt = a[idx] * best_frac
                a[idx] -= amt
                a[best_dst] += amt
                if a[idx] < 0:
                    a[idx] = 0
                s_t = a.sum()
                a = a * norm / s_t
                best_score = best_new_score
                best_a = a.copy()
                improved = True
                stall = 0
                iter_count += 1
                if verbose and iter_count % 5 == 0:
                    print(f"    LocalSearch iter={iter_count}: score={best_score:.8f}")
        
        if not improved:
            stall += 1
            if stall > 5:
                break
    
    return best_a, best_score

def smooth_and_reoptimize(a0, kernel_size, time_limit, solver, verbose=True):

    a_smooth = np.convolve(a0, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a0, fast_autocorr(a0).max()
    a_smooth = a_smooth * np.sqrt(2.0 * n) / s
    return lp_nesterov_enhanced(a_smooth, time_limit, solver, verbose)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

_solver_instance = None


def get_lp_solver():
    global _solver_instance
    if _solver_instance is not None:
        return _solver_instance
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_instance = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver_instance = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _solver_instance


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit):

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_enhanced(a0, time_limit, solver, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        if time.monotonic() - t0 > time_limit:
            break

        if b is None:

        b_sum = b.sum()
        if b_sum < 1e-15:

        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        for w in windows:
            avg_buffers[w].append(a.copy())
            if len(avg_buffers[w]) > w:
                avg_buffers[w].pop(0)
            if len(avg_buffers[w]) == w:
                avg = np.mean(avg_buffers[w], axis=0)
                s_avg = avg.sum()
                if s_avg > 1e-15:
                    avg = avg * norm / s_avg
                    score_avg = fast_autocorr(avg).max()
                    if score_avg < best_score:

                        best_a = avg.copy()

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0

    return best_a, best_score


def local_coordinate_search(a0, time_limit):
    """Greedy coordinate descent with mass redistribution."""
    rng = np.random.default_rng(12345)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0

    # Compute initial peak
    b = fast_autocorr(a)
    peak_idx = int(np.argmax(b))

    # Priority: elements contributing to peak
    js = max(0, peak_idx - n + 1)
    je = min(n - 1, peak_idx)
    peak_elements = list(range(js, je + 1))
    rng.shuffle(peak_elements)

    other_elements = [i for i in range(n) if i not in set(peak_elements)]
    rng.shuffle(other_elements)
    priority = peak_elements + other_elements

    while time.monotonic() - t0 < time_limit:
        improved = False
        for idx in priority:
            if time.monotonic() - t0 > time_limit:
                break
            if a[idx] < norm * 1e-8:
                continue

            best_dst = -1
            best_frac = 0.0
            best_new_score = best_score

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[idx] * frac
                if amt < 1e-15:
                    continue

                # Try a few random destinations
                for trial in range(8):
                    dst = int(rng.integers(n))
                    if dst == idx:
                        continue
                    a_trial = a.copy()
                    a_trial[idx] -= amt
                    a_trial[dst] += amt
                    if a_trial[idx] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_new_score:
                        best_new_score = sc
                        best_frac = frac
                        best_dst = dst

            if best_dst >= 0 and best_new_score < best_score - 1e-15:
                amt = a[idx] * best_frac
                a[idx] -= amt
                a[best_dst] += amt
                if a[idx] < 0:
                    a[idx] = 0.0
                s_t = a.sum()
                a = a * norm / s_t
                best_score = best_new_score
                best_a = a.copy()
                improved = True
                stall = 0
                iter_count += 1
                if iter_count % 10 == 0:
                    print(f"    LocalSearch iter={iter_count}: score={best_score:.8f}")

        if not improved:
            stall += 1
            if stall > 8:
                break

    print(f"    LocalSearch done: {iter_count} improvements, best={best_score:.8f}")
    return best_a, best_score


def smooth_sequence(a0, kernel_size):
    """Apply circular smoothing."""
    n = len(a0)
    kernel = np.ones(kernel_size) / kernel_size
    a_smooth = np.convolve(a0, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a0.copy()
    return a_smooth

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_coordinate_search(a_norm, t3)
        a_out = np.maximum(a_out, 0.0)
        a_out = a_out / a_out.sum()
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [local n{n_best}]: {sc:.6f}")

    # === Phase 4: LP at longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP at n={n2} (budget={t4:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t4, solver, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        a_out = a_out / a_out.sum()
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [lp_long n{n2}]: {sc:.6f}")

    # === Phase 5: Final LP polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Final LP n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t5, solver, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        a_out = a_out / a_out.sum()
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [lp_final n{n_best}]: {sc:.6f}")

    # === Phase 6: Smoothing + local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_smooth = smooth_sequence(a_norm, 3)
        a_smooth = a_smooth * np.sqrt(2.0 * n_best) / a_smooth.sum()
        t6 = min(remaining * 0.3, 100.0)
        if t6 > 10:
            print(f"\n=== Phase 6: Smooth + Local n={n_best} (budget={t6:.0f}s) ===")
            a_out, _ = local_coordinate_search(a_smooth, t6)
            a_out = np.maximum(a_out, 0.0)
            a_out = a_out / a_out.sum()
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [smooth n{n_best}]: {sc:.6f}")

    # === Phase 7: Any remaining time - final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.7, 120.0)
        if t7 > 15:
            print(f"\n=== Phase 7: Last LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov_enhanced(a_norm, t7, solver, verbose=False)
            a_out = np.maximum(a_out, 0.0)
            a_out = a_out / a_out.sum()
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [last_lp n{n_best}]: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
