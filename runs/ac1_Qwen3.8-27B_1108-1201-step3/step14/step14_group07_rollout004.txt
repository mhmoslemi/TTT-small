def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work with sum = 1 first
    v = v / target_sum  # Now project onto sum = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    # Ensure exact sum
    proj = proj / proj.sum() * target_sum
    return proj

                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_target / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_a = a_trial.copy()
                            best_score = score
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_target / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0
                        break

            if stall > 200:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.cos(6.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x)
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = 0.8 + 0.2 * np.sin(2.0 * np.pi * x) ** 2
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Hybrid Nesterov + LP (main workhorse, 55% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Hybrid Nesterov+LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.55
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc_out = hybrid_nesterov_optimize(a_norm, t1, solver,
                                                  K_start=3, K_end=8, T=5.0,
                                                  lp_every=30, verbose=True)
        do_try_update(a_out, f"hybrid_n{n_base}")

    # Phase 2: LSE scipy refinement (15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=30.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Coordinate descent polish (20%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.20
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t3, seed=12345)
        do_try_update(a_out, "cd_polish")

    # Phase 4: Second hybrid pass with noisy restart (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Hybrid noisy restart n={n_best} (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.10
    if t4 > 20:
        noise_scale = 0.01 * best_seq.mean()
        a_noisy = best_seq + rng.normal(0, noise_scale, n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = hybrid_nesterov_optimize(a_noisy, t4, solver,
                                             K_start=5, K_end=10, T=3.0,
                                             lp_every=40, verbose=False)
        do_try_update(a_out, f"hybrid_noisy_n{n_best}")

    # Phase 5: Try slightly longer sequence (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 200
    print(f"\n=== Phase 5: Longer seq n={n_long} (budget={remaining:.0f}s) ===")
    t5 = remaining * 0.10
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = hybrid_nesterov_optimize(a_norm, t5, solver,
                                             K_start=3, K_end=8, T=5.0,
                                             lp_every=30, verbose=False)
        do_try_update(a_out, f"long_n{n_long}")

    # Phase 6: Final coordinate descent on best (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.5
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t6, seed=99999)
        do_try_update(a_out, "cd_final")

    # Phase 7: Quick LSE on final best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t7 = remaining * 0.4
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t7, T_start=15.0, T_end=1.0, n_stages=3)
        do_try_update(a_out, "lse_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
