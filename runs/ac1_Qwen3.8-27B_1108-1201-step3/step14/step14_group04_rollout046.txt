def lp_best_response_reduced(a, M, solver, n_constraints=200, warm_start=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M (reduced constraints), b >= 0."""
    n = len(a)

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)

def local_refine(a0, time_limit, seed=42):
    """Local refinement via pairwise mass transfer targeting convolution peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_a = fast_conv(a)
        # Get top peaks
        n_peaks = min(10, len(conv_a))
        top_peaks = np.argsort(conv_a)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks:
            # For peak at lag k, the contribution is sum_j a[j] * a[k-j]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            # Contribution from each j: a[j] * a[k-j]
            j_idx = np.arange(js, je + 1)
            contrib = a[j_idx] * a[k - j_idx]
            
            # Find the pair with maximum contribution
            max_idx = np.argmax(contrib)
            j_star = j_idx[max_idx]
            j_partner = k - j_star
            
            # Try reducing a[j_star] (or a[j_partner]) and adding to a low-contributing position
            # Find positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2+1] = True
            
            # Also include the partner positions
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    partners = k2 - np.arange(js2, je2+1)
                    valid = (partners >= 0) & (partners < n)
                    peak_mask[partners[valid]] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != j_star)[0]
            
            # Sort by current value (prefer moving to low-value positions)
            candidates = candidates[np.argsort(a[candidates])]
            
            for dst in candidates[:5]:
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                    amt = a[j_star] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[j_star] -= amt
                    a_trial[dst] += amt
                    if a_trial[j_star] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_conv(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 200:
                break
    
    return best_a, best_score

def parametric_init(n, variant=0):
    """Generate parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if variant == 0:
        # Uniform
        a = np.ones(n)
    elif variant == 1:
        # Triangular (peak in middle)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif variant == 2:
        # U-shaped (peaks at edges)
        a = (2.0 * x - 1.0) ** 2
    elif variant == 3:
        # Raised cosine
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif variant == 4:
        # Bimodal
        a = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)
    elif variant == 5:
        # Inverse U (dip in middle)
        a = 1.0 - 0.7 * np.exp(-50.0 * (x - 0.5) ** 2)
    elif variant == 6:
        # Quartic U
        a = (2.0 * x - 1.0) ** 4 + 0.1
    elif variant == 7:
        # Sine squared
        a = np.sin(np.pi * x) ** 2
    elif variant == 8:
        # Heavy edges
        a = (2.0 * x - 1.0) ** 6 + 0.05
    elif variant == 9:
        # Three bumps
        a = (np.exp(-30.0 * (x - 0.2) ** 2) + 
             np.exp(-30.0 * (x - 0.8) ** 2) + 
             0.3 * np.exp(-30.0 * (x - 0.5) ** 2))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s
