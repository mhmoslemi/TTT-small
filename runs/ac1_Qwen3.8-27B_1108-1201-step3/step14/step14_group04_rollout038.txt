import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    return _global_solver

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

# LSE gradient (same as before but cleaned up)
def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 30.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 8.0), n / (T * 1.5))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

# NOVEL: Peak-aware coordinate descent
def peak_aware_cd(a0, time_limit, seed=42, verbose=True):
    """
    Novel coordinate descent that identifies peak-contributing positions
    and transfers mass to low-contribution positions.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(b)[::-1][:10]
        
        # Compute contribution of each position to top peaks
        contrib = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib[idx] += a[idx] * a[k - idx]
        
        # Find source: highest contribution position with substantial value
        src_candidates = np.where(contrib > contrib.max() * 0.5)[0]
        if len(src_candidates) == 0:
            src_candidates = np.where(contrib > contrib.max() * 0.1)[0]
        if len(src_candidates) == 0:
            srcCandidates = np.argsort(contrib)[::-1][:20]
            src_candidates = src_candidates if len(src_candidates) > 0 else srcCandidates
        
        # Find destination: low contribution, low value
        dst_candidates = np.where(contrib < contrib.min() * 3 + 1e-10)[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.argsort(contrib)[:20]
        
        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            break
        
        src = src_candidates[rng.integers(len(src_candidates))]
        dst = dst_candidates[rng.integers(len(dst_candidates))]
        
        if src == dst or a[src] < 1e-12:
            stall += 1
            if stall > 500:
                break
            continue
        
        # Try different transfer amounts
        improved = False
        for frac in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            s_t = a_trial.sum()
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1
            else:
                stall += 1
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Peak-CD iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        if stall > 1000:
            break
    
    return best_a, best_score

def multi_scale_optimize(a0, time_limit, solver, verbose=True):
    """
    Work at reduced resolution first, then refine at full resolution.
    """
    n_full = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()
    
    # Try coarse resolutions
    for scale in [4, 2]:

        if n_coarse < 100:
            continue
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 60:
            break
        t_budget = min(remaining * 0.4, 300.0)
        
        # Downsample
        a_coarse = a0[:n_coarse * scale].reshape(n_coarse, scale).sum(axis=1)
        a_coarse = a_coarse / a_coarse.sum()
        
        print(f"  Multi-scale: coarse n={n_coarse} (scale={scale})")
        a_norm = a_coarse * np.sqrt(2.0 * n_coarse) / a_coarse.sum()
        a_out, sc = lp_nesterov(a_norm, t_budget, solver, verbose=False)
        
        # Upsample
        a_up = np.repeat(a_out, scale)[:n_full]
        a_up = a_up / a_up.sum()
        sc_up = fast_autocorr(a_up).max()
        # Note: scoring at full n would be different, but we use it as init
        best_a = a_up
        best_score = sc_up
        print(f"  Multi-scale: upsampled score={sc_up:.8f}")
    
    # Full resolution refinement
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        n_best = len(best_a)
        a_norm = best_a * np.sqrt(2.0 * n_best) / best_a.sum()
        a_out, sc = lp_nesterov(a_norm, remaining * 0.7, solver, verbose=True)
        if sc < best_score:
            best_a = a_out

        # Peak-aware CD polish
        remaining2 = time_limit - (time.monotonic() - t0)

            a_norm2 = best_a * np.sqrt(2.0 * n_best) / best_a.sum()
            a_out2, sc2 = peak_aware_cd(a_norm2, remaining2, seed=777)
            if sc2 < best_score:
                best_a = a_out2

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s

                return s

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L

            a_new = project_simplex(z, 1.0)

                y_new = a_new.copy()
            else:

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def peak_aware_cd(a0, time_limit, seed=42, verbose=True):
    """
    Novel coordinate descent: computes peak contribution per position,
    transfers mass from high-contribution to low-contribution positions.
    Uses multi-peak awareness and adaptive step sizing.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:12]
        peak_vals = b[top_k]

        # Compute contribution of each position to top peaks
        contrib = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib[idx] += a[idx] * a[k - idx]

        # Find source positions: high contribution AND substantial value
        thresh = contrib.max() * 0.4
        src_candidates = np.where((contrib > thresh) & (a > norm * 1e-10))[0]
        if len(src_candidates) == 0:
            src_candidates = np.where(contrib > contrib.max() * 0.1)[0]
        if len(src_candidates) == 0:
            src_candidates = np.argsort(contrib)[::-1][:30]
        src_candidates = src_candidates[np.argsort(contrib[src_candidates])[::-1]]

        # Find destination: low contribution positions
        thresh_dst = contrib.max() * 0.05
        dst_candidates = np.where(contrib < thresh_dst)[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.argsort(contrib)[:30]
        # Prefer positions that are not in any peak's support
        dst_sorted = dst_candidates[np.argsort(contrib[dst_candidates])]

        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            stall += 1
            if stall > 100:
                break
            continue

        improved = False
        # Try multiple source-destination pairs
        n_tries = min(5, len(src_candidates))
        for trial in range(n_tries):
            if improved:
                break
            src = src_candidates[trial % len(src_candidates)]
            dst = dst_sorted[rng.integers(min(20, len(dst_sorted)))]
            if src == dst or a[src] < norm * 1e-12:
                continue

            for frac in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0004]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial * norm / a_trial.sum()
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Random perturbation to escape local attraction
            n_rand = 3
            for _ in range(n_rand):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-8:
                    continue
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                a_trial = a_trial * norm / a_trial.sum()
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
                if stall > 1500:
                    break

        iter_count += 1
        if verbose and iter_count % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Peak-CD iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f} stall={stall}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_out, _ = lse_fista(best_seq, T_sched, t1)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_out = a_out / s
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [lse]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 2: LP Nesterov (MAIN WORKHORSE - most time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.42, 650.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_out = a_out / s
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [lp1]: {sc:.6f} (n={len(a_out)}, t={time.monotonic()-t_global:.0f}s)")

    # === Phase 3: Peak-aware CD (NOVEL) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 25:
        print(f"\n=== Phase 3: Peak-Aware CD n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_aware_cd(a_norm, t3, seed=42, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:
            a_out = a_out / s
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [paccd]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 4: LP second pass (refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 280.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP polish n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()

            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [lp2]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print(f"\n=== Phase 5: Length exploration ===")
        n_current = len(best_seq)
        for dn in [20, -20, 40, -40]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 35:
                break
            n_try = n_current + dn
            if n_try < 100:
                continue
            t5 = min(remaining * 0.35, 140.0)
            print(f"  Trying n={n_try} (budget={t5:.0f}s)")
            init_try = interpolate_sequence(best_seq, n_try)
            a_norm = init_try * np.sqrt(2.0 * n_try) / init_try.sum()
            a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=False)
            a_out = np.maximum(a_out, 0.0)
            s = a_out.sum()
            if s > 0.01:
                a_out = a_out / s
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [len{n_try}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 6: Final peak-aware CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.25, 180.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Final Peak-CD n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_aware_cd(a_norm, t6, seed=77777, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        s = a_out.sum()
        if s > 0.01:

            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:

                best_seq = a_out.copy()
                print(f"  *** NEW BEST [final_paccd]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t7 = min(remaining * 0.85, 300.0)
        if t7 > 25:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=False)
            a_out = np.maximum(a_out, 0.0)
            s = a_out.sum()
            if s > 0.01:
                a_out = a_out / s
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [lp_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
