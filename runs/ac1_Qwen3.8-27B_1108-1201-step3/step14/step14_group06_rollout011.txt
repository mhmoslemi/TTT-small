def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a, n):
    """Build constraint matrix T where (T @ b)[k] = conv(a, b)[k]."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_T(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lse_obj_and_grad(a_flat, n, T_val):
    """LSE-smoothed max autocorrelation objective and its gradient."""
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_val * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_val
    w = exp_vals / exp_vals.sum()
    
    # Gradient: d/da[m] log_sum = (1/s) * sum_k w[k] * 2*a[k-m] ... 
    # Actually, the full objective is 2*n*log_sum/s^2
    # d/da[m] (2n*log_sum/s^2) = 2n * (d log_sum/da[m]) / s^2 - 2n*log_sum * 2*a[m] / s^3
    # d log_sum/da[m] = (1/T) * sum_k w[k] * T * d b[k]/da[m] = sum_k w[k] * 2*a[k-m]
    
    # C[m] = sum_k w[k] * a[k-m] = np.correlate(a, w, 'valid')[m]
    C = np.correlate(a, w, 'valid')
    d_logsum_da = 2.0 * C  # gradient of log_sum w.r.t. a
    
    obj = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * d_logsum_da / (s * s) - 4.0 * n * log_sum * a / (s * s * s)
    return obj, grad


def lse_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=6):
    """Multi-temperature LSE optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = remaining / (n_stages - si)

        def obj_func(x, T_val=T):
            val, _ = lse_obj_and_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_obj_and_grad(x, n, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 500, 'maxfun': 2000,
                                           'gtol': 1e-10, 'ftol': 1e-12,
                                           'maxls': 50})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_val / s_c
                    score_c = fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        print(f"    LSE T={T:.1f}: max_b={score_c:.10f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score

**Idea: Spectral method**

The autocorrelation `conv(a,a)` in the frequency domain is `|FFT(a)|^2`. So `max_k conv(a,a)[k]` is the maximum of the inverse FFT of `|FFT(a)|^2`.

The problem is equivalent to: minimize the maximum of the autocorrelation, subject to `sum(a) = sqrt(2n)` and `a >= 0`.

In the frequency domain, `sum(a) = FFT(a)[0] / sqrt(n)` (approximately, with the right normalization). And `a >= 0` is a constraint in the time domain.

This is a difficult non-convex problem in either domain.

**Idea: Water-filling**

If we ignore the `a >= 0` constraint, the optimal solution would be to make `conv(a,a)` as flat as possible (all values equal to the minimum). This is related to "flat-top" sequences or "causal minimum phase" sequences.

With the `a >= 0` constraint, the solution is more complex, but the intuition is similar: we want the autocorrelation to be as flat as possible.

The LP best-response approach naturally works towards this: it finds the `g` that maximizes `sum(g)` while keeping `conv(y, g)` bounded, which effectively "pushes" the autocorrelation to be more uniform.

**Idea: Iterative reweighted least squares (IRLS)**

At each iteration, weight the peaks more heavily and solve a weighted least-squares problem:
`minimize sum_k w_k * (conv(a,a)[k])^2` subject to `sum(a) = sqrt(2n)`, `a >= 0`.

This is a QP (quadratic program) that can be solved efficiently. The weights `w_k` are set to be high where the autocorrelation is near its maximum.
