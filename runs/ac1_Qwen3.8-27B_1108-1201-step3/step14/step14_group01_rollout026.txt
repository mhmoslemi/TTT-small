def build_T(a):
    """Build the convolution matrix T such that (T @ b)[m] = (a*b)[m]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        # T[m, k] = a[m-k] for k in [lo, hi)
        # (a*b)[m] = sum_k a[k] * b[m-k]... 
        # Actually we want (a*b)[m] = sum_k a[k] b[m-k]
        # So T[m, k] should give the coefficient of b[k] in (a*b)[m]
        # (a*b)[m] = sum_j a[j] b[m-j], so coefficient of b[k] is a[m-k]
        # Valid when 0 <= m-k < n, i.e., m-n+1 <= k <= m
        for k in range(max(0, m - n + 1), min(n, m + 1)):
            T[m, k] = a[m - k]
    return T

def build_T_fast(a):
    """Build T using vectorized operations."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):

        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        if hi > lo:
            T[m, lo:hi] = a[m - np.arange(lo, hi)]
    return T

def lp_solve(a, M, solver, active=None):

            best_val = cur_max
            best_delta = 0.0
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 20
                deltas = np.linspace(0.0, ai, n_samples)
                for d in deltas:
                    if d < 1e-15:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-13:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_pert = max(5, n // 40)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.05 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_gd(a0, time_limit, verbose=True):
    """LogSumExp smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for _ in range(10):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    if fast_conv_self(a_new).max() <= max_c + 0.01 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng, seed_type=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if seed_type == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif seed_type == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif seed_type == 2:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.8) ** 2)
    elif seed_type == 3:
        a = (1.0 - x) ** 2 + 0.1 * x
    elif seed_type == 4:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x) * (1 - x)
    elif seed_type == 5:
        a = np.exp(-2*x) + 0.3
    elif seed_type == 6:
        a = x * (1 - x) ** 1.5 + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
