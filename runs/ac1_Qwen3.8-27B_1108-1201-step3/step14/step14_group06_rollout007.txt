import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None, regularize=0.0):
    """LP/QP best response. If regularize > 0, adds quadratic penalty for sparsity."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    if regularize > 0:
        objective = cp.Maximize(cp.sum(b_var) - regularize * cp.sum_squares(b_var))
    else:
        objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum() + 1e-300) / T_temp
    w = exp_vals / (exp_vals.sum() + 1e-300)

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g

        try:

                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 800,
                                           'gtol': 1e-7, 'ftol': 1e-11,
                                           'maxls': 40})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            pass

    return best_a, best_score


def adaptive_lp_nesterov(a0, time_limit, solver, verbose=True, use_regularization=True):
    """LP best-response with adaptive M, Nesterov, and optional QP regularization."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    fail_count = 0
    eps = 0.01
    reg = 0.01 if use_regularization else 0.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - last_improve > 90.0 and n_iters > 50:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive epsilon: start small, grow with success, shrink with failure
        eps_target = min(0.08, 0.002 + 0.004 * np.sqrt(t + 1))
        if fail_count > 3:
            eps = max(0.003, eps * 0.7)
            fail_count = 0
        else:
            eps = 0.9 * eps + 0.1 * eps_target

        M_used = M_curr * (1.0 - eps)
        T_mat = build_constraint_matrix(y, n)

        b = lp_best_response(y, M_used, solver, T_mat, regularize=reg)

        if b is None:
            # Backtrack: try less aggressive M
            for backtrack in [0.5, 0.3, 0.1]:
                M_bt = M_curr * (1.0 - eps * backtrack)
                b = lp_best_response(y, M_bt, solver, T_mat, regularize=reg)
                if b is not None:
                    eps *= 0.8
                    break
            if b is None:
                fail_count += 1
                if fail_count > 10:
                    reg *= 0.5  # reduce regularization, try pure LP
                    eps = 0.002
                    fail_count = 0
                    continue
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            fail_count += 1
            continue

        g = b * norm_val / b_sum
        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.5)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall_count = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
            stall_count = 0

        # Periodic averaging
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Momentum restart: if stalling, reset momentum
        if time.monotonic() - last_improve > 30.0:
            stall_count += 1
            if stall_count % 10 == 0:
                y = best_a.copy()
                a = best_a.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                if verbose:
                    print(f"    Momentum restart at iter={n_iters}, eps={eps:.4f}, best={best_score:.8f}")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eps={eps:.4f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = np.abs(a[idx] * np.abs(a[k_peak - idx]))

        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        top_peaks = np.argsort(b)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbations
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Try simultaneous multi-coordinate moves
            found = False
            for _ in range(50):
                i1 = rng.integers(n)
                i2 = rng.integers(n)
                j1 = rng.integers(n)
                j2 = rng.integers(n)
                if i1 == j1 or i2 == j2 or i1 == i2 or a[i1] < 1e-10 or a[i2] < 1e-10:
                    continue
                frac1 = rng.uniform(0.01, 0.1)
                frac2 = rng.uniform(0.01, 0.1)
                amt1 = a[i1] * frac1
                amt2 = a[i2] * frac2
                a_trial = a.copy()
                a_trial[i1] -= amt1
                a_trial[i2] -= amt2
                a_trial[j1] += amt1
                a_trial[j2] += amt2
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    found = True
                    break
            if found:
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def semicircle_init(n):
    """Semicircle distribution - related to Wigner law for flat autocorrelation."""
    x = np.linspace(-1.0, 1.0, n)
    a = np.sqrt(np.maximum(1.0 - x**2, 0.0))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def raised_cosine_init(n):
    """Raised cosine (Hann window) initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def parabolic_init(n):
    """Parabolic (inverted) initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 - 4.0 * (x - 0.5)**2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gaussian_init(n, sigma=0.25):
    """Gaussian initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma)**2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        return semicircle_init(n)
    elif choice == 1:
        return raised_cosine_init(n)
    elif choice == 2:
        return parabolic_init(n)
    elif choice == 3:
        return gaussian_init(n, 0.2)
    elif choice == 4:
        return gaussian_init(n, 0.35)
    else:
        return np.ones(n) / n


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search combining LP, LSE, and coordinate descent with multi-length exploration."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2000
        best_seq = semicircle_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Semicircle init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov with regularization on parent length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest reg n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.30
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = adaptive_lp_nesterov(a_norm, t1, solver, verbose=True, use_regularization=True)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_reg_n{n_best}", t_global)

    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()

        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_n{n_best}", t_global)

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t3, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"coord_n{n_best}", t_global)

    # Phase 4: LP without regularization (different dynamics)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP no-reg n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adaptive_lp_nesterov(a_norm, t4, solver, verbose=False, use_regularization=False)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_noreg_n{n_best}", t_global)

    # Phase 5: Try semicircle init at different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_semi = max(n_best + 200, 1800)
    print(f"\n=== Phase 5: Semicircle LP n={n_semi} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 60:
        init_semi = semicircle_init(n_semi)
        a_norm = init_semi * np.sqrt(2.0 * n_semi) / init_semi.sum()
        a_out, _ = adaptive_lp_nesterov(a_norm, t5, solver, verbose=True, use_regularization=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"semi_lp_n{n_semi}", t_global)

    # Phase 6: Try raised cosine at yet another length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cos = n_best - 200
    if n_cos < 1000:
        n_cos = n_best + 400
    print(f"\n=== Phase 6: Raised cosine LP n={n_cos} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 40:
        init_cos = raised_cosine_init(n_cos)
        a_norm = init_cos * np.sqrt(2.0 * n_cos) / init_cos.sum()
        a_out, _ = adaptive_lp_nesterov(a_norm, t6, solver, verbose=True, use_regularization=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"cos_lp_n{n_cos}", t_global)

    # Phase 7: LSE on best so far with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LSE low-T n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t7, T_start=20.0, T_end=0.5, n_stages=3)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_lowT_n{n_best}", t_global)

    # Phase 8: Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord descent n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, remaining * 0.5, seed=12345)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 9: One more LP round from the very best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Final LP polish n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adaptive_lp_nesterov(a_norm, remaining * 0.6, solver, verbose=False, use_regularization=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Falling back to semicircle.")
        n_fb = 1600
        best_seq = semicircle_init(n_fb)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
