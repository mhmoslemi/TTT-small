def build_conv_matrix(a):
    """Build the sparse convolution matrix T where T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(j, j + n):
            rows.append(i)
            cols.append(j)
            vals.append(a[i - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))

def build_conv_matrix_fast(a):
    """Build the sparse convolution matrix more efficiently."""
    n = len(a)
    size = 2 * n - 1
    # Each column j has non-zeros at rows j, j+1, ..., j+n-1
    # with values a[0], a[1], ..., a[n-1]
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        end = min(j + n, size)
        count = end - j
        rows[idx:idx+count] = np.arange(j, end)
        cols[idx:idx+count] = j
        vals[idx:idx+count] = a[:count]
        idx += count
    
    # Remove zeros
    mask = np.abs(vals) > 1e-15
    return csr_matrix((vals[mask], (rows[mask], cols[mask])), shape=(size, n))

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def mirror_descent_best_response(a, M, n_iters=200, eta=None):
    """First-order approximation to the LP best response using mirror descent."""
    n = len(a)
    if eta is None:
        eta = 2.0 / np.sqrt(n)
    
    b = np.ones(n) / n
    best_b = b.copy()
    best_sum = 1.0
    
    for _ in range(n_iters):
        # Compute conv(a, b)
        ab = fast_conv(a, b)
        
        # Check constraint violation
        violation = ab - M
        max_viol = violation.max()
        
        # If constraint is satisfied, we can increase b
        if max_viol < 1e-10:
            # Scale up
            b = b * (1.0 + 0.01)
            continue
        
        # Gradient of the constraint violation w.r.t. b
        # d/d b_j sum_i max(0, (ab)_i - M) * T_{i,j}
        # Approximate: use the most violated constraint
        k_star = np.argmax(ab)
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = a[idx]
        
        # Mirror descent update (multiplicative)
        b = b * np.exp(-eta * g)
        b = np.maximum(b, 1e-10)
        b = b / b.sum()
        
        s = b.sum()
        if s > best_sum:
            best_sum = s
            best_b = b.copy()
    
    return best_b

def subgradient_step(a, k_star, alpha, norm):
    """Take a subgradient step in the direction of the peak at k_star."""
    n = len(a)
    g = np.zeros(n)
    for m in range(n):
        idx = k_star - m
        if 0 <= idx < n:
            g[m] = 2.0 * a[idx]
    
    a_new = a - alpha * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * norm / s
