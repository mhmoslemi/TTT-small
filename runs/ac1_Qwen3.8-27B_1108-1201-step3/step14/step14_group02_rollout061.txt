```python
    fft_size = 1 << (2*n).bit_length()  # Pad to avoid circular correlation issues
    a_curr = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(n_iter):
        A = np.fft.rfft(a_curr, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Flatten the magnitude spectrum
        # Target: make all magnitudes equal to the mean (except DC)
        mean_mag = mag[1:].mean()
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # Keep DC
        target_mag[1:] = mean_mag * target_factor
        
        # Blend: partially flatten
        mag_new = 0.5 * mag + 0.5 * target_mag
        mag_new[0] = mag[0]
        
        # Reconstruct
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m = fast_autocorr(a_new).max()
        if m < best_m:
            best_m = m
            best_a = a_new.copy()
        
        a_curr = a_new
    
    return best_a, best_m
```

```python
def fast_autocorr(a):
    """Compute full autocorrelation (a convolved with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_flatten(a_init, n_iter=5000, blend=0.3, time_limit=None, t0=None):
    """Flatten the power spectrum iteratively (Gerchberg-Saxton style)."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    if t0 is None:
        t0 = time.monotonic()
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, n=fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flatten the magnitude (keep DC)
        if it % 100 == 0:
            # Periodically update target
            mean_mag = np.mean(mag[1:])
            target_mag = np.full_like(mag, mean_mag)
            target_mag[0] = mag[0]
        
        # Blend towards flat
        mag_new = (1 - blend) * mag + blend * target_mag
        mag_new[0] = mag[0]  # Preserve DC
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=fft_size)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m = fast_autocorr(a_new).max()
        if m < best_m:
            best_m = m
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_m

def mirror_descent_optimize(a0, time_limit, top_k=15, beta=3.0, lr_init=0.1, verbose=False):
    """Mirror descent with adaptive multi-lag subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = lr_init / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute soft subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]
        
        # Mirror descent step (exponentiation map)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def build_T_sparse(y, n):
    """Build sparse constraint matrix T where T[k,j] = y[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_range = range(j, j + n)
        y_range = y
        for i, k in enumerate(k_range):
            rows.append(k)
            cols.append(j)
            vals.append(y_range[i])
    T = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T.tocsr()

def lp_frank_wolfe(a0, time_limit, verbose=True, solver=cp.CLARABEL):
    """Nesterov-accelerated Frank-Wolfe with sparse LP best-response."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build sparse T matrix
        T = build_T_sparse(y, n)
        
        # LP: max sum(b) s.t. T @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent polish: move mass to reduce peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Identify the lag(s) achieving (near) the maximum
        threshold = M_curr * 0.95
        peak_lags = np.where(ac >= threshold)[0]
        
        improved = False
        
        for k in peak_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            
            # Find best destination: position that minimizes increase in peaks
            # Score each candidate by how much it increases the autocorrelation peaks
            best_dst = -1
            best_score = float('inf')
            
            # Sample candidates
            n_cand = min(n, 300)
            candidates = rng.integers(0, n, size=n_cand)
            
            for dst in candidates:
                if dst == src:
                    continue
                # Estimate increase in max autocorrelation
                # Adding mass to dst increases conv[k'] by approximately:
                # delta_conv[k'] ≈ 2 * amt * a[|k' - dst|] (for the relevant k')
                # We want to minimize the max of (ac[k'] + delta)
                peak_increase = 0.0
                for k2 in peak_lags[:5]:
                    k2i = int(k2)
                    d = abs(k2i - dst)
                    if d < n:
                        inc = 2.0 * a[d]  # proportional to a[d]
                    else:
                        inc = 0.0
                    # Also subtract the reduction from removing src
                    d_src = abs(k2i - src)
                    dec = 0.0
                    if d_src < n:
                        dec = 2.0 * a[d_src]
                    peak_increase = max(peak_increase, inc - dec)
                
                if peak_increase < best_score:
                    best_score = peak_increase
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            
            # Try various step sizes
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m * (1 - 1e-10):
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        else:
            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening initialization (FFT-based, Gerchberg-Saxton style)
2. Adaptive mirror descent with multi-lag soft subgradient
3. Sparse LP-based Nesterov Frank-Wolfe (using scipy HiGHS solver)
4. Coordinate descent polish with smart destination selection
5. Multi-length scaling exploration (1600 -> 2000 -> 2500)
"""

import numpy as np
from scipy.fft import rfft, irfft
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_autocorr(a):
    """Compute full autocorrelation (a convolved with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_T_sparse(y, n):
    """Build sparse matrix T where T[k,j] = y[k-j] if 0<=k-j<n, else 0."""
    j_idx = np.repeat(np.arange(n), n)
    offset = np.tile(np.arange(n), n)
    k_idx = j_idx + offset
    vals = np.tile(y, n)
    T = sp.coo_matrix((vals, (k_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return T


def lp_best_response(y, M):
    """Solve LP: max sum(b) s.t. T(y)@b <= M, b >= 0. Returns b or None."""
    n = len(y)
    T = build_T_sparse(y, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None


def spectral_flatten(a_init, n_iter=10000, blend=0.3, time_limit=None, t0=None):
    """Flatten the power spectrum iteratively (Gerchberg-Saxton style)."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 1).bit_length()

    if t0 is None:
        t0 = time.monotonic()

    target_mag = None
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, n=fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        if target_mag is None or it % 200 == 0:
            mean_mag = np.mean(mag[1:]) if len(mag) > 1 else mag[0]
            target_mag = np.full_like(mag, mean_mag)
            target_mag[0] = mag[0]

        mag_new = (1.0 - blend) * mag + blend * target_mag
        mag_new[0] = mag[0]

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=fft_size)[:n]

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        m = fast_autocorr(a_new).max()
        if m < best_m:
            best_m = m
            best_a = a_new.copy()

        a = a_new

    return best_a, best_m


def mirror_descent_optimize(a0, time_limit, top_k=20, beta=3.0, lr_init=0.05, verbose=False):
    """Mirror descent with adaptive multi-lag soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        step = lr_init / (np.sqrt(t) + 1.0)

        ac = fast_autocorr(a)
        M_curr = ac.max()

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, verbose=True):
    """Nesterov-accelerated Frank-Wolfe with LP best-response using HiGHS."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    lp_fail_count = 0

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        b_val = lp_best_response(y, M_curr)
        if b_val is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            continue
        lp_fail_count = 0

        b_sum = b_val.sum()
        g = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=12345):
    """Coordinate descent polish: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        threshold = M_curr * 0.9
        peak_lags = np.where(ac >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.argsort(ac)[::-1][:5]

        improved = False

        for k in peak_lags[:6]:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]

            best_dst = -1
            best_score = float('inf')

            n_cand = min(n, 400)
            candidates = rng.integers(0, n, size=n_cand)

            top_peaks = peak_lags[:4]
            for dst in candidates:
                if dst == src:
                    continue
                peak_increase = 0.0
                for k2 in top_peaks:
                    k2i = int(k2)
                    d_dst = abs(k2i - dst)
                    d_src = abs(k2i - src)
                    inc = 2.0 * a[d_dst] if d_dst < n else 0.0
                    dec = 2.0 * a[d_src] if d_src < n else 0.0
                    pi = inc - dec
                    if pi > peak_increase:
                        peak_increase = pi

                if peak_increase < best_score:
                    best_score = peak_increase
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst

            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m * (1.0 - 1e-10):
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        else:
            stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with spectral init, MD, sparse LP, and coordinate polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral flattening (fast FFT-based) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral flatten n={n0} ===")
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 10:
        a_out, _ = spectral_flatten(best_seq, n_iter=50000, blend=0.3, time_limit=t1, t0=time.monotonic())
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "spectral_n0", t_global)

        # Try different blend
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t1b = min(remaining * 0.5, 60.0)
            a_out, _ = spectral_flatten(best_seq, n_iter=50000, blend=0.5, time_limit=t1b, t0=time.monotonic())
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "spectral_b0.5", t_global)

    # === Phase 2: Mirror descent with multiple parameter settings ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror descent n={n0} ===")
    md_configs = [
        (20, 3.0, 0.05),
        (30, 2.0, 0.03),
        (15, 5.0, 0.08),
    ]
    for cfg_i, (tk, bt, lr) in enumerate(md_configs):
        remaining = time_budget - (time.monotonic() - t_global)
        t2 = min(remaining * 0.08, 80.0)
        if t2 < 10:
            break
        print(f"  MD config {cfg_i}: top_k={tk}, beta={bt}, lr={lr}")
        a_out, _ = mirror_descent_optimize(best_seq, t2, top_k=tk, beta=bt, lr_init=lr, verbose=(cfg_i == 0))
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"md{cfg_i}_n{n0}", t_global)

    # === Phase 3: LP Frank-Wolfe at n0 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe n={n_best} ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_out, _ = lp_frank_wolfe(best_seq, t3, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp_n{n_best}", t_global)

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate polish n={n_best} ===")
    t4 = min(remaining * 0.07, 80.0)
    if t4 > 10:
        a_out, _ = coordinate_polish(best_seq, t4, verbose=True, seed=99)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"coord_n{n_best}", t_global)

    # === Phase 5: Scale to n=2000 ===
    n1 = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Scale to n={n1} ===")
    t5_total = min(remaining * 0.22, 330.0)
    if t5_total > 30:
        a_long = interpolate_sequence(best_seq, n1)

        # Spectral flatten at new length
        t5a = t5_total * 0.2
        a_out, _ = spectral_flatten(a_long, n_iter=50000, blend=0.25, time_limit=t5a, t0=time.monotonic())
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "spectral_n1", t_global)

        # MD at new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t5b = min(remaining * 0.2, 100.0)
            a_out, _ = mirror_descent_optimize(best_seq, t5b, top_k=20, beta=3.0, lr_init=0.03, verbose=True)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "md_n1", t_global)

        # LP at new length (if feasible)
        remaining = time_budget - (time.monotonic() - t_global)
        n_best = len(best_seq)
        if remaining > 30 and n_best <= 2100:
            t5c = min(remaining * 0.4, 180.0)
            if t5c > 20:
                a_out, _ = lp_frank_wolfe(best_seq, t5c, verbose=True)
                best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lp_n1", t_global)

    # === Phase 6: Scale to n=2500 ===
    n2 = 2500
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Scale to n={n2} ===")
    t6_total = min(remaining * 0.15, 200.0)
    if t6_total > 30:
        a_long2 = interpolate_sequence(best_seq, n2)

        t6a = t6_total * 0.3
        a_out, _ = spectral_flatten(a_long2, n_iter=50000, blend=0.2, time_limit=t6a, t0=time.monotonic())
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "spectral_n2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.4, 80.0)
            a_out, _ = mirror_descent_optimize(best_seq, t6b, top_k=25, beta=3.0, lr_init=0.02, verbose=True)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "md_n2", t_global)

    # === Phase 7: Final coordinate polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} ===")
    if remaining > 15:
        t7 = min(remaining * 0.12, 100.0)
        if t7 > 10:
            a_out, _ = coordinate_polish(best_seq, t7, verbose=True, seed=777)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "coord_final", t_global)

    # === Phase 8: Final LP polish (if length allows) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} ===")
    if n_best <= 2100 and remaining > 30:
        t8 = min(remaining * 0.3, 200.0)
        if t8 > 20:
            a_out, _ = lp_frank_wolfe(best_seq, t8, verbose=True)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lp_final", t_global)

    # === Phase 9: Alternative initialization exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Alternative inits n={n_best} ===")

        # Try cosine-based init
        x = np.linspace(0, 1, n_best)
        alt1 = 0.5 * (1.0 + np.cos(2.0 * np.pi * x - np.pi))
        alt1 = np.maximum(alt1, 0.0)
        alt1 = alt1 / alt1.sum()

        # Blend with best
        blend1 = 0.6 * norm_sum1(best_seq) + 0.4 * alt1
        blend1 = norm_sum1(blend1)

        t9 = min(remaining * 0.15, 60.0)
        if t9 > 10:
            a_out, _ = spectral_flatten(blend1, n_iter=50000, blend=0.3, time_limit=t9, t0=time.monotonic())
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "alt1_spectral", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.3, 50.0)
            a_out, _ = mirror_descent_optimize(best_seq, t9b, top_k=20, beta=4.0, lr_init=0.04, verbose=False)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "alt1_md", t_global)

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t10 = min(remaining * 0.2, 60.0)
        if t10 > 5:
            a_out, _ = coordinate_polish(best_seq, t10, verbose=False, seed=31415)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "coord_very_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
