def pair_cd(a, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent for local polish."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()

    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue
            
            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break
                    
                    # Compute delta in autocorrelation from moving mass from i to j
                    # a[i] -= d, a[j] += d
                    # Delta in conv[k] = -d * (row_i + col_i) + d * (row_j + col_j)
                    # where row_i[k] = a[k-i] if valid, col_i[k] = a[i-(k-n+1)] if valid
                    # More precisely: conv[k] changes by -d*a[k-i] - d*a[i-(k)] + d*a[k-j] + d*a[j-k]... 
                    # Actually let's just use the formula:
                    # conv_new[k] = conv[k] + 2d*(cross_corr terms)
                    
                    # Simpler: compute the change vector
                    delta_vec = np.zeros(size)
                    # Effect of decreasing a[i] by d:
                    # conv[k] -= d * (a[k-i] + a[i+k-n+1]...) 
                    # Let me just compute it directly using the structure
                    # conv[k] = sum_j a[j]*a[k-j]
                    # d(conv[k])/d(a[i]) = 2*a[k-i] (if 0<=k-i<n)
                    # So delta from a[i]-=d: -d * 2*a[k-i] for valid k-i
                    # delta from a[j]+=d: +d * 2*a[k-j] for valid k-j
                    # Also need to handle the diagonal terms: conv[2i] gets -d^2 (from removing a[i]^2)
                    # and conv[2j] gets +d^2 + 2*d*a[j]
                    
                    # For the linear part (ignoring d^2 for now):
                    for k in range(max(0, i), min(size, i + n)):
                        delta_vec[k] -= 2.0 * a[k - i]
                    for k in range(max(0, j), min(size, j + n)):
                        delta_vec[k] += 2.0 * a[k - j]
                    
                    # Now search for best d in [0, a[i]]
                    # conv_new = conv + d * delta_vec + quadratic terms
                    # For simplicity, use linear approximation + golden section
                    best_d = 0.0
                    best_val = conv.max()
                    
                    # Sample a few values of d
                    n_samples = 10
                    for d in np.linspace(0, a[i] * 0.9, n_samples):
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        # Add quadratic corrections
                        if 2 * i < size:
                            new_conv[2*i] += -2*d*a[i] + d*d  # was a[i]^2, now (a[i]-d)^2, change = -2*a[i]*d + d^2
                        if 2 * j < size:
                            new_conv[2*j] += 2*d*a[j] + d*d  # was a[j]^2, now (a[j]+d)^2, change = 2*a[j]*d + d^2
                        # Cross terms: conv[i+j] gets +2*d*a[i] - 2*d*a[j]... 
                        # Actually the cross term between i and j: originally a[i]*a[j] appears in conv[i+j] and conv[|i-j|]
                        # After change: (a[i]-d)*(a[j]+d) = a[i]*a[j] + d*(a[i]-a[j]) - d^2
                        # So conv[i+j] (if i+j < size) changes by 2*(d*(a[i]-a[j]) - d^2) [factor 2 because it appears twice]
                        if i + j < size:
                            new_conv[i+j] += 2*(d*(a[i] - a[j]) - d*d)
                        if abs(i-j) < size and i != j:
                            new_conv[abs(i-j)] += 2*(d*(a[i] - a[j]) - d*d)
                        
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d
                    
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def smooth_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated smooth optimization with periodic LP refinement."""
    global n_1
    n_1 = len(a0)
    n = n_1
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    lp_interval = 200

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: start moderate, decrease to sharp
        progress = min(1.0, t / 50000.0)
        T = 0.3 * (1.0 - 0.95 * progress) + 0.015

        # Compute gradient at y
        grad, max_c, w = compute_smooth_gradient(y, fft_size, T)

        if max_c < 1e-14:
            break

        # Periodic LP refinement for precision
        if t % lp_interval == 0:
            M_curr = fast_autocorr(y).max()
            T_mat = np.zeros((size, n))
            for j in range(n):
                T_mat[j:j + n, j] = y[j]
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T_mat @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                prob = None
            if prob is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum > 1e-15:
                    g = b_val * norm / b_sum
                    alpha = 2.0 / (t + 2.0)
                    y_new = (1.0 - alpha) * y + alpha * g
                    y_new = np.maximum(y_new, 0.0)
                    s_y = y_new.sum()
                    if s_y < 1e-15:
                        y_new = y.copy()
                    else:
                        y_new = y_new * norm / s_y

                    if t == 1:
                        a_new = y_new.copy()
                    else:
                        momentum = (t - 1.0) / (t + 2.0)
                        a_new = y_new + momentum * (y_new - y)
                        a_new = np.maximum(a_new, 0.0)
                        s_a = a_new.sum()
                        if s_a < 1e-15:
                            a_new = y_new.copy()
                        else:
                            a_new = a_new * norm / s_a

                    y = y_new
                    a = a_new
                    n_iters += 1

                    score_y = fast_autocorr(y).max()
                    if score_y < best_score:
                        best_score = score_y
                        best_a = y.copy()
                    score_a = fast_autocorr(a).max()
                    if score_a < best_score:
                        best_score = score_a
                        best_a = a.copy()
                    continue

        # Gradient step with adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            lr = 1e-3
        else:
            lr = 0.8 / (grad_norm + 1e-10) * min(1.0, 50.0 / (n * T))

        # Try both additive and multiplicative, use the better one
        a_add = y - lr * grad
        a_add = np.maximum(a_add, 0.0)
        s_add = a_add.sum()
        if s_add < 1e-15:
            a_add = y.copy()
        else:
            a_add = a_add * norm / s_add

        # Multiplicative update
        log_step = np.clip(-lr * grad, -5.0, 5.0)
        a_mult = y * np.exp(log_step)
        s_mult = a_mult.sum()
        if s_mult < 1e-15:
            a_mult = y.copy()
        else:
            a_mult = a_mult * norm / s_mult

        # Score both and pick better
        score_add = fast_autocorr(a_add).max()
        score_mult = fast_autocorr(a_mult).max()
        if score_mult < score_add:
            a_candidate = a_mult
        else:
            a_candidate = a_add

        # Nesterov convex combination
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_candidate
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        n_iters += 1

        # Track best (check every few iterations)
        if n_iters % 3 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth iter={n_iters} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent: move mass between pairs to reduce max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Compute delta vector for moving mass d from i to j
                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] -= a[lo1 - i:hi1 - i]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        idx2 = np.arange(lo2, hi2)
                        src2 = n - 1 - idx2
                        valid = (src2 >= 0) & (src2 < n)
                        delta_vec[lo2:hi2][valid] -= a[src2[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] += a[lo3 - j:hi3 - j]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        idx4 = np.arange(lo4, hi4)
                        src4 = n - 1 - idx4
                        valid4 = (src4 >= 0) & (src4 < n)
                        delta_vec[lo4:hi4][valid4] += a[src4[valid4]]

                    delta_vec *= 2.0

                    # Search for best step
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 12
                    for d in np.linspace(0.0, a[i] * 0.95, n_samples):
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        # Quadratic corrections
                        if 2 * i < size:
                            new_conv[2 * i] += -2 * d * a[i] + d * d
                        if 2 * j < size:
                            new_conv[2 * j] += 2 * d * a[j] + d * d
                        # Cross terms
                        if i + j < size:
                            new_conv[i + j] += 2 * (d * (a[i] - a[j]) - d * d)
                        if abs(i - j) < size and i != j:
                            new_conv[abs(i - j)] += 2 * (d * (a[i] - a[j]) - d * d)

                        new_max = new_conv.max()
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_d = d

                    if best_d > 1e-12:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-14:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Smooth Nesterov - main workhorse (55% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: Smooth Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc = smooth_nesterov_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"smooth1_n{n_base}")

    # Phase 2: Pair CD polish (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Pair CD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_optimize(a_norm, t2, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcd1_n{n_best}")

    # Phase 3: Second smooth pass with lower temperature (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Smooth Nesterov round 2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smooth_nesterov_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"smooth2_n{n_best}")

    # Phase 4: Final pair CD with wider window (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Final Pair CD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_optimize(a_norm, t4, rng, window=50, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 5: Try alternate length if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 90:
        n_alt = n_base + 200
        print(f"\n=== Phase 5: Alternate length n={n_alt} (budget={remaining * 0.7:.0f}s) ===")
        t5 = remaining * 0.7
        if t5 > 40:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            t5a = min(t5 * 0.65, remaining * 0.4)
            a_out_alt, _ = smooth_nesterov_optimize(a_norm_alt, t5a, solver, verbose=False)
            do_try_update(a_out_alt, f"smooth_n{n_alt}")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 20 and best_seq is not None and len(best_seq) == n_alt:
                a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                a_out_alt2, _ = pair_cd_optimize(a_norm_alt2, min(remaining2 * 0.5, 60.0), rng, window=25, verbose=False)
                do_try_update(a_out_alt2, f"pcd_n{n_alt}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
