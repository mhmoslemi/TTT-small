def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def fast_conv_cross(a, b):

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)

def subgradient_nesterov(a0, time_limit, verbose=True):
    """Fast subgradient method with Nesterov acceleration, no LP needed."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Lower bound: 2n/(2n-1) for normalized problem
    f_lb = 2.0 * n / (2 * n - 1)

    y = a.copy()
    x = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < f_lb * (1 + 1e-10):
            break

        # Find active lags (within 1% of max for subgradient averaging)
        threshold = M * 0.99
        active = np.where(conv_y >= threshold)[0]

        # Compute subgradient: weighted average of 2*T_k*a for active k
        grad = np.zeros(n)
        for k in active:
            # (T_k * a)[i] = a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(k + 1, n)
            if hi > lo:
                indices = np.arange(lo, hi)
                grad[indices] += a[k - indices]
            # Diagonal term: if 2*i == k, add a[i]/2
            if k % 2 == 0:
                mid = k // 2
                if mid < n:
                    grad[mid] += 0.5 * a[mid]

        grad *= 2.0  # factor of 2 from the quadratic form gradient

        # Polyak step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        # Use the gap: f(y) - f_lb
        gap = M - f_lb
        if gap < 1e-12:
            break
        step = gap / grad_norm_sq

        # Nesterov update for y
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * x
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Update x with projected gradient step
        x_new = y_new - step * grad
        x_new = np.maximum(x_new, 0.0)
        s_x = x_new.sum()
        if s_x < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new * norm / s_x

        y = y_new
        x = x_new

        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_x = fast_conv_self(x).max()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f}, gap={M-f_lb:.8f}")

    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(80.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP Nesterov with adaptive constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Adaptive screening: start loose, tighten over time
    initial_frac = 0.75
    final_frac = 0.95
    n_total = 100000

    for t in range(1, n_total + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive threshold
        frac = initial_frac + (final_frac - initial_frac) * min(1.0, t / (n_total * 0.5))
        threshold = M * frac
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 10:
            active = np.where(conv_y >= M * 0.5)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints (every 50 iterations or if screening is active)
        if active is not None and len(active) < size and (t % 50 == 0 or t < 10):
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 1e-6))[0]
            if len(violated) > 0:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score

def peak_cd_optimize(a0, time_limit, rng, verbose=True):
    """Coordinate descent targeting the peak lag specifically."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_lag = int(np.argmax(conv))
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Focus on j values that contribute to the peak
            # Peak at lag k*: contribution from a[i] is a[k*-i] + a[i-k*] (if in range)
            # We want to move mass from positions where they contribute to the peak
            # to positions where they don't

            # Candidate j's: positions far from the peak contribution
            j_candidates = set()
            # Random candidates
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            # Structured: positions that minimize contribution to peak
            # If peak at k*, then a[i] contributes a[k*-i] at the peak.
            # Moving to j reduces peak by a[k*-i]+a[i-k*] and adds a[k*-j]+a[j-k*]
            # So we want j where a[k*-j]+a[j-k*] is small
            if 0 <= peak_lag - i < n:
                pass  # i contributes to peak directly
            # Try j = i + offset that moves away from peak
            for offset in [n // 4, n // 3, n // 2, 2 * n // 3, 3 * n // 4]:
                j1 = (i + offset) % n
                j2 = (i - offset) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 25
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d < 1e-15:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v - 1e-14:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 4 or no_improve == 6:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and t0 > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score

def simulated_annealing_lp(a0, time_limit, solver, rng, verbose=True):
    """Simulated annealing with LP restarts to escape local optima."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    current_score = best_score
    T_init = 0.05
    T_min = 0.001
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        # Temperature schedule
        progress = min(1.0, n_iters / 200.0)
        T = T_init * (T_min / T_init) ** progress

        # Random perturbation
        n_perturb = rng.integers(5, max(6, n // 20))
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_pert = a.copy()
        a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p < 1e-15:
            continue
        a_pert = a_pert * norm / s_p

        # Quick LP optimization from perturbed point
        t_lp = min(remaining * 0.1, 30.0)
        a_out, _ = nesterov_lp_optimize(a_pert, t_lp, solver, verbose=False)
        sc_out = fast_conv_self(a_out).max()

        # Accept/reject
        delta = sc_out - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_out
            current_score = sc_out
            if sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                if verbose:
                    print(f"    SA iter={n_iters} ({elapsed:.0f}s): NEW BEST={best_score:.8f}, T={T:.5f}")

        n_iters += 1
        if verbose and n_iters % 20 == 0:
            print(f"    SA iter={n_iters} ({time.monotonic()-t0:.0f}s): cur={current_score:.8f}, best={best_score:.8f}, T={T:.5f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-10.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-4.0 * x) * 0.5 + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x) * np.cos(3 * np.pi * x)
    elif choice == 8:
        # Three peaks
        a = np.exp(-15*(x-0.2)**2) + 0.6*np.exp(-15*(x-0.5)**2) + 0.4*np.exp(-15*(x-0.8)**2)
    else:
        # Broad Gaussian
        a = np.exp(-2.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Fast subgradient (no LP, very fast per iteration)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 1: Subgradient Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = subgradient_nesterov(best_seq, t1, verbose=True)
        try_update(a_out, "subgrad")

    # Phase 2: LSE gradient descent (fast, smooth)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 2: LSE GD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t2, verbose=True)
        try_update(a_out, f"lse_n{len(best_seq)}")

    # Phase 3: Main LP Nesterov (workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp_main_n{len(best_seq)}")

    # Phase 4: Peak-targeted CD
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Peak CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = peak_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")

    # Phase 5: Simulated annealing with LP restarts
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: Simulated Annealing n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = simulated_annealing_lp(best_seq, t5, solver, rng, verbose=True)
        try_update(a_out, "sa")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length exploration (budget={remaining:.0f}s) ===")
    n_current = len(best_seq)
    candidate_lengths = []
    for nn in [n_current - 200, n_current + 200, n_current + 400, n_current + 600]:
        if nn >= 400 and nn <= 4000:
            candidate_lengths.append(nn)
    # Deduplicate and sort
    candidate_lengths = sorted(set(candidate_lengths))

    for n_try in candidate_lengths[:3]:  # Try up to 3 different lengths
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_try = min(remaining * 0.3, 120.0)
        print(f"  Trying n={n_try} (budget={t_try:.0f}s)...")
        init_try = interpolate_seq(best_seq, n_try)
        # Quick LP optimization
        a_out, _ = nesterov_lp_optimize(init_try, min(t_try, 60.0), solver, verbose=False)
        try_update(a_out, f"lp_n{n_try}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = peak_cd_optimize(best_seq if len(best_seq) == n_current else a_out,
                                         min(remaining * 0.4, 40.0), rng, verbose=False)
            try_update(a_out2, f"pcd_n{n_try}")

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    t7 = min(remaining * 0.30, 150.0)
    print(f"\n=== Phase 7: Final LP polish n={n_final} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=True)
        try_update(a_out, "lp_final")

    # Phase 8: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.40, 100.0)
    print(f"\n=== Phase 8: Final CD n={len(best_seq)} (budget={t8:.0f}s) ===")
    if t8 > 15:
        a_out, _ = peak_cd_optimize(best_seq, t8, rng, verbose=True)
        try_update(a_out, "pcd_final")

    # Phase 9: One more subgradient sweep
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 9: Final subgradient (budget={remaining:.0f}s) ===")
        a_out, _ = subgradient_nesterov(best_seq, min(remaining * 0.6, 60.0), verbose=False)
        try_update(a_out, "subgrad_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
