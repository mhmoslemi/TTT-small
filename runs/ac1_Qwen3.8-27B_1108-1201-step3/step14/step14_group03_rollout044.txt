import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")

    _global_solver = cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 efficiently."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    restart_threshold = 50  # restart Nesterov after this many iters without improvement
    last_best_iter = 0
    
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_best_iter = n_iters

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_best_iter = n_iters

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg

            t = 1

def smooth_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) on the smooth approximation of max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    p = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = p.sum()
    if s0 < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s0  # normalize to probability vector

    best_p = p.copy()
    best_score = fast_autocorr(p).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Temperature schedule: start high, decrease
    n_stages = 8
    temps = np.geomspace(50.0, 0.5, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(500, int(remaining * 3)))

        # Learning rate schedule within this stage
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute autocorrelation
            c = fast_autocorr(p)
            max_c = c.max()

            # Soft weights
            shifted = (c - max_c) / T
            # Clip for numerical stability
            shifted = np.clip(shifted, -500, 500)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Compute gradient: g[j] = sum_k w[k] * p[k-j] (with proper bounds)
            # g[j] = sum_i w[j+i] * p[i] for i=0..n-1 where j+i < size
            # Using FFT: g = (w_rev * p)[n-1 : 2n-1][::-1] where w_rev = w[::-1]
            w_rev = w[::-1]
            conv_size = len(w_rev) + n - 1
            fft_conv = 1 << (conv_size - 1).bit_length()
            W_rev_fft = rfft(w_rev, fft_conv)
            P_fft = rfft(p, fft_conv)
            conv_w = irfft(W_rev_fft * P_fft, fft_conv)[:conv_size]
            g = conv_w[n-1:2*n-1][::-1]

            # Gradient of F_T wrt p is (2/T) * g
            grad = (2.0 / T) * g

            # Mirror descent update (multiplicative weights)
            # lr = T / n (standard for mirror descent with entropy)
            lr = T / (n * (1.0 + 0.05 * it))
            log_update = -lr * grad
            log_update = np.clip(log_update, -50, 50)
            p_new = p * np.exp(log_update)
            s_new = p_new.sum()
            if s_new < 1e-15:
                continue
            p_new = p_new / s_new
            p = p_new

            # Track best
            score = fast_autocorr(p).max()
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MDS stage T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_p, best_score
