            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)

        je = min(n - 1, k)
        j_idx = np.arange(js, je + 1)

import time

# Global solver
_lp_solver = None

def _get_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return s
        except:
            continue
    _lp_solver = cp.CLARABEL
    return _lp_solver

def fast_auto(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A*A, fft_size)[:size]

def proj_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * S / len(v)
    return S * v / s

def find_active(a, top_k=200):
    """Find top_k lags of the convolution."""
    c = fast_auto(a)
    return np.argsort(c)[::-1][:top_k]

def lp_solve(a, M, lags, solver):
    """Solve max sum(b) s.t. T[lags]@b <= M, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build constraint matrix rows
    rows = []
    for k in lags:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row[idx] = a[k - idx]
        rows.append(row @ b <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), rows)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def lp_nesterov(a0, t_limit, solver, top_k=200, verbose=True):
    """LP Nesterov with constraint reduction."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = S * a / s
    
    best_a = a.copy()
    best_sc = fast_auto(a).max()
    y = a.copy()
    
    # Geometric average
    geo = a.copy()
    geo_w = 1.0
    gamma = 0.99
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < t_limit:
        it += 1
        M = fast_auto(y).max()
        if M < 1e-14:
            break
        
        lags = find_active(y, top_k)
        b = lp_solve(y, M, lags, solver)
        if b is None:
            b = np.ones(n) / n
        
        g = S * b / b.sum()
        
        # Nesterov step
        alpha = 2.0 / (it + 2.0)
        y_new = (1 - alpha) * y + alpha * g
        y_new = S * np.maximum(y_new, 0.0) / max(np.maximum(y_new, 0.0).sum(), 1e-15)
        
        if it == 1:
            a_new = g.copy()
        else:
            a_new = y_new + ((it - 1.0) / (it + 2.0)) * (y_new - y)
            a_new = S * np.maximum(a_new, 0.0) / max(np.maximum(a_new, 0.0).sum(), 1e-15)
        
        y = y_new
        a = a_new
        
        # Update geometric average
        geo = gamma * geo + (1 - gamma) * a
        geo_w = gamma * geo_w + (1 - gamma)
        
        # Check
        sc = fast_auto(y).max()
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        sc = fast_auto(a).max()
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        sc = fast_auto(geo).max()
        if sc < best_sc:
            best_sc = sc
            best_a = geo.copy()
        
        if verbose and it % 200 == 0:
            el = time.monotonic() - t0
            print(f"  LP iter={it} ({el:.0f}s): best={best_sc:.8f} M={M:.6f}")
        
        # Adapt gamma: increase it over time
        if it % 5000 == 0:
            gamma = min(0.999, gamma + 0.0001)
    
    return best_a, best_sc, it

    return _lp_solver


def fast_auto(a):
    """Fast autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_nn_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * S / len(v)
    return S * v / s


def find_top_lags(a, top_k=200):
    """Find the top_k lags of the full convolution of a with itself."""
    c = fast_auto(a)
    idx = np.argsort(c)[::-1][:top_k]
    return idx


def lp_best_response(a, M, lags, solver):
    """Solve max sum(b) s.t. T_lags @ b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in lags:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        row[idx] = a[k - idx]
        constraints.append(row @ b_var <= M)
    if not constraints:
        return np.ones(n) / n
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_nesterov_run(a0, t_limit, solver, top_k=200, gamma_init=0.99, verbose=True, label=""):
    """LP Nesterov with constraint reduction and geometric ergodic averaging."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = S * a / s

    best_a = a.copy()
    best_sc = fast_auto(a).max()
    y = a.copy()

    # Geometric ergodic average
    geo = a.copy()
    geo_w = 1.0
    gamma = gamma_init

    t0 = time.monotonic()
    it = 0
    last_print = t0

    while time.monotonic() - t0 < t_limit:
        it += 1
        M = fast_auto(y).max()
        if M < 1e-14:
            break

        lags = find_top_lags(y, top_k)
        b = lp_best_response(y, M, lags, solver)
        if b is None:
            b = np.ones(n) / n

        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
            b_sum = 1.0
        g = S * b / b_sum

        # Nesterov acceleration
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = S * y_new / s_y

        if it == 1:
            a_new = g.copy()
        else:
            mom = (it - 1.0) / (it + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = S * a_new / s_a

        y = y_new
        a = a_new

        # Update geometric average
        geo = gamma * geo + (1.0 - gamma) * a
        geo_w = gamma * geo_w + (1.0 - gamma)

        # Check scores
        sc_y = fast_auto(y).max()
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()

        sc_a = fast_auto(a).max()
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a.copy()

        sc_geo = fast_auto(geo).max()
        if sc_geo < best_sc:
            best_sc = sc_geo
            best_a = geo.copy()

        # Periodically increase gamma (more weight on recent iterates)
        if it % 5000 == 0:
            gamma = min(0.9995, gamma + 0.0001)

        # Print progress
        if verbose and time.monotonic() - last_print > 30:
            el = time.monotonic() - t0
            print(f"    LP[{label}] iter={it} ({el:.0f}s): best={best_sc:.8f} M={M:.6f} topk={top_k}")
            last_print = time.monotonic()

    return best_a, best_sc, it


def lse_fista(a0, t_limit, verbose=True, label=""):
    """LSE-smoothed FISTA for warm-up."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_sc = fast_auto(a).max()
    t0 = time.monotonic()
    t_fista = 1
    staging = a.copy()

    T_schedule = [10.0, 3.0, 1.0, 0.3, 0.1]
    L = n / 20.0

    for T in T_schedule:
        L = max(L, n / (T * 1.5))
        last_print = t0
        while time.monotonic() - t0 < t_limit:
            c = fast_auto(y)
            c_max = c.max()
            if c_max < 1e-15:
                break
            shifted = (c - c_max) / T
            shifted = np.clip(shifted, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()

            # Gradient computation via FFT
            N = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N)
            Y_rev = rfft(y[::-1], N)
            cross = irfft(W * Y_rev, N)
            grad = 2.0 * cross[n - 1: n - 1 + n]

            step = 1.0 / L
            z = y - step * grad
            a_new = proj_nn_sum(z, 1.0)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = proj_nn_sum(y_new, 1.0)

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if t_fista % 200 == 0:
                sc = fast_auto(a).max()
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

            if verbose and time.monotonic() - last_print > 20:
                el = time.monotonic() - t0
                print(f"    LSE[{label}] T={T:.1f} iter={t_fista} ({el:.0f}s): best={best_sc:.8f}")
                last_print = time.monotonic()

    sc = fast_auto(a).max()
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a * S, best_sc * S / S  # Return normalized to S


def local_search(a0, t_limit, seed=42, verbose=True, label=""):
    """Local search via mass redistribution with simulated annealing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = S * a / s

    best_a = a.copy()
    best_sc = fast_auto(a).max()
    t0 = time.monotonic()
    stall = 0
    T_anneal = 0.01
    last_print = t0

    while time.monotonic() - t0 < t_limit:
        c = fast_auto(a)
        top_k = np.argsort(c)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find low-contribution destination
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            candidates = candidates[np.argsort(a[candidates])][:max(10, len(candidates) // 4)]
            dst = candidates[rng.integers(len(candidates))]

            for frac in [0.0003, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = S * a_trial / s_t
                sc = fast_auto(a_trial).max()
                delta = sc - best_sc
                if delta < -1e-14:
                    a = a_trial
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
                elif delta < 0.01 * T_anneal:
                    # Accept with some probability (annealing)
                    if rng.random() < np.exp(-delta / max(T_anneal, 1e-10)):
                        a = a_trial
                        improved = True
                        break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > S * 1e-8:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = S * a_trial / s_t
                        sc = fast_auto(a_trial).max()
                        if sc < best_sc:
                            a = a_trial
                            best_sc = sc
                            best_a = a.copy()
                            stall = 0
                        elif sc - best_sc < 0.02 * T_anneal and rng.random() < 0.1:
                            a = a_trial

        # Anneal
        if stall > 50:
            T_anneal *= 0.8
            if T_anneal < 0.001:
                T_anneal = 0.001

        if stall > 2000:
            break

        if verbose and time.monotonic() - last_print > 30:
            el = time.monotonic() - t0
            print(f"    LocalSearch[{label}] ({el:.0f}s): best={best_sc:.8f} stall={stall}")
            last_print = time.monotonic()

    return best_a, best_sc


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            p_score = evaluate_safe(parent)
