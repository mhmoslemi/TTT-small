Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient descent with Adam optimizer (FFT-based gradients)
2. Cyclic coordinate descent with exact 1D optimization
3. LP-based Nesterov acceleration (improved constraint handling)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective: 2*n*max_b/(sum_a)^2"""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_grad_fft(a, w):
    """Compute (2/T) * corr(w, a) where corr[j] = sum_m w[j+m]*a[m].
    
    Uses FFT to compute the cross-correlation.
    """
    n = len(a)
    # We need c[j] = sum_{m=0}^{n-1} w[j+m] * a[m] for j=0,...,n-1
    # This is the cross-correlation of w (length 2n-1) and a (length n)
    # c[j] = conv(a, w_rev)[j + (2n-2)] where w_rev = w[::-1]
    
    w_rev = w[::-1]
    # conv(a, w_rev) has length n + (2n-1) - 1 = 3n-1
    conv_size = 3 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    
    A = rfft(a, fft_size)
    W = rfft(w_rev, fft_size)
    conv = irfft(A * W, fft_size)[:conv_size]
    
    # c[j] = conv[j + (2n-2)] for j=0,...,n-1
    c = conv[(2*n-2):(3*n-2)]
    return c


def lse_loss_and_grad(a, T):
    """Compute LSE_T of autocorrelation and its gradient w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    # Shift for numerical stability
    b_shift = b - max_b
    w = np.exp(b_shift / T)
    w_sum = w.sum()
    w = w / w_sum
    
    loss = max_b + T * math.log(w_sum)
    
    # Gradient: (2/T) * c where c[j] = sum_m w[j+m]*a[m]
    c = lse_grad_fft(a, w)
    grad = (2.0 / T) * c
    
    return loss, grad, b, w


def adam_lse_optimize(a0, time_limit, T_start=None, T_end=None, verbose=True):
    """LSE-smoothed gradient descent with Adam optimizer."""
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    if T_start is None:
        T_start = compute_score(a) * 0.1
    if T_end is None:
        T_end = compute_score(a) * 0.001
    T_start = max(T_start, 1e-4)
    T_end = max(T_end, 1e-6)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    steps = 0
    
    # Adam hyperparameters
    lr = 1.0 / n
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        loss, grad, b, w = lse_loss_and_grad(a, T)
        
        # Project gradient onto tangent space (sum constraint)
        grad -= np.mean(grad)
        
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        score = compute_score(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adapt learning rate based on progress
        if steps % 500 == 0:
            # Check if we're making progress
            current_T_score = max_b_approx = b.max()  # approximate
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AdamLSE step={steps} ({elapsed:.0f}s): T={T:.6f}, score={best_score:.8f}, loss={loss:.6f}")
        
        # Learning rate schedule
        lr_eff = lr * (1.0 - 0.3 * elapsed_frac)
        
    return best_a, best_score

            else:

            ks = np.arange(k_start, k_end + 1)
            ks = ks[ks != 2*j]  # exclude k=2j
            if len(ks) > 0:

                bases = b[ks] - 2.0 * aj * a[ks - j]  # base for each k

            # f(t) = max(t^2 + C2j, max(coeffs*t + bases), max_unaffected)
            # This is convex in t >= 0

            # t^2 + C2j <= current_max → t <= sqrt(current_max - C2j)
            current_max = max(b.max(), 0.0)
            t_hi = math.sqrt(max(current_max - C2j, 0.0)) + aj + 1.0
            t_hi = max(t_hi, 1.0)
            t_lo = 0.0
            
            def f_of_t(t):
                val = t * t + C2j
                if len(ks) > 0:
                    val = max(val, np.max(coeffs * t + bases))
                if max_unaffected > -np.inf:
                    val = max(val, max_unaffected)
                return val
            
            # Golden section search
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            c = t_hi - gr * (t_hi - t_lo)
            d = t_lo + gr * (t_hi - t_lo)
            fc = f_of_t(c)
            fd = f_of_t(d)
            
            for _ in range(40):

                    d = c
                    fd = fc
                    c = t_hi - gr * (t_hi - t_lo)
                    fc = f_of_t(c)
                else:
                    t_lo = c
                    c = d
                    fc = fd
                    d = t_lo + gr * (t_hi - t_lo)
                    fd = f_of_t(d)
            
            t_opt = (t_lo + t_hi) / 2.0
            
            # If t_opt differs significantly from current, update
            if abs(t_opt - aj) > 1e-10 * max(aj, 1.0):
                a[j] = t_opt
                improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a * norm / s
        
        # Recompute autocorrelation
        b = fast_autocorr(a)
        score = b.max()  # Since sum = norm = sqrt(2n), score = max_b
        actual_score = compute_score(a)
        
        if actual_score < best_score:
            best_score = actual_score
            best_a = a.copy()

            break
    
    return best_a, best_score

        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)
        
        # Temperature: use a more aggressive cooling
        # T = T_start * (T_end/T_start)^(elapsed_frac^0.7) for faster initial cooling
        progress = elapsed_frac ** 0.6
        T = T_start * (T_end / T_start) ** progress
        T = max(T, T_end)
        
        lse_val, grad, b, max_b = lse_grad(a, T)
        
        # Project gradient to tangent space (sum constraint)
        grad_mean = np.mean(grad)
        grad_tan = grad - grad_mean
        
        steps += 1
        m = beta1 * m + (1 - beta1) * grad_tan
        v = beta2 * v + (1 - beta2) * grad_tan ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        # Learning rate with warmup and decay
        warmup_steps = 200
        if steps < warmup_steps:
            lr_scale = steps / warmup_steps
        else:
            lr_scale = 1.0 - 0.5 * elapsed_frac
        lr_eff = lr * lr_scale
        
        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        score = compute_score(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Periodic logging
        if verbose and steps % 2000 == 0:
            print(f"    AdamLSE step={steps} ({elapsed:.0f}s): T={T:.6f}, score={best_score:.8f}")
        
        # Stall detection and restart
        if steps % 5000 == 0 and score > best_score * 1.01:
            n_restarts += 1
            if n_restarts > max_restarts:
                break
            # Perturb and reset Adam state
            rng = np.random.default_rng(steps)
            a = a + rng.normal(0, 0.02 * a.mean(), n)
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            steps = 0
            if verbose:
                print(f"    AdamLSE restart #{n_restarts} at step {steps}, score={best_score:.8f}")
    
    return best_a, best_score


def solve_lp(a, M, solver, verbose=False):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0.
    Returns optimal b or None."""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix T: T[k, j] such that (T @ b)[k] = (a*b)[k]
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
    # = sum_{j: 0<=k-j<n} a[k-j] * b[j]
    # For fixed k, j ranges from max(0, k-n+1) to min(n-1, k)
    
    # We'll use a sparse-like approach: for each j, a[j] contributes to b[k]
    # for k in [j, j+n-1]. Specifically, (a*b)[k] includes a[j]*b[k-j] for k in [j, j+n-1].
    # So T[k, j] = a[j] if j <= k <= j+n-1, else 0.
    # But this is the same as: T @ b where T[k,j] = a[k-j] if 0<=k-j<n.
    
    # For efficiency with cvxpy, let's use the full constraint matrix
    # but only include active constraints.
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Compute current a*b to find active constraints
    ab = fast_autocorr(a)  # This is a*a, not a*b... 
    # We need a convolved with b, but b is the variable.
    # The constraint is (a * b_var)[k] <= M for all k.
    # (a * b_var)[k] = sum_j a[j] * b_var[k-j] (with valid indices)
    # = sum_j b_var[j] * a[k-j] (with valid indices, substituting j -> k-j)
    
    # Build T matrix: T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a banded matrix (bandwidth n)
    
    # For efficiency, use the fact that T is a Toeplitz-like matrix
    # T[k, j] = a[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
    
    # Build as a matrix (for n up to ~2000, this is a 3999 x 2000 matrix)
    # That's about 8M entries, which is manageable.
    
    # Actually, let's use a more memory-efficient approach.
    # We'll express the constraint as:
    # For each j: a[j] * b_var[j] appears in (a*b)[k] for k in [j, j+n-1]
    # So (a*b)[k] = sum_{j: j<=k<=j+n-1} a[j] * b_var[j]
    
    # Let's build the matrix row by row
    # T has shape (2n-1, n)
    # T[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # Actually, it's simpler: T[k, j] = a[k-j] if 0 <= k-j <= n-1
    # This means for row k, the non-zero entries are at columns j where 0 <= k-j <= n-1
    # i.e., k-(n-1) <= j <= k, intersected with [0, n-1]
    
    # Let's just build it directly
    T = np.zeros((size, n))
    for j in range(n):
        # a[j] appears in rows k = j, j+1, ..., j+n-1
        # T[k, j] = a[j]... wait no.
        # (a*b)[k] = sum_i a[i]*b[k-i]
        # Term with b[j]: a[k-j]*b[j], valid when 0<=k-j<n
        # So T[k, j] = a[k-j] if 0 <= k-j < n
        ks = range(max(0, k_start := 0), min(size, n + j))
        # Let me just do it more carefully
        pass
    
    # Simpler: for each j, T[k, j] = a[k-j] for k in [j, j+n-1]
    # (since k-j ranges from 0 to n-1)
    for j in range(n):
        k_start = j
        k_end = min(size - 1, j + n - 1)
        # k-j ranges from 0 to k_end-j
        T[k_start:k_end+1, j] = a[0:k_end-j+1]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    
    return b_val


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov (Fessler-Parks) optimization."""
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.6  # Fessler-Parks momentum parameter
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        
        b_y = fast_autocorr(y)
        M = float(b_y.max())
        
        if M < 1e-14:
            break
        
        # Solve LP
        b_opt = solve_lp(y, M, solver)
        
        if b_opt is None:
            # If LP fails, try with relaxed M
            b_opt = solve_lp(y, M * 1.01, solver)
            if b_opt is None:
                n_iters += 1
                continue
        
        b_sum = b_opt.sum()
        if b_sum < 1e-15:
            continue
        
        g = b_opt * norm / b_sum
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        score_y = compute_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = compute_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPNest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
        
        # Adaptive restart
        if stall > 200:
            rng = np.random.default_rng(n_iters * 7 + 3)
            y = np.maximum(y + rng.normal(0, 0.02 * a.mean(), n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = min(gamma + 0.1, 2.0)
            if verbose:
                print(f"    LPNest restart at iter {n_iters}")
        
        # Adaptive gamma
        if n_iters % 200 == 0 and n_iters > 100:
            # Check oscillation
            b_y2 = fast_autocorr(y)
            M2 = float(b_y2.max())
            if abs(M2 - M) / max(M, 1e-14) > 0.02:
                gamma = max(gamma - 0.1, 1.0)
    
    return best_a, best_score


def get_solver():
    """Find the best available LP solver."""
    if not HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    return None


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving (normalized) shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
