"""

1. Fast LP best-response with reduced constraint set (top-K peaks) and Nesterov acceleration
2. Adaptive M targeting with periodic relaxation to escape local minima
3. Spectral-domain perturbations for diversity
4. Targeted coordinate descent for final polish

Key improvements over previous: faster LP via reduced constraints, better M schedule,
spectral restarts to break local minima, and more aggressive peak targeting.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  Solver (fallback): ECOS")
    return cp.ECOS


def _fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    ffs = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), ffs)
    return irfft(A * A, ffs)[:size]


def _eval(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _next_pow2(m):
    return 1 << (m - 1).bit_length()


def _lp_bresp(a, M, solver, K=150):
    """LP best response using only top-K peak constraints for speed."""
    n = len(a)
    b = _fast_conv(a)
    top_k = np.argsort(b)[::-1][:K]
    T = np.zeros((K, n))
    for row, k in enumerate(top_k):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        T[row, idx] = a[k - idx]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = np.asarray(b_var.value, dtype=np.float64)
    if bv is None:
        return None
    bv = np.maximum(bv, 0.0)
    if bv.sum() < 1e-15:
        return None
    return bv


def _spectral_perturb(a, rng, sigma=0.05):
    """Perturb in frequency domain while maintaining non-negativity and sum."""
    n = len(a)
    ffs = _next_pow2(2 * n)
    A = rfft(a, ffs)
    freq = np.arange(len(A))
    w = np.exp(-freq ** 2 / (2.0 * max(n // 8, 1) ** 2))
    noise = rng.standard_normal(len(A)) * sigma * w
    A_new = A * (1.0 + noise)
    a_new = irfft(A_new, ffs)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new * (a.sum() / s)


def _nesterov_lp(a0, time_limit, solver, K=150, verbose=True, eps_base=0.025):
    """Nesterov-accelerated LP best-response with adaptive M and spectral restarts."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    rng = np.random.default_rng(777)

    y = a.copy()
    t0 = time.monotonic()
    last_improve = t0
    stall = 0
    n_iter = 0
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = _fast_conv(y)
        M_curr = b_y.max()
        if M_curr < 1e-14:
            break

        # Adaptive M: starts tight, gradually relaxes to avoid infeasibility
        if t < 50:
            eps = eps_base * 0.5
        elif t < 500:
            eps = eps_base
        else:
            eps = eps_base * (1.0 + 0.5 * min((t - 500) / 2000.0, 1.0))
        M_use = M_curr * (1.0 - eps)

        b_resp = _lp_bresp(y, M_use, solver, K)
        if b_resp is None:
            M_use2 = M_curr * 1.005
            b_resp = _lp_bresp(y, M_use2, solver, K)
            if b_resp is None:
                break

        b_sum = b_resp.sum()
        if b_sum < 1e-15:
            break
        g = b_resp * norm_val / b_sum
        n_iter += 1

        # Nesterov step with adaptive alpha
        alpha = 2.0 / (t + 2.0)
        # Adapt alpha based on improvement ratio
        M_new_g = _fast_conv(g).max()
        if M_curr > 0 and M_new_g > 0:
            ratio = min(M_new_g / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.1), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Track improvement
        sc_y = _fast_conv(y).max()
        if sc_y < best_score - 1e-13:
            best_score = sc_y
            best_a = y.copy()
            last_improve = time.monotonic()

        sc_a = _fast_conv(a).max()
        if sc_a < best_score - 1e-13:
            best_score = sc_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Averaging (Polyak-Ruppert)
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                sc_avg = _fast_conv(avg_check).max()
                if sc_avg < best_score - 1e-13:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Spectral perturbation periodically
        if n_iter % 300 == 0 and time.monotonic() - last_improve > 20:
            a_pert = _spectral_perturb(a, rng, sigma=0.08)
            sc_p = _fast_conv(a_pert).max()
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_pert.copy()
                a = a_pert.copy()
                y = a.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                last_improve = time.monotonic()
            else:
                # Apply mild perturbation to escape, continue
                a = _spectral_perturb(a, rng, sigma=0.03)
                y = a.copy()
                avg_a = np.zeros(n)
                n_avg = 0

        # Hard stall detection
        if time.monotonic() - last_improve > 50.0:
            stall += 1
            if stall > 2:
                if verbose:
                    print(f"    Stall at iter={n_iter}, best={best_score:.8f}")
                break
        else:
            stall = 0

        if verbose and n_iter % 100 == 0:
            print(f"    LP-Nest iter={n_iter} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_conv(a)
        k_peak = int(np.argmax(b))

        # Find elements contributing most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:12]

        # Find low-value targets (away from peak region)
        peak_mask = np.zeros(2 * n - 1, dtype=bool)
        top_peaks = np.argsort(b)[::-1][:12]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            for i in range(js2, je2 + 1):
                peak_mask[i] = True
                if k2 - i < 2 * n:
                    peak_mask[k2 - i] = True
        non_peak = np.where(~peak_mask[:n])[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:12]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = _fast_conv(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random moves
            for _ in range(60):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = _fast_conv(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 100:
                break

    return best_a, best_score


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _structured_init(n, rng):
    choice = rng.integers(0, 5)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(6 * np.pi * x)
    elif choice == 3:
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.7 * np.exp(-5 * (x - 0.7) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _update_best(best_seq, best_score, candidate, label, t_global):
    if candidate is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = _eval(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
