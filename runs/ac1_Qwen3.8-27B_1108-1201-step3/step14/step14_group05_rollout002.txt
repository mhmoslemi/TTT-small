def build_toeplitz(a):
    """Build the Toeplitz matrix T where T @ b = conv(a, b)."""

def smoothed_gradient_descent(a0, time_limit, verbose=True):
    """FFT-based log-sum-exp smoothed gradient descent with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    steps = 0
    
    # Temperature schedule
    c_init = fast_autocorr(a)
    T_init = max(c_init.max() * 0.1, 1e-3)
    T_final = max(c_init.max() * 0.001, 1e-6)
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.5 / n
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    
    # FFT size for gradient computation
    fft_size = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential temperature decay
        T = T_init * (T_final / T_init) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Compute weights for log-sum-exp
        # Use shifted b to avoid overflow
        b_shifted = b - max_b
        log_w = b_shifted / T
        # Clip to avoid overflow in exp
        log_w = np.clip(log_w, -50, 50)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Compute gradient via FFT
        # g[j] = 2 * sum_k w[k] * a[k-j] (where a[m]=0 for m out of [0,n))
        # This equals 2 * cross_corr(a, w)[j]
        # = 2 * convolve(flip(a), w)[n-1+j]
        a_flip = a[::-1]
        A_flip = rfft(a_flip, fft_size)
        W_padded = rfft(w, fft_size)
        C = irfft(A_flip * W_padded, fft_size)
        grad = 2.0 * C[n-1 : 2*n-1]
        
        # Project gradient onto the tangent space of {sum(a) = norm}
        grad -= np.mean(grad)
        
        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        # Adaptive learning rate
        lr = lr_base * (1.0 - 0.5 * progress)  # Decay learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
        
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGrad step={steps} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}")
        
        # Periodic restart if stuck
        if steps % 1000 == 0:
            # Check if we're making progress
            pass
    
    return best_a, best_score
