def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a

def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()

    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    y = a.copy()
    t0 = time.monotonic()

def compute_smooth_grad(a, w):
    """Compute gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    Formula: grad_i = 2 * sum_j w[i+j] * a[j] (with w[k]=0 for k>=2n-1)
    Computed via: grad[i] = 2 * conv(w_rev, a)[2n-2-i]
    """
    n = len(a)
    size = 2 * n - 1
    w_rev = w[::-1].copy()
    conv_len = size + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    WR = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_result = irfft(WR * A, fft_size)
    indices = (2 * n - 2) - np.arange(n)
    return 2.0 * conv_result[indices]


def smoothed_gradient_optimize(a0, time_limit, verbose=True):
    """Smoothed gradient descent with Adam, annealing temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    grad_fft_size = 1 << (size + n - 2).bit_length()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-10
    base_lr = 2.0 / np.sqrt(n)

    # Temperature schedule
    T_start = 8.0
    T_end = 0.03
    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)

    global_t = 0
    stag = 0

    for stage in range(n_stages):
        T = temps[stage]
        lr = base_lr / (1.0 + 0.3 * stage)

        while time.monotonic() - t0 < time_limit:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 0.1:
                break
            # Estimate iterations possible
            iters_this_stage = int(min(remaining * 300, max(50, remaining * 50)))
            if iters_this_stage < 1:
                break

            for it in range(iters_this_stage):
                if time.monotonic() - t0 > time_limit:
                    break

                # Compute convolution
                A = rfft(a, fft_size)
                conv = irfft(A * A, fft_size)[:size]
                max_conv = conv.max()

                # Softmax weights (numerically stable)
                shifted = (conv - max_conv) / T
                exp_vals = np.exp(shifted)
                w = exp_vals / exp_vals.sum()

                # Compute gradient
                grad = compute_smooth_grad(a, w)

                # Scale gradient by 1/T (derivative of log-sum-exp w.r.t. T gives the right scale)
                # Actually the gradient of T*log(sum(exp(conv/T))) w.r.t. a is:
                # sum_k w_k * (1/T) * d(conv_k)/da = (1/T) * grad_smooth
                # But we want to minimize max(conv), and T*log(sum(exp(conv/T))) approximates max(conv)
                # as T->0. The gradient of this approximation is (1/T) * sum_k w_k * d(conv_k)/da
                # = (1/T) * (grad computed above) / 2 * 2 = (1/T) * grad
                # Wait, I computed grad = 2 * sum_k w_k * d(conv_k)/da_i / 2... let me recheck.
                # My compute_smooth_grad returns 2 * sum_j w[i+j]*a[j] which equals
                # 2 * sum_k w_k * d(conv_k)/da_i ... no.
                # Actually d(conv_k)/da_i = 2*a[k-i] (from my derivation)
                # So sum_k w_k * d(conv_k)/da_i = 2 * sum_k w_k * a[k-i] = 2 * sum_j w[i+j]*a[j]
                # And my compute_smooth_grad returns exactly this: 2 * sum_j w[i+j]*a[j]
                # So grad = sum_k w_k * d(conv_k)/da_i (the gradient of sum_k w_k * conv_k w.r.t. a_i)
                # The gradient of T*log(sum(exp(conv_k/T))) w.r.t. a_i is:
                #   sum_k [exp(conv_k/T) / sum_j exp(conv_j/T)] * (1/T) * d(conv_k)/da_i
                #   = (1/T) * sum_k w_k * d(conv_k)/da_i
                #   = grad / T
                # So the actual gradient to use is grad / T.

                grad_scaled = grad / T

                # Adam update
                global_t += 1
                m = beta1 * m + (1 - beta1) * grad_scaled
                v = beta2 * v + (1 - beta2) * grad_scaled * grad_scaled
                m_hat = m / (1 - beta1**global_t)
                v_hat = v / (1 - beta2**global_t)

                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-15:
                    a_new = a.copy()
                else:
                    a_new = a_new * norm / s
                a = a_new

                # Check score (every few iterations for speed)
                if it % 5 == 0:
                    A_check = rfft(a, fft_size)
                    conv_check = irfft(A_check * A_check, fft_size)[:size]
                    score = conv_check.max()
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()

        # End of stage
        if verbose:
            A_check = rfft(a, fft_size)
            conv_check = irfft(A_check * A_check, fft_size)[:size]
            final_score = conv_check.max()
            if final_score < best_score:
                best_score = final_score
                best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    SG stage {stage+1}/{n_stages} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

        # Reset Adam when changing temperature significantly
        if stage < n_stages - 1 and temps[stage + 1] < temps[stage] * 0.5:
            m = np.zeros(n)
            v = np.zeros(n)
            global_t = 0

    return best_a, best_score


def pair_cd_window(a, time_limit, rng, window=30, verbose=True):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Compute delta in convolution from moving mass from i to j
                    delta_vec = np.zeros(size)
                    # Contribution of a[i] to conv
                    delta_vec[i:i+n] -= a[i]
                    lo = max(0, n - 1 - i)
                    hi = min(size, n - 1 + i + 1)
                    delta_vec[lo:hi] -= a[i]
                    # Contribution of a[j] to conv (added)
                    delta_vec[j:j+n] += a[j]
                    lo2 = max(0, n - 1 - j)
                    hi2 = min(size, n - 1 + j + 1)
                    delta_vec[lo2:hi2] += a[j]

                    # Binary search for best d (amount to move from i to j)
                    lo_d = 0.0
                    hi_d = a[i] * 0.999
                    best_d = 0.0
                    best_val = conv.max()

                    for _ in range(8):
                        mid_d = (lo_d + hi_d) / 2.0
                        if mid_d <= 0:
                            lo_d = mid_d
                            continue
                        new_conv = conv + mid_d * delta_vec
                        # Fix diagonal terms
                        new_conv[2 * i] = conv[2 * i] - 2 * a[i] * mid_d + mid_d * mid_d
                        new_conv[2 * j] = conv[2 * j] + 2 * a[j] * mid_d + mid_d * mid_d
                        new_max = new_conv.max()
                        if new_max < best_val:
                            best_val = new_max
                            best_d = mid_d
                            lo_d = mid_d
                        else:
                            hi_d = mid_d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def single_cd(a, time_limit, rng, verbose=True):
    """Single coordinate descent: try to optimally set each coordinate."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    for _ in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]

            # conv_new[k] = conv[k] + d * (a[k-i] + a[i-k]) for valid k
            # = conv[k] + d * delta_k where delta_k = a[k-i] + a[i-k] (0 if out of range)
            # But also the diagonal term: conv[2i] changes by 2*a[i]*d + d^2

            delta = np.zeros(size)
            # Forward part: a[k-i] for k=i..i+n-1
            delta[i:i+n] += a
            # Backward part: a[i-k] for k=.. let's think
            # a[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            # delta[k] += a[i-k] for k in [max(0, i-n+1), min(i, size-1)]
            lo = max(0, i - n + 1)
            hi = min(i + 1, size)
            if hi > lo:
                # a[i-k] for k = lo..hi-1 means a[i-lo..i-hi+1]
                # = a reversed segment
                idxs = np.arange(i - lo, i - hi + 1, -1)
                idxs = idxs[(idxs >= 0) & (idxs < n)]
                if len(idxs) > 0:
                    delta[lo + np.arange(len(idxs))] += a[idxs]

            # The objective after change: conv + d*delta + d^2 at position 2i
            # We want to minimize max(conv + d*delta) subject to a[i]+d >= 0
            # This is a 1D minimization: find d that minimizes max_k (conv[k] + d*delta[k])
            # (ignoring the d^2 term at 2i for now as it's small)

            # The function max_k (conv[k] + d*delta[k]) is piecewise linear in d
            # Minimum occurs where the "active" constraints cross
            # Simple approach: ternary search on d in [-a[i], a[i]*3]

            d_lo = -a[i] * 0.999
            d_hi = a[i] * 3.0
            if a[i] < norm * 1e-12:
                d_hi = norm * 0.1

            best_d = 0.0
            best_val = conv.max()

            # Grid search first
            n_grid = 20
            d_grid = np.linspace(d_lo, d_hi, n_grid)
            vals = np.array([np.max(conv + d * delta) for d in d_grid])
            # Find min
            imin = np.argmin(vals)
            if vals[imin] < best_val - 1e-12:
                # Refine with ternary search
                lo = d_grid[max(0, imin - 1)]
                hi = d_grid[min(n_grid - 1, imin + 1)]
                for _ in range(15):
                    m1 = lo + (hi - lo) / 3
                    m2 = hi - (hi - lo) / 3
                    v1 = np.max(conv + m1 * delta)
                    v2 = np.max(conv + m2 * delta)
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1
                best_d = (lo + hi) / 2
                best_val = np.max(conv + best_d * delta)

            if abs(best_d) > 1e-15:
                a[i] += best_d
                if a[i] < 0:
                    a[i] = 0
                    best_d = -a[i]  # undo
                    a[i] -= best_d
                    continue
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SCDC done ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            p_score = evaluate_safe(p)
            print(f"Parent: n={len(p)}, score={p_score:.6f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov (35% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining * 0.35:.0f}s) ===")
    t1 = remaining * 0.35
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Smoothed gradient Adam (30% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Smoothed Gradient n={n_best} (budget={remaining * 0.30:.0f}s) ===")
    t2 = remaining * 0.30
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_gradient_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sg_n{n_best}")

    # Phase 3: Pair CD (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Pair CD n={n_best} (budget={remaining * 0.20:.0f}s) ===")
    t3 = remaining * 0.20
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t3, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 4: Second round - LP + SG from current best (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP+SG Round 2 n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t4 = remaining * 0.10
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        t4a = t4 * 0.5
        a_out, _ = nesterov_lp_optimize(a_norm, t4a, solver, verbose=False)
        do_try_update(a_out, f"lp2_n{n_best}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10 and best_seq is not None:
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out2, _ = smoothed_gradient_optimize(a_norm2, min(remaining2 * 0.6, 60.0), verbose=False)
            do_try_update(a_out2, f"sg2_n{n_best2}")

    # Phase 5: Alternate length exploration (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_base + 200
        print(f"\n=== Phase 5: Alt length n={n_alt} (budget={remaining * 0.70:.0f}s) ===")
        t5 = remaining * 0.70
        if t5 > 30:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            t5a = min(t5 * 0.5, remaining * 0.3)
            a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t5a, solver, verbose=False)
            do_try_update(a_out_alt, f"lp_n{n_alt}")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 20 and best_seq is not None and len(best_seq) == n_alt:
                a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                a_out_alt2, _ = smoothed_gradient_optimize(a_norm_alt2, min(remaining2 * 0.4, 40.0), verbose=False)
                do_try_update(a_out_alt2, f"sg_n{n_alt}")

    # Phase 6: Final pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 6: Final PCD n={n_best} (budget={remaining * 0.80:.0f}s) ===")
        t6 = remaining * 0.80
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t6, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
