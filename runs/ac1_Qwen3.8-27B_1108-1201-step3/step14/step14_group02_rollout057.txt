Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with adaptive constraint generation and Nesterov acceleration (scipy HiGHS)
2. Hard subgradient mirror descent with adaptive top-k peaks

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_rows(y, top_idx):
    """Build rows of T matrix for given top lags. T[k,j] = y[k-j] if 0<=k-j<n."""
    n = len(y)
    k_act = len(top_idx)
    T_sub = np.zeros((k_act, n))
    for i in range(k_act):
        k_int = int(top_idx[i])
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            T_sub[i, j_start:j_end] = y[k_int - j_start:k_int - j_start + (j_end - j_start)]
    return T_sub

def solve_lp_fast(T_sub, M, n):
    """Solve: max sum(b) s.t. T_sub @ b <= M, b >= 0 using scipy HiGHS."""
    c = -np.ones(n)
    b_ub = np.full(T_sub.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T_sub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        M_curr = ac.max()

        k_act = min(k_act, 2 * n - 1)

        top_idx = np.argsort(ac)[::-1][:k_act]
        T_sub = build_constraint_rows(y, top_idx)

        b_val = solve_lp_fast(T_sub, M_curr, n)
        if b_val is None:
            # Fallback: use uniform
            b_val = np.ones(n) / n

        b_sum = b_val.sum()
        if b_sum < 1e-15:
            b_val = np.ones(n) / n
            b_sum = 1.0
        g = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta_coeff = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_coeff * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}, k_act={k_act}")

        # Early stop if no improvement in last 200 iterations and t > 500
        if t > 500 and t % 200 == 0:
            if best_m > prev_best_m - 1e-10:
                if verbose:
                    print(f"    FW: Stalled at t={t}, J={2.0*n*best_m:.6f}")
                break
            prev_best_m = best_m

    return best_a, best_m

def hard_subgradient(a0, time_limit, top_k=8, gamma=0.5, verbose=True):
    """Mirror descent with hard subgradient (average of top-k peak gradients)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = a.copy()
    n_avg = 1
    best_avg_m = best_m

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        step = gamma / (np.sqrt(t) + 1.0)

        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]

        grad = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * a[k_int - idx]
        grad /= len(top_idx)

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    HSG t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent with peak-aware moves and adaptive step sizes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall_count = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(8, len(ac))]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start >= j_end:
                continue

            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination using peak sensitivity
            n_cand = min(n, 150)
            # Weight towards positions not contributing to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                peak_mask[s2:e2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < n_cand:
                candidates = np.where(rng.random(n) > 0.3)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            candidates = candidates[candidates != src]
            if len(candidates) > n_cand:
                candidates = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_net = float('inf')

            for dst in candidates:
                dst_i = int(dst)
                net = 0.0
                for k2 in top_k:
                    k2i = int(k2)
                    add = 0.0
                    if 0 <= k2i - dst_i < n:
                        add += a[k2i - dst_i]
                    if 0 <= dst_i - k2i < n:
                        add += a[dst_i - k2i]
                    sub = 0.0
                    if 0 <= k2i - src < n:
                        sub += a[k2i - src]
                    if 0 <= src - k2i < n:
                        sub += a[src - k2i]
                    net += max(0.0, add - sub)
                if net < best_net:
                    best_net = net
                    best_dst = dst_i

            if best_dst < 0:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall_count = 0
                    break

        if not improved:
            stall_count += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall_count = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase LP-FW + subgradient + coordinate polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW at parent length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW n={n0} ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, top_k_start=30, top_k_max=100, verbose=True)
        try_update(a_out, f"lp1_n{n0}")

    # === Phase 2: Hard subgradient at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: HSG n={n0} ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = hard_subgradient(a_in, t2, top_k=8, gamma=0.5, verbose=True)
        try_update(a_out, f"hsg_n{n0}")

    # === Phase 3: Coordinate polish at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord n={n0} ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True)
        try_update(a_out, f"coord_n{n0}")

    # === Phase 4: Scale to 2000, LP-FW ===
    n1 = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW n={n1} (current best n={n_best_cur}) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 30:
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        a_out, _ = lp_frank_wolfe(a_in, t4, top_k_start=30, top_k_max=80, verbose=True)
        try_update(a_out, f"lp_n{n1}")

    # === Phase 5: Coordinate polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord polish n={n_best} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True)
        try_update(a_out, f"coord2_n{n_best}")

    # === Phase 6: Try n=2400 ===
    n2 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP-FW n={n2} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_long2 = interpolate_sequence(best_seq, n2)
        a_in = norm_sum1(a_long2)
        a_out, _ = lp_frank_wolfe(a_in, t6, top_k_start=25, top_k_max=60, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # === Phase 7: Final LP-FW at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW n={n_best} ===")
    if n_best <= 2000:
        t7 = min(remaining * 0.15, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, top_k_start=40, top_k_max=120, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 15:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Quick subgradient at best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: HSG final n={n_best} ===")
        t9 = min(remaining * 0.4, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = hard_subgradient(a_in, t9, top_k=12, gamma=0.3, verbose=False)
            try_update(a_out, f"hsg_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        fallback = np.ones(1600) / 1600
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
