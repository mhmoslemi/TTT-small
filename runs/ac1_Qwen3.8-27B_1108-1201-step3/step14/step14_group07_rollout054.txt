    n = len(a)

                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_positions.update(range(js2, je2 + 1))
            
            candidates = [i for i in range(n) if i not in peak_positions and i != src]
            if not candidates:
                candidates = [i for i in range(n) if i != src]
            candidates.sort(key=lambda i: a[i], reverse=True)
            
            for dst in candidates[:10]:
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = sc
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 10 == 0:
                n_pert = max(5, n // 50)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx_p in indices:
                    a_trial[idx_p] *= rng.uniform(0.3, 2.0)
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm_val / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = sc
                        stall = 0
            if stall > 500:
                break
        
        if time.monotonic() - t0 < time_budget and n > 100:
            pass  # Continue loop
    
    return best_a, best_score
