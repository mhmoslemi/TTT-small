             λ_k >= 0

where T[k,j] = a[k-j] * [0<=k-j<n].

The dual has 2n-1 variables (λ_k) and n constraints. The primal has n variables and 2n-1 constraints.

For large n, the dual might be easier to solve if we can reduce the number of active dual variables (which correspond to the active primal constraints).

Actually, by strong duality, at the optimum, the dual variables λ_k that are positive correspond to the primal constraints that are tight. So the number of positive dual variables equals the number of tight primal constraints, which is at most n (by the dimension of the primal).

This means we can solve the dual with at most n positive variables, which might be faster than solving the primal with 2n-1 constraints.

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)

def conv_gradient(a, k):
    """Compute gradient of conv[k] w.r.t. a.
    Result: grad[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # j ranges over 0..n-1, and we need 0 <= k-j < n
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    j_lo = max(0, k - n + 1)
    j_hi = min(n, k + 1)  # exclusive
    if j_hi > j_lo:
        j_idx = np.arange(j_lo, j_hi)
        src = k - j_idx
        grad[j_lo:j_hi] = 2.0 * a[src]
    return grad

def lp_best_response(a, M, solver, active_set=None, max_constraints=500):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is None or len(active_set) == 0:
        active_set = np.arange(size)
    
    # Limit number of constraints for speed
    if len(active_set) > max_constraints:
        # Take the most important constraints (closest to M)
        active_set = active_set[:max_constraints]
    
    # Build constraint matrix rows for active set
    rows = []
    for k in active_set:
        row = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        if j_hi > j_lo:
            j_idx = np.arange(j_lo, j_hi)
            src = k - j_idx
            row[j_lo:j_hi] = a[src]
        rows.append(row)
    
    T = np.array(rows)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, max_active=300):
    """Nesterov-accelerated LP best-response."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_set = None
    stall_count = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Adaptive active set: constraints where conv is close to max
        if active_set is None or t % 50 == 0:
            rel_tol = 0.1 + 0.05 * np.exp(-t / 200.0)
            active_set = np.where(conv_y > M_curr * (1.0 - rel_tol))[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y > M_curr * 0.7)[0]
        
        b = lp_best_response(y, M_curr, solver, active_set, max_active)
        
        if b is None:
            if t % 10 == 0:
                b = lp_best_response(y, M_curr, solver, None, max_active)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0
        
        if stall_count > 80:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

Now for the coordinate descent. The key insight is:
- At the peak index k*, the gradient is g[j] = 2*a[k*-j] (shifted a)
- Moving mass from coordinate i to coordinate j changes the peak by approximately delta * (g[j] - g[i])
- We want to move mass from high-gradient to low-gradient coordinates

But we need to be careful: moving mass changes the entire convolution, not just the peak. So we need to check the actual improvement.

Here's a more practical approach:
- Find k* = argmax conv
- Compute the gradient g at k*
- For each coordinate i (cyclically), try to reduce a[i] by a small amount and compensate by increasing a[j] for the j with minimum gradient
- Use line search to find the optimal amount

Actually, let me try a different approach for coordinate descent. Instead of the gradient-based approach, let me try a more direct method:

For each pair (i, j), compute the exact change in the maximum convolution value when we move δ from a[i] to a[j]. This is:

- The new max is max_k(conv[k] + δ * (a[k-j] - a[k-i]))

We want to find δ > 0 that minimizes this max. This is a 1D problem that can be solved by bisection.

But this is O(n) per pair evaluation, and we have O(n^2) pairs. That's too slow.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive active set screening
2. Gradient-guided coordinate descent exploiting conv gradient structure
3. Multi-length exploration with interpolation and final polish
Key: fast LP via aggressive constraint selection, systematic CD using
the fact that d(conv[k])/d(a[j]) = 2*a[k-j] (shifted sequence).
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def conv_gradient(a, k):
    """Gradient of conv[k] w.r.t. a. Result[j] = 2*a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n, k + 1)
    if j_hi > j_lo:
        grad[j_lo:j_hi] = 2.0 * a[k - np.arange(j_lo, j_hi)]
    return grad


def lp_best_response(a, M, solver, active_set, max_constraints=400):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is None or len(active_set) == 0:
        active_set = np.arange(2 * n - 1)
    
    if len(active_set) > max_constraints:
        active_set = active_set[:max_constraints]
    
    rows = []
    for k in active_set:
        row = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        if j_hi > j_lo:
            row[j_lo:j_hi] = a[k - np.arange(j_lo, j_hi)]
        rows.append(row)
    
    T = np.array(rows)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, max_active=300):

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

            break

            if len(active_set) < 8:
                active_set = np.where(conv_y >= M_curr * 0.6)[0]
        
        b = lp_best_response(y, M_curr, solver, active_set, max_active)
        
        if b is None:
            if t % 5 == 0:

            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g

        s_y = y_new.sum()

def gradient_cd(a0, time_limit, verbose=True, seed=0):
    """Gradient-guided coordinate descent using conv gradient structure."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve_passes = 0
    fractions = np.array([0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5])
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        k_star = int(np.argmax(conv))
        grad = conv_gradient(a, k_star)
        
        positive_mask = a > norm * 5e-8
        n_pos = np.count_nonzero(positive_mask)
        if n_pos < 2:
            break
        
        n_moves = min(30, 2 * n_pos)
        
        for m in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Select source: high gradient among positive elements
            pos_idx = np.where(positive_mask)[0]
            i = int(pos_idx[rng.integers(len(pos_idx))])
            ai = a[i]
            if ai < norm * 5e-8:
                continue
            
            # Select target: low gradient (prefer small values)
            # Score: gradient + small penalty for large values
            scores = grad + 0.1 * (a / norm)
            candidates = np.where(a < norm * 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            candidates = candidates[candidates != i]
            if len(candidates) == 0:
                continue
            j = int(candidates[np.argmin(scores[candidates])])
            
            # Try several fractions and pick the best
            best_delta = 0.0
            best_new_max = best_score
            best_new_conv = None
            
            for frac in fractions:
                delta = ai * frac
                if delta >= ai or delta < 1e-15:
                    continue
                a[i] -= delta
                a[j] += delta
                new_conv = fast_autocorr(a)
                new_max = new_conv.max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_delta = delta
                    best_new_conv = new_conv
                # Revert
                a[j] -= delta
                a[i] += delta
            
            if best_delta > 0:
                a[i] -= best_delta
                a[j] += best_delta
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 6:
                break
        else:
            no_improve_passes = 0
        
        if verbose and n_passes % 50 == 0:
            print(f"    CD pass={n_passes} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def cyclic_cd(a0, time_limit, verbose=True):

    n = len(a0)

    s0 = a.sum()

    fractions = np.array([0.01, 0.05, 0.1, 0.2, 0.4, 0.7])
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        k_star = int(np.argmax(conv))
        grad = conv_gradient(a, k_star)
        
        # Find best target (minimum gradient)
        j_target = int(np.argmin(grad))
        
        for i in range(n):

                break
            if i == j_target:
                continue

                new_max = new_conv.max()
                if new_max < best_new_max - 1e-14:

                    best_delta = delta

def structured_init(n, seed=0):
    """Generate structured initial sequences."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2 + 0.1
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
