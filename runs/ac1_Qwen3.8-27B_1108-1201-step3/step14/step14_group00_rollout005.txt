    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_rows(a, indices):
    """Build rows of convolution matrix T where (T@b)_k = (a*b)_k for given k indices."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    for k in indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        rows.append(row)
    return np.array(rows) if rows else np.zeros((0, n))


def build_conv_matrix_full(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        T = build_conv_rows(a, active_set)
        if len(T) == 0:
            T = np.ones((1, n)) * 1e-10
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_full(a)
        active_set = np.arange(size)
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None

    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass

    return b_val, active_set, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with dual-based screening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    active_set = None
    duals_cache = None
    stall_count = 0
    restart_count = 0
    n_iters = 0
    size = 2 * n - 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive screening using dual-based prediction
        if active_set is None or t % 50 == 0 or restart_count > 0:
            if duals_cache is not None and len(duals_cache) == size:
                # Use dual weights to predict active constraints
                predicted = duals_cache * conv_y
                threshold = np.percentile(predicted, 85.0)
                active_set = np.where(predicted > threshold)[0]
                # Always include the top 10 peaks
                top10 = np.argsort(conv_y)[-10:]
                active_set = np.union1d(active_set, top10)
            else:
                tol = 0.03 + 0.02 * np.exp(-t / 300.0)
                active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
                if len(active_set) < 10:
                    active_set = np.argsort(conv_y)[-max(20, n // 10):]
            restart_count = 0

        b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)

        # Check for constraint violations and expand if needed
        if b is not None and active_set_new is not None and len(active_set_new) < size:
            complement = np.setdiff1d(np.arange(size), active_set_new)
            if len(complement) > 0:
                T_check = build_conv_rows(y, complement)
                violations = T_check @ b
                viol_idx = np.where(violations > M_curr * (1 + 1e-6))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set_new, complement[viol_idx])
                    if len(active_set) > 0.9 * size:
                        active_set = None
                    b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)

        if b is None:
            if t % 30 == 0:
                active_set = None
                b, active_set_new, duals = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        if duals is not None:
            duals_cache = duals.copy()

        active_set = active_set_new
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > 40:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            duals_cache = None
            stall_count = 0
            restart_count += 1
            if verbose:
                print(f"    LP restart at iter {t} ({time.monotonic()-t0:.0f}s), best={best_score:.10f}")

        if verbose and n_iters % 200 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def peak_driven_cd(a0, time_limit, rng, verbose=False):
    """Coordinate descent targeting peak decomposition."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_peaks = 8

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = 0

        # Find top peaks
        peak_indices = np.argsort(conv)[-max_peaks:]
        peak_weights = np.zeros(size)
        for pk in peak_indices:
            peak_weights[pk] = conv[pk] ** 2
        peak_weights /= peak_weights.sum()

        # Build weighted gradient: g_i = 2 * sum_k w_k * a_{k-i}
        grad = np.zeros(n)
        for pk in peak_indices:
            w = peak_weights[pk]
            for i in range(n):
                ki = pk - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * w * a[ki]

        # Process coordinates in order of gradient (highest pressure first)
        order = np.argsort(grad)[::-1]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Find best target j using gradient heuristic
            # For peak k, moving from i to j changes conv_k by:
            # delta * (2*a_{k-j} - 2*a_{k-i}) (approximately, ignoring diagonal terms)
            # We want to MINIMIZE the peak, so move to j where a_{k-j} is small
            best_gain = -np.inf
            best_j = -1
            best_delta = 0.0

            # Candidate targets: positions with low weighted pressure
            candidate_j = np.argsort(grad)[:min(30, n // 3)]
            # Also add some random candidates
            rand_j = rng.integers(0, n, size=min(10, n // 10))
            candidates = np.unique(np.concatenate([candidate_j, rand_j]))

            for j in candidates:
                if j == i:
                    continue
                # Compute gain vector for moving from i to j
                gain = np.zeros(size)
                # Removing mass at i: subtract contribution
                for k in range(size):
                    ki = k - i
                    if 0 <= ki < n:
                        gain[k] -= 2.0 * a[ki]
                    kj = i - k
                    if 0 <= kj < n and ki != kj:
                        pass  # Already handled by the first term when we iterate all k
                # Actually, let's compute it properly using the convolution structure
                # (a*a)_k change when a_i -= d, a_j += d:
                # = -d * (a_{k-i} + a_{i-k} if valid) + d * (a_{k-j} + a_{j-k} if valid) + d^2 * correction
                # For simplicity, compute the linear part:
                # linear_k = a_{j part} - a_{i part}
                gain = np.zeros(size)
                for k in range(size):
                    contrib_i = 0.0
                    ki = k - i
                    if 0 <= ki < n:
                        contrib_i += a[ki]
                    ki2 = i - k + n - 1  # This is wrong, let me redo
                    # Actually (a*a)_k = sum_{i'} a_{i'} a_{k-i'}
                    # d/ddelta [(a'e)(a'e)]_k where a'_i = a_i - delta, a'_j = a_j + delta
                    # = sum_{i'} (a_{i'} + delta*delta_{i',j} - delta*delta_{i',i})(a_{k-i'} + delta*delta_{k-i',j} - delta*delta_{k-i',i})
                    # Linear in delta:
                    # = sum_{i'} a_{i'} a_{k-i'} + delta * [sum_{i'} a_{k-i'}*delta_{i',j} - sum_{i'} a_{k-i'}*delta_{i',i} 
                    #   + sum_{i'} a_{i'}*delta_{k-i',j} - sum_{i'} a_{i'}*delta_{k-i',i}]
                    # = conv_k + delta * [a_{k-j} - a_{k-i} + a_{j} * delta_{k-j, valid} ... ]
                    
                    # Let me just use the formula: grad_k = 2*a_{k-i} if 0<=k-i<n
                    # So change in conv_k = delta * (2*a_{k-j} if 0<=k-j<n else 0) - delta * (2*a_{k-i} if 0<=k-i<n else 0)
                    g_i = 2.0 * a[k - i] if 0 <= k - i < n else 0.0
                    g_j = 2.0 * a[k - j] if 0 <= k - j < n else 0.0
                    gain[k] = g_j - g_i

                # Find optimal delta: minimize max_k (conv_k + delta * gain_k + delta^2 * diag_k)
                # Diagonal correction: when k = 2i, the -d term creates a d^2 correction
                # conv_{2i} changes by -2*a_i*d + d^2 (from the a_i^2 term)
                # conv_{2j} changes by +2*a_j*d + d^2
                # For now, ignore d^2 terms (valid for small delta)
                
                # Find the delta that minimizes the peak (approximate)
                # The peak after change is max_k(conv_k + delta * gain_k)
                # We want to find delta in [0, a_i] that minimizes this
                # Use a few candidate deltas
                best_d = 0.0
                best_val = (conv).max()
                for frac in [0.1, 0.25, 0.4, 0.6, 0.8, 0.95]:
                    d = ai * frac
                    new_conv_max = (conv + d * gain).max()
                    # Add diagonal corrections
                    di = 2 * i
                    if di < size:
                        new_conv_max = max(new_conv_max, conv[di] - 2 * a[i] * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_conv_max = max(new_conv_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_conv_max < best_val:
                        best_val = new_conv_max
                        best_d = d

                if best_d > 0 and best_val < best_score - 1e-12:
                    gain_val = best_score - best_val
                    if gain_val > best_gain:
                        best_gain = gain_val
                        best_j = int(j)
                        best_delta = best_d

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved += 1
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if pass_improved == 0:
            no_improve += 1
            if no_improve >= 6:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            print(f"    PDCD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, improved={pass_improved}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    """Projected subgradient method using peak-weighted gradients."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    step_scale = 1.0 / np.sqrt(n)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        n_iters += 1

        # Weight peaks
        n_peaks = min(16, n)
        peak_idx = np.argsort(conv)[-n_peaks:]
        peak_vals = conv[peak_idx]
        weights = (peak_vals - M * 0.9) ** 2
        if weights.sum() < 1e-15:
            weights = np.ones(len(peak_idx))
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, pk in zip(weights, peak_idx):
            for i in range(n):
                ki = pk - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * w * a[ki]

        # Polyak step size approximation
        # gap = M - lower_bound, where lower_bound ~ (sum a)^2 / (2n-1)
        lower_bound = (norm_val ** 2) / (2 * n - 1)
        gap = M - lower_bound
        if gap < 1e-10:
            break
        step = (gap) / (np.dot(grad, grad) + 1e-15)
        step = min(step, step_scale * (1.0 + 1.0 / np.sqrt(t)))

        # Take step
        a_new = a - step * grad
        # Project onto {a >= 0, sum(a) = norm_val}
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_val / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        else:
            # Contraction step (helps with non-convexity)
            a_new = 0.5 * a + 0.5 * best_a
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm_val / s_new
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

        a = a_new

        if verbose and n_iters % 5000 == 0:
            print(f"    SubGrad iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def fine_cd(a0, time_limit, rng, verbose=False):
    """Fine-grained coordinate descent with exhaustive pair search on top coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = 0
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Compute gain for moving from i to each j
            # gain_k = 2*(a_{k-j} - a_{k-i}) for valid indices
            # Precompute the "removal" vector
            rem = np.zeros(size)
            for k in range(size):
                ki = k - i
                if 0 <= ki < n:
                    rem[k] = 2.0 * a[ki]

            best_gain = -1e-15
            best_j = -1
            best_d = 0.0

            # Check a subset of targets (top 50 by low pressure + random)
            # Pressure at position j: sum over peaks of a_{k-j}
            # Approximate: just use a[j] as a proxy (high a[j] means high pressure)
            targets = np.argsort(a)[:min(40, n // 4)]
            rand_targets = rng.integers(0, n, size=min(15, n // 8))
            all_targets = np.unique(np.concatenate([targets, rand_targets]))

            for j in all_targets:
                if j == i:
                    continue
                # Compute add vector
                add = rem.copy()  # Will overwrite
                for k in range(size):
                    kj = k - j
                    if 0 <= kj < n:
                        add[k] = 2.0 * a[kj]
                    else:
                        add[k] = 0.0

                gain = add - rem

                # Find best delta
                best_d_local = 0.0
                best_val_local = conv.max()
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    d = ai * frac
                    new_max = (conv + d * gain).max()
                    di2 = 2 * i
                    if di2 < size:
                        new_max = max(new_max, conv[di2] - 2 * a[i] * d + d * d)
                    dj2 = 2 * j
                    if dj2 < size:
                        new_max = max(new_max, conv[dj2] + 2 * a[j] * d + d * d)
                    if new_max < best_val_local:
                        best_val_local = new_max
                        best_d_local = d

                if best_d_local > 0:
                    gain_val = best_score - best_val_local
                    if gain_val > best_gain:
                        best_gain = gain_val
                        best_j = int(j)
                        best_d = best_d_local

            if best_j >= 0 and best_d > 0:
                a[i] -= best_d
                a[best_j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[best_j] -= best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-12:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved += 1

        if pass_improved == 0:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            print(f"    FineCD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, imp={pass_improved}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x)
    elif choice == 7:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.15 * np.cos(4 * np.pi * x))
    elif choice == 8:
        # Triangular with oscillation
        a = np.minimum(x, 1 - x) ** 0.8 * (1 + 0.2 * np.sin(10 * np.pi * x))
    elif choice == 9:
        # Flat with edges
        a = np.ones(n) * 0.5
        edge = int(0.1 * n)
        a[:edge] = np.linspace(1.0, 0.5, edge)
        a[-edge:] = np.linspace(0.5, 1.0, edge)
    elif choice == 10:
        # Gaussian mixture
        a = sum(np.exp(-20.0 * (x - c) ** 2) for c in [0.2, 0.5, 0.8])
    else:
        # Power law edge decay
        a = np.minimum(x / 0.2, 1.0) * np.minimum((1 - x) / 0.2, 1.0)
        a = a ** 0.5
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate():
    rng = np.random.default_rng(777)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: If no parent, try multiple initializations ===
    if parent is None:
        print("No parent found, trying multiple initializations...")
        best_init = None
        best_init_score = float('inf')
        for trial in range(10):
            n_try = rng.integers(1200, 2000)
            init = structured_init(n_try, rng)
            sc = evaluate_safe(init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP Nesterov at base length ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Peak-driven coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak-Driven CD n={n_best} ===")
    t2 = min(remaining * 0.18, 300.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_driven_cd(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"pdc1_n{n_best}")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient Polish n={n_best} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t3, verbose=False)
        do_try_update(a_out, f"subg_n{n_best}")

    # === Phase 4: LP Nesterov with perturbation restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP with perturbations ===")
    t4_total = min(remaining * 0.20, 300.0)
    for r in range(3):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t4 = min(t4_total / 3.0, remaining - 50)
        if t4 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        pert = perturb_sequence(best_seq, rng, scale=scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=False, seed=200 + r * 10)
        do_try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 5: Fine CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Fine CD n={n_best} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fine_cd(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"finecd_n{n_best}")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 200
    if remaining > 100:
        print(f"\n=== Phase 6: LP Nesterov n={n2} ===")
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 30:
            init_long = interpolate_sequence(best_seq, n2)
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=42)
            do_try_update(a_out, f"lp_long_n{n2}")

    # === Phase 7: Final polish with fine CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Fine CD n={n_best} ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fine_cd(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"final_cd_n{n_best}")

    # === Phase 8: LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 8: Final LP n={n_best} ===")
        t8 = min(remaining * 0.4, 100.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, seed=99)
            do_try_update(a_out, f"final_lp_n{n_best}")

    # === Phase 9: Subgradient final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final SubGrad n={n_best} ===")
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t9, verbose=False)
            do_try_update(a_out, f"final_subg_n{n_best}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
