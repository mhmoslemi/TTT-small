            src = idx[src_local]

            # Find a good destination: low value, not contributing to any top peak
            # A position j contributes to peak p if js <= j <= je
            # For all top peaks, find positions that don't contribute
            not_contrib = np.ones(n, dtype=bool)
            for p2 in top_k:
                js2 = max(0, p2 - n + 1)
                je2 = min(n - 1, p2)
                not_contrib[js2:je2 + 1] = False
            not_contrib[src] = False

            if not_contrib.sum() == 0:
                not_contrib = np.ones(n, dtype=bool)
                not_contrib[src] = False

            # Pick destination: lowest value among non-contributing
            dst_candidates = np.where(not_contrib)[0]
            dst = dst_candidates[np.argmin(a_work[dst_candidates])]

            # Move a small fraction
            frac = 0.1
            amt = a_work[src] * frac
            a_work[src] -= amt
            a_work[dst] += amt
            a_work = np.maximum(a_work, 0)
            s = a_work.sum()
            if s > 1e-15:
                a_work = a_work * norm_target / s

    return a_work


def spectral_reweight(a, time_limit, verbose=True):
    """
    Use the spectral representation to flatten the self-convolution.
    In frequency domain: conv(a,a) <-> |A(f)|^2.
    We want to minimize the peak of ifft(|A(f)|^2).
    This isiteratively reweight in frequency domain.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    norm_target = a.sum()

    a_work = np.maximum(np.asarray(a, dtype=np.float64), 0.0) * norm_target / max(a.sum(), 1e-15)
    best_a = a_work.copy()
    best_score = fast_conv(a_work).max()
    t_start = time.monotonic()
    iters = 0

    while time.monotonic() - t_start < time_limit:
        # Compute FFT
        A = rfft(a_work, fft_size)
        # |A|^2 in frequency domain
        power = A.real ** 2 + A.imag ** 2

        # Inverse FFT gives the self-convolution
        conv_vals = irfft(power, fft_size)[:2 * n - 1]
        max_conv = conv_vals.max()

        if max_conv < 1e-12:
            break

        # Reweight: modify A to flatten conv_vals
        # conv_vals[k] = ifft(|A|^2)[k]
        # We want to reduce peaks in conv_vals
        # In frequency domain, this means redistributing |A|
        # Use log-domain reweighting: A_new = A * exp(-beta * grad_log)
        # The gradient of max(conv) wrt log|A|^2 is related to the peak position

        # Simpler: use amplitude modification
        # Where conv has peaks, reduce the corresponding frequency components
        # Peak at time lag k corresponds to frequency modulation e^{-2pij*k*f/fft_size}

        p_peak = int(np.argmax(conv_vals))

        # The contribution of frequency bin m to time lag k is |A[m]|^2 * exp(2pij*m*k/fft_size)
        # To reduce the peak at k, reduce |A[m]| where exp(2pij*m*k/fft_size) aligns with the peak

        # Phase alignment: peak at k is enhanced when phase of A[m]^2 ~ 2pi*m*k/fft_size
        # A[m]^2 = |A[m]|^2 * exp(2j*angle(A[m]))
        # So conv[k] gets contribution proportional to |A[m]|^2 * cos(2*angle(A[m]) - 2*pi*m*k/fft_size)

        # Simplified: just reduce |A[m]| uniformly where it contributes to the peak
        # Use the IFFT of the peak indicator
        peak_indicator = np.zeros(fft_size)
        peak_indicator[p_peak] = 1.0
        freq_response = rfft(peak_indicator, fft_size)

        # Modify amplitude: where |freq_response| is high and aligned, reduce
        magnitudes = np.abs(A)
        phases = np.angle(A)

        # Compute alignment
        align_real = A * np.conj(freq_response)
        # Where real part is positive, the frequency contributes positively to the peak
        contribution = magnitudes ** 2 * np.real(align_real / (magnitudes + 1e-30))

        # Reduce amplitudes that contribute to the peak
        beta = 0.1
        scale = np.exp(-beta * np.maximum(contribution, 0) / max(contribution.max(), 1e-30))
        A_new = A * scale

        # Reconstruct
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm_target / s_new

        score_new = fast_conv(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a_work = a_new
        else:
            # Try smaller beta
            beta *= 0.5
            if beta < 0.001:
                # Reset with perturbation
                rng = np.random.default_rng(iters)
                a_work = np.maximum(a_work + rng.normal(0, 0.01 * a_work.mean(), n), 0)
                a_work = a_work * norm_target / a_work.sum()
                beta = 0.1
            else:
                scale = np.exp(-beta * np.maximum(contribution, 0) / max(contribution.max(), 1e-30))
                A_new = A * scale
                a_new = irfft(A_new, fft_size)[:n]
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm_target / s_new
                    a_work = a_new

        iters += 1
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Spectral {iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_start_lp(a0, n_starts, time_limit, verbose=True):
    """Run LP-Nesterov from multiple perturbations and keep the best."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a_base = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a_base.sum()
    if s < 1e-15:
        a_base = np.ones(n)
    a_base = a_base * norm / a_base.sum()

    best_a = a_base.copy()
    best_M = fast_conv(a_base).max()
    t_start = time.monotonic()

    per_start_time = time_limit / n_starts
    rng = np.random.default_rng(12345)

    for i in range(n_starts):
        elapsed = time.monotonic() - t_start
        remaining = time_limit - elapsed
        if remaining < 10:
            break

        t_alloc = min(per_start_time, remaining - 5)
        if i == 0:
            a_init = a_base.copy()
        else:
            # Perturb
            scale = 0.05 + 0.05 * (i / n_starts)
            perturb = rng.normal(0, scale * a_base.mean(), n)
            a_init = np.maximum(a_base + perturb, 0)
            s_i = a_init.sum()
            if s_i > 1e-15:
                a_init = a_init * norm / s_i
            else:
                a_init = a_base.copy()

        a_out, M_out = nesterov_lp_improved(a_init, t_alloc, verbose=(i == 0))
        if a_out is not None and M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()
            if verbose:
                print(f"    Multi-start {i}: improved to M={best_M:.8f}")

    return best_a, best_M


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return a_interp / ss


def symmetry_optimize(a, time_limit, verbose=True):
    """Try to improve by enforcing or breaking symmetry."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a_work = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a_work.sum()
    if s < 1e-15:
        a_work = np.ones(n)
    a_work = a_work * norm / a_work.sum()

    best_a = a_work.copy()
    best_M = fast_conv(a_work).max()

    # Try symmetric version
    a_sym = 0.5 * (a_work + a_work[::-1])
    a_sym = np.maximum(a_sym, 0)
    s_sym = a_sym.sum()
    if s_sym > 1e-15:
        a_sym = a_sym * norm / s_sym
        M_sym = fast_conv(a_sym).max()
        if M_sym < best_M:
            best_M = M_sym
            best_a = a_sym.copy()

    # Run short LP-Nesterov on the symmetric version
    t_start = time.monotonic()
    if time.monotonic() - t_start < time_limit * 0.5:
        a_out, M_out = nesterov_lp_improved(best_a, time_limit * 0.5, verbose=verbose)
        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search combining:
    1. LP-Nesterov (primary)
    2. Spectral reweighting
    3. Peak shaving
    4. Multi-length exploration
    5. Multi-start diversification
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    print(f"=== Starting optimization, budget={time_budget:.0f}s ===")

    # Load parent
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_candidate(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_candidate(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if needed
    if best_seq is None:
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_candidate(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_cur = len(best_seq)

    # === Phase 1: Main LP-Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n_cur} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_improved(a_norm, t1, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Spectral reweighting ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 2: Spectral reweight n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = spectral_reweight(a_norm, t2, verbose=True)
        try_update(a_out, f"spectral_n{n_cur}")

    # === Phase 3: LP-Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 280.0)
    print(f"\n=== Phase 3: LP-Nesterov r2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_improved(a_norm, t3, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Multi-length exploration (budget for this phase) ===")
    n_candidates = [n_cur - 100, n_cur + 100, n_cur + 200]
    n_candidates = [x for x in n_candidates if 500 <= x <= 2500]
    t_per_len = min(remaining * 0.20 / max(len(n_candidates), 1), 150.0)
    for n_try in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30 or t_per_len < 20:
            break
        print(f"  Trying n={n_try} (budget={t_per_len:.0f}s)")
        a_interp = interpolate_to_length(best_seq, n_try)
        a_norm = a_interp * np.sqrt(2.0 * n_try) / a_interp.sum()
        a_out, _ = nesterov_lp_improved(a_norm, t_per_len, verbose=False)
        try_update(a_out, f"lp_n{n_try}")

    # === Phase 5: Peak shaving refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 5: Peak shaving n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_shaved = targeted_peak_shave(best_seq, n_peaks=5)
        try_update(a_shaved, f"peakshave_n{n_cur}")

    # === Phase 6: Symmetry optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 6: Symmetry opt n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = symmetry_optimize(a_norm, t6, verbose=False)
        try_update(a_out, f"sym_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 7: Final LP polish n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_improved(a_norm, t7, verbose=True)
        try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.10, 80.0)
    print(f"\n=== Phase 8: Final peak shave n={n_cur} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_shaved = targeted_peak_shave(best_seq, n_peaks=5)
        try_update(a_shaved, f"peakshave_final_n{n_cur}")
        # Also try with more aggressive shaving
        a_shaved2 = targeted_peak_shave(best_seq, n_peaks=10)
        try_update(a_shaved2, f"peakshave10_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_candidate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
