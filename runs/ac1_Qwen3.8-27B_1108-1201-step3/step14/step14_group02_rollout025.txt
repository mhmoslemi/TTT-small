        tau_start = ac.max() / 5.0
    
    tau = tau_start

    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 1.0
    
    best_a = a.copy()
    best_obj = 2 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute gradient
        ac = fast_autocorr(a)
        tau = max(tau_end, tau_start * 0.98 ** (t // 100))  # decay tau
        max_ac = ac.max()
        w = np.exp((ac - max_ac) / tau)
        S = w.sum()
        
        # grad_a = (2/S) * c where c[i] = sum_j a[j] * w[i+j]
        w_flip = w[::-1]
        fft_size = 1 << (3*n - 2 - 1).bit_length()

        Wf = rfft(w_flip, fft_size)

        grad_a = (2.0 / S) * c

        v = beta2 * v + (1 - beta2) * grad_z**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # Convert back to a
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a = a_new / a_new.sum()

        M = fast_autocorr(a).max()
        obj = 2 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adjust learning rate
        if t % 200 == 0:
            lr *= 0.95  # slowly decrease
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Smooth t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, tau={tau:.4f}")

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, tau):
    """Gradient of tau*logsumexp(autocorr(a)/tau) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / tau)
    S = w.sum()
    w_flip = w[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_aw = irfft(A * Wf, fft_size)[:3 * n - 2]
    c = conv_aw[n - 1: 2 * n - 1][::-1]
    grad = (2.0 / S) * c
    return grad

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smooth_adam_optimize(a0, time_limit, verbose=True, tau_decay=0.95, tau_min=0.005):
    """Optimize smooth logsumexp objective using Adam on log-parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    ac0 = fast_autocorr(a)
    tau = ac0.max() / 4.0
    tau_floor = tau_min * ac0.max()
    z = np.log(a + 1e-30)

    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 2.0

    best_a = a.copy()
    best_obj = 2.0 * n * ac0.max()
    t0 = time.monotonic()
    stall_counter = 0
    best_at_stall = best_obj

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Decay tau periodically
        if t % 50 == 0:
            tau = max(tau_floor, tau * tau_decay)

        ac = fast_autocorr(a)
        max_ac = ac.max()
        obj = 2.0 * n * max_ac

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            stall_counter = 0
            best_at_stall = best_obj
        else:
            stall_counter += 1

        # If stalled, reduce learning rate and reset momentum
        if stall_counter > 300:
            lr *= 0.7
            m *= 0.5
            v *= 0.5
            stall_counter = 0
            if lr < 1e-4:
                break

        # Compute gradient
        max_ac_cur = ac.max()
        w = np.exp((ac - max_ac_cur) / tau)
        S = w.sum()
        w_flip = w[::-1]
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        A = rfft(a, fft_size)
        Wf = rfft(w_flip, fft_size)
        conv_aw = irfft(A * Wf, fft_size)[:3 * n - 2]
        c = conv_aw[n - 1: 2 * n - 1][::-1]
        grad_a = (2.0 / S) * c

        # Convert to log-parameter gradient
        center = np.dot(a, grad_a)
        grad_z = a * (grad_a - center)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back to a, renormalize
        z_shifted = z_new - z_new.max()
        a_new = np.exp(z_shifted)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()
        z = z_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, tau={tau:.5f}, lr={lr:.4f}")

    return best_a, best_obj / (2.0 * n)

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build T matrix: T[k,j] = y[k-j] if 0 <= k-j < n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m - 1e-14:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best_a = x.copy()
            stall = 0

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")

        if stall > 200:
            # Random perturbation to escape
            rng = np.random.default_rng(t)
            idx = rng.integers(n)
            jdx = rng.integers(n)
            if idx != jdx:
                frac = 0.01
                amt = y[idx] * frac
                y[idx] -= amt
                y[jdx] += amt
                y = np.maximum(y, 0.0)
                y = y / y.sum()
                stall = 0

    return best_a, best_m

def subgradient_fw(a0, time_limit, top_k=10, verbose=True):
    """Subgradient Frank-Wolfe: move towards best-response at top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-14:
            break

        # Pick top-k lags with weighting
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** 3
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Build direction: weighted average of shifted-a vectors
        direction = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            for j in range(n):
                idx = k_int - j
                if 0 <= idx < n:
                    direction[j] += wi * 2.0 * a[idx]

        d_sum = direction.sum()
        if d_sum < 1e-15:
            continue
        g = direction / d_sum

        # Frank-Wolfe step with backtracking
        step = 2.0 / (t + 2.0)
        for _ in range(20):
            a_new = (1.0 - step) * a + step * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            M_new = fast_autocorr(a_new).max()
            if M_new <= M + 1e-12:
                a = a_new
                break
            step *= 0.5
        else:
            a = (1.0 - step) * a + step * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        M_cur = fast_autocorr(a).max()
        if M_cur < best_m:
            best_m = M_cur
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def pairwise_exchange(a0, time_limit, verbose=True):
    """Pairwise coordinate exchange with binary search for optimal step."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find best source: coordinate with highest contribution to peaks
        # Source contribution: for source i, contribution = sum of a[k-i]*a[i] over top k's
        top_lags = np.argsort(ac)[::-1][:15]
        src_contrib = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            idx = np.arange(j_start, j_end + 1)
            src_contrib[idx] += a[idx] * a[k_int - idx]

        # Try top sources
        n_try = min(8, n)
        sources = np.argsort(src_contrib)[::-1][:n_try]

        improved = False
        for src in sources:
            src = int(src)
            if a[src] < 1e-10:
                continue

            # Find best destination
            # Score: how much does moving from src to dst reduce the max?
            # delta_b[k] = 2*delta*(a[k-dst] - a[k-src]) (approximately)
            # We want dst that minimizes max_k (ac[k] + 2*delta*(a[k-dst]-a[k-src]))
            # For small delta, this is approximately: minimize sum over top peaks of (a[k-dst]-a[k-src])

            # Sample destinations
            n_cand = min(n, 300)
            # Bias towards low-value positions
            if n_cand < n:
                probs = 1.0 / (a + 0.01)
                probs /= probs.sum()
                cands = rng.choice(n, size=n_cand, replace=False, p=probs)
            else:
                cands = np.arange(n)

            best_dst = -1
            best_score = float('inf')

            for dst in cands:
                dst = int(dst)
                if dst == src:
                    continue
                score = 0.0
                for k in top_lags:
                    k_int = int(k)
                    add_dst = a[k_int - dst] if 0 <= k_int - dst < n else 0.0
                    rem_src = a[k_int - src] if 0 <= k_int - src < n else 0.0
                    score += (add_dst - rem_src)
                if score < best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0 or best_score >= 0:
                continue

            dst = best_dst

            # Binary search for optimal delta
            lo, hi = 0.0, a[src] * 0.9
            for _ in range(12):

                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    hi = mid
                    break
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m:
                    lo = mid
                else:
                    hi = mid

            delta = lo
            if delta < 1e-15:
                continue

            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[dst] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt

                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = a / a.sum()

        if verbose and (t // 100) != (stall // 100):
            elapsed = time.monotonic() - t0

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Smooth Adam at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth Adam n={n0} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam_optimize(a_in, t1, verbose=True)
        try_update(a_out, f"smooth_n{n0}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP FW n={n0} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    # === Phase 3: Smooth Adam again (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Smooth Adam 2 n={n0} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam_optimize(a_in, t3, verbose=True, tau_decay=0.92)
        try_update(a_out, f"smooth2_n{n0}")

    # === Phase 4: Pairwise exchange ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Pairwise n={n0} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_exchange(a_in, t4, verbose=True)
        try_update(a_out, f"pairwise_n{n0}")

    # === Phase 5: Subgradient FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: SubGrad FW n={n0} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_fw(a_in, t5, top_k=8, verbose=True)
        try_update(a_out, f"subgrad_n{n0}")

    # === Phase 6: Scale to 2400 ===
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Scale to n={n1} ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 30:
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        a_out, _ = smooth_adam_optimize(a_in, t6 * 0.5, verbose=True)
        try_update(a_out, f"smooth_n{n1}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, min(remaining * 0.5, 150.0), solver, verbose=True)
            try_update(a_out, f"lp_n{n1}")

    # === Phase 7: Final polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} ===")
    if remaining > 30:
        # Smooth with very low tau
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smooth_adam_optimize(a_in, t7, verbose=True, tau_decay=0.90, tau_min=0.002)
            try_update(a_out, f"smooth_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_exchange(a_in, min(remaining * 0.5, 100.0), verbose=True)
            try_update(a_out, f"pairwise_final_n{n_best}")

    # === Phase 8: Try alternative initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Alt init n={n_best} ===")
        x = np.linspace(0, 1, n_best)
        # Try different analytical shapes
        alts = [
            np.cos(np.pi * (x - 0.5)) ** 4,
            np.exp(-5.0 * (x - 0.5) ** 2),
            (1 + np.cos(2 * np.pi * x)) / 2,
        ]
        for idx, alt in enumerate(alts):
            if remaining < 25:
                break
            remaining = time_budget - (time.monotonic() - t_global)
            blend = 0.6 * norm_sum1(best_seq) + 0.4 * norm_sum1(alt)
            blend = norm_sum1(blend)
            t_alt = min(remaining * 0.4, 60.0)
            if t_alt > 10:
                a_out, _ = smooth_adam_optimize(blend, t_alt, verbose=False)
                try_update(a_out, f"alt{idx}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
