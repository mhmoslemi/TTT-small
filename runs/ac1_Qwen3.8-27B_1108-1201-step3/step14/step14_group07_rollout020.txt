    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    momentum = 0.0

    # Use sparse LP for large n
    use_sparse = n > 1500

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if use_sparse:
            b = lp_best_response_sparse(y, M_curr, solver)
        else:
            b = lp_best_response_dense(y, M_curr, solver)
        if b is None:
            if use_subgrad:
                # Fallback to subgradient step
                b = subgrad_best_response(y, M_curr)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        if use_line_search:
            alpha_base = 2.0 / (t + 2.0)
            best_alpha = alpha_base
            best_obj = fast_autocorr((1 - alpha_base) * y + alpha_base * g).max()
            for trial_alpha in [alpha_base * 0.5, alpha_base * 1.5, alpha_base * 2.0]:
                trial_alpha = min(trial_alpha, 1.0)
                cand = (1 - trial_alpha) * y + trial_alpha * g
                cand = np.maximum(cand, 0.0)
                sc = fast_autocorr(cand).max()
                if sc < best_obj:
                    best_obj = sc
                    best_alpha = trial_alpha
            alpha = best_alpha
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Subgradient peak shaving every 50 iterations
        if use_subgrad and n_iters % 50 == 0:
            y = subgrad_shave(y, norm, n_shaves=3)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a * norm / best_a.sum()
            else:
                y = y * norm / s_y
            a = y.copy()

        if t % 300 == 0:
            stall_count += 1
            if stall_count > 30:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
            elif stall_count > 8:
                # Restart from best with perturbation
                rng_restart = np.random.default_rng(n_iters)
                y = best_a + rng_restart.normal(0, 0.01 * best_a.mean(), n)
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                y = y * norm / s_y
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgrad_best_response(a, M):
    """Compute a subgradient-based approximation of the LP best response."""
    n = len(a)
    c = fast_autocorr(a)
    k_star = np.argmax(c)
    # Subgradient of max(c) w.r.t. a[j]: 2*a[k_star - j] if 0 <= k_star - j < n
    j_max = min(n - 1, k_star)
    j_min = max(0, k_star - n + 1)
    j_range = np.arange(j_min, j_max + 1)
    grad = np.zeros(n)
    grad[j_range] = 2.0 * a[k_star - j_range]
    # The best response tries to maximize sum(b) subject to conv(a,b) <= M
    # A heuristic: set b[j] proportional to 1/grad[j] where grad[j] > 0
    b = np.zeros(n)
    positive = grad > 1e-15
    if positive.any():
        b[positive] = 1.0 / grad[positive]
        # Also allow b where grad is 0 (those don't contribute to the peak)
        zero_grad = ~positive
        b[zero_grad] = 1.0 / max(1.0, M / a.sum() * n)
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s


def subgrad_shave(a, norm, n_shaves=5, lr_schedule='cosine'):
    """Apply subgradient steps to reduce the max autocorrelation peak."""
    n = len(a)
    a = a.copy()
    t0 = time.monotonic()
    max_time = 2.0  # seconds

    for s in range(n_shaves):
        if time.monotonic() - t0 > max_time:
            break
        c = fast_autocorr(a)
        k_star = np.argmax(c)
        max_c = c[k_star]
        
        # Compute subgradient
        j_max = min(n - 1, k_star)
        j_min = max(0, k_star - n + 1)
        j_range = np.arange(j_min, j_max + 1)
        grad = np.zeros(n)
        grad[j_range] = a[k_star - j_range]
        
        # Also consider second peak
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.9 * max_c:
            k_star2 = np.argmax(np.where(c == c_sorted[1], c, -1))
            j_max2 = min(n - 1, k_star2)
            j_min2 = max(0, k_star2 - n + 1)
            j_range2 = np.arange(j_min2, j_max2 + 1)
            grad[j_range2] += a[k_star2 - j_range2]
        
        S = a.sum()
        # Learning rate
        lr = 0.1 / (s + 1)
        # Update: decrease a where it contributes to peak, increase elsewhere
        a_new = a - lr * grad
        # Renormalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * norm / s_new
    
    return a


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    # Compute gradient of LSE w.r.t. a using convolution
    fft_size = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * W_fft, fft_size)[:n]

    conv_grad = 2.0 * cross  # dc/da = 2*a[k-j], sum over k with weights w
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=6):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        x0 = a.copy()

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_grad(x, n, T)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'time_limit': stage_time})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score
