```python
class Adam:
    def __init__(self, dim, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
    
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2
        m_hat = self.m / (1 - self.beta1**self.t)
        v_hat = self.v / (1 - self.beta2**self.t)
        return m_hat / (np.sqrt(v_hat) + self.eps)
```

```python
import numpy as np

# --- Helper functions ---

def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = get_fft_size(n)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_objective_and_grad(a, T):
    """Compute LSE objective value and gradient for a (sum=1 normalization)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    # Shift for numerical stability
    m = shifted.max()  # should be 0
    exp_vals = np.exp(shifted - m)
    S = exp_vals.sum()
    w = exp_vals / S
    lse_val = max_b + (m + np.log(S)) / T
    
    # Gradient via cross-correlation
    w_rev = w[::-1]
    fft_size2 = 1 << (3*n - 2).bit_length()
    a_fft = rfft(a, fft_size2)
    w_fft = rfft(w_rev, fft_size2)
    conv = irfft(a_fft * w_fft, fft_size2)
    cross = conv[2*n-2 - np.arange(n)]
    grad = 2.0 * cross
    return lse_val, grad, b

def normalize_a(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

# --- LP Best Response ---

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build constraint matrix
    T_mat = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T_mat[start:end, i] = a[start - i:end - i]
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize_a(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()

    t0 = time.monotonic()
    
    # Temperature schedule
    temps = [100, 60, 40, 25, 15, 10, 7, 5, 3.5, 2.5, 2, 1.5, 1.2, 1.0, 0.8]
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    adam_t = 0
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()
        
        # Learning rate for this stage
        lr_start = 0.005
        lr_end = 0.0005
        n_steps_target = 3000  # target steps per stage
        
        # Run Adam for this temperature
        stage_steps = 0
        last_check = time.monotonic()
        
        while time.monotonic() - t0 < time_limit:
            adam_t += 1
            stage_steps += 1
            
            # Compute gradient
            lse_val, grad, b = lse_objective_and_grad(a, T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** adam_t)
            v_hat = v / (1 - beta2 ** adam_t)
            
            # Cosine learning rate within stage
            progress = min(1.0, stage_steps / n_steps_target)
            lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
            
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s
            
            # Periodic check
            if stage_steps % 50 == 0:
                current_max = fast_autocorr(a).max()
                if current_max < best_score:
                    best_score = current_max
                    best_a = a.copy()

                    print(f"    T={T}, step={stage_steps}, LSE={lse_val:.6f}, max={current_max:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
        
        # After each stage, check and optionally reset Adam
        current_max = fast_autocorr(a).max()
        if current_max < best_score:
            best_score = current_max
            best_a = a.copy()
        
        # Print stage summary
        elapsed = time.monotonic() - t0

            print(f"  Stage T={T}: steps={stage_steps}, max={current_max:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

        js = max(0, peak_idx - n + 1)

        top_contrib = idx[np.argsort(contrib)[::-1][:8]]

        improved = False
        for src in top_contrib:

                    if not (js <= dst <= je and peak_idx - dst == src):
```

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-stage LSE + Adam optimization with adaptive temperature scheduling (primary)
2. LP best-response with Nesterov acceleration (secondary refinement)
3. Peak-targeting coordinate descent (final polish)
4. Multi-length exploration and noise-based restarts for diversity
Key improvements: faster per-step computation, adaptive LR scheduling,
better temperature annealing, gradient norm clipping, and multi-restart strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = get_fft_size(n)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad_compute(a, T):
    """Compute LSE value, its gradient, and the autocorrelation. Returns (lse_val, grad, b)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    m = shifted.max()
    exp_vals = np.exp(shifted - m)
    S = exp_vals.sum()
    w = exp_vals / S
    lse_val = max_b + (m + np.log(S)) / T

    # Cross-correlation: cross[j] = sum_m a[m] * w[j+m]
    # = conv[a, w_rev][2n-2-j] where w_rev = w[::-1]
    w_rev = w[::-1]
    fft_size2 = 1 << (3 * n - 2).bit_length()
    a_fft = rfft(a, fft_size2)
    w_fft = rfft(w_rev, fft_size2)
    conv = irfft(a_fft * w_fft, fft_size2)
    cross = conv[2 * n - 2 - np.arange(n)]
    grad = 2.0 * cross
    return lse_val, grad, b


def normalize_a(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T_mat[start:end, i] = a[start - i:end - i]
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lse_adam_optimize(a0, time_limit, verbose=True):
    """Multi-stage LSE + Adam optimization."""
    n = len(a0)
    a = normalize_a(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Benchmark per-step time
    bench_start = time.monotonic()
    n_bench = 30
    for _ in range(n_bench):
        _, _, _ = lse_grad_compute(a, 10.0)
    bench_time = max(1e-6, (time.monotonic() - bench_start) / n_bench)
    est_total_steps = int((time_limit - 2.0) / bench_time * 0.85)
    if verbose:
        print(f"  Bench: {bench_time*1000:.3f} ms/step, ~{est_total_steps} steps in {time_limit:.0f}s")

    # Temperature schedule with time weights
    temps = [80, 50, 30, 20, 12, 8, 5, 3.5, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5, 0.4]
    weights = [0.5, 0.7, 0.9, 1.0, 1.2, 1.4, 1.6, 1.8, 2.0, 2.2, 2.5, 2.5, 3.0, 3.0, 3.5, 3.5, 4.0]
    total_w = sum(weights)

    m_ = np.zeros(n)
    v_ = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    adam_t = 0
    total_steps = 0

    for stage_i, (T, w) in enumerate(zip(temps, weights)):
        if time.monotonic() - t0 > time_limit - 2:
            break
        stage_time = time_limit * w / total_w
        stage_deadline = min(time.monotonic() + stage_time, t0 + time_limit - 1)

        # Learning rate: start higher for high T, decrease for low T
        lr = 0.005 / (1.0 + 0.3 * stage_i)
        # Cosine decay within stage
        n_stage_target = max(100, int(stage_time / bench_time * 0.8))

        stage_steps = 0
        last_improve_step = 0

        while time.monotonic() < stage_deadline:
            adam_t += 1
            stage_steps += 1

            lse_val, grad, b = lse_grad_compute(a, T)
            cur_max = b.max()

            # Gradient clipping
            gnorm = np.linalg.norm(grad)
            if gnorm > 5.0:
                grad = grad * 5.0 / gnorm

            # Adam update
            m_ = beta1 * m_ + (1 - beta1) * grad
            v_ = beta2 * v_ + (1 - beta2) * (grad * grad)
            m_hat = m_ / (1.0 - beta1 ** adam_t)
            v_hat = v_ / (1.0 - beta2 ** adam_t)

            # Cosine LR within stage
            progress = min(1.0, stage_steps / n_stage_target)
            lr_eff = lr * 0.5 * (1.0 + np.cos(np.pi * progress))
            lr_eff = max(lr_eff, lr * 0.1)

            a = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s

            total_steps += 1

            if stage_steps % 200 == 0:
                if cur_max < best_score:
                    best_score = cur_max
                    best_a = a.copy()
                    last_improve_step = stage_steps
                elif stage_steps - last_improve_step > 5000:
                    # Plateau: nudge with small noise
                    a = a + 0.001 * np.random.default_rng(stage_steps).normal(0, 1, n)
                    a = normalize_a(a)
                    last_improve_step = stage_steps
                    # Reset Adam partially
                    m_ *= 0.5
                    v_ *= 0.5

        # End of stage
        b_final = fast_autocorr(a)
        if b_final.max() < best_score:
            best_score = b_final.max()
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  T={T:.1f}: {stage_steps} steps, max={b_final.max():.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    if verbose:
        print(f"  Total: {total_steps} steps, best_max={best_score:.8f}")
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = normalize_a(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M = b_y.max()
        if M < 1e-14:
            break

        b = lp_best_response(y, M, solver)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue
        stall = 0

        g = normalize_a(b)
        alpha = 2.0 / (t + 2.0)

        # Line search
        best_alpha = alpha
        best_obj = fast_autocorr(np.maximum((1 - alpha) * y + alpha * g, 0)).max()
        for trial_a in [alpha * 0.5, alpha * 1.5, alpha * 2.0, alpha * 3.0]:
            trial_a = min(trial_a, 1.0)
            cand = np.maximum((1 - trial_a) * y + trial_a * g, 0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial_a
        alpha = best_alpha

        y_new = np.maximum((1 - alpha) * y + alpha * g, 0)
        y_new = normalize_a(y_new)

        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = np.maximum(y_new + momentum * (y_new - y), 0)
            a_new = normalize_a(a_new)
        else:
            a_new = g.copy()

        y = y_new
        a = a_new

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score

                continue

                    continue

                if len(candidates) == 0:
                    continue
                # Sort by current value (prefer moving to low-value positions)
                candidates = candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 5)]]

                for dst in candidates[:4]:
                    for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                        amt = a[src]
