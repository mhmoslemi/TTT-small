def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve the LP best-response: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE objective and gradient for the self-convolution peak."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)

    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    log_sum = max_b + np.log(exp_sum) / T_temp
    w = exp_vals / exp_sum

    # Gradient: d/da_i [log_sum] = sum_k w_k * (a convolved with delta_k + delta_k convolved with a)_i / T
    # Simplified: grad from the LSE smoothed peak
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    # cross = inverse FFT of W_fft * A_fft gives (w * a) which relates to the gradient
    cross = irfft(W_fft * A_fft, fft_size)[:n]
    # Also need the "reflected" convolution
    w_ref = w[::-1]
    w_ref_full = np.zeros(fft_size)
    w_ref_full[:2 * n - 1] = w_ref
    W_ref_fft = rfft(w_ref_full, fft_size)
    cross_ref = irfft(W_ref_fft * A_fft, fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    raw_grad = (cross + cross_ref) / T_temp
    grad = 2.0 * n * raw_grad / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_scipy_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=5):
    """Multi-stage LSE optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)

        # Time-based iteration limit for L-BFGS-B
        max_iter = max(50, int(stage_time * 3))

        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                          'gtol': 1e-9, 'ftol': 1e-12,
                                          'maxls': 40})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_tight_M=True):
    """LP best-response with Nesterov acceleration and adaptive M targets."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    patience = 90.0  # seconds without improvement before stopping

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive M target: start tight, relax if infeasible
        if use_tight_M:
            # Use a schedule: tight early, slightly looser late
            if t < 50:
                eps = 0.01
            elif t < 200:
                eps = 0.015
            elif t < 1000:
                eps = 0.02
            else:
                eps = min(0.025, 0.005 * np.log1p(t))
            M_used = M_curr * (1.0 - eps)
        else:
            M_used = M_curr

        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            # Fallback: use current peak
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                b = lp_best_response(y, M_curr * 1.01, solver, T)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.5)
            # If the new point has a much higher peak, use a smaller step
            if ratio > 1.0:
                alpha = min(alpha * 0.5, 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Track best from y
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        # Track best from a (Nesterov point)
        if t % 3 == 0:  # Check less frequently for speed
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                last_improve = time.monotonic()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > patience:
            stall_count += 1
            if stall_count > 3:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Find elements contributing most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find all significant peaks
        top_peaks = np.argsort(b)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        # Target: move mass from high contributors to low-peak areas
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False
        # Try targeted moves
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        # Random exploration if no improvement
        if not improved:
            n_random = 100
            for _ in range(n_random):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x) * np.exp(-2 * (x - 0.5)**2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.6 * np.sin(np.pi * x) + 0.3 * np.sin(2*np.pi*x)**2 + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    """Try to update best state with candidate."""
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase LP + LSE + coordinate descent optimization."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: Main LP optimization on parent length (45% time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_main = n_base
    t1 = remaining * 0.42
    print(f"\n=== Phase 1: LP-Nest n={n_main} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_main) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_tight_M=True)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_main_n{n_main}", t_global)

    # ============ Phase 2: LSE refinement (12% time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 2: LSE refinement n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=100.0, T_end=3.0, n_stages=5)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_n{n_cur}", t_global)

    # ============ Phase 3: LP round 2 from improved point (15% time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: LP round 2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, use_tight_M=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_r2_n{n_cur}", t_global)

    # ============ Phase 4: Try longer length (15% time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_cur + 400
    t4 = min(remaining * 0.17, 250.0)
    print(f"\n=== Phase 4: LP longer n={n_long} (budget={t4:.0f}s) ===")
    if t4 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, use_tight_M=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # ============ Phase 5: LSE on longer length if it won ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: LSE n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t5, T_start=50.0, T_end=2.0, n_stages=4)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_long_n{n_cur}", t_global)

    # ============ Phase 6: Coordinate descent polish (12% time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.14, 180.0)
    print(f"\n=== Phase 6: Coord descent n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t6, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord1", t_global)

    # ============ Phase 7: Final LP + LSE alternation ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final alternation n={n_cur} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        # Alternate LP (short) and LSE
        n_rounds = max(2, int(remaining / 80))
        for r in range(n_rounds):
            rem_r = time_budget - (time.monotonic() - t_global)
            if rem_r < 30:
                break
            # Short LP
            t_lp = min(rem_r * 0.5, 100.0)
            if t_lp > 20:
                n_cur = len(best_seq)
                a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out, _ = nesterov_lp_optimize(a_norm, t_lp, solver, verbose=False, use_tight_M=True)
                best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"final_lp_{r}", t_global)

            rem_r = time_budget - (time.monotonic() - t_global)
            if rem_r < 20:
                break
            # LSE
            t_lse = min(rem_r * 0.5, 80.0)
            if t_lse > 15:
                n_cur = len(best_seq)
                a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out, _ = lse_scipy_optimize(a_norm, t_lse, T_start=30.0, T_end=1.5, n_stages=3)
                best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"final_lse_{r}", t_global)

    # ============ Final coordinate descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, remaining * 0.8, seed=777)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
