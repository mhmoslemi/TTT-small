        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, norm)
        
        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, norm)
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        M_y = fast_autocorr(y).max()
        M_a = fast_autocorr(a).max()
        
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            stall = 0
        elif M_a < best_M:
            best_M = M_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive gamma
        if len(M_hist) > 200 and len(M_hist) % 100 == 0:
            recent = M_hist[-100:]
            diffs = np.diff(recent)
            osc = np.abs(diffs).mean() / max(recent[0], 1e-10)
            if osc > 0.03:
                gamma = max(1.0, gamma - 0.1)
            elif osc < 0.005:
                gamma = min(2.0, gamma + 0.05)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPNest t={t} ({elapsed:.0f}s): best_M={best_M:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}")
        
        # Restart if stalled
        if stall > 200:
            rng = np.random.default_rng(t * 31 + 7)
            scale = 0.05 * a.mean()
            y = normalize_sum(np.maximum(y + rng.normal(0, scale, n), 0), norm)
            a = normalize_sum(np.maximum(a + rng.normal(0, scale, n), 0), norm)
            prev_y = y.copy()
            prev_b = None
            stall = 0
            gamma = 1.8
            M_hist = []
            if verbose:
                print(f"    [Restart at t={t}]")
    
    return best_a, best_M


def gradient_coord_descent(a0, time_limit, verbose=True):
    """
    Gradient-guided coordinate descent.
    Uses the structure: changing a[j] by delta changes b[k] by 2*delta*a[k-j] + delta^2.
    """
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = normalize_sum(a0.copy(), norm)
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0
    rng = np.random.default_rng(12345)
    
    window = min(50, n // 4)  # Only try coordinates in a window around the peak contributor
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_pos = int(np.argmax(b))
        max_b = b[max_pos]
        
        # Find the top contributors to the peak
        js_start = max(0, max_pos - n + 1)
        js_end = min(n - 1, max_pos)
        if js_start > js_end:
            break
        
        contrib = a[js_start:js_end + 1] * a[max_pos - np.arange(js_start, js_end + 1)]
        
        # Pick the top few contributors to try reducing
        n_try = min(10, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_try]
        top_global = js_start + top_local
        
        improved = False
        
        for j in top_global:
            # Try reducing a[j]
            # delta changes b[k] by 2*delta*a[k-j] + delta^2 for k in [j, j+n-1]
            # We want to find delta < 0 that minimizes max_b
            a_j = a[j]
            if a_j < 1e-15:
                continue
            
            # Try several fractions
            best_score = best_M
            best_frac = 0
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = -a_j * frac
                k_start = j
                k_end = j + n - 1
                # New b values in the affected region
                affected = b[k_start:k_end + 1].copy()
                profile = a[k_start - j:k_end - j + 1]  # a[0:n] = a
                new_affected = affected + 2 * delta * profile + delta * delta
                new_max = max(np.max(new_affected), np.max(np.delete(b, range(k_start, k_end + 1))))
                if new_max < best_score:
                    best_score = new_max
                    best_frac = frac
            
            if best_frac > 0:
                delta = -a[j] * best_frac
                a[j] += delta
                a = np.maximum(a, 0)
                a = normalize_sum(a, norm)
                new_M = fast_autocorr(a).max()
                if new_M < best_M:
                    best_M = new_M
                    best_a = a.copy()
                    improved = True
                    break  # Restart from new peak
        
        if not improved:
            # Try increasing some non-contributing position
            # Find positions that don't contribute to the peak
            non_contrib = np.zeros(n, dtype=bool)
            non_contrib[js_start:js_end + 1] = True
            non_contrib[~non_contrib] = True  # flip: non-contributing
            
            # Actually, let's try random perturbation
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.05, 0.3)
                dst = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[dst] += amt
            a_trial = normalize_sum(np.maximum(a_trial, 0), norm)
            trial_M = fast_autocorr(a_trial).max()
            
            if trial_M < best_M:
                a = a_trial
                best_M = trial_M
                best_a = a.copy()
            elif trial_M < best_M * 1.005 and rng.random() < 0.3:
                a = a_trial  # Annealing acceptance
    
        step += 1
        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent step={step} ({elapsed:.0f}s): best_M={best_M:.8f}")
    
    return best_a, best_M


def multi_scale_explore(a0, time_limit, verbose=True):
    """
    Try multiple scales (lengths) by interpolating and running short optimizations.
    """
    n0 = len(a0)
    best_a = a0.copy()
    best_M = fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    # Try different lengths
    candidates = []
    for delta_n in [-100, -50, 50, 100, 150, 200, 300, 400]:
        new_n = n0 + delta_n
        if 400 <= new_n <= 4000:
            candidates.append(new_n)
    
    # Also try some "round" numbers
    for rn in [1200, 1400, 1600, 1800, 2000, 2400, 2800, 3200]:
        if 400 <= rn <= 4000 and rn not in candidates:
            candidates.append(rn)
    
    candidates = sorted(set(candidates))
    per_candidate_time = max(10, time_limit / max(len(candidates), 1))
    
    for new_n in candidates:
        if time.monotonic() - t0 >= time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        alloc = min(per_candidate_time, remaining)
        if alloc < 10:
            break
        
        print(f"    MultiScale: trying n={new_n} (budget={alloc:.0f}s)")
        init = interp_sequence(a0, new_n)
        a_trial, M_trial = lp_nesterov(init, alloc, max_constraints=150, verbose=False)
        
        if a_trial is not None:
            # Compare using the actual score (which accounts for n)
            score_trial = 2.0 * new_n * M_trial / init.sum() ** 2
            score_best = 2.0 * len(best_a) * best_M / a0.sum() ** 2
            # Actually we should compare using normalized M: score = 2n*M/sqrt(2n)^2 = M
            # Since we normalize to sum=sqrt(2n), score = M directly
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial
                print(f"    MultiScale: improved! n={new_n}, M={M_trial:.8f}")
    
    return best_a, best_M


def annealing_polish(a0, time_limit, seed=42, verbose=True):
    """
    Simulated annealing with structured moves for final polish.
    Uses FFT to quickly evaluate moves.
    """
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = normalize_sum(a0.copy(), norm)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    T_start = best_M * 0.01
    T_end = best_M * 0.0001
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_pos = int(np.argmax(b))
        max_b = b[max_pos]
        
        # Generate a structured move: move mass from peak contributor to a random position
        js_start = max(0, max_pos - n + 1)
        js_end = min(n - 1, max_pos)
        
        if js_start > js_end:
            break
        
        contrib = a[js_start:js_end + 1] * a[max_pos - np.arange(js_start, js_end + 1)]
        src = js_start + int(np.argmax(contrib))
        
        if a[src] < 1e-12:
            stall += 1
            if stall > 500:
                break
            continue
        
        # Pick destination: prefer positions far from the peak
        dst = rng.integers(n)
        if dst == src:
            dst = (src + 1) % n
        
        # Try move
        frac = rng.uniform(0.05, 0.25)
        amt = a[src] * frac
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = normalize_sum(np.maximum(a_trial, 0), norm)
        
        trial_M = fast_autocorr(a_trial).max()
        
        if trial_M < best_M:
            a = a_trial
            best_M = trial_M
            best_a = a.copy()
            stall = 0
        else:
            # Annealing acceptance
            delta = trial_M - best_M
            if delta < T and rng.random() < math.exp(-delta / max(T, 1e-15)):
                a = a_trial
            stall += 1
        
        # Periodic larger perturbation
        if stall > 100:
            n_pert = rng.integers(3, max(4, n // 20))
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in idx:
                amt = a[i] * rng.uniform(0.1, 0.5)
                j = rng.integers(n)
                a_trial[i] -= amt
                a_trial[j] += amt
            a_trial = normalize_sum(np.maximum(a_trial, 0), norm)
            trial_M = fast_autocorr(a_trial).max()
            if trial_M < best_M * 1.02:
                a = a_trial
            stall = 0
        
        if verbose and (time.monotonic() - t0) % 60 < 0.1:
            print(f"    Anneal ({time.monotonic()-t0:.0f}s): best_M={best_M:.8f}, T={T:.6f}, stall={stall}")
    
    return best_a, best_M


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function for optimal step function coefficients.
    Combines LP Nesterov, coordinate descent, multi-scale exploration, and annealing.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    
    print(f"Total time budget: {time_budget:.0f}s, seed={seed}")
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = safe_eval(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        n_base = 1600
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        # Use a raised cosine window as initial guess
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x))
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = safe_eval(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_cur} (budget={t1:.0f}s) ===")
    if t1 > 30:
        norm1 = math.sqrt(2.0 * n_cur)
        a_norm = normalize_sum(best_seq, norm1)
        a_out, M_out = lp_nesterov(a_norm, t1, max_constraints=250, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")
    
    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: CoordDescent n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 10:
        norm2 = math.sqrt(2.0 * n_cur)
        a_norm = normalize_sum(best_seq, norm2)
        a_out, M_out = gradient_coord_descent(a_norm, t2, verbose=True)
        try_update(a_out, f"coord_n{n_cur}")
    
    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:
        norm3 = math.sqrt(2.0 * n_cur)
        a_norm = normalize_sum(best_seq, norm3)
        a_out, M_out = lp_nesterov(a_norm, t3, max_constraints=200, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")
    
    # === Phase 4: Multi-scale exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: MultiScale n0={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 30:
        norm4 = math.sqrt(2.0 * n_cur)
        a_norm = normalize_sum(best_seq, norm4)
        a_out, M_out = multi_scale_explore(a_norm, t4, verbose=True)
        if a_out is not None:
            try_update(a_out, f"multiscale_n{len(a_out)}")
    
    # === Phase 5: Annealing polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: Annealing n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 15:
        norm5 = math.sqrt(2.0 * n_cur)
        a_norm = normalize_sum(best_seq, norm5)
        a_out, M_out = annealing_polish(a_norm, t5, seed=seed + 1, verbose=True)
        try_update(a_out, f"anneal_n{n_cur}")
    
    # === Phase 6: Final LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = max(remaining * 0.4, 30.0) if remaining > 60 else remaining
    print(f"\n=== Phase 6: Final LP Nesterov n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 20:
        norm6 = math.sqrt(2.0 * n_cur)
        a_norm = normalize_sum(best_seq, norm6)
        a_out, M_out = lp_nesterov(a_norm, t6, max_constraints=150, verbose=True)
        try_update(a_out, f"lp_final_n{n_cur}")
    
    # Final check
    if best_seq is not None:
        final_score = safe_eval(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
