def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]


def lse_obj_grad(a, T):
    """Compute LSE objective and gradient for normalized a (sum=1)."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    exp_vals = np.exp(shifted - shifted.max())  # numerical stability
    w = exp_vals / exp_vals.sum()
    lse = c_max + (shifted.max() + np.log(exp_vals.sum())) / T
    
    # Gradient: 2 * cross_correlation(w, a)
    N = 1 << (3 * n - 2).bit_length()
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    
    W = rfft(w_pad, N)
    A = rfft(a_pad, N)
    grad = irfft(np.conj(W) * A, N)[:n]
    grad = 2.0 * grad
    
    return lse, grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    return result


def nesterov_lse_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=8):
    """Nesterov-accelerated projected gradient on LSE objective."""
    n = len(a0)
    a = simplex_project(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    
    # Use the true max for scoring
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / (n_stages - stage)
        stage_t0 = time.monotonic()
        
        y = a.copy()
        t_iter = 0
        stall = 0
        last_best_score = best_score
        
        while time.monotonic() - stage_t0 < stage_time:
            t_iter += 1
            obj, grad = lse_obj_grad(y, T)
            
            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            alpha = 1.0 / (grad_norm + 1e-8)
            
            # Nesterov momentum
            momentum = (t_iter - 1.0) / (t_iter + 2.0) if t_iter > 1 else 0.0
            
            # Gradient step
            y_new = y - alpha * grad
            a_new = y_new + momentum * (y_new - y)
            
            # Project onto simplex
            y_new = simplex_project(y_new)
            a_new = simplex_project(a_new)
            
            # Check with true objective periodically
            if t_iter % 20 == 0:
                score = fast_conv(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    stall = 0
                elif score > last_best_score * 1.01:
                    stall += 1
                else:
                    stall = max(0, stall - 1)
                
                if stall > 50:
                    break
            
            y = y_new
            a = a_new
        
        a = best_a if best_score < fast_conv(a).max() else a
        print(f"    LSE stage {stage+1}/{n_stages} (T={T:.1f}): best={best_score:.8f}")
    
    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    
    # Build constraint matrix (sparse)
    # Row k: a[k], a[k-1], ..., a[0], 0s, a[1], ..., a[k]
    from scipy.sparse import lil_matrix
    T = lil_matrix((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_iteration(a, solver, max_iters=50, time_limit=300.0):
    """Iterative LP best-response with averaging."""
    n = len(a)
    t0 = time.monotonic()
    a_curr = simplex_project(a.copy())
    best_a = a_curr.copy()
    best_score = fast_conv(a_curr).max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_conv(a_curr).max()
        b = lp_best_response(a_curr, M, solver)
        if b is None:
            break
        b = simplex_project(b)
        
        # Check if b improves
        score = fast_conv(b).max()
        if score < best_score:
            best_score = score
            best_a = b.copy()
        
        # Nesterov extrapolation
        if it > 0:
            momentum = (it - 1.0) / (it + 2.0)
            a_new = b + momentum * (b - a_curr)
            a_new = simplex_project(a_new)
            score_new = fast_conv(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a_curr = a_new
        else:
            a_curr = b
    
    return best_a, best_score


def coord_descent(a0, time_limit, seed=42):
    """Coordinate descent: move mass from peak-contributing positions to non-peak positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = simplex_project(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return False
        trial = trial / s
        score = fast_conv(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:10]
        improved = False
        
        for k_peak in top_peaks:
            # Find positions that contribute most to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            
            # Find a position to move mass from (high contrib to peak)
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            
            # Find a position to move mass to (low contribution to all peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Choose destination with lowest current value
            dst = candidates[np.argsort(a[candidates])[:3]]
            
            for d in dst:
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[d] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    if try_update(a_trial):
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 10 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break
            
            if stall > 200:
                break
    
    return best_a, best_score


def structured_init(n, rng, style=None):

        x = np.linspace(0, 1, n)

    elif style == 6:
        # Tapered
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5

        # Random smooth
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += rng.uniform(0.1, 0.3) * np.sin(2*np.pi*freq*np.arange(n)/n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    if len(a) == new_n:
        return a.copy()
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_norm = a / a.sum()
    a_new = np.interp(x_new, x_old, a_norm)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new
