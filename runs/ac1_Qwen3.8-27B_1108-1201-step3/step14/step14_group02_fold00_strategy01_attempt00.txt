                    cand = interpolate_sequence(cand, n_initial)
                
                # Compare new candidate
                J_val_new, n_cur = evaluate_sequence(cand)
                if J_val_new < best_objective:
                    best_candidate = cand.copy()
                    best_objective = J_val_new
                    time_remaining -= 20
                    
                else:
                    time_remaining -= 5
                    
            # Adjust for intensity of improvement
            if J_val == best_objective:
                # Polishing phase with more aggressive methods
                n_avg = 30
                tLP_count = 0
                tLP_minutes = 0
                
                while tLP_count < 500 and time.time() - t_global < time_limit:
                    tLP_count += 1
                    tLP_minutes += 1
                    try:
                        # Tailored optimization using local signal-to-noise ratios
                        local_opt = mirror_descent_optimize(best_candidate, 300, top_k=15, beta=3.0)
                    except:
                        local_opt = polynomial_time_optimization(candidate, adaptive=True)
                        
                    J_opt, n_cur = check_evaluation(local_opt)
                    if J_opt < best_objective:
                        best_candidate = local_opt.copy()
                        best_objective = J_opt
                        tLP_count = 0  # reset for simpler correlation checks
                        
                time_remaining -= 5 * tLP_minutes

            # Update the candidate for the next iteration
            candidate = enhancer(best_candidate)
            
        # One final assessment
        best_objective, _ = check_evaluation(best_candidate)
        return best_candidate.tolist(), best_objective

    except KeyboardInterrupt:
        # Preserve best found so far on manual interruption
        best_objective, _ = check_evaluation(best_candidate)
        print(f"Manual stop saved candidate with score {best_objective}")
        return best_candidate.tolist(), best_objective
    except Exception as e:
        # Ultimate fallback for catastrophic error
        try:
            fallback = np.ones(n_initial)
            # If no other method, set fallback score to infinity to indicate failure
            best_objective = 1e10
            return fallback.tolist(), best_objective
        except:
            return [0.0] * n_initial, float('inf')
