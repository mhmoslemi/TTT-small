def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

    try:
        a_list = [float(max(0.0, x)) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def compute_pressure(a, k_list):
    """Compute pressure vector: for each j, max over k in k_list of a_{k-j} (valid)."""
    n = len(a)
    pressure = np.zeros(n)
    for k in k_list:
        # pressure[j] = a[k-j] if 0 <= k-j < n
        # j ranges over [max(0, k-n+1), min(n-1, k)]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            vals = a[idx]
            pressure[j_lo:j_hi + 1] = np.maximum(pressure[j_lo:j_hi + 1], vals)
    return pressure


def lp_best_response(a, M, solver, active_indices):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_indices, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    constraints = []
    # Build constraint matrix rows for active indices
    n_act = len(active_indices)
    if n_act == 0:
        return None

    # Sparse construction
    rows_list = []
    for k in active_indices:
        row = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            row[j_lo:j_hi + 1] = a[idx]
        rows_list.append(row)

    T = np.array(rows_list)
    constraints.append(T @ b_var <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-12:
        return None
    return b_val


def lp_best_response_with_screening(a, M, solver, tol=0.05):
    """LP best-response with adaptive constraint screening."""
    n = len(a)
    conv = fast_autocorr(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return None

    # Initial active set: constraints near the peak
    threshold = max_conv * (1.0 - tol)
    active = np.where(conv >= threshold)[0]

    # Ensure we have at least a few constraints
    if len(active) < 3:
        # Take top 10
        top_k = min(10, 2 * n - 1)
        active = np.argsort(conv)[-top_k:]

    b = lp_best_response(a, M, solver, active)

    # Check feasibility of the full constraint set
    if b is not None:
        # Check all constraints
        conv_ab = fast_autocorr_check(a, b)
        violations = conv_ab > M * (1 + 1e-8)
        if violations.any():
            # Add violated constraints and re-solve
            new_active = np.union1d(active, np.where(violations)[0])
            b = lp_best_response(a, M, solver, new_active)
            if b is not None:
                conv_ab = fast_autocorr_check(a, b)
                violations = conv_ab > M * (1 + 1e-8)
                if violations.any():
                    # Give up, use unnormalized heuristic
                    return None

    return b


def fast_autocorr_check(a, b):
    """Compute a*b convolution (not a*a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def nesterov_hybrid_optimize(a0, time_limit, solver, verbose=True, seed=0):
    """
    Nesterov-accelerated hybrid: fast repulsive steps + periodic LP corrections.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    momentum_vec = np.zeros(n)
    t0 = time.monotonic()
    iter_count = 0
    lp_count = 0
    stall_count = 0
    stall_threshold = 200

    # Adaptive parameters
    lp_every = 20  # Do LP every 20 iterations
    lp_every_base = 20
    tol_screen = 0.05

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation of y
        conv_y = fast_autocorr(y)
        M_y = conv_y.max()
        if M_y < 1e-14:
            break

        if (t % lp_every) == 0 and solver is not None:
            # LP correction step
            b = lp_best_response_with_screening(y, M_y, solver, tol_screen)
            if b is not None:
                g = normalize(b)
                lp_count += 1
            else:
                g = None
        else:
            g = None

        if g is None:
            # Fast repulsive step
            k_star = int(np.argmax(conv_y))
            # Pressure from top peaks
            n_peaks = min(50, 2 * n - 1)
            top_peaks = np.argsort(conv_y)[-n_peaks:]
            # Only use those > 0.8 * max
            top_peaks = top_peaks[conv_y[top_peaks] > 0.8 * M_y]
            if len(top_peaks) == 0:
                top_peaks = np.array([k_star])

            pressure = compute_pressure(y, top_peaks)
            # Repulsive update: b_j ~ 1/(pressure + eps)
            eps = 0.01 * y.sum() / n
            g = 1.0 / (pressure + eps)
            g = normalize(g)

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)

        prev_y = y.copy()
        y = y_new
        a = a_new
        iter_count += 1

        # Track best
        score_a = fast_autocorr(a).max()
        score_y = fast_autocorr(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        elif score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive LP frequency
        if stall_count > 100:
            lp_every = max(5, lp_every_base // 2)
        elif stall_count < 20:
            lp_every = min(50, lp_every_base)

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Restart at iter {t} ({elapsed:.0f}s), best={best_score:.10f}")

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}, lp={lp_count}")

    return best_a, best_score

def structured_init(n, rng, variant=None):
    """Generate structured initial sequences."""
    if variant is None:
        variant = int(rng.integers(0, 10))

    x = np.linspace(0.0, 1.0, n)

    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x * 3) + 0.1 * np.cos(2 * np.pi * x * 7)
    elif variant == 2:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2) + 0.1
    elif variant == 3:
        a = (1.0 - np.abs(2 * x - 1)) ** 2.5
    elif variant == 4:
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(6 * np.pi * x))
    elif variant == 5:
        a = 1.0 + 0.2 * np.cos(4 * np.pi * x) + 0.1 * np.cos(10 * np.pi * x) + 0.05 * np.cos(14 * np.pi * x)
    elif variant == 6:
        # Random Fourier features
        f = rng.integers(2, 20)
        ph = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f * x + ph)
    elif variant == 7:
        # Multi-scale
        a = np.ones(n)
        for m in range(3):
            f = rng.integers(2, 10 + 5 * m)
            amp = 0.2 / (m + 1)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * x + ph)
    elif variant == 8:
        # Power-law decay from center
        d = np.abs(x - 0.5)
        a = (1.0 - 2 * d) ** 0.8 + 0.1
    else:
        # Pure random smooth
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth
        from scipy.ndimage import uniform_filter
        a = uniform_filter(a, size=max(3, n // 50))

    a = np.maximum(a, 0.0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    return normalize(f_interp)


def perturb_sequence(a, rng, scale=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    return normalize(a_pert)


def direct_subgrad_optimize(a0, time_limit, verbose=False):
    """Direct subgradient method on the max autocorrelation with averaging."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    lr_base = 5.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max_k (a*a)_k wrt a
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            idx1 = k_star - j
            if 0 <= idx1 < n:
                v += a[idx1]
            idx2 = j - k_star
            if 0 <= idx2 < n:
                v += a[idx2]
            grad[j] = v

        # Gradient step with projection
        lr = lr_base / (t ** 0.6)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        a = a_new

        # Track average
        avg_a += a
        n_avg += 1

        # Check best every 100 steps
        if t % 100 == 0:
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            sc_avg = fast_autocorr(normalize(avg_a)).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = normalize(avg_a).copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: CD polish n={n_cur} ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t6, rng, verbose=True)
        do_try_update(a_out, f"cd2_n{n_cur}")

    # === Phase 7: Final hybrid ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final Hybrid n={n_cur} ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 20:
        a_out, _ = nesterov_hybrid_optimize(best_seq, t7, solver, verbose=True, seed=99)
        do_try_update(a_out, f"hybrid_final_n{n_cur}")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_cur} ===")
    t8 = remaining * 0.30
    if t8 > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t8, rng, verbose=True)
        do_try_update(a_out, f"cd_final_n{n_cur}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
