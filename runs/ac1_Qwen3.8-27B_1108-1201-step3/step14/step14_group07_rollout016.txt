Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed Nesterov gradient method (Phase 1)
2. Exact subgradient Nesterov acceleration (Phase 2 - main workhorse)
3. KL-proximal Nesterov (multiplicative weights variant)
4. Coordinate descent polish with peak targeting
5. Multi-length exploration with interpolation
Key innovation: O(n log n) per iteration via FFT-based gradients,
enabling 10-100x more iterations than LP-based methods.

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * np.conj(A), fft_size)[:size]

def score_on_simplex(p):
    """Compute 2*n*max(conv(p,p)) for p on the simplex (sum=1)."""
    b = fast_autocorr(p)
    return 2.0 * len(p) * b.max()

def subgrad_max_autocorr(y, m_star):
    """Subgradient of max_m (y*y)_m at the active index m_star.
    Returns gradient vector of shape (n,).
    ∂(y*y)[m]/∂y[j] = 2*y[m-j] if 0 <= m-j < n
    """
    n = len(y)
    g = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * y[m_star - idx]
    return g

def lse_gradient(y, T):
    """Gradient of log-sum-exp approximation to max(conv(y,y))."""
    n = len(y)
    b = fast_autocorr(y)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n), max_b
    log_sum = np.log(np.sum(np.exp(T * (b - max_b) + 1e-300)))
    log_se = max_b + log_sum / T
    w = np.exp(T * (b - max_b) - log_sum)  # softmax weights
    
    # Gradient: sum_m w[m] * ∇(y*y)_m
    # ∇(y*y)_m[j] = 2*y[m-j] if 0 <= m-j < n
    # grad[j] = 2 * sum_m w[m] * y[m-j] (where 0<=m-j<n)
    # = 2 * (cross-correlation of w and y)
    # conv(w, y)[j] = sum_m w[m] * y[j-m]... hmm let me think
    # Actually: sum_m w[m] * y[m-j] where we need 0 <= m-j < n
    # Let k = m-j, so m = j+k, and we need 0 <= k < n and 0 <= j+k < 2n-1
    # sum_k w[j+k] * y[k] (where valid)
    # This is (w * y)[j] in the convolution sense (cross-correlation)
    
    # Using FFT: compute cross-correlation of w (length 2n-1) and y (length n)
    fft_size = 1 << (3 * n - 2).bit_length()
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    y_pad = np.zeros(fft_size)
    y_pad[:n] = y
    W = rfft(w_pad, fft_size)
    Y = rfft(y_pad, fft_size)
    # We want sum_m w[m] * y[m-j] for j = 0..n-1
    # This is the cross-correlation: (w ⊛ y)[j] = sum_m w[m] * conj(y[m-j])
    # For real: sum_m w[m] * y[m-j]
    # In FFT: IFFT(conj(W) * Y) gives cross-correlation
    cross = irfft(np.conj(W) * Y, fft_size)[:n]
    
    grad = 2.0 * cross
    return grad, log_se

def project_simplex(v, target_sum):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cv = np.arange(1, n + 1)
    cond = u * cv > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_lse(p0, time_limit, T_start=50.0, T_end=5.0, n_stages=4, verbose=True):
    """LSE-smoothed Nesterov gradient method on the simplex."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_val = fast_autocorr(p).max()
    t0_global = time.monotonic()
    
    p_prev = p.copy()
    t = 1
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0_global > time_limit:
            break
        stage_start = time.monotonic()
        
        while time.monotonic() - t0_global < time_limit:
            # Nesterov extrapolation
            momentum = (t - 1.0) / (t + 2.0)
            y = p + momentum * (p - p_prev)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum < 1e-15:
                y = p.copy()
            else:
                y = y / y_sum
            
            # Compute LSE gradient
            g, lse_val = lse_gradient(y, T)
            
            # Step size: for LSE, Lipschitz constant is ~ L*T where L=2
            L_eff = 2.0 * T
            step = L_eff / (t + 2.0)
            p_candidate = y - step * g
            p_new = project_simplex(p_candidate, 1.0)
            
            p_prev = p
            p = p_new
            t += 1
            
            # Check improvement
            b = fast_autocorr(p)
            val = b.max()
            if val < best_val:
                best_val = val
                best_p = p.copy()
            
            if verbose and t % 5000 == 0:
                elapsed = time.monotonic() - t0_global
                print(f"    LSE({T:.1f}) iter={t} ({elapsed:.0f}s): best={best_val:.10f}")
    
    return best_p, best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Euclidean-projected Nesterov subgradient method (O(n log n)/iter)
2. Frank-Wolfe accelerated variant (O(n log n)/iter, no sorting)
3. LSE-smoothed gradient descent for initial smoothing
4. Peak-targeted coordinate descent with pair moves (final polish)
5. Multi-length exploration with interpolation transfer
Innovation: 10-100x faster per iteration than LP-based approach,
enabling orders of magnitude more iterations for better convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def _fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * np.conj(A), fft_size)[:size]


def _subgrad_max_autocorr(y, m_star):
    n = len(y)
    g = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * y[m_star - idx]
    return g


def _project_simplex(v, target_sum):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cv = np.arange(1, n + 1)
    cond = u * cv > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _nesterov_projected(p0, time_limit, verbose=True, label="", restart_every=50000):
    """Nesterov accelerated subgradient method with Euclidean projection onto simplex."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    b0 = _fast_autocorr(p)
    best_val = b0.max()

    y = p.copy()
    p_prev = p.copy()
    t0 = time.monotonic()
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t = iters + 1

        momentum = (t - 1.0) / (t + 2.0)
        y = p + momentum * (p - p_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = p.copy()
        else:
            y = y / y_sum

        b = _fast_autocorr(y)
        m_star = int(np.argmax(b))
        g = _subgrad_max_autocorr(y, m_star)

        step = 2.0 / (t + 2.0)
        p_candidate = y - step * g
        p_new = _project_simplex(p_candidate, 1.0)

        p_prev = p
        p = p_new
        iters += 1

        if iters % 50 == 1:
            b_p = _fast_autocorr(p)
            val = b_p.max()
            if val < best_val - 1e-14:
                best_val = val
                best_p = p.copy()
                stall = 0
            else:
                stall += 1

            if stall > restart_every // 50:
                p_prev = best_p.copy()
                p = best_p.copy() + np.random.default_rng(iters).normal(0, 1e-4, n)
                p = np.maximum(p, 0)
                ps = p.sum()
                if ps < 1e-15:
                    p = best_p.copy()
                else:
                    p = p / ps
                stall = 0

        if verbose and iters % 100000 == 0:
            elapsed = time.monotonic() - t0
            b_check = _fast_autocorr(best_p)
            print(f"    {label} iter={iters} ({elapsed:.0f}s): best={b_check.max():.12f}")

    b_final = _fast_autocorr(p)
    if b_final.max() < best_val:
        best_val = b_final.max()
        best_p = p.copy()

    return best_p, best_val


def _nesterov_fw(p0, time_limit, verbose=True, label=""):
    """Nesterov-style Frank-Wolfe (no projection needed, O(n log n) per iter)."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_val = _fast_autocorr(p).max()

    y = p.copy()
    p_prev = p.copy()
    t0 = time.monotonic()
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t = iters + 1

        momentum = (t - 1.0) / (t + 2.0)
        y = p + momentum * (p - p_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = p.copy()
        else:
            y = y / y_sum

        b = _fast_autocorr(y)
        m_star = int(np.argmax(b))
        g = _subgrad_max_autocorr(y, m_star)

        # FW step: move towards vertex minimizing <g, x> over simplex
        j_min = int(np.argmin(g))
        v = np.zeros(n)
        v[j_min] = 1.0

        gamma = 2.0 / (t + 2.0)
        p_new = (1.0 - gamma) * y + gamma * v
        p_new = np.maximum(p_new, 0.0)
        p_new_sum = p_new.sum()
        if p_new_sum < 1e-15:
            p_new = p.copy()
        else:
            p_new = p_new / p_new_sum

        p_prev = p
        p = p_new
        iters += 1

        if iters % 100 == 1:
            b_p = _fast_autocorr(p)
            val = b_p.max()
            if val < best_val - 1e-14:
                best_val = val
                best_p = p.copy()
                stall = 0
            else:
                stall += 1

            if stall > 1000:
                p_prev = best_p.copy()
                p = best_p.copy()
                stall = 0

        if verbose and iters % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters} ({elapsed:.0f}s): best={best_val:.12f}")

    b_final = _fast_autocorr(p)
    if b_final.max() < best_val:
        best_val = b_final.max()
        best_p = p.copy()

    return best_p, best_val


def _lse_nesterov(p0, time_limit, T_start=100.0, T_end=10.0, n_stages=3, verbose=False):
    """LSE-smoothed Nesterov gradient method."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_val = _fast_autocorr(p).max()
    t0 = time.monotonic()
    p_prev = p.copy()
    t = 1

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        while time.monotonic() - t0 < time_limit:
            momentum = (t - 1.0) / (t + 2.0)
            y = p + momentum * (p - p_prev)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum < 1e-15:
                y = p.copy()
            else:
                y = y / y_sum

            b = _fast_autocorr(y)
            max_b = b.max()
            if max_b < 1e-15:
                break
            shifted = T * (b - max_b)
            log_sum = np.log(np.sum(np.exp(shifted + 1e-300)))
            w = np.exp(shifted - log_sum)

            # grad = 2 * cross_correlation(w, y)
            fft_size = 1 << (3 * n - 2).bit_length()
            w_pad = np.zeros(fft_size)
            w_pad[:2 * n - 1] = w
            y_pad = np.zeros(fft_size)
            y_pad[:n] = y
            W = rfft(w_pad, fft_size)
            Y = rfft(y_pad, fft_size)
            cross = irfft(np.conj(W) * Y, fft_size)[:n]
            g = 2.0 * cross

            L_eff = 2.0 * T
            step = L_eff / (t + 2.0)
            p_candidate = y - step * g
            p_new = _project_simplex(p_candidate, 1.0)

            p_prev = p
            p = p_new
            t += 1

            if t % 200 == 1:
                b_p = _fast_autocorr(p)
                val = b_p.max()
                if val < best_val:
                    best_val = val
                    best_p = p.copy()

        if verbose:
            print(f"    LSE T={T:.1f} done ({time.monotonic()-t0:.0f}s): best={best_val:.10f}")

    return best_p, best_val


def _coord_descent(p0, time_limit, rng, verbose=True):
    """Peak-targeted coordinate descent."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_val = _fast_autocorr(p).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(p)
        top_peaks = np.argsort(b)[::-1][:8]
        improved = False

        for m_star in top_peaks:
            lo = max(0, int(m_star) - n + 1)
            hi = min(n - 1, int(m_star))
            idx = np.arange(lo, hi + 1)
            contrib = p[idx] * p[int(m_star) - idx]
            order = np.argsort(contrib)[::-1]
            top_idx = idx[order[:8]]

            # Build mask of indices involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for mp in top_peaks[:5]:
                l2 = max(0, int(mp) - n + 1)
                h2 = min(n - 1, int(mp))
                peak_mask[l2:h2 + 1] = True

            for src in top_idx:
                if p[src] < 1e-10:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                c_sorted = candidates[np.argsort(p[candidates])]

                for dst in c_sorted[:8]:
                    for frac in [0.01, 0.025, 0.05, 0.1, 0.15]:
                        amt = p[src] * frac
                        if amt < 1e-14:
                            continue
                        p_trial = p.copy()
                        p_trial[src] -= amt
                        p_trial[dst] += amt
                        p_trial = np.maximum(p_trial, 0)
                        st = p_trial.sum()
                        if st < 0.01:
                            continue
                        p_trial = p_trial / st
                        b_trial = _fast_autocorr(p_trial)
                        val = b_trial.max()
                        if val < best_val - 1e-14:
                            p = p_trial
                            best_val = val
                            best_p = p.copy()
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                for _ in range(50):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or p[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.08)
                    amt = p[i] * frac
                    p_trial = p.copy()
                    p_trial[i] -= amt
                    p_trial[j] += amt
                    p_trial = np.maximum(p_trial, 0)
                    st = p_trial.sum()
                    if st < 0.01:
                        continue
                    p_trial = p_trial / st
                    b_trial = _fast_autocorr(p_trial)
                    if b_trial.max() < best_val - 1e-14:
                        p = p_trial
                        best_val = b_trial.max()
                        best_p = p.copy()
                        stall = 0
                        improved = True
                        break

            if stall % 100 == 0 and stall > 0:
                n_pert = max(5, n // 50)
                indices = rng.choice(n, size=n_pert, replace=False)
                p_trial = p.copy()
                for idx_i in indices:
                    p_trial[idx_i] *= rng.uniform(0.7, 1.3)
                p_trial = np.maximum(p_trial, 0)
                st = p_trial.sum()
                if st > 0.01:
                    p_trial = p_trial / st
                    b_trial = _fast_autocorr(p_trial)

        if verbose and stall % 200 == 0 and stall > 0:

    return best_p, best_val


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 1:
        freq = int(rng.integers(2, 8))
        a = 1.0 + 0.4 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-8*(x-c1)**2) + 0.7*np.exp(-8*(x-c2)**2)
    elif choice == 3:
        a = 1.0 + 0.3*np.cos(2*np.pi*rng.uniform(3,7)*x) + 0.15*np.cos(2*np.pi*rng.uniform(7,12)*x)
    elif choice == 4:
        a = np.ones(n)
        n_blocks = int(rng.integers(3, 8))
        for _ in range(n_blocks):
            start = int(rng.integers(0, n - n//5))
            length = int(rng.integers(n//10, n//4))
            a[start:start+length] += rng.uniform(0.5, 1.5)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _evaluate_simplex(p):

    n = len(p)
    b = _fast_autocorr(p)

    print(f"\n=== Phase 2: Nesterov projected n={n_best} (budget={min(remaining*0.35, 500):.0f}s) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 20:
        a_out, _ = _nesterov_projected(best_seq, t2, verbose=True, label=f"NEST_n{n_best}")
        record(a_out, f"nest_n{n_best}")

    # Phase 3: FW variant at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FW Nesterov n={n_best} (budget={min(remaining*0.15, 200):.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, _ = _nesterov_fw(best_seq, t3, verbose=True, label=f"FW_n{n_best}")
        record(a_out, f"fw_n{n_best}")

    # Phase 4: Try longer sequence
    n_long = n_best + 300
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov long n={n_long} (budget={min(remaining*0.15, 250):.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        init_long = _interpolate_seq(best_seq, n_long)
        a_out, _ = _nesterov_projected(init_long, t4, verbose=True, label=f"NEST_long")
        record(a_out, f"nest_long_n{n_long}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={min(remaining*0.15, 200):.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_out, _ = _coord_descent(best_seq, t5, rng, verbose=True)
        record(a_out, f"cd_n{n_best}")

    # Phase 6: Another Nesterov pass (different init from CD output)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Nesterov round 2 n={n_best} (budget={min(remaining*0.15, 200):.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 15:
        noisy = best_seq + rng.normal(0, 0.01 * best_seq.mean(), n_best)
        noisy = np.maximum(noisy, 0)
        noisy = noisy / noisy.sum()
        a_out, _ = _nesterov_projected(noisy, t6, verbose=True, label=f"NEST2_n{n_best}")
        record(a_out, f"nest2_n{n_best}")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.6
    if t7 > 10:
        a_out, _ = _coord_descent(best_seq, t7, rng, verbose=True)
        record(a_out, f"cd_final_n{n_best}")

    # Phase 8: If time remains, try one more Nesterov on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final Nesterov n={n_best} (budget={remaining*0.7:.0f}s) ===")
        a_out, _ = _nesterov_projected(best_seq, remaining * 0.7, verbose=True, label="FINAL")
        record(a_out, "final_nest")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_simplex(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
