import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_lp_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)

        except Exception:

def fast_conv(a):
    """Compute conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_smooth_grad(a, T):
    """
    Compute gradient of T * log(sum(exp(conv(a,a) / T))) with respect to a.
    Uses FFT for efficiency.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n)
    shifted = (c - c_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    # Gradient: g[i] = 2 * sum_m a[m] * w[i+m]
    # = 2 * conv(reverse(a), w)[i + n - 1]
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(w, N)
    A_rev = rfft(a[::-1], N)
    cross = irfft(W * A_rev, N)
    grad = 2.0 * cross[n - 1 : n - 1 + n]
    return grad

def lse_fista_adaptive(a0, T_schedule, time_limit, verbose=True):
    """LSE-smoothed FISTA with adaptive (backtracking) step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_cmax = fast_conv(a).max()
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        L = n / 10.0  # initial Lipschitz estimate
        t_fista = 1
        staging = a.copy()
        
        while time.monotonic() - t0 < time_limit:
            grad = lse_smooth_grad(y, T)
            step = 1.0 / L
            
            # Backtracking: try the step, if it doesn't decrease, halve L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            
            # Check decrease (optional - can be expensive)
            # For now, just proceed with the step
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            # Periodic check
            if t_fista % 200 == 0:
                cmax = fast_conv(a).max()
                if cmax < best_cmax:
                    best_cmax = cmax
                    best_a = a.copy()
        
        # End of stage
        cmax = fast_conv(a).max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best_cmax={best_cmax:.8f}")
        t_fista = 1  # reset momentum for next stage
    
    return best_a, best_cmax

def subgradient_polyak(a0, time_limit, n_topk=5, verbose=True):
    """
    Subgradient method with Polyak step size for min max(conv(a,a)) s.t. a>=0, sum=1.
    Uses random selection from top-k peaks for the subgradient.
    """
    rng = np.random.default_rng(12345)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_cmax = fast_conv(a).max()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    # Polyak: need a lower bound. Use 1/(2n-1) as theoretical lower bound for max(conv) when sum=1
    lower_bound = 1.0 / (2 * n - 1)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        cmax = c.max()
        
        # Select subgradient from top-k peaks (random to avoid cycling)
        top_k = np.argsort(c)[::-1][:n_topk]
        k_star = top_k[rng.integers(n_topk)]
        
        # Subgradient of (a*a)[k_star] w.r.t. a:
        # g[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]
        
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            continue
        
        # Polyak step size
        gap = cmax - lower_bound
        if gap < 1e-15:
            break
        step = gap / g_norm_sq
        
        # Update
        a_new = a - step * g
        a_new = project_simplex(a_new, 1.0)
        
        # Accept if it improves
        c_new = fast_conv(a_new)
        cmax_new = c_new.max()
        if cmax_new < cmax or rng.random() < 0.1:  # occasionally accept non-improving steps
            a = a_new
        
        avg_a += a
        n_avg += 1
        
        # Check ergodic average
        if n_avg % 50 == 0:
            avg = avg_a / n_avg
            avg = project_simplex(avg, 1.0)
            cmax_avg = fast_conv(avg).max()
            if cmax_avg < best_cmax:
                best_cmax = cmax_avg
                best_a = avg.copy()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Subgrad iter={it} ({elapsed:.0f}s): cmax={cmax:.8f}, best={best_cmax:.8f}")
    
    return best_a, best_cmax

def lp_best_response(a, M, solver, n=0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    if n == 0:
        n = len(a)
    
    # conv(a, b)[k] = sum_j a[j] * b[k-j] for k = 0..2n-2
    # Matrix form: T @ b <= M where T[k, i] = a[k-i] if 0 <= k-i < n else 0
    # T is a Toeplitz matrix
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraints efficiently using Toeplitz structure
    # T is (2n-1) x n, T[k, i] = a[k-i] if 0 <= k-i < n
    # We can build this as a block matrix
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j : j + n, j] = a
    
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_cmax = fast_conv(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        cmax_y = fast_conv(y).max()
        if cmax_y < best_cmax:
            best_cmax = cmax_y
            best_a = y.copy()
        cmax_a = fast_conv(a).max()
        if cmax_a < best_cmax:
            best_cmax = cmax_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                cmax_avg = fast_conv(avg_check).max()
                if cmax_avg < best_cmax:
                    best_cmax = cmax_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best_cmax={best_cmax:.8f}")

def coordinate_refine(a0, time_limit, seed=42, verbose=True):
    """
    Local refinement via pairwise mass redistribution.
    For each peak position in conv, identify the most contributing pair and try to redistribute.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_cmax = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        # Get top peaks
        top_k = np.argsort(c)[::-1][:8]
        improved = False
        
        for k in top_k:
            # Find contributing pairs for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            
            # Try redistributing from the largest contributor
            src = idx[np.argmax(contrib)]
            
            # Build peak mask: positions that contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Find destinations (low mask, low value)
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            # Sort by value, take smallest
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = min(20, len(candidates_sorted))
            
            for dst_i in range(n_try):
                dst = candidates_sorted[dst_i]
                for frac in [0.0003, 0.0007, 0.0015, 0.003, 0.007, 0.015]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        break
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    cmax_t = fast_conv(a_trial).max()
                    if cmax_t < best_cmax - 1e-14:
                        a = a_trial
                        best_cmax = cmax_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    cmax_t = fast_conv(a_trial).max()
                    if cmax_t < best_cmax:
                        a = a_trial
                        best_cmax = cmax_t
                        best_a = a.copy()
                        stall = 0
            if stall > 2000:
                break
        
        if verbose and it % 100 == 0:  # (it not defined, use stall)
            pass
    
    return best_a, best_cmax

def mirror_descent(a0, time_limit, verbose=True):
    """
    Mirror descent with negative entropy (exponential weighting).
    Update: a_new[i] proportional to a[i] * exp(-eta * grad[i])
    This naturally maintains non-negativity.
    """
    rng = np.random.default_rng(777)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Ensure no zeros (for log)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_cmax = fast_conv(a).max()
    t0 = time.monotonic()
    
    eta = 1.0 / np.sqrt(n)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        cmax = c.max()
        
        # LSE gradient (smooth subgradient)
        T = max(cmax * 0.1, 1e-6)
        shifted = (c - cmax) / T
        shifted = np.clip(shifted, -80.0, 80.0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        # Compute gradient
        N = 1 << (3 * n - 1).bit_length()
        W = rfft(w, N)
        A_rev = rfft(a[::-1], N)
        cross = irfft(W * A_rev, N)
        grad = 2.0 * cross[n - 1 : n - 1 + n]
        
        # Mirror descent update (multiplicative)
        # a_new[i] = a[i] * exp(-eta * grad[i])
        # Then normalize
        log_update = -eta * grad
        log_update = np.clip(log_update, -5.0, 5.0)  # prevent overflow
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        cmax_new = fast_conv(a_new).max()
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        a = a_new
        
        # Decay eta
        if it % 100 == 0:
            eta *= 0.98
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Mirror iter={it} ({elapsed:.0f}s): cmax={cmax:.8f}, best={best_cmax:.8f}")
    
    return best_a, best_cmax

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- LSE-smoothed FISTA with adaptive temperature annealing
- Mirror descent (multiplicative weight updates)
- LP best-response with Nesterov acceleration
- Subgradient method with Polyak averaging
- Coordinate refinement via pairwise mass redistribution
- Multi-length exploration and restart strategies

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver


def fast_conv(a):
    """Compute linear convolution conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(a, T):
    """Gradient of T*log(sum(exp(conv(a,a)/T))) w.r.t. a, via FFT."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n)
    shifted = (c - c_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(w, N)
    A_rev = rfft(a[::-1], N)
    cross = irfft(W * A_rev, N)
    return 2.0 * cross[n - 1: n - 1 + n]


def lse_fista(a0, T_schedule, time_limit, verbose=True):
    """LSE-smoothed FISTA with Nesterov momentum and per-stage reset."""
    n = len(a0)

    s = a.sum()

        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_cmax = fast_conv(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / 20.0, 1.0)
        t_fista = 1
        staging = a.copy()

        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

def lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
