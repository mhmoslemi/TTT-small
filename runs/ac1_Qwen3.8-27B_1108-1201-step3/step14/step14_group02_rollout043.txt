def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_search(n, time_limit=30.0):
    """Search over parametric shapes for good initialization."""
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')
    x = np.linspace(0, 1, n)

    candidates = []

    # Family 1: (1-x)^p
    for p in np.linspace(0.1, 4.0, 40):
        a = (1.0 - x) ** p
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    # Family 2: exp(-alpha*x)
    for alpha in np.linspace(0.1, 8.0, 50):
        a = np.exp(-alpha * x)
        candidates.append(a / a.sum())

    # Family 3: 1 - alpha*x (linear clip)
    for alpha in np.linspace(0.0, 1.5, 30):
        a = np.maximum(1.0 - alpha * x, 0.0)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    # Family 4: cos(pi*x/2)^p
    for p in np.linspace(0.3, 4.0, 30):
        a = np.cos(np.pi * x / 2.0) ** p
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    # Family 5: (1-x)^p * exp(-beta*x)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.1, 0.5, 1.0, 2.0, 4.0]:
            a = (1.0 - x) ** p * np.exp(-beta * x)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                candidates.append(a / s)

    # Family 6: x^p (increasing)
    for p in np.linspace(0.1, 4.0, 30):
        a = x ** p
        s = a.sum()
        if s > 1e-10:
            candidates.append(a / s)

    # Evaluate all candidates
    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        sc = 2.0 * n * fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    if best_a is None:
        best_a = (1.0 - x) ** 1.0
        best_a = best_a / best_a.sum()
        best_score = 2.0 * n * fast_autocorr(best_a).max()

    return best_a, best_score

def compute_soft_subgrad(a, ac, top_k=50, beta=6.0):
    """Compute soft subgradient using softmax weighting over top lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]

    # Softmax weights
    shifted = top_vals - top_vals.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        # grad[m] += wi * (a[k-m] + a[m-k])
        # a[k-m] is valid when 0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(k, n-1)
        m_lo = max(0, k_int - n + 1)
        m_hi = min(k_int, n - 1)
        if m_lo <= m_hi:
            m_vals = np.arange(m_lo, m_hi + 1)
            idx = k_int - m_vals  # This is a[k-m]
            grad[m_vals] += wi * a[idx]
        # a[m-k] is valid when 0 <= m-k < n, i.e., max(k,0) <= m <= min(n-1, k+n-1)
        m_lo2 = max(k_int, 0)
        m_hi2 = min(n - 1, k_int + n - 1)
        if m_lo2 <= m_hi2:
            m_vals2 = np.arange(m_lo2, m_hi2 + 1)
            idx2 = m_vals2 - k_int  # This is a[m-k]
            grad[m_vals2] += wi * a[idx2]

    return grad

def mirror_descent_optimize(a0, time_limit, top_k=30, beta=5.0, verbose=True):
    """Mirror descent (exponentiated gradient) with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = a.copy()
    n_avg = 1
    momentum = 0.9
    g_mom = np.zeros(n)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        step = 1.0 / (np.sqrt(t) + 1.0) * 1.5
        ac = fast_autocorr(a)
        g = compute_soft_subgrad(a, ac, top_k=top_k, beta=beta)
        g_mom = momentum * g_mom + (1 - momentum) * g

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g_mom
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = np.maximum(avg_a / n_avg, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m - 1e-15:
                    best_m = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def build_T_matrix(a):
    """Build the convolution constraint matrix T where (T@b)[k] = (a*b_conv)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=None):
    """Nesterov-accelerated FW with LP oracle."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0
    max_iters = max_iters if max_iters is not None else 1000000

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build T matrix
        T = build_T_matrix(y)

        # LP: maximize sum(b) s.t. T @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m
