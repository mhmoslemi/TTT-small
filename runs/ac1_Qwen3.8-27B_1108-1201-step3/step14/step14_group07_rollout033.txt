_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def pnorm_and_grad(a, p):
    """Compute p-norm of autocorrelation and its gradient."""
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    
    bp = b ** (p - 1)  # b[k]^(p-1)
    pnorm_val = (np.sum(b ** p)) ** (1.0 / p)
    norm_factor = (np.sum(b ** p)) ** (1.0 / p - 1.0)
    
    # Compute s1[j] = sum_k bp[k] * a[j-k] (full conv of a and bp, take first n)
    s1 = fftconvolve(a, bp, mode='full')[:n]
    
    # Compute s2[j] = sum_k bp[k] * a[j+k]
    # = full_conv(a_rev, bp)[n-1-j] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv2 = fftconvolve(a_rev, bp, mode='full')
    s2 = conv2[n-1:n-1-n:-1]
    
    grad_conv = norm_factor * (s1 + s2)
    
    # Full gradient of 2n * pnorm / s^2
    grad = 2.0 * n * grad_conv / (s * s) - 4.0 * n * pnorm_val * a / (s ** 3)
    
    return pnorm_val, grad

def pnorm_optimize(a0, time_limit, p_start=8.0, p_end=200.0, n_stages=6):
    """Optimize using p-norm surrogate with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    p_list = np.geomspace(p_start, p_end, n_stages)
    for si, p in enumerate(p_list):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 100.0)
        
        def obj_func(x):
            val, _ = pnorm_and_grad(x, p)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            return 2.0 * len(x) * val / (s * s)
        
        def grad_func(x):
            _, g = pnorm_and_grad(x, p)
            return g
        
        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 300, 'maxfun': 600,
                                           'ftol': 1e-12, 'gtol': 1e-8,
                                           'disp': False})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_val / s_c
                    sc = compute_score(a_cand)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception:
            pass
    
    return best_a, best_score

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    y = a.copy()
    t0 = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Line search
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        best_obj = compute_score(np.maximum((1 - alpha_base) * y + alpha_base * g, 0))
        
        for trial_alpha in [alpha_base * 0.5, alpha_base * 0.75, alpha_base * 1.25, alpha_base * 1.5, alpha_base * 2.0]:
            trial_alpha = min(trial_alpha, 1.0)
            cand = np.maximum((1 - trial_alpha) * y + trial_alpha * g, 0)
            s_cand = cand.sum()
            if s_cand < 1e-10:
                continue
            cand = cand * norm_val / s_cand
            sc = compute_score(cand)
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial_alpha
        alpha = best_alpha
        
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = np.maximum(y_new + momentum * (y_new - y), 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = compute_score(a)
        if sc_a < best_score:
            best_score =
            best_a = a.copy()
        
        if verbose and t % 50 == 0:
            print(f"    NLP t={t}: best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, n_steps=50000, lr_start=0.001, lr_end=0.0001):
    """Polish using subgradient of the max objective with momentum."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = compute_score(a)
    velocity = np.zeros(n)
    beta = 0.9
    
    t0 = time.monotonic()
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        lr = lr_start * (lr_end / lr_start) ** (step / n_steps)
        b = fast_autocorr(a)
        max_b = b.max()
        k_star = np.argmax(b)
        
        # Subgradient of max_b w.r.t. a[j]:
        # db/d(a[j]) where b[k*] = sum_i a[i]*a[i+k*]
        # = a[j-k*] + a[j+k*] (clipped)
        sg = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j - k_star < n:
                val += a[j - k_star]
            if 0 <= j + k_star < n:
                val += a[j + k_star]
            sg[j] = val
        
        # Gradient of the full objective 2n*max_b/s^2:
        # = 2n * sg / s^2 - 4n * max_b * a / s^3
        s = a.sum()
        grad = 2.0 * n * sg / (s * s) - 4.0 * n * max_b * a / (s ** 3)
        
        # Momentum update
        velocity = beta * velocity - lr * grad
        a_new = a * norm_val / s + velocity * norm_val
        
        # Project onto non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * norm_val / s_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]

This is like the coefficient of x^k in the polynomial A(x)^2 where A(x) = sum_i a[i] x^i.

So we're looking for a non-negative polynomial A(x) = a[0] + a[1]x + ... + a[n-1]x^(n-1) such that A(x)^2 has as flat a coefficient profile as possible (all coefficients close to the mean).

The mean coefficient of A(x)^2 is: (sum of all coefficients of A(x)^2) / (2n-1) = (A(1))^2 / (2n-1) = (sum a_i)^2 / (2n-1).

We want all coefficients of A(x)^2 to be close to this mean.

This is related to the " Tac (TAC) polynomial" problem or "flat top" polynomial design.

**Connection to binary sequences**: For binary sequences (a_i in {0,1}), the problem becomes finding a subset of {0,...,n-1} such that the number of representations of each integer k as a sum of two elements of the subset is as uniform as possible. This is related to "modular Besicovitch sets" and "Golomb rulers" (but those minimize collisions, which is different).

**For continuous non-negative sequences**: We have more freedom. The optimal sequence likely has a smooth shape.
