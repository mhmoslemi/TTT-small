    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

        return float(sc)
    except:

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=50):
    """Compute soft subgradient of max(ac) using top-k most active lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    
    # Get top-k lags
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Exponential weighting (soft max approximation)
    # Use temperature based on the gap
    w = np.exp(top_vals / max(top_vals[0], 1e-30) * 5.0)  # temperature = max/5
    w = w / w.sum()
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # Vectorized: for each k, the contribution to g is a shifted and flipped a
        # a[k-i] for i in [max(0,k-n+1), min(k, n-1)]
        # a[i-k] for i in [max(k,0), min(n-1, k+n-1)]
        
        # First term: a[k-i], i from max(0, k-n+1) to min(k, n-1)
        i_start1 = max(0, k - n + 1)
        i_end1 = min(k, n - 1)
        if i_start1 <= i_end1:
            idx1 = np.arange(i_start1, i_end1 + 1)
            g[idx1] += wi * a[k - idx1]
        
        # Second term: a[i-k], i from max(k, 0) to min(n-1, k+n-1)
        i_start2 = max(k, 0)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            g[idx2] += wi * a[idx2 - k]
    
    return g

def fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def normalize_simplex(a):
    """Project onto the probability simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_soft_subgrad(a, ac, top_k=40, beta=3.0):
    """Compute soft subgradient of max(ac) using exponential-weighted top-K lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0.0)
    
    # Exponential weighting with adaptive temperature
    max_val = top_vals[0]
    if max_val < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        temp = max_val / beta
        w = np.exp(top_vals / temp - 1.0)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # First term: i ranges over [max(0, k-n+1), min(k, n-1)]
        i1s = max(0, k - n + 1)
        i1e = min(k, n - 1)
        if i1s <= i1e:
            idx1 = np.arange(i1s, i1e + 1)
            g[idx1] += wi * a[k - idx1]
        # Second term: i ranges over [max(k, 0), min(n-1, k+n-1)]
        i2s = max(k, 0)
        i2e = min(n - 1, k + n - 1)
        if i2s <= i2e:
            idx2 = np.arange(i2s, i2e + 1)
            g[idx2] += wi * a[idx2 - k]

    return g


def compute_exact_subgrad(a, ac, tol=1e-12):
    """Compute exact subgradient from all active (near-max) lags."""
    n = len(a)
    max_val = ac.max()
    active = np.where(ac >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(ac))])
    w = np.ones(len(active)) / len(active)
    
    g = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        i1s = max(0, k - n + 1)
        i1e = min(k, n - 1)
        if i1s <= i1e:
            idx1 = np.arange(i1s, i1e + 1)
            g[idx1] += wi * a[k - idx1]
        i2s = max(k, 0)

        if i2s <= i2e:
            idx2 = np.arange(i2s, i2e + 1)

                y = y / sy

        # Compute autoconvolution and subgradient at y
        ac_y = fast_autocorr(y)
        g = compute_soft_subgrad(y, ac_y, top_k=top_k, beta=beta)

        # Adaptive step size
        step = gamma / (np.sqrt(t) + min_step)

        # Mirror descent update: exp map on simplex
        log_y = np.log(y + 1e-30)
        log_new = log_y - step * g
        # Softmax normalization
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        a = a_new

        # Track best
        ac_x = fast_autocorr(x)
        m_x = ac_x.max()
        if m_x < best_m:
            best_m = m_x
            best_a = x.copy()

        # Also check Chebyshev average every 20 steps
        if t % 20 == 0:
            # Weighted average of recent iterates
            avg = (x + a) / 2.0
            avg = normalize_simplex(avg)
            m_avg = fast_autocorr(avg).max()
            if m_avg < best_m:
                best_m = m_avg
                best_a = avg.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def active_set_polish(a0, time_limit, verbose=False):
    """Polish using exact active set and coordinate moves."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        max_val = ac.max()
        active = np.where(ac >= max_val - 1e-10)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(ac))])

        improved = False

        # For each active lag, find the pair (i, j) that contributes most
        for k in active:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i1s = max(0, k - n + 1)
            i1e = min(k, n - 1)
            if i1s > i1e:
                continue
            idx = np.arange(i1s, i1e + 1)
            # Contribution of each i to c[k]: a[i]*a[k-i]
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Try to move mass from src to a position that reduces the peak
            # Score candidate destinations by how much they'd increase peaks
            peak_lags = active
            n_cands = min(n, 300)
            # Sample from positions not in the peak region
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            non_peak = np.where(~in_peak)[0]
            if len(non_peak) >= n_cands:
                candidates = non_peak[rng.permutation(len(non_peak))[:n_cands]]
            else:
                candidates = np.where(rng.random(n) < 0.3)[0]
                if len(candidates) < 10:
                    candidates = rng.integers(0, n, size=n_cands)

            best_dst = -1
            best_score = float('inf')

            for dst in candidates:
                dst = int(dst)
                if dst == int(src):
                    continue
                # Estimate impact on peak lags
                sens = 0.0
                for k2 in peak_lags:
                    k2i = int(k2)
                    add = 0.0
                    d1 = k2i - dst
                    d2 = dst - k2i
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add * 0.5  # factor of 0.5 for the move fraction

                rem = 0.0
                for k2 in peak_lags:
                    k2i = int(k2)
                    d1 = k2i - int(src)
                    d2 = int(src) - k2i
                    if 0 <= d1 < n:
                        rem += a[d1]
                    if 0 <= d2 < n:
                        rem += a[d2]
                    rem += 0.0

                net = sens - rem * a[src]
                if net < best_score:
                    best_score = net
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = int(best_dst)
            # Try multiple step sizes
            for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            if a.sum() > 1e-15:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric_shape(n, shape_type='parabolic', alpha=1.0):
    """ Generate a parametric shape of length n. """
    x = np.linspace(0.0, 1.0, n)
    if shape_type == 'uniform':
        f = np.ones(n)
    elif shape_type == 'parabolic':
        f = (1.0 - (2.0 * x - 1.0) ** 2) ** alpha
    elif shape_type == 'cosine':
        f = np.cos(np.pi * (x - 0.5)) ** (2.0 * alpha)
    elif shape_type == 'super_gaussian':
        f = np.exp(-alpha * (2.0 * x - 1.0) ** 4)
    elif shape_type == 'bicubic':
        f = 1.0 - (2.0 * x - 1.0) ** 4
    elif shape_type == ' raised_cosine':
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def try_update(a, label, verbose=True):
    """Try to update the global best."""
    global _global_best, _global_best_score
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < _global_best_score:
        _global_best_score = sc
        _global_best = a_norm.copy()
        if verbose:
            elapsed = time.monotonic() - _t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(budget_s=1800.0):
    """
    Main search function combining:
    - Multi-initialization with diverse parametric shapes
    - Mirror descent with Nesterov acceleration
    - Active-set coordinate polishing
    - Multi-length scaling
    - Random restarts
    """
    global _global_best, _global_best_score, _t_global_start
    _t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    _global_best = None
    _global_best_score = float('inf')

    rng = np.random.default_rng(2024)
    print(f"Total time budget: {time_budget:.0f}s")

    # --- Load and evaluate parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                _global_best = parent.copy()
                _global_best_score = p_score
    except Exception:
        pass

    n0 = len(parent) if parent is not None else 1600

    # --- Phase 1: Multi-initialization exploration (20% time) ---
    remaining = time_budget - (time.monotonic() - _t_global_start)
    t_phase1 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 1: Multi-init exploration (budget {t_phase1:.0f}s) ===")

    shapes_to_try = [
        ('parabolic', 0.5), ('parabolic', 1.0), ('parabolic', 1.5), ('parabolic', 2.0),
        ('cosine', 0.5), ('cosine', 1.0), ('cosine', 1.5),
        ('super_gaussian', 0.5), ('super_gaussian', 1.0), ('super_gaussian', 2.0),
        ('bicubic', 1.0), ('raised_cosine', 1.0),
    ]

    for shape_type, alpha in shapes_to_try:
        if time.monotonic() - _t_global_start > time_budget - 60:
            break
        f = make_parametric_shape(n0, shape_type, alpha)
        sc = evaluate_safe(f)
        if np.isfinite(sc):
            print(f"  Shape {shape_type}(a={alpha}): J={sc:.6f}")
            if sc < _global_best_score:
                _global_best_score = sc
                _global_best = f.copy()
                print(f"  *** New best from shape: {sc:.6f}")

    # Also try random perturbation of parent
    if parent is not None:
        for trial in range(5):
            if time.monotonic() - _t_global_start > time_budget - 60:
                break
            noise = rng.standard_normal(n0) * 0.1
            perturbed = np.maximum(parent + noise, 0.0)
            perturbed = perturbed / perturbed.sum()
            sc = evaluate_safe(perturbed)
            if np.isfinite(sc):
                print(f"  Perturbed parent trial {trial}: J={sc:.6f}")
                if sc < _global_best_score:
                    _global_best_score = sc
                    _global_best = perturbed.copy()

    # --- Phase 2: Mirror descent at parent length (30% time) ---
    remaining = time_budget - (time.monotonic() - _t_global_start)
    t_phase2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: MD at n={n0} (budget {t_phase2:.0f}s) ===")

    if t_phase2 > 30 and _global_best is not None:
        a_in = normalize_simplex(_global_best)
        # Run MD with different hyperparameters
        t_half = t_phase2 * 0.5
        a_out, _ = mirror_descent_nesterov(a_in, t_half, top_k=40, beta=3.0, gamma=1.0, verbose=True)
        try_update(a_out, f"md1_n{n0}")

        remaining = time_budget - (time.monotonic() - _t_global_start)
        if remaining > 30:
            a_in = normalize_simplex(_global_best)
            a_out, _ = mirror_descent_nesterov(a_in, min(remaining, t_phase2 * 0.4),
                                               top_k=60, beta=4.0, gamma=2.0, verbose=True)
            try_update(a_out, f"md2_n{n0}")

    # --- Phase 3: Active-set polishing (10% time) ---
    remaining = time_budget - (time.monotonic() - _t_global_start)
    t_phase3 = min(remaining * 0.10, 170.0)
    print(f"\n=== Phase 3: Polish at n={n0} (budget {t_phase3:.0f}s) ===")

    if t_phase3 > 15 and _global_best is not None:
        a_in = normalize_simplex(_global_best)
        a_out, _ = active_set_polish(a_in, t_phase3, verbose=True)
        try_update(a_out, f"polish_n{n0}")

    # --- Phase 4: Multi-length scaling (25% time) ---
    remaining = time_budget - (time.monotonic() - _t_global_start)
    t_phase4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: Multi-length scaling (budget {t_phase4:.0f}s) ===")

    target_lengths = [2000, 2500, 3000, 4000]
    for new_n in target_lengths:
        if time.monotonic() - _t_global_start > time_budget - 30:
            break
        if _global_best is None:
            break
        # Only scale if it might help (larger n gives more degrees of freedom)
        elapsed = time.monotonic() - _t_global_start
        remaining = time_budget - elapsed
        if remaining < 40:
            break

        a_long = interpolate_sequence(_global_best, new_n)
        sc_long = evaluate_safe(a_long)
        print(f"  Interpolated to n={new_n}: J={sc_long:.6f}")

        if not np.isfinite(sc_long):
            continue

        t_sub = min(remaining * 0.4, 120.0)
        if t_sub > 20:
            a_in = normalize_simplex(a_long)
            a_out, _ = mirror_descent_nesterov(a_in, t_sub, top_k=50, beta=3.5,
                                               gamma=1.5, verbose=(new_n <= 3000))
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - _t_global_start)
        if remaining > 20 and _global_best is not None and len(_global_best) == new_n:
            t_pol = min(remaining * 0.3, 60.0)
            if t_pol > 10:
                a_in = normalize_simplex(_global_best)
                a_out, _ = active_set_polish(a_in, t_pol, verbose=False)
                try_update(a_out, f"polish_n{new_n}")

    # --- Phase 5: Random restarts (10% time) ---
    remaining = time_budget - (time.monotonic() - _t_global_start)
    t_phase5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Random restarts (budget {t_phase5:.0f}s) ===")

    if t_phase5 > 30 and _global_best is not None:
        n_best = len(_global_best)
        t_per_restart = max(10.0, t_phase5 / 5.0)
        for r in range(5):
            if time.monotonic() - _t_global_start > time_budget - 20:
                break
            # Create diverse initialization
            x = np.linspace(0, 1, n_best)
            blend_alpha = rng.uniform(0.3, 0.7)
            noise_scale = rng.uniform(0.05, 0.2)
            base = make_parametric_shape(n_best, 'cosine', rng.uniform(0.5, 2.0))
            init = blend_alpha * normalize_simplex(_global_best) + (1 - blend_alpha) * base
            init += noise_scale * rng.standard_normal(n_best)
            init = np.maximum(init, 0.0)
            init = init / init.sum()

            a_out, _ = mirror_descent_nesterov(init, t_per_restart, top_k=40, beta=3.0,
                                               gamma=1.0, verbose=False)
            try_update(a_out, f"restart_{r}")

    # --- Phase 6: Final intensive polish (remaining time) ---
    remaining = time_budget - (time.monotonic() - _t_global_start)
    print(f"\n=== Phase 6: Final polish (budget {remaining:.0f}s) ===")

    if remaining > 20 and _global_best is not None:
        n_final = len(_global_best)
        a_in = normalize_simplex(_global_best)

        # Final MD with tuned parameters
        t_md = min(remaining * 0.4, 200.0)
        if t_md > 15:
            a_out, _ = mirror_descent_nesterov(a_in, t_md, top_k=50, beta=3.0,
                                               gamma=0.8, verbose=True)
            try_update(a_out, f"final_md_n{n_final}")

        # Final polish
        remaining = time_budget - (time.monotonic() - _t_global_start)
        if remaining > 15:
            a_in = normalize_simplex(_global_best)
            a_out, _ = active_set_polish(a_in, min(remaining * 0.6, 150.0), verbose=True)
            try_update(a_out, f"final_polish_n{n_final}")

    # --- Final validation ---
    if _global_best is not None:
        fs = evaluate_safe(_global_best)
        if np.isfinite(fs):
            _global_best_score = fs
        print(f"\nFINAL: score={_global_best_score:.6f}, n={len(_global_best)}, "
              f"time={time.monotonic() - _t_global_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Returning default.")
        return [1.0]

    return [float(x) for x in _global_best]
