def lse_objective_grad(x, n, T):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T
    obj = 2.0 * n * lse / (s * s)

    # Gradient via FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    A_f = rfft(a_pad, fft_size)
    W_f = rfft(w_pad, fft_size)
    cross = irfft(A_f * np.conj(W_f), fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad

            break
        remaining = time_limit * 0.9 - elapsed
        if remaining < 5:
            break
        stage_time = remaining / max(1, n_stages - si)
        stage_time = min(stage_time, 120.0)

        x0 = a.copy()

        def obj_wrapper(x):
            return lse_objective_grad(x, n, T)[0]

        def grad_wrapper(x):
            return lse_objective_grad(x, n, T)[1]

        try:

                                 jac=grad_wrapper,

                                 options={'maxiter': 500, 'maxfun': 1200,
                                          'time_limit': stage_time, 'gtol': 1e-10})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm / s_c
                    sc = fast_autocorr(a_cand).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
                        if verbose:
                            print(f"    LSE T={T:.2f}: score={best_score:.8f} ({time.time()-t0:.0f}s)")
        except Exception:
            pass

    return best_a, best_score


def peak_targeted_cd(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.time()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        sc = fast_autocorr(trial).max()
        if sc < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = sc
            stall = 0
            return True
        return False

    eval_count = 0
    while time.time() - t0 < time_limit * 0.95:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:15]
        improved = False

        # Targeted moves: move mass from high contributors to low-contribution positions
        for k_peak in top_peaks[:8]:
            if improved:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:8]]

            # Build peak mask (positions that contribute to any top peak)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                # Find safe destinations (not in peak mask)
                safe_dst = np.where(~peak_mask)[0]
                if len(safe_dst) == 0:
                    safe_dst = np.where(np.arange(n) != src)[0]
                # Prefer low-value positions
                safe_dst = safe_dst[np.argsort(a[safe_dst])][:20]

                for dst in safe_dst[:8]:
                    if a[dst] >= a[src]:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        trial = a.copy()
                        trial[src] -= amt
                        trial[dst] += amt
                        eval_count += 1
                        if try_update(trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Random exploration when stalled
        if not improved:
            stall += 1
            n_random = 20 + stall * 5
            for _ in range(n_random):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                eval_count += 1
                if try_update(trial):
                    break

            # Structured perturbation every 50 stalls
            if stall > 0 and stall % 50 == 0:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                trial = a.copy()
                for idx in indices:
                    trial[idx] *= rng.uniform(0.3, 2.0)
                eval_count += 1
                if try_update(trial):
                    pass

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.7
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 2
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.time()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.time() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    n_parent = len(best_seq) if best_seq is not None else 1600
    scan_lengths = sorted(set([n_parent, n_parent + 200, n_parent + 400, n_parent - 100 if n_parent > 1200 else n_parent]))
    scan_lengths = [n for n in scan_lengths if 800 <= n <= 3000]

    for i_scan, n_scan in enumerate(scan_lengths[:3]):
        elapsed = time.time() - t_global
        if elapsed > time_budget * 0.10:
            break
        remaining = time_budget * 0.10 - elapsed
        if remaining < 15:
            break
        scan_time = min(remaining / max(1, 3 - i_scan), 80.0)
        if scan_time < 10:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = normalize(init_seq, n_scan)
        a_out, sc_out = nesterov_lp(a_norm, scan_time, solver, verbose=False)
        do_try_update(a_out, f"scan_n{n_scan}")
        print(f"    n={n_scan}: internal={sc_out:.8f} ({time.time()-t_global:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Main LP optimization
    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.30
    a_norm = normalize(best_seq, n_base)
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, use_peak_targeting=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: LSE optimization with aggressive cooling
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_norm = normalize(best_seq, n_best)
        a_out, _ = lse_optimize(a_norm, t2, T_start=60.0, T_end=0.3, n_stages=8, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Try longer length
    n_long = n_base + 300
    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 3: LP n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = normalize(init_long, n_long)
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, use_peak_targeting=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Noisy restart LP at best length
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP restart n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        noise_scale = 0.03 * best_seq.mean()
        a_noisy = best_seq + rng.normal(0, noise_scale, n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lp(a_noisy, t4, solver, verbose=False, use_peak_targeting=True)
        do_try_update(a_out, f"lp_restart_n{n_best}")

    # Phase 5: Peak-targeted coordinate descent
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak CD n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 15:
        a_norm = normalize(best_seq, n_best)
        a_out, _ = peak_targeted_cd(a_norm, t5, seed=12345)
        do_try_update(a_out, "peak_cd")

    # Phase 6: Final LSE polish
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 10:
        a_norm = normalize(best_seq, n_best)
        a_out, _ = lse_optimize(a_norm, t6, T_start=8.0, T_end=0.1, n_stages=5, verbose=True)
        do_try_update(a_out, "lse_polish")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = max(remaining * 0.5, 10.0)
    remaining_final = time.time() - t_global
    t7 = min(t7, time_budget - 60.0 - remaining_final)
    if t7 > 10:
        a_norm = normalize(best_seq, n_best)
        a_out, _ = peak_targeted_cd(a_norm, t7, seed=99999)
        do_try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.time()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
