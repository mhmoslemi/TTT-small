"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid multi-phase approach:
1. Trigonometric/Gaussian parameterization + L-BFGS-B for fast basin finding
2. LSE-smoothed gradient descent with L-BFGS-B 
3. Exact max subgradient with Nesterov acceleration and backtracking line search
4. Multi-start with diverse initializations (parent, uniform, structured, random)
5. Coordinate descent polish with peak-targeting and mass redistribution
Key innovation: low-dimensional parameterization for fast global search,
combined with high-dimensional exact subgradient for precise local optimization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

def fast_conv(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def subgrad_max_conv(a, peak_indices):
    """
    Subgradient of max(conv(a,a)) at a.
    Uses top-k peak positions for a smoothed subgradient.
    """
    n = len(a)
    m = len(peak_indices)
    grad = np.zeros(n)
    for k in peak_indices:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx_a = k - np.arange(j_start, j_end + 1)
        grad[j_start:j_end + 1] += 2.0 * a[idx_a]
    grad /= m
    return grad

def lse_smooth_max(b, T):
    """Log-sum-exp smoothing of the max."""
    if T < 1e-6:
        return b.max()
    b_shifted = b - b.max()
    max_val = b.max()
    scaled = T * b_shifted
    # clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    return max_val + np.log(np.sum(np.exp(scaled))) / T

def evaluate_local(a):
    """Compute the objective value (scale-invariant)."""
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_conv(a)
    return 2.0 * len(a) * b.max() / (s * s)

def normalize(a, target_sum=None):
    """Normalize a to be non-negative with the target sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a * target_sum / s

def fourier_param_to_seq(params, n):
    """Convert Fourier parameters to a non-negative sequence."""
    # params = [c0, a1, b1, a2, b2, ..., aK, bK]
    c0 = params[0]
    j = np.arange(n)
    seq = np.full(n, c0)
    num_harmonics = (len(params) - 1) // 2
    for k in range(1, num_harmonics + 1):
        ak = params[2 * k - 1]
        bk = params[2 * k]
        seq += ak * np.cos(2 * np.pi * k * j / n) + bk * np.sin(2 * np.pi * k * j / n)
    return np.maximum(seq, 0.0)

def fourier_objective(params, n):
    """Objective for Fourier parameter optimization."""
    seq = fourier_param_to_seq(params, n)
    s = seq.sum()
    if s < 1e-10:
        return 1e10
    seq_norm = seq * np.sqrt(2.0 * n) / s
    b = fast_conv(seq_norm)
    return b.max()

def gaussian_param_to_seq(params, n):
    """Convert Gaussian mixture parameters to a sequence."""
    # params = [w1, mu1, sig1, w2, mu2, sig2, ..., wL, muL, sigL]
    num_comp = len(params) // 3
    j = np.arange(n, dtype=np.float64)
    seq = np.zeros(n)
    for l in range(num_comp):
        w = max(params[3*l], 0.1)  # minimum weight
        mu = params[3*l+1] / (n - 1) if n > 1 else 0.5  # normalize to [0,1]
        sig = max(params[3*l+2], 0.01) / (n - 1) if n > 1 else 0.1
        seq += w * np.exp(-0.5 * ((j / (n-1) - mu) / sig) ** 2)
    return np.maximum(seq, 1e-10)  # avoid all zeros

def gaussian_objective(params, n):
    """Objective for Gaussian mixture parameter optimization."""
    seq = gaussian_param_to_seq(params, n)
    s = seq.sum()
    if s < 1e-10:
        return 1e10
    seq_norm = seq * np.sqrt(2.0 * n) / s
    b = fast_conv(seq_norm)
    return b.max()

def nesterov_subgrad_optimize(a0, time_limit, num_peaks=5, momentum_init=0.5, beta=0.9, verbose=True):
    """
    Nesterov-accelerated subgradient descent for min max(conv(a,a))
    subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0, target_sum)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    
    y = a.copy()  # Nesterov's y
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    lr = 1.0 / (n ** 0.5)
    stall = 0
    iter_count = 0
    
    patience = 200
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        a_eval = y + momentum_init * (y - a)
        a_eval = normalize(a_eval, target_sum)
        
        # Compute convolution and find peaks
        b = fast_conv(a_eval)
        score = b.max()
        
        top_k = min(num_peaks, len(b))
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        # Compute subgradient
        g = subgrad_max_conv(a_eval, peak_indices)
        
        # Account for the normalization constraint:
        # We're minimizing on the manifold sum(a) = target_sum, a >= 0
        # Project gradient to the tangent space: subtract mean to handle the sum constraint
        g_tangent = g - g.mean()
        
        # Momentum update
        v = beta * v - lr * g_tangent
        y_new = y + v
        
        # Project: clip to non-negative, rescale
        a_new = np.maximum(y_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * target_sum / s
        
        iter_count += 1
        
        # Check improvement
        score_new = fast_conv(a_new).max()
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            # Anneal learning rate
            lr = max(lr * 0.999, 1e-6)
        else:
            no_improve += 1
        
        y = a_new
        
        # Adaptive momentum
        if no_improve > 100:
            momentum_init = max(momentum_init * 0.95, 0.1)
            beta = min(beta * 1.001, 0.999)
            lr = max(lr * 0.9, 1e-7)
        
        if no_improve > patience:
            # Restart with perturbation
            rng = np.random.default_rng(iter_count)
            perturbation = rng.normal(0, 0.05 * best_a.mean(), n)
            a_new = best_a + perturbation
            a_new = normalize(a_new, target_sum)
            y = a_new.copy()
            v = np.zeros(n)
            lr = 1.0 / (n ** 0.5)
            momentum_init = 0.5
            no_improve = 0
            stall += 1
            if stall > 20:
                break
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score

def lse_lbfgs_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=5, verbose=True):
    """LSE-smoothed L-BFGS-B optimization."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0, target_sum)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 200.0)
        
        def obj(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            x_norm = x * target_sum / s
            b = fast_conv(x_norm)
            return lse_smooth_max(b, T)
        
        def grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return np.zeros(n)
            x_norm = x * target_sum / s
            b = fast_conv(x_norm)
            max_b = b.max()
            scaled = T * (b - max_b)
            scaled = np.clip(scaled, -500, 500)
            exp_vals = np.exp(scaled)
            w = exp_vals / exp_vals.sum()
            
            # Gradient of LSE wrt x_norm:
            # dLSE/db * db/dx_norm
            # db[k]/dx_norm[j] = 2*x_norm[k-j] (valid) + 2*x_norm[j-(k)] ... 
            # Actually: db[k]/dx[j] = 2*x[k-j] if 0<=k-j<n, else 0 (for conv(x,x))
            # But we need it for the weighted sum: sum_k w[k] * db[k]
            # = sum_k w[k] * 2*x[k-j] (for valid j)
            # This is 2 * conv(w, x_reversed)[j]... let me compute it directly
            
            grad_xnorm = np.zeros(n)
            # For each k, contribution to grad[j] is w[k] * 2 * x[k-j] (if valid)
            # = 2 * sum_k w[k] * x[k-j] (if 0 <= k-j < n)
            # Let d = k-j, so k = j+d, and we need 0 <= d < n
            # sum_k w[k] * x[k-j] = sum_d w[j+d] * x[d] (for valid j+d)
            # This is a cross-correlation!
            
            # Efficient computation using FFT:
            # sum_d w[j+d] * x[d] for d=0..n-1 (with valid j+d < 2n-1)
            # = (wrev * x)[j] where wrev is w reversed
            # Actually, let me just use the direct subgradient formula
            
            # grad_xnorm[j] = 2 * sum_{k: 0<=k-j<n} w[k] * x[k-j]
            # = 2 * sum_{d=0}^{n-1} w[j+d] * x[d] (where j+d < 2n-1)
            
            # Use FFT: pad w and x to size >= 3n-2 for the correlation
            fft_size = 1 << (3 * n - 2).bit_length()
            w_pad = np.zeros(fft_size)
            w_pad[:len(w)] = w
            x_pad = np.zeros(fft_size)
            x_pad[:n] = x
            # Cross-correlation: sum_d w[j+d] * x[d]
            # = ifft(FFT(w_rev) * FFT(x)) where w_rev is w reversed
            # Or: = ifft(conj(FFT(w)) * FFT(x_shifted))... 
            # Let me use the direct approach for correctness
            
            # Actually: sum_k w[k]*x[k-j] for 0<=k-j<n, 0<=k<2n-1
            # Let me compute it as: for each j, sum over k from j to j+n-1 of w[k]*x[k-j]
            
            # This is the cross-correlation of w and x:
            # corr[j] = sum_k w[k] * x[k-j]
            
            w_full = w  # length 2n-1
            # corr[j] = sum_{d=0}^{n-1} w[j+d] * x[d] (for j+d < 2n-1)
            # = sum_{d=0}^{n-1} x[d] * w[j+d]
            
            # This is a convolution: (x * w_rev)[j + n - 1] where w_rev = w[::-1]
            # Let me just compute it directly using FFT
            x_pad2 = np.zeros(fft_size)
            x_pad2[:n] = x
            w_rev_pad = np.zeros(fft_size)
            w_rev = w_full[::-1]  # reverse w
            w_rev_pad[:len(w_rev)] = w_rev
            # (x * w_rev)[m] = sum_d x[d] * w_rev[m-d]
            # We want corr[j] = sum_d x[d] * w[j+d] = sum_d x[d] * w_rev[(2n-2)-(j+d)]
            # = (x * w_rev)[(2n-2)-j]
            
            conv_result = irfft(rfft(x_pad2, fft_size) * rfft(w_rev_pad, fft_size), fft_size)[:2*n-1+1]
            # corr[j] = conv_result[(2n-2)-j] for j = 0, ..., n-1
            idx = (2*n-2) - np.arange(n)
            corr = conv_result[idx]
            
            grad_xnorm = 2.0 * corr
            
            # Chain rule: x_norm = x * target_sum / s, where s = sum(x)
            # dx_norm[j]/dx[i] = target_sum/s * delta[j,i] - x_norm[j] * x[i] / s^2 * target_sum
            # = target_sum/s * (delta[j,i] - x_norm[j]*x[i]/s)
            # 
            # grad_obj[x[i]] = sum_j grad_xnorm[j] * dx_norm[j]/dx[i]
            # = target_sum/s * (sum_j grad_xnorm[j]*delta[j,i] - sum_j grad_xnorm[j]*x_norm[j]*x[i]/s)
            # = target_sum/s * (grad_xnorm[i] - x[i]/s * sum_j grad_xnorm[j]*x_norm[j])
            
            dot = np.dot(grad_xnorm, x_norm)
            grad_result = (target_sum / s) * (grad_xnorm - x * dot / s)
            
            return grad_result
        
        try:
            res = scipy_minimize(obj, a.copy(), method='L-BFGS-B',
                                  jac=grad,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'ftol': 1e-12, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = normalize(res.x, target_sum)
                score = fast_conv(a_candidate).max()
                if score < best_score:
                    best_score = score
                    best_a = a_candidate.copy()
                    if verbose:
                        print(f"      LSE stage {si} (T={T:.1f}): new best={best_score:.8f}")
        except Exception as e:
            if verbose:
                print(f"      LSE stage {si} error: {e}")
    
    return best_a, best_score

def fourier_optimize(n, time_limit, n_harmonics=15, n_starts=5, verbose=True):
    """Optimize using Fourier parameterization."""
    best_seq = None
    best_score = float('inf')
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    n_params = 1 + 2 * n_harmonics
    
    for start in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / max(1, n_starts - start), 100.0)
        
        if start == 0:
            # Start from a smooth bell shape
            params = np.zeros(n_params)
            params[0] = 1.0
            params[1] = 0.3  # a1
            params[3] = 0.15  # a2
        else:
            # Random start
            params = rng.normal(0, 0.3, n_params)
            params[0] = abs(params[0]) + 1.0  # ensure positive mean
        
        bounds = [(-2, 2)] * n_params
        
        try:
            res = scipy_minimize(fourier_objective, params, args=(n,),
                                  method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': 300, 'maxfun': 800,
                                           'ftol': 1e-14, 'gtol': 1e-10})
            if res.fun < best_score:
                best_score = res.fun
                seq = fourier_param_to_seq(res.x, n)
                s = seq.sum()
                if s > 1e-10:
                    seq = seq * np.sqrt(2.0 * n) / s
                    best_seq = seq.copy()
                    if verbose:
                        print(f"      Fourier start {start}: score={best_score:.8f}")
        except Exception:
            pass
    
    if best_seq is None:
        return None, float('inf')
    return best_seq, best_score

def gaussian_optimize(n, time_limit, n_components=4, n_starts=5, verbose=True):
    """Optimize using Gaussian mixture parameterization."""
    best_seq = None
    best_score = float('inf')
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    n_params = 3 * n_components
    
    for start in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / max(1, n_starts - start), 80.0)
        
        if start == 0:
            # Single Gaussian at center
            params = np.zeros(n_params)
            params[0] = 1.0  # weight
            params[1] = 0.5  # center
            params[2] = 0.2  # width
        elif start == 1:
            # Two Gaussians
            params = np.zeros(n_params)
            params[0] = 1.0
            params[1] = 0.3
            params[2] = 0.15
            params[3] = 1.0
            params[4] = 0.7
            params[5] = 0.15
        else:
            params = np.zeros(n_params)
            for l in range(n_components):
                params[3*l] = rng.uniform(0.5, 2.0)
                params[3*l+1] = rng.uniform(0.1, 0.9)
                params[3*l+2] = rng.uniform(0.05, 0.3)
        
        bounds = []
        for l in range(n_components):
            bounds.extend([(0.01, 5.0), (-0.5, 1.5), (0.01, 1.0)])
        
        try:
            res = scipy_minimize(gaussian_objective, params, args=(n,),
                                  method='Nelder-Mead',
                                  bounds=bounds,
                                  options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < best_score:
                best_score = res.fun
                seq = gaussian_param_to_seq(res.x, n)
                s = seq.sum()
                if s > 1e-10:
                    seq = seq * np.sqrt(2.0 * n) / s
                    best_seq = seq.copy()
                    if verbose:
                        print(f"      Gaussian start {start}: score={best_score:.8f}")
        except Exception:
            pass
    
    if best_seq is None:
        return None, float('inf')
    return best_seq, best_score

def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting coordinate descent."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0, target_sum)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-10:
            return False
        trial = trial * target_sum / s
        score = fast_conv(trial).max()
        if score < best_score - 1e-12:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        top_peaks = np.argsort(b)[::-1][:8]
        
        improved = False
        
        for k_peak in top_peaks:
            if improved:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:4]]
            
            # Build peak influence mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                peak_mask[max(0, k2-n+1):min(n, k2+1)] = True
            
            for src in top_contrib:
                if improved:
                    break
                if a[src] < 1e-8:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Pick candidates with low values
                candidates_sorted = candidates[np.argsort(a[candidates])][:10]
                
                for dst in candidates_sorted:
                    if improved:
                        break
                    for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        if a_trial[src] - amt < -1e-12:
                            continue
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            improved = True
                            break
        
        if not improved:
            stall += 1
            # Random perturbations
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if try_update(a_trial):
                    break
            
            # Larger perturbation
            if stall % 20 == 0:
                n_perturb = max(5, n // 50)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                a_trial[indices] *= rng.uniform(0.4, 1.6, n_perturb)
                if try_update(a_trial):
                    pass
            
            if stall > 500:
                break
        
        if verbose and stall % 100 == 0 and stall > 0:
            print(f"      CD stall={stall}, best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng, style=None):
    """Generate structured initial sequences."""
    if style is None:
        style = rng.integers(0, 8)
    
    if style == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif style == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif style == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.3) ** 2) + 0.7 * np.exp(-6 * (x - 0.7) ** 2)
    elif style == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.1)
    elif style == 4:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x * 3)) ** 0.5
    elif style == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1 + 10 * (x - 0.5) ** 2)
    elif style == 6:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x * 2)) ** 0.7
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def power_iter_refine(a0, time_limit, num_peaks=3, verbose=False):
    """
    Iteratively reduce the top peaks by solving a least-squares subproblem.
    At each step, find the direction that most reduces the current peak.
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0, target_sum)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        peak_idx = b.argmax()
        current_peak = b[peak_idx]
        
        # Get subgradient
        g = subgrad_max_conv(a, [peak_idx])
        g_tangent = g - g.mean()
        
        # Try different step sizes
        best_step_score = best_score
        best_step_a = None
        
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a - lr * g_tangent
            a_trial = normalize(a_trial, target_sum)
            score = fast_conv(a_trial).max()
            if score < best_step_score:
                best_step_score = score
                best_step_a = a_trial
        
        if best_step_a is not None and best_step_score < best_score - 1e-12:
            a = best_step_a
            best_a = best_step_a.copy()
            best_score = best_step_score
            if verbose:
                print(f"      PI: peak={best_score:.8f}")
        else:
            break
    
    return best_a, best_score
