def project_nonneg_sum(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_fista_short(a0, time_limit, verbose=True):
    """Short LSE-FISTA warmup with aggressive annealing."""

        a = a / s

    T_schedule = [10.0, 3.0, 1.0, 0.3, 0.1]
    
    for stage, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), 1.0)
        t_fista = 1
        staging = a.copy()
        stage_time = (time_limit / len(T_schedule))
        
        while time.monotonic() - t0 < time_limit - (len(T_schedule) - stage - 1) * stage_time:
            ac = fast_autocorr(y)
            ac_max = ac.max()
            if ac_max < 1e-15:
                break
            shifted = np.clip((ac - ac_max) / T, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            
            N = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N)
            Y_rev = rfft(y[::-1], N)

            grad = 2.0 * cross[n - 1: n - 1 + n]
            
            step = 1.0 / L

                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_nonneg_sum(y_new, 1.0)
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            if t_fista % 200 == 0:
                sc = fast_autocorr(a).max()

def lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0"""

    # Build convolution matrix efficiently using broadcasting
    # (a*b)_k = sum_j a[j] * b[k-j] for valid j
    # Use the Toeplitz structure
    T_parts = []
    for j in range(n):
        row_start = j
        row_end = j + n
        T_parts.append(None)
    
    # Actually, let's just use the direct formulation with sparsity
    # The constraint matrix has (2n-1) rows and n columns
    # Entry (k,j) = a[|k-j|] if |k-j| < n, else 0... 
    # Wait, no. (conv(a,b))_k = sum_j a[j] * b[k-j] where indices are 0-based
    
    # For full convolution: (a*b)[k] = sum_{j=0}^{min(k,n-1)} a[j] * b[k-j] for k=0..2n-2
    # Let me just build it as a sparse matrix or use the dense formulation
    
    # Dense approach for n=1600: 3199 x 1600 matrix = ~5M entries, manageable
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    
    # Get dual variables for active constraints
    try:
        dual_vals = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        dual_vals = None
    
    return b_val, dual_vals

    log_geom = np.log(a + 1e-30)  # for geometric mean

    count_avg = 1
    
    t0 = time.monotonic()

            continue
        b_sum = b.sum()

        g = b * norm / b_sum
        
        # Nesterov update
        t = n_iters
        alpha = 2.0 / (t + 2.0)

        # Find top peaks

        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        for k in peak_indices:
            # Gradient of (a*a)_k w.r.t. a_i = a_{i+k} + a_{i-k} (with bounds)

            for i in range(n):
                val = 0.0
                j1 = i + k
                j2 = i - k
                if 0 <= j1 < n:
                    val += a[j1]
                if 0 <= j2 < n:
                    val += a[j2]
                grad_k[i] = val
            
            # Find positions with highest gradient (most contribute to this peak)
            # and positions with low gradient (safe to add mass)
            top_contrib = np.argsort(grad_k)[::-1][:max(5, n // 20)]

                for dst in low_contrib[:10]:
                    if src == dst:
                        continue

                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:

            stall += 1
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-8:

def quadratic_peak_reduction(a0, time_limit, K=10, verbose=True):
    """Second-stage LP: minimize max of top-K autocorrelation peaks via perturbation."""
    import cvxpy as cp
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a0, fast_autocorr(a0).max()
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    solver = get_lp_solver()
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(best_a)
        peak_idx = np.argsort(c)[::-1][:K]
        
        # Build the sub-matrix: for each peak k, the row of the convolution matrix
        # conv(a, delta)[k] = sum_j a[j] * delta[k-j] (valid j)
        # We want to minimize: max_i (c[peak_i] + 2 * conv(a, delta)[peak_i])
        
        delta = cp.Variable(n, nonneg=True)  # delta >= 0 (we'll handle the "sum=0" constraint differently)
        # Actually, we want sum(delta) = 0, so let's use delta as a free variable with sum constraint
        # But we need a + delta >= 0
        
        delta = cp.Variable(n)
        vals = []
        T_sub = np.zeros((K, n))
        for i, k in enumerate(peak_idx):
            # conv(a, delta)[k] = sum_j a[j] * delta[k-j] for valid j
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    T_sub[i, j] = a[j]
            vals.append(c[k] + 2.0 * T_sub[i] @ delta)
        
        t_var = cp.Variable()
        constraints = [
            t_var >= vals[i] for i in range(K),
            cp.sum(delta) == 0,  # preserve the sum
            a + delta >= 0  # non-negativity
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        if delta.value is None:
            break
        
        delta_val = np.asarray(delta.value, dtype=np.float64)
        a_new = a + delta_val
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        if score < best_score - 1e-14:

            best_a = a_new.copy()
            a = a_new.copy()
        else:

                    a = a_trial.copy()
                    break
            else:

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_schedule = [10.0, 3.0, 1.0, 0.3, 0.1]
    n_stages = len(T_schedule)

    for stage, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), 1.0)
        t_fista = 1
        staging = a.copy()
        remaining_stages = n_stages - stage - 1
        stage_end = t0 + time_limit - remaining_stages * (time_limit / n_stages)

        while time.monotonic() < stage_end:
            ac = fast_autocorr(y)
            ac_max = ac.max()
            if ac_max < 1e-15:
                break
            shifted = np.clip((ac - ac_max) / T, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()

            N = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N)
            Y_rev = rfft(y[::-1], N)
            cross = irfft(W * Y_rev, N)
            grad = 2.0 * cross[n - 1: n - 1 + n]

            step = 1.0 / L
            z = y - step * grad
            a_new = project_nonneg_sum(z, 1.0)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_nonneg_sum(y_new, 1.0)

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if t_fista % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc = fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    if verbose:
        print(f"  LSE-FISTA done in {time.monotonic()-t0:.1f}s, best={best_score:.8f}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_dual_avg(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = a.copy()
    count_avg = 1
    log_geom = np.log(a + 1e-30)
    count_geom = 1

    t0 = time.monotonic()
    n_iters = 0
    stall_counter = 0
    best_ever = best_score

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        n_iters += 1
        g = b * norm / b_sum

        t = n_iters
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy

        if n_iters == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa

        y = y_new
        a = a_new

        count_geom += 1
        eta_g = 1.0 / (30.0 + count_geom * 0.02)
        log_geom = (1.0 - eta_g) * log_geom + eta_g * np.log(g + 1e-30)

        count_avg += 1
        avg_a = avg_a + (a - avg_a) / count_avg

        if n_iters % 20 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_counter = 0
            else:
                stall_counter += 1

            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_counter = 0

            if n_iters % 50 == 0:
                geom_a = np.exp(log_geom)
                geom_a = np.maximum(geom_a, 0.0)
                sg = geom_a.sum()
                if sg > 1e-15:
                    geom_a = geom_a * norm / sg
                    score_g = fast_autocorr(geom_a).max()
                    if score_g < best_score:
                        best_score = score_g
                        best_a = geom_a.copy()
                        stall_counter = 0

                avg_check = avg_a * norm / avg_a.sum()
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_counter = 0

                elapsed = time.monotonic() - t0
                print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

            if stall_counter > 400 and n_iters > 300:
                y = best_a * norm / best_a.sum()
                a = y.copy()
                log_geom = np.log(y + 1e-30)
                count_geom = 1
                avg_a = y.copy()
                count_avg = 1
                stall_counter = 0

    for candidate in [y, a, np.exp(log_geom), avg_a]:
        candidate = np.maximum(candidate, 0.0)
        sc = candidate.sum()
        if sc > 1e-15:
            candidate = candidate * norm / sc
            score = fast_autocorr(candidate).max()
            if score < best_score:
                best_score = score
                best_a = candidate.copy()

    if verbose:
        print(f"  LP-Nesterov done in {time.monotonic()-t0:.1f}s, iters={n_iters}, best={best_score:.8f}")
    return best_a, best_score


def quadratic_peak_reduction(a0, time_limit, K=8, verbose=True):
    import cvxpy as cp
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a0, fast_autocorr(a0).max()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    solver = get_lp_solver()
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(best_a)
        peak_idx = np.argsort(c)[::-1][:K]

        delta = cp.Variable(n)
        T_sub = np.zeros((K, n))
        for i, k in enumerate(peak_idx):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    T_sub[i, j] = a[idx]

        t_var = cp.Variable()
        constraints = []
        for i in range(K):
            constraints.append(t_var >= c[peak_idx[i]] + 2.0 * T_sub[i] @ delta)
        constraints.append(cp.sum(delta) == 0)
        constraints.append(a + delta >= 0)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        if delta.value is None:
            break

        delta_val = np.asarray(delta.value, dtype=np.float64)
        n_iters += 1

        a_new = a + delta_val
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose and n_iters % 20 == 0:
                print(f"    QPeak iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
        else:
            improved = False
            for alpha in [0.5, 0.2, 0.1, 0.05]:
                a_trial = a + alpha * delta_val
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial * norm / st
                sc = fast_autocorr(a_trial).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    if verbose:
        print(f"  Quadratic peak reduction done in {time.monotonic()-t0:.1f}s, iters={n_iters}, best={best_score:.8f}")
    return best_a, best_score


def peak_directed_search(a0, time_limit, seed=42, verbose=True):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        n_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in peak_indices:
            grad_k = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad_k[i] = 2.0 * a[k - i]

            max_grad = grad_k.max()
            if max_grad < 1e-15:
                continue

            top_contrib = np.where(grad_k > 0.7 * max_grad)[0]
            low_contrib = np.where(grad_k < 0.3 * max_grad)[0]

            if len(top_contrib) == 0 or len(low_contrib) == 0:
                continue

            src_candidates = top_contrib[np.argsort(grad_k[top_contrib])[::-1]][:15]
            dst_candidates = low_contrib[np.argsort(a[low_contrib])][:15]

            for src in src_candidates:
                if a[src] < 1e-10:
                    continue
                for dst in dst_candidates:
                    if src == dst:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0:
                            break
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            break
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < norm * 1e-8:
                    continue
                frac = rng.uniform(0.002, 0.012)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if stall > 200:
                break

    if verbose:
        print(f"  Peak-directed done in {time.monotonic()-t0:.1f}s, best={best_score:.8f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

    except Exception:
        pass

    def try_update(a, label):

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global

        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Multi-start FISTA warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 60.0)

        best_fista_score = best_score

        init_shapes.append(np.sin(np.pi * x) ** 2.0)
        init_shapes.append((1.0 + np.cos(2.0 * np.pi * x)) / 2.0)
        init_shapes.append(np.sin(np.pi * x) ** 0.7)

        t_per_shape = t1 / len(init_shapes)
        for si, shape in enumerate(init_shapes):
            shape = np.maximum(shape, 0.0)
            ss = shape.sum()
            if ss < 1e-15:
                continue
            shape = shape / ss
            rem = time_budget - (time.monotonic() - t_global)
            if rem < 5:
                break
            tl = min(t_per_shape, rem * 0.7)
            if tl < 5:

            a_out, sc = lse_fista_short(shape, tl, verbose=False)

    # === Phase 2: LP-Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)

    # Final verification
    if best_seq is not None:

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
