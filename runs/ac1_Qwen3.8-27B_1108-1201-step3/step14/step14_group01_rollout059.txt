    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0 using MWA."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    b = np.ones(n)  # initial primal
    lam = np.ones(size)  # initial dual
    
    for it in range(n_inner_iters):
        # Compute a*b (linear convolution)
        Ab = fast_conv_cross(a, b)
        
        # Dual update
        violation = np.maximum(Ab - M, 0)
        lam = lam * np.exp(eta * violation)
        
        # Normalize lam to prevent overflow
        lam_max = lam.max()
        if lam_max > 1e10:
            lam = lam / lam_max
        
        # Compute gradient: grad_j = sum_k lam_k * T_{k,j}
        # T_{k,j} = a_{k-j} if 0 <= k-j <= n-1
        # grad_j = sum_k lam_k * a_{k-j} = cross-correlation of lam and a

        # This is np.correlate(lam, a, 'full')[n-1-j] or similar
        
        # Actually: sum_k lam[k] * a[k-j] for k = j to j+n-1
        # = sum_i lam[j+i] * a[i] for i = 0 to n-1
        # This is the cross-correlation of lam and a at lag j
        
        # Using FFT:
        L = rfft(lam, fft_size)
        A = rfft(a, fft_size)

        M = conv_y.max()
        if M < 1e-12:
            break

        b = mwa_lp_solve(y, M, n_inner, eta)

        s_b = b.sum()
        if s_b < 1e-15:
            continue
        g = b * norm / s_b

        alpha = 2.0 / (total_iters + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if total_iters == 1:
            x_new = g.copy()
        else:
            momentum = (total_iters - 1.0) / (total_iters + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x

        y = y_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_x = fast_conv_self(x_new).max()
        if sc_x < best_score - 1e-12:
            best_score = sc_x
            best_a = x_new.copy()
            no_improve = 0

        # Periodic perturbation to escape local optima
        if no_improve > 2000 and no_improve % 500 == 0:
            n_pert = max(10, n // 40)
            idx = np.random.default_rng(total_iters).choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.15 * np.random.default_rng(total_iters + 1).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWA Miter it={total_iters} ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score


def spectral_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain to flatten the power spectrum."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = rfft(a, fft_size)
    freqs = np.arange(fft_size // 2 + 1)

    # Compute current power spectrum
    P = np.abs(A) ** 2
    P_mean = P[P > 0].mean() if (P > 0).any() else 1.0

    # Clip peaks in spectrum
    P_clipped = np.minimum(P, P_mean * 3.0)

    # Reconstruct with modified magnitude, preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_clipped) * np.exp(1j * phase)

    # For real signal, enforce conjugate symmetry
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s

    # Blend with original
    a_result = a * (1 - strength) + a_new * strength
    a_result = np.maximum(a_result, 0.0)
    s_r = a_result.sum()
    if s_r > 1e-15:
        a_result = a_result / s_r
    return a_result


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Gradient-guided pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Select j candidates: random + gradient-informed
            peak_threshold = cur_max * 0.9
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(10):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_struct = int(kl - i)
                if 0 <= j_struct < n:
                    j_candidates.add(j_struct)
                j_struct2 = int(kl + i)
                if 0 <= j_struct2 < n:
                    j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 25
                deltas = np.linspace(0.05, 0.95, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve >= 4:
                n_perturb = max(5, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(30.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(150, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= conv.max() + 0.01 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.4 + 0.4 * np.sin(3 * np.pi * x) ** 2
    elif choice == 8:
        a = 0.5 + 0.25 * np.cos(4 * np.pi * x) + 0.15 * np.cos(7 * np.pi * x)
    else:
        a = np.exp(-5.0 * (x - 0.5) ** 2) + 0.4
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_state, a, label, t0):
    if a is None:
        return best_state
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_state
    a = a / s
    sc = eval_fn(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        elapsed = time.monotonic() - t0
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_state


def mwa_miter_optimize_advanced(a0, time_limit, n_inner, eta, rng, verbose=True):
    """Advanced MWA miter with adaptive inner iterations and perturbation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    no_improve = 0
    total_iters = 0
    eta_current = eta
    n_inner_current = n_inner

    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive: more inner iters when stuck
        if no_improve > 1000:
            n_inner_use = min(n_inner_current * 2, 200)
            eta_use = eta_current * 1.5
        else:
            n_inner_use = n_inner_current
            eta_use = eta_current

        b = mwa_lp_solve(y, M, n_inner_use, eta_use)
        s_b = b.sum()
        if s_b < 1e-15:
            continue
        g = b * norm / s_b

        alpha = 2.0 / (total_iters + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if total_iters == 1:
            x_new = g.copy()
        else:
            momentum = (total_iters - 1.0) / (total_iters + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x

        y = y_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_x = fast_conv_self(x_new).max()
        if sc_x < best_score - 1e-12:
            best_score = sc_x
            best_a = x_new.copy()
            no_improve = 0

        # Spectral perturbation when stuck
        if no_improve > 3000 and no_improve % 1000 == 0:
            y = spectral_perturb(y, rng, strength=0.15)
            norm_y = y.sum()
            if norm_y > 1e-15:
                y = y * norm / norm_y
            no_improve = 0
            # Reset eta to explore
            eta_current = eta * (0.5 + rng.random())
            print(f"    Spectral perturbation at iter {total_iters}, eta={eta_current:.3f}")

        # Standard perturbation
        if no_improve > 5000 and no_improve % 2000 == 0:
            n_pert = max(10, n // 30)
            idx = rng.choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y2 = y.sum()
            if s_y2 > 1e-15:
                y = y * norm / s_y2
            no_improve = 0

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWA+ it={total_iters} ({elapsed:.0f}s): best={best_score:.8f}, no_imp={no_improve}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function that orchestrates all phases."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_state = {'seq': None, 'score': float('inf')}

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_state['seq'] = p / p.sum()
            best_state['score'] = eval_fn(best_state['seq'])
            print(f"Parent loaded: n={len(best_state['seq'])}, score={best_state['score']:.6f}")
    except Exception:
        pass

    # Initialize if needed
    if best_state['seq'] is None:
        n_base = 1600
        best_state['seq'] = structured_init(n_base, rng)
        best_state['score'] = eval_fn(best_state['seq'])
        print(f"Default init: n={len(best_state['seq'])}, score={best_state['score']:.6f}")

    n_base = len(best_state['seq'])

    # === Phase 1: Fast MWA Miter (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: MWA Miter n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = mwa_miter_optimize_advanced(best_state['seq'], t1, 80, 1.0, rng, verbose=True)
        best_state = try_update(best_state, a_out, f"mwa1_n{n_base}", t_global)

    # === Phase 2: LSE Gradient Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 2: LSE GD n={len(best_state['seq'])} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_gradient_descent(best_state['seq'], t2, verbose=True)
        best_state = try_update(best_state, a_out, f"lse_n{len(best_state['seq'])}", t_global)

    # === Phase 3: MWA Miter round 2 (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: MWA Miter round 2 (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = mwa_miter_optimize_advanced(best_state['seq'], t3, 100, 0.8, rng, verbose=True)
        best_state = try_update(best_state, a_out, f"mwa2_n{len(best_state['seq'])}", t_global)

    # === Phase 4: Coordinate Descent Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: Pair CD (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_state['seq'], t4, rng, verbose=True)
        best_state = try_update(best_state, a_out, f"pcd_n{len(best_state['seq'])}", t_global)

    # === Phase 5: Multi-restart with spectral perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_restarts = 3
    t5_per = min(remaining * 0.15, 100.0) / n_restarts
    print(f"\n=== Phase 5: Multi-restart (budget={n_restarts * t5_per:.0f}s) ===")
    for ri in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        a_pert = spectral_perturb(best_state['seq'], rng, strength=0.2 + 0.1 * ri)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s > 1e-15:
            a_pert = a_pert / s
        if t5_per > 15:
            a_out, _ = mwa_miter_optimize_advanced(a_pert, min(t5_per, remaining * 0.3), 60, 1.2, rng, verbose=False)
            best_state = try_update(best_state, a_out, f"restart{ri}", t_global)

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.20, 150.0)
    print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 80:
        init_long = interpolate_seq(best_state['seq'], n_long)
        a_out, _ = mwa_miter_optimize_advanced(init_long, min(t6, remaining * 0.4), 60, 1.0, rng, verbose=False)
        best_state = try_update(best_state, a_out, f"long_{n_long}", t_global)

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t_half = remaining * 0.5
        a_out, _ = mwa_miter_optimize_advanced(best_state['seq'], min(t_half, 80.0), 100, 0.7, rng, verbose=False)
        best_state = try_update(best_state, a_out, "final_mwa", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_state['seq'], min(remaining * 0.7, 60.0), rng, verbose=False)
            best_state = try_update(best_state, a_out, "final_pcd", t_global)

    # Final evaluation
    if best_state['seq'] is not None:
        final_score = eval_fn(best_state['seq'])
        if np.isfinite(final_score):
            best_state['score'] = final_score
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_state['seq']]
