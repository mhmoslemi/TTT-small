def fast_autocorr(a):
    """Compute full convolution conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

    try:

        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)

def compute_gradient(w, a):
    """
    Compute grad[j] = 2 * sum_m w[j+m] * a[m] for j=0,...,n-1.
    Uses FFT: grad[j] = 2 * conv(w, reverse(a))[j + n - 1]
    """
    n = len(a)
    w = np.asarray(w, dtype=np.float64)
    a_flip = a[::-1].copy()
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AF = rfft(a_flip, fft_size)
    conv_result = irfft(W * AF, fft_size)[:size]
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad

        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()

    a_prev = a.copy()
    momentum = 0.0
    # Schedule beta from beta_start to beta_end over the time limit
    n_total_est = int(time_limit * 100)  # estimate ~100 iters/sec

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        frac = min(1.0, it / max(1, n_total_est))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Nesterov extrapolation in linear space
        if it == 1:
            y = a.copy()
        else:
            momentum = (it - 1.0) / (it + 2.0)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

        # Compute objective and gradient
        b = fast_autocorr(y)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Softmax weights (with log-sum-exp for stability)
        b_shifted = b - max_b
        w = np.exp(beta * b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(b)) / len(b)
        else:
            w = w / w_sum

        # Compute gradient
        grad = compute_gradient(w, y)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        grad_normed = grad / grad_max

        # Mirror descent step with backtracking
        eta = eta_start / (1.0 + 0.1 * it)
        improved = False
        for _trial in range(10):
            a_cand = a * np.exp(-eta * grad_normed)
            a_cand = np.maximum(a_cand, 1e-30)
            s_cand = a_cand.sum()
            if s_cand < 1e-15:
                eta *= 0.5
                continue
            a_cand = a_cand / s_cand
            val_cand = fast_autocorr(a_cand).max()
            if val_cand < max_b + 1e-12:
                a = a_cand
                improved = True
                break
            eta *= 0.5

        if not improved:
            # Fallback: smaller step
            a_cand = a * np.exp(-0.01 * grad_normed)
            a_cand = np.maximum(a_cand, 1e-30)
            a_cand = a_cand / a_cand.sum()
            a = a_cand

        a_prev = a.copy()

        # Track best
        val = fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MSD iter={it} ({elapsed:.0f}s): val={best_val:.8f}, beta={beta:.1f}")

    return best_a, best_val


def two_timescale_game(a0, time_limit, eta_a=0.3, eta_q=0.01, seed=42, verbose=True):
    """
    Two-timescale method for the zero-sum game:
    min_a max_k conv(a,a)[k] s.t. sum a = 1, a >= 0
    Fast timescale: multiplicative weights on a (column player)
    Slow timescale: empirical distribution q on k (row player)
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    size = 2 * n - 1
    q = np.ones(size) / size  # distribution over k indices
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution
        b = fast_autocorr(a)
        max_b = b.max()

        # Update q (slow timescale): move towards argmax
        k_star = int(b.argmax())
        q = (1.0 - eta_q) * q + eta_q * np.zeros(size)
        q[k_star] += eta_q
        # Also add some spread to avoid getting stuck
        if it % 50 == 0:
            # Add small amount of uniform to prevent q from concentrating too much
            q = 0.95 * q + 0.05 / size
            q = q / q.sum()

        # Fast timescale: mirror descent step using q
        # grad[j] = 2 * sum_k q[k] * a[k-j] (valid range)
        grad = compute_gradient(q, a)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        # Adaptive step size
        eta = eta_a / (1.0 + 0.05 * it)
        a_new = a * np.exp(-eta * grad / grad_max)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Check improvement
        val_new = fast_autocorr(a_new).max()
        if val_new < max_b + 1e-12:
            a = a_new
        else:
            # Try smaller step
            a_new2 = a * np.exp(-0.01 * grad / grad_max)
            a_new2 = np.maximum(a_new2, 1e-30)
            a_new2 = a_new2 / a_new2.sum()
            a = a_new2

        # Track best
        val = fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    2TS iter={it} ({elapsed:.0f}s): val={best_val:.8f}")

    return best_a, best_val


def accelerated_mirror_descent(a0, time_limit, status_callback=None):
    """
    Nesterov-accelerated mirror descent with adaptive temperature.
    Uses a combination of sharp and diffuse softmax for better exploration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_old = a.copy()
    last_improve = t0

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        gamma = (it - 1.0) / (it + 2.0)
        y = a + gamma * (a - a_old)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        b = fast_autocorr(y)
        max_b = b.max()

        if max_b < best_val * 0.999:
            # y is already better, just project
            a_new = y
            a_old = a.copy()
            a = a_new
            best_val = max_b
            best_a = a.copy()
            last_improve = time.monotonic()
            continue

        # Mixed softmax: combine sharp peak-focusing with diffuse exploration
        b_shifted = b - max_b
        w_sharp = np.exp(30.0 * b_shifted)
        w_sharp = w_sharp / w_sharp.sum()
        w_flat = np.ones(len(b)) / len(b)
        w = 0.7 * w_sharp + 0.3 * w_flat

        grad = compute_gradient(w, y)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        grad_n = grad / grad_max

        # Backtracking line search
        eta = 0.3
        found = False
        for _ in range(8):
            a_cand = y * np.exp(-eta * grad_n)
            a_cand = np.maximum(a_cand, 1e-30)
            a_cand = a_cand / a_cand.sum()
            val_cand = fast_autocorr(a_cand).max()
            if val_cand <= max_b + 1e-10:
                a_old = a.copy()
                a = a_cand
                found = True
                break
            eta *= 0.5
        if not found:
            a_old = a.copy()
            # Tiny step
            a = a * np.exp(-0.005 * grad_n)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()

        val = fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 30:
            # Restart from best with perturbation
            a = best_a.copy()
            y = a.copy()
            a_old = a.copy()
            noise = np.random.default_rng(it).normal(0, 0.01, n)
            a = np.maximum(a + noise * a.mean(), 1e-10)
            a = a / a.sum()
            last_improve = time.monotonic()

    return best_a, best_val


def lse_objective_fn(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shifted = T_temp * (b - max_b)
    log_sum = max_b + np.log(np.exp(shifted).sum()) / T_temp
    return 2.0 * n * log_sum / (s * s)


def lse_grad_fn(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T_temp * (b - max_b)
    w = np.exp(shifted)
    w = w / w.sum()
    lse_val = max_b + np.log(w.sum()) / T_temp  # = max_b + log(sum(exp))/T

    # grad of LSE wrt a is: 2 * sum_k w[k] * a[k-j] (same structure)
    grad = compute_gradient(w, a)
    obj_val = 2.0 * n * lse_val / (s * s)
    result = 2.0 * n * grad / (s * s) - 2.0 * obj_val * a / s
    return result


def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        x0 = a.copy()

        def obj_func(x):
            return lse_objective_fn(x, n, T)

        def grad_func(x):
            return lse_grad_fn(x, n, T)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand / s_c
                    score_c = fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        improved = False

        for k_peak in top_peaks:
            # Find indices contributing to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:5]]

            # Build peak mask (indices involved in top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2 + 1] = True

            for src in top_contrib:
                if a[src] < 1e-14:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by smallest value (move mass to "cold" spots)
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]

                for dst in candidates_sorted[:8]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_val - 1e-14:
                            a = a_trial
                            best_a = a_trial.copy()
                            best_val = score
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random moves
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-14:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_val - 1e-14:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_val = score
                        stall = 0
                        break

            if stall > 500:
                break

    return best_a, best_val


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(5 * np.pi * x)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        a = 1.0 + 0.3 * np.sin(3.0 * np.pi * x) + 0.1 * np.sin(7.0 * np.pi * x)
    elif choice == 7:
        a = 1.0 + 0.4 * (2.0 * x - 1.0) ** 2
    elif choice == 8:
        a = np.exp(x * 2.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with multiple phases."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    n_base = len(best_seq) if best_seq is not None else 1600

    # Phase 1: Accelerated mirror descent at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Accelerated MSD n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.30
    if t1 > 30:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_base, rng)
        a_init = a_init / a_init.sum()
        a_out, _ = accelerated_mirror_descent(a_init, t1)
        do_try_update(a_out, "amd_n"+str(n_base))

    # Phase 2: Softmax mirror descent with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 2: Softmax MSD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.20
    if t2 > 20:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_best, rng)
        a_init = a_init / a_init.sum()
        a_out, _ = softmax_mirror_descent(a_init, t2, beta_start=80.0, beta_end=5.0,
                                           eta_start=0.4, seed=123)
        do_try_update(a_out, "msd_n"+str(n_best))

    # Phase 3: Two-timescale game
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 3: Two-timescale n={n_best} (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 20:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_best, rng)
        a_init = a_init / a_init.sum()
        a_out, _ = two_timescale_game(a_init, t3, eta_a=0.2, eta_q=0.005, seed=456)
        do_try_update(a_out, "2ts_n"+str(n_best))

    # Phase 4: LSE + scipy polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 4: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 15:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_best, rng)
        a_init = a_init / a_init.sum()
        a_out, _ = lse_scipy_optimize(a_init, t4, T_start=60.0, T_end=1.5, n_stages=8)
        do_try_update(a_out, "lse_n"+str(n_best))

    # Phase 5: Try longer length
    n_long = n_best + 300
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MSD n={n_long} (budget={remaining:.0f}s) ===")
    t5 = remaining * 0.12
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n_long) if best_seq is not None else structured_init(n_long, rng)
        a_out, _ = accelerated_mirror_descent(init_long, t5)
        do_try_update(a_out, "amd_long_n"+str(n_long))

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 6: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.15
    if t6 > 15:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_best, rng)
        a_init = a_init / a_init.sum()
        a_out, _ = coordinate_descent_polish(a_init, t6, seed=99999)
        do_try_update(a_out, "cd_n"+str(n_best))

    # Phase 7: Final accelerated MSD from best (fresh momentum)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 7: Final AMD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.35
    if t7 > 10:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_best, rng)
        # Add small noise to escape local optima
        noise = rng.normal(0, 0.005, n_best)
        a_init = np.maximum(a_init + noise * a_init.mean(), 1e-10)
        a_init = a_init / a_init.sum()
        a_out, _ = accelerated_mirror_descent(a_init, t7)
        do_try_update(a_out, "amd_final_n"+str(n_best))

    # Phase 8: Quick coord descent on final best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 8: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t8 = remaining * 0.5
    if t8 > 10:
        a_init = best_seq.copy() if best_seq is not None else structured_init(n_best, rng)
        a_init = a_init / a_init.sum()
        a_out, _ = coordinate_descent_polish(a_init, t8, seed=77777)
        do_try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
