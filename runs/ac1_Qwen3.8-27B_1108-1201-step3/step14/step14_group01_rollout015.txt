def next_pow2(m):
    """Return the next power of 2 >= m."""
    return 1 << (m - 1).bit_length()

def fft_conv(a, b=None):
    """Compute linear convolution via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    n2 = len(b)
    size = n + n2 - 1
    N = next_pow2(size)
    A = rfft(a, N)
    B = rfft(b, N)
    return irfft(A * B, N)[:size]

def normalize(a, target_sum):
    """Normalize a to have the given sum, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a * target_sum / s

def compute_max_conv(a):
    """Compute max of self-convolution."""
    return fft_conv(a).max()

def lp_solve(a, M, solver, active=None, b_prev=None):
    """Solve the LP: max sum(b) s.t. (a*b)(k) <= M for k in active, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    if active is not None:
        rows = active
    else:
        rows = np.arange(size)
    
    T = np.zeros((len(rows), n))
    for m_idx, m in enumerate(rows):
        # T[m_idx, j] = a[m - j] if 0 <= m-j < n else a[j-m] if 0 <= j-m < n else 0
        # Actually T[m, j] is the coefficient of b[j] in (a*b)[m]
        # (a*b)[m] = sum_j a[m-j] * b[j] (when 0 <= m-j < n and 0 <= j < n)
        # So T[m, j] = a[m-j] if 0 <= m-j < n else 0
        j_start = max(0, m - n + 1)
        j_end = min(n, m + 1)
        if j_end > j_start:
            T[m_idx, j_start:j_end] = a[m - (j_start:j_end)][::-1]
    
    b_var = cp.Variable(n, nonneg=True)
    if b_prev is not None:
        b_var.value = b_prev
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Try to get dual variables
    y_dual = None
    try:
        y_dual = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return val, y_dual

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent to minimize max(conv)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_max = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    N = next_pow2(size)
    
    # Geometric temperature schedule
    temps = np.geomspace(30.0, 0.5, 15)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(300, int(remaining * 4)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fft_conv(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE gradient
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k]) / max_c
            # = 2 * (cross_corr(w, a)[i] + conv(w, a)[i]) / max_c ... 
            # Actually: d/max(conv) of max(conv) ≈ 2 * sum_k w_k * (a[k-i]+a[i-k])
            # where w_k = exp((conv_k - max)/T) / sum(exp(...))
            
            W = rfft(w, N)
            A = rfft(a, N)
            
            # First term: sum_k w[k] * a[k-i] for i in [0, n-1]
            # = (w * a_flipped)[i] ... 
            # Let's use the identity: sum_k w[k] * a[k-i] = (a * w_flip)[i] where w_flip[k] = w[size-1-k]
            # Or more directly: ifft(conj(FFT(a)) * FFT(w))[i] gives sum_k a[k] * w[k+i] (circular)
            # We want sum_k w[k] * a[k-i]. Let j = k-i: sum_j w[j+i] * a[j] = sum_j a[j] * w[j+i]
            # = ifft(conj(FFT(a)) * FFT(w))[i] in circular convolution
            # Since N >= 2n-1, this is the same as linear for i in [0, n-1]
            ca = irfft(np.conj(A) * W, N)
            g1 = ca[:n]  # sum_j a[j] * w[j+i] = sum_k w[k] * a[k-i]
            
            # Second term: sum_k w[k] * a[i-k]
            # Let j = i-k: sum_j w[i-j] * a[j] = (a * w)[i] (linear convolution)
            aw = irfft(A * W, N)
            g2 = aw[:n]  # sum_j a[j] * w[i-j] = sum_k w[k] * a[i-k]
            
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            
            if grad_norm < 1e-15:
                break
            
            # Backtracking line search
            step = norm / (T_val * np.sqrt(n))
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                step = 0.0
            else:
                a_trial = a_trial * norm / s_trial
                sc = compute_max_conv(a_trial)
                trial = 0
                while sc > max_c + 0.001 * T_val and trial < 15:
                    step *= 0.5
                    a_trial = a - step * grad
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        break
                    a_trial = a_trial * norm / s_trial
                    sc = compute_max_conv(a_trial)
                    trial += 1
                a = a_trial
                sc = compute_max_conv(a)
                if sc < best_max:
                    best_max = sc
                    best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_max:.8f}")
    
    return best_a, best_max
