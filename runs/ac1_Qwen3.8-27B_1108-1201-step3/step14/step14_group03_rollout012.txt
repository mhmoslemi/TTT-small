"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax gradient descent with momentum (primary, 40% time)
2. LP best-response with Nesterov acceleration (30% time)
3. Window-based pair coordinate descent (20% time)
4. Random restart with local search (10% time)
Key improvements over previous: softmax GD provides smooth gradients
that can escape local minima differently than LP or coordinate descent.
"""

import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

        try:

    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def softmax_gradient_descent(a0, time_limit, solver=None, verbose=True):
    """Softmax gradient descent with momentum for minimizing max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    # FFT size for gradient computation
    size = 2 * n - 1
    grad_fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Parameters
    T_start = 5.0
    T_end = 50.0
    momentum = 0.9
    base_lr = 0.5  # will be scaled by norm/n
    n_warmup = 50
    
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Estimate total iterations
        elapsed = time.monotonic() - t0
        # Assume ~1ms per iteration for n=1600
        est_time_per_iter = max(elapsed / max(it, 1), 0.0005) if it > 0 else 0.0005
        remaining = time_limit - elapsed
        est_total = it + int(remaining / est_time_per_iter)
        
        # Anneal T
        progress = min(it / max(est_total, 1), 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute autocorrelation
        conv = fast_autocorr(a)
        max_conv = conv.max()
        
        # Compute softmax weights
        shifted = conv - max_conv
        # Clip to avoid overflow
        shifted = np.clip(shifted, -100.0 / T, 0.0)
        exp_vals = np.exp(T * shifted)
        p_sum = exp_vals.sum()
        if p_sum < 1e-300:
            break
        p = exp_vals / p_sum
        
        # Compute gradient via FFT
        P = rfft(p, grad_fft_size)
        A = rfft(a, grad_fft_size)
        g1 = irfft(P * np.conj(A), grad_fft_size)[:n]
        g2 = irfft(P * A, grad_fft_size)[:n]
        grad = g1 + g2
        
        # Learning rate (decays over time)
        lr = base_lr * (norm / n) / (1.0 + 0.01 * it)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s
        a = a_new
        
        it += 1
        
        # Check score periodically
        if it % 5 == 0 or it <= 10:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = fast_autocorr(a).max()
            print(f"    SGD iter={it} ({elapsed:.0f}s): T={T:.1f}, best={best_score:.8f}, cur={score:.8f}")
    
    # Final check
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum_coeff = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum_coeff * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_window(a, time_limit, rng, window=40, verbose=True):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Compute delta in conv from moving mass from i to j
                    delta_vec = np.zeros(size)
                    # Removing mass from i
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        src = np.arange(lo1 - i, hi1 - i)
                        delta_vec[lo1:hi1] -= a[src]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        src2 = n - 1 - np.arange(lo2, hi2)
                        valid = (src2 >= 0) & (src2 < n)
                        delta_vec[lo2:hi2][valid] -= a[src2[valid]]
                    # Adding mass to j
                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        src3 = np.arange(lo3 - j, hi3 - j)
                        delta_vec[lo3:hi3] += a[src3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        src4 = n - 1 - np.arange(lo4, hi4)
                        valid4 = (src4 >= 0) & (src4 < n)
                        delta_vec[lo4:hi4][valid4] += a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 10
                    deltas = np.linspace(0.0, a[i] * 0.99, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def single_cd(a, time_limit, rng, verbose=True):
    """Single coordinate descent: adjust individual elements."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            
            # Compute the "effect" of changing a[i]
            # conv[k] depends on a[i] when: k-i in [0,n-1] or i-k in [0,n-1]
            # i.e., k in [i, i+n-1] and k in [i-(n-1), i]
            # Conv contribution from a[i]:
            # delta[k] = a[k-i] + a[i-k] for valid indices
            delta = np.zeros(size)
            # Forward: k = i, i+1, ..., i+n-1 (clipped to [0, size-1])
            lo = max(0, i)
            hi = min(size, i + n)
            if hi > lo:
                indices = np.arange(lo - i, hi - i)
                delta[lo:hi] += a[indices]
            # Backward: k = i-(n-1), ..., i (clipped to [0, size-1])
            lo2 = max(0, i - (n - 1))
            hi2 = min(size, i + 1)
            if hi2 > lo2:
                indices2 = np.arange(lo2, hi2)
                delta[lo2:hi2] += a[i - indices2]
            
            # Remove double-counting at k = i (where a[i]*a[i] appears)
            # Actually, at k=i: delta[i] = a[0] + a[0] = 2*a[0]... no
            # delta[i] = a[i-i] + a[i-i] = a[0] + a[0] = 2*a[0]
            # But the actual contribution of a[i] to conv[i] includes a[i]*a[i] = a[i]^2
            # plus a[i]*a[0] (from i and k-i=0) ... 
            # Wait, I think the delta computation is correct for the linear part.
            # The quadratic correction at k=2i (self-term) needs separate handling.
            
            # We want to find the best a[i]' = a[i] + t that minimizes max(conv + t*delta + t^2 * correction)
            # The correction is at position 2i: conv[2i] gets a[i]^2, and after change it gets (a[i]+t)^2
            # So correction at 2i: 2*a[i]*t + t^2
            # But this is already in delta if we include it... let me re-examine.
            
            # Actually, the issue is that conv[k] = sum_m a[m]*a[k-m]
            # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with a[x]=0 for invalid x)
            # But when k-i = i (i.e., k=2i), the term a[i]*a[i] appears only once in the sum,
            # and its derivative is 2*a[i], not a[i] + a[i] = 2*a[i]. So it's the same!
            # The linear approximation is: conv[k] -> conv[k] + t * delta[k]
            # But the exact change is: conv[k] -> conv[k] + t * delta[k] + t^2 * (1 if k==2i else 0)
            
            # For the optimization, we can include the quadratic term
            # new_conv[k] = conv[k] + t * delta[k] + t^2 * (k == 2i)
            
            best_t = 0.0
            best_val = conv.max()
            
            # Try different t values
            # t can be in [min(0, -a[i]), a[i] * 5] (allow increasing or decreasing)
            t_min = -a[i] * 0.99
            t_max = a[i] * 2.0
            n_samples = 20
            ts = np.linspace(t_min, t_max, n_samples)
            
            for t in ts:
                if t == 0:
                    continue
                new_conv = conv + t * delta
                if 2 * i < size:
                    new_conv[2 * i] += t * t
                new_max = new_conv.max()
                if new_max < best_val - 1e-13:
                    best_val = new_max
                    best_t = t
            
            if best_t != 0:
                a[i] += best_t
                if a[i] < 0:
                    a[i] = 0
                s = a.sum()
                if s < 1e-15:
                    a[i] -= best_t
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    a[i] -= best_t
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def random_restart_local_search(a0, time_limit, rng, solver, verbose=True):
    """Add random perturbations and do local optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_restarts = 0
    time_per_restart = min(30.0, time_limit / 5)
    
    while time.monotonic() - t0 < time_limit:
        n_restarts += 1
        # Add random noise
        noise_scale = 0.1 / np.sqrt(n)
        noise = rng.normal(0, noise_scale, n)
        a_perturbed = a + noise
        a_perturbed = np.maximum(a_perturbed, 0)
        s = a_perturbed.sum()
        if s < 1e-15:
            continue
        a_perturbed = a_perturbed * norm / s
        
        # Do a few iterations of softmax GD on the perturbed sequence
        remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_restart, remaining * 0.3)
        if t_sub > 2:
            a_out, score_out = softmax_gradient_descent(a_perturbed, t_sub, verbose=False)
            if score_out < best_score:
                best_score = score_out
                best_a = a_out.copy()
                a = a_out.copy()  # Update parent
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    Restart {n_restarts} ({elapsed:.0f}s): new best={best_score:.8f}")
    
    return best_a, best_score
