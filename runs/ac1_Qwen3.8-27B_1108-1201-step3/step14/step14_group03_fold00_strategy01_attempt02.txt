        sequence: A list of non-negative floats.

    Returns:
        The objective value (lower is better), or inf if the sequence is invalid.
    """
    # Security checks
    if not isinstance(sequence, list):
        return np.inf
    
    if len(sequence) == 0:
        return np.inf
        
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    
    # Convert all elements to floats and ensure non-negativity via clipping (since max(0,x) is done in evaluate_safe)
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    
    n = len(sequence)
    # Compute the convolution (full mode)
    conv_vals = np.convolve(sequence, sequence, mode='full')
    max_conv = np.max(conv_vals)
    total_sum = np.sum(sequence)
    
    # Return inf if normalized sum was below threshold
    if total_sum < 0.01:
        return np.inf
        
    # Return the scaled objective value
    return float(2.0 * n * max_conv / (total_sum ** 2))
