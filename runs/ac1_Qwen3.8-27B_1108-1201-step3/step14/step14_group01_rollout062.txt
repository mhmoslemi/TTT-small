def build_T_rows(a, active_indices):
    """Build constraint rows for given active indices."""
    n = len(a)
    rows = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        # conv(a,b)[k] = sum_j b[j] * a[k-j] for valid j
        # b[j] coefficient is a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            rows[i, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    return rows

def lp_best_response(a, M, solver, active=None, max_iters=5):

    n = len(a)
    size = 2 * n - 1
    
    if active is None:
        active = np.arange(size)
    
    active = np.asarray(active)

        T = build_T_rows(a, active)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None

            return None

        return b

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov momentum with adaptive alpha
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            break
        y_new = y_new * norm / s_y
        
        if it == 1:
            a_new = y_new.copy()
        else:
            momentum = (it - 1.0) / (it + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_avg_score:
                    best_avg_score = sc_avg
                    best_avg_a = avg_check.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov it={it} ({elapsed:.0f}s): best={best_score:.8f}, avg={best_avg_score:.8f}")
    
    if best_avg_score < best_score:
        best_score = best_avg_score
        best_a = best_avg_a
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with informed neighbor selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            conv = fast_conv_self(a)
            cur_max = conv.max()
            
            j_candidates = list(set(int(x) for x in rng.integers(n, size=12)))
            peak_thresh = cur_max * 0.92
            peak_lags = np.where(conv >= peak_thresh)[0]
            for kl in peak_lags[:4]:
                j_s = (int(kl) - i) % n
                if j_s != i and j_s not in j_candidates:
                    j_candidates.append(j_s)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i or j >= n:
                    continue
                aj = a[j]
                
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi-lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3-lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                n_samples = 25
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                
                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve >= 5:
                n_perturb = max(10, n // 30)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and total_improvements > 0 and total_improvements % 300 == 0:
            print(f"    PCD imp={total_improvements} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def smooth_init(n, rng):
    """Initialize with smooth bump shape."""
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = np.exp(-8.0 * (x - 0.5) ** 2) + 0.15
    elif choice == 3:
        a = 1.0 - (2.0 * x - 1.0) ** 6
    elif choice == 4:
        a = np.sin(np.pi * x) ** 0.7
    else:
        a = np.exp(-4.0 * (x - 0.35) ** 2) + 0.9 * np.exp(-4.0 * (x - 0.65) ** 2) + 0.08
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1730.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        best_seq = smooth_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 1: LSE (fast initial improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE GD n={len(best_seq)} ({t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_subgradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse_n{len(best_seq)}")
    
    # Phase 2: LP Nesterov (main workhorse - allocate most time here)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{len(best_seq)}")
    
    # Phase 3: Active-set subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 3: Active-set polish ({t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = active_set_subgradient_polish(best_seq, t3, verbose=True)
        try_update(a_out, f"asp_n{len(best_seq)}")
    
    # Phase 4: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} ({t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")
    
    # Phase 5: LP round 2 (further refinement)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 5: LP round 2 ({t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")
    
    # Phase 6: Active-set + LSE final
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.30, 150.0)
    print(f"\n=== Phase 6: Final polish ({t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = active_set_subgradient_polish(best_seq, min(t6 * 0.4, 50.0), verbose=False)
        try_update(a_out, "asp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = lse_subgradient_descent(best_seq, min(remaining * 0.5, 40.0), verbose=False)
            try_update(a_out2, "lse_final")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_out3, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 30.0), rng, verbose=False)
                try_update(a_out3, "pcd_final")
    
    # Phase 7: Try longer length if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_long = len(best_seq) + 400
        t7 = min(remaining * 0.5, 150.0)
        print(f"\n=== Phase 7: Long n={n_long} ({t7:.0f}s) ===")
        if t7 > 30:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, t7 * 0.7, solver, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
