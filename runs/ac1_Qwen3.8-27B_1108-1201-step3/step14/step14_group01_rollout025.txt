def fast_conv_self(a):

    A = rfft(a, fft_size)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    a = a * norm / s

    t0 = time.monotonic()

    n_stages = 8
    temps = np.geomspace(30.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, int(remaining * 8))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)

            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = a.copy()
            else:
                a = a * norm / s_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

def gs_ripple(a0, n_iter, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

            break

                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 25
                deltas = np.linspace(0.01, ai * 0.95, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 60)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        centers = rng.uniform(0.1, 0.9, 3)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-30.0 * (x - c) ** 2)
        a += 0.3
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.ones(n) * rng.uniform(0.3, 1.0, n)
    else:
        a = np.sin(np.pi * x) * np.cos(0.5 * np.pi * x) ** 0.5
        a = np.maximum(a, 0.0) + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try loading parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Try multiple initializations at different lengths
    n_values = [1600, 2000, 2400]
    init_results = []

    if best_seq is not None:
        n_base = len(best_seq)
        if n_base not in n_values:
            n_values.insert(0, n_base)
    else:
        n_base = 1600

    # Phase 1: Fast initialization
    print(f"\n=== Phase 1: Initialization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.10, 120.0)
    t_per_init = t_init / max(1, len(n_values))

    for n_try in n_values[:4]:
        if time.monotonic() - t_global > time_budget:
            break
        t_sub = min(t_per_init, time_budget - (time.monotonic() - t_global))
        if t_sub < 10:
            break
        a_init = structured_init(n_try, rng)
        # Quick GS
        a_gs, _ = gs_ripple(a_init, int(t_sub * 3), t_sub * 0.4, verbose=False)
        # Quick LSE
        a_lse, _ = lse_gradient_descent(a_gs, t_sub * 0.5, verbose=False)
        init_results.append((a_lse, n_try))
        try_update(a_lse, f"init_n{n_try}")

    if best_seq is None:
        best_seq = init_results[0][0]
        best_score = eval_fn(best_seq)

    # Phase 2: LP-Nesterov on best initialization
    print(f"\n=== Phase 2: LP-Nesterov ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.45, 700.0)
    print(f"  n={n2}, budget={t2:.0f}s")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, verbose=True)
        try_update(a_out, f"lp1_n{n2}")

    # Phase 3: CD polish
    print(f"\n=== Phase 3: Pair CD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    print(f"  budget={t3:.0f}s")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"cd1_n{len(best_seq)}")

    # Phase 4: Another LP round
    print(f"\n=== Phase 4: LP-Nesterov round 2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 350.0)
    print(f"  n={len(best_seq)}, budget={t4:.0f}s")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Try a larger n by interpolation
    print(f"\n=== Phase 5: Multi-scale ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_long = len(best_seq) + 400
        t5 = min(remaining * 0.4, 200.0)
        print(f"  Interpolating to n={n_long}, budget={t5:.0f}s")
        init_long = interpolate_seq(best_seq, n_long)
        # Quick init at long length
        a_gs_long, _ = gs_ripple(init_long, 200, min(t5 * 0.3, 30), verbose=False)
        a_lse_long, _ = lse_gradient_descent(a_gs_long, min(t5 * 0.3, 30), verbose=False)
        # LP on long
        t5_lp = t5 * 0.4
        if t5_lp > 20:
            a_out_long, _ = nesterov_lp_optimize(a_lse_long, t5_lp, verbose=False)
            try_update(a_out_long, f"lp_long_{n_long}")
        # CD on long
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out_long2, _ = pair_cd_optimize(best_seq, min(remaining * 0.3, 60), rng, verbose=False)
            try_update(a_out_long2, f"cd_long_{n_long}")

    # Phase 6: Final polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6 = min(remaining * 0.6, 100.0)
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        try_update(a_out, f"cd_final")

    # Phase 7: One more LP if time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.7, 80.0)
        a_out, _ = nesterov_lp_optimize(best_seq, t7, verbose=False)
        try_update(a_out, f"lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 1600
        return [float(x) for x in structured_init(n_fb, rng)]

    return [float(x) for x in best_seq]
