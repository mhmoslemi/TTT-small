                break

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Compute d_k = change in autocorrelation per unit delta
            d = np.zeros(size)
            # +delta at position j
            lo1 = max(0, j)
            hi1 = min(size, j + n)
            if hi1 > lo1:
                d[lo1:hi1] += a[lo1 - j:hi1 - j]
            lo2 = max(0, n - 1 - j)
            hi2 = min(size, n + j)
            if hi2 > lo2:
                src = n - 1 - np.arange(lo2, hi2)
                valid = (src >= 0) & (src < n)
                if np.any(valid):
                    d[lo2:hi2] += a[src[valid]] * valid
            # -delta at position i
            lo3 = max(0, i)
            hi3 = min(size, i + n)
            if hi3 > lo3:
                d[lo3:hi3] -= a[lo3 - i:hi3 - i]
            lo4 = max(0, n - 1 - i)
            hi4 = min(size, n + i)
            if hi4 > lo4:
                src = n - 1 - np.arange(lo4, hi4)
                valid = (src >= 0) & (src < n)
                if np.any(valid):
                    d[lo4:hi4] -= a[src[valid]] * valid

            # s_k (second order): nonzero at 2i, 2j, i+j
            s = np.zeros(size)
            if 2 * i < size:
                s[2 * i] += 1.0
            if 2 * j < size:
                s[2 * j] += 1.0
            ij = i + j
            if ij < size:
                s[ij] -= 2.0

            # Evaluate at delta grid
            n_deltas = 25
            deltas = np.linspace(0, ai, n_deltas + 1)[1:]
            # Vectorized: shape (n_deltas, size)
            vals = conv[None, :] + deltas[:, None] * d[None, :] + (deltas[:, None] ** 2) * s[None, :]
            max_vals = vals.max(axis=1)
            best_t = int(np.argmin(max_vals))
            best_d = deltas[best_t]
            best_val = max_vals[best_t]

            if best_d > 0 and best_val < best_score * (1 - 1e-12):
                a[i] -= best_d
                a[j] += best_d
                s_sum = a.sum()
                if s_sum < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue
                a = a * norm / s_sum
                conv = _fast_autocorr(a)
                cur_score = conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
                    conv = _fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def _lp_best_response(a, M, solver, active_set=None):
    """Solve LP: maximize sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0 and len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                ii = k - j
                if 0 <= ii < n:
                    T[idx, j] = a[ii]
        constraints = [T @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j+n, j] = a
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def _nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _normalize(a0, norm)

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_prev = a.copy()
    t_nest = 1.0
    stall = 0
    active_set = None
    n_iters = 0

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = _fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break

        if active_set is None or it % 80 == 0:
            tol = 0.03 + 0.01 * np.exp(-it / 300.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 5:
                active_set = np.where(conv_y > M_curr * 0.75)[0]

        b = _lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            if it % 30 == 0:
                active_set = None
                b = _lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        if it == 1:
            x_new = g
        else:
            momentum = (t_nest - 1.0) / (t_nest + 1.5)
            y_curr = y + momentum * (y - x_prev)
            # LP at y_curr for better quality
            conv_yc = _fast_autocorr(np.maximum(y_curr, 0))
            M_yc = conv_yc.max()
            b2 = _lp_best_response(np.maximum(y_curr, 0), M_yc, solver, active_set)
            if b2 is not None and b2.sum() > 1e-15:
                x_new = b2 * norm / b2.sum()
            else:
                x_new = g

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_nest**2)) / 2.0
        t_nest = t_next

        x_prev = y
        y = x_new

        score = _fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 80:
            y = best_a.copy()
            x_prev = best_a.copy()
            t_nest = 1.0
            active_set = None
            stall = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def _perturb(a, rng, scale=0.03):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    a_p = a * noise
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s

def _interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, a / s)
    interp = np.maximum(interp, 0.0)
    ss = interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return interp / ss

def _mirror_descent(a0, time_limit, verbose=True, seed=0):
    """Mirror descent (KL-based) on the simplex for smoothed max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _normalize(a0, norm)

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    log_a = np.log(a + 1e-30)
    eta = 1.0 / (4.0 * norm)
    n_iters = 0
    stall = 0

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        a_curr = np.exp(log_a)
        a_curr = a_curr * norm / a_curr.sum()

        conv = _fast_autocorr(a_curr)
        max_val = conv.max()

        mu = max(norm * 1e-5, 0.05 * max_val / n)
        shifted = conv / mu - (conv / mu).max()
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros_like(conv)
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum

        # Gradient of smoothed max
        L_f = 1 << (3 * n - 2 - 1).bit_length()
        Y = rfft(a_curr, L_f)
        P = rfft(p, L_f)
        Yr = rfft(a_curr[::-1], L_f)
        g2 = irfft(Y * P, L_f)[:n]
        g1 = irfft(Yr * P, L_f)[n-1:n-1+n]
        grad = g1 + g2

        # Mirror descent update in log space
        log_a = log_a - eta * grad / a_curr
        # Clip for numerical stability
        log_a = np.clip(log_a, -50, 50)

        a_curr = np.exp(log_a)
        s = a_curr.sum()
        if s < 1e-15:
            a_curr = np.ones(n) * norm / n
            log_a = np.log(a_curr / norm)
        else:
            a_curr = a_curr * norm / s

        score = _fast_autocorr(a_curr).max()
        n_iters += 1

        if score < best_score:
            best_score = score
            best_a = a_curr.copy()
            stall = 0
        else:
            stall += 1

        # Decay eta
        eta *= 0.9999
        eta = max(eta, 1e-8)

        if stall > 300:
            log_a = np.log(best_a + 1e-30)
            a_curr = best_a.copy()
            stall = 0
            eta = 1.0 / (4.0 * norm)

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score
