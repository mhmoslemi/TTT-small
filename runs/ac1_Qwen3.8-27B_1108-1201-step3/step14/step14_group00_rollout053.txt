import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific k indices."""
    n = len(a)
    size = 2 * n - 1
    m = len(k_indices)
    T = np.zeros((m, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            r = k - j
            if 0 <= r < n:
                T[idx, j] = a[r]
    return T

**Flattening via multiplicative updates**:

The idea is:
- We want (a*a)_k to be as equal as possible (for k where it's near the max)
- For each position i in a, it contributes to (a*a)_k for all k, with weight a_{k-i}
- If (a*a)_k is too high at some k, we want to decrease the a_i that contribute most to it
- Multiplicative update: a_i *= (1 - alpha * contribution_i) where contribution_i measures how much a_i contributes to the peak

- Compute the "gradient" of the max: for k* = argmax(a*a), grad_i = 2*a_{k*-i}
- Do a multiplicative step: a_i *= exp(-alpha * grad_i / max_grad)
- Renormalize

def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0, lp_interval=5):
    """Nesterov-accelerated LP best-response with periodic LP re-solves."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize_seq(a0, 1.0)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    x = a.copy()
    avg_x = np.zeros(n)
    n_avg = 0
    g = a.copy()  # current LP direction
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    stall_thresh = 30

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Periodically re-solve LP
        if t == 1 or (t - 1) % lp_interval == 0:
            conv_y = fast_autocorr(y)
            M_curr = conv_y.max()
            if M_curr < 1e-15:
                break
            b = lp_best_response_lazy(y, M_curr, solver)
            if b is not None and b.sum() > 1e-10:
                g = normalize_seq(b, 1.0)

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_seq(y_new, 1.0)

        if t == 1:
            x_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            x_new = normalize_seq(x_new, 1.0)

        y = y_new
        x = x_new
        n_iters += 1

        # Track best
        sc = fast_autocorr(x).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Ergodic average
        avg_x += x
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = normalize_seq(avg_x / n_avg, 1.0)
            sc_avg = fast_autocorr(avg_check).max()
            if sc_avg < best_score - 1e-12:
                best_score = sc_avg
                best_a = avg_check.copy()
                stall = 0

        # Restart on stall
        if stall > stall_thresh:
            y = best_a.copy()
            x = best_a.copy()
            avg_x = np.zeros(n)
            n_avg = 0
            stall = 0
            if verbose:
                print(f"    NLP restart at iter {t} ({time.monotonic()-t0:.0f}s), best={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def multiplicative_flattening(a0, time_limit, rng, n_steps=2000, verbose=False):

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        if M < 1e-15:
            break

        # Find active set (near-maximal autocorrelation lags)
        active_thresh = M * (1.0 - 0.05)
        active_mask = conv >= active_thresh
        n_active = max(1, np.sum(active_mask))

        # Compute pressure: for each i, how much does it contribute to active lags?
        # pressure[i] = sum_{k in active} a_{k-i} (with bounds)
        # This is a convolution of a with the active mask indicator
        active_indicator = active_mask.astype(np.float64)
        pressure = convolve_full_fft(a, active_indicator)
        pressure = pressure / n_active  # average pressure per active lag

        # Multiplicative update: reduce high-pressure positions
        # Use exponential with adaptive rate
        max_p = pressure.max()
        if max_p < 1e-15:
            break
        rate = 0.1 / max_p
        a_new = a * np.exp(-rate * pressure)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, 1.0)

        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve the rate on failure
            rate = rate * 0.5
            a_new = a * np.exp(-rate * pressure)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, 1.0)
            sc = fast_autocorr(a_new).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # No improvement, try smaller perturbation
                break

        if verbose and step % 200 == 0:
            print(f"    FLAT step={step} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=False):
    """Pair swap coordinate descent with efficient evaluation."""
    n = len(a0)
    a = normalize_seq(a0, 1.0)
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue

            # Try a few random partners
            for _attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute the change vector: moving delta from i to j
                # delta_conv[k] = a_{k-j} - a_{k-i} (approximately, ignoring delta^2 terms)
                delta_conv = np.zeros(size)
                lo_j = max(0, j)
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    delta_conv[lo_j:hi_j] += a[np.arange(0, hi_j - lo_j)]
                lo_i = max(0, i)
                hi_i = min(size, i + n)
                if hi_i > lo_i:
                    delta_conv[lo_i:hi_i] -= a[np.arange(0, hi_i - lo_i)]

                # Also account for the reverse (j-i and i-j shifts)
                # Actually the full conv change is more complex. Let's just evaluate directly.
                # Try a few candidate deltas
                candidates = [ai * 0.1, ai * 0.3, ai * 0.5, ai * 0.7]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    a[i] -= d
                    a[j] += d
                    a = normalize_seq(a, 1.0)
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d
                        best_conv = new_conv.copy()
                        best_a_tmp = a.copy()

                if best_d > 0:
                    a = best_a_tmp.copy()
                    conv = best_conv.copy()
                    if best_val < best_score - 1e-13:
                        best_score = best_val
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                else:
                    # Restore
                    a = normalize_seq(a, 1.0)
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        if verbose and pass_num % 5 == 0:
            print(f"    PCD pass={pass_num} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def stochastic_subgradient(a0, time_limit, rng, verbose=False, n_steps=50000):
    """Stochastic subgradient method on the max autocorrelation."""
    n = len(a0)
    a = normalize_seq(a0, 1.0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    lr0 = 0.01
    for step in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        # Randomly pick from top lags
        top_k = int(min(20, 2 * n - 1))
        top_indices = np.argsort(-conv)[:top_k]
        k_star = int(top_indices[int(rng.integers(top_k))])

        # Subgradient of (a*a)_{k_star}
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        js = np.arange(lo, hi)
        rs = k_star - js
        grad[js] = 2.0 * a[rs]

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            continue

        # Learning rate
        lr = lr0 / np.sqrt(step + 1)

        # Gradient step
        a_new = a - lr * grad / g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, 1.0)

        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if verbose and step % 10000 == 0:
            print(f"    SGD step={step} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, type_idx=0):
    """Generate a structured initial sequence."""
    choice = type_idx % 10
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(6 * np.pi * x))
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1 + np.sin(3 * np.pi * x)) * (1 + 0.3 * np.cos(7 * np.pi * x))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.2)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def perturb_structural(a, rng, scale=0.05):
    """Add structured perturbation (sinusoid or bump)."""
    n = len(a)
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 4)
    if choice == 0:
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1 + scale * np.sin(2 * np.pi * freq * x + phase))
    elif choice == 1:
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.2)
        a_new = a * (1 + scale * np.exp(-((x - center) / width) ** 2))
    elif choice == 2:
        freq = rng.uniform(2, 8)
        a_new = a * (1 + scale * np.cos(2 * np.pi * freq * x))
    else:
        a_new = a * (1 + scale * rng.standard_normal(n) / np.sqrt(n) * 3)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate():
    """Main search function: find best sequence minimizing the evaluation objective."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1740.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize from parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget {t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, seed=0, lp_interval=4)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Multiplicative flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 2: Flattening n={n_cur} (budget {t2:.0f}s) ===")
    if t2 > 10:
        a_out, _ = multiplicative_flattening(best_seq, t2, rng, verbose=True)
        do_try_update(a_out, f"flat_n{n_cur}")

    # === Phase 3: LP Nesterov after flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 280.0)
    print(f"\n=== Phase 3: LP Nesterov n={n_cur} (budget {t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True, seed=42, lp_interval=3)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Stochastic subgradient for diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: Stochastic SG n={n_cur} (budget {t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = stochastic_subgradient(best_seq, t4, rng, verbose=True)
        do_try_update(a_out, f"sgd_n{n_cur}")

    # === Phase 5: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Perturbation restarts ===")
    n_restarts = 3
    t5_total = min(remaining * 0.20, 250.0)
    t5_each = t5_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t5 = min(t5_each, remaining - 40)
        if t5 < 20:
            break
        n_cur = len(best_seq)
        if r % 2 == 0:
            pert = perturb_sequence(best_seq, rng, scale=0.02 + 0.01 * r)
        else:
            pert = perturb_structural(best_seq, rng, scale=0.03)
        a_out, _ = nesterov_lp_optimize(pert, t5, solver, verbose=False, seed=100 + r * 7, lp_interval=4)
        do_try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length exploration ===")
    for delta_n in [100, -100, 200, -200]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        n_try = len(best_seq) + delta_n
        if n_try < 500 or n_try > 5000:
            continue
        t6 = min(remaining * 0.12, 100.0)
        if t6 < 20:
            continue
        print(f"  Trying n={n_try} (budget {t6:.0f}s)")
        init_new = interpolate_sequence(best_seq, n_try)
        a_out, _ = nesterov_lp_optimize(init_new, t6, solver, verbose=False, seed=n_try % 100, lp_interval=4)
        do_try_update(a_out, f"len{n_try}")
        # Also try flattening
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = multiplicative_flattening(best_seq if len(best_seq) == n_try else interpolate_sequence(best_seq, n_try),
                                                  min(remaining * 0.05, 40.0), rng, verbose=False)
            do_try_update(a_out2, f"flat_n{n_try}")

    # === Phase 7: Pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 120.0)
    print(f"\n=== Phase 7: Pair CD n={n_cur} (budget {t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = pair_coordinate_descent(best_seq, t7, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_cur}")

    # === Phase 8: Final LP + Flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.5, 150.0)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget {t8:.0f}s) ===")
    if t8 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t8, solver, verbose=True, seed=999, lp_interval=2)
        do_try_update(a_out, f"lp_final_n{n_cur}")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, _ = multiplicative_flattening(best_seq, min(remaining * 0.3, 40.0), rng, verbose=True)
        do_try_update(a_out, f"flat_final_n{n_cur}")

    # === Phase 9: Final pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t9 = min(remaining * 0.5, 60.0)
    print(f"\n=== Phase 9: Final PCD n={n_cur} (budget {t9:.0f}s) ===")
    if t9 > 10:
        a_out, _ = pair_coordinate_descent(best_seq, t9, rng, verbose=False)
        do_try_update(a_out, f"pcd_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
