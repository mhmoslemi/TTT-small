def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_xcorr(a, w):
    """
    Compute cross-correlation c_j = sum_k w_k * a_{k-j} for j=0..n-1.
    Here w has length 2n-1 (full autocorrelation length).
    """

    flip_a = a[::-1]
    conv_len = n + len(w) - 1

    FA = rfft(flip_a, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(FA * W, fft_size)
    # c_j = conv[n-1+j] for j=0..n-1
    return conv[n-1 : n-1+n]

    top_idx = np.argsort(b)[::-1][:k]
    w = np.zeros(len(b))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based gradient descent
with Nesterov momentum, top-k peak targeting, LSE refinement, and
coordinate descent polish. Key advantage: O(n log n) per iteration
enables millions of optimization steps.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_xcorr_for_grad(a, w):
    """
    Compute c_j = sum_{k=0}^{2n-2} w_k * a_{k-j} for j=0..n-1.
    This is the gradient-relevant cross-correlation.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    flip_a = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    FA = rfft(flip_a, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(FA * W, fft_size)
    return conv[n - 1: n - 1 + n]


def compute_topk_weights(b, k):
    """Return weights vector (length = len(b)) with 1/k on top-k indices."""
    top_idx = np.argpartition(b, -k)[-k:]
    w = np.zeros(len(b))
    w[top_idx] = 1.0 / k
    return w, b[top_idx].max()


def compute_lse_weights(b, T):
    """Return softmax weights and LSE value."""
    max_b = b.max()
    shifted = T * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    lse_val = max_b + np.log(sum_exp) / T
    return w, lse_val


def grad_of_lse_over_s2(a, b, w, lse_val):
    """
    Compute gradient of F(a) = 2*n*LSE / S^2.
    Returns gradient vector of length n.
    """
    n = len(a)
    S = a.sum()
    c = fast_xcorr_for_grad(a, w)
    # dF/da_j = 4*n*(c_j*S - lse_val) / S^3
    grad = 4.0 * n * (c * S - lse_val) / (S ** 3)
    return grad


def grad_of_lse_constrained(a, b, w, lse_val):
    """
    Gradient of LSE (without the 1/S^2 factor), for use with
    sum(a) = const constraint.
    dLSE/da_j = 2 * c_j
    """
    n = len(a)
    c = fast_xcorr_for_grad(a, w)
    return 2.0 * c


def project_nonneg_sum(x, target_sum):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.full(len(x), target_sum / len(x))
    return x * (target_sum / s)


def objective_from_autocorr(a, b):
    """Compute F = 2*n*max_b/S^2 from autocorrelation."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (S * S)


def nesterov_topk_descent(a0, time_limit, n_topk=20, lr_init=0.5,
                           verbose=True, seed=42):
    """
    Nesterov accelerated gradient descent using top-k peak approximation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    y = a.copy()
    mu = 0.0  # momentum velocity
    best_a = a.copy()
    best_val = objective_from_autocorr(a, fast_autocorr(a))
    
    t0 = time.monotonic()
    lr = lr_init
    n_iter = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation of y
        b = fast_autocorr(y)
        
        # Top-k weights
        w, max_b = compute_topk_weights(b, n_topk)
        
        # Gradient (constrained: minimize LSE s.t. sum(a)=norm)
        c = fast_xcorr_for_grad(y, w)
        grad = 2.0 * c  # gradient of LSE approximation
        
        # Nesterov step
        mu = 0.9 * mu - lr * grad
        y_new = y + mu
        
        # Project
        y_new = project_nonneg_sum(y_new, norm)
        
        # Evaluate
        b_new = fast_autocorr(y_new)
        val = objective_from_autocorr(y_new, b_new)
        
        if val < best_val:
            best_val = val
            best_a = y_new.copy()
            stagnant = 0
            # Increase lr slightly
            lr = min(lr * 1.02, lr_init * 5)
        else:
            stagnant += 1
            if stagnant % 50 == 0:
                lr *= 0.7
            if stagnant > 500:
                # Restart with noise
                noise = rng.normal(0, 0.03 * y.mean(), n)
                y = project_nonneg_sum(y + noise, norm)
                mu = np.zeros(n)
                lr = lr_init * 0.5
                stagnant = 0
        
        y = y_new
        n_iter += 1
        
        if verbose and n_iter % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NGD-topk iter={n_iter} ({elapsed:.1f}s): "
                  f"best={best_val:.8f}, lr={lr:.4f}, stag={stagnant}")
    
    return best_a, best_val


def lse_scipy_optimize(a0, time_limit, T_values=None, seed=42):
    """
    LSE optimization using scipy L-BFGS-B with decreasing temperature.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    if T_values is None:
        T_values = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0]
    
    best_a = a.copy()
    best_val = objective_from_autocorr(a, fast_autocorr(a))
    t0 = time.monotonic()
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / max(1, len(T_values) - T_values.index(T)), 120.0)
        
        # Define objective and gradient for scipy
        n_local = n
        norm_local = norm
        T_local = T
        
        # We'll use a wrapper that calls top-level functions
        # Since we can't use closures, we'll compute manually
        x0 = a.copy()
        
        try:
            res = scipy_minimize(
                _lse_scipy_obj, x0,
                method='L-BFGS-B',
                args=(n_local, norm_local, T_local),
                jac=_lse_scipy_grad,
                bounds=[(0, None)] * n,
                options={'maxiter': 200, 'maxfun': 500,
                         'time_limit': stage_time, 'ftol': 1e-12}
            )
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm / s_c
                    b_c = fast_autocorr(a_cand)
                    val_c = objective_from_autocorr(a_cand, b_c)
                    if val_c < best_val:
                        best_val = val_c
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception:
            pass
    
    return best_a, best_val


def _lse_scipy_obj(x, n, norm, T):
    """LSE objective for scipy (with sum constraint handled by rescaling)."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a * norm / s
    b = fast_autocorr(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    lse = max_b + np.log(exp_vals.sum()) / T
    return 2.0 * n * lse / (a.sum() ** 2)


def _lse_scipy_grad(x, n, norm, T):
    """LSE gradient for scipy."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    a = a * norm / s
    b = fast_autocorr(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    lse = max_b + np.log(sum_exp) / T
    
    c = fast_xcorr_for_grad(a, w)
    grad_conv = 2.0 * c  # dLSE/da
    
    S = a.sum()
    # d(2n*LSE/S^2)/da = 2n*(grad_conv*S - 2*LSE*1*S... 
    # Actually: F = 2n*LSE/S^2
    # dF/da_j = 2n*(dLSE/da_j * S^2 - LSE * 2S * da_j... 
    # Wait, S = sum(a), dS/da_j = 1
    # dF/da_j = 2n*(dLSE/da_j * S^2 - 2S*LSE) / S^4
    #         = 2n*(dLSE/da_j * S - 2*LSE) / S^3
    #         = 2n*(2*c_j * S - 2*LSE) / S^3
    #         = 4n*(c_j*S - LSE) / S^3
    grad = 4.0 * n * (c * S - lse) / (S ** 3)
    return grad


def coordinate_descent(a0, time_limit, seed=42):
    """
    Coordinate descent that targets the top peaks of the autocorrelation.
    Moves mass from peak-contributing positions to low-contribution positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_val = objective_from_autocorr(a, fast_autocorr(a))
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        improved = False
        
        for k_peak in top_peaks:
            # Find which (i, k-i) pairs contribute most to b[k_peak]
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:8]]
            
            # Find positions with low contribution to any peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            free_positions = np.where(~peak_mask)[0]
            if len(free_positions) == 0:
                free_positions = np.where(np.arange(n) != top_idx[0])[0]
            
            # Sort free positions by current value (move from high to low)
            free_sorted = free_positions[np.argsort(a[free_positions])][:20]
            
            for src in top_idx:
                if a[src] < 1e-10:
                    continue
                for dst in free_sorted:
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-10:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        b_t = fast_autocorr(a_trial)
                        val_t = objective_from_autocorr(a_trial, b_t)
                        if val_t < best_val - 1e-12:
                            a = a_trial
                            best_val = val_t
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 0 and stall % 10 == 0:
                # Random perturbations
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    b_t = fast_autocorr(a_trial)
                    val_t = objective_from_autocorr(a_trial, b_t)
                    if val_t < best_val - 1e-12:
                        a = a_trial
                        best_val = val_t
                        best_a = a.copy()
                        stall = 0
                        break
            
            if stall > 200:
                break
    
    return best_a, best_val
