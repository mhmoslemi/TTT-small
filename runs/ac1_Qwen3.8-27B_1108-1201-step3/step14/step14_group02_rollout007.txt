    n, m = len(a), len(b)

    A = rfft(a.astype(np.float64), fft_size)

    """Interpolate sequence to new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s

    new_x = np.linspace(0.0, 1.0, new_n)
