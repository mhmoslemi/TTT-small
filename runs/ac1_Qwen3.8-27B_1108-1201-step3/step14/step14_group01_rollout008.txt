        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    print(f"  Using solver (fallback): SCS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve_primal(a, M, solver, active=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = build_T(a)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lp_solve_dual(a, M, solver, active=None):
    """Solve dual LP: min M*sum(y) s.t. T^T @ y >= 1, y >= 0"""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    if active is not None and 0 < len(active) < size:
        T_use = T[active, :]
    else:
        T_use = T
    m = T_use.shape[0]
    y_var = cp.Variable(m, nonneg=True)
    constraints = [T_use.T @ y_var >= 1]
    prob = cp.Problem(cp.Minimize(M * cp.sum(y_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return prob.value  # This is the optimal value of the primal

def lse_gradient_descent(a0, time_limit, verbose=True):
    """Smooth gradient descent with adaptive temperature and Nesterov momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Adaptive temperature schedule
    current_T = 20.0
    min_T = 0.2
    cool_rate = 0.995
    momentum = 0.9
    velocity = np.zeros(n)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 50 == 0:
            current_T = max(min_T, current_T * cool_rate)

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Smooth max approximation
        shifted = (conv - max_c) / current_T
        shifted = np.maximum(shifted, -50.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()

        # Gradient computation via FFT
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # grad_i = 2 * (sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k])
        # First term: cross-correlation
        ca = irfft(np.conj(A) * W, fft_size)
        g1 = ca[:n]
        # Second term: convolution
        aw = irfft(A * W, fft_size)
        g2 = aw[:n]
        grad = 2.0 * (g1 + g2)

        # Nesterov momentum + backtracking
        velocity = momentum * velocity - grad
        trial_step = norm / (current_T * np.sqrt(n) * (1.0 + 0.01 * it))
        
        a_new = a + trial_step * velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Backtracking
        for bt in range(15):
            sc = fast_conv_self(a_new).max()
            if sc <= max_c + 1e-10:
                break
            trial_step *= 0.5
            velocity = momentum * velocity - grad
            a_new = a + trial_step * velocity
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
        else:
            a_new = a.copy()

        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} T={current_T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Adaptive screening threshold
    screen_frac = 0.85
    min_active = max(30, n // 10)

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive screening
        adaptive_threshold = M * screen_frac
        active = np.where(conv_y > adaptive_threshold)[0]
        if len(active) < min_active:
            active = np.where(conv_y > M * 0.5)[0]
        if len(active) < min_active:
            active = None  # Use all constraints

        if t % 20 == 0:
            # Occasionally tighten screening
            screen_frac = max(0.95, screen_frac + 0.005)

        b = lp_solve_primal(y, M, solver, active)
        if b is None:
            b = lp_solve_primal(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * (1 + 1e-3):
                b_full = lp_solve_primal(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            n_active = len(active) if active is not None else size
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s) active={n_active}: best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with vectorized 1D minimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Candidate j selection
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            # Peak-lag informed candidates
            peak_threshold = cur_max * 0.9
            peak_lags = np.where(conv >= peak_threshold)[0]
            for kl in peak_lags[:3]:
                j_struct = (kl - i) % n
                j_candidates.add(j_struct)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:8]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # 1D minimization: find delta that minimizes max(conv + delta*g)
                # with second-order corrections
                n_samples = 20
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d < 1e-15:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v - 1e-12:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    x = np.linspace(0.0, 1.0, n)
    if mode == 0:
        # Power law decay from left
        a = (1.0 + x * 5.0) ** (-1.5)
    elif mode == 1:
        # Two-sided decay
        a = np.minimum(1.0 + x * 4.0, 1.0 + (1-x) * 4.0) ** (-1.2)
    elif mode == 2:
        # Edge-heavy
        a = 0.3 + 0.7 * (x ** 0.3 + (1-x) ** 0.3) / 2
    elif mode == 3:
        # Cosine window
        a = np.cos(np.pi * x / 2.0) ** 1.5 + np.cos(np.pi * (1-x) / 2.0) ** 1.5
    elif mode == 4:
        # Geometric
        a = 0.97 ** np.arange(n, 0, -1)
    elif mode == 5:
        # Random smooth
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-3.0 * (x - 0.7) ** 2) + 0.3
    elif mode == 6:
        # Sawtooth-like
        a = 1.0 - np.abs(2.0 * x - 1.0) ** 0.5
    else:
        # Uniform with noise
        a = 1.0 + 0.2 * np.sin(6 * np.pi * x) * np.cos(3 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
