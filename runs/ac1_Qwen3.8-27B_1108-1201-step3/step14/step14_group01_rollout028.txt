def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent: transfer mass between pairs (i,j)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            ai = a[i]
            if ai < 1e-9:
                continue

            # Candidate j selection
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.95)[0]
            for kl in peak_lags[:4]:
                j1 = abs(int(kl) - i)
                if j1 < n:
                    j_candidates.add(j1)
                j2 = n - 1 - abs(int(kl) - i)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_val = cur_max
            best_delta = 0.0
            best_j = -1

            for j in list(j_candidates)[:8]:
                if j == i or j < 0 or j >= n:
                    continue
                aj = a[j]
                # Gain vector: how conv changes when transferring delta from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Sample deltas
                n_samples = 20
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a_i_old = a[i]
                a_j_old = a[best_j]
                a[i] = a_i_old - best_delta
                a[best_j] = a_j_old + best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = a_i_old
                    a[best_j] = a_j_old
                else:
                    a = a / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[i] = a_i_old
                        a[best_j] = a_j_old
                        a = a / a.sum()

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 4:
                break
            if no_improve_passes == 2:
                n_perturb = max(3, n // 80)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
        else:
            no_improve_passes = 0

        if verbose and (idx // max(1, n // 10)) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) * np.exp(-2.0 * (x - 0.5) ** 2) + 0.2
    else:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.8) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def optimize_at_length(a_init, time_budget, rng, verbose=True):
    """Run full optimization pipeline at a given length."""
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = fast_conv_self(best_a).max()

    def check(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if verbose:
                print(f"      [{label}] {sc:.8f} (t={time.monotonic()-t0:.0f}s)")

    remaining = time_budget - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score

    # Phase 1: Multiplicative weights (new)
    t1 = min(remaining * 0.30, 300.0)
    if t1 > 20 and verbose:
        print(f"    Phase 1: MW (budget={t1:.0f}s)")
    a_out, _ = mul_weights_optimize(best_a, t1, verbose=verbose)
    check(a_out, "MW")

    remaining = time_budget - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score

    # Phase 2: LSE GD
    t2 = min(remaining * 0.25, 200.0)
    if t2 > 15 and verbose:
        print(f"    Phase 2: LSE GD (budget={t2:.0f}s)")
    a_out, _ = lse_gradient_descent(best_a, t2, verbose=verbose)
    check(a_out, "LSE")

    remaining = time_budget - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score

    # Phase 3: LP Nesterov
    t3 = min(remaining * 0.35, 400.0)
    if t3 > 20 and verbose:
        print(f"    Phase 3: LP Nest (budget={t3:.0f}s)")
    a_out, _ = lp_nesterov_optimize(best_a, t3, verbose=verbose)
    check(a_out, "LP")

    remaining = time_budget - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score

    # Phase 4: Pair CD
    t4 = min(remaining * 0.30, 200.0)
    if t4 > 10 and verbose:
        print(f"    Phase 4: PCD (budget={t4:.0f}s)")
    a_out, _ = pair_cd_optimize(best_a, t4, rng, verbose=verbose)
    check(a_out, "PCD")

    remaining = time_budget - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score

    # Phase 5: Another LP round (shorter)
    t5 = min(remaining * 0.5, 100.0)
    if t5 > 15 and verbose:
        print(f"    Phase 5: LP round 2 (budget={t5:.0f}s)")
    a_out, _ = lp_nesterov_optimize(best_a, t5, verbose=False)
    check(a_out, "LP2")

    remaining = time_budget - (time.monotonic() - t0)
    if remaining < 10:
        return best_a, best_score

    # Phase 6: Final PCD
    t6 = min(remaining * 0.6, 80.0)
    if t6 > 10 and verbose:
        print(f"    Phase 6: Final PCD (budget={t6:.0f}s)")
    a_out, _ = pair_cd_optimize(best_a, t6, rng, verbose=False)
    check(a_out, "PCD_final")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple lengths and optimization strategies
    to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = compute_score_fast(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_check(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = compute_score_fast(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Determine which lengths to try
    lengths = []
    if n_base >= 100:
        # Focus on parent length and nearby
        lengths.append(n_base)
        if n_base > 1200:
            lengths.append(1200)
        if n_base < 2000:
            lengths.append(min(n_base + 400, 2400))
        if n_base > 1600:
            lengths.append(1600)
    else:
        lengths = [1200, 1600, 2000]

    # Sort by priority (parent length first)
    lengths = [n_base] + [l for l in lengths if l != n_base]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen and l >= 100:
            seen.add(l)
            unique_lengths.append(l)
    lengths = unique_lengths[:4]  # max 4 lengths

    # Allocate time
    n_lengths = len(lengths)
    # Give more time to the first (parent) length
    weights = []
    for i, l in enumerate(lengths):
        if l == n_base:
            weights.append(1.5)
        else:
            weights.append(1.0)
    total_w = sum(weights)
    times = [time_budget * w / total_w for w in weights]

    print(f"Lengths to try: {lengths}, times: {[f'{t:.0f}s' for t in times]}")

    for li, (length, t_alloc) in enumerate(zip(lengths, times)):
        if time.monotonic() - t_global > time_budget - 30:
            break

        remaining_total = time_budget - (time.monotonic() - t_global)
        t_alloc = min(t_alloc, remaining_total - 20)
        if t_alloc < 30:
            break

        print(f"\n{'='*50}")
        print(f"Length {length} (budget={t_alloc:.0f}s)")
        print(f"{'='*50}")

        # Initialize at this length
        if length == len(best_seq):
            a_init = best_seq.copy()
        else:
            a_init = interpolate_seq(best_seq, length)

        a_out, sc_out = optimize_at_length(a_init, t_alloc, rng, verbose=True)
        try_update(a_out, f"len_{length}")

        # Try a different initialization for this length
        remaining_total = time_budget - (time.monotonic() - t_global)
        if remaining_total > 60 and li == 0:
            a_init2 = structured_init(length, rng)
            sc_init2 = compute_score_fast(a_init2)
            if sc_init2 < best_score * 1.1:  # only if competitive
                t_alloc2 = min(remaining_total * 0.3, 200.0)
                print(f"\n  Alternative init for n={length} (budget={t_alloc2:.0f}s)")
                a_out2, _ = optimize_at_length(a_init2, t_alloc2, rng, verbose=False)
                try_update(a_out2, f"alt_{length}")

    # Final polish on the best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\nFinal polish: n={len(best_seq)}, budget={remaining:.0f}s")
        t_lp = min(remaining * 0.5, 100.0)
        if t_lp > 15:
            a_out, _ = lp_nesterov_optimize(best_seq, t_lp, verbose=False)
            try_update(a_out, "final_LP")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_pc = min(remaining * 0.7, 60.0)
            a_out, _ = pair_cd_optimize(best_seq, t_pc, rng, verbose=False)
            try_update(a_out, "final_PCD")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_check(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
