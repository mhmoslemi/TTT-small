def lp_best_response_fast(a, M, solver):
    """Solve LP best response: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1

    # Build constraint matrix sparsely
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        rows.extend(idx.tolist())
        cols.extend([i] * len(idx))
        vals.extend(a[idx - i].tolist())

    from scipy.sparse import coo_matrix
    T = coo_matrix((vals, (rows, cols)), shape=(size, n)).tocsr()
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, use_line_search=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_fast(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        if use_line_search:
            alpha_base = 2.0 / (t + 2.0)
            best_alpha = alpha_base
            best_obj = fast_autocorr((1 - alpha_base) * y + alpha_base * g).max()
            for trial in [0.5 * alpha_base, 1.5 * alpha_base, 2.0 * alpha_base, 0.25 * alpha_base]:
                trial = min(trial, 1.0)
                cand = np.maximum((1 - trial) * y + trial * g, 0.0)
                sc = fast_autocorr(cand).max()
                if sc < best_obj:
                    best_obj = sc
                    best_alpha = trial
            alpha = best_alpha
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0

        if t % 200 == 0:
            stall_count += 1
            if stall_count > 30:
                if verbose:
                    print(f"    LP Stalling at iter={n_iters}, best={best_score:.8f}")
                break
            elif stall_count > 10:
                y = best_a * norm / best_a.sum()
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42, verbose=False):
    """Aggressive coordinate descent with gradient-guided peak reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    total_tries = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall, total_tries
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        total_tries += 1
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    # Precompute for speed
    frac_list = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        max_peak_val = b.max()

        for k_peak in top_peaks[:10]:
            if b[k_peak] < max_peak_val * 0.95:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            if contrib.sum() < 1e-15:
                continue
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            # Build mask of positions involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-12:
                    continue
                # Target: positions NOT in peak mask, with lowest a values
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]

                for dst in candidates_sorted[:6]:
                    for frac in frac_list:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 5 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            if stall % 10 == 0:
                # Multi-point perturbation
                n_perturb = max(3, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.4, 1.6)
                try_update(a_trial)

            if stall > 500:
                break

        if verbose and total_tries % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD tries={total_tries} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) * (1 + 0.2 * np.sin(6 * np.pi * x))
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    # Phase 1: Adam LSE at base length (main workhorse - 40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam LSE n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = adam_lse_optimize(a_norm, t1, T_init=3.0, T_end=0.05,
                                     lr_init=0.03, verbose=True)
        do_try_update(a_out, f"adam_n{n_base}")

    # Phase 2: LP Nesterov at base length (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.20
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, use_line_search=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 3: Adam LSE with different hyperparameters (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Adam LSE v2 n={n_best} (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        # Use different initial temperature and learning rate
        a_out, _ = adam_lse_optimize(a_norm, t3, T_init=1.0, T_end=0.02,
                                     lr_init=0.01, verbose=False)
        do_try_update(a_out, f"adam2_n{n_best}")

    # Phase 4: Try a longer length (10% time)
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Adam LSE n={n_long} (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.10
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = adam_lse_optimize(a_norm, t4, T_init=3.0, T_end=0.1,
                                     lr_init=0.03, verbose=False)
        do_try_update(a_out, f"adam_long_n{n_long}")

    # Phase 5: Coordinate descent polish (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CD polish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = remaining * 0.15
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345, verbose=False)
        do_try_update(a_out, "cd_polish")

    # Phase 6: Another Adam pass at best (refined temps) (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Adam refine n={n_best} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.10
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adam_lse_optimize(a_norm, t6, T_init=0.5, T_end=0.01,
                                     lr_init=0.005, verbose=False)
        do_try_update(a_out, "adam_refine")

    # Phase 7: Final coordinate descent (5% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.5
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=77777, verbose=False)
        do_try_update(a_out, "cd_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
