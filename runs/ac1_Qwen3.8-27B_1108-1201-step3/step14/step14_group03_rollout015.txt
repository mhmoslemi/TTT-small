def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def soft_grad_descent(a0, time_limit, T0=5.0, T_min=0.05, verbose=True):
    """Soft-max gradient descent with multiplicative (mirror) updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    T = T0
    n_iters = 0
    n_anneal = 0
    
    while time.monotonic() - t0 < time_limit:
        if T < T_min:
            break
        
        c = fast_autocorr(a)
        c_max = c.max()
        
        # Soft-max weights (shifted to avoid overflow)
        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(size)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: grad[j] = 2 * (w ⋆ a)[j] = 2 * real(ifft(conj(fft(w)) * fft(a)))[:n]
        W_fft = fft(w, fft_size)
        A_fft = fft(a, fft_size)
        cross = np.real(ifft(np.conj(W_fft) * A_fft, fft_size))[:n]
        grad = 2.0 * cross
        
        # Adaptive step size
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            eta = 0.1
        else:
            eta = 0.5 / max_grad
        
        # Multiplicative update
        log_update = -eta * grad
        # Clip to avoid underflow/overflow
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        n_iters += 1
        
        # Check score periodically
        if n_iters % 50 == 0 or n_iters < 10:
            score = fast_autocorr(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
        
        # Anneal
        if n_iters % 100 == 0:
            T *= 0.9
            n_anneal += 1
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            cur_score = fast_autocorr(a).max()
            print(f"    SGD iter={n_iters} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    # Final check
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_exchange_search(a0, time_limit, rng, window=50, verbose=True):
    """
    Exhaustive pair-exchange local search.
    For each pair (i,j) within a window, find the optimal mass transfer
    that minimizes max autocorrelation, using quadratic-aware evaluation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_samples = 30  # number of points to evaluate for each pair
    
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Try pairs within window
            max_offset = min(window, n - 1)
            for offset in range(1, max_offset + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    
                    # Compute delta_vec: change in autocorrelation per unit d
                    # delta_c[k] = 2d * u[k] + d^2 * v[k]
                    # where u[k] = a[k-j] - a[k-i] (with a[m]=0 outside [0,n-1])
                    # and v[k] = 1_{k=2i} + 1_{k=2j} - 2*1_{k=i+j}
                    
                    u = np.zeros(size)
                    # a[k-j] for k-j in [0,n-1], i.e., k in [j, j+n-1]
                    lo1 = j
                    hi1 = min(size, j + n)
                    if hi1 > lo1:
                        u[lo1:hi1] += a[:hi1-lo1]
                    # a[k-i] for k-i in [0,n-1], i.e., k in [i, i+n-1]
                    lo2 = i
                    hi2 = min(size, i + n)
                    if hi2 > lo2:
                        u[lo2:hi2] -= a[:hi2-lo2]
                    u *= 2.0  # factor of 2
                    
                    v = np.zeros(size)
                    k_2i = 2 * i
                    if 0 <= k_2i < size:
                        v[k_2i] += 1.0
                    k_2j = 2 * j
                    if 0 <= k_2j < size:
                        v[k_2j] += 1.0
                    k_ij = i + j
                    if 0 <= k_ij < size:
                        v[k_ij] -= 2.0
                    
                    # f(d) = max(conv + d*u + d^2*v) for d in [0, ai]
                    # Also try negative d (transfer from j to i)
                    # f(-d') = max(conv - d'*u + d'^2*v) for d' in [0, a[j]]
                    
                    # Forward: transfer from i to j
                    best_d = 0.0
                    best_val = conv.max()
                    if ai > 1e-15:
                        ds = np.linspace(0.0, ai * 0.999, n_samples)
                        for d in ds[1:]:  # skip d=0
                            new_conv = conv + d * u + d * d * v
                            new_max = new_conv.max()
                            if new_max < best_val - 1e-14:
                                best_val = new_max
                                best_d = d
                    
                    # Backward: transfer from j to i
                    if a[j] > 1e-15:
                        ds = np.linspace(0.0, a[j] * 0.999, n_samples)
                        for d in ds[1:]:
                            new_conv = conv - d * u + d * d * v
                            new_max = new_conv.max()
                            if new_max < best_val - 1e-14:
                                best_val = new_max
                                best_d = -d  # negative means backward
                    
                    if best_d != 0.0:
                        # Apply the move
                        if best_d > 0:
                            a[i] -= best_d
                            a[j] += best_d
                        else:
                            a[i] += (-best_d)
                            a[j] -= (-best_d)
                        
                        # Renormalize
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += abs(best_d)
                            a[j] -= abs(best_d)
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        
                        if cur_max < best_score - 1e-14:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            # Revert
                            if best_d > 0:
                                a[i] += best_d
                                a[j] -= best_d
                            else:
                                a[i] += (-best_d)
                                a[j] -= (-best_d)
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PXS pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_samples = 20

        n_passes += 1
        pass_improved = False

            if time.monotonic() - t0 > time_limit:

            ai = a[i]
            
            # If we change a[i] to a[i] + delta, the autocorrelation changes by:
            # delta_c[k] = 2*delta*a[k-i] + delta^2 * 1_{k=2i}
            # (with a[m] = 0 outside [0,n-1])
            
            # So new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2 * 1_{k=2i}
            # We want to minimize max_k new_conv[k] over delta in [-ai, 1000]
            
            # Precompute the linear term
            linear = np.zeros(size)
            lo = i
            hi = min(size, i + n)
            if hi > lo:
                linear[lo:hi] = 2.0 * a[:hi-lo]
            
            quad_coeff = np.zeros(size)
            k_2i = 2 * i
            if 0 <= k_2i < size:
                quad_coeff[k_2i] = 1.0
            
            # Find best delta
            best_delta = 0.0
            best_val = conv.max()
            
            # Try a range of deltas
            # Most useful deltas are in [-ai, ai] (small perturbations)
            # But also try larger ones
            deltas = np.concatenate([
                np.linspace(-ai, ai, n_samples),
                np.array([ai * 2, ai * 5, -ai * 0.5])
            ])
            deltas = deltas[(deltas > -ai + 1e-12) & (deltas < 1000.0)]
            
            for d in deltas:
                if abs(d) < 1e-15:
                    continue
                new_conv = conv + d * linear + d * d * quad_coeff
                new_max = new_conv.max()
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_delta = d
            
            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(0.0, a[i])
                s = a.sum()
                if s < 1e-15:
                    a[i] -= best_delta
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

Actually, let me reconsider. For each pair `(i, j)`:
- Computing `u`: O(n) (just two shifted copies of `a`)
- Evaluating `f(d)` at one point: O(n) (just `max(conv + d*u + d^2*v)`)
- With 30 sample points: O(30n)
- Total per pair: O(31n)

With window W and n elements: n * W * 2 * 31n = 62 * W * n^2 operations per pass.
For n=1600, W=40: 62 * 40 * 1600^2 = 62 * 40 * 2.56M = 6.3 billion. Too slow.

I need to either:
1. Reduce the window
2. Reduce the number of sample points
3. Use a smarter line search (ternary search, which needs O(log(1/eps)) evaluations)
4. Use the fact that `f(d)` is the max of linear/quadratic functions to find the minimum more efficiently

Option 4 is the best. Since `f(d) = max_k [conv[k] + 2d*u[k] + d^2*v[k]]`, and `v[k]` is non-zero at at most 3 positions, we have:

For most `k`: `f_k(d) = conv[k] + 2d*u[k]` (linear)
For `k = 2i`: `f_k(d) = conv[2i] + 2d*u[2i] + d^2`
For `k = 2j`: `f_k(d) = conv[2j] + 2d*u[2j] + d^2`
For `k = i+j`: `f_k(d) = conv[i+j] + 2d*u[i+j] - 2d^2`

The maximum of linear functions is a convex piecewise-linear function. Adding the quadratic terms (with positive coefficients at `2i` and `2j`) keeps it convex. The term at `i+j` has a negative coefficient, but it's only active in a limited range.

Actually, for the purpose of finding the minimum, I can use ternary search since the function is "mostly" convex. Let me use 15 iterations of ternary search, which gives precision of about (2/3)^15 ≈ 0.002 of the initial range. That's good enough.

With ternary search: O(15 * n) per pair instead of O(30 * n). Not a huge improvement.

def pair_exchange_search(a0, time_limit, rng, window=30, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            if ai < norm * 1e-12:

                    u_j = _shifted_a(a, j, size)  # u_j[k] = a[k-j] if valid, else 0
                    u_i = _shifted_a(a, i, size)  # u_i[k] = a[k-i] if valid, else 0
                    u = 2.0 * (u_j - u_i)
                    
                    # v vector (nonzero at at most 3 positions)

                    if 0 <= k_2i < size:

                    if 0 <= k_2j < size:

                            a[j] += best_d
                        else:
                            a[i] += (-best_d)
                            a[j] -= (-best_d)
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s < 1e-15:
                            # revert
                            if best_d > 0:
                                a[i] += best_d
