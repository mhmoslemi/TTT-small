    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    
    return b_val

def lp_nesterov(a0, time_limit, solver, n_constraints=300, verbose=True):
    """LP-based minimax with Nesterov acceleration and reduced constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-12:
            break
        
        # Select top n_constraints lags
        if n_constraints < 2 * n - 1:
            top_idx = np.argpartition(ac, -n_constraints)[-n_constraints:]
        else:
            top_idx = np.arange(2 * n - 1)
        
        b = lp_best_response(y, M_curr, solver, constraint_indices=top_idx)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check score
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Ergodic averaging
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coord_search(a0, time_limit, seed=42):
    """Local coordinate search for final polish."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        peak_k = np.argmax(ac)
        
        # Find coordinates that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Try moving mass from high-contribution to low-contribution positions
        src = idx[np.argmax(contrib)]
        
        # Find good destinations
        dest_candidates = np.where(a < a.mean() * 0.5)[0]
        if len(dest_candidates) == 0:
            dest_candidates = np.where(np.arange(n) != src)[0]
        dst = dest_candidates[rng.integers(len(dest_candidates))]
        
        improved = False
        for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            s_t = a_trial.sum()
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            stall += 1
            if stall > 500:
                break
    
    return best_a, best_score
