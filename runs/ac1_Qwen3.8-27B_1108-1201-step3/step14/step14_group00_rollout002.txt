def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a, target_sum=1.0):

    s = a.sum()
    if s < 1e-15:

def build_conv_matrix(a, k_indices):
    """Build rows of convolution matrix for specific lag indices.
    Row k: (T @ b)_k = sum_j T[k,j] * b_j = (a*b)_k
    T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    """
    n = len(a)
    rows = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        # (a*b)_k = sum_j a[k-j] * b_j where 0 <= k-j < n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows[idx, j] = a[k - j]
    return rows


def build_conv_matrix_full(a):
    """Build full (2n-1) x n convolution matrix."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lp_solve(a, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) < size:
        T = build_conv_matrix(a, active_set)
    else:
        T = build_conv_matrix_full(a)
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None, None
    
    # Get dual values
    duals = None
    try:
        if isinstance(constraints[0], cp.Inequality) or hasattr(constraints[0], 'dual_value'):
            dual_val = constraints[0].dual_value
            if dual_val is not None:
                duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, b_sum, duals


def nesterov_lp(a0, time_limit, solver, seed=0, verbose=True):
    """Nesterov-accelerated LP best-response with adaptive active set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    
    a = normalize(a0, norm)
    y = a.copy()
    
    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()
    
    # Active set: start with top lags
    conv0 = fast_autocorr(a)
    M0 = conv0.max()
    initial_frac = 0.3
    active_set = np.where(conv0 > M0 * (1.0 - 0.15 * initial_frac))[0]
    if len(active_set) < 10:
        active_set = np.argsort(conv0)[-max(10, int(0.1 * size)):]
    
    t0 = time.monotonic()
    t = 0
    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0
    
    # Momentum state
    prev_y = y.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        M = fast_autocorr(y).max()
        if M < 1e-12:
            break
        
        # Solve LP
        b, b_sum, duals = lp_solve(y, M, active_set, solver)
        
        if b is None:
            # Retry with full constraint set
            b, b_sum, duals = lp_solve(y, M, None, solver)
            if b is None:
                # Maybe M is too tight, increase slightly
                b, b_sum, duals = lp_solve(y, M * 1.01, active_set, solver)
            if b is None:
                continue
        
        # Check for constraint violations outside active set
        if active_set is not None and len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_inact = build_conv_matrix(y, inactive)
                viol = T_inact @ b
                violated = inactive[viol > M * 1.001]
                if len(violated) > 0:
                    active_set = np.union1d(active_set, violated)
                    if len(active_set) > 0.9 * size:
                        active_set = None
                    # Re-solve with expanded active set
                    b, b_sum, duals = lp_solve(y, M, active_set, solver)
                    if b is None:
                        b, b_sum, duals = lp_solve(y, M, None, solver)
                        if b is None:
                            continue
        
        # Update active set based on current autocorrelation
        if t % 50 == 0 or (duals is not None and len(duals) == len(active_set)):
            conv_y = fast_autocorr(y)
            M_y = conv_y.max()
            # Include lags that are close to max
            tol = 0.05
            new_active = np.where(conv_y > M_y * (1.0 - tol))[0]
            # Also include any that were previously active
            if active_set is not None:
                new_active = np.union1d(new_active, active_set)
            # Limit size
            if len(new_active) > 0.7 * size:
                # Keep only top 70%
                new_active = np.argsort(conv_y)[-int(0.7 * size):]
            active_set = new_active
        
        g = normalize(b, norm)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_norm = normalize(avg_a / n_avg, norm)
            score_avg = fast_autocorr(avg_norm).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_norm.copy()
                stall = 0
        
        # Restart if stalled
        if stall > 80:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0
            # Reset active set
            conv_best = fast_autocorr(best_a)
            M_best = conv_best.max()
            active_set = np.where(conv_best > M_best * 0.85)[0]
            if verbose:
                print(f"    Restart at t={t}, score={best_score:.10f}")
        
        if verbose and t % 200 == 0:
            print(f"    LP t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, |active|={0 if active_set is None else len(active_set)}")
    
    return best_a, best_score


def subgrad_cd(a0, time_limit, rng, verbose=True):
    """Subgradient-based coordinate descent for peak shaving.
    
    Strategy: Identify top-K peaks, compute combined subgradient,
    then move mass from high-subgradient coordinates to low-subgradient ones.
    """
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    K = 10  # Number of top peaks to consider
    
    while time.monotonic() - t0 < time_limit:
        # Find top K peaks
        top_k = np.argsort(conv)[-K:]
        top_k_sorted = top_k[np.argsort(conv[top_k])[::-1]]
        
        # Compute combined subgradient
        # For lag k: dc_k/da_i = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # But since we're looking at the weighted sum of top peaks:
        # d/d(a_i) sum_{k in top_k} w_k * c_k where w_k are weights
        # Use softmax weights over top K
        top_vals = conv[top_k]
        w = np.exp(top_vals / max(top_vals.max(), 1e-15) * 10)  # Softmax with temperature
        w = w / w.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(top_k):
            if w[idx] < 1e-10:
                continue
            # dc_k/da_i = a[k-i] + a[i-k] (boundary handled)
            # For k: i ranges such that 0 <= k-i < n => k-n+1 <= i <= k
            # For k: i ranges such that 0 <= i-k < n => k <= i <= k+n-1
            # Combined: max(0, k-n+1) <= i <= min(n-1, k+n-1)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k + n - 1)
            for i in range(i_lo, i_hi + 1):
                # Term 1: a[k-i] if 0 <= k-i < n
                if 0 <= k - i < n:
                    grad[i] += w[idx] * 2.0 * a[k - i]
                # Term 2: a[i-k] if 0 <= i-k < n (but this is the same as term 1 when we swap)
                # Actually for autocorrelation: c_k = sum_i a_i * a_{i+k} (for k >= 0)
                # d(c_k)/d(a_i) = a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
                # Wait, let me be more careful.
                # c_k = sum_{i: 0<=i<n, 0<=i+k<n} a_i * a_{i+k}
                # d(c_k)/d(a_j) = a_{j+k} (if j+k < n) + a_{j-k} (if j-k >= 0)
                # But we already added this above. Let me redo.
            pass
        
        # Actually, let me be more careful with the gradient.
        # For 0 <= k <= n-1: c_k = sum_{i=0}^{n-1-k} a_i * a_{i+k}
        #   d(c_k)/d(a_j) = a_{j+k} (if j+k < n) + a_{j-k} (if j-k >= 0)
        #   Note: the second term appears because a_j also appears as a_{(j-k)+k} = a_j
        #   when i = j-k, contributing a_{j-k} * a_j, derivative w.r.t. a_j is a_{j-k}
        # For n <= k <= 2n-2: let k' = 2n-2-k (by symmetry of autocorrelation)
        #   Actually, (a*a)_k = (a*a)_{2n-2-k} only if a is palindromic.
        #   In general: (a*a)_k = sum_{i=k-n+1}^{n-1} a_i * a_{k-i} for n <= k <= 2n-2
        #   Let's just handle it directly.
        
        # Let me recompute the gradient properly.
        grad = np.zeros(n)
        for idx, k in enumerate(top_k):
            wi = w[idx]
            if wi < 1e-12:
                continue
            # c_k = sum_i a_i * a[k-i] where 0<=i<n and 0<=k-i<n
            # d(c_k)/d(a_j) = sum_i [delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j}]
            #                = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=j<n and k-j is the i value)
            # Wait, let me be more careful.
            # c_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * a_{k-i}
            # d(c_k)/d(a_j) = sum_{i} [delta_{i,j} a_{k-i} + a_i delta_{k-i,j}]
            # = a_{k-j} (if 0<=k-j<n) + sum_{i: k-i=j} a_i = a_{k-j} + a_{k-j} = 2*a_{k-j} (if 0<=k-j<n)
            # Wait that's only if both i=j and k-i=j are valid, which means i=j and i=k-j, so j=k/2.
            # Let me redo this.
            # c_k = sum_i a_i * a_{k-i}  (sum over valid i)
            # d/d(a_j): 
            #   - a_j appears as the first factor when i=j: contributes a_{k-j} (if k-j valid)
            #   - a_j appears as the second factor when k-i=j, i.e., i=k-j: contributes a_{k-j} (if k-j valid)
            #   So d(c_k)/d(a_j) = a_{k-j} * [1_{0<=k-j<n}] + a_{k-j} * [1_{0<=k-j<n}]
            #   = 2 * a_{k-j} if 0 <= k-j < n
            #   Wait, that's not right either. Let me be very explicit.
            #   d/d(a_j) [a_j * a_{k-j}] = a_{k-j}  (when i=j is a valid term)
            #   d/d(a_j) [a_{k-j} * a_j] = a_{k-j}  (when i=k-j is a valid term, and the term is a_{k-j} * a_{j})
            #   But a_{k-j} * a_j = a_j * a_{k-j}, so these are the SAME term!
            #   Unless j = k-j, i.e., j = k/2, in which case it's a_j^2 and the derivative is 2*a_j.
            #   
            # OK so the correct formula is:
            # d(c_k)/d(a_j) = a_{k-j} if 0 <= k-j < n and j != k-j
            #               = 2*a_j if j == k-j (i.e., k = 2j)
            #               = a_{k-j} + a_{j-k} if both are valid and different
            #
            # Actually wait. The autocorrelation (a*a)_k = sum_i a_i * a_{i+k} for k >= 0.
            # But in the "full" convolution, (a*a)_k = sum_i a_i * a_{k-i} for all k.
            # These are the same! Because (a*a)_k = sum_i a_i * a_{k-i} = sum_j a_j * a_{k-j}
            # and (a*a)_{2n-2-k} = sum_i a_i * a_{2n-2-k-i}
            # Let i' = n-1-i, then 2n-2-k-i = 2n-2-k-n+1+i' = n-1-k+i'
            # So (a*a)_{2n-2-k} = sum_{i'} a_{n-1-i'} * a_{n-1-k+i'}
            # This equals sum_{i} a_i * a_{i+k} only if a is palindromic.
            #
            # Let me just use the direct formula: (a*a)_k = sum_i a_i * b_{k-i} where b = a.
            # = sum_i a_i * a_{k-i}
            # Valid when 0<=i<n and 0<=k-i<n, i.e., max(0,k-n+1) <= i <= min(n-1,k).
            #
            # d/d(a_j) of sum_i a_i * a_{k-i}:
            # The term i=j gives: a_j * a_{k-j}, derivative = a_{k-j} (if k-j valid)
            # The term i=k-j gives: a_{k-j} * a_j, derivative = a_{k-j} (if k-j valid)
            # But if j = k-j (i.e., j = k/2), these are the same term: a_j * a_j = a_j^2, derivative = 2*a_j.
            # If j != k-j but both are valid, we get a_{k-j} + a_{k-j} = 2*a_{k-j}.
            # Wait no! If j != k-j, the term i=j gives a_j * a_{k-j} and the term i=k-j gives a_{k-j} * a_j.
            # These are two DIFFERENT terms in the sum (different i values), and both contain a_j.
            # So the derivative is a_{k-j} (from first) + a_{k-j} (from second) = 2*a_{k-j}.
            # If j = k/2, it's a single term a_j^2, derivative = 2*a_j = 2*a_{k-j}. Same!
            #
            # So: d(c_k)/d(a_j) = 2 * a_{k-j} if 0 <= k-j < n, else 0.
            #
            # Wait, but what about negative k? In the full convolution, k ranges from 0 to 2n-2.
            # For k >= 0: (a*a)_k = sum_i a_i * a_{k-i}
            #   d/d(a_j) = 2 * a_{k-j} if 0 <= k-j < n
            # For k < 0 (if we extend): not applicable since we're using 0 to 2n-2.
            #
            # Actually I realize the full convolution has k from 0 to 2n-2.
            # (a*a)_k = sum_i a_i * a_{k-i} for k = 0, 1, ..., 2n-2.
            # For this to be non-zero, we need 0 <= i < n and 0 <= k-i < n.
            # So i ranges from max(0, k-n+1) to min(n-1, k).
            #
            # d/d(a_j) = 2 * a_{k-j} if 0 <= k-j < n (and j is in valid range)
            #
            # Let me verify: if k = 0, d/d(a_j) = 2*a_j (since k-j = -j, which is < 0 for j > 0).
            # Hmm, that gives 0 for j > 0. But (a*a)_0 = sum_i a_i^2, so d/d(a_j) = 2*a_j.
            # The issue is that for k=0, the valid range is i from max(0, -(n-1)) to min(n-1, 0) = 0.
            # So only i=0 is valid? That gives (a*a)_0 = a_0 * a_0 = a_0^2. That's wrong!
            #
            # I think I'm confusing the indexing. Let me use the standard definition:
            # (a*b)_k = sum_i a_i * b_{k-i} where the sum is over all i where both indices are valid.
            # For a = b, (a*a)_k = sum_i a_i * a_{k-i}.
            # For k = 0: sum_i a_i * a_{-i}. Only i=0 gives valid index (-0=0). So (a*a)_0 = a_0^2.
            # That's clearly wrong. The issue is that in numpy's convolve, the indexing is:
            # np.convolve(a, a)[k] = sum_i a[i] * a[k-i] where 0<=i<len(a) and 0<=k-i<len(a).
            # For k=0: only i=0 works (k-i=0), giving a[0]*a[0].
            # For k=1: i=0 gives a[0]*a[1], i=1 gives a[1]*a[0]. Sum = 2*a[0]*a[1].
            # For k=n-1: i=0 gives a[0]*a[n-1], ..., i=n-1 gives a[n-1]*a[0]. Sum = 2*sum_{i<j} a[i]*a[j] + ... 
            # Wait, for k=n-1: valid i are 0 to n-1. a[i]*a[n-1-i]. This includes i and n-1-i.
            # If n is odd, the middle term i=(n-1)/2 gives a[(n-1)/2]^2.
            #
            # So the formula d/d(a_j) of (a*a)_k:
            # a_j appears in the term i=j (giving a_j * a_{k-j}) if 0 <= k-j < n
            # a_j appears in the term i=k-j (giving a_{k-j} * a_j) if 0 <= k-j < n
            # These are the same term when j = k-j, i.e., k = 2j.
            #
            # So: d(a*a)_k / da_j = a_{k-j} * [0<=k-j<n] + a_{k-j} * [0<=k-j<n]
            # = 2 * a_{k-j} * [0<=k-j<n]
            #
            # For k=0: d/d(a_j) = 2*a_j * [0<=-j<n] = 2*a_j * [j=0] = 2*a_0 if j=0, else 0.
            # And (a*a)_0 = a_0^2. d/d(a_0) = 2*a_0. Correct!
            #
            # For k=1: d/d(a_j) = 2*a_{1-j} * [0<=1-j<n]
            # j=0: 2*a_1. j=1: 2*a_0. Others: 0.
            # (a*a)_1 = a_0*a_1 + a_1*a_0 = 2*a_0*a_1.
            # d/d(a_0) = 2*a_1. d/d(a_1) = 2*a_0. Correct!
            #
            # Great, so the formula is: d(a*a)_k / da_j = 2 * a_{k-j} if 0 <= k-j < n, else 0.
            
            # Apply the formula:
            # d(c_k)/d(a_j) = 2 * a[k-j] if 0 <= k-j < n
            # So for a given k, the gradient w.r.t. a_j is non-zero when k-n < j <= k (i.e., 0 <= k-j < n means 0 <= k-j, so j <= k, and k-j < n means j > k-n).
            # Combined with 0 <= j < n: max(0, k-n+1) <= j <= min(n-1, k).
            
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                grad[j] += wi * 2.0 * a[k - j]
        
        # Now grad[i] is the subgradient component for a_i (weighted by top-K peaks)
        # We want to REDUCE the top peaks, so we want to move in the -grad direction.
        # But we need to maintain sum(a) = norm and a >= 0.
        # 
        # Coordinate move: decrease a[p] (where grad[p] is large) and increase a[q] (where grad[q] is small).
        # This is a pair move: a[p] -= d, a[q] += d.
        
        # Find best p (to decrease) and q (to increase)
        # We want to minimize the increase in the objective.
        # First-order change in c_k from moving d from p to q:
        # d(c_k) = d * (grad_q - grad_p) for each peak k
        # So we want grad_p >> grad_q to reduce all peaks.
        
        # Strategy: pick p = argmax(grad * a) [high contribution] and q = argmin(grad) [low sensitivity]
        # But we need a[p] > 0 and the move should actually help.
        
        # Try several candidate moves
        best_move = None
        best_first_order = 0.0
        
        # Get a ranking
        grad_rank = np.argsort(grad)  # ascending
        # q candidates: low gradient (first few)
        # p candidates: high gradient and a[p] > 0 (last few)
        
        q_candidates = grad_rank[:min(20, n)]
        p_candidates = grad_rank[-min(20, n):]
        
        for qi in q_candidates:
            for pi in p_candidates:
                if pi == qi or a[pi] < norm * 1e-8:
                    continue
                # First-order change in weighted peak sum:
                # delta = d * (grad[qi] - grad[pi])
                # We want this negative (reduce peaks), so grad[pi] > grad[qi]
                first_order = grad[pi] - grad[qi]
                if first_order > best_first_order:
                    best_first_order = first_order
                    best_move = (pi, qi)
        
        if best_move is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        pi, qi = best_move
        
        # Line search: try several step sizes
        best_d = 0.0
        best_max = conv.max()
        
        # Compute the effect of moving d from pi to qi on the autocorrelation
        # delta_conv[k] = -d * 2*a[k-pi] + d * 2*a[k-qi] (with boundary handling)
        # = 2d * (a[k-qi] - a[k-pi]) (with validity checks)
        # Actually this is just d * (grad_width[qi] - grad_width[pi]) where grad_width includes all lags.
        # For simplicity, let's just compute the new conv for a few d values.
        
        a[pi] -= a[pi] * 0.5  # temporary for computing direction
        a[qi] += a[pi] * 0.5 + a[pi]  # this is getting messy, let me just do direct evaluation
        
        # Simpler: just try several fractions of a[pi]
        d_candidates = [0.01, 0.025, 0.05, 0.1, 0.2, 0.3, 0.5]
        a_save = a[pi]
        for frac in d_candidates:
            d = a[pi] * frac
            if d < 1e-15:
                continue
            a[pi] -= d
            a[qi] += d
            conv_new = fast_autocorr(a)
            max_new = conv_new.max()
            if max_new < best_max - 1e-14:
                best_max = max_new
                best_d = d
            a[pi] = a_save
            a[qi] -= d
        
        if best_d > 0:
            a[pi] -= best_d
            a[qi] += best_d
            a = normalize(a, norm)
            conv = fast_autocorr(a)
            cur_max = conv.max()
            if cur_max < best_score - 1e-14:
                best_score = cur_max
                best_a = a.copy()
                no_improve = 0
            else:
                # Revert
                a[qi] -= best_d
                a[pi] += best_d
                a = normalize(a, norm)
                conv = fast_autocorr(a)
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
        
        if verbose and no_improve % 100 == 0:
            pass  # reduce output
    
    return best_a, best_score


def peak_shave_cd(a0, time_limit, rng, verbose=True):
    """Faster coordinate descent: directly shave the peak at each step."""
