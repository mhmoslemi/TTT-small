def project_simplex(v, s):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)

    if s <= 1e-15:

    # cross[i] for i in [0, 2n-2] gives sum_k w[k]*a[k-i] (circular)
    # For linear, we want sum over valid k,i
    # Since w has support [0, 2n-2] and a has support [0, n-1]:
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation at lag i (linear, not circular)
    # Using FFT: pad both to size >= 3n
    pad_size = 1 << (3 * n - 1).bit_length()
    W_pad = np.zeros(pad_size)
    W_pad[:size] = w
    A_pad = np.zeros(pad_size)
    A_pad[:n] = a
    W_fft_pad = rfft(W_pad, pad_size)
    A_fft_pad = rfft(A_pad, pad_size)
    cross_pad = irfft(W_fft_pad * np.conj(A_fft_pad), pad_size)
    # cross_pad[i] = sum_k W_pad[k] * A_pad[k-i] (circular)
    # For i in [0, n-1]: cross_pad[i] = sum_k w[k] * a[k-i] (linear, since no wrap-around)
    # But we want: grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * sum_k w[k] * a[k-i]
    # So grad[i] = 2 * cross_pad[i] for i in [0, n-1]
    grad = 2.0 * cross_pad[:n]
    return grad


def smoothed_nesterov_optimize(a0, time_limit, verbose=True):
    """Smoothed gradient descent with Nesterov acceleration and adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    pad_size = 1 << (3 * n - 1).bit_length()

    t0 = time.monotonic()
    n_total_stages = 8
    T_start = 5.0
    T_end = 0.05
    temps = np.geomspace(T_start, T_end, n_total_stages)

    n_stuck = 0
    overall_best_score = best_score

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(300, int(remaining * 3)))

        y = a.copy()
        a_prev = a.copy()
        stage_best = best_score
        stuck = 0

        for it in range(1, n_steps + 1):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1))) / 2.0
            beta = (it - 1) / (it + 1.5)
            y = a + beta * (a - a_prev)
            y = project_simplex(y, 1.0)

            # Compute gradient at y
            conv_y = fast_autocorr(y)
            grad = compute_gradient_smoothing(y, conv_y, T, pad_size)

            # Step size: use Lipschitz-based estimate
            # The Hessian of T*log-sum-exp(conv/T) is bounded by ~ 2/T * ||a||^2 * something
            # Use adaptive step: step = T / (max|grad| + epsilon)
            g_norm = np.abs(grad).max()
            if g_norm < 1e-12:
                step = T / max(n, 1)
            else:
                step = T / (g_norm + T / n)
            # Cap step size
            step = min(step, 1.0 / max(n, 10))

            # Gradient step
            y_new = y - step * grad
            a_new = project_simplex(y_new, 1.0)

            # Evaluate
            score = fast_autocorr(a_new).max()
            if score < overall_best_score - 1e-12:
                overall_best_score = score
                best_a = a_new.copy()
                n_stuck = 0
            else:
                n_stuck += 1

            if score < stage_best:
                stage_best = score

            a_prev = a
            a = a_new

            if n_stuck > 50:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smoothed Nesterov stage {si+1}/{n_total_stages} T={T:.3f} ({elapsed:.0f}s): best={overall_best_score:.8f}")

    return best_a, overall_best_score


def build_conv_matrix(a):
    """Build the convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b) <= M."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_active(a, M, solver, warm_start=None):
    """Active-set LP: solve with only active constraints, add violated ones."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)

    # Start with constraints at the top positions of conv(a, ones)
    ones = np.ones(n) / n
    conv_ones = T @ ones
    top_idx = np.argsort(conv_ones)[::-1][:min(20, size)]

    active_set = set(top_idx.tolist())

    b = warm_start if warm_start is not None else None
    max_iters = 50

    for _ in range(max_iters):
        sorted_active = sorted(active_set)
        T_active = T[sorted_active, :]
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_active @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_new.sum() < 1e-15:
            return None

        # Check for violated constraints
        conv_b = T @ b_new
        violated = np.where(conv_b > M + 1e-10)[0]
        if len(violated) == 0:
            return b_new
        # Add the most violated constraint
        worst = violated[np.argmax(conv_b[violated])]
        active_set.add(worst)
        b = b_new

    # If we didn't converge, solve the full LP
    return lp_best_response(a, M, solver)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_active_set=True):
    """LP Nesterov acceleration for the minimax problem."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    warm_b = None

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if use_active_set:
            b = lp_best_response_active(y, M_curr, solver, warm_start=warm_b)
        else:
            b = lp_best_response(y, M_curr, solver)

        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        warm_b = b.copy()
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = project_simplex(y_new, 1.0)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new, 1.0)

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, window=50, verbose=True):
    """Pair coordinate descent: move mass between pairs to reduce max convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue

            for offset in range(1, min(window, n) + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < 1e-12:
                        break

                    # Compute delta_conv: change in conv when moving d from i to j
                    # delta_conv[k] = 2*d*(a[k-j]*I(0<=k-j<n) - a[k-i]*I(0<=k-i<n))
                    delta_conv = np.zeros(size)

                    # Contribution from a[j] (adding d)
                    lo_j = j
                    hi_j = min(size, j + n)
                    if hi_j > lo_j:
                        delta_conv[lo_j:hi_j] += 2.0 * a[j - j:j - j + (hi_j - lo_j)]
                        # Actually: a[k-j] for k in [j, j+n-1], so a[0], a[1], ..., a[n-1]
                        delta_conv[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]

                    # Remove contribution from a[i]
                    lo_i = i
                    hi_i = min(size, i + n)
                    if hi_i > lo_i:
                        delta_conv[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]

                    # Find best d
                    # We want to minimize max(conv + d * delta_conv)
                    # Also need to handle the diagonal terms
                    best_d = 0.0
                    best_val = conv.max()
                    max_d = a[i] * 0.99

                    # Sample d values
                    n_samples = 10
                    if max_d > 0:
                        deltas = np.linspace(0.0, max_d, n_samples)
                        for d in deltas:
                            if d <= 1e-15:
                                continue
                            new_conv = conv + d * delta_conv
                            # Diagonal corrections
                            if 2 * i < size:
                                new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                            if 2 * j < size:
                                new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                            new_max = new_conv.max()
                            if new_max < best_val - 1e-12:
                                best_val = new_max
                                best_d = d

                    if best_d > 1e-15:
                        a[i] -= best_d
                        a[j] += best_d
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-12:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            # Revert
                            a[j] -= best_d
                            a[i] += best_d
                            a = a / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def single_coord_update(a, conv, i, size):
    """Find optimal value for a[i] given all other a[j] fixed (j != i)."""
    n = len(a)
    # conv[k] = a[i]*a[k-i] + a[k-i]*a[i] + rest (if i != k-i)
    # or conv[k] = a[i]^2 + rest (if i == k-i, i.e., k = 2i)
    # We want: minimize max_k (a_new[i] * contrib[k] + base[k])
    # where base[k] = conv[k] - contribution from a[i]
    # contrib[k] = 2*a[k-i] if k != 2i, or 2*a[i] if k == 2i (but a[i] is the variable)

    # Actually, let's think of it as:
    # conv_new[k] = conv[k] + (a_new - a_old) * c[k] + (a_new - a_old)^2 * d[k]
    # where c[k] = 2*a[k-i] - 2*a[i] (if k != 2i) or 2*a[i] - 2*a[i] = 0 (if k=2i, but then d=1)
    # This is getting complex. Let's just use the quadratic in a_new[i].

    # conv_new[k] as a function of x = a_new[i]:
    # = (x + (a[k-i]-x if k!=2i else 0)) * (a[k-i] + (x - a[i] if k!=2i else 0))
    # Hmm, let me just compute it directly for the top positions.

    # For simplicity, use the linear approximation and find the best x
    # linear: conv_new[k] ≈ conv[k] + (x - a[i]) * 2 * a[k-i] (for k where 0<=k-i<n)
    # We want min max_k conv_new[k]

    # The optimal x satisfies: the maximum is balanced among the top positions
    # This is a 1D minimax problem.

    a_old = a[i]
    # c[k] = 2*a[k-i] for k in [i, i+n-1] (valid range)
    # base[k] = conv[k] - a_old * c[k] + a_old^2 * (1 if k==2i else 0)
    # Actually: conv[k] = sum_j a[j]*a[k-j]
    # If we replace a[i] with x:
    # new_conv[k] = conv[k] - a_old*(a[k-i] + a[i-k] if valid) + x*(a[k-i] + a[i-k] if valid) + (x-a_old)*(x-a_old)*delta_{k,2i}
    # where a[i-k] is 0 if i-k < 0 or i-k >= n

    # For efficiency, just check the top 20 positions of conv
    top_idx = np.argsort(conv)[::-1][:30]

    best_x = a_old
    best_max = conv.max()
    x_candidates = np.linspace(0, a_old * 1.5, 20)
    for x in x_candidates:
        if x < 0:
            continue
        new_conv = conv.copy()
        for k in range(size):
            if 0 <= k - i < n:
                new_conv[k] += (x - a_old) * a[k - i]
            if 0 <= i - k < n and i - k != k - i:
                new_conv[k] += (x - a_old) * a[i - k]
            if k == 2 * i:
                new_conv[k] += (x - a_old) ** 2
        # Renormalize (approximately)
        s_new = a.sum() - a_old + x
        if s_new < 1e-15:
            continue
        new_conv = new_conv * (a.sum() / s_new) ** 2
        m = new_conv.max()
        if m < best_max:
            best_max = m
            best_x = x

    return best_x


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Smoothed Nesterov (main workhorse, 40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    n1 = n_base
    print(f"\n=== Phase 1: Smoothed Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = smoothed_nesterov_optimize(best_seq, t1, verbose=True)
        do_try_update(a_out, f"smooth_n{n1}")

    # === Phase 2: LP Nesterov (active set, 25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True, use_active_set=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: Pair CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Pair CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pair_cd_polish(best_seq, t3, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # === Phase 4: LP Nesterov round 2 (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, use_active_set=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # === Phase 5: Smoothed Nesterov round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Smoothed Nesterov round 2 n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = smoothed_nesterov_optimize(best_seq, t5, verbose=True)
        do_try_update(a_out, f"smooth2_n{n_best}")

    # === Phase 6: Final pair CD with wider window ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = remaining * 0.12
    print(f"\n=== Phase 6: Final Pair CD n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = pair_cd_polish(best_seq, t6, rng, window=80, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # === Phase 7: Try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        for n_alt in [n_base + 100, n_base - 100, n_base + 200]:
            if n_alt < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_alt = min(remaining * 0.3, 180)
            print(f"\n=== Phase 7: Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            init_alt = interpolate_sequence(best_seq, n_alt)
            # Quick smoothed optimization
            a_alt, _ = smoothed_nesterov_optimize(init_alt, t_alt * 0.5, verbose=False)
            do_try_update(a_alt, f"smooth_n{n_alt}")
            # Quick LP
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_lp = min(remaining * 0.3, 80)
                if len(best_seq) == n_alt:
                    a_alt2, _ = nesterov_lp_optimize(best_seq, t_lp, solver, verbose=False, use_active_set=True)
                    do_try_update(a_alt2, f"lp_n{n_alt}")
            # Quick CD
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20 and best_seq is not None and len(best_seq) == n_alt:
                t_cd = min(remaining * 0.2, 30)
                a_alt3, _ = pair_cd_polish(best_seq, t_cd, rng, window=30, verbose=False)
                do_try_update(a_alt3, f"pcd_n{n_alt}")

    # === Final: one more polish on the best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Final Polish n={n_best} (budget={remaining:.0f}s) ===")
        a_out, _ = pair_cd_polish(best_seq, remaining * 0.5, rng, window=60, verbose=True)
        do_try_update(a_out, "final_pcd")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = smoothed_nesterov_optimize(best_seq, remaining * 0.5, verbose=False)
            do_try_update(a_out2, "final_smooth")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
