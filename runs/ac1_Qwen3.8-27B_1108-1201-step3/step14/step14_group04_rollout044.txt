def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lp_best_response(a, M, solver, solver_opts):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)(k) <= M for all k."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False, **solver_opts)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    best_score = fast_autocorr(a).max()
    y = a.copy()

    # Tail averaging: start averaging after (1-tail_frac) fraction of iterations
    tail_avg = np.zeros(n)
    tail_count = 0
    total_iters = 0
    t0 = time.monotonic()
    iter_list = []

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, solver_opts)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        total_iters += 1
        iter_list.append(g.copy())

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Check current iterate
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Tail averaging: average the last tail_frac of iter_list
        if total_iters > 10:
            tail_start = max(0, total_iters - max(10, int(total_iters * tail_frac)))
            if total_iters % 5 == 0:
                tail_window = iter_list[tail_start:]
                tail_avg_candidate = np.mean(tail_window, axis=0)
                tail_avg_candidate = np.maximum(tail_avg_candidate, 0.0)
                s_tc = tail_avg_candidate.sum()
                if s_tc > 1e-15:
                    tail_avg_candidate = tail_avg_candidate * norm / s_tc
                    sc_tail = fast_autocorr(tail_avg_candidate).max()
                    if sc_tail < best_score:
                        best_score = sc_tail
                        best_a = tail_avg_candidate.copy()

        if verbose and total_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={total_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}")

    return best_a, best_score


def polish_phase(a0, time_limit, solver, solver_opts, verbose=True):
    """Polishing: vanilla LP iterations with cumulative averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    all_solutions = []
    iters = 0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(a).max()

        b = lp_best_response(a, M_curr, solver, solver_opts)
        if b is None:
            # Try with slightly larger M
            b = lp_best_response(a, M_curr * 1.001, solver, solver_opts)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b * norm / b_sum
        iters += 1
        all_solutions.append(b.copy())

        # Check b itself
        sc_b = fast_autocorr(b).max()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        # Check average of all solutions
        if iters >= 5:
            avg_all = np.mean(all_solutions, axis=0)
            avg_all = np.maximum(avg_all, 0.0)
            s_avg = avg_all.sum()
            if s_avg > 1e-15:
                avg_all = avg_all * norm / s_avg
                sc_avg = fast_autocorr(avg_all).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_all.copy()

        # Check tail average (last 30 solutions)
        if iters >= 10:
            tail_len = min(30, iters)
            avg_tail = np.mean(all_solutions[-tail_len:], axis=0)
            avg_tail = np.maximum(avg_tail, 0.0)
            s_tail = avg_tail.sum()
            if s_tail > 1e-15:
                avg_tail = avg_tail * norm / s_tail
                sc_tail = fast_autocorr(avg_tail).max()
                if sc_tail < best_score:
                    best_score = sc_tail
                    best_a = avg_tail.copy()

        # Update a with small step
        alpha = 0.05
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * norm / s_new
        else:
            a = a.copy()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish iter={iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}")

    return best_a, best_score


def targeted_shave(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find destination
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])]
            dst = candidates_sorted[rng.integers(max(1, len(candidates_sorted) // 4))]

            for frac in [0.0003, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial * norm / a_trial.sum()
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial * norm / a_trial.sum()
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LSE-FISTA warm-up at parent length
    Phase 2: LP Nesterov at parent length
    Phase 3: LP Nesterov at longer lengths (interpolated init)
    Phase 4: Polishing phase on best found
    Phase 5: Targeted peak shaving
    Phase 6: Final LP polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)

    solver, solver_opts = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1800
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP Nesterov at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 40:
        print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = lp_nesterov(a_norm, t2, solver, solver_opts, verbose=True, tail_frac=0.5)
        try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Try longer sequences ===
    longer_lengths = []
    n_cur = len(best_seq)
    for delta in [200, 400, 600, 800]:
        nl = n_cur + delta
        if nl <= 3200:
            longer_lengths.append(nl)

    for i, n_long in enumerate(longer_lengths):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        # Allocate more time to the first longer length
        frac = max(0.15, 0.35 / (i + 1))
        t3 = min(remaining * frac, 350.0)
        if t3 < 40:
            t3 = min(remaining * 0.3, 200.0)
        if t3 < 30:
            break

        print(f"\n=== Phase 3.{i+1}: LP-Nesterov n={n_long} (budget={t3:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, sc_out = lp_nesterov(a_norm, t3, solver, solver_opts, verbose=True, tail_frac=0.5)
        try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 4: Polishing phase ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 30:
        print(f"\n=== Phase 4: Polishing n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = polish_phase(a_norm, t4, solver, solver_opts, verbose=True)
        try_update(a_out, f"polish_n{n_best}")

    # === Phase 5: Targeted peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.1, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Targeted shave n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = targeted_shave(a_norm, t5, seed=777)
        try_update(a_out, f"shave_n{n_best}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.4, 200.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Final LP n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = lp_nesterov(a_norm, t6, solver, solver_opts, verbose=False, tail_frac=0.4)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final polish if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.3, 100.0)
    if t7 > 20:
        print(f"\n=== Phase 7: Final polish n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = polish_phase(a_norm, t7, solver, solver_opts, verbose=True)
        try_update(a_out, "polish_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
