def lp_best_response_cutting(a, M, solver, max_cuts=30, initial_topk=12):
    """LP best-response using cutting plane method."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    b_curr = fast_autocorr(a)
    top_k = np.argsort(b_curr)[::-1][:initial_topk]
    active = [int(k) for k in top_k]

    for iteration in range(max_cuts):
        m = len(active)
        T = np.zeros((m, n))
        for row, k in enumerate(active):
            T[row] = build_row(a, k, n)

        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None:
            return None
        b_val = np.maximum(b_val, 0.0)

        b_conv = fast_autocorr(b_val)
        violated = np.where(b_conv > M + 1e-8)[0]
        if len(violated) == 0:
            return b_val

        n_add = min(8, len(violated))
        for k in violated[:n_add]:
            k_int = int(k)
            if k_int not in active:
                active.append(k_int)

    # Final solve with all collected constraints
    m = len(active)
    T = np.zeros((m, n))
    for row, k in enumerate(active):
        T[row] = build_row(a, k, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)

        return None
    return np.maximum(b_val, 0.0)

        else:

        b_sum = b.sum()

        n_iters += 1

def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    log_sum = max_b + (shifted.max() + np.log(exp_vals.sum())) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 120:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gradient_optimize_long(a0, time_limit):
    """Gradient-based optimization for longer sequences where LP is too slow."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Use LSE with multiple temperatures
    for T in [40.0, 15.0, 5.0, 2.0]:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 100, 'maxfun': 400,
                                           'gtol': 1e-7, 'ftol': 1e-10,
                                           'maxls': 30})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    GradLong T={T:.0f}: score={score_c:.8f}")
        except Exception:
            pass

    return best_a, best_score


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
