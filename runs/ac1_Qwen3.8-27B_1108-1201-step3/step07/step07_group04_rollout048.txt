def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name

            continue
    _lp_solver = cp.SCS

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using optimized multi-phase approach.
Improvements: adaptive FP, multi-temperature L-BFGS, LP-guided peak shaving,
multi-length search with diverse initializations.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CBC, cp.GLPK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return cp.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    last_exc = None
    for s in [solver] + [s2 for s2 in [cp.PDLP, cp.SCS, cp.ECOS, cp.CBC] if s2 != solver]:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception as e:
            last_exc = e
            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_adaptive(n, time_limit, initial=None, solver=None, verbose=True):
    """Adaptive fictitious play with exponentially weighted average."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    # Exponentially weighted average
    ewma = a.copy().astype(np.float64)
    # Arithmetic average
    arith_avg = a.copy().astype(np.float64)
    iters = 0
    lp_fail_count = 0
    adapt_alpha = 0.3  # Start with moderate step size

    while monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            lp_fail_count += 1
            if lp_fail_count > 50:
                break
            continue
        lp_fail_count = 0

        # Update exponentially weighted average
        ewma = (1 - adapt_alpha) * ewma + adapt_alpha * b_new

        # Update arithmetic average
        arith_avg = (iters * arith_avg + b_new) / (iters + 1)

        # Solve LP best response for a given ewma (or b_new)
        ref = ewma if iters > 10 else b_new
        a_new, M_a = solve_lp_best_response(ref, solver)
        if a_new is None:
            a_new = b_new

        # Update a
        a = (1 - adapt_alpha) * a + adapt_alpha * a_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Evaluate candidates
        candidates = [
            ('a', a),
            ('ewma', ewma),
            ('arith', arith_avg),
            ('mix_ewma_arith', 0.5 * ewma + 0.5 * arith_avg),
            ('a_ewma', 0.5 * a + 0.5 * ewma),
        ]
        for name, cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            max_conv = fast_convolve(cand_n).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_n.copy()

        # Adapt step size: increase if improving, decrease if not
        if best_score < 2.0 - 0.4:  # Making progress
            adapt_alpha = min(0.5, adapt_alpha * 1.02)
        elif iters % 50 == 0:
            adapt_alpha = max(0.05, adapt_alpha * 0.95)

        # Perturbation to escape plateaus
        if iters % 200 == 0 and iters > 100:
            noise = np.random.standard_normal(n) * 0.003
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            ewma = ewma + np.random.standard_normal(n) * 0.001
            ewma = np.maximum(ewma, 0)
            ewma = ewma / ewma.sum()

        if verbose and iters % 50 == 0:
            elapsed = monotonic() - t0
            ma_str = f"{M_a:.8f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.8f}" if M_b is not None else "N/A"
            print(f"    FP n={n} it={iters} t={elapsed:.0f}s: M_a={ma_str} M_b={mb_str} best={best_score:.8f} α={adapt_alpha:.3f}")

    return best_a, best_score, iters


def lbgfs_refine(a0, time_limit, T=50, maxiter=5000):
    """L-BFGS refinement in log-space with softmax smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        # Gradient: d/d a of sum_k w[k] * conv(a,a)[k] = 2 * sum_k w[k] * (a convolved with delta at k)
        # conv(a,a)[k] = sum_j a[j] * a[k-j]
        # d/d a[i] conv(a,a)[k] = a[k-i] + a[i-k] (with bounds)
        # So grad_a[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k])_clipped
        # = 2 * sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) + 2 * sum_k w[k] * (a[i-k] if 0<=i-k<n else 0)
        # = 2 * (sum_k w[k] * a[k-i]) + 2 * (sum_k w[k] * a[i-k])
        # By symmetry of the problem, these two terms are related. Let's just compute directly.
        grad_a = np.zeros(n)
        # Efficient computation using convolution
        # sum_k w[k] * a[k-i] for each i = conv(w_padded, a_padded) appropriately
        # Actually, let's use the fact that:
        # grad_a[i] = 2 * sum_k w[k] * (a[k-i] if valid) 
        # This is 2 * conv(w, a_reversed)[i + n - 1] or similar...
        # Let me just do it directly for correctness (n is at most ~2048)
        # For each k with w[k] > threshold:
        #   for i in range(max(0,k-n+1), min(n, k+1)):
        #     grad_a[i] += 2*w[k]*a[k-i]
        # This is O(n^2) which is fine for n <= 4096
        k_actual = min(len(b), 2*n-1)
        for k in range(k_actual):
            wk = w[k]
            if wk < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad_a[i] += 2.0 * wk * a[k - i]

        # Transform to z-space: a = softmax(z), so da/dz[i] = a[i](delta[i,j] - a[j])
        # grad_z = a * (grad_a - a * dot(grad_a, a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_s = z_final - z_final.max()
            a_out = np.exp(z_s)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    return a_out


def peak_coordinate_descent(a0, time_limit):
    """Peak-focused coordinate descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    stalled = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        current_peak_val = b[peak_k]
        threshold = best_score / (2.0 * n)

        if current_peak_val <= threshold + 1e-15:
            break

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-20:
            break

        # Try reducing top contributors
        top_n = min(8, len(contribs))
        top_order = np.argsort(contribs)[::-1][:top_n]
        improved = False

        for ci in top_order:
            i = int(idx[ci])
            partner = peak_k - i
            if i < 0 or i >= n:
                continue
            old_ai = a[i]
            if old_ai < 1e-15:
                continue

            # Try several reduction factors
            for factor in [0.85, 0.7, 0.5, 0.3]:
                new_val = old_ai * factor
                delta = old_ai - new_val
                if delta < 1e-15:
                    continue

                # Redistribute the reduced amount to non-peak contributors
                a_trial = a.copy()
                a_trial[i] = new_val
                # Distribute to positions that don't contribute to the peak
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in range(max(0, peak_k - n + 1), min(n, peak_k + 1)):
                    j2 = k2
                    peak_mask[j2] = True
                    if 0 <= peak_k - j2 < n:
                        peak_mask[peak_k - j2] = True

                non_peak_mask = ~peak_mask & (np.arange(n) != i)
                if non_peak_mask.any():
                    np_sum = a_trial[non_peak_mask].sum()
                    if np_sum > 1e-15:
                        a_trial[non_peak_mask] += delta * a_trial[non_peak_mask] / np_sum
                else:
                    # Distribute to all non-i positions
                    mask_all = np.ones(n, dtype=bool)
                    mask_all[i] = False
                    all_sum = a_trial[mask_all].sum()
                    if all_sum > 1e-15:
                        a_trial[mask_all] += delta * a_trial[mask_all] / all_sum

                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Try random perturbation
            stall_candidate = a + np.random.standard_normal(n) * 0.002
            stall_candidate = np.maximum(stall_candidate, 0)
            stall_candidate = stall_candidate / stall_candidate.sum()
            score = 2.0 * n * fast_convolve(stall_candidate).max()
            if score < best_score - 1e-12:
                a = stall_candidate
                best_score = score
                best_a = a.copy()
                improved = True

        if not improved:
            stalled += 1
            if stalled > 80:
                # More aggressive restart
                a = best_a + np.random.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
                stalled = 0
                if monotonic() - t0 > time_limit * 0.9:
                    break

        if improved:
            stalled = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
