def lbgfs_optimize(a0, time_limit, T_start, T_end, n_stages=5, maxiter_per_stage=3000):
    """L-BFGS optimization with soft-max smoothing in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    z = z0.copy()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters = maxiter_per_stage
    
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters_this = min(iters, int(remaining * 50))  # rough estimate
        
        def func(z_inner):
            a_inner = np.exp(z_inner - z_inner.max())
            a_inner = a_inner / a_inner.sum()
            c = fast_convolve(a_inner)
            Tc = T_val * c
            max_Tc = Tc.max()
            return (2.0 * n / T_val) * (max_Tc + np.log(np.exp(Tc - max_Tc).sum()))
        
        def jac(z_inner):
            a_inner = np.exp(z_inner - z_inner.max())
            a_inner = a_inner / a_inner.sum()
            c = fast_convolve(a_inner)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            # grad_a[i] = 2 * sum_k w[k] * a[k-i] * I(0<=k-i<n)
            grad_a = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
            # Convert to log-space: grad_z = a * (grad_a - a·grad_a)
            S = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_this, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_this * 100, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at T={T_val}: {e}")
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def mirror_descent(a0, time_limit, lr_init=0.05, n_steps=50000):
    """Mirror descent (exponentiated gradient) on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    lr = lr_init
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        threshold = max_c - 0.01 * max_c
        active = np.where(c >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        z = z - lr * g
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            z = z + lr * g  # undo
        
        # Adaptive lr
        if step % 5000 == 0:
            lr *= 0.9
            lr = max(lr, 1e-6)
        
        if step % 10000 == 0:
            print(f"    MirrorDesc {time.monotonic()-t0:.0f}s: score={best_score:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=10000):
    """Targeted peak shaving."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find the peak (excluding endpoints which are just a[0]^2 and a[n-1]^2)
        c_mid = c.copy()
        c_mid[0] = -np.inf
        c_mid[-1] = -np.inf
        peak_k = int(np.argmax(c_mid))
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.9, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute the removed mass proportionally to others
                removed = a[i] - a_trial[i]
                if removed > 0:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a[mask].sum()
                    if others_sum > 1e-15:
                        a_trial[mask] += removed * a[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing the peak contributor and its partner
            if len(top) >= 2:
                for ci in top[:3]:
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[i] *= 0.95
                    a_trial[j] *= 0.95
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_a, best_score

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: low-temperature L-BFGS in log-space with soft-max smoothing,
alternating exact LP best-response, subgradient mirror descent,
targeted peak shaving, and multi-restart exploration.
Key improvements: ultra-low temperature L-BFGS, more LP iterations with
dual tracking, adaptive active-set targeting, and structured randomization.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_LBFGS_STATE = {}
_LP_SOLVER = None


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _get_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.SCIP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = s
                return s
        except Exception:
            continue
    _LP_SOLVER = cp.ECOS
    return _LP_SOLVER


def simplex_project(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def _build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = _get_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _lbgfs_func(z):
    T_val = _LBFGS_STATE['T']
    n = _LBFGS_STATE['n']
    t0 = _LBFGS_STATE['t0']
    time_limit = _LBFGS_STATE['time_limit']
    if time.monotonic() - t0 > time_limit:
        return 1e20
    a_inner = np.exp(z - z.max())
    a_inner = a_inner / a_inner.sum()
    c = fast_convolve(a_inner)
    Tc = T_val * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    return (2.0 * n / T_val) * (max_Tc + np.log(exp_Tc.sum()))


def _lbgfs_jac(z):
    T_val = _LBFGS_STATE['T']
    n = _LBFGS_STATE['n']
    t0 = _LBFGS_STATE['t0']
    time_limit = _LBFGS_STATE['time_limit']
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z)
    a_inner = np.exp(z - z.max())
    a_inner = a_inner / a_inner.sum()
    c = fast_convolve(a_inner)
    Tc = T_val * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    grad_a = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
    S = np.dot(grad_a, a_inner)
    return a_inner * (grad_a - S)


def lbgfs_optimize(a0, time_limit, T_start, T_end, n_stages=5, maxiter_per_stage=3000):
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    z = z0.copy()
    _LBFGS_STATE['n'] = n
    _LBFGS_STATE['t0'] = t0
    _LBFGS_STATE['time_limit'] = time_limit

    temps = np.geomspace(T_start, T_end, n_stages)
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        _LBFGS_STATE['T'] = T_val
        iters = min(maxiter_per_stage, max(50, int(remaining * 30)))
        try:
            res = minimize(
                fun=_lbgfs_func, x0=z, jac=_lbgfs_jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters * 100, 'disp': False}
            )
            if res.fun < 1e19:
                z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def alternating_lp(a0, b0, time_limit, solver=None, n_iters=200):
    if solver is None:
        solver = _get_solver()
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    b = np.maximum(b0, 0).astype(np.float64)
    b = b / b.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if it % 20 == 0:
            print(f"    AltLP iter={it} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def mirror_descent_opt(a0, time_limit, lr_init=0.03, n_steps=50000):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    lr = lr_init

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        threshold = max_c - 0.005 * max_c
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        z = z - lr * g
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            z = z + lr * g

        if step % 5000 == 0 and step > 0:
            lr = max(lr * 0.9, 1e-7)

        if step % 10000 == 0:
            print(f"    MirrorDesc {time.monotonic()-t0:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def peak_shave_opt(a0, time_limit, n_steps=20000):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_mid = c.copy()
        c_mid[0] = -np.inf
        c_mid[-1] = -np.inf
        peak_k = int(np.argmax(c_mid))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.94, 0.90]:
                a_trial = a.copy()
                a_trial[i] *= factor
                removed = a[i] * (1 - factor)
                if removed > 1e-16:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a[mask].sum()
                    if others_sum > 1e-15:
                        a_trial[mask] += removed * a[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            if len(top) >= 2:
                for ci in top[:4]:
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[i] *= 0.96
                    a_trial[j] *= 0.96
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            top3 = top[:3]
            a_trial = a.copy()
            for ci in top3:
                i = idx[ci]
                a_trial[i] *= 0.97
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def subgradient_opt(a0, time_limit, n_steps=10000, lr_init=0.005):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        eps = 1e-5 * max_c
        active = np.where(c >= max_c - eps)[0]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        lr = lr_init / (1.0 + 0.002 * step)
        a_new = simplex_project(a - lr * g)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new

        if step % 5000 == 0:
            print(f"    SubGrad {time.monotonic()-t0:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def random_restart_opt(a0, time_limit, n_restarts=4, pert_scale=0.03):
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    rng = np.random.default_rng(777)
    t0 = time.monotonic()

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 40:
            break

        noise = rng.normal(0, pert_scale, n)
        a_pert = best_a + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        t_budget = min(remaining * 0.85 / max(1, n_restarts - r), 150.0)
        a_opt = lbgfs_optimize(a_pert, t_budget, T_start=15, T_end=0.5,
                               n_stages=4, maxiter_per_stage=1500)
        score = 2.0 * n * fast_convolve(a_opt).max()
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"    Restart {r}: new best {score:.6f}")

    return best_a, best_score
