    Compute G_i = sum_k w_k * a_{k-i} for i = 0, ..., n-1.
    w has length 2n-1, a has length n.

    # d/da_i [T * log(sum exp(conv_k/T))] = 2 * sum_k w_k * a_{k-i}

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated gradient descent
in log-space with adaptive temperature annealing, peak-aware coordinate descent,
and multi-length exploration. Key improvements over previous approach:
1. Correct and efficient cross-correlation gradient via real FFT
2. Cosine-annealed temperature schedule for smooth-to-sharp transition
3. Peak-aware coordinate descent using the insight that at the optimum,
   mass should flow from high-contribution to low-contribution positions
4. Multi-length exploration (768, 1024, 2048) with interpolation
5. Adaptive learning rate based on gradient norm

import numpy as np
import time
from scipy.fft import rfft, irfft


def next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Score = 2*n*max(conv) for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def cross_corr_w_a(w, a):
    """
    Compute G_i = sum_k w_k * a_{k-i} for i = 0, ..., n-1.
    w has length 2n-1, a has length n.
    Uses real FFT with size >= 3n-2.
    """
    n = len(a)
    N = next_pow2(3 * n - 2)
    W = rfft(w, N)
    A = rfft(a, N)
    P = W * np.conj(A)
    C = irfft(P, N)[:n]
    return C


def compute_smooth_grad_z(z, T, n):
    """
    Compute gradient of 2n*T*LSE(conv/T) in log-space.
    Returns (gradient, smooth_obj, true_score).
    """
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    
    shifted = conv / T
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    w_full = exp_shifted / exp_shifted.sum()
    
    # Gradient of T*LSE(conv/T) w.r.t. a is 2*(w cross-correl a)
    G = cross_corr_w_a(w_full, a)
    grad_a = 2.0 * G
    
    # Project to simplex in log-space (variance-preserving)
    S = np.dot(grad_a, a)
    grad_z = 2.0 * n * a * (grad_a - S)
    
    # Smooth objective value
    smooth_obj = 2.0 * n * (conv.max() + T * np.log(exp_shifted.sum()))
    true_score = 2.0 * n * conv.max()
    
    return grad_z, smooth_obj, true_score


def nesterov_anneal(z0, time_limit, T_start=5000.0, T_end=1.0,
                    lr_init=0.5, momentum=0.9, verbose=True):
    """
    Nesterov-accelerated gradient descent in log-space with cosine temperature annealing.
    """
    n = len(z0)
    z = z0.copy().astype(np.float64)
    v = np.zeros(n)
    a = softmax_from_z(z)
    best_score = compute_score_normalized(a)
    best_z = z.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    iters = 0
    last_report = t0
    
    # Track gradient norm for adaptive learning rate
    grad_norm_history = []
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)

        T = T_end + 0.5 * (T_start - T_end) * (1.0 + np.cos(np.pi * progress))
        
        # Compute gradient
        grad_z, smooth_obj, true_score = compute_smooth_grad_z(z, T, n)
        
        # Adaptive learning rate: scale by inverse gradient norm
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        effective_lr = lr / (1.0 + grad_norm * 0.1)

        v = momentum * v + effective_lr * grad_z
        z_new = z - v
        
        # Numerical safety: clip z values
        z_new = np.clip(z_new, -40.0, 40.0)

        score = compute_score_normalized(a_new)

            print(f"    Nesterov iter={iters}, T={T:.1f}, score={best_score:.8f}, "

def coordinate_descent_peak_aware(a0, time_limit, verbose=True):

    Coordinate descent that uses peak-aware transfer decisions.
    At the main peak k*, transfer mass from i to j when a_{k*-i} > a_{k*-j}.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)

    a = a / s
    best_a = a.copy()

    iters = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Find the main peak
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution array: contrib_i = a_{k_star - i} (circular-ish)
        # For each i in 0..n-1, the "contribution" to peak k_star from position i is a_{k_star-i}
        # (with a_j = 0 if j < 0 or j >= n)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        # Try transfers: move from high-contrib to low-contrib positions
        # Sample pairs biased towards improvement
        n_trials = max(100, n // 4)
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick source i (high contribution) and target j (low contribution)
            if rng.random() < 0.7:
                # Biased: source from top half of contrib, target from bottom half
                top_idx = np.argsort(contrib)[::-1][:n // 4]
                bot_idx = np.argsort(contrib)[:n // 4]
                if len(top_idx) == 0 or len(bot_idx) == 0:
                    continue
                i = top_idx[rng.integers(0, len(top_idx))]
                j = bot_idx[rng.integers(0, len(bot_idx))]
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            
            if i == j or a[i] < 1e-14:
                continue
            
            # Try several transfer fractions
            base = a[i]
            best_t = 0.0
            best_local = best_score
            
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:

                a_trial = a.copy()

                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)

                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss

                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_mr, sc_mr = nesterov_multi_restart(best_seq, t_alloc, n_restarts=3, verbose=True)
        best_seq, best_score = evaluate_and_check(a_mr, "multi_restart", best_seq, best_score)
    
    # === Phase 4: Try n=1024 ===
    print(f"\n=== Phase 4: n=1024 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_alloc = min(rem * 0.4, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_1024 = interpolate_sequence(best_seq, 1024)
        z0_1024 = z_from_a(a_1024)
        a_opt_1024, sc_1024 = nesterov_anneal(z0_1024, t_alloc * 0.7,
                                               T_start=5000.0, T_end=0.5,
                                               lr_init=0.3, momentum=0.9, verbose=True)
        best_seq, best_score = evaluate_and_check(a_opt_1024, "n1024", best_seq, best_score)
        
        # CD polish at 1024
        rem2 = deadline - time.monotonic() - 300
        if rem2 > 20:
            t_cd = min(rem2 * 0.3, 80.0)
            a_cd_1024, _ = coordinate_descent_peak_aware(a_opt_1024, t_cd, verbose=False)
            best_seq, best_score = evaluate_and_check(a_cd_1024, "cd_n1024", best_seq, best_score)
    
    # === Phase 5: Try n=2048 ===
    print(f"\n=== Phase 5: n=2048 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 60:
        t_alloc = min(rem * 0.5, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_2048 = interpolate_sequence(best_seq, 2048)
        z0_2048 = z_from_a(a_2048)
        a_opt_2048, sc_2048 = nesterov_anneal(z0_2048, t_alloc,
                                               T_start=5000.0, T_end=0.5,
                                               lr_init=0.2, momentum=0.9, verbose=True)
        best_seq, best_score = evaluate_and_check(a_opt_2048, "n2048", best_seq, best_score)
    
    # === Phase 6: Try n=1536 ===
    print(f"\n=== Phase 6: n=1536 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 60:
        t_alloc = min(rem * 0.5, 120.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_1536 = interpolate_sequence(best_seq, 1536)
        z0_1536 = z_from_a(a_1536)
        a_opt_1536, sc_1536 = nesterov_anneal(z0_1536, t_alloc,
                                               T_start=3000.0, T_end=0.5,
                                               lr_init=0.3, momentum=0.9, verbose=True)
        best_seq, best_score = evaluate_and_check(a_opt_1536, "n1536", best_seq, best_score)
    
    # === Phase 7: Final Nesterov polish on best ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_alloc = min(rem * 0.6, 60.0)
        print(f"  Budget: {t_alloc:.0f}s")
        z0_final = z_from_a(best_seq)
        a_final, sc_final = nesterov_anneal(z0_final, t_alloc,
                                            T_start=500.0, T_end=0.1,
                                            lr_init=0.1, momentum=0.95, verbose=False)
        best_seq, best_score = evaluate_and_check(a_final, "final_nesterov", best_seq, best_score)
    
    # === Phase 8: Final CD polish ===
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.7, 30.0)
        a_cd_final, _ = coordinate_descent_peak_aware(best_seq, t_alloc, verbose=False)
        best_seq, best_score = evaluate_and_check(a_cd_final, "final_cd", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
    
    # Final verification with evaluator
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
