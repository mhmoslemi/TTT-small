Key innovation: direct subgradient optimization in log-space instead of fictitious play.

import numpy as np

from scipy.optimize import minimize

from time import monotonic


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, peaks, weights):
    """Compute weighted sum of subgradients at given peaks."""
    n = len(a)
    g = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        w = weights[i]
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * w * a[k - idx]
    return g


def mirror_descent(a0, time_limit, eta_init=0.5, eta_decay=0.9999, peak_frac=0.01, verbose=True):
    """Mirror descent (multiplicative weights) with peak-weighted subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    eta = eta_init
    iters = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        b = fast_convolve(a)
        M = b.max()
        
        # Find peaks: top fraction or top-k
        k_peaks = max(1, int(peak_frac * len(b)))
        peak_indices = np.argpartition(b, -k_peaks)[-k_peaks:]
        peak_values = b[peak_indices]
        
        # Soft weights based on peak values
        temp = 0.005 * max(M, 1e-10)
        scaled = (peak_values - M) / temp
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Compute subgradient
        g = compute_subgradient(a, peak_indices, w)
        
        # Mirror descent update (multiplicative weights)
        # a_new[i] = exp(log(a[i]) - eta * g[i]) / sum
        log_a = np.log(a)
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()  # numerical stability
        a = np.exp(log_a_new)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            log_a = np.log(a)
        else:
            a = a / s
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        eta *= eta_decay
        
        if verbose and iters % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    MD iter={iters} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score
