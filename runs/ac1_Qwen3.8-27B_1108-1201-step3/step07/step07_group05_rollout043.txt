```
def interpolate_sequence(f,new_n):
    old_n=len(f)
    if old_n==new_n: return f.copy()
    s=f.sum()
    if s<=0: return np.ones(new_n)/new_n
    f=f/s
    x_old=np.linspace(0,1,old_n)
    x_new=np.linspace(0,1,new_n)
    out=np.interp(x_new,x_old,f)
    out=np.maximum(out,0)
    ss=out.sum()
    if ss>0: out=out/ss
    return out
```

```python
"""
Search for short non-negative sequences minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

The program combines several high-precision local search methods:
1. Corrected L-BFGS optimization in log/softmax space for a smoothed
   log-sum-exp surrogate of the maximum convolution coefficient.
2. A very fast exact pair-move coordinate descent.  For each pair (i,j)
   the effect on the full convolution is computed in O(n) and the best
   transfer over the pair is sampled and locally refined.
3. A mirror-descent / multiplicative-weights method using subgradients
   from top active convolution peaks.
4. Interpolation and random spectral/binary initializations for several
   sequence lengths, followed by final polishing.

The parent construction height_sequence_1 is used when available.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_lbfgs_deadline = 0.0


def normalize_nonneg(a):
    """Return a nonnegative vector of sum 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        a[:] = 1.0
        return a / float(max(1, a.size))
    return a / s


def fast_convolve(a):
    """Linear convolution a*a using FFT."""
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = rfft(a, L)
    return irfft(A * A, L)[:size]


def fast_corr(w, a):
    """
    r_k = sum_m w[m+k] a[m], for k=0..len(a)-1, using FFT.
    """
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = int(w.size) + n - 1
    L = 1 << (size - 1).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    r = irfft(W * np.conj(A), L)
    return r[:n]


def score_from_conv(c, n):
    """Exact score from a full convolution array."""
    if c.size == 0:
        return np.inf
    return 2.0 * float(n) * float(np.max(c))


def internal_total_score(a):
    """Fast internal score (same as evaluator for normalized nonnegative data)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    c = fast_convolve(a)
    return score_from_conv(c, n)


def actual_score(a):
    """Call the provided evaluator, if accessible."""
    try:
        val = float(evaluate_sequence([float(x) for x in a]))
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def softmax_from_z(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    zm = float(np.max(z))
    e = np.exp(z - zm)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def z_from_a(a):
    """Map a nonnegative vector to logit variables, avoiding exact zeros."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([0.0])
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.zeros(a.size, dtype=np.float64)
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / float(a.sum())
    return np.log(a)


def top_weights(c, K):
    """Uniform weights on the K largest convolution coefficients."""
    L = int(c.size)
    w = np.zeros(L, dtype=np.float64)
    if L == 0:
        return w
    if K is None or K <= 0 or K >= L:
        w.fill(1.0 / float(L))
        return w
    idx = np.argpartition(c, -int(K))[-int(K):]
    w[idx] = 1.0 / float(int(K))
    return w


def _lbfgs_obj(z, T):
    """Smoothed score in logit space."""
    if time.monotonic() > _lbfgs_deadline:
        return 1e100
    n = int(z.size)
    if n <= 0:
        return 1e100
    a = softmax_from_z(z)
    c = fast_convolve(a)
    if c.size == 0:
        return 1e100
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e100
    if m <= 0.0:
        return 0.0
    T = float(T)
    if not np.isfinite(T) or T <= 0.0:
        T = 1e-12
    shifted = (c - m) / T
    s = float(np.exp(shifted).sum())
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return 2.0 * float(n) * (m + T * np.log(s))


def _lbfgs_jac(z, T):
    """Gradient of smoothed score in logit space."""
    if time.monotonic() > _lbfgs_deadline:
        return np.zeros(int(z.size), dtype=np.float64)
    n = int(z.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    a = softmax_from_z(z)
    c = fast_convolve(a)
    if c.size == 0:
        return np.zeros(n, dtype=np.float64)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.zeros(n, dtype=np.float64)
    if m <= 0.0:
        return np.zeros(n, dtype=np.float64)
    T = float(T)
    if not np.isfinite(T) or T <= 0.0:
        T = 1e-12
    shifted = (c - m) / T
    w = np.exp(shifted)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    w = w / s
    grad_a = 4.0 * float(n) * fast_corr(w, a)
    if not np.all(np.isfinite(grad_a)):
        return np.zeros(n, dtype=np.float64)
    scalar = float(np.dot(grad_a, a))
    return a * (grad_a - scalar)


def lbfgs_optimize(a, time_budget, T):
    """Run L-BFGS in logit space for a fixed smoothing temperature."""
    a = normalize_nonneg(a)
    n = int(a.size)
    best = a.copy()
    best_s = internal_total_score(best)
    if n <= 0 or time_budget <= 1.0:
        return best, best_s

    if T is None or not np.isfinite(float(T)) or float(T) <= 0.0:
        c = fast_convolve(a)
        T = 0.1 * max(1e-12, float(np.max(c)))
    T = float(T)

    z0 = np.clip(z_from_a(a), -25.0, 25.0)
    t0 = time.monotonic()
    global _lbfgs_deadline
    _lbfgs_deadline = t0 + max(0.5, float(time_budget) - 0.5)

    maxiter = max(500, min(20000, int(float(time_budget) / 0.002)))
    bounds = [(-25.0, 25.0)] * n

    try:
        res = minimize(
            _lbfgs_obj,
            z0,
            jac=_lbfgs_jac,
            args=(T,),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-16,
                "gtol": 1e-10,
                "maxfun": int(max(5000, maxiter * 40)),
                "disp": False,
            },
        )
        if res is not None and hasattr(res, "x"):
            x = np.asarray(res.x, dtype=np.float64)
            if x.size == n and np.all(np.isfinite(x)):
                a2 = softmax_from_z(np.clip(x, -25.0, 25.0))
                s2 = internal_total_score(a2)
                if np.isfinite(s2) and s2 < best_s - 1e-15:
                    best = a2
                    best_s = s2
    except Exception:
        pass

    return best, best_s


def lbfgs_continuation(a, time_budget, alphas):
    """Continuation method: decrease smoothing temperature."""
    a = normalize_nonneg(a)
    best = a.copy()
    best_s = internal_total_score(best)
    if time_budget <= 5.0 or len(alphas) == 0:
        return best, best_s

    t0 = time.monotonic()
    deadline = t0 + float(time_budget)
    m = len(alphas)
    for idx, alpha in enumerate(alphas):
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        alloc = max(5.0, min(150.0, rem / float(m - idx)))
        c = fast_convolve(best)
        maxc = float(np.max(c)) if c.size else 0.0
        T = max(1e-12, float(alpha) * maxc)
        a_new, s_new = lbfgs_optimize(best, min(alloc, rem - 0.1), T)
        if s_new < best_s - 1e-15:
            best = a_new
            best_s = s_new
    return best, best_s


def optimize_pair_delta(c, B, A, ai, aj):
    """
    Sample and locally refine the best change delta for:
        a[i] += delta, a[j] -= delta, delta in [-ai, aj].
    Convolution update:
        c' = c + 2*delta*B + delta^2*A.
    """
    dmin = -float(ai)
    dmax = float(aj)
    total = dmax - dmin
    current = float(np.max(c)) if c.size else 0.0
    if total <= 1e-15:
        return 0.0, current

    deltas = np.linspace(dmin, dmax, 32)
    fracs = np.array([0.01, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 0.999])
    if ai > 0.0:
        deltas = np.concatenate([deltas, -ai * fracs])
    if aj > 0.0:
        deltas = np.concatenate([deltas, aj * fracs])
    deltas = np.concatenate([deltas, np.array([0.0])])
    deltas = deltas[(deltas >= dmin - 1e-14) & (deltas <= dmax + 1e-14)]
    if deltas.size == 0:
        deltas = np.array([0.0])

    dcol = deltas.reshape(-1, 1)
    scores = (c.reshape(1, -1) + 2.0 * dcol * B.reshape(1, -1)
              + (dcol * dcol) * A.reshape(1, -1)).max(axis=1)
    idx = int(np.argmin(scores))
    best_d = float(deltas[idx])
    best_sc = float(scores[idx])

    h = total / 100.0
    for scale in (1.0, 0.25):
        lo = best_d - scale * h
        hi = best_d + scale * h
        loc = np.linspace(lo, hi, 25)
        loc = loc[(loc >= dmin - 1e-14) & (loc <= dmax + 1e-14)]
        if loc.size <= 1:
            continue
        lcol = loc.reshape(-1, 1)
        loc_scores = (c.reshape(1, -1) + 2.0 * lcol * B.reshape(1, -1)
                      + (lcol * lcol) * A.reshape(1, -1)).max(axis=1)
        j2 = int(np.argmin(loc_scores))
        if float(loc_scores[j2]) < best_sc - 1e-18:
            best_sc = float(loc_scores[j2])
            best_d = float(loc[j2])

    return best_d, best_sc


def pair_arrays(n, a, i, j):
    """Compute linear and quadratic contributions for a pair move."""
    L = 2 * n - 1
    B = np.zeros(L, dtype=np.float64)
    A = np.zeros(L, dtype=np.float64)

    s1 = int(i)
    e1 = min(s1 + int(n), L)
    if e1 > s1:
        B[s1:e1] += a[:e1 - s1]

    s2 = int(j)
    e2 = min(s2 + int(n), L)
    if e2 > s2:
        B[s2:e2] -= a[:e2 - s2]

    A[2 * int(i)] += 1.0
    A[2 * int(j)] += 1.0
    A[int(i) + int(j)] -= 2.0
    return B, A


def pair_polish_gradient(a, time_budget, K_frac=0.05, moves_per_round=150, seed=123):
    """
    Fast coordinate-wise pair-move local search.
    Uses top-peak subgradient to select promising (source, destination) pairs.
    """
    a = normalize_nonneg(a)
    n = int(a.size)
    if n < 2 or time_budget <= 1.0:
        return a, internal_total_score(a)

    c = fast_convolve(a)
    c = np.maximum(c, 0.0)
    current_s = score_from_conv(c, n)
    best = a.copy()
    best_s = current_s

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(time_budget)
    if K_frac is None or K_frac <= 0.0:
        K_frac = 0.05
    K = max(3, int(c.size * float(K_frac)))
    no_improve = 0
    accepted_since_recompute = 0

    while time.monotonic() < deadline:
        if deadline - time.monotonic() < 0.5:
            break

        w = top_weights(c, K)
        g = 4.0 * float(n) * fast_corr(w, a)
        if not np.all(np.isfinite(g)):
            break
        gc = g - float(np.mean(g))

        order = np.argsort(gc)
        pool_size = max(8, int(0.05 * n))
        src = order[::-1][:pool_size]
        dst = order[:pool_size]

        improved_round = False
        pair_idx = 0
        rounds = int(moves_per_round)
        for _ in range(rounds):
            if time.monotonic() >= deadline:
                break

            if len(src) == 0 or len(dst) == 0 or rng.random() < 0.08:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
            else:
                i = int(src[pair_idx % len(src)])
                j = int(dst[(pair_idx // len(src)) % len(dst)])
                pair_idx += 1

            if i == j:
                continue
            ai = float(a[i])
            aj = float(a[j])
            if ai + aj <= 1e-15:
                continue

            B, A = pair_arrays(n, a, i, j)
            delta, trial_s_formula = optimize_pair_delta(c, B, A, ai, aj)
            if not np.isfinite(delta) or not np.isfinite(trial_s_formula):
                continue
            if trial_s_formula >= current_s - 1e-14:
                continue

            new_ai = ai + delta
            new_aj = aj - delta
            eps = 1e-12
            boundary = (new_ai <= eps) or (new_aj <= eps)

            if boundary:
                old_i, old_j = a[i], a[j]
                a[i] = max(0.0, new_ai)
                a[j] = max(0.0, new_aj)
                c_trial = fast_convolve(a)
                c_trial = np.maximum(c_trial, 0.0)
                s_trial = score_from_conv(c_trial, n)
                if np.isfinite(s_trial) and s_trial < current_s - 1e-14:
                    c = c_trial
                    current_s = s_trial
                    improved_round = True
                    no_improve = 0
                    accepted_since_recompute += 1
                    if current_s < best_s - 1e-14:
                        best_s = current_s
                        best = a.copy()
                    if accepted_since_recompute >= 50:
                        c = fast_convolve(a)
                        c = np.maximum(c, 0.0)
                        current_s = score_from_conv(c, n)
                        accepted_since_recompute = 0
                else:
                    a[i] = old_i
                    a[j] = old_j
                    no_improve += 1
            else:
                change = 2.0 * delta * B + (delta * delta) * A
                c_trial = c + change
                c_trial = np.maximum(c_trial, 0.0)
                s_trial = score_from_conv(c_trial, n)
                if np.isfinite(s_trial) and s_trial < current_s - 1e-14:
                    a[i] = new_ai
                    a[j] = new_aj
                    c = c_trial
                    current_s = s_trial
                    improved_round = True
                    no_improve = 0
                    accepted_since_recompute += 1
                    if current_s < best_s - 1e-14:
                        best_s = current_s
                        best = a.copy()
                    if accepted_since_recompute >= 50:
                        c = fast_convolve(a)
                        c = np.maximum(c, 0.0)
                        current_s = score_from_conv(c, n)
                        accepted_since_recompute = 0
                else:
                    no_improve += 1

        if not improved_round:
            no_improve += 500

        if no_improve > 5000:
            a = best.copy()
            z = z_from_a(a) + rng.normal(0.0, 0.01, n)
            a = normalize_nonneg(softmax_from_z(z))
            c = fast_convolve(a)
            c = np.maximum(c, 0.0)
            current_s = score_from_conv(c, n)
            no_improve = 0
            accepted_since_recompute = 0

    best_c = fast_convolve(best)
    best_c = np.maximum(best_c, 0.0)
    best_s = score_from_conv(best_c, n)
    return best, best_s


def mirror_top(a, time_budget, K_fracs=(0.02, 0.1, 0.3), lr=0.05, seed=555):
    """
    Multiplicative weights / mirror descent using top active peaks.
    """
    a = normalize_nonneg(a)
    n = int(a.size)
    if n < 2 or time_budget <= 2.0:
        return a, internal_total_score(a)

    c = fast_convolve(a)
    c = np.maximum(c, 0.0)
    current_s = score_from_conv(c, n)
    best = a.copy()
    best_s = current_s

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(time_budget)

    KF = list(K_fracs)
    if len(KF) == 0:
        KF = [0.1]
    kf_idx = 0
    eta = float(lr)
    no_improve = 0

    while time.monotonic() < deadline:
        if deadline - time.monotonic() < 0.5:
            break

        Kfr = float(KF[kf_idx % len(KF)])
        K = max(3, int(c.size * Kfr))
        w = top_weights(c, K)
        g = 4.0 * float(n) * fast_corr(w, a)
        if not np.all(np.isfinite(g)):
            break
        gc = g - float(np.mean(g))

        if float(np.max(np.abs(gc))) < 1e-18:
            a = best.copy()
            z = z_from_a(a) + rng.normal(0.0, 0.02, n)
            a = normalize_nonneg(softmax_from_z(z))
            c = fast_convolve(a)
            c = np.maximum(c, 0.0)
            current_s = score_from_conv(c, n)
            no_improve = 0
            eta = float(lr)
            kf_idx = (kf_idx + 1) % len(KF)
            continue

        best_trial = None
        best_trial_s = current_s
        for f in (1.0, 0.5, 0.25, 2.0, 0.1, 4.0):
            e = eta * float(f)
            lc = -e * gc
            lc = np.clip(lc, -2.0, 2.0)
            at = a * np.exp(lc)
            at = normalize_nonneg(at)
            sc = internal_total_score(at)
            if np.isfinite(sc) and sc < best_trial_s - 1e-15:
                best_trial_s = sc
                best_trial = at

        if best_trial is not None:
            a = best_trial
            c = fast_convolve(a)
            c = np.maximum(c, 0.0)
            current_s = best_trial_s
            no_improve = 0
            eta = min(1.0, eta * 1.03)
            if current_s < best_s - 1e-14:
                best_s = current_s
                best = a.copy()
        else:
            no_improve += 1
            eta = max(1e-4, eta * 0.8)

        if no_improve > 300:
            a = best.copy()
            z = z_from_a(a) + rng.normal(0.0, 0.02, n)
            a = normalize_nonneg(softmax_from_z(z))
            c = fast_convolve(a)
            c = np.maximum(c, 0.0)
            current_s = score_from_conv(c, n)
            no_improve = 0
            eta = float(lr)
            kf_idx = (kf_idx + 1) % len(KF)

    best_s = internal_total_score(best)
    return best, best_s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    f = np.asarray(f, dtype=np.float64)
    old_n = int(f.size)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    if old_n == new_n:
        return normalize_nonneg(f)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)

    s = float(f.sum())
    if s <= 0.0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    f = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(x_new, x_old, f)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0.0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return out / ss


def random_z(n, rng, style):
    """Generate random logit vector."""
    n = int(n)
    if n <= 0:
        return np.array([0.0])
    i = np.arange(n, dtype=np.float64)
    if style == 0:
        z = np.zeros(n, dtype=np.float64)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 20 + 1)))
            amp = float(rng.uniform(0.5, 2.0))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z += amp * np.cos(2.0 * np.pi * f * i / float(n) + ph)
            z += amp * np.sin(2.0 * np.pi * f * i / float(n) + ph)
        z += rng.normal(0.0, 0.05, n)
        return z
    elif style == 1:
        f = int(rng.integers(1, max(2, n // 10 + 1)))
        amp = float(rng.uniform(1.0, 3.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z = amp * np.cos(2.0 * np.pi * f * i / float(n) + ph)
        z += rng.normal(0.0, 0.05, n)
        return z
    else:
        z = rng.uniform(-1.0, 1.0, n)
        scale = float(rng.uniform(0.5, 3.0))
        z = scale * z + rng.normal(0.0, 0.05, n)
        return z


def random_sequence(n, rng, style):
    """Random normalized nonnegative sequence."""
    z = random_z(n, rng, style)
    return normalize_nonneg(softmax_from_z(z))


def trim_endpoints(a, tol=1e-12):
    """Remove very small endpoints, which can only help when they are zeros."""
    a = normalize_nonneg(a)
    changed = True
    while changed:
        changed = False
        if a.size > 2 and float(a[0]) < tol:
            a = a[1:]
            changed = True
        if a.size > 2 and float(a[-1]) < tol:
            a = a[:-1]
            changed = True
    return normalize_nonneg(a)


def improve_sequence(a, time_budget, mode="main", seed=1000):
    """
    Multi-stage local improvement for one length.
    mode='main' uses more expensive polishing; mode='explore' is shorter.
    """
    a = normalize_nonneg(a)
    best = a.copy()
    best_s = internal_total_score(best)
    if float(time_budget) <= 0.0:
        return best, best_s

    t0 = time.monotonic()
    deadline = t0 + float(time_budget)

    # Stage 1: L-BFGS continuation
    if mode == "main":
        alphas = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]
        lb_cap = 700.0
        lb_frac = 0.40
    else:
        alphas = [10.0, 3.0, 1.0, 0.3]
        lb_cap = 160.0
        lb_frac = 0.45

    rem = deadline - time.monotonic()
    alloc = min(lb_cap, rem * lb_frac)
    if rem > 15.0 and alloc > 5.0:
        a1, s1 = lbfgs_continuation(best, min(alloc, rem - 1.0), alphas)
        if s1 < best_s - 1e-15:
            best = a1
            best_s = s1

    # Stage 2: pair polish focused on top peaks
    rem = deadline - time.monotonic()
    if mode == "main":
        pair_alloc = min(650.0, rem * 0.35)
    else:
        pair_alloc = min(90.0, rem * 0.45)
    if rem > 10.0 and pair_alloc > 4.0:
        a2, s2 = pair_polish_gradient(
            best,
            min(pair_alloc, rem - 1.0),
            K_frac=0.05,
            moves_per_round=160,
            seed=seed + 1,
        )
        if s2 < best_s - 1e-15:
            best = a2
            best_s = s2

    # Stage 3: small-T L-BFGS
    rem = deadline - time.monotonic()
    if rem > 10.0:
        c = fast_convolve(best)
        maxc = float(np.max(c)) if c.size else 0.0
        T = 0.1 * max(1e-12, maxc)
        if mode == "main":
            alloc = min(180.0, rem * 0.20)
        else:
            alloc = min(35.0, rem * 0.30)
        if alloc > 4.0:
            a3, s3 = lbfgs_optimize(best, min(alloc, rem - 1.0), T)
            if s3 < best_s - 1e-15:
                best = a3
                best_s = s3

    # Stage 4: mirror descent
    rem = deadline - time.monotonic()
    if mode == "main":
        alloc = min(140.0, rem * 0.25)
    else:
        alloc = min(35.0, rem * 0.30)
    if rem > 12.0 and alloc > 4.0:
        a4, s4 = mirror_top(
            best,
            min(alloc, rem - 1.0),
            K_fracs=(0.02, 0.1, 0.3),
            lr=0.05,
            seed=seed + 3,
        )
        if s4 < best_s - 1e-15:
            best = a4
            best_s = s4

    # Stage 5: final pair polish
    rem = deadline - time.monotonic()
    if rem > 6.0:
        alloc = min(rem - 1.0, max(5.0, rem * (0.85 if mode == "main" else 0.70)))
        a5, s5 = pair_polish_gradient(
            best,
            alloc,
            K_frac=0.10,
            moves_per_round=200,
            seed=seed + 4,
        )
        if s5 < best_s - 1e-15:
            best = a5
            best_s = s5

    best_s = internal_total_score(best)
    return best, best_s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Search for a sequence of non-negative heights minimizing the evaluator.
    Must run within budget_s and return the best list found.
    """
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 45.0)
    rng_main = np.random.default_rng(seed)

    best = None
    best_score = np.inf

    # Load the parent sequence if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            if float(p.sum()) > 0.0:
                parent = normalize_nonneg(p)
    except Exception:
        parent = None

    if parent is None:
        n0 = 768
        i = np.arange(n0, dtype=np.float64)
        z = (
            0.7 * np.cos(2.0 * np.pi * i / float(n0))
            + 0.3 * np.cos(4.0 * np.pi * i / float(n0))
            + rng_main.normal(0.0, 0.01, n0)
        )
        parent = normalize_nonneg(softmax_from_z(z))

    best = normalize_nonneg(parent)
    best_score = internal_total_score(best)
    es = actual_score(best)
    if np.isfinite(es):
        best_score = min(best_score, es)

    # Main local improvement from the parent / best known sequence.
    rem = deadline - time.monotonic()
    if rem > 40.0:
        alloc = min(950.0, rem * 0.42)
        a, sc = improve_sequence(best, min(alloc, rem - 5.0), "main", seed + 1)
        if sc < best_score - 1e-14:
            best = a
            best_score = sc

    # If enough time remains, explore other lengths and random starts.
    base_n = int(best.size)
    ns = []
    for n in (
        base_n,
        base_n - 1,
        base_n + 1,
        base_n - 2,
        base_n + 2,
        base_n - 4,
        base_n + 4,
        512,
        640,
        768,
        1024,
        1536,
        2048,
        3072,
    ):
        if 256 <= n <= 3072 and n not in ns:
            ns.append(n)
    ns = ns[:5]

    rem = deadline - time.monotonic()
    explore_total = min(480.0, max(0.0, rem * 0.35))
    if explore_total > 100.0 and ns:
        per_n = explore_total / float(len(ns))
        for idx_n, n in enumerate(ns):
            rem = deadline - time.monotonic()
            if rem < 45.0:
                break
            budget_n = min(180.0, per_n, rem * 0.95)
            if budget_n < 40.0:
                continue

            # Start 1: interpolated current best.
            if int(best.size) == n:
                a_start = best.copy()
            else:
                a_start = interpolate_sequence(best, n)
            b1 = min(budget_n * 0.55, rem - 5.0)
            a, sc = improve_sequence(a_start, b1, "explore", seed + 100 + n * 7)
            if sc < best_score - 1e-14:
                best = a
                best_score = sc

            # Start 2: random construction.
            rem = deadline - time.monotonic()
            if rem > 30.0:
                a_start = random_sequence(n, rng_main, idx_n % 3)
                b2 = min(budget_n * 0.45, rem * 0.6)
                a, sc = improve_sequence(a_start, b2, "explore", seed + 200 + n * 7)
                if sc < best_score - 1e-14:
                    best = a
                    best_score = sc

    # Final polish on the best sequence found so far.
    rem = deadline - time.monotonic()
    if rem > 25.0:
        a, sc = improve_sequence(best, min(rem - 5.0, rem * 0.92), "main", seed + 999)
        if sc < best_score - 1e-15:
            best = a
            best_score = sc

    # Trim removable endpoint mass.
    for _ in range(6):
        trimmed = trim_endpoints(best, 1e-12)
        sc = internal_total_score(trimmed)
        if sc < best_score - 1e-12:
            best = trimmed
            best_score = sc
        else:
            break

    # A small final pair polish after trimming, if time remains.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        a, sc = pair_polish_gradient(
            best,
            min(rem - 5.0, rem * 0.7),
            K_frac=0.08,
            moves_per_round=200,
            seed=1234567,
        )
        if sc < best_score - 1e-15:
            best = a
            best_score = sc

    # Final safety checks and normalization.
    best = normalize_nonneg(best)
    best_c = fast_convolve(best)
    best_score = score_from_conv(best_c, int(best.size))
    es = actual_score(best)
    if np.isfinite(es):
        best_score = es

    if not np.isfinite(best_score) or best.size == 0 or not np.all(np.isfinite(best)):
        best = np.ones(768, dtype=np.float64) / 768.0

    return [float(x) for x in best]
```
