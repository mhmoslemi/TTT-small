"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a QP-based alternating method
that replaces the slow LP with a fast projected-gradient QP solver, combined
with Nesterov acceleration, adaptive temperature softmax dual updates,
multiple sequence lengths, and a final soft-max L-BFGS polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def qa_product(w, a):
    """Compute Q_w @ a where Q_w[i,j] = w[i+j]. Uses FFT for speed."""
    n = len(a)
    # result[i] = sum_j w[i+j] * a[j]
    # This is a cross-correlation type operation
    # Use the direct approach with vectorized numpy for moderate n
    if n <= 3000:
        # Build index matrix and use fancy indexing
        idx = np.add.outer(np.arange(n), np.arange(n))
        Q = w[idx]
        return Q @ a
    else:
        # FFT-based fallback
        # result[i] = sum_j w[i+j]*a[j]; define b[t] = w[t], use correlate
        # Use the formula: result = ifft(fft(w_padded) * conj(fft(a_padded)))
        # Actually let's just do the loop for correctness
        result = np.zeros(n)
        for i in range(n):
            result[i] = np.dot(w[i:i + n], a)
        return result


def proj_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_stable(x, beta=1.0):
    """Compute softmax(beta * x) with numerical stability."""
    bx = beta * x
    bx = bx - bx.max()
    e = np.exp(bx)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_of_normalized(a):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def qnesterov_qp(w, a0, n_iters=200, time_limit=None, t0=None):
    """
    Solve min a^T Q_w a subject to sum(a)=1, a>=0 using
    Nesterov-accelerated projected gradient.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a.copy()
    t_k = 1.0

    # Estimate step size: alpha = 1 / (2 * est_lambda_max)
    # Power iteration for lambda_max estimate
    v = a.copy()
    for _ in range(10):
        Qv = qa_product(w, v)
        nv = np.linalg.norm(Qv)
        if nv < 1e-15:
            break
        v = Qv / nv
    lam_est = abs(np.dot(v, qa_product(w, v)))
    alpha = 1.0 / max(2.0 * lam_est, 1e-10)
    alpha = min(alpha, 1.0)  # cap step size

    for k in range(n_iters):
        if time_limit is not None and t0 is not None:
            if time.monotonic() - t0 > time_limit:
                break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)  # keep in domain
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        # Gradient step
        Qy = qa_product(w, y)
        grad = 2.0 * Qy
        a_new = y - alpha * grad
        a_new = proj_simplex(a_new, 1.0)

        a_prev = a
        a = a_new
        t_k = t_new

    return a


def alternating_qp_softmax(a0, time_limit, beta_init=50.0, beta_final=2.0,
                           n_inner=150, n_outer=None, verbose=True):
    """
    Main alternating method:
    - Dual update: w = softmax(beta * conv(a,a) / max)
    - Primal update: solve QP via Nesterov projected gradient
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_of_normalized(a)
    t0 = time.monotonic()

    elapsed_limit = time_limit
    max_outer = int(elapsed_limit / (n_inner * (n * np.log2(n) * 1e-6 + 0.001)))
    if n_outer is None:
        n_outer = max_outer
    n_outer = max(n_outer, 50)

    # Geometric beta schedule
    n_total = n_outer
    beta_schedule = np.geomspace(beta_init, beta_final, n_total)

    for outer in range(n_total):
        if time.monotonic() - t0 > elapsed_limit:
            break

        beta = beta_schedule[outer]

        # Dual update
        b = fast_convolve(a)
        b_max = b.max()
        if b_max < 1e-30:
            break
        w = softmax_stable(b / b_max, beta)

        # Primal update: solve QP
        a_new = qnesterov_qp(w, a, n_iters=n_inner,
                             time_limit=elapsed_limit, t0=t0)

        # Check if improvement
        sc = score_of_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a smaller step: mix with old a
            for t_mix in [0.5, 0.3, 0.1]:
                a_mix = (1 - t_mix) * a + t_mix * a_new
                a_mix = np.maximum(a_mix, 0)
                sm = a_mix.sum()
                if sm > 1e-15:
                    a_mix = a_mix / sm
                    sc_mix = score_of_normalized(a_mix)
                    if sc_mix < best_score - 1e-14:
                        best_score = sc_mix
                        best_a = a_mix.copy()
                        a = a_mix
                        break
            # If no improvement, keep a unchanged but still update for next iter
            # (the alternating might need to explore)

        if verbose and (outer + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AltQP outer={outer+1} ({elapsed:.0f}s): "
                  f"best={best_score:.6f} beta={beta:.1f}")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T_start=10.0, T_end=0.5, maxiter=3000):
    """L-BFGS polish using soft-max smoothed objective."""
    from scipy.optimize import minimize

    n = len(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            s = exp_z.sum()
            if s < 1e-300:
                return 1e15
            a = exp_z / s
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            s = exp_z.sum()
            if s < 1e-300:
                return np.zeros_like(z)
            a = exp_z / s
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # grad_a[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)

    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 30,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a_out = exp_z / exp_z.sum()
    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def mirror_descent_steps(a, n_steps=1000, lr=0.02, time_limit=None, t0=None):
    """Quick mirror descent (exponentiated gradient) on the simplex."""
    n = len(a)
    a = np.maximum(a, 1e-30).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of_normalized(a)
    z = np.log(a)

    for step in range(n_steps):
        if time_limit is not None and t0 is not None:
            if time.monotonic() - t0 > time_limit:
                break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-30:
            break

        # Active set: top 20% of convolution values
        threshold = max_b * 0.8
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent update
        z = z - lr * g
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_new = exp_z / exp_z.sum()

        sc = score_of_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            z = z + lr * g  # undo

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining QP-softmax alternating method,
    mirror descent, and L-BFGS polish across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    print(f"Total budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_parent = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: QP-Softmax on parent length (most time) ===
    print(f"\n=== Phase 1: QP-Softmax n={n_parent} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 800.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_qp, sc_qp = alternating_qp_softmax(
            best_seq, t1, beta_init=80.0, beta_final=2.0,
            n_inner=100, verbose=True)
        try_update(a_qp, "qp_main")

    # === Phase 2: Mirror descent polish ===
    print(f"\n=== Phase 2: Mirror Descent Polish ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 120.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_md, _ = mirror_descent_steps(
            best_seq, n_steps=20000, lr=0.01,
            time_limit=t2, t0=time.monotonic())
        try_update(a_md, "mirror")

    # === Phase 3: L-BFGS soft-max polish ===
    print(f"\n=== Phase 3: L-BFGS Polish ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a_lb = lbgfs_polish(best_seq, t3, T_start=10.0, T_end=0.5, maxiter=4000)
        try_update(a_lb, "lbgfs1")

        # Second polish at lower temperature
        rem = deadline - time.monotonic()
        t3b = min(rem * 0.08, 100.0)
        if t3b > 20:
            a_lb2 = lbgfs_polish(best_seq, t3b, T_start=3.0, T_end=0.3, maxiter=3000)
            try_update(a_lb2, "lbgfs_lowT")

    # === Phase 4: Try different n values ===
    print(f"\n=== Phase 4: Alternative Lengths ===")
    alt_ns = sorted(set([
        int(n_parent * 0.75),
        int(n_parent * 1.25),
        int(n_parent * 1.5),
        int(n_parent * 0.5),
    ]))
    alt_ns = [x for x in alt_ns if 256 <= x <= 2500 and x != n_parent]

    for alt_n in alt_ns[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alt = min(rem * 0.25, 200.0)
        if t_alt < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, sc_alt = alternating_qp_softmax(
            init_alt, t_alt, beta_init=50.0, beta_final=3.0,
            n_inner=80, verbose=False)
        try_update(a_alt, f"alt_n{alt_n}")

        # Quick L-BFGS polish on the best alt
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_pol = min(rem * 0.3, 60.0)
            if t_pol > 15:
                a_pol = lbgfs_polish(best_seq, t_pol, T_start=5.0,
                                     T_end=1.0, maxiter=2000)
                try_update(a_pol, f"lbgfs_alt{alt_n}")

    # === Phase 5: More QP iterations on the best ===
    print(f"\n=== Phase 5: Extended QP on Best ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.5, 300.0)
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
        a_qp2, _ = alternating_qp_softmax(
            best_seq, t5, beta_init=30.0, beta_final=1.0,
            n_inner=150, verbose=True)
        try_update(a_qp2, "qp_extended")

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq) if best_seq is not None else 768
    for restart in range(5):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_r = min(rem * 0.15, 100.0)
        if t_r < 20:
            break

        # Random smooth initialization
        base = interpolate_sequence(best_seq, n_best)
        noise_scale = 0.1 + 0.05 * restart
        noise = rng.normal(0, noise_scale, n_best)
        a_rand = base + noise
        a_rand = np.maximum(a_rand, 0)
        sr = a_rand.sum()
        if sr > 1e-10:
            a_rand = a_rand / sr
            a_r, sc_r = alternating_qp_softmax(
                a_rand, t_r, beta_init=30.0, beta_final=2.0,
                n_inner=80, verbose=False)
            try_update(a_r, f"restart{restart}")

    # === Phase 7: Final L-BFGS + mirror descent ===
    print(f"\n=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_f = min(rem * 0.4, 80.0)
        a_f = lbgfs_polish(best_seq, t_f, T_start=5.0, T_end=0.3, maxiter=2000)
        try_update(a_f, "final_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_fm = min(rem * 0.5, 40.0)
            a_fm, _ = mirror_descent_steps(
                best_seq, n_steps=10000, lr=0.005,
                time_limit=t_fm, t0=time.monotonic())
            try_update(a_fm, "final_mirror")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
