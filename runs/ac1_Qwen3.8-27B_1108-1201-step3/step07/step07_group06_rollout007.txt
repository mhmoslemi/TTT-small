```python
def solve_lp(f, M):
    n = len(f)
    b = cp.Variable(n)
    # Convolution matrix: A[k,j] = f[k-j] if valid
    # (f*b)[k] = sum_j A[k,j] * b[j]
    constraints = [b >= 0]
    for k in range(2*n - 1):
        # sum_j f[k-j] * b[j] <= M where 0 <= k-j <= n-1
        # j ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        coeffs = f[k-je:k-js+1][::-1]  # f[k-je], f[k-je+1], ..., f[k-js]
        constraints.append(np.dot(coeffs, b[js:je+1]) <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    return b.value
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
(iterated linear programming for the autocorrelation peak reduction problem),
combined with gradient-based polishing and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute the objective score for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """
    Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0 <= k-j < n.
    This satisfies: A @ b = f * b (linear convolution).
    """
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # A[k, js:je+1] = f[k-je : k-js+1][::-1]
        A[k, js:je+1] = f[k-je:k-js+1][::-1]
    return A


def solve_lp_for_b(f, M):
    """
    Solve: maximize sum(b) subject to (f*b)[k] <= M for all k, b >= 0.
    Returns the optimal b vector.
    """
    n = len(f)
    b = cp.Variable(n)
    A = build_conv_matrix(f)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CLPK)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK)
            except Exception:
                prob.solve()
    if b.value is None:
        return None
    b_val = np.asarray(b.value).flatten()
    b_val = np.maximum(b_val, 0)
    return b_val


def fictitious_play(a0, time_limit, step_size=0.1, verbose=True):
    """
    Iterated LP-based fictitious play for minimizing max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), get_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    
    M = fast_convolve(a).max()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        t_start = time.monotonic()
        iteration += 1
        
        # Solve LP
        b = solve_lp_for_b(a, M)
        if b is None:
            break
        
        sb = b.sum()
        if sb < 1e-15:
            break
        
        b = b / sb  # Normalize to sum = 1
        
        # Convex combination
        a_new = (1 - step_size) * a + step_size * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        score_new = get_score(a_new)
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FP iter={iteration} score={best_score:.6f} "
                  f"new_score={score_new:.6f} M={M:.6f} "
                  f"elapsed={elapsed:.0f}s")
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        # Always take the step (even if score didn't improve immediately)
        a = a_new
        M = fast_convolve(a).max()
        
        # Check convergence
        if iteration > 10 and M < best_score / (2.0 * n) * 1.0001:
            if verbose:
                print(f"      FP converged at iter={iteration}")
            break
        
        dt = time.monotonic() - t_start
        if dt > time_limit / 20:
            break  # LP is taking too long
    
    return best_a, best_score


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='linear',
                  verbose=False, verbose_interval=2000):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 3000
    last_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        # Gradient
        W = rfft(w, fft_size)
        A_ft = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A_ft), fft_size)[:n]
        grad_a = 2.0 * n * corr
        
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.7

        last_lr = lr * (1.0 - 0.0001)
        lr = max(0.0001, last_lr)

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```
