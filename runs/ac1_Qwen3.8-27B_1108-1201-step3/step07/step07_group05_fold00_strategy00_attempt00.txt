            gradient_raw[i] = impact * (2.0 * T_in - (scores[i] if i < 2*T_in else scores[i - 2*T_in])) / T_in
        
        # Transform gradient to log-space
        gradient_log = z_in * gradient_raw
        
        return gradient_log, log_sum_exp * T_in
    
    # Check feasibility first
    z = np.array([float(x) for x in z])
    if not check_feasibility(z):
        return z, compute_score(softmax_with_min(z))
    
    # Perform gradient descent
    t0 = time.time()
    z_cur = z.copy()
    grad = None
    obj = None
    converged = False
    
    if time_limit > 0:
        # Set a reasonable number of iterations per time minute
        iterations_per_10s = max_iter * 10
        
        while (time.time() - t0 < time_limit) and not converged:
            grad, obj = single_pass(z_cur, T)
            # Primal check: scale by gradient estimate
            if grad is not None:
                z_prev = z_cur.copy()
                z_cur = z_cur - learning_rate * grad  # project to max-exp direction
            
            # Check for convergence (grad is less than a threshold)
            if np.max(np.abs(grad)) < 1e-12 or (time.time() - t0 > time_limit * 0.99):
                converged = True
    
    else:
        converged = True
    
    # Final evaluation ensures we are in terms of the exact objective
    score = 2.0 * len(z_cur) * obj / ( (np.sum(z_cur)) ** 2 )
    return [float(x) for x in z_cur], score
