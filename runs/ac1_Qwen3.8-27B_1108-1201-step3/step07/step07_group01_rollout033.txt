import cvxpy as cp

def lp_refine(a, M, n):
    """Solve LP to find best direction while keeping convolution <= M."""
    b = cp.Variable(n, nonneg=True)
    # Constraint: conv(a, b) <= M for all lags
    # conv(a,b)[l] = sum_j a[j] * b[l-j] (with b out of range = 0)
    constraints = []
    for l in range(2*n - 1):
        # sum_j a[j] * b[l-j] where 0 <= j < n and 0 <= l-j < n
        # i.e., j in [max(0, l-n+1), min(n-1, l)]
        js = max(0, l - n + 1)
        je = min(n - 1, l)
        expr = sum(a[j] * b[l-j] for j in range(js, je+1))
        constraints.append(expr <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS)  # or another solver
    return b.value

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
line search, subgradient multiplicative weights, simulated annealing, and
coordinate descent polishing. Key innovation over previous approach: implements
the LP-based SOTA method (finding optimal direction within the convolution

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max())

def eval_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    """Build C (shape (2n-1, n)) such that C @ b = full conv(a, b)."""
    n = len(a)
    N = 2 * n - 1
    C = np.zeros((N, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def build_conv_matrix_sparse_rows(a, rows):
    """Build C for only specified rows (for active set LP)."""
    n = len(a)
    C = np.zeros((len(rows), n))
    for ii, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            C[ii, j] = a[k - j]
    return C

def lp_direction(a, M, active_rows=None):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[active_rows] <= M."""
    n = len(a)
    if active_rows is not None:
        C = build_conv_matrix_sparse_rows(a, active_rows)
        b_ub = np.full(len(active_rows), M)
    else:
        C = build_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(a_init, time_limit, verbose=True, label=""):
    """LP-based iterative refinement with active set and line search."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0

    for it in range(1, 2000):
        if time.monotonic() - t0 > time_limit:
            break

        b_conv = fast_convolve(a)
        M = b_conv.max()

        # Find active set: top 20% of convolution values
        threshold = np.percentile(b_conv, 80)
        active_mask = b_conv >= threshold
        active_rows = np.where(active_mask)[0]

        # Try active set LP first (faster), fall back to full LP
        b = lp_direction(a, M, active_rows=active_rows)
        if b is None or b.sum() < 1.0 + 1e-6:
            # Try full LP
            b = lp_direction(a, M)
            if b is None or b.sum() < 1.0 + 1e-6:
                stall_count += 1
                if stall_count > 5:
                    break
                # Perturb and retry

                a = a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                continue

        # Line search
        best_t = 0.0
        best_trial = best_score
        for t in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0)
            s2 = a_trial.sum()
            if s2 < 1e-15:
                continue
            a_trial = a_trial / s2
            sc = score_normalized(a_trial)
            if sc < best_trial - 1e-13:
                best_trial = sc
                best_t = t

        if best_t == 0.0:
            stall_count += 1
            if stall_count > 5:
                break
            # Try smaller steps
            for t in [0.001, 0.002, 0.003]:
                a_trial = (1 - t) * a + t * b
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()

                    break

                a = np.maximum(a, 0)
                a = a / a.sum()
                continue

            stall_count = 0
        else:

    return best_a, best_score

def subgradient_optimize(a_init, time_limit, eta_start=0.01, decay=0.999,
                         verbose=True, label=""):
    """Subgradient method with multiplicative updates targeting the peak."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        b_conv = fast_convolve(a)
        k_peak = int(np.argmax(b_conv))

        # Subgradient
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        for i in idx:
            j = k_peak - i
            if j == i:
                grad[i] = a[j]
            else:
                grad[i] = 2.0 * a[j]

        # Project to tangent space
        grad = grad - np.dot(a, grad)

        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        current_score = score_normalized(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        eta *= decay

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SG it={it} ({elapsed:.0f}s) eta={eta:.6f} "
                  f"sc={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score

def simulated_annealing(a_init, time_limit, T_start=0.1, T_end=0.001,
                        verbose=True, label=""):
    """Simulated annealing with targeted perturbations on peak-contributing coords."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    current_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    total_iters_est = 50000
    for it in range(1, total_iters_est + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / total_iters_est
        T = T_start * (T_end / T_start) ** progress

        # Find peak and its contributing coordinates
        b_conv = fast_convolve(a)
        k_peak = int(np.argmax(b_conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        peak_coords = list(range(js, je + 1))

        # Pick 1-3 random coordinates to perturb (bias toward peak-coords)
        n_perturb = rng.integers(1, 4)
        if peak_coords and rng.random() < 0.7:
            chosen = rng.choice(peak_coords, size=min(n_perturb, len(peak_coords)),
                                replace=False)
        else:
            chosen = rng.choice(n, size=n_perturb, replace=False)

        a_trial = a.copy()
        for ci in chosen:
            factor = rng.uniform(0.5, 1.5)
            a_trial[ci] *= factor
        a_trial = np.maximum(a_trial, 0)
        s2 = a_trial.sum()
        if s2 < 1e-15:
            continue
        a_trial = a_trial / s2

        trial_score = score_normalized(a_trial)
        delta = trial_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-10)):
            a = a_trial
            current_score = trial_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SA it={it} ({elapsed:.0f}s) T={T:.5f} "
                  f"sc={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score

def coordinate_descent_polish(a_init, time_limit, verbose=True, label=""):
    """Targeted coordinate descent on peak-contributing coordinates."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    factors = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]

    no_improve = 0
    max_no_improve = 50

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        b_conv = fast_convolve(a)
        k_peak = int(np.argmax(b_conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Sort coordinates by contribution to peak
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        top_order = idx[np.argsort(contribs)[::-1][:10]]

        improved = False
        for i in top_order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            for f in factors:
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                diff = new_val - old_val
                # Redistribute diff to other coords proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    {label} CD ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score

def u_shape_init(n, sharpness=1.0):
    """Create a U-shaped initialization (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    # Bell-shaped dip in the middle
    a = 1.0 / (0.1 + sharpness * (x - 0.5) ** 2)
    # Or: a = 1/(1 + sharpness*exp(-((x-0.5)**2)/width))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def cosine_u_init(n, power=1.0):
    """Cosine-based U-shape."""
    x = np.linspace(0, np.pi, n)
    a = np.abs(np.cos(x / 2)) ** (2 * power)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

def random_u_init(n, seed=42):
    """Random initialization with U-shape bias."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    base = 1.0 / (0.05 + (x - 0.5) ** 2)
    noise = rng.standard_normal(n) * 0.3
    a = base * (1.0 + noise)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp

        a_cd, sc_cd = coordinate_descent_polish(best_seq, t_alloc, label="CD")
        try_update(a_cd, "CD")

    # === Phase 7: Subgradient on best ===
    print(f"\n=== Phase 7: SG Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t_alloc = min(rem * 0.3, 150.0)
        a_sg2, _ = subgradient_optimize(best_seq, t_alloc, eta_start=0.002,
                                        decay=0.9997, label="SG2")
        try_update(a_sg2, "SG2")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        t_alloc = min(rem * 0.4, 150.0)
        a_final, _ = lp_optimize(best_seq, t_alloc, label="LP_final")
        try_update(a_final, "LP_final")

    # === Phase 9: Final CD ===
    rem = deadline - time.monotonic()
    if rem > 10 and best_seq is not None:
        t_alloc = min(rem * 0.5, 60.0)
        a_cd2, _ = coordinate_descent_polish(best_seq, t_alloc, label="CD_final")
        try_update(a_cd2, "CD_final")

    # Fallback
    if best_seq is None:
        best_seq = u_shape_init(n)
        best_score = eval_safe(best_seq)

    # Final verification
    final_score = eval_safe(best_seq)
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
