import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, use_full=True, top_k=None, eps=1e-7):
    """
    Solve: minimize T subject to:
      (f*f)[k] + 2*sum_j g[j]*f[k-j] <= T, for all k (or top-k)
      g >= 0, sum(g) = 1
    Returns (g, T) or (None, None) on failure.
    """
    n = len(f)
    b = fast_convolve(f)
    num_constraints = len(b)

    if use_full:
        indices = np.arange(num_constraints)
    else:
        if top_k is None:
            top_k = 100
        if top_k < num_constraints:
            indices = np.argpartition(b, -top_k)[-top_k:]
        else:
            indices = np.arange(num_constraints)

    K = len(indices)

    # Build constraint matrix: A[k, j] = f[k-j] if valid
    A = np.zeros((K, n))
    rhs = np.zeros(K)
    for row_i in range(K):
        k = indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
        rhs[row_i] = -b[k]

    g = cp.Variable(n)
    T = cp.Variable()
    constraints = [
        g >= 0,
        cp.sum(g) == 1,
        A @ g <= T + rhs  # (f*f)[k] + 2*(f*g)[k] <= T => 2*(f*g)[k] <= T - (f*f)[k]
    ]
    # Actually: (f*f)[k] + 2*(f*g)[k] <= T => 2*(f*g)[k] <= T - (f*f)[k]
    # So: 2*A@g <= T + rhs where rhs = -(f*f)[k]
    # Let me redo: constraint is 2*A@g - T <= rhs
    constraints = [
        g >= 0,
        cp.sum(g) == 1,
        2 * A @ g - T <= rhs
    ]

    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=300000, abs_acc=eps, rel_acc=eps)
        except Exception:
            return None, None

    if g.value is None or T.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s < 1e-10:
        return None, None
    g_val /= s
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, None
    return g_val, float(T.value)


def solve_lp_reduced(f, top_k=50, eps=1e-7):
    """Solve LP with only top-k peak constraints (faster)."""
    return solve_lp_direction(f, use_full=False, top_k=top_k, eps=eps)


def run_lp_optimization(f_init, time_limit, label="lp", use_full=True):
    """
    Main LP-based optimization loop.
    At each step: solve LP for best direction, take a step with line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    top_k = 50

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = fast_convolve(f).max()

        # Solve LP for best direction
        if use_full and n <= 2000:
            g, T = solve_lp_direction(f, use_full=True, eps=1e-7)
        else:
            g, T = solve_lp_reduced(f, top_k=top_k, eps=1e-7)

        if g is None:
            # Try with fewer constraints
            g, T = solve_lp_reduced(f, top_k=max(20, top_k // 2), eps=1e-6)
            if g is None:
                no_improve += 5
                if no_improve > 30:
                    break
                continue

        # Check if improvement is possible
        if T is not None and T >= M - 1e-10:
            no_improve += 1
            top_k = min(top_k * 2, 2 * n - 1)
            if no_improve > 50:
                break
            continue

        # Line search: find best step size
        best_alpha = 0.1
        best_alpha_obj = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = (1.0 - alpha) * f + alpha * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_alpha_obj:
                best_alpha_obj = obj
                best_alpha = alpha

        # Update
        f = (1.0 - best_alpha) * f + best_alpha * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if best_alpha < 0.05:
                top_k = min(top_k * 2, 2 * n - 1)
        else:
            no_improve += 1
            if no_improve > 40:
                break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"t={elapsed:.0f}s stuck={no_improve}")

    return best_f, best_obj


def run_exp_weights(f_init, time_limit, label="ew"):
    """
    Exponential weights (multiplicative) optimization.
    f[i] <- f[i] * exp(-eta * grad[i]), where grad comes from weighted peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = 0.1
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Compute peak weights (softmax over top peaks)
        top_k = min(200, len(b))
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        # Weight peaks by how close they are to the max
        temp = max(0.01 * M, 1e-10)
        w = np.exp((b_top - M) / temp)
        w /= w.sum()

        # Compute gradient: grad[i] = sum_k w[k] * 2*f[k-i]
        grad = np.zeros(n)
        for ii in range(len(idx)):
            k = idx[ii]
            wi = w[ii]
            if wi < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += wi * 2.0 * f[lo:hi + 1][::-1]

        # Normalize gradient
        gmax = grad.max()
        if gmax < 1e-15:
            break
        grad /= gmax

        # Multiplicative update
        log_f = np.log(f + 1e-30)
        log_f -= eta * grad
        # Subtract max for numerical stability
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new /= s

        f = f_new
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Reduce eta and restart
                eta *= 0.8
                no_improve = 0
                if eta < 0.001:
                    break

        if it % 100 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"eta={eta:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements with聪明的 step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_obj = fast_convolve(f).max()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        b = fast_convolve(f)
        M = b.max()

        # Focus on coordinates that contribute to the peak
        peak_idx = b.argmax()
        # Coordinates that affect the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)

        # Try adjusting coordinates near the peak contribution
        for j in range(lo, hi + 1):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            # Try scaling this coordinate
            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 0.3]:
                new_val = f[j] * mult
                if new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True

        if not improved:
            # Try adding mass to zero coordinates
            zero_coords = np.where(f < 1e-12)[0]
            if len(zero_coords) > 0:
                # Try the coordinate most "away" from peak
                rng_local = np.random.default_rng(it)
                for attempt in range(min(10, len(zero_coords))):
                    j = zero_coords[rng_local.integers(len(zero_coords))]
                    f_trial = f.copy()
                    f_trial[j] = 0.01 / n
                    s = f_trial.sum()
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-12:
                        f = f_trial
                        best_obj = obj
                        improved = True
                        break

        if not improved:
            break

        if it % 10 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} round={it} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return f, best_obj


def make_asymmetric_start(n, variant=0, rng=None):
    """Create an asymmetric starting sequence."""
    if rng is None:
        rng = np.random.default_rng(42)

    if variant == 0:
        # Exponential decay from left
        f = np.exp(np.linspace(0, -3, n))
    elif variant == 1:
        # Triangle on first half
        f = np.zeros(n)
        half = n // 2
        f[:half] = np.linspace(1, 0, half)
        f[:half] += 0.5
    elif variant == 2:
        # Gaussian centered at n/4
        x = np.arange(n)
        f = np.exp(-((x - n * 0.25) ** 2) / (2 * (n * 0.15) ** 2))
    elif variant == 3:
        # Cosine bump on first 60%
        f = np.zeros(n)
        m = int(n * 0.6)
        x = np.linspace(0, np.pi, m)
        f[:m] = np.sin(x) ** 2 + 0.1
    elif variant == 4:
        # Random asymmetric
        f = rng.random(n)
        f = f * np.exp(-np.linspace(0, 2, n))
    elif variant == 5:
        # Bell at 1/3 position
        x = np.arange(n)
        f = np.exp(-((x - n / 3.0) ** 2) / (2 * (n * 0.2) ** 2))
    else:
        f = np.ones(n)

    f = np.maximum(f, 0)
    s = f.sum()
    if s > 0:
        f /= s
    return f


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: LP-based direction finding with full constraint set,
    exponential weights polishing, multi-length exploration, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_start(1024, variant=2)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using asymmetric start n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: LP on parent with full constraints ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: Full LP (n={n_parent}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_optimization(parent, time_limit=t1, label="lp_full", use_full=True)
        score1 = 2.0 * len(f1) * obj1
        # Verify with actual evaluator
        score1_eval = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1_eval):
            score1 = score1_eval
        if score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_full]: {score1:.6f}")

    # === Phase 2: LP with reduced constraints (faster iterations) ===
    print(f"\n=== Phase 2: Reduced LP (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, 250.0)
    if t2 > 40:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = run_lp_optimization(best_seq, time_limit=t2, label="lp_red", use_full=False)
        score2_eval = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2_eval) and score2_eval < best_score:
            best_score = score2_eval
            best_seq = f2.copy()
            print(f"  NEW BEST [lp_red]: {score2_eval:.6f}")

    # === Phase 3: Exponential weights on best so far ===
    print(f"\n=== Phase 3: Exp Weights (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = run_exp_weights(best_seq, time_limit=t3, label="ew")
        score3_eval = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3_eval) and score3_eval < best_score:
            best_score = score3_eval
            best_seq = f3.copy()
            print(f"  NEW BEST [ew]: {score3_eval:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = set()
    for factor in [0.5, 0.7, 0.85, 1.1, 1.25, 1.5, 2.0]:
        tl = int(n_base * factor)
        tl = max(256, min(4096, tl))
        target_lengths.add(tl)
    # Also try some "natural" lengths
    for nat_n in [512, 768, 1024, 1536, 2048, 2560, 3072]:
        if 256 <= nat_n <= 4096:
            target_lengths.add(nat_n)
    target_lengths.discard(n_base)
    # Sort by proximity to base length
    target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_base))

    explored = 0
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if explored >= 4:
            break
        t4 = min(rem * 0.15, 120.0)
        if t4 < 25:
            break
        explored += 1

        # Interpolate from best_seq
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = make_asymmetric_start(target_n, variant=2)

        print(f"  LP n={target_n} (interp), budget={t4:.0f}s")
        f4, obj4 = run_lp_optimization(f_interp, time_limit=t4,
                                       label=f"lp_n{target_n}", use_full=(target_n <= 1500))
