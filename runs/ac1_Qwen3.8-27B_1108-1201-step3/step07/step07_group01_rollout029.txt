def lp_direction(f, M, time_limit):
    """Find LP descent direction: max sum(b) s.t. b >= 0, f*b <= M."""
    n = len(f)
    # Constraints: for each k in 0..2n-2, sum_i f[i] * b[k-i] <= M
    # where b[k-i] is 0 if k-i is outside [0, n-1]
    # This means: for each k, the constraint involves b[max(0,k-n+1)] to b[min(n-1,k)]
    
    # Build constraint matrix A_ub (each row is a constraint)
    # A_ub @ b_vec <= M * ones(2n-1)
    
    # For efficiency, build the matrix as a sparse-like structure
    # Actually for n=768, a dense 1535x768 matrix is fine (~4MB)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, res.fun * -1  # return (b, sum(b))
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided descent combined with
adaptive mirror descent. Key novelty: uses linear programming to find optimal
descent directions (maximizing sum(b) s.t. f*b <= M), combined with multiplicative
updates between LP calls for efficiency. Multiple starting points and adaptive
temperature schedules improve robustness.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_lp_matrix(f):
    """Build constraint matrix for LP: rows are lags, columns are b indices."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. f*b <= M, b >= 0."""
    n = len(f)
    A_ub = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_guided_descent(f_init, time_limit, lp_interval=5, verbose=True, label=""):
    """LP-guided descent with mirror steps between LP calls."""
    n = len(f_init)
    a = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    eta = 0.01
    T = 0.01
    it = 0
    lp_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Every lp_interval iterations, do an LP step
        if it % lp_interval == 1:
            b = fast_convolve(a)
            M = b.max()
            
            # Try LP
            b_lp = lp_solve(a, M)
            if b_lp is not None and b_lp.sum() > 1.0 + 1e-6:
                g = b_lp / b_lp.sum()
                # Update
                for t in [0.3, 0.2, 0.1, 0.05]:
                    a_trial = (1 - t) * a + t * g
                    a_trial = np.maximum(a_trial, 1e-15)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break
            lp_count += 1
            if verbose and lp_count % 10 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} LP {lp_count} ({elapsed:.0f}s): score={best_score:.7f}")
        
        # Between LPs, do mirror descent step
        b = fast_convolve(a)
        M = b.max()
        current_score = 2.0 * n * M
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
        
        # Softmax gradient
        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()
        
        # Gradient via cross-correlation
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]
        
        # Mirror update
        mean_grad = np.dot(a, grad_a)
        log_update = -eta * (grad_a - mean_grad)
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()
        
        # Adaptive step size
        if current_score > best_score * (1 + 1e-4):
            eta *= 0.8
        elif it % 100 == 0:
            eta = min(eta * 1.05, 0.05)
        
        # Temperature annealing
        progress = min(1.0, it / 5000)
        T = 0.01 * (0.001 / 0.01) ** progress
        
    return best_a, best_score

            a = best_a.copy()

def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(20000):
        if time.monotonic() - t0 > time_limit or no_improve > 100:
            break
        
        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        
        if b[peak_k] <= best_score / (2.0 * n) + 1e-16:
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]
        
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue

def lbgfs_refine(a_init, time_limit, T_list=None, maxiter=3000):
    """L-BFGS refinement in log-space with multiple temperatures."""
    n = len(a_init)
    if T_list is None:
        T_list = [100, 50, 20, 10, 5, 2, 1]
    
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        
        z0 = np.log(a)
        z0 = z0 - z0.mean()
        call_count = [0]
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z):
            call_count[0] += 1
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_z = softmax_from_z(z)
            b = fast_convolve(a_z)
            scaled = b / T
            m = scaled.max()
            lse = m + np.log(np.exp(scaled - m).sum())
            return 2.0 * n * T * lse
        
        def jac(z):
            call_count[0] += 1
            if time.monotonic() - t0 > time_limit:
                return np.zeros(n)
            a_z = softmax_from_z(z)
            b = fast_convolve(a_z)
            scaled = b / T
            m = scaled.max()
            w = np.exp(scaled - m)
            w = w / w.sum()
            fft_size = 1 << (3 * n - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a_z, fft_size)
            correl = irfft(np.conj(W) * A, fft_size)
            grad_a = 4.0 * n * correl[:n]
            mean_grad = np.dot(a_z, grad_a)
            return a_z * (grad_a - mean_grad)
        
        try:
            res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': maxiter * 20, 'disp': False})
            z_opt = res.x
            a_new = softmax_from_z(z_opt)
            sc = score_from_a(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
        except Exception:
            pass
        
        print(f"    L-BFGS T={T}: score={best_score:.7f}")
    
    return best_a, best_score


def softmax_from_z(z):
    z = z - z.max()
    exp_z = np.exp(z)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def chirp_sequence(n):
    """Generate a chirp-like sequence."""
    x = np.arange(n) / n
    # Linear chirp in frequency domain, then take magnitude
    freq = np.fft.rfftfreq(n)
    phase = np.pi * (freq * n) ** 2
    a = np.abs(np.fft.irfft(np.exp(1j * phase), n))
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

    n = len(f)
    b_ff = fast_convolve(f)
    peak_indices = np.argsort(b_ff)[::-1][:top_k]

    A_ub = np.zeros((top_k, n))
    for ii, k in enumerate(peak_indices):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        cols = np.arange(js, je + 1)
        A_ub[ii, cols] = f[k_int - cols]

    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_budget, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_guided_descent(f_init, time_limit, lp_interval=2, top_k=30,
                      verbose=True, label=""):
    """LP-guided descent with mirror steps between LP calls."""
    n = len(f_init)
    a = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    eta = 0.01
    T = 0.01
    it = 0
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it % lp_interval == 1:
            b = fast_convolve(a)
            M = b.max()

            # Try LP with progressively more constraints if needed
            b_lp = lp_solve_subset(a, M, top_k=top_k, time_budget=3.0)
            if b_lp is None:
                b_lp = lp_solve_subset(a, M, top_k=15, time_budget=3.0)

            if b_lp is not None and b_lp.sum() > 1.0 + 1e-5:
                g = b_lp / b_lp.sum()
                # Line search for step size
                best_t = 0
                best_sc = best_score
                for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
                    a_trial = (1 - t) * a + t * g
                    a_trial = np.maximum(a_trial, 1e-15)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_t = t
                if best_t > 0:
                    a = (1 - best_t) * a + best_t * g
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    if best_sc < best_score:
                        best_score = best_sc
                        best_a = a.copy()
            lp_count += 1
            if verbose and lp_count % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} LP {lp_count} ({elapsed:.0f}s): score={best_score:.7f}")

        # Mirror descent step
        b = fast_convolve(a)
        M = b.max()
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Adaptive temperature
        progress = min(1.0, it / 8000)
        T = 0.015 * (0.0005 / 0.015) ** progress

        # Softmax weights
        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        # Gradient via cross-correlation
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        # Mirror update
        mean_grad = np.dot(a, grad_a)
        log_update = -eta * (grad_a - mean_grad)
        log_update = np.clip(log_update, -2.5, 2.5)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()

        # Adaptive step size
        if current_score > best_score * (1 + 1e-3):
            eta *= 0.7
        elif it % 200 == 0:
            eta = min(eta * 1.08, 0.06)

    return best_a, best_score


def mirror_descent_only(a_init, time_limit, T_start=0.01, T_end=0.0005,
                        eta_start=0.02, eta_end=0.001, max_iters=50000,
                        verbose=True, label=""):
    """Pure mirror descent with temperature annealing and restarts."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        mean_grad = np.dot(a, grad_a)
        log_update = -eta * (grad_a - mean_grad)
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()

        if stuck > 400:
            rng = np.random.default_rng(it * 104729 + 7)
            noise = rng.standard_normal(n) * 0.04
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            eta = min(eta * 2.0, eta_start)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD {it} ({elapsed:.0f}s): T={T:.5f} eta={eta:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def multiplicative_weights_exact(a_init, time_limit, eta_start=0.008,
                                 decay=0.9998, max_iters=50000,
                                 verbose=True, label=""):
    """Multiplicative weights using exact peak subgradient."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Find top-3 peaks and average their subgradients
        peak_idx = np.argsort(b)[::-1][:3]
        grad = np.zeros(n)
        for pk in peak_idx:
            k = int(pk)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= 3.0

        # Project to tangent space
        grad = grad - np.dot(a, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()

        eta *= decay

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MW {it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def lbgfs_refine(a_init, time_limit, T_list=None, maxiter=2000):
    """L-BFGS refinement in log-space with multiple temperatures."""
    n = len(a_init)
    if T_list is None:
        T_list = [50, 20, 10, 5, 2, 1]

    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for T in T_list:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        _lb_state['n'] = n
        _lb_state['T'] = T
        _lb_state['t0'] = t0
        _lb_state['time_limit'] = time_limit

        z0 = np.log(a)
        z0 = z0 - z0.mean()

        try:
            res = minimize(fun=_lb_func, x0=z0, jac=_lb_jac,
                          method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': maxiter * 15,
                                   'disp': False})
            z_opt = res.x
            a_new = softmax_from_z(z_opt)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc = score_from_a(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            print(f"    L-BFGS T={T}: score={best_score:.7f}")
        except Exception as e:
            print(f"    L-BFGS T={T}: failed ({e})")

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving: reduce contributions to top peaks."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(30000):
        if time.monotonic() - t0 > time_limit or no_improve > 150:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        if b[peak_k] <= best_score / (2.0 * n) * 1.001:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pairwise reduction
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1, f2 in [(0.9, 0.9), (0.8, 0.8), (0.85, 0.75)]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.9, 1.1, 1.25, 1.4]
    no_improve = 0
    max_no_improve = 250

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            no_improve += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-guided descent (primary, uses subset constraints for speed)
    2. Mirror descent with adaptive temperature
    3. Multiplicative weights with top-3 peak subgradient
    4. L-BFGS in log-space
    5. Peak shaving and coordinate descent for final polish
    6. Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n=768, score={best_score:.7f}")

    n_best = len(best_seq)

    # === Phase 1: LP-Guided Descent (main driver) ===
    print(f"\n=== Phase 1: LP-Guided Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t_alloc = min(rem * 0.3, 240.0)
        print(f"  LP-GD: n={n_best}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_guided_descent(best_seq, t_alloc, lp_interval=2, top_k=30,
                                     label="LP-GD")
        best_seq, best_score = check_and_update(a_lp, "LP-GD", best_seq, best_score)

    # === Phase 2: Mirror Descent configs ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    md_configs = [
        (0.015, 0.001, 0.025, 0.001, "MD_mid"),
        (0.005, 0.0003, 0.012, 0.0005, "MD_low"),
        (0.03, 0.002, 0.04, 0.002, "MD_high"),
    ]
    for T_s, T_e, eta_s, eta_e, lbl in md_configs:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 100.0)
        print(f"  {lbl}: budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_only(best_seq, t_alloc, T_start=T_s, T_end=T_e,
                                       eta_start=eta_s, eta_end=eta_e, label=lbl)
        best_seq, best_score = check_and_update(a_md, lbl, best_seq, best_score)

    # === Phase 3: LP-GD Round 2 ===
    print(f"\n=== Phase 3: LP-GD Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 40:
        t_alloc = min(rem * 0.3, 120.0)
        print(f"  LP-GD2: budget={t_alloc:.0f}s")
        a_lp2, _ = lp_guided_descent(best_seq, t_alloc, lp_interval=2, top_k=40,
                                      label="LP-GD2")
        best_seq, best_score = check_and_update(a_lp2, "LP-GD2", best_seq, best_score)

    # === Phase 4: Multiplicative Weights ===
    print(f"\n=== Phase 4: Multiplicative Weights (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.3, 90.0)
        print(f"  MW: budget={t_alloc:.0f}s")
        a_mw, _ = multiplicative_weights_exact(best_seq, t_alloc,
                                                eta_start=0.006, decay=0.9998,
                                                label="MW")
        best_seq, best_score = check_and_update(a_mw, "MW", best_seq, best_score)

    # === Phase 5: L-BFGS ===
    print(f"\n=== Phase 5: L-BFGS (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_alloc = min(rem * 0.45, 100.0)
        print(f"  L-BFGS: budget={t_alloc:.0f}s")

                               maxiter=2000)

    # === Phase 6: Peak Shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.7f}) ===")

    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.7f}) ===")

    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")

    rem = deadline - time.monotonic() - 6
    if rem > 4:
        a_cd2, _ = coordinate_descent(best_seq, min(rem * 0.6, 8.0), label="final_CD")
        best_seq, best_score = check_and_update(a_cd2, "final_CD", best_seq, best_score)

    # === Phase 9: Alternative lengths ===
    print(f"\n=== Phase 9: Alt Lengths (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 30 and best_seq is not None:
        n_best = len(best_seq)
        for factor in [1.25, 0.8, 1.5]:
            if rem < 15:
                break
            alt_n = int(n_best * factor)
            if alt_n < 256 or alt_n > 3000 or alt_n == n_best:
                continue
            t_alloc = min(rem * 0.25, 25.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = lp_guided_descent(init_alt, t_alloc, lp_interval=3,
                                          top_k=20, label=f"alt_n{alt_n}")
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}",
                                                     best_seq, best_score)
            rem = deadline - time.monotonic() - 10

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
