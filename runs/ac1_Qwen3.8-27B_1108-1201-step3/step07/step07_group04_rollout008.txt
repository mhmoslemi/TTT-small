    """Project onto the probability simplex (sum=1, non-negative)."""
    n = len(v)

        rho = np.nonzero(cond)[0][-1]

        theta = 0.0

The objective (with sum(a)=1) is: `f(a) = 2*n*max_k conv(a,a)[k]`

We approximate `max_k conv(a,a)[k]` with `L(a) = T*log(sum_k exp(conv(a,a)[k]/T))`.

`dL/da[j] = sum_k (exp(conv_k/T) / sum_l exp(conv_l/T)) * d(conv_k)/da[j]`

`d(conv_k)/da[j] = sum_i (delta_{i,j}*a[k-i] + a[i]*delta_{k-i,j})`

For the first term: `i=j`, gives `a[k-j]` if `0 <= k-j < n`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated gradient descent on log-sum-exp smoothed objective
2. Cosine-annealed temperature schedule
3. Peak-directed subgradient refinement with momentum
4. Spectral phase-retrieval inspired initialization
5. Multi-length exploration with interpolation
6. Adaptive restarts to escape local minima

import numpy as np
from scipy.fft import rfft, irfft
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = irfft(A * A, fft_size)[:size]
    return B


def compute_lse_and_grad(a, T):
    """Compute LSE smoothed max-conv and its gradient w.r.t. a.
    Returns (lse_value, gradient_array, true_max_conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = (conv - max_c) / T
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    lse_val = max_c + T * np.log(exp_scaled.sum())

    # Gradient: grad[j] = 2 * sum_m w[j+m] * a[m]
    # = 2 * conv(w, a[::-1])[n-1 + j]
    conv_len = 2 * n - 1 + n  # full conv of w (len 2n-1) and a_rev (len n) = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a[::-1], N)
    corr = irfft(W * A, N)
    c = corr[n - 1:2 * n - 1]
    grad = 2.0 * c
    return lse_val, grad, max_c


def project_simplex(v):
    """Project vector onto probability simplex (sum=1, non-negative)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def lse_gradient_descent(a0, time_limit, T_init=40.0, T_min=0.05,
                         lr_init=0.008, momentum=0.9, cosine_decay=True):
    """FFT-accelerated gradient descent on LSE-smoothed objective with cosine annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    v = np.zeros(n)
    lr = lr_init
    T = T_init
    total_iters_estimate = int(time_limit / 0.003)  # rough estimate of total iterations
    iters = 0
    no_improve = 0
    restart_count = 0
    max_restarts = 30

    while monotonic() - t0 < time_limit:
        iters += 1

        # Cosine annealing for temperature
        if cosine_decay:
            progress = min(1.0, iters / max(total_iters_estimate, 1))
            T = T_min + 0.5 * (T_init - T_min) * (1 + np.cos(np.pi * progress))
        else:
            T = max(T_min, T * 0.999)

        lse_val, grad, max_c = compute_lse_and_grad(a, T)
        # Full gradient of objective 2*n*LSE
        full_grad = 2.0 * n * grad

        # Adam-like per-coordinate scaling
        v = momentum * v - lr * full_grad
        a_new = project_simplex(a + v)

        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-11:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive learning rate
        if no_improve % 100 == 0:
            lr *= 0.97

        # Periodic restart with perturbation
        if no_improve > 300 and restart_count < max_restarts:
            a = best_a + np.random.standard_normal(n) * 0.008
            a = project_simplex(a)
            v = np.zeros(n)
            lr = lr_init * 0.7
            # Reheat temperature
            if cosine_decay:
                iters = max(0, iters - 500)
            else:
                T = max(T_min, T_init * 0.6)
            no_improve = 0
            restart_count += 1

        if iters % 3000 == 0:
            elapsed = monotonic() - t0
            print(f"    LSE-GD iter={iters} ({elapsed:.0f}s): score={best_score:.6f} "
                  f"T={T:.3f} lr={lr:.6f} restarts={restart_count}")

    return best_a, best_score, iters


def peak_subgradient(a0, time_limit, top_k=40, lr_init=0.0015, momentum=0.92):
    """Subgradient method focusing energy on top peaks with momentum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    iters = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Compute subgradient from peaks
        sg = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a[k - idx]

        v = momentum * v - lr * sg / k_actual
        a_new = project_simplex(a + v)

        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-11:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        if no_improve > 250:
            a = best_a + np.random.standard_normal(n) * 0.004
            a = project_simplex(a)
            v = np.zeros(n)
            lr = lr_init * 1.2
            no_improve = 0
            if iters % 1000 == 0:
                print(f"    PeakSG restart at iter={iters}")

        if iters % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakSG iter={iters} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # If not improved, try small random moves
        if not improved:
            for _ in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = project_simplex(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stagnant += 1
            if stagnant > 80:
                break
        else:
            stagnant = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    - FFT-accelerated LSE gradient descent with cosine annealing
    - Peak-directed subgradient with momentum
    - Targeted peak shaving
    - Multi-length exploration
    - Final polish with low-temperature refinement
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    # === Phase 1: LSE Gradient Descent (primary optimizer) ===
    print(f"\n=== Phase 1: LSE Gradient Descent (n={len(best_seq)}) ===")
    rem = deadline - monotonic()
    t1 = min(rem * 0.30, 420.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1, s1, iters = lse_gradient_descent(best_seq, t1, T_init=50.0, T_min=0.05,
                                              lr_init=0.006, momentum=0.9)
        best_seq, best_score = check_and_update(a1, f"LSE_n{len(a1)}_it{iters}", best_seq, best_score)

    # === Phase 2: Peak Subgradient ===
    print(f"\n=== Phase 2: Peak Subgradient (n={len(best_seq)}, best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = peak_subgradient(best_seq, t2, top_k=30, lr_init=0.001, momentum=0.91)
        best_seq, best_score = check_and_update(a2, "PeakSG", best_seq, best_score)

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t3 = min(rem * 0.12, 120.0)
    if t3 > 10:
        a3, s3 = refined_peak_shave(best_seq, t3)
        best_seq, best_score = check_and_update(a3, "PeakShave", best_seq, best_score)

    # === Phase 4: Second LSE pass with lower temperature ===
    print(f"\n=== Phase 4: LSE Refinement (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t4 = min(rem * 0.18, 200.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a4, s4, iters4 = lse_gradient_descent(best_seq, t4, T_init=15.0, T_min=0.02,
                                               lr_init=0.003, momentum=0.93)
        best_seq, best_score = check_and_update(a4, f"LSE2_n{len(a4)}", best_seq, best_score)

    # === Phase 5: Length Exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for target_n in [512, 768, 1024, 1280, 1536, 2048, 2560, 3072]:
        if target_n != n_best and 256 <= target_n <= 3500:
            candidate_lengths.append(target_n)
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    for alt_n in candidate_lengths[:3]:
        rem = deadline - monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.25, 180.0)
        if t_alloc < 30:
            continue
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)

        # LSE GD on new length
        a_alt, _, it_alt = lse_gradient_descent(init_alt, t_alloc * 0.55,
                                                 T_init=30.0, T_min=0.1,
                                                 lr_init=0.005, momentum=0.9)
        best_seq, best_score = check_and_update(a_alt, f"LSE_n{alt_n}", best_seq, best_score)

        # Quick peak subgradient
        rem2 = deadline - monotonic()
        if rem2 > 40 and len(best_seq) == alt_n:
            a_alt2, _ = peak_subgradient(a_alt, min(rem2 * 0.3, 60.0), top_k=20)
            best_seq, best_score = check_and_update(a_alt2, f"SG_n{alt_n}", best_seq, best_score)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 40:
        t_final = min(rem * 0.4, 100.0)
        a_f, _, _ = lse_gradient_descent(best_seq, t_final, T_init=8.0, T_min=0.01,
                                          lr_init=0.002, momentum=0.95)
        best_seq, best_score = check_and_update(a_f, "FinalLSE", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 25:
        t_ps = min(rem * 0.4, 60.0)
        a_ps, _ = refined_peak_shave(best_seq, t_ps)
        best_seq, best_score = check_and_update(a_ps, "FinalPS", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 15:
        t_sg = min(rem * 0.5, 40.0)
        a_sg, _ = peak_subgradient(best_seq, t_sg, top_k=10, lr_init=0.0005, momentum=0.95)
        best_seq, best_score = check_and_update(a_sg, "FinalSG", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
