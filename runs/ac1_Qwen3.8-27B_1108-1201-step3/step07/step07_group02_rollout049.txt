def lbgfs_smoothed(a0, time_limit, T_start=100, T_end=1, maxiter=5000):

def interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n: return f.copy()
    s = f.sum()
    if s < 1e-15: return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15: fi /= ss
    return fi
