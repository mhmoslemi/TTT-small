"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth approximation with L-BFGS (softmax parameterization)
2. Simulated annealing over temperature tau
3. LP-based fictitious play for initialization and refinement
4. Coordinate descent polishing
5. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * b.max()

def lse_and_grad(a, tau):
    """Compute log-sum-exp of conv(a,a)/tau and its gradient w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    shifted = (b - b_max) / tau
    # For numerical stability
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    lse = b_max + tau * (max_shifted + np.log(exp_shifted.sum()))
    weights = exp_shifted / exp_shifted.sum()
    
    # Gradient: dL/da_m = (2/tau) * sum_k weights[k] * a[k-m] (with bounds)
    # = (2/tau) * correlate(weights, a, mode='valid')[m]
    grad = (2.0 / tau) * np.correlate(weights, a, mode='valid')
    return lse, grad

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def lse_softmax_obj(x, tau):
    """Objective and gradient for optimization over x where a = softmax(x)."""
    a = softmax(x)
    val, grad_a = lse_and_grad(a, tau)
    # Gradient through softmax: d/dx = a * (grad_a - a @ grad_a)
    grad_x = a * (grad_a - a @ grad_a)
    return val, grad_x

def lse_lbfgs(a0, tau, max_iter=500, time_limit=60.0):
    """Optimize using L-BFGS-B with softmax parameterization."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    
    t0 = time.monotonic()
    
    def fun(x):
        if time.monotonic() - t0 > time_limit:
            return np.inf, np.zeros(len(x))
        return lse_softmax_obj(x, tau)
    
    res = minimize(fun, x0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'maxfun': 2000, 'gtol': 1e-8})
    
    x_opt = res.x
    a_opt = softmax(x_opt)
    a_opt = a_opt / a_opt.sum()
    return a_opt, fast_score(a_opt)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():

    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:

                _lp_solver = solver_name

        except Exception:

    _lp_solver = cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:

            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = fast_score(a)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = solve_lp_best_response(a, solver)

        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)

            cn = cand / s
            sc = fast_score(cn)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth approximation optimized with L-BFGS (softmax parameterization)
2. Continuation method with decreasing temperature tau
3. LP-based fictitious play for diversification
4. Efficient coordinate descent with binary search for final polish
5. Random restarts and multi-length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly assuming a>=0."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * b.max()

def lse_and_grad(a, tau):
    """Compute log-sum-exp of conv(a,a)/tau and gradient w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    shifted = (b - b_max) / tau
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    exp_sum = exp_shifted.sum()
    lse = b_max + tau * (max_shifted + np.log(exp_sum))
    weights = exp_shifted / exp_sum
    grad = (2.0 / tau) * np.correlate(weights, a, mode='valid')
    return lse, grad

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def lse_softmax_obj(x, tau):
    """Objective and gradient for optimization over x where a = softmax(x)."""
    a = softmax(x)
    val, grad_a = lse_and_grad(a, tau)
    grad_x = a * (grad_a - a @ grad_a)
    return val, grad_x

def lse_lbfgs(a0, tau, max_iter=1000, time_limit=60.0):
    """Optimize using L-BFGS-B with softmax parameterization."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    t0 = time.monotonic()
    stop_flag = [False]

    def fun(x):
        if time.monotonic() - t0 > time_limit:
            stop_flag[0] = True
            return 0.0, np.zeros(len(x))
        return lse_softmax_obj(x, tau)

    res = minimize(fun, x0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'maxfun': 5000, 'gtol': 1e-10,
                            'factr': 1e4})
    a_opt = softmax(res.x)
    a_opt = a_opt / a_opt.sum()
    return a_opt, fast_score(a_opt)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

        a = a / s if s > 1e-15 else np.ones(n) / n

    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = fast_score(a)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = solve_lp_best_response(a, solver)

            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:

        a_avg = (iters * a_avg + a) / (iters + 1)

                a[i] = new_val
                a = a / a.sum()
                sc = fast_score(a)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    any_imp = True
                    break
                else:
                    a[i] = old
                    a = a / a.sum()
            if not any_imp and sweep > 30:
                break
    return best_a, best_score

def interp_seq(f, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s if s > 1e-15 else np.ones(old_n) / old_n
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    return f_interp / ss if ss > 1e-15 else np.ones(new_n) / new_n

def adam_optimize(a0, tau, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the softmax-parameterized log-sum-exp objective."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    x = np.log(a0 / a0.sum())
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = (a0 / a0.sum()).copy()
    best_score = fast_score(best_a)
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax(x)
        val, grad_x = lse_softmax_obj(x, tau)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = softmax(x)
        sc = fast_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        lr_t = lr * (0.999 ** (t // 500))
    return best_a, best_score
