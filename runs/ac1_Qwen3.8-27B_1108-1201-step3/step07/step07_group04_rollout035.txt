    A = rfft(a, fft_size)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
with log-sum-exp smoothing in log-space (L-BFGS-B), mirror descent,
and local peak shaving. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_obj_grad(z, T):
    """
    Compute smooth objective and gradient in log-space.
    a = softmax(z), minimize 2n * LSE_T(conv(a,a))
    Returns (objective, gradient_wrt_z)
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_b = np.exp(scaled)
    lse_sum = exp_b.sum()
    LSE = b_max + T * np.log(lse_sum)
    w = exp_b / lse_sum
    
    # c[m] = sum_j a[j] * w[j+m], computed via FFT
    a_rev = a[::-1].copy()
    conv_len = n + (2*n - 1) - 1
    fft_c = 1 << (conv_len - 1).bit_length()
    Ar = rfft(a_rev, fft_c)
    W = rfft(w, fft_c)
    c_full = irfft(Ar * W, fft_c)[:conv_len]
    c = c_full[n-1 : 2*n-1]
    
    # grad_a[m] = 2*c[m] - a[m]*w[2m] (if 2m < 2n-1)
    grad_a = 2.0 * c
    # Correction for diagonal
    diag_indices = np.arange(0, 2*n-1, 2)  # 0, 2, 4, ..., 2n-2
    diag_a = a[diag_indices // 2]
    diag_w = w[diag_indices]
    grad_a[diag_indices // 2] -= diag_a * diag_w
    
    obj = 2.0 * n * LSE
    
    # Transform to z-space
    ga_dot_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * ga_dot_a)
    grad_z *= 2.0 * n
    
    return obj, grad_z


def mirror_descent(a0, time_budget, n_iters=None):
    """Mirror descent (multiplicative weight) for the autocorrelation minimization."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = monotonic()
    lr = 0.1
    
    for it in range(n_iters if n_iters else 100000):
        if monotonic() - t0 > time_budget:
            break
        if (it + 1) % 5000 == 0:
            lr = max(0.01, 0.1 / np.sqrt((it+1)/5000.0 + 1))
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Subgradient: grad_a[m] = 2*a[peak_k - m] (if valid) - a[m] (if 2m==peak_k)
        grad_a = np.zeros(n)
        for m in range(n):
            j = peak_k - m
            if 0 <= j < n:
                grad_a[m] = 2.0 * a[j]
            if 2 * m == peak_k:
                grad_a[m] -= a[m]
        
        # Mirror descent update: a_new[m] proportional to a[m] * exp(-lr * grad_a[m])
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad_a
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update if worse (or use averaging)
            pass
    
    return best_a, best_score


def lbgfs_optimize(a0, time_budget, T_start=50.0, T_end=1.0, n_temps=5):
    """L-BFGS optimization in log-space with decreasing temperature."""
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    t0 = monotonic()
    temps = np.geomspace(T_start, T_end, n_temps)
    
    z = np.log(np.maximum(a0, 1e-30))
    z = z - z.mean()
    
    for T in temps:
        if monotonic() - t0 > time_budget - 5:
            break
        
        def func(z_input):
            obj, _ = compute_obj_grad(z_input, T)
            return obj
        
        def jac(z_input):
            _, g = compute_obj_grad(z_input, T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 20000, 'disp': False}
            )
            if not res.success and res.status != 0:
                pass  # continue anyway
            
            z = res.x
            z_shifted = z - z.max()
            a_new = np.exp(z_shifted)
            a_new = a_new / a_new.sum()
            
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
            break
    
    return best_a, best_score


def peak_shave_local(a0, time_budget):

    t0 = monotonic()
    no_improve = 0

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        # Strategy: reduce the top contributor
        top_i = int(idx[np.argmax(contribs)])
        reduced = False

            a_trial = a.copy()
            diff = a_trial[top_i] * (1 - factor)
            a_trial[top_i] *= factor
            
            # Redistribute to other elements (those not contributing to peak)
            # Or just redistribute proportionally to all others
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others_sum = a_trial[mask].sum()
            if others_sum > 1e-15:
                a_trial[mask] += diff * a_trial[mask] / others_sum
            
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()

                best_score = score
                best_a = a.copy()
                reduced = True
                break
        
        if not reduced:

            for ci in top2:
                i1 = int(idx[ci])
                i2 = peak_k - i1
                if i1 == i2 or i2 < 0 or i2 >= n:
                    continue
                for f1 in [0.9, 0.85]:
                    for f2 in [0.9, 0.85]:
                        a_trial = a.copy()
                        a_trial[i1] *= f1
                        a_trial[i2] *= f2

                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            reduced = True
                            break
                    if reduced:
                        break
                if reduced:
                    break
        
        if not reduced:
            no_improve += 1
            if no_improve > 20:

                a = a / a.sum()
                no_improve = 0
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
with log-sum-exp smoothing in log-space (L-BFGS-B), mirror descent,
subgradient with top-k peaks, and local peak shaving.
Explores multiple sequence lengths with adaptive time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_obj_grad(z, T):
    """
    Compute smooth objective and gradient in log-space (softmax parameterization).
    a = softmax(z), minimize 2n * LSE_T(conv(a,a))
    Returns (objective, gradient_wrt_z)
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_b = np.exp(scaled)
    lse_sum = exp_b.sum()
    LSE = b_max + T * np.log(lse_sum)
    w = exp_b / lse_sum

    # c[m] = sum_j a[j] * w[j+m], computed via FFT cross-correlation
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_c = 1 << (conv_len - 1).bit_length()
    Ar = rfft(a_rev, fft_c)
    W = rfft(w, fft_c)
    c_full = irfft(Ar * W, fft_c)[:conv_len]
    c = c_full[n - 1:2 * n - 1]

    # grad_a[m] = 2*c[m] - a[m]*w[2m] (if 2m < 2n-1)
    grad_a = 2.0 * c
    diag_idx = np.arange(0, 2 * n - 1, 2)
    grad_a[diag_idx // 2] -= a[diag_idx // 2] * w[diag_idx]

    obj = 2.0 * n * LSE

    # Transform gradient to z-space (softmax chain rule)
    ga_dot_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * ga_dot_a)
    grad_z *= 2.0 * n

    return obj, grad_z


def mirror_descent(a0, time_budget):
    """Mirror descent (multiplicative weight updates) for autocorrelation minimization."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = monotonic()
    lr = 0.05

    it = 0
    while monotonic() - t0 < time_budget:
        it += 1
        if it % 2000 == 0:
            lr = max(0.005, 0.05 / np.sqrt(it / 2000.0))

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Subgradient at the peak
        grad_a = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for m in range(n):
            j = peak_k - m
            if 0 <= j < n:
                grad_a[m] = 2.0 * a[j]
        if 0 <= peak_k // 2 < n and 2 * (peak_k // 2) == peak_k:
            grad_a[peak_k // 2] -= a[peak_k // 2]

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad_a
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def subgradient_topk(a0, time_budget, top_k=20, lr_init=0.01):
    """Subgradient method using weighted combination of top-k peaks with momentum."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    momentum = 0.85
    t0 = monotonic()
    no_improve = 0

    it = 0
    while monotonic() - t0 < time_budget:
        it += 1
        if it % 5000 == 0:
            lr = max(0.001, lr_init / np.sqrt(it / 5000.0 + 1.0))

        b = fast_convolve(a)
        peak_val = b.max()

        # Get top-k peaks with softmax weighting
        k_actual = min(top_k, len(b))
        top_idx = np.argpartition(b, -k_actual)[-k_actual:]
        top_vals = b[top_idx]
        temp = 0.005 * max(peak_val, 1e-10)
        w_top = np.exp((top_vals - top_vals.max()) / temp)
        w_top = w_top / w_top.sum()

        # Compute weighted subgradient
        grad_a = np.zeros(n)
        for ii in range(len(top_idx)):
            if w_top[ii] < 1e-10:
                continue
            k = int(top_idx[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                j = k - m
                if 0 <= j < n:
                    grad_a[m] += w_top[ii] * 2.0 * a[j]
            if k % 2 == 0 and 0 <= k // 2 < n:
                grad_a[k // 2] -= w_top[ii] * a[k // 2]

        # Momentum update
        v = momentum * v - lr * grad_a

        # Project onto simplex {a >= 0, sum(a) = 1}
        a_candidate = a + v
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0])
        theta = (cssv[rho] - 1.0) / (rho + 1)
        a_new = np.maximum(a_candidate - theta, 0.0)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturb to escape
                a = best_a + np.random.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = max(lr * 1.5, lr_init * 0.5)
                no_improve = 0

    return best_a, best_score


def lbgfs_optimize(a0, time_budget, T_start=50.0, T_end=1.0, n_temps=5):
    """L-BFGS optimization in log-space with decreasing temperature schedule."""
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    t0 = monotonic()
    temps = np.geomspace(T_start, T_end, n_temps)

    z = np.log(np.maximum(a0, 1e-30))
    z = z - z.mean()

    for T in temps:
        if monotonic() - t0 > time_budget - 3:
            break

        def func(z_in):
            obj, _ = compute_obj_grad(z_in, T)
            return obj

        def jac(z_in):
            _, g = compute_obj_grad(z_in, T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': 30000, 'disp': False}
            )
            z = res.x
            z_shifted = z - z.max()
            a_new = np.exp(z_shifted)
            a_new = a_new / a_new.sum()

            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

        except Exception:
            break

    return best_a, best_score

            others_sum = a_trial[mask].sum()
            if others_sum > 1e-15:
                a_trial[mask] += diff * a_trial[mask] / others_sum
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                reduced = True
                break

        # Strategy 2: Reduce pair
        if not reduced:
            top2 = np.argsort(contribs)[::-1][:4]
            for ci in top2:
                i1 = int(idx[ci])
                i2 = peak_k - i1
                if i1 == i2 or i2 < 0 or i2 >= n:
                    continue
                for f1 in [0.92, 0.88, 0.82]:
                    for f2 in [0.92, 0.88, 0.82]:
                        a_trial = a.copy()
                        a_trial[i1] *= f1
                        a_trial[i2] *= f2
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            reduced = True
                            break
                    if reduced:
                        break
                if reduced:
                    break

        # Strategy 3: Small random perturbation directed away from peak
        if not reduced:
            for attempt in range(3):
                direction = np.random.standard_normal(n)
                # Bias away from peak contributors
                for ci in top2[:2]:
                    ii = int(idx[ci])
                    jj = peak_k - ii
                    if 0 <= jj < n:
                        direction[ii] -= 0.5
                        direction[jj] -= 0.5
                a_trial = a + direction * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    reduced = True
                    break

        if reduced:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                a = best_a + np.random.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
            if no_improve > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    """

    2. L-BFGS in log-space with log-sum-exp smoothing (coarse to fine)
    3. Subgradient with top-k peaks and momentum

    5. Multi-length exploration

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()

    else:
