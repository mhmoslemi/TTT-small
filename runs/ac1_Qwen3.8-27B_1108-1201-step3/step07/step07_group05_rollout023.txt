def coord_descent_exact(a0, time_limit, n_sweeps=None):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Precompute random ordering
    while time.monotonic() - t0 < time_limit:
        # Random permutation of indices
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if a[i] < 1e-15:
                continue
            
            # Try transferring to a few random targets
            for _ in range(3):
                j = rng.integers(0, n)
                if i == j:
                    continue
                
                # Golden section search for optimal t
                lo = -a[j] + 1e-15
                hi = a[i] - 1e-15
                
                if hi - lo < 1e-15:
                    continue
                
                def score_at_t(t):
                    a_trial = a.copy()
                    a_trial[i] -= t
                    a_trial[j] += t
                    conv = fast_convolve(a_trial)
                    return 2.0 * n * conv.max()
                
                # Golden section search
                gr = (np.sqrt(5) - 1) / 2
                t1 = hi - gr * (hi - lo)
                t2 = lo + gr * (hi - lo)
                f1 = score_at_t(t1)
                f2 = score_at_t(t2)
                
                for _ in range(30):
                    if hi - lo < 1e-10:
                        break
                    if f1 < f2:
                        hi = t2
                        t2 = t1
                        f2 = f1
                        t1 = hi - gr * (hi - lo)
                        f1 = score_at_t(t1)
                    else:
                        lo = t1
                        t1 = t2
                        f1 = f2
                        t2 = lo + gr * (hi - lo)
                        f2 = score_at_t(t2)
                
                t_opt = (lo + hi) / 2
                score_opt = score_at_t(t_opt)
                
                if score_opt < best_score - 1e-14:
                    a[i] -= t_opt
                    a[j] += t_opt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_score = score_opt
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. L-BFGS with adaptive soft-max smoothing (multiple temperatures)
2. Multiplicative weight reweighting (fixed-point iteration toward KKT)
3. Coordinate descent with exact golden section line search
4. Spectral perturbation in frequency domain
5. Multi-length search with interpolation
6. Nelder-Mead on active peak-contributing coordinates
Key improvements: reweighting method targets KKT conditions directly,
exact line search in CD, and frequency-domain moves for correlated changes.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, minimize_scalar
from scipy.ndimage import gaussian_filter1d


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    # Cross-correlation of w with a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result


def compute_soft_gradient_conv(a, T):
    """Compute soft gradient of max(conv) using soft-max weights."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    M = np.exp((conv - max_c) / T)
    M /= M.sum()

    # g[i] = 2 * sum_k M[k] * (a[k-i] + a[i-k])
    N = 1 << (3 * n).bit_length()
    M_pad = np.zeros(N)
    M_pad[:2 * n - 1] = M
    a_pad = np.zeros(N)
    a_pad[:n] = a

    M_F = fft(M_pad)
    a_F = fft(a_pad)

    g1 = ifft(np.conj(M_F) * a_F).real[:n]
    g2 = ifft(a_F * M_F).real[:n]
    g = 2.0 * (g1 + g2)
    return M, g, conv, max_c


def reweighting_optimize(a0, time_limit, T_init=0.05, T_min=0.005,
                         step_init=0.3, verbose=False):
    """
    Multiplicative weight reweighting: fixed-point iteration toward KKT.
    At KKT: g[i] = constant for all i with a[i] > 0.
    Update: a[i] *= (g_max / g[i])^step
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = step_init
    T = T_init
    iters = 0
    no_improve = 0
    rng = np.random.default_rng(123456)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        M, g, conv, max_c = compute_soft_gradient_conv(a, T)
        score = 2.0 * n * max_c

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Multiplicative weight update
        g_safe = np.maximum(g, 1e-15)
        g_max = g_safe.max()
        if g_max < 1e-15:
            break

        ratio = g_max / g_safe
        a_new = a * np.power(ratio, step)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        a = a_new

        # Adaptive parameters
        if no_improve > 50:
            step *= 0.85
            T = min(T * 1.3, 50.0)
            no_improve = 0
        elif no_improve > 20:
            step *= 0.95
        if iters % 200 == 0:
            T = max(T_min, T * 0.995)
        if iters > 5000 and T <= T_min * 1.05:
            # Restart with perturbation if stuck at low T
            perturbation = rng.standard_normal(n) * 0.005
            a = best_a + perturbation
            a = np.maximum(a, 0)
            a = a / a.sum()
            step = step_init * 0.5
            T = T_init * 0.5
            no_improve = 0

        if verbose and iters % 500 == 0:
            print(f"    Reweight iter={iters}, best={best_score:.8f}, T={T:.4f}, step={step:.4f}")

    return best_a, best_score


def coord_descent_exact(a0, time_limit, targets_per_coord=5, max_iter_gs=30,
                        verbose=False):
    """
    Coordinate descent with exact golden section search.
    For each coordinate i, tries transferring to several target coordinates j,
    finding the optimal transfer amount via golden section search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77777)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            # Pick targets: mix of random and "gradient-informed"
            targets = []
            for _ in range(targets_per_coord):
                targets.append(rng.integers(0, n))

            for j in targets:
                if i == j:
                    continue

                lo = -a[j] + 1e-14
                hi = a[i] - 1e-14
                if hi - lo < 1e-13:
                    continue

                # Precompute the base convolution for efficiency
                # C'(k) = C(k) + 2*t*(a[k-j]-a[k-i]) + t^2*(delta_{k,2j}-delta_{k,2i})
                # We'll use golden section on the score function

                t1 = hi - gr * (hi - lo)
                t2 = lo + gr * (hi - lo)

                # Compute scores at t1 and t2
                def get_score_at_t(t, a_base, ii, jj):
                    a_trial = a_base.copy()
                    a_trial[ii] -= t
                    a_trial[jj] += t
                    conv = fast_convolve(a_trial)
                    return 2.0 * n * conv.max()

                f1 = get_score_at_t(t1, a, i, j)
                f2 = get_score_at_t(t2, a, i, j)

                for _ in range(max_iter_gs):
                    if hi - lo < 1e-11:
                        break
                    if f1 < f2:
                        hi = t2
                        t2 = t1
                        f2 = f1
                        t1 = hi - gr * (hi - lo)
                        f1 = get_score_at_t(t1, a, i, j)
                    else:
                        lo = t1
                        t1 = t2
                        f1 = f2
                        t2 = lo + gr * (hi - lo)
                        f2 = get_score_at_t(t2, a, i, j)

                t_opt = (lo + hi) / 2.0

                if abs(t_opt) < 1e-13:
                    continue

                # Evaluate at t_opt
                score_opt = get_score_at_t(t_opt, a, i, j)

                if score_opt < best_score - 1e-14:
                    a[i] -= t_opt
                    a[j] += t_opt
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss < 1e-15:
                        a = best_a.copy()
                        continue
                    a = a / ss
                    best_score = score_opt
                    best_a = a.copy()
                    improved = True

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep={sweep}, best={best_score:.8f}")

        if not improved and sweep > 5:
            # Perturb to escape
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = compute_score(a)

    return best_a, best_score
