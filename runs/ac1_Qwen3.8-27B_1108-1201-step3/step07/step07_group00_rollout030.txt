"""
Improved LP-based optimization with constraint generation, quadratic line search,
multi-length exploration, and peak shaving. Targets score < 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve_with_generation(f, n, top_k_init=30, max_constraints=500, eps=1e-8):
    """
    Solve LP with constraint generation:
    maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for active constraints.
    Start with top_k_init, add violated constraints iteratively.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    # Start with top-k peaks
    top_idx = np.argpartition(b, -top_k_init)[-top_k_init:]
    top_idx = np.sort(top_idx)
    active_set = set(top_idx.tolist())

    g = None
    for cg_iter in range(20):  # Max constraint generation iterations
        K = len(active_set)
        if K > max_constraints:
            # Keep only the most violated ones
            break
        sorted_idx = sorted(active_set)

        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = sorted_idx[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )
        try:
            prob.solve(solver=cp.PDLP, max_iters=200000)
        except Exception:

            except Exception:
                return None, M

        if g_var.value is None:
            return None, M
        g = np.maximum(g_var.value, 0.0)

        worst = np.argsort(fg[violations])[-min(10, len(violations)):]
        for v in worst:

    return g, M


def fast_convolve_imine(f, g):
    """Cross-convolution f*g via FFT."""
    n = max(len(f), len(g))
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def _steepest_desc_dir(f, peak_idx, n):
    """Steepest descent direction for reducing b[peak_idx]."""
    # Gradient of b[peak_idx] w.r.t. f[j] = f[peak_idx - j] (for valid j)
    g = np.zeros(n)
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            g[j] = f[k]
    return g


def peak_shaving(f, time_limit, n_peaks=5, shrink=0.01):
    """
    Direct peak shaving: for each top peak, reduce its value
    by proportionally decreasing the contributing elements.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    max_iters = 200

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(f)
        # Find top peaks
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        top_idx = top_idx[b[top_idx] > best_obj * 0.99]  # Only near-maxima
        
        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            # Elements contributing to b[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contributors = f[lo:hi + 1]
            partners = f[k - hi:k - lo + 1][::-1]
            
            # Try to reduce by scaling down large contributors
            mask = np.zeros(n, dtype=bool)
            mask[lo:hi + 1] = True
            # Scale down the top contributors
            contrib_vals = f.copy()
            contrib_vals[~mask] = 0
            top_contrib = np.where(contrib_vals > np.median(contrib_vals[mask]))[0]
            
            if len(top_contrib) == 0:
                continue
            
            # Try different shrink factors
            for sf in [shrink * 0.5, shrink, shrink * 1.5, shrink * 2]:
                f_trial = f.copy()
                f_trial[top_contrib] *= (1 - sf)
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj
                    improved = True
                    break
        
        if not improved:
            # Try increasing small elements (spread mass)
            b = fast_convolve(f)
            small_idx = np.argsort(f)[:max(1, n // 10)]
            for idx in small_idx[:5]:
                for inc in [0.001, 0.005, 0.01, 0.02]:
                    f_trial = f.copy()
                    f_trial[idx] += inc * f.mean()
                    s = f_trial.sum()
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-14:
                        f = f_trial
                        best_obj = obj
                        improved = True
                        break
                if time.monotonic() - t0 > time_limit:
                    break
            if not improved:
                break
    
    return f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.002, n_top=50):
    """Subgradient optimization with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = 0.0005
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Batch of subgradient steps
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            b = fast_convolve(f)
            
            # Weight peaks
            scaled = b / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum

            # Top-k weighting
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx].copy()
                ws = w_sel.sum()
                if ws > 1e-30:
                    w_sel /= ws
                else:
                    w_sel = np.ones(len(top_idx)) / len(top_idx)
                w_new = np.zeros(len(b))
                w_new[top_idx] = w_sel
                w = w_new

            # Compute gradient
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * f[k - idx]

            v = 0.9 * v - lr * g
            f_new = project_simplex(f + v)
            f = f_new

        # Check improvement
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.01)
        else:
            no_improve += 1
            if no_improve > 3:
                lr *= 0.9
                no_improve = 0
            if no_improve > 15:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0

        if it % 500 == 0:
            temp = min(temp * 1.2, 0.5)
            print(f"    sg iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj

"""
Improved LP-based optimization with constraint generation, focused descent,
multi-length exploration with window initialization, and peak shaving.
Targets score < 1.5027 (lower is better) for the auto-convolution peak problem.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lag_matrix(n, indices):
    """Build constraint matrix rows for given lag indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f_slice(n, lo, hi)
    return A


def f_slice(n, lo, hi):
    return np.ones(hi - lo + 1)


def lp_solve_with_generation(f, n, top_k_init=40, max_constraints=400, time_per_solve=30.0):
    """
    Solve LP with constraint generation:
    maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for active constraints.
    """
    t0 = time.monotonic()
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    # Start with top-k peaks
    if top_k_init < len(b):
        top_idx = np.argpartition(b, -top_k_init)[-top_k_init:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    active_list = top_idx.tolist()
    active_set = set(active_list)

    g = None
    for cg_iter in range(15):
        if time.monotonic() - t0 > time_per_solve * 2:
            break

        K = len(active_list)
        if K > max_constraints:
            # Keep only top constraints
            b_active = b[active_list]
            keep = np.argsort(b_active)[-max_constraints:]
            active_list = [active_list[i] for i in keep]
            active_set = set(active_list)
            K = len(active_list)

        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = active_list[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )
        t_solve_start = time.monotonic()

        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=300000)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            if g is not None:
                break
            return None, M

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)) or np.any(np.isinf(g)):
            if cg_iter > 0:
                break
            return None, M

        # Check for violated constraints (only if we have some headroom)
        if time.monotonic() - t0 > time_per_solve:
            break

        fg = fast_cross_convolve(f, g)
        # Find violations outside active set
        mask = np.zeros(len(b), dtype=bool)
        mask[active_list] = True
        violations = np.where((fg > M + 1e-9) & (~mask))[0]

        if len(violations) == 0:
            break

        # Add most violated constraints (up to 5 at a time)
        worst = np.argsort(fg[violations])[-min(5, len(violations)):]
        for v in worst:
            vi = int(v)
            if vi not in active_set:
                active_set.add(vi)
                active_list.append(vi)

    return g, M


def quadratic_line_search(f, direction, max_evals=20):
    """Find best step size using golden section search on the peak objective."""
    n = len(f)

    def peak_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # Initial coarse search
    t_candidates = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    best_t = 0.05
    best_obj = peak_at(0.05)
    for t in t_candidates:
        obj = peak_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Refine with golden section
    idx = t_candidates.index(best_t)
    lo = t_candidates[max(0, idx - 1)]
    hi = t_candidates[min(len(t_candidates) - 1, idx + 1)]
    if lo == hi:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = peak_at(c)
    fd = peak_at(d)

    for _ in range(12):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = peak_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = peak_at(d)
        if hi - lo < 1e-5:
            break

    best_t = (lo + hi) / 2
    return best_t

        if g is None:

            b = fast_convolve(f)

def focused_descent(f, time_limit, n_peaks=5):
    """Direct steepest descent on the top peaks with adaptive step size."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        # Find top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Compute combined gradient (direction to reduce peaks)
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]

        if grad.max() < 1e-15:
            break

        # Try different step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            # Try very small steps
            for alpha in [0.0005, 0.001]:
                f_trial = f - alpha * grad
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                if obj < best_trial - 1e-14:
                    best_trial = obj
                    best_alpha = alpha

        if best_alpha is None:
            break

        f_new = f - best_alpha * grad
        f_new = project_simplex(f_new)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.001, n_top=30):

            if w_sum < 1e-30:

def flat_top_window(n):
    """Generate a normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def raised_cosine_window(n):
    """Generate a normalized raised cosine (tukey-like) window."""
    x = np.linspace(0, 1, n)
    w = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

                best_for_len = f_reg.copy()

        if best_for_len is not None and best_score_len < best_score:
            best_score = best_score_len
            best_seq = best_for_len.copy()
            print(f"  NEW BEST [n{target_n}]: {best_score:.8f}")
        else:
            print(f"  n{target_n}: best={best_score_len:.8f} (no improvement over {best_score:.8f})")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 100.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = subgradient_polish(best_seq, time_limit=t6, lr_init=0.001,
                                       n_top=max(20, n_b // 12))
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [sg]: {score6:.8f}")
        else:
            print(f"  sg: {score6:.8f} (no improvement)")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.35, 120.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t7:.0f}s, n={n_b}")
        f7, obj7 = run_lp_phase(best_seq, time_limit=t7, top_k_init=80,
                                 top_k_max=200, label="lp_final")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {score7:.8f}")

    # === Phase 8: Final focused descent ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t8 = min(rem * 0.3, 30.0)
        if t8 > 5:
            f8, obj8 = focused_descent(best_seq, time_limit=t8, n_peaks=10)
            score8 = float(evaluate_sequence(f8.tolist()))
            if np.isfinite(score8) and score8 < best_score:
                best_score = score8
                best_seq = f8.copy()
                print(f"  NEW BEST [final_focused]: {score8:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Improvement needed: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
