def smooth_max(x, beta):
    """Smooth approximation to max(x) using log-sum-exp."""
    x_max = x.max()
    return x_max + np.log(np.exp((x - x_max) * beta).sum()) / beta

    """Weights for the smooth max (softmax of beta*x)."""
    x_max = x.max()
    e = np.exp((x - x_max) * beta)
    return e / e.sum()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute the objective score (assuming sum(a) = 1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

    n = len(a0)
    z = np.log(a0 + 1e-30)
    z -= z.mean()  # center for numerical stability
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    N = 1 << (3*n - 2 - 1).bit_length()  # FFT size for xcorr
    
    best_a = a0.copy()
    best_score = score_of(a0)
    
    for beta in betas:
        for step in range(steps_per_beta):
            t += 1
            # Compute a from z
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            
            # Compute convolution
            A = rfft(a, N)
            conv = irfft(A * A, N)[:2*n-1]
            
            # Compute weights
            conv_max = conv.max()
            w = np.exp(beta * (conv - conv_max))
            w /= w.sum()
            
            # Gradient in a-space: 2 * xcorr(w, a)
            W = rfft(w, N)
            grad_a = 2.0 * irfft(W * np.conj(A), N)[:n]
            
            # Gradient in z-space
            dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - dot)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Track best
            s = score_of(a)
            if s < best_score - 1e-15:
                best_score = s
                best_a = a.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based Adam optimization in log-space
with adaptive beta (smoothed-max) schedule, diverse initializations, multi-length
exploration, and local peak shaving. Key insight: the gradient of the smoothed max
of the auto-convolution is a cross-correlation, computable in O(n log n) via FFT.
"""

import numpy as np
from scipy.fft import rfft, irfft
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute objective score assuming sum(a)=1, a>=0."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_logspace(a0, betas, iters_per_beta, lr=0.005, deadline=None, verbose=False):
    """
    Adam optimization in log-space minimizing smoothed max of auto-convolution.
    a0: initial point on simplex (sum=1, >=0)
    betas: list of sharpness parameters (increasing)
    iters_per_beta: iterations at each beta
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z -= z.mean()

    N = 1 << (3 * n - 2).bit_length()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_score = score_of(a0)

    for beta in betas:
        # Scale lr inversely with beta (sharper objective needs smaller steps)
        cur_lr = lr / max(1.0, np.sqrt(beta))
        for step in range(iters_per_beta):
            if deadline is not None and monotonic() > deadline:
                return best_a, best_score

            t += 1
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a = exp_z / exp_z.sum()

            A = rfft(a, N)
            conv = irfft(A * A, N)[:2 * n - 1]

            cmax = conv.max()
            shifted = beta * (conv - cmax)
            # Clip to avoid overflow
            shifted = np.clip(shifted, -50, 50)
            w = np.exp(shifted)
            wsum = w.sum()
            if wsum < 1e-30:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(conv))] = 1.0
            else:
                w /= wsum

            W = rfft(w, N)
            grad_a = 2.0 * irfft(W * np.conj(A), N)[:n]

            dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - dot)

            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            z = z - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

            if t % 500 == 0:
                z -= z.mean()

            if step % 5 == 0 or step == iters_per_beta - 1:
                s = score_of(a)
                if s < best_score - 1e-15:

                    best_a = a.copy()
                    if verbose and step % 200 == 0:

def subgradient_descent(a0, max_iters, lr=0.003, momentum=0.85, top_k=50, deadline=None, verbose=False):

    Faster than full gradient when only a few peaks matter.
    """

    best_a = a.copy()

            break

        k_actual = min(top_k, len(conv))

        pv = conv[peaks]

        scaled = np.clip((pv - pv.max()) / temp, -50, 50)
        w = np.exp(scaled)

    2. FFT-based Adam in log-space with adaptive beta schedule
    3. Subgradient descent refinement
    4. Local peak shaving
    5. Final polish with tighter parameters
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Starting uniform n=768: score={best_score:.8f}")

    n_parent = len(best_seq)

    # Determine which lengths to try
    n_candidates = sorted(set([n_parent, 512, 768, 1024, 1536, 2048, 3072]))
    # Prioritize lengths close to parent
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_parent))
    n_candidates = n_candidates[:6]

    print(f"\n=== Phase 1: Adam Log-Space at multiple lengths ===")
    print(f"Lengths to try: {n_candidates}")

    # Time allocation: more time for parent length
    total_time = deadline - monotonic() - 120
    time_per_n = max(30, total_time / len(n_candidates))

    for n_try in n_candidates:
        if monotonic() > deadline - 100:
            break
        rem = deadline - monotonic() - 80
        if rem < 20:
            break
        t_alloc = min(time_per_n, rem)

        print(f"\n  n={n_try}, time={t_alloc:.0f}s")

        # Get initializations for this length
        if parent is not None and len(parent) == n_try:
            inits = [('parent', parent.copy())]
        elif parent is not None:
            inits = [('parent_interp', interpolate_to_length(parent, n_try))]
        else:
            inits = []

        # Add structured inits (limit to 3 to save time)
        standard_inits = make_initializations(n_try, seed_val=n_try)
        for name, arr in standard_inits[:3]:
            inits.append((name, arr))

        # Run Adam for each init
        for init_name, init_a in inits:
            if monotonic() > deadline - 60:
                break

            # Adaptive beta schedule
            beta_sched = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
            # More iterations at low beta, fewer at high beta
            iters = [max(50, int(t_alloc * 8 / len(beta_sched))),
                     max(50, int(t_alloc * 6 / len(beta_sched))),
                     max(40, int(t_alloc * 5 / len(beta_sched))),
                     max(30, int(t_alloc * 4 / len(beta_sched))),
                     max(20, int(t_alloc * 3 / len(beta_sched))),
                     max(20, int(t_alloc * 3 / len(beta_sched))),
                     max(10, int(t_alloc * 2 / len(beta_sched))),
                     max(10, int(t_alloc * 2 / len(beta_sched)))]

            d_inner = deadline - 60
            a_result, s_result = adam_logspace(
                init_a, beta_sched, iters,
                lr=0.008, deadline=d_inner, verbose=False
            )
            best_seq, best_score = check_and_update(a_result, f"adam_n{n_try}_{init_name}", best_seq, best_score)

    # === Phase 2: Subgradient refinement on best ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 60
        if rem > 30:
            t_sg = min(rem * 0.4, 120.0)
            n_best = len(best_seq)
            top_k = max(30, n_best // 8)
            max_it = max(1000, int(t_sg * 800))  # estimate iterations
            a_sg, _ = subgradient_descent(
                best_seq, max_it, lr=0.002, momentum=0.88,
                top_k=top_k, deadline=monotonic() + t_sg
            )
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 40
        if rem > 15:
            t_ps = min(rem * 0.5, 100.0)
            max_it = int(t_ps * 500)
            a_ps, _ = peak_shave(best_seq, max_it, deadline=monotonic() + t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 4: Second Adam pass with tighter params ===
    print(f"\n=== Phase 4: Adam Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 30
        if rem > 20:
            t_polish = min(rem * 0.4, 80.0)
            beta_tight = [50.0, 200.0, 1000.0, 5000.0, 20000.0]
            iters_tight = [max(20, int(t_polish * 4 / len(beta_tight)))] * len(beta_tight)
            a_pol, _ = adam_logspace(
                best_seq, beta_tight, iters_tight,
                lr=0.002, deadline=monotonic() + t_polish
            )
            best_seq, best_score = check_and_update(a_pol, "adam_polish", best_seq, best_score)

    # === Phase 5: Final peak shaving ===
    print(f"\n=== Phase 5: Final Peak Shave (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 8:
            t_final = min(rem * 0.6, 30.0)
            max_it = int(t_final * 500)
            a_fin, _ = peak_shave(best_seq, max_it, deadline=monotonic() + t_final)
            best_seq, best_score = check_and_update(a_fin, "final_shave", best_seq, best_score)

    # === Phase 6: Try a few more peaks shaves with different strategies ===
    print(f"\n=== Phase 6: Extended Local Search (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 5
        if rem > 10:
            t_ext = min(rem * 0.7, 50.0)
            # Run subgradient with different params
            n_best = len(best_seq)
            max_it = int(t_ext * 1000)
            a_ext, _ = subgradient_descent(
                best_seq, max_it, lr=0.001, momentum=0.92,
                top_k=max(20, n_best // 12), deadline=monotonic() + t_ext
            )
            best_seq, best_score = check_and_update(a_ext, "ext_subgrad", best_seq, best_score)

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
