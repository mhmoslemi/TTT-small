1. Guess the active set (which lags have the maximum convolution).
2. Solve for a that satisfies the KKT conditions for that active set.
3. Check if the guess was correct (i.e., the computed a really has those lags as its max).
4. If not, adjust the active set and repeat.

This is the "active set method" mentioned in the state of the art!

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

    """LP-based refinement using sparse matrices."""
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = max_b_of(a)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        if M < 1e-15:
            break
        
        # Build sparse LP
        A_sp = build_sparse_lp_matrix(a)
        b_lp = M * np.ones(2 * n - 1)
        c_vec = -np.ones(n)
        
        try:
            res = linprog(c_vec, A_ub=A_sp, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            best_t = 0.0
