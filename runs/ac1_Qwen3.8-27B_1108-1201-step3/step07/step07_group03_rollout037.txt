```python
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    t = cp.Variable()

        i_start = max(0, k - n + 1)

    a_val = np.asarray(a.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
```

Since the constraint is `a >= 0` (the probability simplex), mirror descent with the negative entropy function (which gives the multiplicative weights update) is natural:


This is equivalent to the "exponential weights" or "Hedge" algorithm in online learning.

For our problem, the subgradient at `a_t` is:
`g_t(i) = 2 * sum_{k in active peaks} w_k * a_t(k-i)` (for valid indices)

where the active peaks are those where `conv(a_t, a_t)_k` is close to the max.

```python
def mirror_descent(a0, time_limit, eta_init=0.1, decay=0.999):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    eta = eta_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks (within epsilon of max)
        eps = 0.01 * max_b
        active = np.where(b >= max_b - eps)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            for i in range(i_start, i_end + 1):
                j = int(k) - i
                g[i] += a[j]
        
        # Mirror descent update
        log_a = np.log(a + 1e-300)
        log_a_new = log_a - eta * g
        # Normalize (project onto simplex via exp)
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        eta *= decay
    
    return best_a, best_score
```

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. LP-based fictitious play with multiple diverse initializations
2. Exponentially weighted running average for faster convergence
3. Mirror descent (exponential weights / multiplicative weights) refinement
4. Simulated annealing for local escape from poor local minima
5. Full coordinate descent final polish
6. Multi-length exploration around the best found length
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False, ewma_alpha_decay=0.9995):
    """Alternating LP best responses with exponentially weighted moving average."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:

        if a_new is not None:
            a = a_new
            a_avg = (1 - alpha) * a_avg + alpha * a

def compute_subgrad(a, top_k=30):
    """Compute weighted subgradient of max(conv(a,a)) at point a."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    temp = 0.005 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        sg[j] += 2.0 * w[idx] * a[k - j]
    return sg

def mirror_descent(a0, time_limit, eta_init=0.05, top_k=30):
    """Mirror descent (exponential weights) on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    eta = eta_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = compute_subgrad(a, top_k=top_k)

        log_a = np.log(a + 1e-300)
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        sc = score_of(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 200:
            eta *= 0.8
            no_improve = 0
        eta *= 0.9999

        # Periodic restart from best
        if it % 5000 == 0 and no_improve > 1000:
            a = best_a.copy()
            eta = eta_init * 0.5

    return best_a, best_score

def simulated_annealing(a0, time_limit, t_init=0.01, t_final=1e-6):
    """Simulated annealing on the simplex with scaling perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    n_iter = 50000
    total_iters_est = 1.0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        progress = it / total_iters_est
        T = t_init * (t_final / t_init) ** progress

        # Random perturbation: pick 2-5 coordinates and scale them
        n_perturb = rng.integers(2, 6)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_trial = a.copy()
        for i in idx:
            factor = rng.uniform(0.5, 1.5)
            a_trial[i] *= factor
        a_trial = np.maximum(a_trial, 0)
        s_try = a_trial.sum()
        if s_try < 1e-15:
            continue
        a_trial = a_trial / s_try

        sc_trial = score_of(a_trial)
        delta = sc_trial - score_of(a)
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))

    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_blocks(n, rng):

    print(f"\n=== Phase 5: Simulated Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sa = min(rem * 0.6, 120.0)
            print(f"  SimAnn budget={t_sa:.0f}s")
            a_sa, s_sa = simulated_annealing(best_seq, t_sa, t_init=0.008, t_final=1e-7)
            best_seq, best_score = check_and_update(a_sa, best_seq, best_score, "sim_anneal")

    # === Phase 6: Coordinate descent + peak shaving ===
    print(f"\n=== Phase 6: CoordDescent + PeakShave (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 100.0)
            print(f"  CoordDescent budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent_polish(best_seq, t_cd, n_sweeps=500)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_descent")

        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_ps = min(rem * 0.6, 60.0)
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_targeted_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "peak_shave")

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.85, 1.15, 0.75, 1.25]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2560 and alt_n != n_best:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alt = min(rem * 0.5, 80.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fictitious_play(alt_n, t_alt, initial=init_alt,
                                               solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

            # Quick polish
            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_pol, _ = coord_descent_polish(best_seq, min(rem * 0.5, 30.0), n_sweeps=100)
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "alt_polish")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_final = min(rem * 0.4, 40.0)
            a_fin, _ = coord_descent_polish(best_seq, t_final, n_sweeps=300)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fin2, _ = peak_targeted_shave(best_seq, min(rem * 0.5, 20.0))
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_ps")

        # One more mirror descent pass with small eta
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_fin3, _ = mirror_descent(best_seq, min(rem * 0.6, 10.0), eta_init=0.01, top_k=50)
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
