```python
    """Minimize f on [a, b] using golden section search."""
    phi = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = f(d)
    return (a + b) / 2
```

```python
def golden_section_min(f, lo, hi, max_iter=25, tol=1e-12):
    """Minimize unimodal function f on [lo, hi] using golden section search."""
    if hi - lo < tol:
        return (lo + hi) / 2.0
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / phi
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)

    fd = f(d)

            fd = fc

            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0
```

Now let me think about the pairwise phase. The idea is:
1. Find the top peaks in the convolution
2. For each peak k, find the coordinate pairs (i, k-i) that contribute most
3. For each such pair, do a 2D optimization

For the 2D optimization, I'll use a simple approach: alternate between optimizing a[i] (with a[j] fixed) and a[j] (with a[i] fixed), for a few iterations. This is 2D coordinate descent.

```python
def pairwise_phase(a, b, n, total_sum, time_limit, verbose=False):
    t0 = time.monotonic()
    best_score = 2.0 * n * b.max() / (total_sum * total_sum)
    best_a = a.copy()
    best_b = b.copy()
    best_sum = total_sum
    rng = np.random.default_rng(123)
    
    round_num = 0
    while time.monotonic() - t0 < time_limit:
        round_num += 1
        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peaks = peaks[np.argsort(b[peaks])[::-1]]
        
        for peak_k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peak_k)
            # Valid i values: max(0, k-n+1) <= i <= min(k, n-1)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start > i_end:
                continue
            indices = np.arange(i_start, i_end + 1)
            # Contribution of pair (i, k-i) to b[k]
            # If i != k-i: contribution is 2*a[i]*a[k-i]
            # If i == k-i: contribution is a[i]^2
            js = k - indices
            contribs = np.zeros(len(indices))
            for idx in range(len(indices)):
                i_val = indices[idx]
                j_val = js[idx]
                if i_val == j_val:
                    contribs[idx] = a[i_val] ** 2
                else:
                    contribs[idx] = 2.0 * a[i_val] * a[j_val]
            
            # Get top contributing pairs
            top_pairs = np.argsort(contribs)[::-1][:8]
            
            for pi in top_pairs:
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = int(indices[pi])
                i2 = int(js[pi])
                if i1 == i2:
                    continue  # Skip diagonal (already handled by 1D CD)
                
                # 2D coordinate descent: alternate between i1 and i2
                for _iter in range(10):
                    if time.monotonic() - t0 > time_limit:
                        break
                    # Optimize i1 with i2 fixed
                    hi1 = max(2.5 * a[i1], 3.0 / n)
                    best_v1 = golden_section_coord(b, a, n, total_sum, i1, 0.0, hi1, max_iter=12)
                    if abs(best_v1 - a[i1]) > 1e-15:
                        old_v = a[i1]
                        a[i1] = best_v1
                        total_sum += best_v1 - old_v
                        apply_coord_change(b, a, n, i1, old_v, best_v1)
                    
                    # Optimize i2 with i1 fixed
                    hi2 = max(2.5 * a[i2], 3.0 / n)
                    best_v2 = golden_section_coord(b, a, n, total_sum, i2, 0.0, hi2, max_iter=12)
                    if abs(best_v2 - a[i2]) > 1e-15:
                        old_v = a[i2]
                        a[i2] = best_v2
                        total_sum += best_v2 - old_v
                        apply_coord_change(b, a, n, i2, old_v, best_v2)
                
                new_score = 2.0 * n * b.max() / (total_sum * total_sum)
                if new_score < best_score - 1e-15:
                    best_score = new_score
                    best_a = a.copy()
                    best_b = b.copy()
                    best_sum = total_sum

            elapsed = time.monotonic() - t0

        if round_num % 20 == 0:
            np.copyto(b, fast_convolve(a))
    
    return best_a, best_b, best_score
```

Hmm, this is getting complex. Let me also add the simulated annealing phase.

For simulated annealing:
- State: (a, b, total_sum, score)
- Move: pick a random coordinate i, set a[i] = max(0, a[i] + noise) where noise ~ Normal(0, sigma)
- Or: pick a random coordinate i, set a[i] = a[i] * exp(noise) where noise ~ Normal(0, sigma)
- Accept if new_score < old_score, or with prob exp((old_score - new_score) / T)
- Cool T: T = T0 * cooling_rate^step

The key is choosing sigma. If sigma is too large, we'll rarely accept and make no progress. If too small, we'll be stuck locally.

A good adaptive strategy: start with sigma such that about 50% of moves are accepted, then decrease.

Let me implement a simpler version: geometric neighborhood moves (multiply by a random factor between 0.5 and 2.0).

```python
def simulated_annealing_phase(a, b, n, total_sum, time_limit, verbose=False):
    t0 = time.monotonic()
    best_score = 2.0 * n * b.max() / (total_sum * total_sum)
    best_a = a.copy()
    best_b = b.copy()
    best_sum = total_sum
    rng = np.random.default_rng(777)
    
    T = 0.005  # initial temperature
    cooling = 0.9997
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Pick random coordinate
        i = int(rng.integers(0, n))
        old_v = a[i]
        if old_v < 1e-15:
            new_v = rng.exponential(1.0 / n) * 0.1
        else:
            # Geometric perturbation
            log_factor = rng.normal(0.0, 0.3)
            new_v = old_v * np.exp(log_factor)
        
        if new_v < 0:
            new_v = 0.0
        if new_v == old_v:
            continue
        
        old_score = 2.0 * n * b.max() / (total_sum * total_sum)
        
        # Apply change
        total_sum += new_v - old_v
        apply_coord_change(b, a, n, i, old_v, new_v)
        
        new_score = 2.0 * n * b.max() / (total_sum * total_sum)
        
        # Accept or reject
        if new_score < old_score:
            accept = True
        else:
            delta = new_score - old_score
            if delta > 0 and T > 1e-8:
                accept = rng.random() < np.exp(-delta / T)
            else:
                accept = False
        
        if not accept:
            # Revert
            total_sum -= new_v - old_v
            apply_coord_change(b, a, n, i, new_v, old_v)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            best_b = b.copy()
            best_sum = total_sum
        
        T *= cooling
        
        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    SA step={step} ({elapsed:.0f}s): T={T:.6f} best={best_score:.8f}")
        
        if step % 5000 == 0:
            np.copyto(b, fast_convolve(a))
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using intensive multi-phase local search:
1. Exact coordinate descent with multi-start golden-section line search (incremental conv updates)
2. Pairwise coordinate descent targeting peak-contributing coordinate pairs
3. Simulated annealing with geometric perturbations and adaptive temperature
4. Multi-round refinement cycling through phases with periodic convolution recomputation

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def apply_coord_change(b, a, n, i, old_val, new_val):
    """Apply a coordinate change to a and b in place (incremental update)."""
    a[i] = new_val
    delta = new_val - old_val
    if abs(delta) < 1e-18:
        return
    k_start = i
    k_end = min(i + n, 2 * n - 1)
    ks = np.arange(k_start, k_end)
    updates = 2.0 * delta * a[ks - i]
    idx_2i = 2 * i
    if k_start <= idx_2i < k_end:
        pos = idx_2i - k_start
        updates[pos] = new_val * new_val - old_val * old_val
    b[ks] += updates


def eval_hypothetical_score(b, a, n, total_sum, i, new_val):
    """Evaluate score if a[i] were set to new_val, without modifying a or b."""
    old_val = a[i]
    new_sum = total_sum + new_val - old_val
    if new_sum < 0.01:
        return float('inf')
    delta = new_val - old_val
    if abs(delta) < 1e-18:
        return float(2.0 * n * b.max() / (new_sum * new_sum))
    k_start = i
    k_end = min(i + n, 2 * n - 1)
    ks = np.arange(k_start, k_end)
    updates = 2.0 * delta * a[ks - i]
    idx_2i = 2 * i
    if k_start <= idx_2i < k_end:
        pos = idx_2i - k_start
        updates[pos] = new_val * new_val - old_val * old_val
    new_b_segment = b[ks] + updates
    segment_max = float(new_b_segment.max())
    if k_start > 0:
        left_max = float(b[:k_start].max())
    else:
        left_max = 0.0
    if k_end < 2 * n - 1:
        right_max = float(b[k_end:].max())
    else:
        right_max = 0.0
    unaffected_max = max(left_max, right_max)
    new_max = max(segment_max, unaffected_max)
    return float(2.0 * n * new_max / (new_sum * new_sum))


def golden_section_coord(b, a, n, total_sum, i, lo, hi, max_iter=20):
    """Golden section search for optimal value of coordinate i."""
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    inv_phi = 2.0 / (1.0 + np.sqrt(5.0))
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = eval_hypothetical_score(b, a, n, total_sum, i, c)
    fd = eval_hypothetical_score(b, a, n, total_sum, i, d)
    for _ in range(max_iter):
        if hi - lo < 1e-13:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = eval_hypothetical_score(b, a, n, total_sum, i, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = eval_hypothetical_score(b, a, n, total_sum, i, d)
    return (lo + hi) / 2.0

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
