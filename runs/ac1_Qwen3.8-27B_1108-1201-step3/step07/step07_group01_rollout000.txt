```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Key mathematical insight: By Cauchy-Schwarz, sum(a_i)^2 <= n * sum(a_i^2),
and since max(conv(a,a)) >= conv(a,a)[n-1] = sum(a_i^2), we always have
score >= 2, with equality iff the sequence is uniform.

Therefore the optimal solution is the uniform distribution. We verify this
with a uniform sequence and attempt limited optimization to confirm.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-300:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def try_gradient_descent(a_init, time_limit, lr=0.01, max_iters=10000):
    """
    Subgradient descent on the simplex. Since the minimum is 2 (achieved by
    uniform), this should converge to uniform. We use it to verify.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_peak = int(np.argmax(b))

        # Subgradient of max_k b_k w.r.t. a
        # b_k = sum_i a_i * a_{k-i}, grad_{a_j} b_k = a_{k-j} (for valid indices)
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = a[k_peak - idx]

        # Project gradient to simplex tangent space
        grad = grad - np.dot(a, grad)

        # Multiplicative update (mirror descent on simplex)
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        sc = score_from_a(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

    rng = np.random.default_rng(12345)

    uniform = np.ones(n) / n
    uniform_score = score_from_a(uniform)

    # The uniform sequence is provably optimal (score = 2)
    if uniform_score < best_score:
        best_a = uniform.copy()
        best_score = uniform_score

    # Try a few random directions to confirm uniform is best
    for i in range(min(1000, max(1, int((time_limit - (time.monotonic() - t0)) * 100)))):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb toward uniform
        t = rng.uniform(0.01, 0.5)
        a_trial = (1 - t) * best_a + t * uniform
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        sc = score_from_a(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()

    # Also try random perturbations away from current to confirm no better exists
    for i in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        noise = rng.standard_normal(n) * 0.05
        a_trial = best_a + noise
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        sc = score_from_a(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Find the optimal non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Mathematical proof that the minimum is 2 (achieved by uniform):
    - max(conv(a,a)) >= conv(a,a)[n-1] = sum(a_i^2) [zero-lag autocorrelation]
    - By Cauchy-Schwarz: sum(a_i)^2 <= n * sum(a_i^2)
    - Therefore: 2*n*max(conv(a,a))/(sum(a))^2 >= 2*n*(sum(a)^2/n)/(sum(a))^2 = 2
    
    Equality holds iff all a_i are equal (uniform distribution).
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    # Check for parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    # The provably optimal solution is the uniform distribution
    # Choose a reasonable length
    if parent is not None:
        n = len(parent)
    else:
        n = 768

    # Try a few lengths to see if any gives a better numerical score
    best_seq = None
    best_score = float('inf')

    candidate_lengths = [n, 512, 1024, 256, 2048]
    candidate_lengths = sorted(set(l for l in candidate_lengths if 2 <= l <= 4096))

    for n_try in candidate_lengths:
        if time.monotonic() > deadline:
            break
        a_uniform = np.ones(n_try) / n_try
        real_score = float(evaluate_sequence(a_uniform.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = a_uniform.copy()
            print(f"Uniform n={n_try}: score={real_score:.10f}")

    # Verify with internal score function
    print(f"Internal check - uniform n={len(best_seq)}: internal_score={score_from_a(best_seq):.10f}")
    print(f"Internal check - parent: internal_score={score_from_a(parent):.10f}" if parent is not None else "")

    # Try optimization from parent (should not improve on 2.0)
    if parent is not None:
        p_score_internal = score_from_a(parent)
        print(f"Parent internal score: {p_score_internal:.10f}")

        # Only attempt optimization if parent is close to optimal
        if p_score_internal < 2.01:
            rem = deadline - time.monotonic() - 30
            if rem > 10:
                t_alloc = min(rem * 0.3, 120.0)
                print(f"\nAttempting gradient descent from parent (budget={t_alloc:.0f}s)...")
                a_opt, s_opt = try_gradient_descent(parent, t_alloc, lr=0.005, max_iters=5000)
                real_s = float(evaluate_sequence(a_opt.tolist()))
                print(f"GD result: internal={s_opt:.10f}, real={real_s:.10f}")
                if np.isfinite(real_s) and real_s < best_score:
                    best_score = real_s
                    best_seq = a_opt.copy()

            rem = deadline - time.monotonic() - 20
            if rem > 10:
                t_alloc = min(rem * 0.5, 120.0)
                print(f"\nAttempting perturbation search (budget={t_alloc:.0f}s)...")
                a_pert, s_pert = try_perturbation(parent, t_alloc)
                real_s = float(evaluate_sequence(a_pert.tolist()))
                print(f"Perturbation result: internal={s_pert:.10f}, real={real_s:.10f}")
                if np.isfinite(real_s) and real_s < best_score:
                    best_score = real_s
                    best_seq = a_pert.copy()

    # If we have no result (shouldn't happen), use uniform
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    print(f"Theoretical minimum is 2.0 (achieved by uniform distribution).")

    return [float(x) for x in best_seq]
```
