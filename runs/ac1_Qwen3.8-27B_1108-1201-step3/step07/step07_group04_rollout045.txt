import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] s.t. sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

    t0 = monotonic()
    pass_count = 0
    no_improve_passes = 0

    while monotonic() - t0 < time_limit:
        improved_this_pass = False
        for i in range(n):
            if monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                # Try adding a small amount
                for dv in [a.max() * 0.01, a.max() * 0.005]:
                    a_trial = a.copy()
                    a_trial[i] = dv
                    s_trial = a_trial.sum()
                    a_trial = a_trial / s_trial
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved_this_pass = True
                        break
                continue

            # Try multiplicative and additive perturbations
            candidates = [
                v * 0.5, v * 0.7, v * 0.85, v * 1.15, v * 1.3, v * 1.5,
                v + a.max() * 0.02, v - a.max() * 0.02
            ]
            candidates = [max(0.0, c) for c in candidates]

            best_trial = None
            best_trial_score = best_score

            for new_val in candidates:
                if abs(new_val - v) < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_trial = a_trial / s_trial
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_trial_score - 1e-14:
                    best_trial_score = score
                    best_trial = a_trial

            if best_trial is not None:
                a = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved_this_pass = True

        pass_count += 1
        if not improved_this_pass:
            no_improve_passes += 1
            if no_improve_passes > 5:
                break
        else:
            no_improve_passes = 0

        if verbose and pass_count % 10 == 0:
            elapsed = monotonic() - t0
            print(f"  CCD pass={pass_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def cyclic_cd_focused(a0, time_limit, top_k=50, verbose=True):
    """CCD that focuses on coordinates contributing to the top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    iters = 0
    no_improve = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find top peak positions
        k_actual = min(top_k, len(b))
        peak_positions = np.argpartition(b, -k_actual)[-k_actual:]

        # Find which coordinates contribute most to these peaks
        contrib_score = np.zeros(n)
        for k in peak_positions:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contrib_score[j] += b[k] * a[j] * a[k - j]

        # Focus on top contributing coordinates
        focus_indices = np.argsort(contrib_score)[::-1][:min(100, n // 3)]
        # Add random coordinates for exploration
        random_indices = np.random.choice(n, size=min(50, n // 10), replace=False)
        indices = np.unique(np.concatenate([focus_indices, random_indices]))

        improved = False
        for i in indices:
            if monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                for dv in [a.max() * 0.005, a.max() * 0.002]:
                    a_trial = a.copy()
                    a_trial[i] = dv
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                continue

            # Fine-grained search around current value
            step = max(v * 0.1, a.max() * 0.005)
            for offset in [-2 * step, -step, -step * 0.5, step * 0.5, step, 2 * step]:
                new_val = v + offset
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_trial = a_trial / s_trial
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        iters += 1
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            print(f"  CCD_focused iter={iters} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        # Gradient of the smoothed objective
        # d/da [sum_k w_k * (a*a)_k] = 2 * sum_k w_k * a shifted
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        # Transform to z-space: a = softmax(z), da/dz_i = a_i*(delta_ij - a_j)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def peak_shave_aggressive(a0, time_limit, verbose=True):
    """Aggressive peak shaving with quadratic redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    steps = 0
    no_improve = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)
        if peak_val <= target + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-20:
            break

        improved = False

        # Strategy 1: Reduce top contributors proportionally
        top_n = min(10, len(contribs))
        top_contrib = np.argsort(contribs)[::-1][:top_n]

        for reduce_frac in [0.3, 0.2, 0.1, 0.05]:
            a_trial = a.copy()
            reduction = 0.0
            for ci in top_contrib:
                i = int(idx[ci])
                old_val = a_trial[i]
                new_val = old_val * (1 - reduce_frac)
                reduction += old_val - new_val
                a_trial[i] = new_val

            # Redistribute reduction to non-contributing positions
            mask = np.ones(n, dtype=bool)
            for ci in top_contrib:
                i = int(idx[ci])
                mask[i] = False
                j = peak_k - i
                if 0 <= j < n:
                    mask[j] = False

            if not mask.any():
                continue

            others_sum = a_trial[mask].sum()
            if others_sum < 1e-15:
                continue

            a_trial[mask] += reduction * a_trial[mask] / others_sum
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        # Strategy 2: Move mass from peak contributors to random positions
        if not improved:
            for attempt in range(10):
                i_src = int(idx[top_contrib[np.random.randint(len(top_contrib))]])
                i_dst = np.random.randint(n)
                if i_dst == i_src:
                    continue
                move_frac = 0.1
                amount = a[i_src] * move_frac
                a_trial = a.copy()
                a_trial[i_src] -= amount
                a_trial[i_dst] += amount
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                a_trial = a_trial / s_trial
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        # Strategy 3: Small random perturbation
        if not improved:
            for attempt in range(5):
                scale = 0.001 + 0.001 * (no_improve / 50)
                a_trial = a + np.random.standard_normal(n) * scale
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        steps += 1
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

        if verbose and steps % 500 == 0:
            elapsed = monotonic() - t0
            print(f"  PeakShave step={steps} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP Fictitious Play with exponential weighting
    2. Cyclic Coordinate Descent (focused on peak contributors)
    3. L-BFGS soft-max polish at multiple temperatures
    4. Aggressive peak shaving
    5. Multi-length exploration
    6. Final polish
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting score: {best_score:.6f}")

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: FP Exponential (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t_fp = min(rem * 0.25, 400.0)
    if t_fp > 30:
        print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
        a_fp, score_fp, iters = fictitious_play_exp(n_parent, t_fp, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}_it{iters}", best_seq, best_score)

    # === Phase 2: CCD Focused ===
    print(f"\n=== Phase 2: CCD Focused (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t_ccd = min(rem * 0.25, 350.0)
    if t_ccd > 20:
        print(f"  CCD n={len(best_seq)}, budget={t_ccd:.0f}s")
        a_ccd, _ = cyclic_cd_focused(best_seq, t_ccd, top_k=60, verbose=True)
        best_seq, best_score = check_and_update(a_ccd, "ccd_focused", best_seq, best_score)

    # === Phase 3: L-BFGS with multiple T ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    for T in [20, 50, 150]:
        rem = deadline - monotonic()
        if rem < 15:
            break
        t_lb = min(rem * 0.15, 80.0)
        if t_lb < 10:
            break
        print(f"  L-BFGS T={T}, budget={t_lb:.0f}s")
        a_lb = lbgfs_smooth(best_seq, t_lb, T=T, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t_ps = min(rem * 0.2, 200.0)
    if t_ps > 15:
        print(f"  PeakShave budget={t_ps:.0f}s")
        a_ps, _ = peak_shave_aggressive(best_seq, t_ps, verbose=True)
        best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: More CCD ===
    print(f"\n=== Phase 5: CCD Round 2 (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t_ccd2 = min(rem * 0.2, 200.0)
    if t_ccd2 > 15:
        print(f"  CCD2 budget={t_ccd2:.0f}s")
        a_ccd2, _ = cyclic_cd_focused(best_seq, t_ccd2, top_k=40, verbose=True)
        best_seq, best_score = check_and_update(a_ccd2, "ccd_round2", best_seq, best_score)

    # === Phase 6: Full CCD sweep ===
    print(f"\n=== Phase 6: Full CCD (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    t_ccd3 = min(rem * 0.15, 120.0)
    if t_ccd3 > 10:
        print(f"  FullCCD budget={t_ccd3:.0f}s")
        a_ccd3, _ = cyclic_coordinate_descent(best_seq, t_ccd3, verbose=True)
        best_seq, best_score = check_and_update(a_ccd3, "full_ccd", best_seq, best_score)

    # === Phase 7: Try alternative length ===
    print(f"\n=== Phase 7: Alt Length (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 100:
        n_best = len(best_seq)
        # Try a nearby length
        alt_n = n_best + 64 if n_best < 1024 else n_best - 64
        if 256 <= alt_n <= 2048 and alt_n != n_best:
            t_alt = min(rem * 0.3, 200.0)
            print(f"  Alt length n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play_exp(alt_n, min(t_alt * 0.5, 80.0), initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"alt_fp_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:

                if t_ccd_alt > 10:
                    a_alt2, _ = cyclic_cd_focused(best_seq, t_ccd_alt, top_k=50, verbose=False)
                    best_seq, best_score = check_and_update(a_alt2, f"alt_ccd_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final L-BFGS + peak shave ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 15:
        t_final_lb = min(rem * 0.3, 40.0)
        a_fin = lbgfs_smooth(best_seq, t_final_lb, T=30, maxiter=2000)
        best_seq, best_score = check_and_update(a_fin, "final_lbgfs", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 10:
        t_final_ps = min(rem * 0.4, 25.0)
        a_fin2, _ = peak_shave_aggressive(best_seq, t_final_ps, verbose=False)
        best_seq, best_score = check_and_update(a_fin2, "final_ps", best_seq, best_score)

    # Final CCD squeeze
    rem = deadline - monotonic()
    if rem > 8:
        t_final_ccd = min(rem * 0.5, 15.0)
        a_fin3, _ = cyclic_cd_focused(best_seq, t_final_ccd, top_k=30, verbose=False)
        best_seq, best_score = check_and_update(a_fin3, "final_ccd", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
