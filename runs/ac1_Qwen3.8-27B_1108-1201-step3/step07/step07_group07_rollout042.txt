def fast_convolve(a):

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        a = a_new

            no_improve = 0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with 
log-sum-exp smooth max, LP refinement, and progressive length expansion.

Strategy:
- Polish parent sequence with fast gradient-based optimization
- Use LP for principled peak reduction
- Expand length (interpolate + re-optimize) to approach theoretical limit of ~1.5
- Multiple restarts for diversity
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_max_b(a):
    """Return max of auto-convolution (assumes sum(a)=1)."""
    b = fast_convolve(a)
    return float(b.max())

def project_simplex(v, target_sum=1.0):
    """Projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, weights):
    """Gradient of sum_k weights[k] * conv(a,a)[k]."""
    n = len(a)
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(weights, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * corr

def smooth_max_and_weights(a, beta):
    """Compute smooth max and weights via log-sum-exp."""
    b = fast_convolve(a)
    b_max = b.max()
    if b_max < 1e-30:
        return b_max, np.ones(len(b)) / len(b)
    scaled = beta * (b - b_max)
    exp_scaled = np.exp(np.clip(scaled, -50, 0))
    w_sum = np.sum(exp_scaled)
    weights = exp_scaled / w_sum
    smooth_val = b_max + np.log(w_sum) / beta
    return smooth_val, weights

def adam_smooth_max(a0, time_limit, seed=42, lr_init=0.005, 
                     beta_start=30, beta_end=500, beta_growth=1.01,
                     m1=0.6, m2=0.999, restart_every=500):
    """
    Adam optimizer minimizing smooth max(conv(a,a)) s.t. a>=0, sum(a)=1.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    eps = 1e-8
    lr = lr_init

    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        _, weights = smooth_max_and_weights(a, beta)
        grad = compute_gradient(a, weights)
        
        m = m1 * m + (1 - m1) * grad
        v = m2 * v + (1 - m2) * grad ** 2
        m_hat = m / (1 - m1 ** t)
        v_hat = v / (1 - m2 ** t)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = project_simplex(a - lr * update)
        obj_new = eval_max_b(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        beta = min(beta * beta_growth, beta_end)
        
        # Adaptive learning rate
        if no_improve > restart_every:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(lr * 0.5, 1e-5)
            no_improve = 0
            beta = beta_start
        elif no_improve > restart_every // 2:
            lr *= 0.95
        elif no_improve > restart_every // 4:
            lr *= 0.98

    return best_a, best_obj

def lp_refine(f, max_iters=20, time_limit=60.0, verbose=False):

    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)

    best_a = a.copy()
    best_obj = eval_max_b(a)

        b = fast_convolve(a)
        M = b.max()

        rows = []
        cols = []
        data = []

            rows.extend(r_range)
            cols.extend([j] * n)
            data.extend(f_list)
        
        A = csr_matrix((data, (rows, cols)), shape=(n_constraints, n))
        b_lp = np.full(n_constraints, M)
        c_obj = -np.ones(n)

        try:

            if not res.success:
                break
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum

            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()

                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.2f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break
        except Exception:
            break

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

    return f_interp


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def update_best_state(best_seq, best_score, a, label):
    """Try to update best. Returns (new_best_seq, new_best_score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses progressive length expansion with Adam+LP optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 40.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1024)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using uniform n=1024, score={best_score:.6f}")

    # === Phase 1: Adam polish on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam polish on parent (n={len(parent)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 900
    if rem > 20:
        t1 = min(rem, 200.0)
        print(f"  Time: {t1:.0f}s")
        a1, _ = adam_smooth_max(parent, time_limit=t1, seed=seed,
                                lr_init=0.004, beta_start=25)
        best_seq, best_score = update_best_state(best_seq, best_score, a1, "p1_adam")

    # === Phase 3: Expand to 2x ===

    print(f"\n{'='*60}")
    print(f"Phase 3: Expand to n={n3_target}")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 400
    if rem > 20 and n3_target > len(best_seq):

        t3 = min(rem, 200.0)
        print(f"  Time: {t3:.0f}s")

                                lr_init=0.003, beta_start=20)
        best_seq, best_score = update_best_state(best_seq, best_score, a3, f"p3_n{n3_target}")

    # === Phase 4: LP at expanded length ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP at n={len(best_seq)}")
    print(f"{'='*60}")

    print(f"{'='*60}")

    return [float(x) for x in best_seq]
