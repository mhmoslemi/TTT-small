def simplex_project(v):
    """Project v onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_convolve(a):

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def simplex_project(v):
    """Project v onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_list = np.nonzero(cond)[0]
    if len(rho_list) == 0:
        return np.ones(n) / n
    rho = rho_list[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smoothed_grad(a, T):
    """Compute smoothed gradient using log-sum-exp approximation."""
    n = len(a)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()

    # = 2 * cross_corr(w, a) [linear version]
    size = 2 * n - 1

    W = rfft(w, fft_size)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_subgradient(a_init, time_limit, T_schedule=None, verbose=True, label=""):
    """Projected subgradient with Nesterov acceleration."""
    n = len(a_init)
    a = simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    alpha = 0.0005
    t_k = 1.0
    prev_a = a.copy()
    no_improve = 0
    max_no_improve = 2000

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        t_k_new = (1 + np.sqrt(1 + 4 * t_k ** 2)) / 2
        beta = (t_k - 1) / t_k_new
        y = a + beta * (a - prev_a)
        y = np.maximum(y, 0)
        y = y / max(y.sum(), 1e-15)
        t_k = t_k_new

        # Compute gradient (smoothed for stability)
        if T_schedule is not None:
            progress = min(1.0, it / 50000)
            T = T_schedule[0] * (T_schedule[1] / T_schedule[0]) ** progress
        else:
            T = 0.5

        b = fast_convolve(y)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Get gradient
        if T < 0.1:
            k_peak = int(np.argmax(b))
            g = subgrad_max_b(y, k_peak)
        else:
            g = smoothed_grad(y, T)

        # Project gradient onto tangent space
        g = g - np.mean(g)

        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        a_new = simplex_project(y - alpha * g)

        # Check improvement
        sc_new = score_from_a(a_new)
        if sc_new < current_score + 1e-12:
            a = a_new
        else:
            alpha *= 0.7
            a = a_new  # Still move, just smaller next time

        if no_improve > max_no_improve:
            # Restart with perturbation
            a = best_a + np.random.randn(n) * 0.005
            a = simplex_project(a)
            alpha = 0.001
            no_improve = 0
            t_k = 1.0
            prev_a = a.copy()

        if verbose and it % 10000 == 0:

        # Gradient of Var(P) w.r.t. a
        # P[k] = |A[k]|^2, dP[k]/da = 2*Re(conj(A[k]) * dA[k]/da)
        # dA[k]/da[i] = exp(-2*pi*i*k/fft_size)
        # This is complex, so let's use a simpler approach
        # Just do coordinate descent on spectral variance

        # Instead, use the actual objective with a different step strategy
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        g = subgrad_max_b(a, k_peak)
        g = g - np.mean(g)

        a_new = simplex_project(a - alpha * g)
        sc = score_from_a(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                alpha *= 0.8

        if no_improve > 2000:
            a = best_a + np.random.randn(n) * 0.01
            a = simplex_project(a)
            no_improve = 0
            alpha = 0.01

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def ultra_fine_coord(a_init, time_limit, verbose=True, label=""):
    """Ultra-fine coordinate descent with very small factors."""
    n = len(a_init)
    a = simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    # Very fine factors
    factors = [0.995, 0.997, 0.998, 0.999, 1.001, 1.002, 1.003, 1.005]
    no_improve = 0
    max_no_improve = 100
    pass_count = 0

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            if a[i] < 1e-15:
                continue

            for f in factors:
                new_val = a[i] * f
                if new_val < 0 or 1.0 - new_val < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                target = 1.0 - new_val
                if others_sum < 1e-15 or target < 1e-12:
                    continue
                a_trial[mask] *= target / others_sum
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-15:
                    best_score = sc
                    a = a_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1

        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} pass={pass_count} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def multi_peak_equalize(a_init, time_limit, verbose=True, label=""):
    """Try to equalize the top few peaks to reduce the maximum."""
    n = len(a_init)
    a = simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for rnd in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top 3 peaks
        b_sorted = np.sort(b)[::-1]
        top3_vals = b_sorted[:3]

        # For the peak, find contributors and try to reduce them
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            # Binary search for optimal reduction factor
            lo, hi = 0.5, 1.0
            best_f = 1.0
            for _ in range(10):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[i] *= mid
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                sc = score_from_a(a_trial)
                if sc < best_score:
                    best_f = mid
                    hi = mid
                else:
                    lo = mid

            if best_f < 1.0:
                a_trial = a.copy()
                a_trial[i] *= best_f
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-15:
                    best_score = sc
                    a = a_trial
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

        if verbose and rnd % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} rnd={rnd} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses projected subgradient, coordinate descent, 2D moves, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n=768, score={best_score:.7f}")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")

    # === Phase 1: Nesterov subgradient (30% time) ===
    print("\n=== Phase 1: Nesterov Subgradient ===")
    rem = deadline - time.monotonic()
    t1 = rem * 0.30
    if t1 > 30:
        a_res, _ = nesterov_subgradient(
            best_seq, t1, T_schedule=(5.0, 0.3), label="Nest"
        )
        check_update(a_res, "nesterov")

    # === Phase 2: 1D Coordinate Descent (30% time) ===
    print(f"\n=== Phase 2: 1D Coord Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t2 = rem * 0.30
    if t2 > 30:
        a_res, _ = coord_descent_1d(best_seq, t2, label="CD1")
        check_update(a_res, "coord_1d")

    # === Phase 3: 2D Peak-Targeted Moves (15% time) ===
    print(f"\n=== Phase 3: 2D Peak Moves (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t3 = rem * 0.15
    if t3 > 20:
        a_res, _ = peak_targeted_2d(best_seq, t3, label="2D")
        check_update(a_res, "2d_peak")

    # === Phase 4: Multi-peak equalization (10% time) ===
    print(f"\n=== Phase 4: Multi-Peak Equalize (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t4 = rem * 0.10
    if t4 > 15:
        a_res, _ = multi_peak_equalize(best_seq, t4, label="Eq")
        check_update(a_res, "multi_eq")

    # === Phase 5: Alternative lengths (10% time) ===
    print(f"\n=== Phase 5: Alternative Lengths (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = [720, 832, 896, 1024]
        for alt_n in alt_lengths:
            if time.monotonic() > deadline - 30:
                break
            if alt_n == n_best:
                continue
            t_alloc = min((deadline - time.monotonic()) * 0.4, 50.0)
            if t_alloc < 15:
                continue
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = coord_descent_1d(init_alt, t_alloc, label=f"n{alt_n}")
            check_update(a_alt, f"alt_n{alt_n}")

    # === Phase 6: Fine 1D coordinate descent (remaining time) ===
    print(f"\n=== Phase 6: Fine Coord Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t6 = rem * 0.50
    if t6 > 10:
        a_res, _ = coord_descent_1d(best_seq, t6, label="CD2")
        check_update(a_res, "coord_fine")

    # === Phase 7: Ultra-fine polish ===
    print(f"\n=== Phase 7: Ultra-Fine Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t7 = rem * 0.40
    if t7 > 8:
        a_res, _ = ultra_fine_coord(best_seq, t7, label="UF")
        check_update(a_res, "ultra_fine")

    # === Phase 8: Final 2D moves ===
    print(f"\n=== Phase 8: Final 2D (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t8 = rem * 0.50
    if t8 > 5:
        a_res, _ = peak_targeted_2d(best_seq, t8, label="Fin2D")
        check_update(a_res, "final_2d")

    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        print(f"\nFALLBACK: n=768")

    return [float(x) for x in best_seq]
