from scipy.fft import rfft, irfft

import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    """Build the matrix A where A @ g = f * g (full convolution)."""
    # A is (2n-1) x n
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    return A

        return None, M, None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_conv_matrix(f, n):
    """Build sparse matrix A where A @ g = f * g (full linear convolution)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def solve_lp(f, n, top_k=None, solver=None, eps=1e-9):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for selected k.
    Returns (g, M, sum_g, duals).
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M, 0.0, None
    
    if top_k is None or top_k >= 2*n - 1:
        idx = np.arange(2*n-1)
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
    
    # Build constraint matrix (only for selected rows)
    rows = []
    cols = []
    vals = []
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(idx), n))
    rhs = np.full(len(idx), M)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= rhs]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    if solver is None:
        solver = cp.SCS
    
    try:
        if solver == cp.SCS:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
        elif solver == cp.ECOS:
            prob.solve(solver=cp.ECOS, abs_acc=eps, rel_acc=eps, max_iters=100000)
        else:
            prob.solve(solver=solver)
    except Exception as e:
        try:
            prob.solve(solver=cp.SCS, max_iters=1000000, eps=1e-7)
        except Exception:
            return None, M, 0.0, None
    
    if g.value is None:
        return None, M, 0.0, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M, 0.0, None
    
    duals = None
    try:
        d = prob.constraints[1].dual_value
        if d is not None:
            duals = np.array(d)
    except Exception:
        pass
    
    return g_val, M, g_val.sum(), duals


def line_search(f, h, obj_current, n_evals=12):
    """Adaptive line search: log-spaced grid + refinement."""
    best_t = 0.05
    best_obj = obj_current
    
    # Log-spaced grid
    ts = np.logspace(np.log10(0.01), np.log10(0.8), n_evals)
    
    for t in ts:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t, best_obj

2. FFT-based gradient computation for fast subgradient
3. Adaptive line search with log-spaced grid
4. Exponential (multiplicative) subgradient polish
5. Multi-length exploration with interpolation
6. Periodic perturbations to escape local optima
7. Better time management and convergence detection
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy import sparse
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_conv_rows(f, n, idx):
    """Build sparse CSR matrix for selected convolution rows."""
    rows = []
    cols = []
    vals = []
    for ri, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(ri)
            cols.append(i)
            vals.append(f[i])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(len(idx), n))


def solve_lp(f, n, top_k=None, solver_name="SCS", eps=1e-9):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for selected k.
    Returns (g_val, M, sum_g, duals) or (None, M, 0, None) on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M, 0.0, None

    if top_k is None or top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
        idx = np.sort(idx)

    A = build_sparse_conv_rows(f, n, idx)
    rhs = np.full(len(idx), M)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= rhs]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    try:
        if solver_name == "ECOS":
            prob.solve(solver=cp.ECOS, abs_acc=1e-8, rel_acc=1e-8, max_iters=100000)
        elif solver_name == "SCS":
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
        else:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
        solved = True
    except Exception:
        pass

    if not solved or g.value is None:
        try:
            prob.solve(solver=cp.SCS, max_iters=1000000, eps=1e-7)
            solved = True
        except Exception:
            pass

    if not solved or g.value is None:
        return None, M, 0.0, None

    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M, 0.0, None

    duals = None
    try:
        d = prob.constraints[1].dual_value
        if d is not None:
            duals = np.asarray(d, dtype=np.float64)
    except Exception:
        pass

    return g_val, M, float(g_val.sum()), duals


def line_search(f, h, obj_ref, n_grid=16):
    """Log-spaced line search with refinement."""
    best_t = 0.05
    best_obj = obj_ref

    ts = np.logspace(np.log10(0.01), np.log10(0.7), n_grid)
    for t in ts:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = float(t)

    # Refine around best
    if best_t > 0.01:
        ts_refine = np.linspace(best_t * 0.5, min(best_t * 1.5, 0.9), 8)
        for t in ts_refine:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = float(t)

    return best_t, best_obj


def run_lp_phase(f_init, time_limit, label="lp", top_k=None, use_full=True):
    """Main LP optimization loop with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    current_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        # Adaptive: if stuck, try more peaks
        if stuck > 5 and current_top_k is not None and current_top_k < 2 * n - 1:
            current_top_k = min(current_top_k * 2, 2 * n - 1)
            stuck = 0

        tk = 2 * n - 1 if use_full else current_top_k
        g, M, S_g, duals = solve_lp(f, n, top_k=tk, solver_name="SCS", eps=1e-9)

        if g is None:
            stuck += 1
            if stuck > 15:
                break
            continue

        if S_g <= 1.0 + 5e-6:
            stuck += 1
            if stuck > 30:
                break
            if use_full and current_top_k is not None and current_top_k < 2 * n - 1:
                current_top_k = min(2 * current_top_k, 2 * n - 1)
            continue

        stuck = 0
        h = g / S_g
        best_t, best_obj_trial = line_search(f, h, best_obj)

        if best_t < 0.003:
            stuck += 1
            if stuck > 20:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0 * n * best_obj:.8f} t={elapsed:.0f}s S_g={S_g:.4f}")

    return best_f, best_obj


def compute_conv_gradient(f, top_idx, n):
    """Compute subgradient of max-convolution using FFT."""
    indicator = np.zeros(2 * n - 1)
    indicator[top_idx] = 1.0
    fft_size = 1 << (3 * n - 1).bit_length()
    F = np.fft.fft(f, fft_size)
    I = np.fft.fft(indicator, fft_size)
    corr = np.fft.ifft(I * np.conj(F)).real[:n]
    return 2.0 * corr


def exponential_subgradient(f_init, time_limit, label="exp_sg"):
    """Exponential (multiplicative) subgradient with momentum and restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    lr = 0.005
    top_k = min(80, n)
    it = 0
    no_improve = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if top_k < 2 * n - 1:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(2 * n - 1)

        grad = compute_conv_gradient(f, top_idx, n)

        v = 0.7 * v - lr * grad
        f = f * np.exp(v)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = best_f.copy()
            v = np.zeros(n)

        if it % 200 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.92
                    no_improve = 0

            if no_improve > 15:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                lr = 0.005

            if it % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} iter={it} obj={best_obj:.12f} "
                      f"score={2.0 * n * best_obj:.8f} lr={lr:.5f} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Simple coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.6, 0.8, 0.9, 1.1, 1.25, 1.5])
    improved_any = True

    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved_any = True

    return f, best_obj

    # === Phase 2: Reduced LP (adaptive top-k, can escape if full LP stuck) ===

    # === Phase 4: Multi-length exploration ===
