```python
def lp_linearization_step(a, b, M_current):
    """
    Solve the linearized LP step.
    a: current sequence (normalized, sum=1)
    b: current auto-convolution (length 2n-1)
    Returns: optimal δ (length n) and t (new max estimate)
    """
    n = len(a)
    num_peaks = 2 * n - 1
    
    # Build constraint matrix
    # Variables: [t, δ_0, δ_1, ..., δ_{n-1}]
    # Total variables: n + 1
    
    # A_ub will have num_peaks + n rows
    A_ub = np.zeros((num_peaks + n, n + 1))
    b_ub = np.zeros(num_peaks + n)
    
    # Peak constraints: -t + 2*sum_i a_{k-i}*δ_i <= -b_k
    for k in range(num_peaks):
        A_ub[k, 0] = -1.0  # coefficient of t
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[k, i + 1] = 2.0 * a[k - i]
        b_ub[k] = -b[k]
    
    # Non-negativity: -δ_j <= a_j
    for j in range(n):
        A_ub[num_peaks + j, j + 1] = -1.0
        b_ub[num_peaks + j] = a[j]
    
    # Equality: sum δ_j = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] + [(None, None)] * n,
                  method='highs')
    
    if not res.success:
        return None, None
    
    t_new = res.x[0]
    delta = res.x[1:]
    
    return delta, t_new
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP successive linearization.

Core idea: Linearize the auto-convolution peaks around the current point,
solve an LP to find the optimal direction, take a damped step, and repeat.
This is a convex optimization approach that converges reliably.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    return float(fast_convolve(a).max())

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T

def lp_successive_linearization(a_init, time_limit, max_n_full_lp=2048, verbose=False):
    """
    Main optimization: successive LP linearization with damping.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    # Pre-compute convolution matrix (changes only when a changes)
    T = build_conv_matrix(a)
    
    # Prepare LP structure
    num_peaks = 2 * n - 1
    nvar = n + 1
    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])
    c_obj = np.zeros(nvar)
    c_obj[0] = 1.0
    
    # Determine peak subset strategy
    use_full = n <= max_n_full_lp
    
    no_improve = 0
    it = 0
    alpha = 1.0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Update T if a changed significantly
        T = build_conv_matrix(a)
        
        # Build A_ub
        if use_full:
            peak_idx = np.arange(num_peaks)
        else:
            n_top = min(num_peaks, 500)
            peak_idx = np.argpartition(b, -n_top)[-n_top:]
        
        npeaks = len(peak_idx)
        A_ub = np.zeros((npeaks + n, nvar))
        b_ub = np.zeros(npeaks + n)

        A_ub[:npeaks, 0] = -1.0
        A_ub[:npeaks, 1:] = 2.0 * T[peak_idx, :]
        b_ub[:npeaks] = -b[peak_idx]

        idx = np.arange(n)
        A_ub[npeaks:, idx + 1] = -1.0  # This creates a diagonal
        b_ub[npeaks:] = a

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                use_full = False
                continue
            else:

        delta = res.x[1:]
        t_predicted = res.x[0]

        best_actual_obj = b_max

            a_trial = np.maximum(a_trial, 0.0)

            if s_trial < 1e-15:

                best_alpha = test_alpha

            a = a / a.sum()
            it += 1
            
            if best_actual_obj < best_obj - 1e-12:

                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if verbose and it % 100 == 0:

            no_improve = 0

        # Check time
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_obj
```
