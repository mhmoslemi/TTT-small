def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_conv(w, a, fft_size):
    """
    Compute g[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0..n-1.
    This is the linear cross-correlation of w and a at lags 0..n-1.
    """
    corr = irfft(rfft(w, fft_size) * np.conj(rfft(a, fft_size)), fft_size)
    return corr[:len(a)]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_score_norm(a):
    """Score with sum(a)=1 assumed."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_obj_and_grad(a, T, fft_size):
    """Compute soft-max objective value and gradient (with sum(a)=1)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    S = exp_b.sum()
    w = exp_b / S
    obj = 2.0 * n * T * (max_b / T + np.log(S))
    # grad = 4n * sum_j w[i+j] * a[j] (with sum=1 normalization, the 
    # normalization gradient is a constant that doesn't affect the direction
    # when we project onto the simplex)
    grad = 4.0 * n * compute_grad_conv(w, a, fft_size)
    return obj, grad

            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            contribs = a[idx] * a[k - idx]

def coordinate_descent_exact(a0, time_limit, n_sweeps=100):
    """
    Coordinate descent with exact 1D line search for each coordinate.
    """

            if rest_sum < 1e-15:
                continue
            
            best_val = a[i]
            best_local_score = best_score

            for val in [a[i] * f for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]]:

                a_trial[i] = val
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s

                    best_val = val

def interpolate_sequence(f, new_n):
    """Interpolate a probability vector to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp = f_interp / s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid soft-max Adam, exact
subgradient descent on the non-smooth max function, intensive pairwise
peak-shaving with adaptive line search, and multi-length exploration.
Key improvements: (1) exact subgradient steps targeting the true peak,
(2) pairwise optimization with redistribution to safe positions,
(3) adaptive switching between smooth and non-smooth methods,
(4) multi-length transfer via interpolation.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_corr(w, a, fft_size):
    """
    Compute g[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0..n-1.
    Linear cross-correlation of w (length 2n-1) with a (length n).
    """
    corr = irfft(rfft(w, fft_size) * np.conj(rfft(a, fft_size)), fft_size)
    return corr[:len(a)]


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_score_norm(a):
    """Score with sum(a)≈1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_obj_grad(a, T, fft_size):
    """Soft-max surrogate objective and gradient (on simplex, sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    S = exp_b.sum()
    w = exp_b / S
    obj = 2.0 * n * T * (max_b / T + np.log(S))
    grad = 4.0 * n * compute_corr(w, a, fft_size)
    return obj, grad


def subgrad_direction(a, fft_size):
    """
    Compute the exact subgradient of max(conv(a,a)) at point a.
    Returns the subgradient vector (before scaling by 2n).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    tol = max(1e-12, max_b * 1e-10)
    peak_mask = b >= (max_b - tol)
    peak_indices = np.nonzero(peak_mask)[0]

    subgrad = np.zeros(n)
    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            subgrad[i_start:i_end + 1] += a[k - np.arange(i_start, i_end + 1)]
    subgrad *= 2.0 / len(peak_indices)
    return subgrad, len(peak_indices)

    best_score = get_score_norm(a)
    no_improve = 0
    max_no_improve = 4000
    lr_base = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = max(T_final, T_init * (T_final / T_init) ** progress)

        _, grad_sm = softmax_obj_grad(a, T, fft_size)

        if use_subgrad_mix and t % 5 == 0:
            sg, n_peaks = subgrad_direction(a, fft_size)
            grad = 0.5 * grad_sm + 0.5 * (2.0 * n * sg)
        else:
            grad = grad_sm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        score = get_score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > max_no_improve:
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.008)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr_base * 0.6)

        lr = max(0.0003, lr * 0.99995)

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")

    return best_a, best_score


def subgrad_descent(a0, time_limit, lr_init=0.001, verbose=False):
    """
    Exact subgradient descent with backtracking line search.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64).copy())
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score_norm(a)
    no_improve = 0

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        sg, _ = subgrad_direction(a, fft_size)
        direction = -sg

        # Line search: try a few step sizes
        best_lr = lr
        best_a_trial = a
        best_s_trial = best_score

        for lr_try in [lr, lr * 0.5, lr * 0.2, lr * 0.1, lr * 2.0]:
            a_trial = project_simplex(a + lr_try * direction)
            s_trial = get_score_norm(a_trial)
            if s_trial < best_s_trial:
                best_s_trial = s_trial
                best_a_trial = a_trial
                best_lr = lr_try

        a = best_a_trial
        lr = best_lr * 1.05  # Slightly increase if it worked

        if get_score_norm(a) < best_score - 1e-14:
            best_score = get_score_norm(a)
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.01)
            no_improve = 0
            lr = lr_init * 0.5

        if verbose and t % 2000 == 0:
            print(f"      SubGrad t={t} lr={lr:.7f} score={best_score:.6f}")

    return best_a, best_score


def peak_shave_line_search(a0, time_limit, n_trials=20000):
    """
    Peak-targeted local search with adaptive line search for optimal
    redistribution amounts.
    """
    n = len(a0)
    a = project_simplex(np.maximum(np.asarray(a0, dtype=np.float64), 0))
    best_a = a.copy()
    best_score = get_score_norm(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Get top peak lags
        sorted_lags = np.argsort(b)[::-1]
        top_lags = sorted_lags[:3]

        improved = False

        for k in top_lags:
            if improved:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            contribs = a[idx] * a[k - idx]
            top_n = min(6, len(idx))
            top_local = np.argsort(contribs)[::-1][:top_n]

            for ci in top_local:
                if improved:
                    break
                i = int(idx[ci])
                j = int(k - i)

                if i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue

                # Try reducing a[i] and redistributing to safe positions
                for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor

                    # Find safe positions (don't contribute to top peaks)
                    safe_mask = np.ones(n, dtype=bool)
                    for ll in top_lags[:2]:
                        ll = int(ll)
                        ls = max(0, ll - n + 1)
                        le = min(ll, n - 1)
                        safe_mask[ls:le + 1] = False
                    safe_mask[i] = False
                    safe_mask[j] = False

                    safe_count = safe_mask.sum()
                    if safe_count < 3:
                        safe_mask = np.ones(n, dtype=bool)
                        safe_mask[i] = False
                        safe_mask[j] = False

                    safe_sum = a_trial[safe_mask].sum()
                    if safe_sum < 1e-14:
                        continue

                    a_trial[safe_mask] += diff * a_trial[safe_mask] / safe_sum
                    a_trial = project_simplex(a_trial)

                    score = get_score_norm(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break

        if not improved:
            if trial > 500 and trial % 200 == 0:
                # Escape via perturbation
                pert_size = 0.01 + 0.02 * (trial % 5) / 5.0
                a = project_simplex(best_a + rng.standard_normal(n) * pert_size)

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=60):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = project_simplex(np.maximum(np.asarray(a0, dtype=np.float64), 0))
    best_a = a.copy()
    best_score = get_score_norm(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        any_improve = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            best_val = a[i]
            best_local = best_score

            for factor in [0.75, 0.85, 0.92, 1.08, 1.15, 1.25]:
                val = a[i] * factor
                if val < 0 or val > 0.9:
                    continue
                a_trial = a.copy()
                a_trial[i] = val
                s = a_trial.sum()
                if s < 1e-14:
                    continue
                a_trial = a_trial / s
                score = get_score_norm(a_trial)
                if score < best_local:
                    best_local = score
                    best_val = factor

            if best_val != a[i]:
                a[i] = a[i] * best_val
                a = a / a.sum()
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()
                    any_improve = True

        if not any_improve:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a probability vector to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-14:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate via official evaluator and update best."""
    a_clean = np.maximum(a, 0.0)
    s = a_clean.sum()
    if s < 0.01:
        return best_seq, best_score
    a_eval = a_clean / s
    try:
        score = float(evaluate_sequence(a_eval.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_eval.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_eval)})")
    return best_seq, best_score


def full_optimize(a0, time_budget):
    """Run the full hybrid optimization pipeline."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = get_score_norm(a)
    t_start = time.monotonic()

    # Stage 1: Adam soft-max (50% of budget)
    t1 = time_budget * 0.45
    if t1 > 20:
        a, sc = adam_simplex(a, t1, T_init=0.04, T_final=0.002,
                             lr=0.006, schedule='hold_anneal',
                             use_subgrad_mix=True)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 2: Subgradient (20%)
    rem = time_budget - (time.monotonic() - t_start)
    t2 = rem * 0.22
    if t2 > 15:
        a, sc = subgrad_descent(a, t2, lr_init=0.0005)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 3: Peak shave (20%)
    rem = time_budget - (time.monotonic() - t_start)
    t3 = rem * 0.35
    if t3 > 15:
        a, sc = peak_shave_line_search(a, t3, n_trials=50000)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 4: Coordinate descent (remaining)
    rem = time_budget - (time.monotonic() - t_start)
    if rem > 10:
        a, sc = coordinate_descent(a, rem, n_sweeps=40)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score
