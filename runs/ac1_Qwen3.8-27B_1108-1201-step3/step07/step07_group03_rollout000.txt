def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(w, a):
    """Compute cross-correlation: result[m] = sum_j w[m+j]*a[j] for m=0..n-1."""
    n = len(a)
    nw = len(w)
    N = 1 << (nw + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(A) * W, N)
    return corr[:n]

def softmax_stable(v, T):
    """Stable softmax with temperature."""
    v_shift = v - v.max()
    e = np.exp(v_shift / T)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

                a_new = e / e.sum()
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    score = 2.0 * n * fast_convolve(a_new).max()
                    if score < best_score - 1e-10:
                        best_score = score
                        best_a = a_new.copy()
                        if verbose:
                            print(f"    L-BFGS round {n_rounds}: score={best_score:.8f}, T={T_current:.2e}")
                x = x_new
        except Exception:
            pass
        
        # Decrease temperature
        T_current *= 0.6
        if T_current < T_min:
            T_current = T_min
        
        # Check convergence
        if n_rounds > 30:
            break
    
    return best_a, best_score

def multiplicative_weights(a0, time_limit, T=0.001, verbose=False):
    """Mirror descent with entropy Bregman divergence (multiplicative weights)."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    T_current = T
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        T_curr = max(T_current, b.max() * 0.01)
        w = softmax_stable(b, T_curr)
        
        # Gradient: (2/T) * cross_corr(w, a)
        grad = (2.0 / T_curr) * cross_corr(w, a)
        
        # Multiplicative update: a_new = a * exp(-eta * grad) / sum
        a_new = a * np.exp(-eta * grad)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay eta and T
        eta *= 0.9995
        T_current *= 0.999
        
        if verbose and it % 1000 == 0:
            print(f"    MW it={it}: score={best_score:.8f}, eta={eta:.6e}")
    
    return best_a, best_score

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed):
    """Given fixed a (sum=1), find b minimizing max conv(a,b)."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def fictitious_play(a0, time_limit, verbose=False):
    """Alternating LP best responses."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    b = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        b_new = solve_lp_best_response(a)
        if b_new is not None:
            b = b_new
        
        a_new = solve_lp_best_response(b)
        if a_new is not None:
            a = a_new
        
        # Check candidates
        for cand in [a, b, 0.5*(a+b)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_n.copy()
        
        if verbose and iters % 10 == 0:
            print(f"    FP it={iters}: score={best_score:.8f}")
    
    return best_a, best_score, iters

def coord_descent(a0, time_limit, n_sweeps=500):
    """Coordinate descent with multiple trial factors."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    factors_coarse = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    factors_fine = [0.97, 0.985, 1.015, 1.03]
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            
            # Try coarse factors
            for factor in factors_coarse:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                sc = a.sum()
                if sc < 1e-15:
                    a[i] = old_val
                    continue
                a_tmp = a / sc
                score = 2.0 * n * fast_convolve(a_tmp).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            
            # Try fine factors if no coarse improvement
            if not improved and old_val > 1e-10:
                for factor in factors_fine:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    sc = a.sum()
                    if sc < 1e-15:
                        a[i] = old_val
                        continue
                    a_tmp = a / sc
                    score = 2.0 * n * fast_convolve(a_tmp).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a_tmp.copy()
                        a = a_tmp.copy()
                        old_val = a[i]
                        improved = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
        
        if not improved and sweep > 20:
            break
    
    return best_a, best_score

def peak_shave(a0, time_limit):
    """Targeted peak reduction by adjusting top contributors."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peaks = np.argsort(b)[::-1][:3]
        
        improved = False
        for pk in peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            
            if contribs.max() < 1e-15:
                continue
            
            # Top contributors
            top_c = np.argsort(contribs)[::-1][:5]
            
            for ci in top_c:
                i = int(idx[ci])
                j = pk - i
                if a[i] < 1e-14:
                    continue
                
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    sc = a_trial.sum()
                    if sc < 1e-15:
                        continue
                    a_trial = a_trial / sc
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved and i != j and a[j] >= 1e-14:
                for f1 in [0.85, 0.75]:
                    for f2 in [0.85, 0.75]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        sc = a_trial.sum()
                        if sc < 1e-15:
                            continue
                        a_trial = a_trial / sc
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng, n_harmonics=None, amp_scale=0.3):
    """Fourier-based initialization."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(3, 15))
    amplitudes = rng.random(n_harmonics) * amp_scale
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * (j + 1) * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_ultraparabolic(n, rng):
    """Initialize with ultra-parabolic (Tukey-like) window."""
    r = rng.uniform(0.2, 0.8)
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    # Apply window
    for i in range(n):
        if t[i] < r / 2:
            a[i] = 0.5 * (1 + np.cos(np.pi * t[i] / (r / 2)))
        elif t[i] > 1 - r / 2:
            a[i] = 0.5 * (1 + np.cos(np.pi * (t[i] - (1 - r / 2)) / (r / 2)))
    a = np.maximum(a, 1e-6)
    s = a.sum()
    return a / s

def get_parent():
    """Try to get the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            return p / p.sum()
    except Exception:
        pass
    return None

def check_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-start L-BFGS with diverse initializations
    Phase 2: LP fictitious play
    Phase 3: L-BFGS refinement with decreasing temperature
    Phase 4: Multiplicative weights
    Phase 5: Coordinate descent + peak shaving
    Phase 6: Final polish
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(2024)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent
    parent = get_parent()
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_score):
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    
    # === Phase 1: Multi-start L-BFGS ===
    print("\n=== Phase 1: Multi-start L-BFGS ===")
    t1 = time.monotonic()
    t_phase1 = 200.0
    
    n_values = [512, 640, 768, 896, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
    
    phase1_results = []
    
    for n in n_values:
        if time.monotonic() - t1 > t_phase1:
            break
        remaining = t_phase1 - (time.monotonic() - t1)
        if remaining < 10:
            break
        
        t_per_init = min(remaining / 4, 15.0)
        
        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('fourier1', init_fourier(n, rng, n_harmonics=5, amp_scale=0.25)))
        inits.append(('fourier2', init_fourier(n, rng, n_harmonics=10, amp_scale=0.4)))
        inits.append(('ultra', init_ultraparabolic(n, rng)))
        
        for name, init_a in inits:
            elapsed = time.monotonic() - t1
            if elapsed > t_phase1:
                break
            t_alloc = min(t_per_init, t_phase1 - elapsed)
            if t_alloc < 2:
                continue
            
            print(f"  L-BFGS n={n} init={name} budget={t_alloc:.0f}s")
            a_opt, s_opt = softmax_lbfgs(init_a, t_alloc, verbose=False)
            phase1_results.append((s_opt, a_opt, n, name))
    
    phase1_results.sort(key=lambda x: x[0])
    for s_c, a_c, n_c, name_c in phase1_results[:3]:
        best_seq, best_score = check_update(a_c, best_seq, best_score, f"p1_n{n_c}_{name_c}")
    
    print(f"  Phase 1 done, best={best_score:.8f}")
    
    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_fp = min(rem * 0.5, 200.0)
            print(f"  FP n={n_best} budget={t_fp:.0f}s")
            a_fp, s_fp, iters = fictitious_play(best_seq, t_fp, verbose=True)
            best_seq, best_score = check_update(a_fp, best_seq, best_score, f"fp_n{n_best}")
            
            # Also try at a different length
            alt_n = int(n_best * 1.33)
            if 300 <= alt_n <= 2000 and alt_n != n_best:
                rem = deadline - time.monotonic() - 500
                if rem > 20:
                    t_fp2 = min(rem * 0.3, 60.0)
                    init_alt = interpolate_seq(best_seq, alt_n)
                    a_fp2, s_fp2, _ = fictitious_play(init_alt, t_fp2)
                    best_seq, best_score = check_update(a_fp2, best_seq, best_score, f"fp_alt_n{alt_n}")
    
    # === Phase 3: L-BFGS refinement with decreasing T ===
    print(f"\n=== Phase 3: L-BFGS Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_lbf = min(rem * 0.5, 200.0)
            print(f"  L-BFGS refine n={len(best_seq)} budget={t_lbf:.0f}s")
            a_lb, s_lb = softmax_lbfgs(best_seq, t_lbf, T_init=None, T_min=1e-6)
            best_seq, best_score = check_update(a_lb, best_seq, best_score, "lbfgs_refine")
    
    # === Phase 4: Multiplicative Weights ===
    print(f"\n=== Phase 4: Multiplicative Weights (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 15:

            print(f"  MW budget={t_mw:.0f}s")

            best_seq, best_score = check_update(a_mw, best_seq, best_score, "mw")
    
    # === Phase 5: Coordinate Descent + Peak Shaving ===
    print(f"\n=== Phase 5: Coord Descent + Peak Shave (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

            t_cd = min(rem * 0.5, 100.0)
            print(f"  CoordDescent budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent(best_seq, t_cd)
            best_seq, best_score = check_update(a_cd, best_seq, best_score, "coord_desc")
        
        rem = deadline - time.monotonic() - 100

            t_ps = min(rem * 0.5, 50.0)
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_update(a_ps, best_seq, best_score, "peak_shave")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # L-BFGS with very small T
        rem = deadline - time.monotonic() - 30
        if rem > 10:

            print(f"  Final L-BFGS budget={t_fin1:.0f}s")
            a_f1, s_f1 = softmax_lbfgs(best_seq, t_fin1, T_min=1e-7)
            best_seq, best_score = check_update(a_f1, best_seq, best_score, "final_lbfgs")
        
        # Another coord descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_fin2 = min(rem * 0.5, 30.0)
            print(f"  Final CD budget={t_fin2:.0f}s")
            a_f2, s_f2 = coord_descent(best_seq, t_fin2, n_sweeps=200)
            best_seq, best_score = check_update(a_f2, best_seq, best_score, "final_cd")
        
        # Peak shave
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_f3, _ = peak_shave(best_seq, min(rem * 0.6, 15.0))
            best_seq, best_score = check_update(a_f3, best_seq, best_score, "final_ps")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
