Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: LP-based iterative refinement (SOTA approach using cvxpy).

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k.
    Returns g (unnormalized) or None if infeasible."""
    n = len(a)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]
    
    # Build constraints: (a*g)[k] <= M
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            coeff = a[i_min:i_max + 1]
            idx = k - np.arange(i_min, i_max + 1)
            constraints.append(coeff @ g[idx] <= M)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except:
            return None
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0)


def lp_optimize(a_init, time_limit, verbose=True):
    """LP-based iterative optimization.
    
    Iteratively solves LP to find direction g, then line searches
    to update a <- (1-t)*a + t*g.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    
    M_fraction = 1.0
    stall_count = 0
    max_stall = 50
    
    for it in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Solve LP
        g_raw = lp_direction(a, M * M_fraction)
        if g_raw is None:
            stall_count += 1
            M_fraction *= 0.9
            if M_fraction < 0.5:
                M_fraction = 1.0
                # Random perturbation
                rng = np.random.default_rng(it)
                a = a + rng.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_fraction = 1.0
            continue
        
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            stall_count += 1
            continue
        
        g = g_raw / g_sum  # Normalize to sum = 1
        
        # Line search
        improved = False
        for t in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = score_norm(a_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                stall_count = 0
                break
        
        if not improved:
            stall_count += 1
            if stall_count > max_stall:
                # Reset M_fraction and try perturbation
                M_fraction = max(0.5, M_fraction * 0.8)
                rng = np.random.default_rng(it * 31)
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): score={best_score:.7f}, M_frac={M_fraction:.3f}, stall={stall_count}")
    
    return best_a, best_score


def subgradient_mirror_descent(a_init, time_limit, eta=0.01, verbose=True):
    """Subgradient mirror descent on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    eta_current = eta
    stall = 0
    
    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        
        # Subgradient: grad[j] = 2*a[k_peak - j] if 0 <= k_peak - j < n
        grad = np.zeros(n)
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)
        grad[j_min:j_max + 1] = 2.0 * a[k_peak - j_max:k_peak - j_min + 1][::-1]
        
        # Project to tangent space (simplex)
        grad = grad - np.dot(a, grad)
        
        # Mirror (multiplicative) update
        log_update = -eta_current * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        
        sc = score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Decay eta
        if it % 100 == 0:
            eta_current *= 0.95
        
        # Stuck detection
        if stall > 200:
            rng = np.random.default_rng(it * 7919)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0
            eta_current = min(eta_current * 2, eta)
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True):
    """Coordinate descent with multiple step sizes."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    
    factors = [0.5, 0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5, 2.0]
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000))
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                for f in [0.1, 0.5, 1.0]:
                    a_trial = a.copy()
                    a_trial[i] = 1e-6
                    a_trial = a_trial / a_trial.sum()
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
                continue
            
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] -= diff * a_trial[mask] / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_norm(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
(SOTA approach with cvxpy), subgradient mirror descent, and coordinate descent.
The LP solves: max sum(g) s.t. g>=0, (a*g)[k]<=M for all k, providing
structured descent directions that reduce the autocorrelation peak.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k]<=M for all k."""
    n = len(a)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            coeff = a[i_min:i_max + 1]
            idx = k - np.arange(i_min, i_max + 1)
            constraints.append(coeff @ g[idx] <= M)
    prob = cp.Problem(objective, constraints)
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL]
    last_err = None
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception as e:
            last_err = e
            continue
    if g.value is None:
        return None
    return np.maximum(g.value, 0)


def lp_optimize(a_init, time_limit, m_frac_start=1.0, verbose=True):
    """LP-based iterative optimization with line search."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    m_frac = m_frac_start
    stall = 0
    lp_count = 0

    for it in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        M = b.max()
        g_raw = lp_direction(a, M * m_frac)
        if g_raw is None:
            stall += 1
            m_frac = max(0.5, m_frac * 0.9)
            if stall > 30:
                rng = np.random.default_rng(it)
                a = best_a + rng.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
                m_frac = m_frac_start
                stall = 0
            continue
        lp_count += 1
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            stall += 1
            continue
        g = g_raw / g_sum
        improved = False
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = score_norm(a_trial)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 40:
                m_frac = max(0.5, m_frac * 0.85)
                if m_frac < 0.55:
                    rng = np.random.default_rng(it * 31)
                    a = best_a + rng.standard_normal(n) * 0.005
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    m_frac = m_frac_start
                stall = 0
        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): score={best_score:.7f}, mfrac={m_frac:.3f}, lps={lp_count}, stall={stall}")
    return best_a, best_score


def mirror_descent(a_init, time_limit, eta=0.008, verbose=True):
    """Subgradient mirror descent on simplex with adaptive step size."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    eta_cur = eta
    stall = 0

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        grad = np.zeros(n)
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)
        grad[j_min:j_max + 1] = 2.0 * a[k_peak - j_max:k_peak - j_min + 1][::-1]
        grad = grad - np.dot(a, grad)
        log_update = -eta_cur * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        sc = score_norm(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if it % 200 == 0:
            eta_cur *= 0.97
        if stall > 300:
            rng = np.random.default_rng(it * 7919)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0
            eta_cur = min(eta_cur * 1.5, eta)
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} ({elapsed:.0f}s): score={best_score:.7f}, eta={eta_cur:.6f}")
    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True):
    """Targeted peak shaving: reduce top contributing pairs."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()

    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        if b[peak_k] <= b.max() * (1 - 1e-8):
            break
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_norm(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            for ci in top_idx[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_norm(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    factors = [0.5, 0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5]
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_pass = False
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] -= diff * a_trial[mask] / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_norm(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            no_improve += 1
        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.7f}, no_improve={no_improve}")
    return best_a, best_score


def cosine_window(n):
    """Generate cosine-squared window normalized to sum=1."""
    x = np.linspace(0, np.pi, n)
    w = np.cos(x / 2.0) ** 2
    return w / w.sum()


def triangular_window(n):
    """Generate triangular window normalized to sum=1."""
    mid = n // 2
    if n % 2 == 0:
        w = np.concatenate([np.arange(1, mid + 1), np.arange(mid, 0, -1)])
    else:
        w = np.concatenate([np.arange(1, mid + 1), np.arange(mid - 1, 0, -1)])
    w = w.astype(np.float64)
    return w / w.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
