"""
Fast LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy's HiGHS solver for speed, multiplicative weight updates for fast iterations,
and multi-length exploration to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    # Scale to sum=1 first
    v = v / max(v.max(), 1e-30) * (target_sum / n)  # rough scaling
    # Actually, let's do the standard projection
    # Project onto {x >= 0, sum(x) = 1} then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, top_k):
    """
    Solve the LP to find the best direction.
    Primal: max sum(g) s.t. g >= 0, (f*g)[k_i] <= M for top_k peaks.
    Uses scipy HiGHS solver.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix A (K x n)
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    # Primal: max sum(g) s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = M * np.ones(K)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'maxiter': 50000})
    except Exception:
        return None, M

    if not res.success or res.x is None:
        return None, M

    g = np.maximum(res.x, 0.0)
    if np.any(np.isnan(g)):
        return None, M
    return g, M


def solve_dual_lp(f, top_k):

    Solve the dual LP: min M * sum(y) s.t. A^T y >= 1, y >= 0.
    Then recover primal solution.

    M = b.max()

        A[i, lo:hi + 1] = f[lo:hi + 1]

    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0

    A_ub = -A.T  # n x K

    if not res.success or res.x is None:

def multiplicative_weight_step(f, b, top_k, eta):
    """
    Multiplicative weight update - much faster than LP.
    f_new[j] = f[j] * exp(-eta * sum_{k in top} w_k * f[k-j])
    """
    n = len(f)
    # Select top-k peaks
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    
    # Compute weights (proportional to peak height)
    peaks = b[top_idx]
    w = peaks / peaks.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k, wi in zip(top_idx, w):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += w[k] * f[k - j] for j in [lo, hi]
        # f[k-j] for j in [lo, hi] is f[k-lo : k-hi+1] reversed
        j_range = np.arange(lo, hi + 1)
        f_vals = f[lo:hi + 1]  # f[lo], f[lo+1], ..., f[hi]

        # For j in [lo, hi]: f[k-j] where k-j ranges from k-lo to k-hi
        # k-lo >= 0, k-hi <= n-1
        # f[k-j] for j=lo is f[k-lo], for j=hi is f[k-hi]
        # So grad[j] += w * f[k-j] = w * f[lo : hi+1] reversed? No...
        # grad[j] += w * f[k-j]
        # When j = lo: f[k-lo]
        # When j = lo+1: f[k-lo-1]
        # ...
        # When j = hi: f[k-hi]
        # So it's f[k-hi : k-lo+1] in reverse... 
        # Actually: j goes from lo to hi, k-j goes from k-lo down to k-hi

        grad[lo:hi+1] += wi * f[k-hi:k-lo+1][::-1]

    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-30:
        f_new /= s
    else:

def run_lp_optimization(f_init, time_limit, top_k_start=20, top_k_end=150,
                        use_dual=False, label="lp"):
    """LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    stuck = 0
    top_k = top_k_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Grow top_k gradually
        progress = min(1.0, it / 200.0)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k = min(top_k, 2 * n - 1)
        
        # Solve LP
        if use_dual:
            g, M = solve_dual_lp(f, top_k)
        else:
            g, M = solve_lp_direction(f, top_k)
        
        if g is None:
            stuck += 1
            if stuck > 20:
                break
            top_k = min(top_k * 2, 2 * n - 1)
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            continue
        
        # Normalize direction
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        
        f_trial = (1.0 - 0.05) * f + 0.05 * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s > 1e-10:
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = 0.05
        
        for t in [0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-13:
                best_t_obj = obj
                best_t = t
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        # Check improvement
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_score / (2.0 * n) - 1e-14:
            # Improved (in terms of the normalized objective)
            new_score = 2.0 * n * cur_obj  # since sum=1
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
                stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
        
        if it % 50 == 0:

            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"top_k={top_k} t={elapsed:.0f}s")
    
    return best_f, best_score

def subgradient_polish(f_init, time_limit, n_top=50, label="sg"):
    """Subgradient method with momentum for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = 0.002
    momentum = 0.9
    temp = 0.0005
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    stuck = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Softmax weights over top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        peaks = b[top_idx]
        scaled = peaks / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k, wi in zip(top_idx, w):
            if wi < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            grad[lo:hi+1] += wi * f[k-hi:k-lo+1][::-1]
        
        # Update with momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        f = f_new
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            score = 2.0 * n * obj
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                stuck = 0
                lr = min(lr * 1.03, 0.01)
            else:
                stuck += 1
                if stuck > 4:
                    lr *= 0.9
                    stuck = 0
        
        if it % 200 == 0:
            temp = min(temp * 1.2, 0.1)
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"lr={lr:.5f} t={elapsed:.0f}s")
        
        if stuck > 15:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            stuck = 0
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    multipliers = [0.7, 0.85, 1.0, 1.15, 1.4]
    
    improved = True
    it = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        it += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_score
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = fast_convolve(f).max()
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_score / (2.0 * n) - 1e-12:
                    best_score = 2.0 * n * best_j_obj
                    improved = True
    
    return f, best_score

"""
Fast LP-based optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS solver for fast LP direction finding, multiplicative weight
warmup, adaptive peak selection, multi-length exploration, and subgradient/
coordinate polish to drive the score below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, top_idx, weights):
    """
    Compute weighted gradient: grad[j] = sum_i weights[i] * f[k_i - j].
    """
    n = len(f)
    grad = np.zeros(n)
    for k, wi in zip(top_idx, weights):
        if wi < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += wi * f[k - hi:k - lo + 1][::-1]
    return grad


def solve_lp_direction(f, top_k):
    """
    Solve primal LP: max sum(g) s.t. g >= 0, (f*g)[k_i] <= M.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix (K x n)
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    # Primal: max sum(g) s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'maxiter': 100000,
                                               'dual_feasibility_tolerance': 1e-8})
    except Exception:
        return None, M

    if not res.success or res.x is None:
        return None, M

    g = np.maximum(res.x, 0.0)
    if np.any(~np.isfinite(g)):
        return None, M
    return g, M


def solve_dual_lp(f, top_k):
    """
    Solve dual LP: min M*sum(y) s.t. A^T y >= 1, y >= 0.
    Recover primal via complementary slackness.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    # Dual: min M*sum(y) s.t. A^T y >= 1, y >= 0
    c = M * np.ones(K)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0.0, None)] * K

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'maxiter': 100000})
    except Exception:
        return None, M

    if not res.success or res.x is None:
        return None, M

    y = res.x
    Ay = A.T @ y
    tol = 1e-5
    support = np.where(Ay <= 1.0 + tol)[0]

    if len(support) == 0:
        return None, M

    active_i = np.where(y > 1e-6)[0]
    if len(active_i) == 0:
        g = np.ones(n) / n
        return g, M

    n_s = len(support)
    A_red = A[active_i, support]
    c_red = -np.ones(n_s)
    b_eq = M * np.ones(len(active_i))

    try:
        res2 = linprog(c_red, A_eq=A_red, b_eq=b_eq,
                       bounds=[(0.0, None)] * n_s,
                       method='highs', options={'maxiter': 50000})
    except Exception:
        return None, M

    if not res2.success or res2.x is None:
        return None, M

    g = np.zeros(n)
    g[support] = np.maximum(res2.x, 0.0)
    return g, M


def run_lp_optimization(f_init, time_limit, top_k_start=20, top_k_end=120,
                        use_dual=False, label="lp"):
    """LP-based optimization with adaptive peak count and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptively grow top_k
        progress = min(1.0, it / 150.0)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k = max(10, min(top_k, 2 * n - 1))

        # Solve for direction
        if use_dual and top_k <= 80:
            g, M = solve_dual_lp(f, top_k)
        else:
            g, M = solve_lp_direction(f, top_k)

        if g is None:
            stuck += 1
            if stuck > 30:
                break
            top_k = min(top_k + 10, 2 * n - 1)
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                break
            continue

        h = g / S_g

        # Line search (geometric)
        best_t = 0.0
        cur_obj = fast_convolve(f).max()
        best_t_obj = cur_obj

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-14:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 20:
                # Random restart around best
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                v_reset = np.zeros(n)
                stuck = 0
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_score = 2.0 * n * fast_convolve(f).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_score


def run_mw_optimization(f_init, time_limit, top_k=40, label="mw"):
    """Multiplicative weight method - fast per-iteration."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    eta = 0.05
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Get top peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))

        peaks = b[top_idx]
        w = peaks / peaks.sum()

        # Compute gradient
        grad = compute_gradient(f, top_idx, w)

        # Multiplicative update
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-30:
            f = f_new / s
        else:
            f = f.copy()

        if it % 20 == 0:
            score = 2.0 * n * fast_convolve(f).max()
            if score < best_score - 1e-11:
                best_score = score
                best_f = f.copy()
                stuck = 0
                eta = min(eta * 1.01, 0.5)
            else:
                stuck += 1
                if stuck > 5:
                    eta = max(eta * 0.85, 0.005)
                    stuck = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"eta={eta:.4f} t={elapsed:.0f}s")

    return best_f, best_score


def run_subgradient(f_init, time_limit, n_top=50, label="sg"):
    """Subgradient with momentum for fine polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = 0.002
    momentum = 0.9
    temp = 0.0005
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    stuck = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        peaks = b[top_idx]
        scaled = peaks / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        grad = compute_gradient(f, top_idx, w)

        # Momentum update
        v = momentum * v - lr * grad
        f = project_simplex(f + v)

        if it % 50 == 0:
            score = 2.0 * n * fast_convolve(f).max()
            if score < best_score - 1e-11:
                best_score = score
                best_f = f.copy()
                stuck = 0
                lr = min(lr * 1.02, 0.015)
            else:
                stuck += 1
                if stuck > 3:
                    lr = max(lr * 0.92, 0.0005)
                    stuck = 0

        if it % 200 == 0:
            temp = min(temp * 1.1, 0.05)
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} score={best_score:.6f} "
                  f"lr={lr:.5f} t={elapsed:.0f}s")

        if stuck > 20:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            v = np.zeros(n)
            stuck = 0

    return best_f, best_score


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    multipliers = [0.75, 0.9, 1.0, 1.1, 1.3]
    improved = True

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_score
            if f[j] < 1e-13:
                continue
            cur_obj = None
            best_j_val = f[j]
            for mult in multipliers:
                if mult == 1.0:
                    continue
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                if cur_obj is None:
                    if improved:
                        cur_obj = 2.0 * n * fast_convolve(f).max() / (2.0 * n)
                obj = fast_convolve(f_trial).max()
                best_obj_now = fast_convolve(f).max() if cur_obj is None else cur_obj
                if obj < best_obj_now - 1e-13:
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                new_score = 2.0 * n * fast_convolve(f).max()
                if new_score < best_score:
                    best_score = new_score
                    improved = True

    return f, best_score
