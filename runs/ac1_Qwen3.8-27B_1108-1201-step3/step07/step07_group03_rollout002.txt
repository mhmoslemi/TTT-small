    # conv(w, a_flip)[j] = sum_k w[k] * a_flip[j-k] = sum_k w[k] * a[n-1-(j-k)] = sum_k w[k] * a[n-1-j+k]
    # We want grad[i] = sum_k w[k] * a[k-i]
    # So we need n-1-j+k = k-i, i.e., n-1-j = -i, i.e., j = n-1+i
    # So grad[i] = conv(w, a_flip)[n-1+i]
    conv_result = irfft(W * A_flip, fft_size)
    result = conv_result[n-1:n-1+n]
    return result

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def logsumexp(x, temp):
    """Compute log(sum(exp(x/temp))) stably."""
    x_max = x.max()
    if x_max < -1000:
        return 0.0
    return x_max + temp * np.log(np.sum(np.exp((x - x_max) / temp)))

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    if '_lp_solver_cached' in globals() and globals()['_lp_solver_cached'] is not None:
        return globals()['_lp_solver_cached']
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.XPRESS, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                globals()['_lp_solver_cached'] = solver_name
                return solver_name
        except Exception:
            continue
    globals()['_lp_solver_cached'] = cp.ECOS
    return cp.ECOS

def solve_lp_primal(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b, M_value).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraints: (T @ b)_k <= M for k = 0,...,2n-2
    # (T @ b)_k = sum_j a[j] * b[k-j] (with appropriate bounds)
    # This is conv(a, b)[k] = sum_j a[j] * b[k-j]
    
    # Efficient way: use cp's built-in convolution or build the matrix
    # For speed, build the matrix explicitly
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def solve_lp_dual(a_fixed, solver=None):
    """
    Given fixed a, solve the dual LP: max sum y s.t. (T^T y)_j <= 1, y >= 0.
    Returns (y, dual_value).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    y_var = cp.Variable(2 * n - 1, nonneg=True)
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [
        T.T @ y_var <= 1,
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(y_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Maximize(cp.sum(y_var)), constraints)
        for alt in [cp.ECOS, cp.GLPK, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    y_val = np.asarray(y_var.value, dtype=np.float64)
    y_val = np.maximum(y_val, 0)
    return y_val, float(prob.value)

def smooth_gradient_descent(a0, time_limit, temp_init=None, temp_min=0.001, verbose=False):
    """
    Gradient descent on the smooth approximation:
    f(a) = 2n * logsumexp(conv(a,a)/temp) / (sum a)^2
    with projection onto the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if temp_init is None:
        conv_a = fast_convolve(a)
        temp_init = 0.05 * conv_a.max()
    
    temp = temp_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.5 / n  # Initial learning rate
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute conv(a, a)
        conv_aa = fast_convolve(a)
        
        # Compute softmax weights
        temp_clipped = max(temp, 1e-10)
        shifted = (conv_aa - conv_aa.max()) / temp_clipped
        weights = np.exp(shifted)
        weights = weights / weights.sum()
        
        # Compute gradient: 2n/temp * T^T @ weights
        # T^T @ weights is the cross-correlation we computed above
        grad = (2.0 * n / temp_clipped) * conv_gradient(weights, a)
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    SGD it={it}: score={best_score:.6f} temp={temp:.6f} lr={lr:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        
        # Anneal temperature
        if it % 50 == 0:
            temp *= 0.95
            if temp < temp_min:
                temp = temp_min
            lr *= 0.98
        
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
    
    return best_a, best_score

def fictitious_play_mirror(a0, time_limit, solver=None, verbose=False):
    """
    Fictitious play with mirror descent (multiplicative weights) for averaging.
    Alternates between primal and dual LPs.
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    # Mirror descent state: log-weights for a and b
    log_a = np.full(n, np.log(1.0 / n))
    log_b = np.full(n, np.log(1.0 / n))
    
    # Initialize with the given a
    mask = a > 1e-15
    if mask.sum() > 0:
        log_a[mask] = np.log(a[mask] / a[mask].sum())
        log_b[mask] = np.log(a[mask] / a[mask].sum())
    
    # Running averages (for tracking)
    a_avg = a.copy()
    b_avg = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    eta = 1.0  # Step size for mirror descent
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve primal LP: given current a, find best b
        a_current = np.exp(log_a)
        a_current = a_current / a_current.sum()
        
        b_new, M_b = solve_lp_primal(a_current, solver)
        if b_new is not None:
            # Mirror descent update for b
            mask_b = b_new > 1e-15
            if mask_b.any():
                log_b[mask_b] = np.log(b_new[mask_b] / b_new[mask_b].sum())
            # Keep small values as is (or set to log of small epsilon)
        
        # Solve dual LP: given current b, find best y (which informs a update)
        b_current = np.exp(log_b)
        b_current = b_current / b_current.sum()
        
        a_new, M_a = solve_lp_primal(b_current, solver)
        if a_new is not None:
            mask_a = a_new > 1e-15
            if mask_a.any():
                log_a[mask_a] = np.log(a_new[mask_a] / a_new[mask_a].sum())
        
        # Compute running averages
        a_avg = 0.5 * a_avg + 0.5 * np.exp(log_a)
        a_avg = a_avg / a_avg.sum()
        b_avg = 0.5 * b_avg + 0.5 * np.exp(log_b)
        b_avg = b_avg / b_avg.sum()
        
        # Evaluate candidates
        for cand in [np.exp(log_a), np.exp(log_b), a_avg, b_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP-Mirror it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def fp_with_exponential_weights(a0, time_limit, solver=None, verbose=False):
    """
    Fictitious play with exponential weighting for the iterate average.
    This is equivalent to the multiplicative weights / mirror descent algorithm.
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    # Exponential weights accumulator
    log_weights = np.full(n, -np.log(n))  # log(1/n)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    b = a.copy()  # Current b (opponent's strategy)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Best response: given b, find a that minimizes max_k (a*b)_k
        a_br, _ = solve_lp_primal(b, solver)
        if a_br is None:
            a_br = a
        a = a_br
        
        # Update exponential weights
        mask = a > 1e-15
        if mask.any():
            log_weights[mask] = np.log(a[mask] / a[mask].sum())
        
        # Compute weighted average
        w = np.exp(log_weights - log_weights.max())
        w = w / w.sum()
        
        # Evaluate
        score = 2.0 * n * fast_convolve(w).max()
        if score < best_score:
            best_score = score
            best_a = w.copy()
        
        # Best response: given a, find b that maximizes... 
        # Actually, in this formulation, both players solve the same type of LP.
        # Given a, find b minimizing max_k (a*b)_k.
        b_br, _ = solve_lp_primal(a, solver)
        if b_br is not None:
            b = b_br
            # Also update weights with b
            mask_b = b_br > 1e-15
            if mask_b.any():
                # Combine: average the log-weights
                log_weights = 0.5 * log_weights + 0.5 * np.log(np.maximum(b_br, 1e-15))
                # Renormalize
                log_weights = log_weights - np.max(log_weights)
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP-ExpW it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def damped_fp(a0, time_limit, solver=None, verbose=False):
    """
    Fictitious play with damped updates (like the previous approach but with
    adaptive damping and more careful evaluation).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP for b given a
        b_new, _ = solve_lp_primal(a, solver)
        if b_new is None:
            continue
        b = b_new
        
        # Solve LP for a given b
        a_new, _ = solve_lp_primal(b, solver)
        if a_new is None:
            continue
        
        # Damped update
        alpha = min(1.0, 2.0 / (it ** 0.5))
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Also try the undamped a_new and b
        for cand in [a_new, b, 0.5*(a + b)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Damped-FP it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def coordinate_descent_v2(a0, time_limit, verbose=False):
    """
    Improved coordinate descent with:
    - Golden section search for optimal scaling
    - Pa coordinate updates (adjusting two coordinates simultaneously)
    - Adaptive step selection
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    no_improve_streak = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve_streak < max_no_improve:
        # Identify peak locations
        conv_a = fast_convolve(a)
        top_k = min(10, len(conv_a))
        peaks = np.argpartition(conv_a, -top_k)[-top_k:]
        peak_vals = conv_a[peaks]
        
        # For each peak, identify the contributing coordinate pairs
        # conv_a[k] = sum_j a[j] * a[k-j]
        # The largest contribution is from j where a[j]*a[k-j] is largest
        
        improved = False
        
        # Try adjusting individual coordinates
        # Focus on coordinates that contribute to the top peaks
        candidate_indices = set()
        for k in peaks[:5]:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                if a[j] * a[k - j] > 0.1 * peak_vals.max():
                    candidate_indices.add(j)
                    candidate_indices.add(k - j)
        
        candidate_indices = list(candidate_indices)
        rng.shuffle(candidate_indices)
        
        for i in candidate_indices:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            
            if a[i] < 1e-10:
                continue
            
            # Try to find optimal scaling for coordinate i
            # We want to minimize max_k conv(a_scaled)[k]
            # where a_scaled = a with a[i] scaled by factor f, then renormalized
            
            # Quick binary search for best factor
            best_factor = 1.0
            best_local_score = best_score
            
            # Try a range of factors
            for f in [0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5, 2.0]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_factor = f
            
            if best_factor != 1.0:
                a[i] *= best_factor
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_local_score < best_score - 1e-14:
                    best_score = best_local_score
                    best_a = a.copy()
                    improved = True
                    no_improve_streak = 0
                else:
                    # Revert
                    a[i] /= best_factor
                    a = a / a.sum()
        
        if not improved:
            # Try pairwise adjustments
            pair_improved = False
            n_pairs = min(100, len(candidate_indices) * (len(candidate_indices) - 1) // 2)
            for _ in range(n_pairs):
                if time.monotonic() - t0 > time_limit:
                    return best_a, best_score
                i, j = rng.choice(len(candidate_indices), size=2, replace=False)
                i, j = candidate_indices[i], candidate_indices[j]
                if a[i] < 1e-10 or a[j] < 1e-10:
                    continue
                
                for f1 in [0.5, 0.7, 0.9]:
                    for f2 in [0.5, 0.7, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            pair_improved = True
                            no_improve_streak = 0
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break
            
            if not improved:
                no_improve_streak += 1
        
        if verbose and no_improve_streak % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD2 ({elapsed:.0f}s): best={best_score:.6f} no_improve={no_improve_streak}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_shave_v2(a0, time_limit):
    """
    Targeted peak shaving: for each peak in the autocorrelation,
    find the coordinate pair that contributes most and reduce it.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_convolve(a)
        # Find the global peak
        peak_k = int(np.argmax(conv_a))
        peak_val = conv_a[peak_k]
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
        
        # Find the top contributing pair
        top_ci = int(np.argmax(contribs))
        i = int(idx[top_ci])
        j = peak_k - i
        
        # Try reducing a[i] (and possibly a[j])
        improved = False
        
        if i == j:
            # Single coordinate at the peak center
            for factor in [0.9, 0.8, 0.7, 0.6, 0.5]:
                a_trial = a.copy()
                a_trial[i] *= factor
