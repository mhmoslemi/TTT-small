import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def make_palindromic(a):
    n = len(a)
    half = np.maximum(a[:n//2 + n%2], a[:n//2 + n%2][::-1])
    # Actually, let's just average:
    half = (a[:n//2 + n%2] + a[:n//2 + n%2][::-1]) / 2
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])

def score_of(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    return 2.0 * len(a) * fast_convolve(a).max()

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_refine(a0, time_limit, solver=None, palindromic=False):
    """LP-based refinement with damped updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    if palindromic:
        a = make_palindromic(a)
        a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        alpha = min(1.0, 5.0 / (it + 1.0))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        if palindromic:
            a_new = make_palindromic(a_new)
        s = a_new.sum()
        if s < 1e-15:
            break

        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

            # DIFFERENT j values (unless i=k-i). If i=k-i (k=2i), it's the same j,
            # and d(a[i]*a[i])/d(a[i]) = 2*a[i]. So the formula gives 2*a[k-i] = 2*a[i], correct.
            # If i != k-i, they're different j values, each contributing a[k-i], total 2*a[k-i]. Correct.
            
            if 0 <= k - i and k - i < n:
                grad[i] += 2 * wk * a[k - i]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Projected gradient step
        lr = 0.01
        a_new = a - lr * grad / gnorm
        if palindromic:
            a_new = make_palindromic(a_new)
        a_new = simplex_project(a_new)
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective value for a non-negative sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    return 2.0 * len(a_n) * fast_convolve(a_n).max()


def make_palindromic(a):
    """Make sequence palindromic by averaging with its reverse."""
    n = len(a)
    rev = a[::-1]
    return (a + rev) / 2.0


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

    n = len(a)

    for j in range(n):
        T[j:j + n, j] = a

def gradient_step_smooth(a, time_limit, temp_init=0.005, lr_init=0.02, palindromic=False):
    """
    Gradient descent on smooth approximation of max(conv(a,a)).
    Uses FFT to compute gradient in O(n log n).
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if palindromic:
        a = make_palindromic(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    temp = temp_init
    lr = lr_init
    it = 0

    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute convolution
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:size]
        M = b.max()

        # Soft-max weights
        w = np.exp((b - M) / temp)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(size)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        # Gradient via circular convolution (FFT-based)
        # grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * (wrap(w * a))[i]
        W = rfft(w, fft_size)
        grad = 2.0 * irfft(A * W, fft_size)[:n]

        # Normalize and step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        a_new = a - lr * grad
        if palindromic:
            a_new = make_palindromic(a_new)
        a_new = simplex_project(a_new)

        if palindromic:
            a_new = make_palindromic(a_new)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

        sc = score_of(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            # Anneal temperature slowly
            temp *= 0.999
        else:
            temp *= 1.001
            if temp > temp_init * 3:
                temp = temp_init * 3

        a = a_new
        lr *= 0.9995

    return best_a, best_score


def binary_search_coord_descent(a0, time_limit, palindromic=False, max_sweeps=500):
    """
    Coordinate descent where for each coordinate, we binary-search
    for the optimal value (exploiting convexity in each coordinate).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if palindromic:
        a = make_palindromic(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                # Try small positive values for zero coordinates
                for trial in [0.001, 0.005, 0.01]:
                    a[i] = trial
                    if palindromic:
                        a[i] = make_palindromic(a)[i]
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    sc = score_of(a)
                    if sc < best_score - 1e-15:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
                continue

            # Trisection search for optimal a[i]
            # The function f(v) = score(a with a[i]=v, renormalized) is convex in v
            lo = old_val * 0.1
            hi = old_val * 5.0
            best_v = old_val
            best_f = score_of(a)

            for _ in range(5):  # 5 trisection steps
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                a[i] = m1
                if palindromic:
                    a = make_palindromic(a)
                a = np.maximum(a, 0)
                a = a / a.sum()
                f1 = score_of(a)

                a[i] = m2
                if palindromic:
                    a = make_palindromic(a)
                a = np.maximum(a, 0)
                a = a / a.sum()
                f2 = score_of(a)

                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

                if min(f1, f2) < best_f:
                    if f1 <= f2:
                        best_f = f1
                        best_v = m1
                    else:
                        best_f = f2
                        best_v = m2

            # Set to best found
            a[i] = best_v
            if palindromic:
                a = make_palindromic(a)
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = score_of(a)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val
                if palindromic:
                    a = make_palindromic(a)
                a = np.maximum(a, 0)
                a = a / a.sum()

        if not improved and sweep > 5:
            # Try fine-grained adjustments
            any_improve = False
            for i in rng.choice(n, size=min(n, 150), replace=False):
                old_val = a[i]
                if old_val < 1e-14:
                    continue
                for factor in [0.98, 0.99, 1.01, 1.02]:
                    a[i] = old_val * factor
                    if palindromic:
                        a = make_palindromic(a)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    sc = score_of(a)
                    if sc < best_score - 1e-16:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        any_improve = True
                        break
                if any_improve:
                    break
            if not any_improve:
                break

    return best_a, best_score


def simulated_annealing(a0, time_limit, palindromic=False, temp_init=0.005):
    """SA with adaptive move size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if palindromic:
        a = make_palindromic(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    best_a = a.copy()
    best_score = score_of(a)
    cur_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    temp = temp_init

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        temp = max(temp * 0.99995, 1e-8)

        # Random move
        n_mod = rng.integers(1, 4)
        idxs = rng.choice(n, size=n_mod, replace=False)
        a_trial = a.copy()
        for idx in idxs:
            factor = rng.uniform(0.6, 1.4)
            a_trial[idx] *= factor
        if palindromic:
            a_trial = make_palindromic(a_trial)
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st

        new_score = score_of(a_trial)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_trial
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_trial.copy()

    return best_a, best_score


def peak_shave(a0, time_limit, palindromic=False):
    """Targeted peak reduction by adjusting coordinates contributing to peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if palindromic:
        a = make_palindromic(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find peaks
        n_peaks = min(8, len(b))
        peaks = np.argpartition(b, -n_peaks)[-n_peaks:]

        improved = False
        for pk in peaks:
            pk = int(pk)
            # Find contributing coordinates
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[pk - idx]
            # Try reducing top contributors
            top_c = np.argsort(contribs)[::-1][:4]
            for ci in top_c:
                i = int(idx[ci])
                j = pk - i
                if a[i] < 1e-14:
                    continue
                # Try reducing a[i]
                for factor in [0.85, 0.75, 0.65]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    if palindromic:
                        a_trial = make_palindromic(a_trial)
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a_trial.copy()
                        improved = True
                        break

                # Try simultaneous reduction of pair
                if not improved and i != j and a[j] >= 1e-14:
                    for f1, f2 in [(0.9, 0.9), (0.85, 0.85), (0.9, 0.8)]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        if palindromic:
                            a_trial = make_palindromic(a_trial)
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a_trial.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Random Fourier superposition initialization."""
    n_harm = int(rng.integers(4, 30))
    amps = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1).astype(np.float64)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_chebyshev(n, rng):
    """Chebyshev-based initialization with random coefficients."""
    t = np.linspace(-1, 1, n)
    a = np.ones(n) * 0.5
    n_terms = int(rng.integers(3, 15))
    for k in range(1, n_terms + 1):
        coeff = rng.uniform(-0.3, 0.3)
        # Evaluate Chebyshev polynomial T_k
        a += coeff * np.cos(k * np.arccos(np.clip(t, -1, 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Quick diverse starts with LP refine
    Phase 2: Intensive LP with patience (free + palindromic)
    Phase 3: Gradient-based flattening (FFT-accelerated)
    Phase 4: Binary-search coordinate descent
    Phase 5: Peak shaving
    Phase 6: SA polish
    Phase 7: Final LP round
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(7)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse quick starts ===
    print(f"\n=== Phase 1: Diverse Starts (best={best_score:.6f}) ===")
    t1 = time.monotonic()
    n_values = [512, 640, 768, 896, 1024, 1152]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []
    for n in n_values:
        if time.monotonic() - t1 > 120:
            break
        # Parent interpolation
        if parent is not None:
            cand = interpolate_seq(parent, n)
            a_q, s_q, _ = lp_refine_patience(cand, 10.0, solver, palindromic=False, patience=15)
            candidates.append((s_q, a_q, n, 'parent'))
        # Fourier
        cand = init_fourier(n, rng)
        a_q, s_q, _ = lp_refine_patience(cand, 8.0, solver, palindromic=False, patience=15)
        candidates.append((s_q, a_q, n, 'fourier'))
        # Chebyshev
        cand = init_chebyshev(n, rng)
        a_q, s_q, _ = lp_refine_patience(cand, 8.0, solver, palindromic=False, patience=15)
        candidates.append((s_q, a_q, n, 'cheb'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top={candidates[0][0]:.6f}")
    for s_c, a_c, n_c, lbl in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{lbl}_n{n_c}")

    # === Phase 2: Intensive LP (free) ===
    print(f"\n=== Phase 2: LP Free (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 800
        t_lp = min(rem * 0.4, 350.0)
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, s_lp, it = lp_refine_patience(best_seq, t_lp, solver, palindromic=False, patience=40)
        best_seq, best_score = check_and_update(a_lp, best_seq, best_score, f"lp_free_it{it}")

    # === Phase 3: LP Palindromic ===
    print(f"\n=== Phase 3: LP Palindromic (best={best_score:.6f}) ===")
    if best_seq is not None:
        a_pal = make_palindromic(best_seq)
        a_pal = np.maximum(a_pal, 0)
        a_pal = a_pal / a_pal.sum()
        rem = deadline - time.monotonic() - 600
        t_lp2 = min(rem * 0.4, 200.0)
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, s_lp2, it2 = lp_refine_patience(a_pal, t_lp2, solver, palindromic=True, patience=40)
        best_seq, best_score = check_and_update(a_lp2, best_seq, best_score, f"lp_palin_it{it2}")

    # === Phase 4: Gradient flatten ===
    print(f"\n=== Phase 4: Gradient Flatten (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t_gf = min(rem * 0.4, 150.0)
        print(f"  Budget: {t_gf:.0f}s")
        a_gf, s_gf = gradient_step_smooth(best_seq, t_gf, temp_init=0.004, lr_init=0.015, palindromic=False)
        best_seq, best_score = check_and_update(a_gf, best_seq, best_score, "grad_flat")

        # Also try palindromic gradient
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            a_gfp, s_gfp = gradient_step_smooth(best_seq, min(rem * 0.4, 60.0),
                                                temp_init=0.004, lr_init=0.015, palindromic=True)
            best_seq, best_score = check_and_update(a_gfp, best_seq, best_score, "grad_flat_palin")

    # === Phase 5: Binary-search coordinate descent ===
    print(f"\n=== Phase 5: BinarySearch CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t_cd = min(rem * 0.45, 150.0)
        print(f"  Budget: {t_cd:.0f}s")
        a_cd, s_cd = binary_search_coord_descent(best_seq, t_cd, palindromic=False, max_sweeps=300)
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "bin_cd")

        # Palindromic variant
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            a_pal3 = make_palindromic(best_seq)
            a_pal3 = np.maximum(a_pal3, 0)
            a_pal3 = a_pal3 / a_pal3.sum()
            a_cd2, s_cd2 = binary_search_coord_descent(a_pal3, min(rem * 0.5, 100.0),
                                                       palindromic=True, max_sweeps=200)
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "bin_cd_palin")

    # === Phase 6: Peak shave + SA ===
    print(f"\n=== Phase 6: Peak Shave + SA (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        t_ps = min(rem * 0.4, 80.0)
        a_ps, s_ps = peak_shave(best_seq, t_ps, palindromic=False)
        best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "peak_shave")

        rem = deadline - time.monotonic() - 150
        t_sa = min(rem * 0.5, 80.0)
        a_sa, s_sa = simulated_annealing(best_seq, t_sa, palindromic=False, temp_init=0.003)
        best_seq, best_score = check_and_update(a_sa, best_seq, best_score, "sa")

    # === Phase 7: Random restarts from best ===
    print(f"\n=== Phase 7: Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_r = min(rem * 0.25, 60.0)
            # Perturb best
            noise = rng.standard_normal(n_best) * 0.08
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            a_r, s_r, _ = lp_refine_patience(a_pert, t_r, solver, palindromic=False, patience=25)
            best_seq, best_score = check_and_update(a_r, best_seq, best_score, f"restart_{restart}")

            # Quick CD after restart
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_r2, _ = binary_search_coord_descent(best_seq, min(rem * 0.5, 40.0),
                                                       palindromic=False, max_sweeps=100)
                best_seq, best_score = check_and_update(a_r2, best_seq, best_score, f"cd_r{restart}")

    # === Phase 8: Try alternative lengths ===
    print(f"\n=== Phase 8: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for factor in [0.9, 1.1]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2560 and alt_n != n_best:
                rem = deadline - time.monotonic() - 50
                if rem < 25:
                    break
                t_alt = min(rem * 0.4, 50.0)
                a_alt = interpolate_seq(best_seq, alt_n)
                a_alt2, s_alt, _ = lp_refine_patience(a_alt, t_alt, solver, palindromic=False, patience=20)
                best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Final LP
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_fin, _, _ = lp_refine_patience(best_seq, min(rem * 0.5, 50.0),
                                             solver, palindromic=False, patience=20)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_lp")

        # Final CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_fin2, _ = binary_search_coord_descent(best_seq, min(rem * 0.6, 30.0),
                                                     palindromic=False, max_sweeps=100)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_cd")

        # Final gradient
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a_fin3, _ = gradient_step_smooth(best_seq, min(rem * 0.7, 20.0),
                                             temp_init=0.002, lr_init=0.01, palindromic=False)
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_grad")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
