```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def best_response_lp(a, M, n):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    # A matrix: A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # Shape: (2n-1) x n
    # Build A efficiently
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    # LP: max 1^T g s.t. A g <= M*1, g >= 0
    # In scipy linprog form: min -1^T g s.t. A g <= M*1, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Compute score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_conv_matrix(a):
    """Build the matrix A such that (a * g)[k] = sum_j A[k,j] * g[j]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fictitious_play(a_init, time_limit, t_init=0.5, verbose=True):
    """Fictitious play with LP best response."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    iteration = 0
    t = t_init
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        g = solve_lp(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check convergence
        diff = np.abs(a - g).max()
        if diff < 1e-10:
            if verbose:
                print(f"  FP converged at iter {iteration}")
            break
        
        # Update
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = compute_score_normalized(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive step size
        if iteration > 100:
            t = max(0.1, t_init * 0.999 ** (iteration - 100))
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP iter={iteration}, score={best_score:.8f}, M={M:.6f}, diff={diff:.2e}, t={t:.3f}, time={elapsed:.0f}s")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play.
The key algorithm is:
1. At each iteration, compute M = max(a*a)
2. Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
3. Update a = (1-t)*a + t*(g/sum(g))
This is a principled best-response iteration that converges to a local minimum.
Polished with active-set gradient descent and coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build A matrix: A[k,j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    """LP-based fictitious play."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    iteration = 0
    t = 1.0
    lp_fail_count = 0
    
    while time.monotonic() - t0 < time_limit and iteration < max_iter:
        iteration += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_best_response(a, M)
        if g is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                break
            continue
        
        lp_fail_count = 0
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        accepted = False
        for trial_t in [t, t*0.5, t*0.25, t*0.1, 0.01]:

def active_set_gradient(a_init, time_limit, K=10, lr_init=0.01, verbose=True):
    """Gradient descent using log-sum-exp smoothing over top K peaks."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    lr = lr_init
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        # Use log-sum-exp over top K peaks
        top_k_idx = np.argpartition(conv, -K)[-K:]
        top_k_vals = conv[top_k_idx]
        max_vk = top_k_vals.max()
        log_sum_exp = max_vk + np.log(np.exp(top_k_vals - max_vk).sum())
        
        # Smoothed objective: 2n * log_sum_exp (approximately)
        # Gradient: 2n * sum_{k in top_K} w_k * grad_k
        # where w_k = exp(conv[k] - max_vk) / sum exp
        weights = np.exp(top_k_vals - max_vk)
        weights = weights / weights.sum()
        
        # Gradient of the smoothed max w.r.t. a
        grad = np.zeros(n)
        for w, k in zip(weights, top_k_idx):
            k = int(k)
            # grad_k[j] = 2 * a[k-j] if 0 <= k-j < n
            js = np.maximum(0, k - n + 1)
            je = np.minimum(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * w * a[k - j]
        
        # Project gradient: ensure sum(grad) = 0 (for the simplex constraint)
        grad = grad - grad.mean()
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        sc = compute_score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate
            lr *= 0.9
        
        if verbose and iteration % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  AG iter={iteration}, score={best_score:.8f}, lr={lr:.6f}, time={elapsed:.0f}s")
        
        if lr < 1e-10:
            break
    
    return best_a, best_score

def coordinate_descent(a_init, time_limit, verbose=True):
    """Coordinate descent with ternary search on each coordinate."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()

    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            orig = a[j]
            if orig < 1e-15:
                continue
            
            # Try different values of a[j]
            # Keep sum = 1 by redistributing
            best_val = orig
            best_local = compute_score_normalized(a)
            
            for frac in [0.5, 0.25, 0.75, 0.1, 0.9, 0.0, 0.33, 0.67]:
                val = frac  # a[j] = frac (since sum=1, this means a[j] goes from orig to frac)
                if abs(val - orig) < 1e-12:
                    continue
                a[j] = val
                # Redistribute the difference to other coordinates proportionally
                diff = orig - val
                if abs(diff) > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others = a[mask]
                    os = others.sum()
                    if os > 1e-15:
                        a[mask] = others + diff * others / os
                a_trial = np.maximum(a, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    a[j] = orig
                    continue
                a_trial = a_trial / ss
                sc = compute_score_normalized(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = val
                    best_a_trial = a_trial.copy()
            
            if best_val != orig:
                a = best_a_trial.copy()
                improved = True
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()
            else:
                a[j] = orig

            # Perturb
            idx = rng.integers(0, n, size=10)
            for i in idx:
                a[i] += rng.standard_normal() * 0.005 / n
            a = np.maximum(a, 0)
            a = a / a.sum()

            elapsed = time.monotonic() - t0
            print(f"  CD pass={pass_count}, score={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_lp_matrix(a):
    """Build matrix A where (a*g)[k] = sum_j A[k,j]*g[j]."""

        max_vk = top_k_vals.max()
        weights = np.exp(top_k_vals - max_vk)
        weights = weights / weights.sum()

        # Compute gradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_k_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * w_k * z_la[k - j]

        # Project to simplex (zero-mean gradient)
        grad = grad - grad.mean()

        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        sc = compute_score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            a = a_new  # still follow the trajectory
            stagnant += 1

        # Adaptive learning rate
        if stagnant > 50:
            lr *= 0.95
            stagnant = 0

        if lr < 1e-12:
            break

        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AG iter={iteration}, score={best_score:.8f}, lr={lr:.2e}, "
                  f"elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_fast(a_init, time_limit, verbose=True):
    """Fast coordinate descent using FFT for score evaluation."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    pass_count = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            orig = a[j]
            if orig < 1e-15:
                continue

            best_val = orig
            best_local = best_score
            best_trial = None

            # Try several fractions
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                val = frac
                if abs(val - orig) < 1e-12:
                    continue
                a[j] = val
                diff = orig - val
                if abs(diff) > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others = a[mask]
                    os_sum = others.sum()
                    if os_sum > 1e-15:
                        a[mask] = others + diff * others / os_sum
                a_trial = np.maximum(a, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score_normalized(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = val
                    best_trial = a_trial.copy()

            if best_trial is not None:
                a = best_trial
                improved = True
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()
            else:
                a[j] = orig

        if not improved:
            no_improve_passes += 1
            # Perturb to escape
            if no_improve_passes > 3:
                n_perturb = max(5, n // 50)
                idx = rng.integers(0, n, size=n_perturb)
                for i in idx:
                    a[i] += rng.standard_normal() * (0.01 / np.sqrt(n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve_passes = 0
        else:
            no_improve_passes = 0

        if verbose and pass_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_count}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best if improved."""
    a = np.array(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def multiplicative_weights_update(a_init, time_limit, verbose=True):
    """
    Multiplicative weights / best-response alternative.
    Approximates the LP best response using exponential tilting.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    iteration = 0
    eta = 1.0
    t_step = 1.0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Approximate best response: g_j proportional to exp(-eta * constraint_value_j)
        # The constraint value for coordinate j is the max over k of A[k,j] * g_j / M
        # Approximation: g_j = a_j * exp(-eta * (max_k A[k,j] * a_j) / M)
        # Better: g_j = exp(-eta * c_j) where c_j = max_k (a[k-j] * a_j) / M
        # Even better: use the actual LP-like structure
        # For each j, the "tightness" is max_k { sum_i a_i g_{k-i} } / M
        # Since we don't know g yet, approximate with g = a:
        # tightness_j = max_k { A[k,j] * a_j } / M = a_j * max_k A[k,j] / M

        # max_k A[k,j] = max_k a[k-j] (for valid k) = max(a) (since a[k-j] ranges over all of a)
        # So tightness is uniform! This approximation doesn't work.

        # Better approximation: use the gradient of the objective at a
        # grad_j = 2 * sum_k w_k * a[k-j] where w_k are weights on active peaks
        top_k = min(5, 2 * n - 1)
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
        w = np.exp(top_vals / (0.1 * M))
        w = w / w.sum()

        # Direction: reduce coordinates that contribute most to peaks
        penalty = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                penalty[j] += wk * a[k - j]

        # Multiplicative update: g_j = a_j * (1 + eta * (1/n - penalty_j))
        g = a * (1.0 + eta * (1.0 / n - penalty))
        g = np.maximum(g, 1e-30)
        g = g / g.sum()

        # Update
        a_trial = (1 - t_step) * a + t_step * g
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        sc = compute_score_normalized(a_trial)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
        a = a_trial

        # Decay
        if iteration % 100 == 0:
            eta *= 0.99
            t_step = max(0.1, t_step * 0.999)

        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={iteration}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP-based fictitious play as the primary algorithm,
    supplemented by active-set gradient descent and coordinate descent for polishing.
    Tries multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = np.where(np.isfinite(p), p, 0.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score_normalized(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_p = len(parent)

    # === Phase 1: Fictitious Play on parent (main algorithm) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Fictitious Play n={n_p} (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        a_fp, sc_fp = fictitious_play(parent, t1, max_iter=100000, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"fp_n{n_p}", best_seq, best_score)

    # === Phase 2: Active-set gradient polishing ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Active-Set Gradient (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 150.0)
    if t2 > 20:
        a_ag, sc_ag = active_set_gradient(best_seq, t2, K=8, lr_init=0.005, verbose=True)
        best_seq, best_score = check_and_update(a_ag, "active_grad_1", best_seq, best_score)

    # === Phase 3: Second round of fictitious play (refined) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: FP Refined n={len(best_seq)} (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 300.0)
    if t3 > 30:
        a_fp2, sc_fp2 = fictitious_play(best_seq, t3, max_iter=100000, verbose=True)
        best_seq, best_score = check_and_update(a_fp2, f"fp2_n{len(best_seq)}", best_seq, best_score)

    # === Phase 4: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 150.0)
    if t4 > 20:
        a_cd, sc_cd = coordinate_descent_fast(best_seq, t4, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc_1", best_seq, best_score)

    # === Phase 5: Try larger sequence lengths ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Larger Lengths (best={best_score:.8f})")
    print(f"{'='*60}")
    for alt_n in [1024, 1536, 2048]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t5 = min(rem * 0.15, 200.0)
        if t5 < 40:
            break
        if alt_n > 3000:
            continue
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  FP n={alt_n}, budget={t5:.0f}s")
        a_alt, sc_alt = fictitious_play(init_alt, t5, max_iter=50000, verbose=True)
        best_seq, best_score = check_and_update(a_alt, f"fp_n{alt_n}", best_seq, best_score)

    # === Phase 6: Try different initialization at parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Alternative Inits n={n_p} (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Chirp-like initialization
        t_chirp = np.linspace(0, 2 * np.pi * 5, n_p)
        a_chirp = 1.0 + 0.3 * np.cos(t_chirp) + 0.2 * np.sin(2 * t_chirp)
        a_chirp = np.maximum(a_chirp, 0)
        a_chirp = a_chirp / a_chirp.sum()
        t_alloc = min(rem * 0.15, 120.0)
        if t_alloc > 20:
            a_c, sc_c = fictitious_play(a_chirp, t_alloc, max_iter=30000, verbose=True)
            best_seq, best_score = check_and_update(a_c, "fp_chirp", best_seq, best_score)

    # === Phase 7: Final gradient polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final Polish (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.25, 100.0)
        a_f7, _ = active_set_gradient(best_seq, t7, K=15, lr_init=0.002, verbose=True)
        best_seq, best_score = check_and_update(a_f7, "final_grad", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.3, 60.0)
        a_f8, _ = coordinate_descent_fast(best_seq, t8, verbose=True)
        best_seq, best_score = check_and_update(a_f8, "final_cd", best_seq, best_score)

    # === Phase 8: One more FP round if time allows ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.5, 60.0)
        a_f9, _ = fictitious_play(best_seq, t9, max_iter=20000, verbose=False)
        best_seq, best_score = check_and_update(a_f9, "final_fp", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = parent
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    fs = a_sum = best_seq.sum()
    if fs > 0:
        best_seq = best_seq / fs
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
```
