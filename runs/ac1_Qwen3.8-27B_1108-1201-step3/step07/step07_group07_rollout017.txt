"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Nesterov acceleration
2. LP-based peak reduction (active set method)
3. Fine coordinate descent polish
Multi-length exploration with adaptive time allocation.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of auto-convolution.
    Returns (gradient, smooth_value).
    """
    n = len(a)
    b = fast_convolve(a)
    # Log-sum-exp for stability
    b_max = b.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(tau * (b - b_max))))
    smooth_val = log_sum_exp / tau
    
    # Softmax weights
    p = np.exp(tau * (b - b_max))
    p /= p.sum()
    
    # Gradient: g_i = 2 * sum_m a_m * p_{i+m}
    # This is 2 * (convolve(reversed(a), p))[n-1 : 2n-2]
    r = a[::-1]
    fft_size = 1 << (len(r) + len(p) - 2).bit_length()
    R = rfft(r, fft_size)
    P = rfft(p, fft_size)
    conv = irfft(R * P, fft_size)
    g = 2.0 * conv[n-1:2*n-1]
    
    return g, smooth_val


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_b(a):
    """Compute max of auto-convolution."""
    b = fast_convolve(a)
    return float(b.max())


def nesterov_smooth_descent(a0, time_limit, tau_start=20.0, tau_end=200.0,
                             momentum=0.9, lr_init=0.01, seed=42):
    """
    Nesterov accelerated gradient descent on the log-sum-exp smoothed objective.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = smooth_max_b(a)
    
    cur_tau = tau_start
    k = 0  # iteration counter for Nesterov
    
    while time.monotonic() - t0 < time_limit:
        k += 1
        # Nesterov step
        t_k = (1 + np.sqrt(1 + 4 * k * k)) / 2
        momentum_k = (t_k - 1) / t_k
        
        y = a + momentum_k * (a - y if k > 1 else np.zeros(n))
        # Actually standard Nesterov: y = a + (t_{k-1}-1)/t_k * (a - y_prev)
        # Let me use the simpler form: y = a + momentum * v
        
        # Compute gradient at y
        g_y, _ = compute_smooth_gradient(y, cur_tau)
        
        # Step
        lr = lr_init / (1 + k * 0.01)  # decaying learning rate
        v = y - a  # velocity
        a_new = y - lr * g_y
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj = smooth_max_b(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodically increase tau
        if k % 500 == 0:
            cur_tau = min(cur_tau * 1.5, tau_end)
        
        # Check time
        if time.monotonic() - t0 > time_limit * 0.95:
            break
    
    return best_a, best_obj


def lp_refine_v2(f, max_iters=50, time_limit=60.0, verbose=False):

    Improved LP-based refinement with proper line search.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = smooth_max_b(a)

        M = b.max()

        num_constraints = 2 * n - 1

            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]
        
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)
        
        try:

                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Golden section line search for optimal blending
            best_t = 0.0
            best_blend_obj = best_obj
            
            # First try a range of t values
            t_candidates = np.linspace(0.01, 1.0, 30)

                blend = np.maximum(blend, 0.0)

                blend = blend / s

def coordinate_descent_fine(a0, time_limit, n_passes=3):
    """
    Fine coordinate descent with binary search on optimal multiplier.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = smooth_max_b(a)
    
    improved_global = True
    for pass_num in range(n_passes):
        while improved_global and time.monotonic() - t0 < time_limit:
            improved_global = False
            for j in range(n):
                if time.monotonic() - t0 > time_limit:
                    return best_a, best_obj
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                
                # Binary search for optimal scaling of a[j]
                lo = 1e-8
                hi = min(0.9999, 10 * old_val)
                
                # First, check if current is already good by trying a few values
                best_j_obj = best_obj
                best_j_val = old_val
                
                # Try a coarse grid
                for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.9999:
                        continue
                    scale = (1.0 - new_val) / denom
                    if scale < -0.01 or scale > 50:
                        continue
                    a[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    val = smooth_max_b(a)
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                
                # Restore if no improvement
                if best_j_val == old_val:
                    continue
                
                # Apply best
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - best_j_val) / denom
                a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    
    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

4. Multi-length exploration with adaptive time allocation
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def max_conv(a):
    """Return max of auto-convolution."""
    b = fast_convolve(a)
    return float(b.max())


def compute_smooth_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*b_k)) for b = a*a.
    Returns (gradient array, smooth value).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(tau * (b - b_max))))
    smooth_val = log_sum_exp / tau
    p = np.exp(tau * (b - b_max))
    p /= p.sum()
    # g_i = 2 * sum_m a_m * p_{i+m}
    r = a[::-1]
    total_len = len(r) + len(p) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    R = rfft(r, fft_size)
    P = rfft(p, fft_size)
    conv = irfft(R * P, fft_size)
    g = 2.0 * conv[n - 1: 2 * n - 1]
    return g, smooth_val


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smooth(a0, time_limit, tau_start=30.0, tau_end=300.0,
                    lr_init=0.015, seed=42):
    """
    Nesterov accelerated projected gradient on log-sum-exp smoothed max-conv.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = max_conv(a)
    cur_tau = tau_start
    k = 0
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        k += 1
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        mu = (t_k - 1.0) / t_k
        v = mu * v
        y = a + v

        # Gradient at y
        g_y, _ = compute_smooth_grad(y, cur_tau)

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.005 * k)
        a_new = project_simplex(y - lr * g_y)

        # Evaluate true objective
        obj = max_conv(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Gradually increase tau
        if k % 800 == 0:
            cur_tau = min(cur_tau * 1.4, tau_end)

        # Restart if stuck
        if no_improve > 3000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def lp_refine(f, max_iters=60, time_limit=120.0, verbose=False, relax=1.0):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M*relax, g>=0.
    Uses golden-section line search for blending.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / ss

    best_a = a.copy()
    best_obj = max_conv(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Build LP matrix
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]

        b_lp = np.full(num_constraints, M * relax)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Check if g is too similar to best_a (stagnation)
            cos_sim = np.dot(g_norm, best_a) / (np.linalg.norm(g_norm) * np.linalg.norm(best_a) + 1e-15)
            if cos_sim > 0.9999 and iteration > 0:
                # Try slight relaxation
                relax *= 1.005
                continue

            # Golden section line search
            best_t = 0.0
            best_blend_obj = best_obj

            # Coarse scan
            t_scan = np.linspace(0.005, 0.8, 20)
            best_coarse_t = 0.0
            for t in t_scan:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s2 = blend.sum()
                if s2 < 1e-10:
                    continue
                blend /= s2
                obj_blend = max_conv(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_coarse_t = t
                    best_t = t

            if best_t <= 0.0:
                # Try very small steps
                for t in [0.001, 0.002, 0.003, 0.005]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend /= blend.sum()
                    obj_blend = max_conv(blend)
                    if obj_blend < best_obj - 1e-15:
                        best_t = t
                        best_blend_obj = obj_blend
                        break

            if best_t > 0.0:
                # Golden section refinement
                span = 0.05
                lo = max(0.0, best_t - span)
                hi = min(1.0, best_t + span)
                gr = (np.sqrt(5.0) - 1.0) / 2.0

                for _ in range(15):
                    m1 = hi - gr * (hi - lo)
                    m2 = lo + gr * (hi - lo)

                    blend1 = (1.0 - m1) * best_a + m1 * g_norm
                    blend1 = np.maximum(blend1, 0.0)
                    blend1 /= blend1.sum()
                    obj1 = max_conv(blend1)

                    blend2 = (1.0 - m2) * best_a + m2 * g_norm
                    blend2 = np.maximum(blend2, 0.0)
                    blend2 /= blend2.sum()
                    obj2 = max_conv(blend2)

                    if obj1 < obj2:
                        hi = m2
                    else:
                        lo = m1

                    # Track best
                    if obj1 < best_blend_obj:
                        best_blend_obj = obj1
                        best_t = m1
                    if obj2 < best_blend_obj:
                        best_blend_obj = obj2
                        best_t = m2

                # Apply
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a /= a.sum()
                obj = max_conv(a)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP {iteration}: t={best_t:.4f}, obj={best_obj:.12f}, relax={relax:.4f}")
                else:
                    a = best_a.copy()
            else:
                break

        except Exception:
            break

    return best_a, best_obj


def coord_descent(a0, time_limit, n_passes=5):
    """Fine coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = max_conv(a)

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.92, 0.97,
                            1.03, 1.08, 1.15, 1.3, 1.5, 2.0, 3.0])

    for _ in range(n_passes):
        improved = True
        while improved and time.monotonic() - t0 < time_limit:
            improved = False
            for j in range(n):
                if time.monotonic() - t0 > time_limit:
                    return best_a, best_obj
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue

                best_j_obj = best_obj
                best_j_val = old_val

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0.0 or new_val > 0.99999:
                        continue
                    scale = (1.0 - new_val) / denom
                    if scale < -0.05 or scale > 100.0:
                        continue
                    a[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    val = max_conv(a)
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):

            best_seq = a_norm.copy()

            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [t={elapsed:.0f}s]")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:

    if parent is None:
        parent = make_uniform_init(1152)

    n_parent = len(parent)

    rem = deadline - time.monotonic() - 600

        t_sm = min(rem * 0.3, 80.0)
        print(f"  Smooth: t={t_sm:.0f}s")
        a_sm, _ = nesterov_smooth(parent, time_limit=t_sm,
                                  tau_start=20, tau_end=150,
                                  lr_init=0.02, seed=seed)
        update_best(a_sm, "warmup")

    # === Phase 2: Heavy LP optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_lp = min(rem * 0.6, 180.0)
            print(f"  LP: t={t_lp:.0f}s")
            a_lp, _ = lp_refine(best_seq, max_iters=80, time_limit=t_lp,

    print(f"\n{'='*60}")
    print(f"Phase 3: Smooth+LP Cycle (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        for cycle in range(3):

            if rem < 20:

            t_s = min(rem * 0.25, 40.0)

                                         tau_end=200 + cycle * 100,

                update_best(a_s, f"cycle{cycle}_sm")

            # LP step

            t_l = min(rem * 0.3, 50.0)

            if tl not in target_lengths:

            if t_alloc < 6:

            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_ln, _ = nesterov_smooth(f_int, time_limit=t_alloc * 0.4,
                                      tau_start=50, tau_end=200,
                                      lr_init=0.012, seed=seed + target_n)
            rem = deadline - time.monotonic() - 80
            if rem > 5:

            update_best(a_ln, f"len_{target_n}")

    # === Phase 6: Deep LP on best ===
    print(f"\n{'='*60}")

            a_cd, _ = coord_descent(best_seq, time_limit=t_cd, n_passes=3)
            update_best(a_cd, "cd")

    # === Phase 8: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Final (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:

        if rem > 8:
            a_f, _ = lp_refine(best_seq, max_iters=10, time_limit=min(rem, 20),
                               verbose=True)
            update_best(a_f, "final")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
