"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel multi-phase approach:
(1) Two-scale initialization (optimize short, interpolate to target),
(2) Power-mean based gradient descent with adaptive temperature,
(3) Simulated annealing with peak-targeted redistribution moves,
(4) Multi-length exploration, and (5) Final L-BFGS polish.
Key novelty: power-mean objective creates different optimization landscape
than soft-max, SA with informed moves escapes local minima more effectively.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_normalized(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def soft_max_obj_fn(b, T):
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    return T * (max_b / T + np.log(exp_b.sum()))


def power_mean(b, p):
    b_clipped = np.maximum(b, 1e-30)
    return (np.sum(b_clipped ** p)) ** (1.0 / p)


def power_mean_grad_a(a, b, p, n, fft_size):
    """Gradient of power_mean(conv(a,a), p) w.r.t. a."""
    b_clipped = np.maximum(b, 1e-30)
    pm = power_mean(b, p)
    if pm < 1e-300:
        return np.zeros(n)
    # d/d a[i] pm = (1/p) * pm^(1/p - 1) * p * sum_k b[k]^(p-1) * a[k-i]
    #             = pm^(1/p - 1) * sum_k b[k]^(p-1) * a[k-i]
    w = b_clipped ** (p - 1.0)
    scale = pm ** (1.0 / p - 1.0)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return scale * corr


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def adam_power_mean(a0, time_limit, p_init=4.0, p_final=200.0, lr=0.02,
                    beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimizer using power-mean objective with increasing p."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 2000
    last_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Power parameter schedule (exponential increase)
        p = p_init * (p_final / p_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)

        # Gradient of power mean w.r.t. a
        grad_a = power_mean_grad_a(a_curr, b, p, n, fft_size)

        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Gradient clipping
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm > 10.0:
            grad_z = grad_z / grad_norm * 10.0

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.8

        # Cosine LR decay
        lr = max(0.001, last_lr * (0.5 + 0.5 * np.cos(np.pi * progress)))

        if verbose and t % 5000 == 0:
            print(f"    PM-Adam t={t} p={p:.1f} lr={lr:.5f} score={best_score:.6f}")

    return best_a, best_score


def sa_peak_search(a0, time_limit, T_init=0.01, T_final=0.0001,
                   n_max_iter=100000, verbose=False):
    """
    Simulated annealing with peak-targeted redistribution moves.
    Moves: pick peak lag, reduce top contributor, add to a 'safe' element.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / n_max_iter)
        T = T_init * (T_final / T_init) ** progress

        b = fast_convolve(a)
        cur_score = 2.0 * n * b.max()
        cur_peak = b.max()

        # Find peak lag (prefer non-trivial lags)
        b_search = b.copy()
        b_search[0] -= 0.01  # slight penalty for edges
        peak_k = int(np.argmax(b_search))

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            continue

        # Pick a contributor (biased towards top)
        top_size = min(5, len(idx))
        top_order = np.argsort(contribs)[::-1][:top_size]
        # Weighted random pick from top
        weights = contribs[top_order] / contribs[top_order].sum()
        ci = np.random.choice(top_size, p=weights)
        i = int(idx[top_order[ci]])
        j = int(peak_k - i)

        if i == j or a[i] < 1e-15:
            continue

        # Determine "safe" target elements (low contribution to peak and other high lags)
        # Compute for each element how much it would increase the peak if increased
        # Element m increases b[peak_k] by a[peak_k - m] (if valid)
        safe_scores = np.zeros(n)
        if 0 <= peak_k - 0 and peak_k < 2 * n - 1:
            for m in range(n):
                pk_m = peak_k - m
                if 0 <= pk_m < n:
                    safe_scores[m] = a[pk_m]
                # Also consider second-highest lag
            # We want low safe_scores (low contribution to peak)
            # And prefer elements that are currently small
            safe_combined = safe_scores + 0.5 * a  # prefer small elements with low peak contribution
        else:
            safe_combined = a

        # Pick a safe target (exclude i)
        safe_mask = np.ones(n, dtype=bool)
        safe_mask[i] = False
        safe_candidates = np.where(safe_mask)[0]
        if len(safe_candidates) < 2:
            continue

        # Sample from safe candidates, weighted by inverse safe_combined
        inv_scores = 1.0 / (safe_combined[safe_candidates] + 1e-10)
        inv_scores /= inv_scores.sum()
        m = int(safe_candidates[np.random.choice(len(safe_candidates), p=inv_scores)])

        # Propose move: move fraction of a[i] to a[m]
        fractions = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1]
        f = fractions[np.random.randint(len(fractions))]
        delta = a[i] * f

        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[m] += delta
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()

        new_score = 2.0 * n * fast_convolve(a_trial).max()
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score < 0 or (T > 1e-15 and np.random.random() < np.exp(-delta_score / T)):
            a = a_trial
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a.copy()

        if verbose and it % 10000 == 0:
            print(f"    SA it={it} T={T:.6f} score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, n_sweeps=500, verbose=False):
    """
    Improved coordinate descent: for each element, try multiplicative changes
    and also try paired moves (decrease one, increase another).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    fft_size = 1 << (2 * n - 2).bit_length()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            b = fast_convolve(a)
            cur_score = 2.0 * n * b.max()

            # Single element: try decreasing
            for factor in [0.85, 0.9, 0.95]:
                a_trial = a.copy()
                a_trial[i] *= factor
                s = a_trial.sum()
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    sweep_improved = True
                    break

            if sweep_improved:
                continue

            # Paired move: decrease i, increase j (pick j randomly)
            j = int(rng.integers(0, n))
            if j == i or a[j] < 1e-15:
                continue
            for f_dec in [0.9, 0.8]:
                for f_inc in [1.1, 1.2, 1.3]:
                    a_trial = a.copy()
                    transfer = a_trial[i] * (1.0 - f_dec)
                    a_trial[i] *= f_dec
                    a_trial[j] += transfer
                    a_trial = np.maximum(a_trial, 1e-15)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        sweep_improved = True
                        break
                if sweep_improved:
                    break

        if not sweep_improved and sweep > 50:
            # Stuck: perturbation
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=50, maxiter=5000):
    """L-BFGS polish with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()

    def obj(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * T * (max_b / T + np.log(exp_b.sum()))

    def grad(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(fun=obj, x0=z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': maxiter * 30, 'disp': False})
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()
    return a_out


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    candidates = []

    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 1e-6)
        p = p / p.sum()
        candidates.append((p.copy(), 'parent'))
        for scale in [0.02, 0.05, 0.1, 0.2]:
            pert = p + rng.standard_normal(n) * scale
            pert = np.maximum(pert, 1e-6)
            pert = pert / pert.sum()
            candidates.append((pert, f'pert_{scale}'))

    # Random
    for trial in range(3):
        rand = np.abs(rng.standard_normal(n)) + 0.01
        rand = rand / rand.sum()
        candidates.append((rand, f'random_{trial}'))

    # Uniform
    candidates.append((np.ones(n) / n, 'uniform'))

    # Bimodal (high at edges)
    bi = np.ones(n)
    bi[:n // 4] = 2.0
    bi[-n // 4:] = 2.0
    bi = bi / bi.sum()
    candidates.append((bi, 'bimodal'))

    # Gaussian
    x = np.linspace(-4, 4, n)
    gau = np.exp(-x ** 2 / 6)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append((gau, 'gaussian'))

    # Power law
    idx = np.arange(1, n + 1)
    pw = idx ** (-0.3)
    pw = pw / pw.sum()
    candidates.append((pw, 'power_law'))

    # Sine-modulated
    t = np.linspace(0, 2 * np.pi, n)
    sine = 1.0 + 0.3 * np.cos(2 * t)
    sine = np.maximum(sine, 0.01)
    sine = sine / sine.sum()
    candidates.append((sine, 'sine'))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def two_scale_init(n_target, time_budget, rng):
    """Optimize a short sequence then interpolate."""
    short_n = max(32, n_target // 8)
    init = np.ones(short_n) / short_n
    # Quick Adam on short
    a_short, _ = adam_power_mean(init, time_budget, p_init=2.0, p_final=50.0, lr=0.03)
    # Interpolate to target
    a_target = interpolate_sequence(a_short, n_target)
    return a_target


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Two-scale + diverse multi-start Adam (30%)
    Phase 2: Intensive power-mean Adam on best (25%)
    Phase 3: SA peak search + coordinate descent (25%)
    Phase 4: Multi-length + L-BFGS (15%)
    Phase 5: Final polish (5%)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Multi-start with power-mean Adam (30% time) ===
    print("\n=== Phase 1: Multi-Start Power-Mean Adam ===")
    t_p1 = min(540.0, (deadline - time.monotonic()) * 0.30)

    # Generate inits at parent length
    inits = generate_diverse_inits(n_parent, parent, rng)

    # Score and sort
    inits_scored = []
    for a_init, label in inits:
        s = 2.0 * n_parent * fast_convolve(a_init).max()
        inits_scored.append((s, label, a_init))
    inits_scored.sort(key=lambda x: x[0])

    # Also do two-scale init
    rem = deadline - time.monotonic() - 300
    if rem > 30 and n_parent >= 256:
        a_ts = two_scale_init(n_parent, min(20.0, rem * 0.3), rng)
        s_ts = 2.0 * n_parent * fast_convolve(a_ts).max()
        inits_scored.append((s_ts, 'two_scale', a_ts))
        inits_scored.sort(key=lambda x: x[0])

    # Run Adam on top candidates
    n_starts = min(5, len(inits_scored))
    time_per_start = t_p1 / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(time_per_start, rem * 0.85)
        if t_alloc < 20:
            continue

        s_init, label, init_a = inits_scored[idx]
        print(f"  Start {idx + 1}/{n_starts}: {label} (init={s_init:.6f}), budget={t_alloc:.0f}s")

        # Try different p ranges
        for p_i, p_f, lr_val in [(4.0, 200.0, 0.02), (8.0, 300.0, 0.015), (2.0, 100.0, 0.025)]:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_sub = min(t_alloc / 3.0, rem * 0.8)
            if t_sub < 10:
                continue

            a_res, score_res = adam_power_mean(init_a, t_sub, p_init=p_i, p_final=p_f, lr=lr_val)
            best_seq, best_score = evaluate_and_track(a_res, f"pm_{label}_p{p_i}-{p_f}", best_seq, best_score)

        print(f"    Phase 1 best: {best_score:.6f}")

    # === Phase 2: Intensive refinement on best (25% time) ===
    print(f"\n=== Phase 2: Intensive Refinement (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 200

    if rem > 30:
        t_sub = min(rem * 0.5, 300.0)
        a_res, _ = adam_power_mean(best_seq, t_sub, p_init=10.0, p_final=500.0, lr=0.01)
        best_seq, best_score = evaluate_and_track(a_res, "pm_intensive", best_seq, best_score)
        print(f"  After intensive PM: {best_score:.6f}")

    # Also try soft-max Adam (different landscape)
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_sub = min(rem * 0.4, 150.0)
        # Use a soft-max variant
        a_res, _ = adam_power_mean(best_seq, t_sub, p_init=200.0, p_final=1000.0, lr=0.005)
        best_seq, best_score = evaluate_and_track(a_res, "pm_sharp", best_seq, best_score)
        print(f"  After sharp PM: {best_score:.6f}")

    # === Phase 3: SA + Coordinate Descent (25% time) ===
    print(f"\n=== Phase 3: SA Peak Search + Coord Descent (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 120

    if rem > 30:
        t_sub = min(rem * 0.5, 200.0)
        a_sa, _ = sa_peak_search(best_seq, t_sub, T_init=0.005, T_final=0.00005,
                                 n_max_iter=500000, verbose=True)
        best_seq, best_score = evaluate_and_track(a_sa, "sa_peak", best_seq, best_score)
        print(f"  After SA: {best_score:.6f}")

    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_sub = min(rem * 0.6, 100.0)
        a_cd, _ = coordinate_descent_v2(best_seq, t_sub, n_sweeps=500)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.6f}")

    # Another SA round after coord descent
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t_sub = min(rem * 0.7, 80.0)
        a_sa2, _ = sa_peak_search(best_seq, t_sub, T_init=0.002, T_final=0.00002,
                                  n_max_iter=300000)
        best_seq, best_score = evaluate_and_track(a_sa2, "sa_peak_2", best_seq, best_score)
        print(f"  After SA2: {best_score:.6f}")

    # === Phase 4: Multi-length + L-BFGS (15% time) ===
    print(f"\n=== Phase 4: Multi-Length + L-BFGS (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 60

    # L-BFGS polish at different temperatures
    if rem > 20:
        for T in [100, 30, 10]:
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            t_sub = min(rem * 0.3, 50.0)
            a_lb = lbgfs_polish(best_seq, t_sub, T=T, maxiter=2000)
            best_seq, best_score = evaluate_and_track(a_lb, f"lbgfs_T{T}", best_seq, best_score)
            print(f"  L-BFGS T={T}: {best_score:.6f}")

    # Try different lengths
    rem = deadline - time.monotonic() - 40
    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 1.5, 0.75, 2.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_sub = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = adam_power_mean(init_alt, t_sub, p_init=5.0, p_final=200.0, lr=0.015)
            best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final Polish (5% time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10

    if rem > 10:
        t_sub = min(rem * 0.5, 40.0)
        a_fin, _ = adam_power_mean(best_seq, t_sub, p_init=200.0, p_final=1000.0, lr=0.003)
        best_seq, best_score = evaluate_and_track(a_fin, "final_pm", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.8, 30.0)
        a_cd_fin, _ = coordinate_descent_v2(best_seq, t_sub, n_sweeps=200)
        best_seq, best_score = evaluate_and_track(a_cd_fin, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
