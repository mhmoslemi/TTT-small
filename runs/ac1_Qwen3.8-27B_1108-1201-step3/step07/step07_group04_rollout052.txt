import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def score_normalized(a):
    """Compute the objective for a non-negative sequence (assumes sum=1)."""
    n = len(a)
    if a.sum() < 1e-15:
        return float('inf')
    a = a / a.sum()
    return 2.0 * n * fast_convolve(a).max()


def fictitious_play_enhanced(n, time_limit, initial=None, solver=None, verbose=False):
    """Enhanced fictitious play with adaptive averaging and period resets."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_score = score_normalized(a)
    best_a = a.copy()
    iters = 0
    stale_count = 0
    best_streak_score = best_score

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Adaptive averaging: increase weight of recent iterates over time
        w = min(1.0, 5.0 / iters)
        a_avg = (1.0 - w) * a_avg + w * a

        # Evaluate multiple candidates
        candidates = [a, 0.5 * (a + b), a_avg, 0.7 * a_avg + 0.3 * a]
        improved = False
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            sc = score_normalized(cand_norm)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = cand_norm.copy()
                improved = True
                stale_count = 0
            else:
                stale_count += 1

        # Periodic reset to escape stagnation
        if stale_count > 100:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = a.copy()
            a_avg = a.copy()
            stale_count = 0

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            ma_str = f"{M_a:.8f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.8f}" if M_b is not None else "N/A"
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): M_a={ma_str} M_b={mb_str} best={best_score:.8f}")

    return best_a, best_score


def multiplicative_update(a0, time_limit, alpha=0.5, verbose=False):
    """Multiplicative power iteration exploiting bilinear structure."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = monotonic()
    it = 0

    while monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        # Gradient-like direction: for each j, the "contribution" to peaks
        # Update: a_new[j] = a[j] / (sum_k c[k] * a[j+k-n+1..])^alpha
        # Simplified: use the diagonal of the "interaction matrix"
        # For the center peak: a[j] interacts with a[n-1-j]
        # General approach: compute g[j] = sum_k c[k] * a[k-j] (valid indices)
        g = np.zeros(n)
        for k in range(len(c)):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += c[k] * a[k - idx]

        g = np.maximum(g, 1e-30)
        a_new = a / (g ** alpha)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 50 == 0:
            print(f"    MU iter={it}: score={best_score:.8f}")

    return best_a, best_score


def damped_lp_with_linesearch(a0, time_limit, solver=None, verbose=False):
    """Damped LP refinement with backtracking line search on the step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = monotonic()
    it = 0

    while monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(10):
            a_trial = (1 - alpha) * a + alpha * b_new
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / s
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
            alpha *= 0.5

        if not improved and it > 100:
            # Add perturbation to escape
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 30 == 0:
            print(f"    DampedLS iter={it} ({monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS in log-space with soft-max smoothing of the max function."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-25)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_sh = z - z.max()
        ez = np.exp(z_sh)
        a = ez / ez.sum()
        b = fast_convolve(a)
        bmax = b.max()
        scaled = (b - bmax) / T
        eb = np.exp(scaled)
        lse = bmax + T * np.log(eb.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_sh = z - z.max()
        ez = np.exp(z_sh)
        a = ez / ez.sum()
        b = fast_convolve(a)
        bmax = b.max()
        scaled = (b - bmax) / T
        eb = np.exp(scaled)
        w = eb / eb.sum()

        # Gradient of sum_k w[k]*(a*a)[k] w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        grad_a *= 2.0 * n

        # Transform to z-space (log-parametrization)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_sh = z_final - z_final.max()
            a_out = np.exp(z_sh)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.92, top_k=50):
    """Subgradient with adaptive step size and momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.001 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        sc = score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 500:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.002)
            no_improve = 0

    return best_a, best_score


def peak_shave_enhanced(a0, time_limit, verbose=False):
    """Enhanced peak shaving with multiple strategies and adaptive steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = monotonic()
    failed = 0
    step = 0

    while monotonic() - t0 < time_limit and failed < 100:
        step += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        target = best_score / (2.0 * n)

        if b[peak_k] <= target + 1e-18:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False

        # Strategy 1: Reduce single top contributor with adaptive factor
        for ci in top_contrib[:4]:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Reduce top pair
        if not improved:
            for ci in top_contrib[:4]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-15:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s2 = a_trial.sum()
                        if s2 < 1e-15:
                            continue
                        a_trial = a_trial / s2
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Pair swap (redistribute mass)
        if not improved:
            for ci in top_contrib[:3]:
                i = int(idx[ci])
                # Try moving mass from i to a low-contribution position
                j = int(peak_k - i)
                if i == j:
                    continue
                # Find a position far from the peak
                target_pos = (i + n // 2) % n
                if a[target_pos] < 1e-15:
                    continue
                for eps in [0.001, 0.002, 0.005]:
                    a_trial = a.copy()
                    a_trial[i] -= eps * a[i]
                    a_trial[target_pos] += eps * a[i]
                    a_trial = np.maximum(a_trial, 0)
                    s2 = a_trial.sum()
                    if s2 < 1e-15:
                        continue
                    a_trial = a_trial / s2
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 4: Tiny random perturbation
        if not improved:
            for attempt in range(8):
                noise_scale = 0.0003 * (1.0 + 0.1 * step % 10)
                a_trial = a + np.random.standard_normal(n) * noise_scale
                a_trial = np.maximum(a_trial, 0)
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
        else:
            failed = 0

        if verbose and step % 200 == 0:
            print(f"    PS step={step} ({monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, max_rounds=50):
    """Fine-grained coordinate descent: adjust one element at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = monotonic()

    for rnd in range(max_rounds):
        if monotonic() - t0 > time_limit:
            break
        # Focus on elements that contribute most to peaks
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = idx[np.argsort(contribs)[::-1][:min(20, len(idx))]]

        improved = False
        for i in top_idx:
            if monotonic() - t0 > time_limit:
                break
            old = a[int(i)]
            if old < 1e-15:
                continue
            # Try small multiplicative adjustments
            for factor in [0.995, 1.005, 0.99, 1.01, 0.98, 1.02]:
                a_trial = a.copy()
                a_trial[int(i)] = old * factor
                a_trial = np.maximum(a_trial, 0)
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            # Broader random coordinate selection
            for _ in range(5):
                i = np.random.randint(n)
                old = a[i]
                if old < 1e-15:
                    continue
                for factor in [0.99, 1.01]:
                    a_trial = a.copy()
                    a_trial[i] = old * factor
                    a_trial = np.maximum(a_trial, 0)
                    s2 = a_trial.sum()
                    if s2 < 1e-15:
                        continue
                    a_trial = a_trial / s2
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP fictitious play (adaptive)
    2. Damped LP with line search
    3. Multiplicative updates
    4. Multi-temperature L-BFGS
    5. Subgradient + peak shaving
    6. Coordinate descent polish
    7. Multi-length exploration
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n=768, score={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 600
    if rem > 30:
        t_fp = min(rem * 0.7, 500.0)
        print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
        a_fp, sc_fp = fictitious_play_enhanced(n_parent, t_fp, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}", best_seq, best_score)

    # === Phase 2: Damped LP with Line Search ===
    print(f"\n=== Phase 2: Damped LP + LS (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 400
        if rem > 20:
            t_dl = min(rem * 0.6, 200.0)
            print(f"  DampedLS n={len(best_seq)}, budget={t_dl:.0f}s")
            a_dl, _ = damped_lp_with_linesearch(best_seq, t_dl, solver=solver, verbose=True)
            best_seq, best_score = check_and_update(a_dl, "damped_ls", best_seq, best_score)

    # === Phase 3: Multiplicative Update ===
    print(f"\n=== Phase 3: Multiplicative Update (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 15:
            t_mu = min(rem * 0.5, 80.0)
            a_mu, _ = multiplicative_update(best_seq, t_mu, alpha=0.3)
            best_seq, best_score = check_and_update(a_mu, "multiplicative", best_seq, best_score)

    # === Phase 4: Multi-temperature L-BFGS ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        for T in [50, 20, 8, 3]:
            rem = deadline - monotonic() - 150
            if rem < 10:
                break
            t_lb = min(rem * 0.35, 60.0)
            print(f"  L-BFGS T={T}, budget={t_lb:.0f}s")
            a_lb = lbgfs_smooth(best_seq, t_lb, T=T, maxiter=2000)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Subgradient ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 120
        if rem > 15:
            t_sg = min(rem * 0.5, 80.0)
            n_best = len(best_seq)
            a_sg, _ = subgradient_optimize(best_seq, t_sg, lr_init=0.002, momentum=0.93, top_k=max(30, n_best // 8))
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 6: Peak Shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 80
        if rem > 10:
            t_ps = min(rem * 0.6, 80.0)
            a_ps, _ = peak_shave_enhanced(best_seq, t_ps, verbose=True)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 7: Second FP round ===
    print(f"\n=== Phase 7: FP Round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 80
        if rem > 30:
            t_fp2 = min(rem * 0.5, 100.0)
            a_fp2, _ = fictitious_play_enhanced(len(best_seq), t_fp2, initial=best_seq, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_fp2, "FP2", best_seq, best_score)
            # Quick polish after FP2
            rem = deadline - monotonic() - 30
            if rem > 10:
                a_ps2 = lbgfs_smooth(best_seq, min(rem * 0.5, 30.0), T=5, maxiter=1000)
                best_seq, best_score = check_and_update(a_ps2, "post_fp2_lbgfs", best_seq, best_score)

    # === Phase 8: Coordinate Descent Polish ===
    print(f"\n=== Phase 8: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 40
        if rem > 10:
            t_cd = min(rem * 0.6, 50.0)
            a_cd, _ = coordinate_descent_polish(best_seq, t_cd)
            best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # === Phase 9: Try alternative lengths ===
    print(f"\n=== Phase 9: Length Exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5, 2.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic() - 30
            if rem < 25:
                break
            t_alt = min(rem * 0.5, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = fictitious_play_enhanced(alt_n, t_alt, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)
            # Quick polish on new length
            rem = deadline - monotonic() - 15
            if rem > 8 and best_seq is not None and len(best_seq) == alt_n:
                a_pol = lbgfs_smooth(best_seq, min(rem * 0.5, 20.0), T=10, maxiter=500)
                best_seq, best_score = check_and_update(a_pol, f"alt_n{alt_n}_polish", best_seq, best_score)
                a_ps = peak_shave_enhanced(best_seq, min(rem * 0.3, 10.0))
                best_seq, best_score = check_and_update(a_ps, f"alt_n{alt_n}_ps", best_seq, best_score)

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n=== Phase 10: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 5
        if rem > 5:
            t_final = min(rem * 0.5, 15.0)
            a_fin = lbgfs_smooth(best_seq, t_final, T=2, maxiter=2000)
            best_seq, best_score = check_and_update(a_fin, "final_lbgfs_T2", best_seq, best_score)

        rem = deadline - monotonic() - 3
        if rem > 3:
            a_ps_fin = peak_shave_enhanced(best_seq, min(rem * 0.6, 8.0))
            best_seq, best_score = check_and_update(a_ps_fin, "final_ps", best_seq, best_score)

        rem = deadline - monotonic() - 2
        if rem > 2:
            t_cd = min(rem * 0.5, 5.0)
            a_cd = coordinate_descent_polish(best_seq, t_cd, max_rounds=20)
            best_seq, best_score = check_and_update(a_cd, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
