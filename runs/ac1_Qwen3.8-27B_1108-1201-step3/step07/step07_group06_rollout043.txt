"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase hybrid:
Phase 1: Fictitious play (multiplicative weights on peak subgradient) with adaptive step
Phase 2: Log-sum-exp L-BFGS with temperature annealing in log-space

Phase 4: Multi-start exploration at different sequence lengths

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def softmax_of_x(x):
    """Numerically stable softmax."""
    xm = x - np.max(x)
    e = np.exp(xm)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def lse_obj_func(x, T, n):
    """Log-sum-exp smoothed objective (top-level for scipy compatibility)."""
    a = softmax_of_x(x)
    b = fast_convolve(a)
    mb = b.max()
    eb = np.exp((b - mb) / T)
    s = eb.sum()
    return T * (mb / T + np.log(s))


def lse_grad_func(x, T, n):
    """Gradient of LSE smoothed objective w.r.t. log-space variables."""
    a = softmax_of_x(x)
    b = fast_convolve(a)
    mb = b.max()
    eb = np.exp((b - mb) / T)
    w = eb / eb.sum()
    grad_a = 2.0 * np.correlate(w, a, 'valid')
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def fictitious_play(a0, time_limit, eta_init=None, verbose=False):
    """
    Fictitious play: multiplicative weights targeting the current autocorrelation peak.
    a[i] *= exp(-eta * 2*a[k*-i]) then normalize. Adaptive step size.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-300)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_b_max = fast_convolve(a).max()
    t0 = time.monotonic()

    if eta_init is None:
        eta = 0.3 / max(a.max(), 1e-10)
    else:
        eta = eta_init

    it = 0
    no_improve = 0
    patience = 8000

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        if b_max < best_b_max - 1e-15:
            best_b_max = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            eta *= 0.5
            no_improve = 0
            if eta < 1e-12:
                break
            # Also try a small perturbation to escape
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 1e-300)
            a = a / a.sum()
            no_improve = 0
            patience = int(patience * 1.3)

        k_star = int(np.argmax(b))

        # Gradient: shifted copy of a
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            shifted = a[k_star - hi:k_star - lo + 1]
            # Multiplicative update only on the affected range
            a[lo:hi + 1] *= np.exp(-2.0 * eta * shifted)
            a = np.maximum(a, 1e-300)
            a = a / a.sum()

        it += 1
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={it} eta={eta:.6f} best_bmax={best_b_max:.10f} t={elapsed:.0f}s")

def fictitious_play_smoothed(a0, time_limit, T_init=2.0, T_final=0.1, verbose=False):
    """
    Fictitious play with smoothed subgradient (log-sum-exp weights).
    Uses the full smoothed gradient for more stable updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-300)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_b_max = fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.1 / max(a.max(), 1e-10)
    it = 0
    no_improve = 0
    patience = 2000

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_total)
        T = T_init * (T_final / T_init) ** progress

        b = fast_convolve(a)
        b_max = b.max()

        if b_max < best_b_max - 1e-15:
            best_b_max = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-12:
                break

        # Smoothed weights
        mb = b.max()
        eb = np.exp((b - mb) / T)
        w = eb / eb.sum()

        # Smoothed gradient
        grad = 2.0 * np.correlate(w, a, 'valid')

        # Multiplicative update
        log_update = -eta * grad
        # Clip to avoid numerical issues
        log_update = np.clip(log_update, -5.0, 5.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

def lbgfs_phase(a0, time_limit, T_start=30.0, T_end=0.3, n_temps=15):
    """
    L-BFGS on the log-sum-exp smoothed objective with decreasing temperature.
    Works in log-space (softmax parameterization).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-300)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_temps)
    time_per_T = time_limit / n_temps

    for idx, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        t_alloc = min(time_per_T, (time_limit - (time.monotonic() - t0)) * 0.85)
        if t_alloc < 3:
            break

        z = np.log(a + 1e-300)

        max_iters = min(3000, int(t_alloc * 200))
        max_fun = min(50000, max_iters * 30)

        try:
            res = minimize(
                fun=lse_obj_func,
                x0=z,
                jac=lse_grad_func,
                args=(T, n),
                method='L-BFGS-B',
                options={'maxiter': max_iters, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': max_fun, 'disp': False}
            )
            a_new = softmax_of_x(res.x)
        except Exception:
            a_new = a.copy()

        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        if idx % 3 == 0:
            print(f"  L-BFGS T={T:.3f}: score={score:.8f} (best={best_score:.8f})")

    return best_a, best_score


def peak_shave(a0, time_limit, n_trials=5000):
    """
    Peak-targeted local search: identify contributing elements to the max peak
    and try pairwise reductions with mass redistribution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the max peak (interior preferred)
        b_search = b.copy()
        b_search[0] = -np.inf
        if len(b_search) > 1:
            b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        if peak_val <= best_score / (2.0 * n) * 1.0001:
            # All peaks are balanced - try random perturbation
            if trial % 20 == 0:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 > 1e-15:
                    a = a / s2
            continue

        # Find top contributors to this peak
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_ci = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_ci:
            if improved:
                break
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[i] < 1e-15 or a[j] < 1e-15:
                continue

            # Try reducing i and/or j, redistributing mass
            for f_i in [0.85, 0.75, 0.6]:
                if improved:
                    break
                for f_j in [1.0, 0.95, 0.9]:
                    a_trial = a.copy()
                    diff_i = a_trial[i] * (1.0 - f_i)
                    diff_j = a_trial[j] * (1.0 - f_j)
                    a_trial[i] *= f_i
                    a_trial[j] *= f_j
                    total_diff = diff_i + diff_j
                    if total_diff < 1e-15:
                        continue
                    # Redistribute to random positions or to elements far from peak
                    n_redist = min(5, n // 4)
                    # Choose redistribution targets: elements NOT contributing to this peak
                    non_contrib = np.setdiff1d(np.arange(n), idx)
                    if len(non_contrib) < n_redist:
                        non_contrib = np.setdiff1d(np.arange(n), [i, j])
                    targets = rng.choice(non_contrib, size=n_redist, replace=False)
                    for t_idx in targets:
                        a_trial[int(t_idx)] += total_diff / n_redist

                    a_trial = np.maximum(a_trial, 0)
                    s3 = a_trial.sum()
                    if s3 < 1e-15:
                        continue
                    a_trial = a_trial / s3
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True

        if not improved and trial % 50 == 0 and trial > 100:
            # Stuck - perturbation from best
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            s4 = a.sum()
            if s4 > 1e-15:
                a = a / s4

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=100):
    """
    Coordinate descent: for each element in random order, try multiplicative
    adjustments and keep the best.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score
            cur_sum = a.sum()

            for factor in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score - 1e-15:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length (preserving shape)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            best_seq, best_score = eval_and_track(a_alt, f"alt_n{alt_n}_fp", best_seq, best_score)
            if len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 30
                if rem > 15:
                    t_sub2 = min(rem * 0.6, 50.0)
                    a_alt2, _ = lbgfs_phase(best_seq, t_sub2, T_start=10.0, T_end=0.3, n_temps=8)
                    best_seq, best_score = eval_and_track(a_alt2, f"alt_n{alt_n}_lb", best_seq, best_score)
            print(f"  Alt n={alt_n}: {best_score:.8f}")

        # Try structured initialization at original length
        rem = deadline - time.monotonic() - 25
        if rem > 30:
            # Bimodal initialization
            n_try = n_curr
            mid = n_try // 2
            bi = np.zeros(n_try)
            bi[max(0, mid - n_try // 5):mid + n_try // 5] = 1.0
            bi[max(0, mid - n_try // 3):mid + n_try // 3] = 0.5
            bi = np.maximum(bi, 0.001)
            bi = bi / bi.sum()
            # Blend with current best
            blend = 0.5 * best_seq + 0.5 * bi
            if len(blend) == n_try:
                blend = blend / blend.sum()
            t_sub = min(rem * 0.5, 60.0)
            a_bi, _ = fictitious_play(blend, t_sub)
            best_seq, best_score = eval_and_track(a_bi, "bimodal_blend", best_seq, best_score)
            if len(best_seq) == n_try:
                rem = deadline - time.monotonic() - 15
                if rem > 10:
                    a_bi2, _ = lbgfs_phase(best_seq, rem * 0.7, T_start=5.0, T_end=0.2, n_temps=5)
                    best_seq, best_score = eval_and_track(a_bi2, "bimodal_lb", best_seq, best_score)
            print(f"  Bimodal: {best_score:.8f}")

    # === PHASE 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 5

    if rem > 10:
        # Final L-BFGS at very low temp
        t_sub = min(rem * 0.4, 40.0)
        a_fin, _ = lbgfs_phase(best_seq, t_sub, T_start=1.0, T_end=0.05, n_temps=5)
        best_seq, best_score = eval_and_track(a_fin, "final_lb", best_seq, best_score)

    rem = deadline - time.monotonic() - 3
    if rem > 5:
        t_sub = min(rem * 0.7, 20.0)
        a_ps_fin, _ = peak_shave(best_seq, t_sub, n_trials=2000)
        best_seq, best_score = eval_and_track(a_ps_fin, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f}")

    return [float(x) for x in best_seq]
