import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_a(a, w, n, fft_size):
    """
    Compute gradient of 2n * soft_max(conv(a,a), T) w.r.t. a.
    grad_a[i] = 2n * sum_{j=0}^{n-1} w[j+i] * a[j]
    Uses FFT-based cross-correlation.
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def get_score(a):

def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='linear',
                  verbose=False, verbose_interval=2000):
    """
    Adam optimizer in log-space with temperature annealing for soft-max smoothing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()

    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()

    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 3000
    last_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Temperature schedule
        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        # Current a from z
        a_curr = softmax_from_z(z)

        # Compute soft-max and weights
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_grad_a(a_curr, w, n, fft_size)

        # Transform to z-space: grad_z = a * (grad_a - a^T * grad_a)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        conv_new = fast_convolve(a_new)

            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.7

        # Slow LR decay
        last_lr = lr * (1.0 - 0.0001)
        lr = max(0.0001, last_lr)

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")

def lbgfs_logspace(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * (T * (max_b / T + np.log(exp_b.sum())))

    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = compute_grad_a(a, w, n, fft_size)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=soft_max_obj,
            x0=z0,
            jac=soft_max_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()

    return a_out

            break

        b = fast_convolve(a)
        # Find peak (prefer interior peaks)

        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)
        if peak_val <= target * 1.001:
            continue

        # Elements contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue

            for f1 in [0.95, 0.9, 0.85, 0.8]:
                for f2 in [1.0, 1.02, 1.05]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try single-element reduction
            for ci in top_idx[:3]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    # Redistribute to maintain sum
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 100:
            # Stuck - try random perturbation
            if trial % 50 == 0:
                a = best_a + np.random.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=200):
    """
    Coordinate descent: for each element, try multiplicative updates
    and keep the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Random order
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_initializations(parent, n, rng):
    """Generate diverse starting points for multi-start optimization."""
    candidates = []

    # Parent-based (if available and right length)
    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            # Perturbed parent
            for scale in [0.01, 0.03, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'parent_pert_{scale}', pert))
            # Structured perturbation
            for freq in [2, 3, 5, 8]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p + 0.03 * np.cos(t)
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    pert = pert / sp
                    candidates.append((f'parent_cos_{freq}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Triangular (concave)
    tri = np.linspace(0, 1, n)
    tri += np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    tri = tri / tri.sum()
    candidates.append(('triangular', tri))

    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    bi[max(0, mid - n // 6):mid + n // 6] = 1.0
    bi[max(0, mid - n // 4):mid + n // 4] = 0.5
    bi = np.maximum(bi, 0.001)
    bi = bi / bi.sum()
    candidates.append(('bimodal', bi))

    # Gaussian-like
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 4)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append(('gaussian', gau))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    # Exponential decay
    expo = np.exp(-np.linspace(0, 3, n))
    expo = expo / expo.sum()
    candidates.append(('expo', expo))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
