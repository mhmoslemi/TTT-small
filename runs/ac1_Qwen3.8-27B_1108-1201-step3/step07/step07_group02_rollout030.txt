def _fast_conv(a):

    return irfft(A * A, fsize)[:size]

def _softmax(z):

def _lp_best_response(a_fixed, solver):
    """Given a (sum=1, a>=0), find b minimizing max_k conv(a,b)[k] s.t. sum(b)=1, b>=0."""
    n = len(a_fixed)

    M_var = cp.Variable()
    # Build T implicitly: (T @ b)[k] = sum_j a_fixed[k-j]*b[j] for valid k-j
    # T is (2n-1) x n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

def _lp_alternate(a0, time_budget, solver, n_iters=2000):
    """LP alternating (block coordinate descent on the convex problem)."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_sc = _obj_norm(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        # Given a, find best b
        b_new = _lp_best_response(a, solver)
        if b_new is None:
            break
        # Update a = b_new (or averaged)
        # Use averaging for stability
        alpha = 2.0 / (it + 2)
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a /= a.sum()
        
        sc = _obj_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if (it + 1) % 500 == 0:

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    else:
        fi = np.ones(new_n) / new_n
    return fi

def _peak_shave(a0, time_budget, n_steps=3000):

            break

        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1]

                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_t = a.copy()
                a_t[i] = new_val
                a_t[mask] = others + diff * others / os
                a_t /= a_t.sum()
                sc = _obj_norm(a_t)
                if sc < best_sc - 1e-14:
                    a = a_t
                    best_sc = sc

def _structured_init(n, rng, kind=0):
    """Generate structured initializations."""
    if kind == 0:  # Asymmetric exponential
        x = np.arange(n) / n
        c = np.exp(-2.0 * x)
    elif kind == 1:  # Asymmetric quadratic  
        x = np.arange(n) / n
        c = (1.0 - x) ** 2
    elif kind == 2:  # Power law

        c = x ** 0.5
    elif kind == 3:  # Bimodal
        x = np.arange(n) / n
        c = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif kind == 4:  # Sine-squared (shifted)
        x = np.arange(n) / n
        c = np.sin(np.pi * (x + 0.1)) ** 2
    else:  # Random
        c = rng.random(n)

    for name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = name
                return name
        except Exception:
            continue
    _cached_solver = cp.SCS
    return _cached_solver


def _fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = rfft(a, fsize)
    return irfft(A * A, fsize)[:size]


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _log_map(a):
    return np.log(np.maximum(a, 1e-300))


def _obj_norm(a):
    return 2.0 * len(a) * _fast_conv(a).max()


def _cross_corr_sliding(w, a):
    """r[i] = sum_{m=0}^{n-1} w[i+m]*a[m] for i=0..n-1 via FFT."""
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    return irfft(W * np.conj(A), L)[:n]


# === Soft-max objective and gradient (top-level callbacks) ===

def _sm_func(z):
    sm = _global_sm
    if time.monotonic() - sm.t0 > sm.time_lim:
        return 1e15
    a = _softmax(z)
    c = _fast_conv(a)
    T = sm.T
    Tc = T * c
    mx = Tc.max()
    ev = np.exp(Tc - mx)
    return (2.0 * sm.n / T) * (mx + np.log(ev.sum()))


def _sm_jac(z):
    sm = _global_sm
    if time.monotonic() - sm.t0 > sm.time_lim:
        return np.zeros(sm.n)
    a = _softmax(z)
    c = _fast_conv(a)
    T = sm.T
    Tc = T * c
    mx = Tc.max()
    ev = np.exp(Tc - mx)
    S = ev.sum()
    w = ev / S
    n = sm.n
    r = _cross_corr_sliding(w, a)
    s = np.dot(w, c)
    return 4.0 * n * a * (r - s)


# === Optimization routines ===

def _lbfgs_smooth(a0, time_budget, T_hi=80.0, T_lo=2.0, max_iter=3000):
    """L-BFGS-B on log-simplex with soft-max smoothing and geometric T schedule."""
    global _global_sm
    n = len(a0)
    z = _log_map(a0).copy()
    t0 = time.monotonic()
    
    class _SM:
        pass
    sm = _SM()
    sm.n = n
    sm.t0 = t0
    sm.time_lim = time_budget
    sm.T = T_hi
    _global_sm = sm
    
    n_stages = 4
    temps = np.geomspace(T_hi, T_lo, n_stages)
    per = max(50, max_iter // n_stages)
    
    for Tv in temps:
        if time.monotonic() - t0 > time_budget:
            break
        if time_budget - (time.monotonic() - t0) < 3.0:
            break
        sm.T = Tv
        try:
            res = minimize(_sm_func, z, jac=_sm_jac, method='L-BFGS-B',
                           options={'maxiter': per, 'ftol': 1e-14, 'gtol': 1e-8,
                                    'maxfun': per * 30, 'disp': False})
            z = res.x
        except Exception:
            pass
        sc = _obj_norm(_softmax(z))
        print(f"    L-BFGS T={Tv:.1f}: {sc:.6f} [{time.monotonic()-t0:.0f}s]")
    
    return _softmax(z)


def _build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def _lp_alternate(a0, time_budget, solver, n_iters=5000):
    """LP alternating with Nesterov-style averaging."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_sc = _obj_norm(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        b_new = _lp_best_response(a, solver)
        if b_new is None:
            break
        alpha = 2.0 / (it + 2)
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a /= a.sum()
        sc = _obj_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if (it + 1) % 1000 == 0:
            print(f"    LP-alt iter={it+1}: {best_sc:.6f} [{time.monotonic()-t0:.0f}s]")
    
    return best_a, best_sc


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    else:
        fi = np.ones(new_n) / new_n
    return fi


def _peak_shave(a0, time_budget, n_steps=3000):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_sc = _obj_norm(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = _fast_conv(a)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1]
        
        improved = False
        for ci in order[:6]:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.93, 0.88, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_t = a.copy()
                a_t[i] = new_val
                a_t[mask] = others + diff * others / os
                a_t /= a_t.sum()
                sc = _obj_norm(a_t)
                if sc < best_sc - 1e-14:
                    a = a_t
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            if len(order) >= 2:
                i1 = idx[order[0]]
                i2 = idx[order[1]]
                if i1 != i2 and a[i1] > 1e-14 and a[i2] > 1e-14:
                    for f1, f2 in [(0.95, 0.95), (0.9, 0.95), (0.95, 0.9), (0.85, 0.9)]:
                        a_t = a.copy()
                        a_t[i1] *= f1
                        a_t[i2] *= f2
                        a_t = np.maximum(a_t, 0)
                        a_t /= a_t.sum()
                        sc = _obj_norm(a_t)
                        if sc < best_sc - 1e-14:
                            a = a_t
                            best_sc = sc
                            best_a = a.copy()
                            improved = True
                            break
            if not improved:
                top3 = order[:3]
                a_trial = a.copy()
                for ci in top3:
                    i = idx[ci]
                    a_trial[i] *= 0.95
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                sc = _obj_norm(a_trial)
                if sc < best_sc - 1e-14:
                    a = a_trial
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    break
    
    return best_a, best_sc


def _structured_init(n, rng, kind=0):
    if kind == 0:
        x = np.arange(n, dtype=np.float64) / n
        c = np.exp(-2.5 * x)
    elif kind == 1:
        x = np.arange(n, dtype=np.float64) / n
        c = (1.0 - x) ** 2.5
    elif kind == 2:
        x = (np.arange(n, dtype=np.float64) + 1) / (n + 1)
        c = x ** 0.7
    elif kind == 3:
        x = np.arange(n, dtype=np.float64) / n
        c = np.exp(-30.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-30.0 * (x - 0.75) ** 2)
    elif kind == 4:
        x = np.arange(n, dtype=np.float64) / n
        c = np.sin(np.pi * (0.15 + 0.7 * x)) ** 3
    elif kind == 5:
        x = np.arange(n, dtype=np.float64) / n
        c = 1.0 / (1.0 + 50.0 * x)
    else:
        c = rng.random(n)
    c = np.maximum(c, 1e-10)
    c /= c.sum()
    return c


def _eval_safe(a):
    try:
        al = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(al))
    except Exception:
        return float('inf')
