"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative gradient descent
with ultra-low-temperature peak targeting, aggressive coordinate descent,
multi-length exploration, and structured initializations. Key improvements:
(1) much lower soft-max temperatures to target individual peaks,
(2) faster per-iteration updates with vectorized FFT ops,
(3) multi-peak balancing with adaptive weights,
(4) structured initializations exploiting spectral properties,
(5) longer focused polishing phases.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_normalized(p):
    """Compute 2n*max_b for normalized p (sum=1)."""
    n = len(p)
    b = fast_convolve(p)
    return 2.0 * n * b.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_p(p):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(p, 1e-300))


def compute_grad_b_k(p, k, n, fft_size):
    """
    Compute gradient of b[k] = sum_i p[i]*p[k-i] w.r.t. p.
    grad[i] = p[k-i] + p[i-k] (0 if index out of range [0, n-1]).
    Uses FFT-based cross-correlation.
    """
    # Build the shifted version
    shifted = np.zeros(n)
    # p[k - i] for valid i: need 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        shifted[lo:hi+1] = p[k - 1 : lo - 1 : -1] if False else p[k - np.arange(lo, hi+1)]
    # p[i - k] for valid i: need 0 <= i-k < n, i.e., max(0, k) <= i <= min(n-1, k+n-1)
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        shifted[lo2:hi2+1] += p[np.arange(lo2, hi2+1) - k]
    return shifted


def compute_grad_all_peaks(p, weights, n, fft_size):
    """
    Compute weighted sum of gradients for multiple peaks.
    weights: list of (k, w) pairs.
    """
    n = len(p)
    grad = np.zeros(n)
    for k, w in weights:
        shifted = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            shifted[lo:hi+1] = p[k - np.arange(lo, hi+1)]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            shifted[lo2:hi2+1] += p[np.arange(lo2, hi2+1) - k]
        grad += w * shifted
    return grad


def multiplicative_gd(p0, time_limit, T_init=0.001, T_final=0.00005,
                      lr=0.1, n_top=3, verbose=False, verbose_interval=5000,
                      hold_frac=0.2, seed=0):
    """
    Multiplicative gradient descent in log-space with temperature annealing.
    Targets the top n_top peaks with soft-max weighting.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 1e-300)
    s = p.sum()
    p = p / s
    z = z_from_p(p)
    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    max_no_improve = 5000
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Temperature: hold at T_init for hold_frac, then anneal
        if progress < hold_frac:
            T = T_init
        else:
            p2 = (progress - hold_frac) / (1.0 - hold_frac)
            T = T_init * (T_final / T_init) ** p2

        p_curr = softmax_from_z(z)
        b = fast_convolve(p_curr)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Soft-max weights
        b_shifted = b - max_b
        exp_b = np.exp(b_shifted / T)
        w = exp_b / exp_b.sum()

        # Use only top n_top peaks for gradient computation (speed)
        top_idx = np.argsort(b)[::-1][:n_top]
        peak_weights = [(int(k), w[k]) for k in top_idx if w[k] > 1e-6]
        if not peak_weights:
            peak_weights = [(int(np.argmax(b)), 1.0)]

        # Compute combined gradient
        grad_p = compute_grad_all_peaks(p_curr, peak_weights, n, fft_size)

        # Transform to z-space: grad_z = p * (grad_p - p'^T * grad_p)
        S = np.dot(grad_p, p_curr)
        grad_z = p_curr * (grad_p - S)

        # Adam-like update (simple momentum)
        lr_t = lr * (1.0 - 0.5 * progress)
        z = z - lr_t * grad_z

        # Evaluate
        p_new = softmax_from_z(z)
        score_new = 2.0 * n * fast_convolve(p_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > max_no_improve:
            z = z_from_p(best_p) + rng.standard_normal(n) * 0.01
            no_improve = 0
            lr = max(0.01, lr * 0.6)

        if verbose and t % verbose_interval == 0:
            print(f"      t={t} T={T:.6f} lr={lr_t:.4f} score={best_score:.6f}")

    return best_p, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm, period=1)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(parent, n, rng):
    """Generate diverse starting points."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.005, 0.01, 0.02, 0.05]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'pert_{scale}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('interp', interp))

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Raised cosine (periodic)
    x = np.arange(n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 1e-6)
    rc = rc / rc.sum()
    candidates.append(('raised_cos', rc))

    # Smooth periodic (sinc-like)
    x = (np.arange(n) - n/2 + 0.5) / n
    sinc_seq = np.sinc(3 * x)
    sinc_seq = np.maximum(sinc_seq, 1e-6)
    sinc_seq = sinc_seq / sinc_seq.sum()
    candidates.append(('sinc', sinc_seq))

    # Two-harmonic
    for freq in [1, 2, 3]:
        x = np.arange(n)
        har = 1 + 0.3 * np.cos(2 * np.pi * freq * x / n)
        har = np.maximum(har, 1e-6)
        har = har / har.sum()
        candidates.append((f'harmonic_{freq}', har))

    # Gaussian in time
    mid = (n - 1) / 2.0
    gauss = np.exp(-((x - mid) ** 2) / (2 * (n/6) ** 2))
    gauss = np.maximum(gauss, 1e-6)
    gauss = gauss / gauss.sum()
    candidates.append(('gaussian', gauss))

    # Triangle
    tri = 1 - np.abs(2 * x / n - 1)
    tri = np.maximum(tri, 0.01)
    tri = tri / tri.sum()
    candidates.append(('triangle', tri))

    # Random smooth
    n_freq = 5
    coeffs = rng.standard_normal(n_freq)
    phases = rng.uniform(0, 2*np.pi, n_freq)
    rand_seq = np.ones(n) / n
    for f in range(1, n_freq+1):
        rand_seq += 0.1 * coeffs[f-1] * np.cos(2*np.pi*f*x/n + phases[f-1])
    rand_seq = np.maximum(rand_seq, 0.001)
    rand_seq = rand_seq / rand_seq.sum()
    candidates.append(('rand_smooth', rand_seq))

    return candidates

            s = 2.0 * n * fast_convolve(init_a).max()
            scored.append((s, label, init_a))
        scored.sort(key=lambda x: x[0])
        all_inits[n] = scored

    # Run GD on top initializations
    time_per_length = (t_p1_end - time.monotonic()) / len(lengths)

    for n_cur in lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc_n = min(time_per_length, rem * 0.9)
        t_p1_start = time.monotonic()
        print(f"\n  n={n_cur} (alloc={t_alloc_n:.0f}s)")

        # Pick top 2-3 initializations
        top_inits = all_inits[n_cur][:3]
        time_per_init = t_alloc_n / len(top_inits)

        for idx, (s_init, label, init_a) in enumerate(top_inits):
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_sub = min(time_per_init, rem * 0.85)
            if t_sub < 20:
                continue

            print(f"    {label} (init={s_init:.6f}), budget={t_sub:.0f}s")

            # Run with different temperature settings
            best_local_p, best_local_s = multiplicative_gd(
                init_a, t_sub,
                T_init=0.002, T_final=0.0001,
                lr=0.15, n_top=3,
                verbose=False, seed=seed + idx
            )
            best_seq, best_score = evaluate_and_track(best_local_p, f"gd_n{n_cur}_{label}", best_seq, best_score)

            # Second run with different params
            rem = deadline - time.monotonic() - 200
            if rem > 30:
                t_sub2 = min(t_sub * 0.6, rem * 0.5)
                if t_sub2 > 15:
                    best_local_p2, best_local_s2 = multiplicative_gd(
                        init_a, t_sub2,
                        T_init=0.005, T_final=0.0002,
                        lr=0.08, n_top=5,
                        verbose=False, seed=seed + idx + 100
                    )
                    best_seq, best_score = evaluate_and_track(best_local_p2, f"gd2_n{n_cur}_{label}", best_seq, best_score)

        print(f"  After n={n_cur}: best={best_score:.6f}")

    # === Phase 2: Focused long run on best ===
    print(f"\n=== Phase 2: Focused Refinement (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_p2_end = time.monotonic() + (deadline - time.monotonic()) * 0.20
    t_p2 = t_p2_end - time.monotonic()
    print(f"  n={n_best}, time={t_p2:.0f}s")

    # Long run with very low final temperature
    a_res, score_res = multiplicative_gd(
        best_seq, t_p2 * 0.6,
        T_init=0.001, T_final=0.00005,
        lr=0.1, n_top=2,
        verbose=True, verbose_interval=10000,
        hold_frac=0.3, seed=999
    )
    best_seq, best_score = evaluate_and_track(a_res, "focused_gd", best_seq, best_score)
    print(f"  After focused GD: {best_score:.6f}")

    # Another long run with different params
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        a_res2, score_res2 = multiplicative_gd(
            best_seq, min(rem * 0.5, t_p2 * 0.4),
            T_init=0.003, T_final=0.0001,
            lr=0.05, n_top=4,
            verbose=False, seed=888
        )
        best_seq, best_score = evaluate_and_track(a_res2, "focused_gd2", best_seq, best_score)
        print(f"  After focused GD2: {best_score:.6f}")

    # === Phase 3: Aggressive local search ===
    print(f"\n=== Phase 3: Local Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    n_best = len(best_seq)

    # Peak shaving
    if rem > 30:
        t_sub = min(rem * 0.4, 120.0)
        a_ps, _ = peak_shave_v2(best_seq, t_sub, n_trials=10000)
        best_seq, best_score = evaluate_and_track(a_ps, "peak_shave", best_seq, best_score)
        print(f"  After peak shave: {best_score:.6f}")

    # Coordinate descent
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_sub = min(rem * 0.5, 100.0)
        a_cd, _ = coord_desc_v2(best_seq, t_sub, n_sweeps=200)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord desc: {best_score:.6f}")

    # Another peak shave round
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_sub = min(rem * 0.6, 80.0)
        a_ps2, _ = peak_shave_v2(best_seq, t_sub, n_trials=8000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave2", best_seq, best_score)
        print(f"  After peak shave 2: {best_score:.6f}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        # Very short GD with minimal temperature
        t_sub = min(rem * 0.4, 40.0)
        a_fin, _ = multiplicative_gd(
            best_seq, t_sub,
            T_init=0.0005, T_final=0.00002,
            lr=0.05, n_top=1,
            verbose=False, seed=555
        )
        best_seq, best_score = evaluate_and_track(a_fin, "final_gd", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.7, 20.0)
        a_ps_fin, _ = peak_shave_v2(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_ps_fin, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
