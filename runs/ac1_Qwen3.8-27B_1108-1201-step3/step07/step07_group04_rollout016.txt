combined with LP fictitious play for good initialization and
L-BFGS polish for final refinement.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from time import monotonic

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_gradient_peak(a, k):
    """Compute the gradient of conv(a,a)[k] with respect to a.
    grad[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
    """
    n = len(a)
    g = np.zeros(n)
    # First term: a[k-m] for m such that 0 <= k-m < n, i.e., k-n+1 <= m <= k
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start <= m_end:
        g[m_start:m_end+1] += a[k - np.arange(m_start, m_end+1)]
    # Second term: a[m-k] for m such that 0 <= m-k < n, i.e., k <= m <= k+n-1
    m_start2 = max(0, k)
    m_end2 = min(n - 1, k + n - 1)
    if m_start2 <= m_end2:
        g[m_start2:m_end2+1] += a[np.arange(m_start2, m_end2+1) - k]
    return g

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                        n_restarts=5, verbose=True):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = monotonic()
    it = 0
    global_no_improve = 0
    
    for restart in range(n_restarts + 1):
        if monotonic() - t0 > time_limit:
            break
        
        v = np.zeros(n)  # momentum
        lr = lr_init * (0.8 ** restart)  # reduce lr on restarts
        no_improve = 0
        restart_time = min(60.0, time_limit / (n_restarts + 1))
        restart_start = monotonic()
        
        while monotonic() - t0 < time_limit and monotonic() - restart_start < restart_time:
            it += 1
            
            c = fast_convolve(a)
            max_c = c.max()
            score = 2.0 * n * max_c
            
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                global_no_improve = 0
            else:
                no_improve += 1
                global_no_improve += 1
            
            # Find top-K peaks
            k_actual = min(top_k, len(c))
            peaks = np.argpartition(c, -k_actual)[-k_actual:]
            peak_vals = c[peaks]
            
            # Softmax weighting
            temp = max(0.001 * max_c, 1e-15)
            scaled = (peak_vals - peak_vals.max()) / temp
            w = np.exp(scaled)
            w /= w.sum()
            
            # Compute weighted gradient
            g = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-10:
                    continue
                g += w[i] * conv_gradient_peak(a, int(peaks[i]))
            
            # Nesterov momentum step
            a_lookahead = a + momentum * v
            # Recompute gradient at lookahead? (expensive, skip for now)
            # Just use the gradient at current a
            
            v = momentum * v - lr * g
            a_candidate = a + v
            
            # Project onto simplex
            a_new = project_simplex(a_candidate)
            
            # Ensure non-negative and valid
            a_new = np.maximum(a_new, 0)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / a_new_sum
            
            a = a_new
            
            # Adaptive lr
            if no_improve > 100:
                lr *= 0.95
            if no_improve > 500:
                break
        
        # Check if we should restart
        if monotonic() - t0 > time_limit:
            break
        
        # Perturb for next restart
        if restart < n_restarts:
            a = best_a + np.random.standard_normal(n) * (0.001 * (1 + 0.5 * restart))
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            if verbose:
                elapsed = monotonic() - t0
                print(f"  SubGrad restart={restart+1} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines parametric initialization, LP fictitious play (PDLP),
fast subgradient descent with Nesterov momentum, L-BFGS smoothing,
and greedy peak shaving. Explores multiple sequence lengths with

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_grad_at_peak(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. O(n) time."""
    n = len(a)
    g = np.zeros(n)
    # First term: a[k-m] for valid m
    ms = max(0, k - n + 1)
    me = min(n - 1, k)
    if ms <= me:
        g[ms:me + 1] += a[k - np.arange(ms, me + 1)]
    # Second term: a[m-k] for valid m
    ms2 = max(0, k)
    me2 = min(n - 1, k + n - 1)
    if ms2 <= me2:
        g[ms2:me2 + 1] += a[np.arange(ms2, me2 + 1) - k]
    return g


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)

    cssv = np.cumsum(u)

    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)

            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.SCS
    return cp.SCS

    if solver is None:
        solver = get_lp_solver()

    half = (n + 1) // 2
    b_half = cp.Variable(half, nonneg=True)
    M_var = cp.Variable()

    if n % 2 == 0:
        b_full = cp.hstack([b_half, cp.flip(b_half)])
    else:
        b_full = cp.hstack([b_half, cp.flip(b_half[1:])])

    # Build conv matrix T: T @ b_full = conv(a_fixed, b_full)
    T = np.zeros((2 * n - 1, n))

        T[j:j + n, j] = a_fixed

        T @ b_full <= M_var,
        cp.sum(b_full) == 1
    ]

    try:

        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Top-K peaks with softmax weighting
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        peak_vals = c[peaks]
        temp = max(0.002 * max_c, 1e-15)
        scaled = (peak_vals - peak_vals.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / ws

        # Weighted gradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            g += w[i] * conv_grad_at_peak(a, int(peaks[i]))

        # Nesterov momentum
        v = momentum * v - lr * g
        a_candidate = a + v
        a_new = project_simplex(a_candidate)
        a = a_new

        # Adaptive learning rate
        lr *= 0.9995
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
        if no_improve > 500:
            # Restart with perturbation
            stall_restarts += 1
            a = best_a + np.random.standard_normal(n) * 0.001 * (1 + 0.5 * stall_restarts)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr_init * 0.7 ** stall_restarts, 0.0005)
            no_improve = 0
            stall_restarts = min(stall_restarts, 10)

        if verbose and it % 5000 == 0:
            print(f"  SG iter={it} ({monotonic()-t0:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS polish using log-sum-exp smoothing of the max."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        zs = z - z.max()
        exp_z = np.exp(zs)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        zs = z - z.max()
        exp_z = np.exp(zs)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        k_actual = min(50, len(w))
        top_peaks = np.argpartition(w, -k_actual)[-k_actual:]
        for i in range(k_actual):
            ki = int(top_peaks[i])
            wi = w[ki]
            if wi < 1e-12:
                continue
            grad_a += wi * conv_grad_at_peak(a, ki)

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            zs = z_final - z_final.max()
            a_out = np.exp(zs)
            a_out = a_out / a_out.sum()
        else:
            a_out = np.maximum(a0, 1e-15)
            a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    return a_out


def peak_shave(a0, time_limit, max_steps=5000):
    """Greedy peak shaving local search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    for step in range(max_steps):
        if monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        k_peak = int(np.argmax(c))

        if c[k_peak] <= best_score / (2.0 * n) + 1e-15:
            break

        # Find top contributors to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False

        # Strategy 1: Scale down single element
        for ci in top_contrib[:4]:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.9, 0.85]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Scale down pair
        if not improved:
            for ci in top_contrib[:4]:
                i = int(idx[ci])
                j = int(k_peak - i)
                if i >= n or j >= n or i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.9]:
                    for f2 in [0.93, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Small perturbation
        if not improved:
            for attempt in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.0003
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 80:
                break
        else:
            failed = 0

    return best_a, best_score


def parametric_init(n, alpha=0.5, beta=0.0):
    """Create a bowl-shaped initial sequence.
    a[i] proportional to 1 + alpha * (2*i/(n-1) - 1)^2 + beta * cos(2*pi*i/n)
    """
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 + alpha * (2.0 * t - 1.0)**2
    if beta != 0:
        a += beta * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Force a sequence to be palindromic."""
    n = len(a)
    half = (n + 1) // 2
    a_half = a[:half]
    if n % 2 == 0:
        out = np.concatenate([a_half, a_half[::-1]])
    else:
        out = np.concatenate([a_half, a_half[1:][::-1]])
    return out / out.sum()


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate via evaluate_sequence and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Parametric initialization (bowl-shaped families)
    2. LP Fictitious Play at multiple lengths (palindromic)
    3. Subgradient descent (fast, many iterations)
    4. L-BFGS polish with adaptive temperature
    5. Peak shaving (greedy local search)
    6. Final multi-round refinement
    """
    np.random.seed(12345)
    deadline = monotonic() + 1780.0  # Leave 20s buffer

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # --- Get parent sequence ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n0 = 768
        best_seq = parametric_init(n0, alpha=0.4)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial parametric: n={n0}, score={best_score:.6f}")

    # === Phase 1: Parametric search for good initialization ===
    print(f"\n=== Phase 1: Parametric Initialization ===")
    n_parent = len(best_seq)
    candidate_inits = {}

    for n_try in [n_parent, 1024, 1536, 2048]:
        if n_try in candidate_inits:
            continue
        for alpha in [0.2, 0.4, 0.6, 1.0]:
            a_p = parametric_init(n_try, alpha=alpha)
            score_p = 2.0 * n_try * fast_convolve(a_p).max()
            key = f"n{n_try}_a{alpha}"
            candidate_inits[key] = (a_p, score_p)
            if score_p < best_score:
                best_seq, best_score = check_and_update(a_p, key, best_seq, best_score)

    # Pick the best parametric init for each n
    best_inits_by_n = {}
    for key, (a_p, score_p) in candidate_inits.items():
        n = len(a_p)
        if n not in best_inits_by_n or score_p < best_inits_by_n[n][1]:
            best_inits_by_n[n] = (a_p, score_p)

    print(f"  After parametric: best={best_score:.6f}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play ===")
    # Try multiple lengths, prioritize parent length
    n_list = sorted(set([n_parent, 1024, 1536]), key=lambda x: abs(x - n_parent))

    for n_try in n_list[:3]:
        rem = deadline - monotonic() - 300
        if rem < 60:
            break

        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            initial = best_seq.copy()
        elif n_try in best_inits_by_n:
            initial = best_inits_by_n[n_try][0]
        else:
            initial = interpolate_sequence(best_seq, n_try)

        # Make palindromic
        initial = make_palindromic(initial)

        t_alloc = min(rem * 0.4, 300.0)
        print(f"  FP n={n_try}, budget={t_alloc:.0f}s, init_score={2*n_try*fast_convolve(initial).max():.6f}")

        a_fp, score_fp, iters = fictitious_play(initial, t_alloc, solver=solver, verbose=True, use_palin=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_try}_it{iters}", best_seq, best_score)

    # === Phase 3: Subgradient Descent (fast iterations) ===
    print(f"\n=== Phase 3: Subgradient Descent ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic() - 250
        if rem > 30:
            t_sg = min(rem * 0.5, 180.0)
            print(f"  SG n={n_best}, budget={t_sg:.0f}s")
            a_sg, _ = subgradient_descent(best_seq, t_sg, lr_init=0.004, momentum=0.9, top_k=40, verbose=True)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 4: L-BFGS Polish ===
    print(f"\n=== Phase 4: L-BFGS Polish ===")
    if best_seq is not None:
        for T in [20, 50, 150]:
            rem = deadline - monotonic() - 150
            if rem < 15:
                break
            t_lb = min(rem * 0.35, 60.0)
            print(f"  L-BFGS T={T}, n={len(best_seq)}, budget={t_lb:.0f}s")
            a_lb = lbgfs_smooth(best_seq, t_lb, T=T, maxiter=2000)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 100
        if rem > 10:
            t_ps = min(rem * 0.5, 60.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: Second round FP + SG ===
    print(f"\n=== Phase 6: Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic() - 100
        if rem > 60:
            t_fp2 = min(rem * 0.3, 120.0)
            print(f"  FP2 n={n_best}, budget={t_fp2:.0f}s")
            init2 = make_palindromic(best_seq)
            a_fp2, _, iters2 = fictitious_play(init2, t_fp2, solver=solver, verbose=False, use_palin=True)
            best_seq, best_score = check_and_update(a_fp2, f"FP2_n{n_best}", best_seq, best_score)

            # Quick subgradient
            rem = deadline - monotonic() - 50
            if rem > 15:
                t_sg2 = min(rem * 0.5, 60.0)
                a_sg2, _ = subgradient_descent(best_seq, t_sg2, lr_init=0.002, momentum=0.9, top_k=50)
                best_seq, best_score = check_and_update(a_sg2, "subgrad2", best_seq, best_score)

    # === Phase 7: Try a different length ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_n = n_best + 256 if n_best < 2000 else n_best - 256
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            rem = deadline - monotonic() - 50
            if rem > 40:
                t_alt = min(rem * 0.4, 80.0)
                a_alt = interpolate_sequence(best_seq, alt_n)
                a_alt = make_palindromic(a_alt)
                print(f"  Alt length n={alt_n}, budget={t_alt:.0f}s")
                a_fp_alt, _, _ = fictitious_play(a_alt, t_alt * 0.5, solver=solver, verbose=False, use_palin=True)
                rem2 = deadline - monotonic() - 30
                if rem2 > 10:
                    a_sg_alt, _ = subgradient_descent(a_fp_alt, rem2 * 0.5, lr_init=0.003, momentum=0.9, top_k=30)
                    a_fp_alt = a_sg_alt
                best_seq, best_score = check_and_update(a_fp_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 10:
            t_final = min(rem * 0.4, 30.0)
            a_final = lbgfs_smooth(best_seq, t_final, T=30, maxiter=2000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - monotonic() - 5
        if rem > 5:
            a_ps_final, _ = peak_shave(best_seq, min(rem * 0.8, 15.0), max_steps=2000)
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

        # One more subgradient sweep
        rem = deadline - monotonic() - 2
        if rem > 5:
            a_sg_final, _ = subgradient_descent(best_seq, min(rem * 0.8, 10.0), lr_init=0.001, momentum=0.9, top_k=50)
            best_seq, best_score = check_and_update(a_sg_final, "final_sg", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
