def fast_correlate(lam, a):
    """
    Compute c[i] = sum_j lam[i+j] * a[j] for i = 0, ..., n-1.
    This is the cross-correlation of lam (length 2n-1) with a (length n).
    """
    n = len(a)
    # c[i] = sum_j lam[i+j] * a[j] = (lam cross-correlated with a)[i]
    # This is the same as: convolve(a, lam reversed)[offset]
    # Or: it's the cross-correlation
    # Let's use FFT: c = ifft(fft(a, N) * conj(fft(lam, N)))
    # Wait, let me think more carefully.
    # c[i] = sum_{j=0}^{n-1} lam[i+j] * a[j]
    # This is the cross-correlation of lam and a, evaluated at lag i
    # cross_corr(lam, a)[i] = sum_j lam[j+i] * conj(a[j])
    # Since a is real: cross_corr(lam, a)[i] = sum_j lam[j+i] * a[j]
    # To compute via FFT: cross_corr = ifft(fft(lam, N) * conj(fft(a, N)))
    # But we need the "positive lag" part
    N = 1 << (2 * n - 1).bit_length()
    L = fft(lam, N)
    A = fft(a, N)
    cc = ifft(L * np.conj(A)).real
    # cc[k] = sum_j lam[j+k] * a[j] (with wraparound, but since N >= 2n-1, no wraparound for k in [0, n-1])
    return cc[:n]


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def score_normalized(a):
    """Compute score assuming sum(a) = 1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_soft_max(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and adaptive temperature."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    
    def func(z, T_val):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp
    
    def jac(z, T_val):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break
        try:
            res = minimize(
                fun=lambda z: func(z, stage_T),
                x0=z,
                jac=lambda z: jac(z, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def primal_dual_optimize(a0, time_limit, n_outer=2000, n_inner_max=50):
    """
    Primal-dual optimization for minimizing max(conv(a,a)).
    Dual: weights on each lag position
    Primal: minimize weighted sum of auto-conv lags via projected gradient
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    # Initialize dual weights
    b = fast_convolve(a)
    max_b = b.max()
    # Soft-max weighting: emphasize peaks
    T_dual = 10.0  # temperature for dual weighting
    lam = np.exp(T_dual * (b - max_b))
    lam = lam / lam.sum()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    eta_primal = 0.01  # primal step size
    eta_dual = 0.1     # dual step size
    
    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Inner loop: minimize sum_k lam[k] * b_k(a) via projected gradient
        n_inner = min(n_inner_max, 5 + outer // 50)
        for inner in range(n_inner):
            # Compute gradient: g[i] = 2 * sum_k lam[k] * a[k-i] (with bounds)
            # This is 2 * correlate(lam, a)[i]
            c = fast_correlate(lam, a)
            g = 2.0 * c
            
            # Projected gradient step
            a_new = simplex_project(a - eta_primal * g)
            
            # Line search (simple: just accept if score doesn't increase too much)
            a = a_new
        
        # Compute new auto-conv
        b = fast_convolve(a)
        score = score_normalized(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Update dual weights
        max_b = b.max()
        # Multiplicative weight update
        lam = lam * np.exp(eta_dual * (b - max_b))
        lam = np.maximum(lam, 1e-30)
        lam = lam / lam.sum()
        
        # Adapt primal step size
        if outer % 50 == 0 and outer > 0:
            eta_primal *= 0.95
        eta_primal = max(eta_primal, 1e-6)
        
        if outer % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PrimalDual iter={outer} ({elapsed:.0f}s): score={best_score:.6f}, max_b={max_b:.8f}")
    
    return best_a, best_score


def freq_domain_perturb(a0, time_limit, n_trials=500):
    """
    Try frequency-domain perturbations to reduce the peak.
    Add small perturbations in the frequency domain that are "anti-peak" 
    (reduce the maximum of the auto-convolution).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current FFT
        A = fft(a, fft_size)
        
        # Random perturbation in frequency domain
        # Perturb the phase of a few random frequency bins
        n_perturb = rng.integers(3, 10)
        freqs = rng.integers(1, fft_size // 2, size=n_perturb)
        phases = rng.uniform(-0.1, 0.1, size=n_perturb)
        amps = rng.uniform(0.001, 0.01, size=n_perturb)
        
        A_new = A.copy()
        for f, p, amp in zip(freqs, phases, amps):
            # Perturb the phase while keeping magnitude approximately the same
            A_new[f] *= np.exp(1j * p)
            A_new[fft_size - f] *= np.exp(-1j * p)
            # Small amplitude change
            A_new[f] *= (1 + amp * rng.standard_normal())
            A_new[fft_size - f] = np.conj(A_new[f]) if f != fft_size - f else A_new[f]
        
        # Inverse FFT and check non-negativity
        a_new = ifft(A_new, fft_size).real[:n]
        
        # Check if mostly non-negative
        if a_new.min() < -0.01 * a.max():
            continue
        
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 0.01:
            continue
        a_new = a_new / sn
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    
    return best_a, best_score


def peak_balance_local(a0, time_limit, n_steps=10000):
    """
    Aggressive peak balancing: identify all active peaks and try to equalize them
    by adjusting coordinates that contribute to multiple peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all active peaks (within 0.5% of max)
        threshold = max_b * 0.997
        active_peaks = np.where(b >= threshold)[0]
        
        if len(active_peaks) == 0:
            break
        
        # For each active peak, find the top contributors
        found_improvement = False
        for pk in active_peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            
            for ci in top_idx:
                i = idx[ci]
                j = pk - i
                
                # Try reducing a[i] and redistributing
                for factor in [0.98, 0.96, 0.94, 0.92, 0.90]:
                    new_val = a[i] * factor
                    diff = a[i] - new_val
                    if diff < 1e-15:
                        continue
                    
                    # Redistribute to minimize impact on peaks
                    # Try redistributing to positions far from all peaks
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a[mask].copy()
                    
                    # Weight redistribution by distance from peaks
                    positions = np.arange(n)
                    # Avoid positions that would create new peaks
                    target_weights = np.ones(n - 1)
                    for p in active_peaks:
                        # Reduce weight at positions that would increase b[p]
                        for t in np.arange(max(0, p - n + 1), min(n, p + 1)):
                            if t != i:
                                local_idx = t if t < i else t - 1
                                target_weights[local_idx] *= 0.5
                    target_weights = np.maximum(target_weights, 0.1)
                    target_weights = target_weights / target_weights.sum()
                    
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    a_trial[mask] = others + diff * target_weights
                    # Wait, this isn't right. Let me redo.
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    # Redistribute diff to other positions
                    other_positions = np.where(mask)[0]
                    a_trial[other_positions] = a[other_positions] + diff * target_weights
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        found_improvement = True
                        break
                if found_improvement:
                    break
            if found_improvement:
                break
        
        if not found_improvement:
            # Try a more global adjustment: reduce all top contributors slightly
            if len(active_peaks) >= 2:
                all_top = set()
                for pk in active_peaks[:5]:
                    js = max(0, pk - n + 1)
                    je = min(n - 1, pk)
                    idx = np.arange(js, je + 1)
                    contribs = a[idx] * a[pk - idx]
                    top = np.argsort(contribs)[::-1][:3]
                    all_top.update(idx[top].tolist())
                
                if all_top:
                    a_trial = a.copy()
                    for i in all_top:
                        a_trial[i] *= 0.97
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 0.01:
                        a_trial = a_trial / st
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            found_improvement = True
            
            if not found_improvement:
                # Try cosine taper modification
                # Modulate a[j] by (1 + eps * cos(2*pi*j/n))
                for eps in [0.01, 0.02, -0.01, -0.02]:
                    modulation = 1 + eps * np.cos(2 * np.pi * np.arange(n) / n)
                    a_trial = a * modulation
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 0.01:
                        a_trial = a_trial / st
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            found_improvement = True
                            break
    
    return best_a, best_score


def subgradient_descent(a0, time_limit, n_steps=20000, lr_init=0.01):
    """
    Subgradient descent on the simplex with averaging.
    Uses the subgradient of max_k b_k(a).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    lr = lr_init
    x_avg = a.copy()  # running average
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active set (peaks within 2% of max)
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average of subgradients at active points
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Step
        lr_step = lr / (1 + step * 0.01)
        a_new = simplex_project(a - lr_step * g)
        
        # Update running average (Polyak averaging)
        x_avg = ((step + 1) * x_avg + a_new) / (step + 2)
        
        a = a_new
        lr = max(lr * 0.999, 1e-8)
        
        # Also check the averaged iterate
        if step % 100 == 0:
            sc_avg = score_normalized(x_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = x_avg.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization using primal-dual, L-BFGS, subgradient,
    and peak balancing methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 0: Try structured initializations ===
    print(f"\n=== Phase 0: Initializations (best={best_score:.6f}) ===")
    n = n_best
    
    inits = {}
    # Cosine taper
    j = np.arange(n)
    init_cos = np.cos(np.pi * j / (n - 1)) ** 2
    init_cos = np.maximum(init_cos, 0)
    inits['cos2'] = init_cos / init_cos.sum()
    
    # Gaussian
    sigma = n / 4
    init_gauss = np.exp(-j ** 2 / (2 * sigma ** 2))
    inits['gauss'] = init_gauss / init_gauss.sum()
    
    # Sinc-squared (windowed)
    w = n / 3
    init_sinc = np.sinc(j / w) ** 2
    init_sinc = np.maximum(init_sinc, 0)
    inits['sinc2'] = init_sinc / init_sinc.sum()
    
    # Triangle
    init_tri = 1 - 2 * np.abs(j - n / 2 + 0.5) / n
    init_tri = np.maximum(init_tri, 0)
    inits['tri'] = init_tri / init_tri.sum()
    
    # Super-Gaussian (order 4)
    init_sg4 = np.exp(-np.abs(2 * (j - n / 2 + 0.5) / n) ** 4)
    inits['sg4'] = init_sg4 / init_sg4.sum()
    
    # Power cosine (order 4)
    init_pc4 = np.cos(np.pi * j / (n - 1)) ** 4
    init_pc4 = np.maximum(init_pc4, 0)
    inits['pc4'] = init_pc4 / init_pc4.sum()
    
    for name, init in inits.items():
        sc = evaluate_safe(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = init.copy()
            print(f"  Better init [{name}]: {sc:.6f}")
    
    # === Phase 1: Primal-dual optimization (400s) ===
    print(f"\n=== Phase 1: Primal-Dual (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  PrimalDual n={n_best}, budget={t_alloc:.0f}s")
        a_pd = primal_dual_optimize(best_seq, t_alloc)
        try_update(a_pd, "primal_dual")

    print(f"\n=== Phase 2: L-BFGS Soft-Max (best={best_score:.6f}) ===")

        a_lb = lbgfs_soft_max(best_seq, t_alloc, T_start=100.0, T_end=1.0, maxiter=6000)
        try_update(a_lb, "lbgfs_main")
        
        # Second pass at lower temperature
        rem = deadline - time.monotonic()
        t_alloc2 = min(rem * 0.12, 150.0)

        a_sg = subgradient_descent(best_seq, t_alloc)
        try_update(a_sg, "subgrad")
    
    # === Phase 4: Peak balancing (250s) ===
    print(f"\n=== Phase 4: Peak Balancing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
