    n = len(a)

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Scale to sum=1 first
    v_scaled = v / target_sum if target_sum > 0 else v
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v_scaled - theta, 0.0) * target_sum

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)

    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(f, beta):
    """Compute smooth max value and gradient."""
    n = len(f)

    w = np.exp(shifted)

    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
        w_sum = 1.0
    w /= w_sum

    # Gradient via FFT
    size = 2 * n - 1

    conv = irfft(W * F_rev, fft_size)[:size]
    grad = 2.0 * conv[n-1 : 2*n-1]
    
    return smooth_val, grad, w

        return None, M

    # Row k: A[k, j] = f[j] if 0 <= k-j < n, else 0
    # This is a "anti-bidiagonal" like structure
    
    # For efficiency, build as dense (for n up to ~2000, size*n = ~4M entries, manageable)
    A = np.zeros((size, n))

        A[k, lo:hi+1] = f[lo:hi+1]

                return None, M

        return None, M

    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):

    return g_val, M

    stuck = 0

        it += 1
        g, M, sum_g = full_lp_solve(f)
        
        if g is None:
            stuck += 1
            if stuck > 5:
                break
            continue
        
        if sum_g <= 1.0 + 1e-5:
            # LP says we can't improve - try with slightly relaxed bound
            # or perturb and restart
            stuck += 1
            if stuck > 20:

                f_pert = f + rng.standard_normal(n) * 0.002 * f.max()
                f = project_simplex(np.maximum(f_pert, 0))

            continue

        h = g / sum_g

        best_t = 0.05

            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue

            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

            elapsed = time.monotonic() - t0
            print(f"    lp iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "

            # Aggressive restart: concave transform
            alpha = 0.7
            f = np.power(f + 1e-15, alpha)
            f = f / f.sum()
            stuck = 0
    
    return best_f, best_obj


def fista_smooth(f_init, beta, time_limit, max_iter=20000):
    """
    FISTA on smooth max: (1/beta)*log(sum exp(beta*(f*f)[k]))
    Projected onto simplex.
    """
    t0 = time.monotonic()
    n = len(f_init)
    x = f_init.copy()
    y = x.copy()
    t_mom = 1.0
    best_x = x.copy()
    best_obj = fast_convolve(x).max()
    
    # Initial step size via power iteration on Hessian
    alpha = 0.01 / max(n, 10)
    
    it = 0
    no_improve = 0
    
    while it < max_iter and time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at y
        b_y = fast_convolve(y)
        shifted = beta * b_y - beta * b_y.max()
        max_shift = shifted.max()
        w = np.exp(shifted - max_shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(b_y)
            w[b_y.argmax()] = 1.0
            w_sum = 1.0
        else:
            w /= w_sum
        
        # Gradient via FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        f_rev = y[::-1]
        F_rev = rfft(f_rev, fft_size)
        conv = irfft(W * F_rev, fft_size)[:size]
        grad = 2.0 * conv[n-1:2*n-1]
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            alpha = min(alpha * 1.05, 1.0 / (beta * 4.0 * y.max() + 1e-10))
        
        # FISTA update
        x_new = project_simplex(y - alpha * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + (t_mom - 1.0) / (t_new + 1.0) * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y /= s_y
        else:
            y = x_new.copy()
        x = x_new
        t_mom = t_new
        
        # Track best
        if it % 200 == 0:
            cur_obj = fast_convolve(x).max()
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    # Reduce step size
                    alpha *= 0.8
                    no_improve = 0
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            cur_obj = fast_convolve(x).max()
            print(f"    fista iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} alpha={alpha:.6e} t={elapsed:.0f}s")
    
    return best_x, best_obj


def interp_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    best_obj = fast_convolve(x).max()

    # Conservative initial step size
    alpha = 1.0 / (beta * 4.0 * y.max() + 1.0)
    alpha = min(alpha, 1.0 / (2.0 * n))

    it = 0
    no_improve = 0
    b = fast_convolve(x)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while it < max_iter and time.monotonic() - t0 < time_limit:
        it += 1

        # Compute weights and gradient at y
        b_y = fast_convolve(y)
        shifted = beta * b_y - beta * b_y.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(size)
            w[b_y.argmax()] = 1.0
            w_sum = 1.0
        else:
            w /= w_sum

        # Gradient via FFT: grad_j = 2 * sum_k w[k] * f[k-j] for valid k-j
        W = rfft(w, fft_size)
        f_rev = y[::-1]
        F_rev = rfft(f_rev, fft_size)
        conv = irfft(W * F_rev, fft_size)[:size]
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # FISTA update
        x_new = project_simplex(y - alpha * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum_coeff = (t_mom - 1.0) / (t_new + 1.0)
        y = x_new + momentum_coeff * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y /= s_y
        else:
            y = x_new.copy()
        x = x_new
        t_mom = t_new

        # Track best periodically
        if it % 500 == 0:
            cur_obj = fast_convolve(x).max()
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_x = x.copy()
                no_improve = 0
                # Slightly increase step
                alpha = min(alpha * 1.02, 1.0 / (beta * 2.0 * x.max() + 1e-10))
            else:
                no_improve += 1
                if no_improve > 5:
                    alpha *= 0.85
                    no_improve = 0

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            cur_obj = fast_convolve(x).max()
            print(f"    fista iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} alpha={alpha:.3e} t={elapsed:.0f}s")

    return best_x, best_obj


def interp_sequence(f, new_n):
    """Interpolate sequence to new length and normalize."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def topk_lp_solve(f, k):
    """Solve LP with only top-K peaks as constraints."""
    n = len(f)
    b = fast_convolve(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, 0.0

    if k < len(b):
        top_idx = np.argpartition(b, -k)[-k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        kk = int(top_idx[row_i])
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])

    for solver in [cp.GLPK, cp.CBC, cp.SCS]:
        try:
            if solver == cp.SCS:
                prob.solve(solver=solver, max_iters=300000, eps=1e-9)
            else:
                prob.solve(solver=solver)
            break
        except Exception:
            continue

    if g.value is None:
        return None, M, 0.0
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    if np.any(np.isnan(g_val)):
        return None, M, 0.0
    return g_val, M, float(g_val.sum())


def lp_iterate_topk(f_init, time_limit, k_init=50, k_max=300, verbose=True):
    """LP iteration with adaptive top-K peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    k = k_init
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and k < k_max:
            k = min(k + 10, k_max)

        g, M, sum_g = topk_lp_solve(f, k)

        if g is None:
            stuck += 2
            if stuck > 8:
                break
            continue

        if sum_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 25:
                alpha = 0.6 + 0.3 * rng.random()
                f_pert = np.power(f + 1e-15, alpha)
                f_pert += rng.standard_normal(n) * 0.002 * f.max()
                f = project_simplex(np.maximum(f_pert, 0.0))
                k = min(k * 2, k_max)
                stuck = 0
            continue

        h = g / sum_g
        best_t = 0.1
        best_t_obj = fast_convolve(f).max()
        for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    lp_topk iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"k={k} sum_g={sum_g:.4f} t={elapsed:.0f}s")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: Hybrid full-LP + top-K LP + FISTA + multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
    p_score = float(evaluate_sequence(parent.tolist()))

    best_seq = parent.copy()
    best_score = p_score
    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Top-K LP on parent (fast, get quick improvement) ===
    print(f"\n=== Phase 1: Top-K LP (n={n_base}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = lp_iterate_topk(best_seq, time_limit=t1, k_init=40, k_max=200)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [topk_lp]: {score1:.6f}")

    # === Phase 2: Full LP (all constraints) ===
    print(f"\n=== Phase 2: Full LP (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s, n={n_b}")
        f2, _ = lp_iterate(best_seq, time_limit=t2)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [full_lp]: {score2:.6f}")

    # === Phase 3: FISTA polish ===
    print(f"\n=== Phase 3: FISTA polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        n_b = len(best_seq)
        for beta in [300.0, 800.0, 2000.0]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(rem * 0.3, 80.0)
            if t_sub < 10:
                break
            print(f"  FISTA beta={beta:.0f}, budget={t_sub:.0f}s")
            f3, _ = fista_smooth(best_seq, beta, time_limit=t_sub)
            score3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best_score = score3
                best_seq = f3.copy()
                print(f"  NEW BEST [fista_b{beta:.0f}]: {score3:.6f}")

    # === Phase 4: Another round of full LP ===
    print(f"\n=== Phase 4: Full LP round 2 (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 250.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, _ = lp_iterate(best_seq, time_limit=t4)
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp_r2]: {score4:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidates = [int(n_best * f) for f in [0.75, 1.2, 1.5, 2.0]]
    for fl in [512, 768, 1024, 1280, 1536, 2048, 2560]:
        if fl not in candidates:
            candidates.append(fl)
    candidates = [c for c in set(candidates) if 256 <= c <= 3000 and c != n_best]
    candidates.sort(key=lambda x: abs(x - n_best))

    for target_n in candidates[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t5 = min(rem * 0.25, 180.0)
        if t5 < 25:
            break

        f_interp = interp_sequence(best_seq, target_n)
        print(f"  n={target_n}, budget={t5:.0f}s")

        # Top-K LP on new length
        f5, _ = lp_iterate_topk(f_interp, time_limit=t5 * 0.6, k_init=40, k_max=150)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.5, 200.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, _ = lp_iterate(best_seq, time_limit=t6)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {score6:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        if parent is not None:
            best_seq = parent.copy()
        else:
            best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
