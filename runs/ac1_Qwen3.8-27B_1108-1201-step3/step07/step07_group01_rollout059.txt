"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based block coordinate descent
with adaptive step size, targeted peak shaving, and spectral initialization.
Main innovation: uses a linear program to find the most spread-out sequence g
that maintains bounded cross-correlation with the current sequence, combined
with line search for optimal mixing and periodic restarts to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def build_A_ub(f):
    """Build the constraint matrix for the LP.
    A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    """
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(2*n-2, j+n-1)
        # A[k, j] = f[k-j] for k in [j, j+n-1] (clamped to 2n-2)
        idx = np.arange(k_start, k_end+1)
        A[idx, j] = f[idx - j]
    return A

def lp_find_spread(f, M, time_limit=30.0):
    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    A_ub = build_A_ub(f)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': time_limit - (time.monotonic() - t0), 'presolve': True})
    
    if res.success:
        return np.maximum(res.x, 0)
    return None

def line_search(f, g, t_max=1.0, ftol=1e-12):
    """Find best t in [0, t_max] such that score((1-t)f + tg) is minimized."""
    n = len(f)
    base_score = score_from_a(f)
    best_t = 0.0
    best_score = base_score
    
    # Try a few values of t
    ts = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    ts = [t for t in ts if t <= t_max]
    
    for t in ts:
        a_trial = (1-t)*f + t*g
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        sc = score_from_a(a_trial)
        if sc < best_score - ftol:
            best_score = sc
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        t_lo = max(0, best_t * 0.5)
        t_hi = min(1.0, best_t * 1.5)
        for _ in range(10):
            t_mid = (t_lo + t_hi) / 2
            a_trial = (1-t_mid)*f + t_mid*g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                t_hi = t_mid
                continue
            a_trial = a_trial / s
            sc = score_from_a(a_trial)
            if sc < best_score:
                best_score = sc
                best_t = t_mid
                t_lo = t_mid
            else:
                t_hi = t_mid
    
    return best_t, best_score

            if others_sum < 1e-15:

            a_trial = np.maximum(a_trial, 0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

The auto-convolution of a probability vector of length n is a probability vector of length 2n-1. The maximum entry is at least 1/(2n-1) by the pigeonhole principle. But we can do much better.

For a "good" sequence, the auto-convolution should be close to uniform on its support. The most uniform auto-convolution would have all entries equal to 1/(2n-1), giving a score of 2n/(2n-1) ≈ 1. But this is impossible because the auto-convolution is a triangular shape (the number of terms in the sum varies with k).

The triangular envelope means b[k] ≈ (min(k+1, n, 2n-1-k)) * (1/n)^2 for a uniform sequence. The peak is at k=n-1 with value n/n^2 = 1/n, giving score 2.

"""
LP-based optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Combines linear programming for finding spread-out sequences, adaptive
line search, multiplicative weight updates, peak shaving, and multi-scale
exploration to achieve a sharp bound.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fast(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * float(b.max())


def build_lp_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        ks = j
        ke = min(2 * n - 2, j + n - 1)
        rows = np.arange(ks, ke + 1)
        A[rows, j] = f[rows - j]
    return A


def solve_lp_spread(f, M, timeout=30):
    n = len(f)
    A_ub = build_lp_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    remaining = max(1.0, timeout - (time.monotonic() - t0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': remaining, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def multiplicative_update(a, grad, eta, clip_val=2.0):
    n = len(a)
    grad_centered = grad - np.dot(a, grad)
    log_update = -eta * grad_centered
    log_update = np.clip(log_update, -clip_val, clip_val)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def subgradient_at_peak(a):
    b = fast_convolve(a)
    k_peak = int(np.argmax(b))
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k_peak - idx]
    return grad, k_peak


def peak_shave(a, time_limit=30.0, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_score = score_fast(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        improved = False
        for ci in np.argsort(contribs)[::-1][:8]:
            i = int(idx[ci])
            j = k_peak - i
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.95, 0.92, 0.9, 0.85, 0.8, 0.7]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                sc = score_fast(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    improved = True
                    break
            if improved:
                break
        
        if not improved and i != j and a[j] > 1e-14:
            for f1 in [0.95, 0.9, 0.85, 0.8]:
                for f2 in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    s2 = a_trial.sum()
                    if s2 < 1e-15:
                        continue
                    a_trial = a_trial / s2
                    sc = score_fast(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return a, best_score


def mirror_descent(a_init, time_limit, eta_start=0.05, eta_end=0.001,
                   max_iters=10000, verbose=False, label=""):
    n = len(a_init)
    a = np.maximum(a_init.copy().astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    eta = eta_start
    stuck = 0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, it / max_iters)
        eta = eta_start * (eta_end / eta_start) ** progress
        
        grad, _ = subgradient_at_peak(a)
        a = multiplicative_update(a, grad, eta)
        
        sc = score_fast(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 200:
            rng = np.random.default_rng(it * 31 + 7)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            eta = min(eta * 5, eta_start * 0.3)
        
        if verbose and it % 5000 == 0:
            print(f"  {label} MD it={it} t={time.monotonic()-t0:.0f}s "
                  f"eta={eta:.5f} cur={sc:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def lp_optimize_loop(a_init, time_limit, verbose=True, label="LP"):
    """Main LP-based optimization loop."""
    n = len(a_init)
    a = np.maximum(a_init.copy().astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    no_improve_count = 0
    max_no_improve = 30
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_convolve(a)
        M = float(b.max())
        current_score = 2.0 * n * M
        
        # Solve LP to find spread-out g
        g = solve_lp_spread(a, M, timeout=15)
        if g is None:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                # Perturb and retry
                rng = np.random.default_rng(iteration * 101 + 3)
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve_count = 0
            continue
        
        g = g / g.sum()
        
        # Line search for optimal t
        best_t = 0.0
        best_trial_score = current_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = score_fast(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_t = t
        
        if best_trial_score < best_score - 1e-14:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_trial_score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if verbose and (iteration % 10 == 0 or no_improve_count == 1):
            elapsed = time.monotonic() - t0
            print(f"  {label} it={iteration} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f} "
                  f"t={best_t:.2f} no_improve={no_improve_count}")
        
        if no_improve_count > max_no_improve:
            # Try perturbation
            rng = np.random.default_rng(iteration * 7919 + 1)
            a = best_a * (1 + rng.standard_normal(n) * 0.02)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve_count = 0
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fast(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * float(b.max())


def build_lp_matrix(f):
    """Build A_ub for LP: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        ks = j
        ke = min(2 * n - 2, j + n - 1)
        rows = np.arange(ks, ke + 1)
        A[rows, j] = f[rows - j]
    return A


def solve_lp_spread(f, M, timeout=20.0):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    n = len(f)
    A_ub = build_lp_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    remaining = max(0.5, timeout - (time.monotonic() - t0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': remaining, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def subgradient_at_peak(a):

    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k_peak - idx]

def multiplicative_update(a, grad, eta, clip_val=2.5):
    """Mirror descent step on the simplex."""
    n = len(a)
    grad_centered = grad - np.dot(a, grad)

    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-15:

    return a_new / s

def interpolate_seq(f, new_n):
    """Interpolate to new length with spectral awareness."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Use cubic-like interpolation via FFT
    fft_size = 1 << (4 * max(old_n, new_n)).bit_length()
    F = rfft(f_norm, fft_size)
    # Zero-pad or truncate in frequency domain
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(x_new, x_old, f_norm)
    # Also try FFT-based
    F_padded = rfft(f_norm, new_n * 4)
    f_fft = irfft(F_padded, new_n * 4)[:new_n]
    # Combine
    f_interp = 0.5 * f_interp + 0.5 * np.maximum(f_fft, 0)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if rem > 20:
            t_alloc = min(rem * 0.4, 180.0)
            print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent(best_seq, t_alloc, eta_start=0.03, eta_end=0.001,
                                     max_iters=20000, label="MD1")
            best_seq, best_score = check_update(a_md, "MD1", best_seq, best_score)

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.4, 120.0)
            print(f"  budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_update(a_ps, "PS1", best_seq, best_score)

    # === Phase 4: Second LP-Hybrid Round ===
    print(f"\n=== Phase 4: LP-Hybrid Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  budget={t_alloc:.0f}s")
            a_lp2, _ = lp_hybrid_loop(best_seq, t_alloc, verbose=True, label="LP2")
            best_seq, best_score = check_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  budget={t_alloc:.0f}s")
            a_cd, _ = coord_desc_polish(best_seq, t_alloc)
            best_seq, best_score = check_update(a_cd, "CD1", best_seq, best_score)

    # === Phase 6: Alternative Lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_candidates = set()
        for factor in [0.75, 1.25, 1.5, 2.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 3000 and alt_n != n_best:
                alt_candidates.add(alt_n)
        # Also try specific good lengths
        for fixed_n in [512, 768, 1024, 1536]:
            if 256 <= fixed_n <= 3000 and fixed_n != n_best:
                alt_candidates.add(fixed_n)

        for alt_n in sorted(alt_candidates)[:3]:
            rem = deadline - time.monotonic() - 40
            if rem < 40:
                break
            t_alloc = min(rem * 0.35, 60.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, _ = lp_hybrid_loop(init_alt, t_alloc, verbose=False, label=f"alt{alt_n}")
            best_seq, best_score = check_update(a_alt, f"alt_{alt_n}", best_seq, best_score)
            if len(best_seq) == alt_n:
                # Quick MD on the new length
                rem = deadline - time.monotonic() - 30
                if rem > 10:
                    a_md_alt, _ = mirror_descent(best_seq, min(rem * 0.5, 30.0),
                                                 eta_start=0.02, eta_end=0.001)
                    best_seq, best_score = check_update(a_md_alt, f"md_alt_{alt_n}",
                                                        best_seq, best_score)

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Short MD
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_fin_md, _ = mirror_descent(best_seq, min(rem * 0.35, 30.0),
                                         eta_start=0.01, eta_end=0.0005, max_iters=5000)
            best_seq, best_score = check_update(a_fin_md, "MD_fin", best_seq, best_score)

        # Short peak shave
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_fin_ps, _ = peak_shave(best_seq, min(rem * 0.5, 15.0))
            best_seq, best_score = check_update(a_fin_ps, "PS_fin", best_seq, best_score)

        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_fin_cd, _ = coord_desc_polish(best_seq, min(rem * 0.6, 10.0))
            best_seq, best_score = check_update(a_fin_cd, "CD_fin", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
