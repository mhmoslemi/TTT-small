"""
Minimax LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key improvement over previous approach: Instead of maximizing sum(g) subject to
(f*g) <= M, we directly solve a minimax LP: minimize T subject to
b[k] + 2*(f*d)[k] <= T, d >= -f, sum(d) = 0. This properly accounts for
current peak values and allows both increases and decreases in f.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minimax_lp(f, use_all_constraints=True):
    """
    Solve: minimize T
    subject to: b[k] + 2*(f*d)[k] <= T for all k
                d >= -f
                sum(d) = 0
    
    Returns (d, T_predicted) or (None, None) on failure.
    """
    n = len(f)
    b = fast_convolve(f)
    size = 2 * n - 1
    
    # Build constraint matrix A (size x n) where row k is 2*f reversed/shifted
    # (f*d)[k] = sum_i f[i] * d[k-i]
    # So the k-th row of A has A[k, i] = 2*f[k-i] if 0 <= k-i < n, else 0
    
    # For efficiency, build as sparse or use the structure
    # Row k: A[k, i] = 2*f[k-i] for i in [max(0, k-n+1), min(n-1, k)]
    
    # Build the matrix
    A = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # A[k, lo:hi+1] = 2*f[lo:hi+1]  ... wait
        # (f*d)[k] = sum_{i=lo}^{hi} f[i] * d[k-i]
        # So the coefficient of d[j] in row k is 2*f[k-j] where j = k-i, i.e., i = k-j
        # A[k, j] = 2*f[k-j] for j in [max(0, k-n+1), min(n-1, k)]
        for j in range(lo, hi + 1):
            A[k, j] = 2.0 * f[k - j]
    
    d = cp.Variable(n)
    T = cp.Variable()
    
    constraints = [
        A @ d <= T - b,  # b[k] + 2*(f*d)[k] <= T
        d >= -f,
        cp.sum(d) == 0
    ]
    
    prob = cp.Problem(cp.Minimize(T), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, abs_acc=1e-8)
        except Exception:
            return None, None
    
    if d.value is None or T.value is None:
        return None, None
    
    d_val = np.asarray(d.value).flatten()
    T_val = float(T.value)
    
    if np.any(np.isnan(d_val)):
        return None, None
    
    return d_val, T_val

Let me also think about the step size more carefully. After the LP gives us d*, we want to find alpha that minimizes the actual objective of f + alpha*d*.

The actual objective is: 2*n*max((f+alpha*d*)*(f+alpha*d*)) / (sum(f+alpha*d*))^2

Since sum(f+alpha*d*) = sum(f) + alpha*sum(d*) = 1 (because sum(d*) = 0 from the LP constraint), the denominator is always 1.

So we just need to minimize max((f+alpha*d*)*(f+alpha*d*)) over alpha in [0, alpha_max].

This is a 1D minimization problem. We can use golden section search or just evaluate at a few points.

Actually, since the convolution is a polynomial in alpha:
(f+alpha*d)* (f+alpha*d) = f*f + alpha*(f*d + d*f) + alpha^2 * d*d = b + 2*alpha*(f*d) + alpha^2*(d*d)

So each component is a quadratic in alpha. The max of quadratics is a convex function of alpha. We can minimize it using convex optimization (or just golden section).

Let me use simple numerical optimization: evaluate at alpha = 0, alpha_max/4, alpha_max/2, 3*alpha_max/4, alpha_max, and pick the best. Then refine with golden section if needed.

"""
Minimax LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovation: Direct minimax LP formulation that properly linearizes the
auto-convolution around the current point, allowing both increases and
decreases. This is more principled than the previous "maximize sum(g) s.t. f*g<=M"
approach. Combined with adaptive step size, multi-length exploration, and
polishing phases.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A @ d) = convolve(f, d) (linear, length 2n-1)."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for k in range(n):
        A[k, 0:k+1] = f[k::-1]
    for k in range(n, size):
        A[k, k-n+1:n] = f[n-1:k-n+1:-1]
    return A


def solve_minimax_lp(f, verbose=False):
    """
    Solve: minimize T
    subject to: b[k] + 2*(f*d)[k] <= T for all k = 0,...,2n-2
                d >= -f
                sum(d) = 0
    """
    n = len(f)
    b = fast_convolve(f)
    size = 2 * n - 1
    
    A = 2.0 * build_conv_matrix(f)
    
    d = cp.Variable(n)
    T = cp.Variable()
    
    constraints = [
        A @ d <= T - b,
        d >= -f,
        cp.sum(d) == 0
    ]
    
    prob = cp.Problem(cp.Minimize(T), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=200000, abs_acc=1e-9)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL, max_iters=100000)
            except Exception:
                return None, None
    
    if d.value is None or T.value is None:
        return None, None
    
    d_val = np.asarray(d.value).flatten()
    T_val = float(T.value)
    
    if np.any(~np.isfinite(d_val)):
        return None, None
    
    return d_val, T_val


def line_search_max_step(f, d):
    """Find maximum alpha such that f + alpha*d >= 0."""
    neg_mask = d < -1e-15
    if not neg_mask.any():
        return 1.0  # Can go full step
    ratios = f[neg_mask] / (-d[neg_mask])
    return max(0.01, np.min(ratios)) * 0.99  # Safety margin


def evaluate_step(f, d, alpha):
    """Evaluate the objective at f + alpha*d (normalized)."""
    h = f + alpha * d
    h = np.maximum(h, 0.0)
    s = h.sum()
    if s < 1e-15:
        return np.inf
    h = h / s
    b = fast_convolve(h)
    return b.max()


def run_lp_phase(f_init, time_limit, label="lp"):
    """Main LP-based optimization loop with minimax formulation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy() / f_init.sum()  # Normalize to sum=1
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    
    print(f"  [{label}] Starting: n={n}, obj={best_obj:.10f}, score={2*n*best_obj:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        d, T_pred = solve_minimax_lp(f)
        
        if d is None:
            print(f"  [{label}] LP failure at iter {it}")
            break
        
        # Check if the LP predicts improvement
        if T_pred >= best_obj - 1e-10:
            stuck += 1
            if stuck > 5:
                break
            continue
        
        # Find valid step size
        alpha_max = line_search_max_step(f, d)
        
        # Try several step sizes
        best_alpha = 0.0
        best_step_obj = best_obj
        
        candidate_alphas = []
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a = frac * alpha_max
            if a > 0 and a <= alpha_max * 1.001:
                candidate_alphas.append(a)
        
        # Always include a very small step
        candidate_alphas.append(0.01 * alpha_max)
        candidate_alphas = sorted(set(candidate_alphas))
        
        for alpha in candidate_alphas:
            obj = evaluate_step(f, d, alpha)
            if obj < best_step_obj - 1e-12:
                best_step_obj = obj
                best_alpha = alpha
        
        if best_alpha == 0.0:
            stuck += 1
            if stuck > 10:
                break
            # Try even smaller step
            alpha = 0.005 * alpha_max
            obj = evaluate_step(f, d, alpha)
            if obj < best_obj:
                best_alpha = alpha
                best_step_obj = obj
            else:
                continue
        
        # Update
        f = f + best_alpha * d
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                break
        
        if it % 20 == 0:
            print(f"  [{label}] iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"alpha={best_alpha:.4f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def subgradient_step(f, lr, v, momentum, temp, n_top):
    """One projected subgradient step with momentum."""
    n = len(f)
    b = fast_convolve(f)
    size = 2 * n - 1
    
    # Softmax weights on peaks
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(size)
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Restrict to top peaks
    if n_top < size:
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:

    # Actually db[k]/df[j] = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n)
    # Wait: b[k] = sum_i f[i]*f[k-i]. db[k]/df[j] = f[k-j] + f[j-k] (when valid)
    # But since the sum includes both i=j and the term where k-i=j (i=k-j), 
    # db[k]/df[j] = f[k-j] + (f[j-k] if j!=k-j else 0)
    # Actually more carefully: b[k] = sum_i f[i]*f[k-i]
    # db[k]/df[j] = f[k-j] + sum_i (if i==j then f[k-i] else 0) + ...
    # Let me just use: db[k]/df[j] = f[k-j] (for k-j in [0,n-1]) + f[j-k] (for j-k in [0,n-1])
    # But when k-j == j-k (i.e., j = k/2), we count it only once.
    # Actually: b[k] = sum_i f[i]*f[k-i]. The derivative wrt f[j]:
    # = f[k-j] (from the i=j term) + f[j-k] (from the k-i=j term, i.e., i=k-j)

    # And the i=j term is f[j]*f[k-j]. Derivative wrt f[j] is f[k-j].

    # db[k]/df[j] = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n and j != k-j)
    # Wait no. Let me use the product rule properly.
    # b[k] = sum_i f[i]*f[k-i]
    # db[k]/df[j] = d/d f[j] [sum_i f[i]*f[k-i]]
    # = f[k-j] (from i=j, derivative of f[j] is 1, so we get f[k-j])
    #   + sum_i f[i] * d/d f[j] f[k-i]
    #   = f[k-j] + f[k-j] (from i=k-j, derivative of f[k-(k-j)]=f[j] is 1, so we get f[k-j])
    #   But only if i=k-j is valid (0 <= k-j < n) and i=k-j != j (i.e., k != 2j)

    # If k != 2j and 0<=k-j<n: total is f[k-j] + f[k-j] = 2*f[k-j]

    # If j appears as the second factor: term is f[k-j]*f[j], derivative = f[k-j]

    # So b[k] contains the term f_j * f_{k-j} (exactly once, as long as it's valid).
    # db[k]/df_j = f_{k-j} (if 0 <= k-j < n)

    # - If j != k-j: 2*f_j*f_{k-j} (from i=j and i=k-j)
    #   Derivative: 2*f_{k-j}
    # - If j = k-j (i.e., k=2j): f_j*f_j = f_j^2 (from i=j only, since i=k-j=j is the same)

    # So in all cases: db[k]/df_j = 2*f_{k-j} (if 0 <= k-j < n), else 0.
    
    g = np.zeros(n)
    for k in range(size):

    return f_new, v

def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-16:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj

"""
Minimax LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovation over previous approach: Uses a direct minimax LP formulation:
  minimize T s.t. b[k] + 2*(f*d)[k] <= T, d >= -f, sum(d) = 0
This properly linearizes the auto-convolution around the current point,
allowing both increases and decreases, and directly targets the minimax
objective. Combined with adaptive 1D line search (exploiting convexity of
phi(alpha) = max_k [quadratic in alpha]), multi-length exploration,
subgradient polish, and coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A @ d)[k] = (f * d)[k] for linear convolution."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for k in range(n):
        A[k, 0:k + 1] = f[k::-1]
    for k in range(n, size):
        A[k, k - n + 1:n] = f[n - 1:k - n + 1:-1]
    return A


def solve_minimax_lp(f, top_k=None):
    """
    Solve: minimize T
    subject to: b[k] + 2*(f*d)[k] <= T for all k (or top_k peaks)
                d >= -f
                sum(d) = 0
    Returns (d, T_predicted) or (None, None) on failure.
    """
    n = len(f)
    b = fast_convolve(f)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        A_full = 2.0 * build_conv_matrix(f)
        A = A_full[top_idx, :]
        b_sel = b[top_idx]
    else:
        A = 2.0 * build_conv_matrix(f)
        b_sel = b

    d = cp.Variable(n)
    T = cp.Variable()

    constraints = [
        A @ d <= T - b_sel,
        d >= -f,
        cp.sum(d) == 0
    ]

    prob = cp.Problem(cp.Minimize(T), constraints)

    solvers_to_try = [
        (cp.SCS, {"max_iters": 500000, "eps": 1e-9}),
        (cp.CLARABEL, {}),
        (cp.ECOS, {"max_iters": 200000, "abs_acc": 1e-9}),
    ]

    for solver, kwargs in solvers_to_try:
        try:
            prob.solve(solver=solver, **kwargs)
            break
        except Exception:
            continue

    if d.value is None or T.value is None:
        return None, None

    d_val = np.asarray(d.value).flatten()
    T_val = float(T.value)

    if np.any(~np.isfinite(d_val)):
        return None, None

    return d_val, T_val


def alpha_max_safe(f, d):
    """Find maximum alpha such that f + alpha*d >= 0."""
    neg_mask = d < -1e-15
    if not neg_mask.any():
        return 1.0
    ratios = f[neg_mask] / (-d[neg_mask])
    return max(0.01, np.min(ratios)) * 0.999


def convex_1d_minimize(f, d, alpha_max, n_coarse=20):
    """
    Minimize phi(alpha) = max_k [b[k] + 2a*(f*d)[k] + a^2*(d*d)[k]]
    over alpha in [0, alpha_max], exploiting convexity.
    Uses coarse grid + golden section refinement.
    """
    b = fast_convolve(f)
    fd = fast_convolve(f * d)  # (f*d)[k] -- wait, this is element-wise product convolved
    # Actually we need (f*d) as convolution: conv(f, d)
    fd_conv = np.zeros(2 * len(f) - 1)
    n = len(f)
    fft_size = 1 << (2 * n - 2).bit_length()
    F = rfft(f, fft_size)
    D = rfft(d, fft_size)
    fd_conv = irfft(F * D, fft_size)[:2 * n - 1]

    dd = fast_convolve(d)

    def phi(alpha):
        return np.max(b + 2.0 * alpha * fd_conv + alpha * alpha * dd)

    # Coarse grid
    alphas = np.linspace(0, alpha_max, n_coarse)
    vals = np.array([phi(a) for a in alphas])
    best_idx = np.argmin(vals)

    # Golden section refinement around best
    lo = alphas[max(0, best_idx - 1)]
    hi = alphas[min(n_coarse - 1, best_idx + 1)]
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = phi(c)
    fd_val = phi(d_pt)

    for _ in range(30):
        if fc < fd_val:
            hi = d_pt
            d_pt = c
            fd_val = fc
            c = hi - gr * (hi - lo)
            fc = phi(c)
        else:
            lo = c
            c = d_pt
            fc = fd_val
            d_pt = lo + gr * (hi - lo)
            fd_val = phi(d_pt)

    alpha_opt = (lo + hi) / 2.0
    return alpha_opt, phi(alpha_opt)


def run_lp_phase(f_init, time_limit, label="lp", use_top_k=None):
    """
    Main LP-based optimization loop with minimax formulation.
    If use_top_k is set, only use top_k constraints (faster but less accurate).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

    print(f"  [{label}] Start: n={n}, obj={best_obj:.10f}, score={2*n*best_obj:.6f}")

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        d, T_pred = solve_minimax_lp(f, top_k=use_top_k)

        if d is None:
            print(f"  [{label}] LP failure at iter {it}")
            break

        # Check predicted improvement
        if T_pred >= best_obj - 1e-10:
            stuck += 1
            if stuck > 8:
                break
            continue

        # 1D convex line search
        am = alpha_max_safe(f, d)
        alpha_opt, obj_opt = convex_1d_minimize(f, d, am)

        # Verify with actual evaluation
        h = f + alpha_opt * d
        h = np.maximum(h, 0.0)
        sh = h.sum()
        if sh < 1e-15:
            stuck += 1
            continue
        h = h / sh
        actual_obj = fast_convolve(h).max()

        if actual_obj < best_obj - 1e-13:

            best_obj = actual_obj
            best_f = f.copy()
            stuck = 0
        else:
            # Try smaller step

                h2 = f + frac * alpha_opt * d
                h2 = np.maximum(h2, 0.0)
                sh2 = h2.sum()
                if sh2 < 1e-15:
                    continue
                h2 = h2 / sh2
                obj2 = fast_convolve(h2).max()
                if obj2 < best_obj - 1e-13:
                    f = h2
                    best_obj = obj2
                    best_f = f.copy()
                    stuck = 0
                    break
            else:
                stuck += 1
                if stuck > 15:
                    break

        if it % 10 == 0:
            print(f"  [{label}] iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"alpha={alpha_opt:.4f} t={elapsed:.0f}s")

    return best_f, best_obj


def compute_subgradient(f, w):
    """Compute subgradient g[j] = sum_k w[k] * 2*f[k-j] (valid j only)."""
    n = len(f)
    size = 2 * n - 1
    g = np.zeros(n)
    for k in range(size):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            g[j] += wk * 2.0 * f[k - j]
    return g


def run_subgradient_phase(f_init, time_limit, lr_init=0.003, n_top=50):
    """Subgradient optimization with momentum and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = 0.0003
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        size = 2 * n - 1

        # Softmax weights
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(size)
            w[b.argmax()] = 1.0
        else:
            w /= wsum

        # Restrict to top peaks
        if n_top < size:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w_new = np.zeros(size)
            w_new[top_idx] = w_sel
            w = w_new

        g = compute_subgradient(f, w)
        v = 0.9 * v - lr * g
        f = project_simplex(f + v)

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.92
                    no_improve = 0

            if it % 500 == 0:
                temp = min(temp * 1.2, 0.5)
                print(f"    sg iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")

            if no_improve > 25:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                lr = max(lr * 1.3, lr_init * 0.5)

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.0, 3.0]

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-16:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True

    return f, best_obj

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 120.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, obj4 = run_subgradient_phase(best_seq, time_limit=t4,
                                         lr_init=0.002,
                                         n_top=max(30, n_b // 8))
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sg]: {score4:.6f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.1, 80.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = coordinate_polish(best_seq, time_limit=t5)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = max(rem - 15, 0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_lp_phase(best_seq, time_limit=t6, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
