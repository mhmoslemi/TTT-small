new_conv[k] = old_conv[k] + delta * a[k-i] + a[i] * delta + delta * a[k-i] * 0 + ...

No wait. The terms involving a[i] in conv[k] are:
- j = i: a[i] * a[k-i] (if 0 <= k-i < n)
- k-j = i, i.e., j = k-i: a[j] * a[i] = a[k-i] * a[i] (if 0 <= k-i < n)

If i != k-i (i.e., i != k/2), both terms exist and the total contribution of a[i] is 2*a[i]*a[k-i].
If i == k-i (i.e., i == k/2), there's only one term: a[i]^2.

After changing a[i] to a[i]+delta:
- If i != k/2: new contribution = 2*(a[i]+delta)*a[k-i] = old + 2*delta*a[k-i]
- If i == k/2: new contribution = (a[i]+delta)^2 = old + 2*a[i]*delta + delta^2

So: new_conv[k] = old_conv[k] + 2*delta*a[k-i] (if 0<=k-i<n and i!=k/2)
                 = old_conv[k] + 2*a[i]*delta + delta^2 (if k=2i)
                 = old_conv[k] (otherwise)

And the new sum = old_sum + delta.

So the score as a function of delta:
score(delta) = 2n * max_k (b[k] + 2*delta*a[k-i] * [0<=k-i<n, k!=2i] + (2*a[i]*delta + delta^2) * [k==2i]) / (S + delta)^2

This is piecewise defined (max of linear/quadratic functions divided by a quadratic). Near the current point, it should be unimodal.

For the line search, I'll use:
1. Compute the top few peaks and their "slopes" with respect to delta
2. Estimate the optimal delta by equalizing the top peaks

def solve_lp_best_response(a_fixed, solver=None):

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()

    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        # Damped update
        alpha = damp * min(1.0, 10.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Early stop if no progress
        if it > 10 and abs(score - best_score) < 1e-10:
            if it > 50:
                break

    return best_a, best_score


def init_fourier(n, rng):
    """Fourier-based initialization."""
    n_harmonics = int(rng.integers(3, min(20, n // 4)))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * (j + 1) * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_structured(n, rng):
    """Structured initialization using periodic patterns."""
    pattern_len = int(rng.integers(2, min(10, n // 2)))
    pattern = rng.random(pattern_len) + 0.1
    a = np.tile(pattern, n // pattern_len + 1)[:n]
    # Add small randomness
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_symmetric(n, rng):
    """Symmetric initialization (a[i] = a[n-1-i])."""
    half = n // 2
    a_half = rng.random(half) + 0.1
    a = np.zeros(n)
    a[:half] = a_half
    a[half:] = a_half[::-1]
    if n % 2 == 0:
        a[half - 1] = a_half[-1]
    a += rng.standard_normal(n) * 0.03
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def refine_at_length(a, n_target, time_budget, solver):
    """Refine sequence at a specific length."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15 or len(a) == 0:
        return np.ones(n_target) / n_target, float('inf')

    if len(a) != n_target:
        old_x = np.linspace(0, 1, len(a))
        new_x = np.linspace(0, 1, n_target)
        a_interp = np.interp(new_x, old_x, a / s)
        a = np.maximum(a_interp, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n_target) / n_target
        else:
            a = a / ss
    else:
        a = a / s

    n = len(a)
    score = 2.0 * n * fast_convolve(a).max()

    # Phase A: Coordinate descent (70% of budget)
    t_cd = time_budget * 0.55
    a, score = coord_descent_vectorized(a, t_cd, top_peaks=5, use_sa=False,
                                        max_sweeps=50000, seed=12345)

    # Phase B: LP refinement (20% of budget)
    t_lp = time_budget * 0.25
    rem = time_budget - (time.monotonic() - (time.monotonic() - t_cd))
    rem = min(rem, t_lp)
    if rem > 10:
        a_lp, score_lp = lp_refine(a, rem, solver=solver, damp=0.2)
        if score_lp < score:
            a, score = a_lp, score_lp
            # Polish after LP
            a, score2 = coord_descent_vectorized(a, min(30, rem * 0.3), top_peaks=5,
                                                  use_sa=False, max_sweeps=10000, seed=999)
            if score2 < score:
                a, score = a, score2

    # Phase C: SA + coord descent (15% of budget)
    t_sa = time_budget * 0.15
    rem = time_budget - t_cd - t_lp
    if rem > 10:
        a_sa, score_sa = coord_descent_vectorized(a, rem, top_peaks=5, use_sa=True,
                                                   sa_temp=0.005, max_sweeps=20000, seed=777)
        if score_sa < score:
            a, score = a_sa, score_sa

    return a, score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Diverse initialization at multiple lengths
    2. Smart coordinate descent with peak-equalizing line search
    3. LP-based refinement
    4. Simulated annealing escape
    5. Final intensive polish
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(2024)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization (120s) ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t1 = time.monotonic()
    t_phase1 = 120.0

    n_values = [384, 512, 640, 768, 896, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []
    for n in n_values:
        if time.monotonic() - t1 > t_phase1 - 30:
            break
        # Multiple inits per length
        inits = []
        if parent is not None:
            old_x = np.linspace(0, 1, len(parent))
            new_x = np.linspace(0, 1, n)
            p_interp = np.interp(new_x, old_x, parent)
            inits.append(np.maximum(p_interp, 0))
        inits.append(init_fourier(n, rng))
        inits.append(init_structured(n, rng))
        inits.append(init_symmetric(n, rng))

        for idx, init in enumerate(inits):
            if time.monotonic() - t1 > t_phase1 - 10:
                break
            a_norm = np.maximum(init, 0)
            s = a_norm.sum()
            if s < 1e-10:
                continue
            a_norm = a_norm / s

            # Quick CD to evaluate
            t_quick = 5.0
            a_opt, s_opt = coord_descent_vectorized(a_norm, t_quick, top_peaks=4,
                                                     use_sa=False, max_sweeps=5000,
                                                     seed=100 + n + idx)
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init score: {candidates[0][0]:.6f}")

    for s_c, a_c, n_c in candidates[:3]:
        if s_c < best_score:
            best_score = s_c
            best_seq = a_c.copy()
            print(f"  New best from init: {s_c:.6f} (n={n_c})")

    # === Phase 2: Intensive refinement at best length (800s) ===
    print(f"\n=== Phase 2: Intensive Refinement (best={best_score:.6f}) ===")
    t2 = time.monotonic()
    t_phase2 = 800.0

    if best_seq is not None:
        n_best = len(best_seq)
        a_work = best_seq.copy()

        # Round 1: Long CD
        t_r1 = min(300, t_phase2 * 0.4)
        print(f"  CD round 1, n={n_best}, budget={t_r1:.0f}s")
        a_r1, s_r1 = coord_descent_vectorized(a_work, t_r1, top_peaks=6,
                                               use_sa=False, max_sweeps=100000, seed=42)
        if s_r1 < best_score:
            best_score = s_r1
            best_seq = a_r1.copy()
            print(f"  After CD1: {s_r1:.6f}")

        # Round 2: LP refinement
        rem = t_phase2 - (time.monotonic() - t2)
        if rem > 60:
            t_r2 = min(200, rem * 0.5)
            print(f"  LP refinement, budget={t_r2:.0f}s")
            a_r2, s_r2 = lp_refine(best_seq, t_r2, solver=solver, damp=0.15)
            if s_r2 < best_score:
                best_score = s_r2
                best_seq = a_r2.copy()
                print(f"  After LP: {s_r2:.6f}")

            # CD after LP
            rem = t_phase2 - (time.monotonic() - t2)
            if rem > 30:
                a_r2b, s_r2b = coord_descent_vectorized(best_seq, min(rem * 0.7, 120),
                                                         top_peaks=6, use_sa=False,
                                                         max_sweeps=50000, seed=88)
                if s_r2b < best_score:
                    best_score = s_r2b
                    best_seq = a_r2b.copy()
                    print(f"  After CD post-LP: {s_r2b:.6f}")

        # Round 3: SA escape
        rem = t_phase2 - (time.monotonic() - t2)
        if rem > 30:
            t_r3 = min(150, rem * 0.7)
            print(f"  SA escape, budget={t_r3:.0f}s")
            a_r3, s_r3 = coord_descent_vectorized(best_seq, t_r3, top_peaks=5,
                                                   use_sa=True, sa_temp=0.008,
                                                   max_sweeps=30000, seed=314)
            if s_r3 < best_score:
                best_score = s_r3
                best_seq = a_r3.copy()
                print(f"  After SA: {s_r3:.6f}")

    # === Phase 3: Try other lengths (300s) ===
    print(f"\n=== Phase 3: Other Lengths (best={best_score:.6f}) ===")
    t3 = time.monotonic()
    t_phase3 = 300.0

    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 0.875, 1.125, 1.25, 1.5]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2048 and alt_n != n_best:
                alt_lengths.append(alt_n)
        # Also try round numbers
        for m in [512, 768, 1024, 1536]:
            if m not in alt_lengths and m != n_best and 256 <= m <= 2048:
                alt_lengths.append(m)

        for alt_n in alt_lengths[:3]:
            rem = t_phase3 - (time.monotonic() - t3)
            if rem < 30:
                break
            t_alt = min(rem * 0.6, 100.0)
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")

            a_alt, s_alt = refine_at_length(best_seq, alt_n, t_alt, solver)
            if s_alt < best_score:
                best_score = s_alt
                best_seq = a_alt.copy()
                print(f"  New best at n={alt_n}: {s_alt:.6f}")

    # === Phase 4: Final intensive polish (remaining time - 30s) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if best_seq is not None and rem > 30:
        n_final = len(best_seq)

        # Very fine CD
        t_fin1 = min(rem * 0.5, 120.0)
        print(f"  Fine CD, budget={t_fin1:.0f}s")
        a_fin, s_fin = coord_descent_vectorized(best_seq, t_fin1, top_peaks=8,
                                                 use_sa=False, max_sweeps=200000, seed=271)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
            print(f"  After fine CD: {s_fin:.6f}")

        # Short LP
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_fin2 = min(rem * 0.5, 60.0)
            a_fin2, s_fin2 = lp_refine(best_seq, t_fin2, solver=solver, damp=0.1)
            if s_fin2 < best_score:
                best_score = s_fin2
                best_seq = a_fin2.copy()
                print(f"  After final LP: {s_fin2:.6f}")

                # Tiny CD
                a_fin3, s_fin3 = coord_descent_vectorized(best_seq, min(20, rem * 0.2),
                                                           top_peaks=8, use_sa=False,
                                                           max_sweeps=20000, seed=999)
                if s_fin3 < best_score:
                    best_score = s_fin3
                    best_seq = a_fin3.copy()

        # Tiny SA
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_fin4, s_fin4 = coord_descent_vectorized(best_seq, rem * 0.6, top_peaks=8,
                                                       use_sa=True, sa_temp=0.003,
                                                       max_sweeps=10000, seed=555)
            if s_fin4 < best_score:
                best_score = s_fin4
                best_seq = a_fin4.copy()
                print(f"  After final SA: {s_fin4:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
