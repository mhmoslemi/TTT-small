import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CBC, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CBC, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play_nesterov(n, time_limit, initial=None, solver=None, verbose=False):
    """Fictitious play with Nesterov-style averaging for better convergence."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    # Store history for averaging
    history_a = [a.copy()]
    history_b = [a.copy()]
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    x = a.copy()  # current "position" (like momentum)
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Compute best responses
        b_new, M_b = solve_lp_best_response(x, solver)
        if b_new is None:
            break
        a_new, M_a = solve_lp_best_response(b_new, solver)
        if a_new is None:
            break
        
        # Nesterov acceleration: x_{k+1} = (t_k - 1)/t_{k+1} * x_k + 1/t_{k+1} * a_new
        t_k = (1 + np.sqrt(1 + 4 * (iters - 1) ** 2)) / 2
        t_k1 = (1 + np.sqrt(1 + 4 * iters ** 2)) / 2
        beta = (t_k - 1) / t_k1
        x = beta * x + (1 - beta) * a_new
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            x = np.ones(n) / n
        else:
            x = x / s
        
        history_a.append(a_new.copy())
        history_b.append(b_new.copy())
        
        # Compute running average (uniform average over all iterates)
        if len(history_a) % 10 == 0:
            avg = np.mean(history_a[-100:], axis=0)  # average of last 100
            avg = np.maximum(avg, 0)
            s = avg.sum()
            if s > 1e-15:
                avg = avg / s
                score = 2.0 * n * fast_convolve(avg).max()
                if score < best_score:
                    best_score = score
                    best_a = avg.copy()
        
        # Also check current iterates
        for cand in [x, a_new, b_new, 0.5 * (a_new + b_new)]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        # Clean up history to prevent memory issues
        if len(history_a) > 500:
            history_a = history_a[-200:]
            history_b = history_b[-200:]
        
        if verbose and iters % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP-Nesterov n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def coord_descent_ternary(a0, time_limit, max_sweeps=200):
    """Coordinate descent with ternary search for each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    
    def score_of_a(arr):
        s = arr.sum()
        if s < 1e-15:
            return float('inf')
        return 2.0 * n * fast_convolve(arr / s).max()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            s_others = 1.0 - old_val  # since a is normalized
            
            if old_val < 1e-14 and s_others > 0.999:
                # Try adding a small value
                for t_try in [0.001, 0.005, 0.01, 0.02]:
                    a_trial = a.copy()
                    a_trial[i] = t_try
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                continue
            
            # Ternary search for optimal value of a[i]
            # a[i] should be in [0, 1] (normalized)
            lo = max(0, old_val * 0.3)
            hi = min(1.0, old_val * 3.0 + 0.05)
            
            for _ in range(12):  # 12 iterations of ternary search
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                
                a[i] = m1
                a_norm1 = a.copy()
                s1 = a_norm1.sum()
                sc1 = 2.0 * n * fast_convolve(a_norm1 / s1).max()
                
                a[i] = m2
                a_norm2 = a.copy()
                s2 = a_norm2.sum()
                sc2 = 2.0 * n * fast_convolve(a_norm2 / s2).max()
                
                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1
            
            a[i] = (lo + hi) / 2
            a_final = a.copy()
            s_final = a_final.sum()
            a_final = a_final / s_final
            sc_final = 2.0 * n * fast_convolve(a_final).max()
            
            if sc_final < best_score - 1e-13:
                best_score = sc_final
                best_a = a_final.copy()
                a = a_final.copy()
                improved = True
            else:
                a[i] = old_val
        
        if not improved:
            break
    
    return best_a, best_score

def compute_subgrad_softmax(a, top_k=30, temperature=0.01):
    """Compute subgradient with softmax weighting over top peaks."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    
    # Softmax weighting
    max_pv = pv.max()
    w = np.exp((pv - max_pv) / (temperature * max_pv + 1e-10))
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * a[k - j]
    return sg

def accelerated_subgrad(a0, time_limit, top_k=30, lr_init=0.01, temp_schedule=True):
    """Accelerated projected subgradient with softmax peak weighting."""

        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        
        y_new = y - lr * v
        y_projected = simplex_project(y_new)
        
        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()

        if no_improve > 100:
            lr *= 0.9
            no_improve = 0

            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.02
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 1.5)
            temp = max(0.005, temp * 2)

def peak_equalize(a0, time_limit, n_peaks=5):

        top_k = min(n_peaks, len(b))

                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()

                    a_trial = a_trial / s_t
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:

        if not improved:

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():

            lo = max(0.0, center - span)
            hi = min(1.0, center + span)
            
            if hi - lo < 1e-8:
                continue
            
            best_t = old_val
            best_sc = best_score
            
            # Coarse ternary search
            for _ in range(n_iters_coarse):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                a[i] = m1
                sc1 = score_trial(a)
                a[i] = m2
                sc2 = score_trial(a)
                
                if sc1 < sc2:
                    hi = m2
                    if sc1 < best_sc:
                        best_sc = sc1
                        best_t = m1
                else:
                    lo = m1
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_t = m2
            
            # Fine ternary search around the best found
            if best_sc < best_score - 1e-14:
                fine_span = (hi - lo)
                lo_f = max(0.0, best_t - fine_span)
                hi_f = min(1.0, best_t + fine_span)
                
                for _ in range(n_iters_fine):
                    m1 = lo_f + (hi_f - lo_f) / 3.0
                    m2 = hi_f - (hi_f - lo_f) / 3.0
                    
                    a[i] = m1
                    sc1 = score_trial(a)
                    a[i] = m2
                    sc2 = score_trial(a)
                    
                    if sc1 < sc2:
                        hi_f = m2
                        if sc1 < best_sc:
                            best_sc = sc1
                            best_t = m1
                    else:
                        lo_f = m1
                        if sc2 < best_sc:
                            best_sc = sc2
                            best_t = m2
                
                a[i] = best_t
                if best_sc < best_score - 1e-13:
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = old_val
        
        if not improved and sweep > 10:
            break
    
    return best_a, best_score

def compute_subgrad_softmax(a, top_k=30, temperature=0.01):
    """Compute subgradient with softmax weighting over top peaks."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    
    max_pv = pv.max()
    if max_pv < 1e-15:
        return np.ones(n) / n
    
    w = np.exp((pv - max_pv) / (temperature * max_pv + 1e-10))
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * a[k - j]
    return sg

def accelerated_subgrad(a0, time_limit, top_k=30, lr_init=0.01, verbose=False):
    """Accelerated projected subgradient with adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temp = 0.015

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * max(1, it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k
        
        if it % 40 == 0:
            temp *= 0.92
            temp = max(temp, 0.003)
        
        sg = compute_subgrad_softmax(y, top_k=top_k, temperature=temp)
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        
        y_new = y - lr * v
        y_projected = simplex_project(y_new)
        
        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = y_projected
        
        if no_improve > 80:
            lr *= 0.92
            no_improve = 0
        
        if no_improve > 350:
            noise = np.random.default_rng(it * 17 + 3).standard_normal(n) * 0.015
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.4, lr * 1.8)
            temp = max(0.01, temp * 2.0)
            no_improve = 0
        
        lr *= 0.9996
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score

def peak_equalize(a0, time_limit, n_peaks=5):
    """Iteratively reduce the top peaks by adjusting contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        top_k = min(n_peaks, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peaks = peaks[np.argsort(b[peaks])[::-1]]
        
        # Check if peaks are already well-equalized
        peak_vals = b[peaks]
        if peak_vals.max() - peak_vals.min() < 1e-8 * peak_vals.max():
            pass  # Still try to reduce the global max
        
        improved = False
        
        # For the highest peak, try reducing top contributors
        k = int(peaks[0])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k - idx]
        
        if contribs.max() > 1e-15:
            top_c = np.argsort(contribs)[::-1][:4]
            for ci in top_c:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                    a[i] *= factor
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t < 1e-15:
                        continue
                    a = a / s_t
                    sc = 2.0 * n * fast_convolve(a).max()
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        # Undo
                        a[i] /= 1.0  # Can't easily undo, so restore from best_a
                        a = best_a.copy()
                        break

        if not improved:
            # Try random single-coordinate perturbations
            for _ in range(3):
                i = rng.integers(0, n)
                factor = rng.uniform(0.85, 1.15)
                a_trial = best_a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            a = best_a.copy()
            if step > 500:
                break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(4, 25))
    amplitudes = rng.random(n_harmonics) * 0.25
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_blocks(n, rng):
    """Initialize with random block amplitudes."""
    n_blocks = int(rng.integers(5, 35))
    block_size = max(2, n // n_blocks)
    amplitudes = rng.random(n_blocks) + 0.05
    a = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        a[start:end] = amplitudes[b]
    ksize = min(5, block_size)
    if ksize > 1:
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
