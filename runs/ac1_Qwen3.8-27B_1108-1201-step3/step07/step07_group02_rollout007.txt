"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return np.real(irfft(A * A, fft_size)[:size])


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Score for a normalized (sum=1, non-neg) sequence: 2*n*max_conv."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbfgs_soft_max(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=10000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # grad_a[i] = sum_k w[k] * a[k-i] * 2 * T_val / T_val = 2 * sum_k w[k] * a[k-i]
            # (the T_val cancels because d/dT log_sum_exp has the T in numerator)
            # Actually: obj = (2n/T) * log_sum_exp(T*b)
            # d obj / d a_i = (2n/T) * sum_k w_k * b_k' / b_k ... 
            # Let me recompute.
            # log_sum_exp = log(sum_k exp(T*b_k))
            # d log_sum_exp / d b_k = T * w_k
            # d b_k / d a_i = a[k-i] (the contribution of a_i to conv at position k)
            # d obj / d a_i = (2n/T) * sum_k T*w_k * a[k-i] = 2n * sum_k w_k * a[k-i]
            # In z-space: grad_z = a * (grad_a - sum(grad_a * a))
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            grad_a *= 2.0 * n
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    # Geometric temperature schedule
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z_cur = z.copy()
    iters_per_stage = maxiter // n_stages

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        # Scale iterations by remaining time
        eff_iters = max(100, int(iters_per_stage * remaining / time_limit * 1.5))
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z_cur,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': eff_iters, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': eff_iters * 100, 'disp': False}
            )
            z_cur = res.x
            a_check = softmax_from_z(z_cur)
            print(f"    L-BFGS stage {si+1}/{n_stages} T={stage_T:.1f}: score={score_normalized(a_check):.6f}")
        except Exception as e:
            print(f"    L-BFGS stage {si+1} error: {e}")

    a_out = softmax_from_z(z_cur)
    return a_out


def hankel_fixed_point(a0, time_limit, n_iters=1000):
    """
    Fixed-point iteration exploiting KKT conditions.
    At optimum: sum_k w_k * a[k-j] = c for all j, where w = softmax(T*conv(a,a)).
    This gives a Hankel system M@a = c*1 which we solve directly.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Start with a high temperature for the soft-max (smooth weights)
    T_schedule_start = 50.0
    T_schedule_end = 10.0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Animate temperature down
        frac = it / max(1, n_iters - 1)
        T_val = T_schedule_start * (T_schedule_end / T_schedule_start) ** frac
        
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Build Hankel matrix M where M[j,m] = w[j+m]
        # M is n x n
        M = np.zeros((n, n))
        for j in range(n):
            M[j, :] = w[j:j+n]
        
        # Solve M @ a_new = c * 1, sum(a_new) = 1
        # This is overdetermined (n equations + 1 constraint for n unknowns)
        # Use least squares with constraint: minimize ||M@a - c||^2 + penalty*(sum(a)-1)^2
        # Or: solve the augmented system
        # 
        # Better approach: solve M @ a = c * 1 in least squares, then project to simplex
        # Even better: solve (M^T M + lambda*I) a = M^T * (c*1) + mu * 1 for the constraint
        
        # Actually, let's just solve M @ a = c*1 as a least squares problem
        # and then normalize
        a_new, residuals, rank, sv = np.linalg.lstsq(M, np.ones(n), rcond=None)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        # Blend with current a
        alpha = 0.3 + 0.4 * frac  # increase blending as we go
        a_blend = (1 - alpha) * a + alpha * a_new
        a_blend = normalize_nonneg(a_blend)
        
        sc = score_normalized(a_blend)
        if sc < best_score:
            best_score = sc
            best_a = a_blend.copy()
        
        a = a_blend
        
        if (it + 1) % 200 == 0:
            print(f"    Hankel FP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def lp_peak_shave(a0, time_limit, n_steps=500, step_size=0.05):
    """
    LP-based peak shaving: at each step, solve an LP to find the best
    small perturbation that reduces the near-peak values of the convolution.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    solver = get_lp_solver()
    
    T = build_conv_matrix(a)  # Will be updated as a changes
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Identify near-peak positions (within 10% of max)
        threshold = max_b * 0.90
        near_peaks = np.where(b >= threshold)[0]
        if len(near_peaks) == 0:
            near_peaks = np.array([np.argmax(b)])
        # Limit to top 20 for LP efficiency
        if len(near_peaks) > 20:
            top_idx = np.argsort(b[near_peaks])[::-1][:20]
            near_peaks = near_peaks[top_idx]
        
        # Build T for current a
        T = build_conv_matrix(a)
        
        # LP: minimize M
        # s.t. (T @ delta)[k] <= M for k in near_peaks
        #      sum(delta) = 0
        #      -a[i] * 0.5 <= delta[i] <= step_size  (limit change)
        delta_var = cp.Variable(n)
        M_var = cp.Variable()
        
        constraints = [cp.sum(delta_var) == 0]
        for k in near_peaks:
            constraints.append(T[k, :] @ delta_var <= M_var)
        constraints.append(delta_var >= -a * 0.3)
        constraints.append(delta_var <= step_size)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
                try:
                    prob.solve(solver=alt)
                    break
                except Exception:
                    continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if step > 50:
                break
            continue
        
        delta_val = np.asarray(delta_var.value, dtype=np.float64)
        if delta_val is None or not np.all(np.isfinite(delta_val)):
            break
        
        a_trial = a + delta_val
        a_trial = normalize_nonneg(a_trial)
        
        sc = score_normalized(a_trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            # No improvement; reduce step size
            step_size *= 0.7
            if step_size < 1e-5:
                break
    
    return best_a, best_score


def nesterov_lp_alternate(a0, time_limit, solver=None, n_iters=2000):
    """Nesterov-accelerated LP alternating optimization."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()  # virtual iterate
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        alpha = 2.0 / (it + 2)
        
        # Solve LP: given y, find best a_new
        a_new, M_a = solve_lp_best_response(y, solver)
        if a_new is None:
            break
        
        # Nesterov update
        y = (1 - alpha) * y + alpha * a_new
        y = normalize_nonneg(y)
        
        # Also evaluate y
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        # Get best response to a_new to get another iterate
        b_new, M_b = solve_lp_best_response(a_new, solver)
        if b_new is not None:
            y2 = (1 - alpha) * y + alpha * b_new
            y2 = normalize_nonneg(y2)
            sc_y2 = score_normalized(y2)
            if sc_y2 < best_score:
                best_score = sc_y2
                best_a = y2.copy()
            y = y2
        
        if (it + 1) % 100 == 0:
            print(f"    NesterovLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    f_padded = np.concatenate([f_norm, [0]])
    new_x = np.linspace(0, 1, new_n + 1)
    f_interp = np.interp(new_x, old_x, f_padded, right=0)
    f_interp = f_interp[:new_n]
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with soft-max smoothed objective and multi-stage geometric temperature annealing
2. Nesterov-accelerated LP alternating optimization
3. LP-based peak shaving (solves small LPs for optimal local perturbations)
4. Multi-length exploration with interpolation
New vs previous: more L-BFGS stages, LP-based peak shaving (not heuristic),
better time allocation, and structured initialization attempts.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return np.real(irfft(A * A, fft_size)[:size])


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_rows(a, row_indices):
    """Build only specific rows of the conv matrix (for LP constraints)."""
    n = len(a)
    rows = len(row_indices)
    T = np.zeros((rows, n))
    for r, k in enumerate(row_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[r, idx] = a[k - idx]
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Score for a normalized (sum=1, non-neg) sequence: 2*n*max_conv."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * float(b.max())


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbfgs_soft_max(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=10000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()

    n_stages = max(3, int(maxiter / 1000))
    temps = np.geomspace(T_start, T_end, n_stages)
    z_cur = z.copy()
    iters_per_stage = max(100, maxiter // n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        eff_iters = max(100, int(iters_per_stage * (remaining / time_limit) * 1.3))

        def func_stage(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = np.max(Tb)
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(np.sum(exp_Tb))
            return (2.0 * n / stage_T) * log_sum_exp

        def jac_stage(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = np.max(Tb)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / np.sum(exp_Tb)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            grad_a *= 2.0 * n
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func_stage,
                x0=z_cur,
                jac=jac_stage,
                method='L-BFGS-B',
                options={'maxiter': eff_iters, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': eff_iters * 80, 'disp': False}
            )
            z_cur = res.x
            a_check = softmax_from_z(z_cur)
            sc = score_normalized(a_check)
            if (si + 1) % 2 == 0 or si == n_stages - 1:
                print(f"    L-BFGS stage {si+1}/{n_stages} T={stage_T:.2f}: score={sc:.6f}")
        except Exception as e:
            print(f"    L-BFGS stage {si+1} error: {e}")

    a_out = softmax_from_z(z_cur)
    return a_out


def lp_peak_shave(a0, time_limit, n_steps=500, step_frac=0.05):
    """
    LP-based peak shaving: solve an LP at each step to find the best
    small perturbation that reduces near-peak convolution values.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    solver = get_lp_solver()
    step_size = step_frac

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = float(b.max())

        # Identify near-peak positions (within 15% of max)
        threshold = max_b * 0.85
        near_peaks = np.where(b >= threshold)[0]
        if len(near_peaks) == 0:
            near_peaks = np.array([np.argmax(b)])
        # Limit to top 15 for LP efficiency
        if len(near_peaks) > 15:
            top_idx = np.argsort(b[near_peaks])[::-1][:15]
            near_peaks = near_peaks[top_idx]

        # Build constraint matrix (only near-peak rows)
        T_rows = build_conv_matrix_rows(a, near_peaks)

        # LP: minimize M
        # s.t. T_rows @ delta <= M * ones
        #      sum(delta) = 0
        #      -0.3*a[i] <= delta[i] <= step_size
        delta_var = cp.Variable(n)
        M_var = cp.Variable()

        constraints = [
            T_rows @ delta_var <= M_var * np.ones(len(near_peaks)),
            cp.sum(delta_var) == 0,
            delta_var >= -0.3 * a,
            delta_var <= step_size
        ]

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
                try:
                    prob.solve(solver=alt)
                    break
                except Exception:
                    continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if step > 30:
                break
            continue

        delta_val = np.asarray(delta_var.value, dtype=np.float64)
        if delta_val is None or not np.all(np.isfinite(delta_val)):
            break

        a_trial = a + delta_val
        a_trial = normalize_nonneg(a_trial)

        sc = score_normalized(a_trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            step_size *= 0.7
            if step_size < 1e-5:
                break

        if (step + 1) % 50 == 0:
            print(f"    LP PeakShave step={step+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f} step_sz={step_size:.6f}")

    return best_a, best_score


def nesterov_lp_alternate(a0, time_limit, solver=None, n_iters=2000):
    """Nesterov-accelerated LP alternating optimization."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        alpha = 2.0 / (it + 2)

        a_new, _ = solve_lp_best_response(y, solver)
        if a_new is None:
            break

        y = (1 - alpha) * y + alpha * a_new
        y = normalize_nonneg(y)

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        # Alternating: get response to a_new
        b_new, _ = solve_lp_best_response(a_new, solver)
        if b_new is not None:
            y = (1 - alpha) * y + alpha * b_new
            y = normalize_nonneg(y)
            sc2 = score_normalized(y)
            if sc2 < best_score:
                best_score = sc2
                best_a = y.copy()

        if (it + 1) % 100 == 0:
            print(f"    NesterovLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1.0, old_n)
    new_x = np.linspace(0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_structured_init(n):
    """Create a structured initialization that's better than uniform."""
    # Slight edge emphasis: a_i proportional to 1 + eps*cos(2*pi*i/n)
    # This creates a mild bimodal structure
    i = np.arange(n)
    a = 1.0 + 0.15 * np.cos(2.0 * np.pi * i / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()
