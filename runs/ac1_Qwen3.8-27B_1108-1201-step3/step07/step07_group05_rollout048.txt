```python
def lp_fictitious_play(f, M, solver='CLARABEL'):
    """Solve the LP to find the best direction g."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)

    # (f * g)(k) = sum_i f[i] * g[k-i] where g is 0 outside [0, n-1]
    constraints = []

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play (novel approach using cvxpy)
2. Log-space interpolation updates (mirror descent-like)
3. Exact 1D line search for step sizes
4. L-BFGS with multiple smoothing temperatures
5. Coordinate descent with golden-section search
6. Length exploration near the parent length

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    return C

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_C(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.CLARABEL)
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

def lp_fictitious_play(f_init, time_limit, use_average=True, verbose=True):
    """LP-based fictitious play."""
    f = f_init.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    
    if use_average:
        f_avg = f.copy()
    else:
        f_avg = None
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        g = solve_lp(f, M)
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Log-space interpolation (geometric mean)
        t_step = 2.0 / (k + 2)
        # a_new = f^(1-t) * g^t (elementwise, then renormalize)
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_f = np.log(f_safe)
        log_g = np.log(g_safe)
        log_new = (1 - t_step) * log_f + t_step * log_g
        a_new = np.exp(log_new - log_new.max())
        a_new = a_new / a_new.sum()
        
        if use_average:
            f_avg = ((k * f_avg) + a_new) / (k + 1)
            current = f_avg
        else:
            f_avg = a_new
            current = a_new
        
        score = compute_score(current)
        if score < best_score - 1e-12:
            best_score = score
            best_f = current.copy()
        
        f = a_new  # for next LP iteration
        
        if verbose and k % 100 == 0:
            print(f"  LP iter={k}, M={M:.6e}, score={score:.8f}, best={best_score:.8f}")
    
    return best_f, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with geometric mean (log-space) updates (main novelty)
2. Greedy LP approximation for fast direction finding
3. L-BFGS in log-space with multiple smoothing temperatures
4. Coordinate descent with exact golden-section line search
5. Length exploration near the parent
6. Peak-focused random search
The key new idea is the LP fictitious play: at each step, solve an LP to find
the "widest" non-negative g that fits under the current max-convolution ceiling,
then update via geometric mean interpolation in log-space.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_cvxpy(f, M, solver='CLARABEL'):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0 using cvxpy."""
    import cvxpy as cp
    n = len(f)
    C = build_C(f)
    g_var = cp.Variable(n, nonneg=True)
    constraints = [C @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0)


def solve_lp_scipy(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0 using scipy linprog."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def greedy_lp(f, M):
    """Greedy approximation to the LP: build g one coordinate at a time."""
    n = len(f)
    g = np.zeros(n)
    conv_f = f.copy()  # (f * g)(k) accumulated
    conv_vals = np.zeros(2 * n - 1)
    
    order = np.arange(n)
    for j in order:
        # Maximum g[j] such that for all k: conv_vals[k] + f[k-j]*g[j] <= M
        # k ranges from j to j+n-1
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]  # f[k-j] for k in [j, j+n-1] is f[0], f[1], ..., f[n-1]
        # Actually f[k-j] where k goes from j to j+n-1, so k-j goes from 0 to n-1
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space: a = f^(1-t) * g^t, normalized."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, use_greedy=False, verbose=True):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    
    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        if use_greedy:
            g = greedy_lp(f, M)
        else:
            g = solve_lp_cvxpy(f, M)
            if g is None:
                g = solve_lp_scipy(f, M)
            if g is None:
                g = greedy_lp(f, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Adaptive step size: larger early, smaller later
        t_step = 2.0 / (k + 2)
        if k > 500:
            t_step = max(0.01, 2.0 / (k + 2))
        
        a_new = geometric_mean_update(f, g, t_step)
        
        # Update average
        f_avg = ((k * f_avg) + a_new) / (k + 1)
        
        score_avg = compute_score(f_avg)
        score_cur = compute_score(a_new)
        
        if score_avg < best_score - 1e-12:
            best_score = score_avg
            best_f = f_avg.copy()
        if score_cur < best_score - 1e-12:

        f = a_new

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective_and_grad(z, T, n):
    """Compute smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    
    # Objective: (2n/T) * sum(w_k) * exp(T * (conv@w - max_shifted))... 
    # Actually: soft_max of conv = T * log(sum(exp(conv/T)))
    # = T * (max_shifted + log(sum(exp(shifted - max_shifted))))
    log_sum_exp = max_shifted + np.log(w.sum())
    obj = (2.0 * n) * T * log_sum_exp
    
    # Gradient w.r.t. a: 2n * (w * a) cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    
    # Project to log-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, (2.0 * n) * grad_z


def lbgfs_logspace(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with smooth max approximation."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_tmp = softmax_from_z(z)
        sc = compute_score(a_tmp)
        if sc < best_score:
            nonlocal_best = sc
        obj, _ = smoothed_objective_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = smoothed_objective_and_grad(z, T, n)
        return g

    try:

                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)

                else:

                    x1 = x2
                    f1 = f2

        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Identify contributing pairs for this peak
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[int(pk) - idx]
            
            # Try reducing the top contributing element
            top_contrib_idx = np.argsort(contribs)[::-1][:3]
            
            for ci in top_contrib_idx:
                i_idx = idx[ci]
                if a[i_idx] < 1e-14:
                    continue
                
                # Try different amounts of reduction
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[i_idx] * frac
                    a_trial = a.copy()
                    a_trial[i_idx] -= delta
                    # Redistribute to neighbors
                    mask = np.ones(n, dtype=bool)
                    mask[i_idx] = False
                    others = a_trial[mask]
                    os_sum = others.sum()
                    if os_sum < 1e-15:
                        continue
                    a_trial[mask] = others + delta * others / os_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved:
                # Try adding to a position that doesn't contribute to this peak
                # (i.e., a position far from the peak's contributing range)
                noncontrib = np.setdiff1d(np.arange(n), idx)
                if len(noncontrib) > 0:
                    j_idx = noncontrib[rng.integers(0, len(noncontrib))]
                    for frac in [0.01, 0.02, 0.05]:
                        delta = a[j_idx] * frac
                        a_trial = a.copy()
                        a_trial[j_idx] -= delta
                        mask = np.ones(n, dtype=bool)
                        mask[j_idx] = False
                        others = a_trial[mask]
                        os_sum = others.sum()
                        if os_sum < 1e-15:
                            continue
                        a_trial[mask] = others + delta * others / os_sum
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
        
        if not improved and iters % 50 == 0:
            # Random restart perturbation
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 30 == 0:
            print(f"  PF iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    - LP fictitious play (geometric mean updates) - main novel approach
    - LP fictitious play (linear updates)
    - L-BFGS in log-space with multiple temperatures
    - Coordinate descent with golden-section search
    - Peak-focused search
    - Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t_arr = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = parent + 0.05 * np.cos(t_arr)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: LP Fictitious Play (Geometric Mean) ===
    print(f"\n=== Phase 1: LP Fictitious Play [geometric] (best={best_score:.8f}) ===")
    t1_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    rem = deadline - time.monotonic()
    if rem > t1_budget + 100:
        t_alloc = t1_budget
    else:
        t_alloc = max(30, rem * 0.2)
    print(f"  Budget: {t_alloc:.0f}s")
    a_lp1, sc_lp1 = lp_fictitious_play(best_seq, t_alloc, use_greedy=False, verbose=True)
    best_seq, best_score = check_and_update(a_lp1, "lp_geo", best_seq, best_score)
    
    # === Phase 2: LP Fictitious Play (Linear) ===
    print(f"\n=== Phase 2: LP Fictitious Play [linear] (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 50:
        t_alloc = min(rem * 0.2, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play_linear(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_lp2, "lp_lin", best_seq, best_score)
    
    # === Phase 3: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 3: L-BFGS Log-Space (best={best_score:.8f}) ===")
    for T_val in [200, 800, 3000]:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(rem * 0.2, 100.0)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_logspace(best_seq, t_alloc, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)
    
    # === Phase 4: Coordinate Descent with Exact Search ===
    print(f"\n=== Phase 4: Coordinate Descent [exact] (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem * 0.3, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent_exact(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd_exact", best_seq, best_score)
    
    # === Phase 5: Peak-Focused Search ===
    print(f"\n=== Phase 5: Peak-Focused Search (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.3, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_pf, sc_pf = peak_focused_search(best_seq, t_alloc, verbose=True)

    # === Phase 6: LP on the refined solution ===
    print(f"\n=== Phase 6: LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp3, sc_lp3 = lp_fictitious_play(best_seq, t_alloc, use_greedy=True, verbose=True)
        best_seq, best_score = check_and_update(a_lp3, "lp_polish", best_seq, best_score)
    
    # === Phase 7: Try nearby lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = [n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
                 n_parent - 4, n_parent + 3, n_parent + 4]
    for alt_n in nearby_ns:
        if alt_n < 64 or alt_n > 3000:
            continue
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        t_alloc = min(rem * 0.15, 60.0)
        if t_alloc < 15:
            break

        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_lbgfs", best_seq, best_score)

            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_lp", best_seq, best_score)
    
    # === Phase 8: Final coordinate descent polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_alloc = min(rem * 0.5, 60.0)
        a_final, sc_final = coordinate_descent_exact(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_final, "final_cd", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 10:

        a_fin2, sc_fin2 = lbgfs_logspace(best_seq, t_alloc, T=1000, maxiter=2000)
        best_seq, best_score = check_and_update(a_fin2, "final_lbgfs", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
