with L-BFGS polishing. The core idea is an iterative algorithm where:
1. Given current sequence f, compute M = max(f*f)
2. Solve LP to find g that maximizes sum(g) subject to (f*g)_k <= M, g >= 0
3. Mix f with normalized g using adaptive step size
This is a game-theoretic approach (fictitious play) that converges to
the minimax optimal sequence. L-BFGS with tmax-min smoothing is used

import time
from scipy.fft import rfft, irfft

    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A such that A @ g = convolve(f, g)."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""

    c = -np.ones(n)

    grad_z = a * (grad_a - S)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
with L-BFGS polishing and coordinate descent refinement.

Core algorithm (LP Fictitious Play):
  1. Given current non-negative sequence f (normalized to sum=1), compute M = max(f*f)
  2. Solve LP: maximize sum(g) subject to (f*g)_k <= M for all lags k, g >= 0
  3. Mix: f <- (1-alpha)*f + alpha*(g/sum(g))
  4. Repeat with decreasing alpha

This is a game-theoretic approach where the LP finds the "best response"
that spreads mass toward under-represented lags, progressively equalizing
the auto-convolution. Final polishing uses L-BFGS on a log-sum-exp
smoothed surrogate and coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score_normalized(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def solve_lp_best_response(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Build A_lp: A_lp[k, j] = f[k-j] if 0<=k-j<=n-1 else 0
    A_lp = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_lp[j:j + n, j] = f

    c = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_lp, b_ub=b_vec, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def fictitious_play(f0, time_limit, max_iters=10000, alpha_start=0.5,
                    alpha_decay=0.02, verbose=True, seed=0):
    """LP-based fictitious play to minimize max(auto_conv)."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        return f0.copy(), compute_score_normalized(f0)
    f = f / s

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ff = fast_convolve(f)
        M = ff.max()

        g = solve_lp_best_response(f, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Adaptive mixing rate
        alpha = alpha_start / (1.0 + alpha_decay * it)
        alpha = min(alpha, 1.0)

        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss

        score = compute_score_normalized(f_new)
        if score < best_score - 1e-15:
            best_score = score
            best_f = f_new.copy()
        f = f_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={it}, best_score={best_score:.8f}, M={M:.6e}, "
                  f"alpha={alpha:.4f}, time={elapsed:.1f}s")

    return best_f, best_score

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def smoothed_objective(z, T, n):
    """Soft-max smoothed log-sum-exp objective."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z

            if i == j or a[j] < 1e-12:
                continue
            t = 0.1 * a[j]
            a_trial = a.copy()
            a_trial[j] -= t
            a_trial[i] += t
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score_normalized(a_trial)
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved and iters > 20:
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = compute_score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def peak_equalize_polish(a0, time_limit, n_top_peaks=10):
    """Targeted reduction of the tallest peaks in the auto-convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score_normalized(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(n_top_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]

        improved = False
        # Focus on the single tallest peak
        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]

        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.05, 0.1, 0.15, 0.25, 0.4]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbation at peak-contributing positions
            peak_positions = set()
        for pi in peak_indices[:5]:
                si = max(0, int(pi) - n + 1)
                ei = min(n - 1, int(pi))
                for p in range(si, ei + 1):
                    peak_positions.add(p)
            if peak_positions:
                pos_list = list(peak_positions)
                for _ in range(15):
                    p = pos_list[rng.integers(0, len(pos_list))]
                    if a[p] < 1e-14:
                        continue
                    delta = a[p] * 0.05 * rng.standard_normal()
                    a_trial = a.copy()
                    a_trial[p] -= delta
                    mask2 = np.ones(n, dtype=bool)
                    mask2[p] = False
                    oth2 = a_trial[mask2]
                    o2s = oth2.sum()
                    if o2s > 1e-15:
                        a_trial[mask2] = oth2 + delta * oth2 / o2s
                    a_trial = np.maximum(a_trial, 0)
                    ss2 = a_trial.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial = a_trial / ss2
                    sc2 = compute_score_normalized(a_trial)
                    if sc2 < best_score:
                        a = a_trial
                        best_score = sc2
                        best_a = a.copy()
                        improved = True
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP Fictitious Play (arithmetic mixing)
    Phase 2: L-BFGS polish at multiple temperatures
    Phase 3: LP Fictitious Play (geometric mixing)
    Phase 4: Peak equalization + Coordinate descent
    Phase 5: Try nearby sequence lengths
    Phase 6: Final L-BFGS polish
    """
    deadline = time.monotonic() + max(1.0, budget_s - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = 1.0 + 0.3 * np.cos(t) + 0.1 * np.cos(2 * t + 0.5)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score_normalized(parent)
        print(f"Init: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP Fictitious Play (arithmetic) ===
    print(f"\n=== Phase 1: LP Fictitious Play (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    fp_budget = min(500.0, rem * 0.35)
    print(f"  Budget: {fp_budget:.0f}s")
    a_fp, sc_fp = fictitious_play(best_seq, fp_budget, max_iters=20000,
                                   alpha_start=0.4, alpha_decay=0.01, verbose=True)
    best_seq, best_score = check_and_update(a_fp, "FP_arith", best_seq, best_score)
    print(f"  Score after Phase 1: {best_score:.8f}")

    # === Phase 2: L-BFGS Polish ===
    print(f"\n=== Phase 2: L-BFGS Polish ===")
    for T_val in [200, 800, 3000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        lb_budget = min(80.0, rem * 0.1)
        print(f"  T={T_val}, budget={lb_budget:.0f}s")
        a_lb, _ = lbgfs_polish(best_seq, lb_budget, T=T_val, maxiter=5000)
        best_seq, best_score = check_and_update(a_lb, f"LBFGS_T{T_val}",
                                                 best_seq, best_score)
    print(f"  Score after Phase 2: {best_score:.8f}")

    # === Phase 3: Geometric Fictitious Play ===
    print(f"\n=== Phase 3: Geometric FP (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        gfp_budget = min(300.0, rem * 0.3)
        print(f"  Budget: {gfp_budget:.0f}s")
        a_gfp, _ = geometric_fictitious_play(best_seq, gfp_budget,
                                              max_iters=15000,
                                              alpha_start=0.2, alpha_decay=0.005,
                                              verbose=True)
        best_seq, best_score = check_and_update(a_gfp, "GFP", best_seq, best_score)
        print(f"  Score after Phase 3: {best_score:.8f}")

    # === Phase 4: Peak Equalization + Coordinate Descent ===
    print(f"\n=== Phase 4: Peak Eq + CD ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        pe_budget = min(120.0, rem * 0.2)
        print(f"  Peak Eq budget: {pe_budget:.0f}s")
        a_pe, _ = peak_equalize_polish(best_seq, pe_budget, n_top_peaks=12)
        best_seq, best_score = check_and_update(a_pe, "peak_eq", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 30:
        cd_budget = min(120.0, rem * 0.2)
        print(f"  CD budget: {cd_budget:.0f}s")
        a_cd, _ = coordinate_polish(best_seq, cd_budget, n_trials=200)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)
    print(f"  Score after Phase 4: {best_score:.8f}")

    # === Phase 5: Try nearby lengths ===
    print(f"\n=== Phase 5: Nearby Lengths ===")
    nearby_ns = [n_parent + 1, n_parent - 1, n_parent + 2, n_parent - 2,
                 n_parent + 4, n_parent - 4, n_parent + 8, n_parent - 8]
    for alt_n in nearby_ns:
        rem = deadline - time.monotonic()
        if rem < 50 or alt_n < 128 or alt_n > 4096:
            continue
        a_init = interpolate_sequence(best_seq, alt_n)
        # Quick FP + L-BFGS
        fp_b = min(40.0, rem * 0.15)
        a_alt, _ = fictitious_play(a_init, fp_b, max_iters=3000,
                                    alpha_start=0.3, alpha_decay=0.02, verbose=False)
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            lb_b = min(30.0, rem2 * 0.15)
            a_alt, _ = lbgfs_polish(a_alt, lb_b, T=500, maxiter=3000)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        for T_val in [5000, 20000]:
            rem2 = deadline - time.monotonic()
            if rem2 < 15:
                break
            lb_b = min(30.0, rem2 * 0.2)
            a_fin, _ = lbgfs_polish(best_seq, lb_b, T=T_val, maxiter=5000)
            best_seq, best_score = check_and_update(a_fin, f"final_T{T_val}",
                                                     best_seq, best_score)

    # Final coordinate polish
    rem = deadline - time.monotonic()
    if rem > 15:
        cd_b = min(30.0, rem * 0.4)
        a_fin2, _ = coordinate_polish(best_seq, cd_b, n_trials=500)
        best_seq, best_score = check_and_update(a_fin2, "final_cd", best_seq, best_score)

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
