def structured_peak_reduction(a, n_iters=1000):
    """Reduce peak by directly modifying the contributing pairs."""
    n = len(a)
    a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    for it in range(n_iters):
        b = fast_convolve(a)
        # Find peaks (exclude edges)
        b_inner = b.copy()
        b_inner[:3] = -1
        b_inner[-3:] = -1
        peak_k = argmax(b_inner)
        
        # Find contributing pairs
        # (a*a)[peak_k] = sum_i a_i * a[peak_k - i]
        # The pairs are (i, peak_k - i)
        
        # Reduce the largest contributing pair
        contribs = {}
        for i in range(max(0, peak_k - n + 1), min(n, peak_k + 1)):
            j = peak_k - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Sort by contribution
        sorted_pairs = sorted(contribs.items(), key=lambda x: -x[1])
        
        improved = False
        for i, c in sorted_pairs[:10]:
            j = peak_k - i
            if i == j:
                # Single element: reduce it
                for factor in [0.99, 0.98, 0.95, 0.9]:
                    a_new = a.copy()
                    a_new[i] *= factor
                    # Redistribute
                    diff = a[i] - a_new[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    # Redistribute to non-contributing positions
                    non_contrib = mask.copy()
                    for i2, _ in sorted_pairs[:5]:
                        non_contrib[i2] = False
                        non_contrib[peak_k - i2] = False
                    if non_contrib.sum() > 0:
                        a_new[non_contrib] += diff * a[non_contrib] / a[non_contrib].sum()
                    score = 2*n*fast_convolve(a_new).max()
                    if score < best_score:
                        a = a_new
                        best_score = score
                        improved = True
                        break
            else:
                # Pair: reduce both
                for f1, f2 in [(0.99, 0.99), (0.98, 0.98), (0.95, 0.95), (0.9, 0.9), (0.99, 0.95)]:
                    a_new = a.copy()
                    a_new[i] *= f1
                    a_new[j] *= f2
                    diff = (a[i] - a_new[i]) + (a[j] - a_new[j])
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    # Redistribute to non-contributing positions
                    non_contrib = mask.copy()
                    for i2, _ in sorted_pairs[:5]:

                    score = 2*n*fast_convolve(a_new).max()

                        improved = True
                        break
            if improved:

        if not improved:
            break
    
    return a, best_score

2. Nesterov-accelerated structured subgradient descent exploiting reflection structure

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

def simplex_project(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_aggressive(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS with multiple temperature stages, very tight tolerances."""
    n = len(a0)
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.1]
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    n_stages = len(T_stages)
    time_per_stage = time_limit / n_stages
    z = z0.copy()

    for stage_idx, stage_T in enumerate(T_stages):
        stage_deadline = t0 + (stage_idx + 1) * time_per_stage
        maxiter_this = max(100, int(maxiter_per_stage * min(1.0, time_per_stage / (time_limit / n_stages))))

        def func(z_inner):
            if time.monotonic() > stage_deadline:
                return 1e15
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            Tb = stage_T * b_inner
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_inner):
            if time.monotonic() > stage_deadline:
                return np.zeros_like(z_inner)
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            Tb = stage_T * b_inner
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_inner)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
            S = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_this, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': maxiter_this * 100, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = evaluate_safe(a_check)
            if stage_idx % 2 == 0 or stage_idx == n_stages - 1:

        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} error: {e}")

    a_out = softmax_from_z(z)
    return a_out


def nesterov_subgradient(a0, time_limit, n_max_iters=20000):
    """Nesterov-accelerated subgradient method using exact subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    momentum = 0.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_k = int(np.argmax(b_inner))
        max_b = b[peak_k]

        # Exact subgradient: g_i = 2*a[peak_k - i] for valid i
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * y[peak_k - idx]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Nesterov momentum update
        momentum = 0.9 * momentum + 0.1 * g
        lr = 2.0 / ((it + 1) * (it + 1) ** 0.5 + n ** 0.5)
        lr = min(lr, 0.05)

        y_new = y - lr * momentum
        y_new = simplex_project(y_new)

        # Weighted average for x (actual iterate)
        a = ((it + 1) * a + y_new) / (it + 2)
        a = simplex_project(a)

        # Evaluate both
        sc_a = evaluate_safe(a)
        sc_y = evaluate_safe(y_new)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_new.copy()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        y = y_new

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
