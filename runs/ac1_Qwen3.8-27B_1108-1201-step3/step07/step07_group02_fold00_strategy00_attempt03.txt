def interpolate_sequence(a, new_n):
    """Interpolate a sequence of length n to new length new_n."""
    if isinstance(a, list):
        a = np.array(a, dtype=float)
    if len(a) == new_n:
        return a.copy()
    old_n = len(a)
    s = a.sum()
    if s < 1e-15:  # Avoid division by zero
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a_norm)
