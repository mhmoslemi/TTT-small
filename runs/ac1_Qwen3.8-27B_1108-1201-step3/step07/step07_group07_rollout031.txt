def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlation(a, w):
    """
    Compute cross-correlation (a ⋆ w)[i] = sum_k a[k] * w[k+i] for i = 0, ..., len(a)-1.
    Using FFT.
    """

    m = len(w)

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return float(b.max())


def smoothed_objective_grad(a, beta):
    """
    Compute the smoothed objective value and gradient.
    G(a) = (1/beta) * log(sum_k exp(beta * b_k))
    grad = sum_k p_k * grad_b_k
    where p_k = exp(beta * b_k) / sum exp(beta * b_j)
    """

    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i]
    g = smooth_gradient(a, p)
    
    return G, g, b, p

from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, p, fft_size=None):
    """
    Compute smooth gradient: g[i] = 2 * sum_j a[j] * p[j+i]
    where p has length 2n-1 (the convolution output length).
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(np.conj(A) * P, fft_size)
    return 2.0 * corr[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Max of auto-convolution for unit-sum sequence."""
    return float(fast_convolve(a).max())

    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, p, fft_size=None):
    """
    Compute smooth gradient: g[i] = 2 * sum_j a[j] * p[j+i].
    a has length n, p has length 2n-1.
    Returns g of length n.
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(np.conj(A) * P, fft_size)
    return 2.0 * corr[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)
