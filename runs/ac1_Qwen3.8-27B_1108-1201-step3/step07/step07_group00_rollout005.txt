"""
Improved LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses quadratic exact line search, adaptive peak selection, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def quadratic_line_search(f0, g, b0, bfg, bgg, max_iters=20):
    """
    Find t in [0,1] that minimizes max_k [(1-t)^2*b0[k] + 2*t*(1-t)*bfg[k] + t^2*bgg[k]]
    Uses ternary search since the objective is convex in t.
    """
    lo, hi = 0.0, 1.0
    
    def obj(t):
        val = (1-t)**2 * b0 + 2*t*(1-t) * bfg + t**2 * bgg
        return val.max()
    
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-12:
            break
    
    t_star = (lo + hi) / 2
    return t_star


def lp_solve_direction(f, M, top_idx, solver='SCS', eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top_idx.
    Returns g or None.
    """
    n = len(f)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        if solver == 'SCS':
            prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
        elif solver == 'ECOS':
            prob.solve(solver=cp.ECOS, max_iters=100000)
        elif solver == 'MOSEK':
            prob.solve(solver=cp.MOSEK)
        else:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

def subgradient_polish(f_init, time_limit, lr_init=0.002, n_top=50, label="sg"):

        b = fast_convolve(f)
        M = b.max()
        
        # Weighted subgradient using soft-max over top peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()

                w_sel /= ws
            w_new = np.zeros(len(b))

            w = w_new
        
        # Compute subgradient
        g = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]
        
        # Update with momentum
        v = 0.85 * v - lr * g
        f_new = project_simplex(f + v)
        f = f_new
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0

            temp = min(temp * 1.1, 0.5)

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with geometric step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_obj = fast_convolve(f).max()
    multipliers = [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]
    improved_total = True
    
    print(f"  [{label}] Start: n={n}, obj={best_obj:.10f}")
    
    while improved_total and time.monotonic() - t0 < time_limit:
        improved_total = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999 or new_val < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved_total = True
    
    return f, best_obj


def resample_sequence(f, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Improved LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: exact quadratic line search, multi-solver LP,
adaptive peak selection with margin, perturbation restarts, and better time allocation.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def quadratic_line_search(b0, bfg, bgg, max_iters=50):
    """
    Find t in [0,1] minimizing max_k [(1-t)^2*b0[k] + 2*t*(1-t)*bfg[k] + t^2*bgg[k]].
    The objective is convex in t (max of quadratics), so ternary search works.
    """
    lo, hi = 0.0, 1.0

    def obj(t):
        val = (1.0 - t) ** 2 * b0 + 2.0 * t * (1.0 - t) * bfg + t ** 2 * bgg
        return val.max()

    for _ in range(max_iters):
        if hi - lo < 1e-12:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1

    return (lo + hi) / 2.0


def lp_solve_direction(f, M, top_idx, solver_name='SCS', eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top_idx.
    Returns g array or None on failure.
    """
    n = len(f)
    K = len(top_idx)

    # Build constraint matrix (sparse-friendly)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(top_idx[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        if solver_name == 'ECOS':
            prob.solve(solver=cp.ECOS, max_iters=200000)
        elif solver_name == 'MOSEK':
            prob.solve(solver=cp.MOSEK)
        elif solver_name == 'CLARABEL':
            prob.solve(solver=cp.CLARABEL)
        else:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

def subgradient_polish(f_init, time_limit, lr_init=0.0015, n_top=50, label="sg"):
    """Subgradient optimization with momentum and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    v = np.zeros(n)
    lr = lr_init
    temp = 0.0005
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42424)

    print(f"  [{label}] Start: n={n}, obj={best_obj:.10f}")

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Soft-max weighting over peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w = w / w_sum

        # Restrict to top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        # Compute subgradient (gradient of max w.r.t. f)
        g = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]

        f = project_simplex(f + v)

        if it % 30 == 0:
            obj = fast_convolve(f).max()

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with geometric step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_obj = fast_convolve(f).max()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.7, 2.2, 3.0]
    improved_total = True
    passes = 0

    print(f"  [{label}] Start: n={n}, obj={best_obj:.10f}")

    while improved_total and time.monotonic() - t0 < time_limit:
        improved_total = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-13:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 0.999 or new_val < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    improved_total = True

    return f, best_obj


def resample_sequence(f, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    -> subgradient polish -> coordinate descent -> final LP burst.

    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Main LP on parent length ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: LP main (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 620.0)
    if t1 > 40:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_optimize(parent, time_limit=t1, top_k_init=40,
                                    top_k_max=200, label="lp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {score1:.6f}")

    # === Phase 2: Deeper LP (more peaks, continued) ===
    print(f"\n=== Phase 2: LP deep (n={len(best_seq)}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 300.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = run_lp_optimize(best_seq, time_limit=t2, top_k_init=100,
                                    top_k_max=400, label="lp2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {score2:.6f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    candidates_n = []
    for factor in [0.5, 0.66, 0.75, 1.25, 1.5, 2.0]:
        tl = int(round(n_base * factor))
        tl = max(256, min(4096, tl))
        if tl != n_base:
            candidates_n.append(tl)
    for fixed_n in [512, 768, 1024, 1536, 2048]:
        if fixed_n != n_base and fixed_n not in candidates_n:
            candidates_n.append(fixed_n)
    candidates_n = sorted(set(candidates_n), key=lambda x: abs(x - n_base))

    for target_n in candidates_n[:4]:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        t3 = min(rem * 0.10, 100.0)
        if t3 < 15:
            break

        f_interp = resample_sequence(best_seq, target_n)
        print(f"  LP n={target_n}, budget={t3:.0f}s")
        f3, obj3 = run_lp_optimize(f_interp, time_limit=t3, top_k_init=40,
                                    top_k_max=150, label=f"lp_n{target_n}")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [n{target_n}]: {score3:.6f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 90.0)
    if t4 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, obj4 = subgradient_polish(best_seq, time_limit=t4,
                                       lr_init=0.001,
                                       n_top=max(30, n_b // 8),
                                       label="sg")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sg]: {score4:.6f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 50.0)
    if t5 > 8:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = coordinate_descent(best_seq, time_limit=t5, label="cd")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.30, 100.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_lp_optimize(best_seq, time_limit=t6, top_k_init=150,
                                    top_k_max=500, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

    # === Phase 7: Final coordinate polish ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.12, 35.0)
    if t7 > 5:
        f7, obj7 = coordinate_descent(best_seq, time_limit=t7, label="cd_final")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd_final]: {score7:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
