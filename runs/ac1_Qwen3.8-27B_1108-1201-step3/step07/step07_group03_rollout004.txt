    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def score_local(a):
    """Compute the local score 2*n*max_conv / (sum)^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def normalize_sum1(a):
    """Normalize a to be non-negative with sum exactly 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses with averaging and candidate tracking."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = normalize_sum1(np.asarray(initial, dtype=np.float64))

    b = a.copy().astype(np.float64)
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = score_local(a)
    iters = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.7 * a + 0.3 * a_avg, 0.3 * a + 0.7 * a_avg]
        improved = False
        for cand in candidates:
            cand = normalize_sum1(cand)
            sc = score_local(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                improved = True

        if not improved:
            stuck_count += 1
        else:
            stuck_count = 0

        if stuck_count > 30:
            # Perturb to escape
            rng = np.random.default_rng(iters * 31 + 7)
            noise = rng.standard_normal(n) * 0.02
            a = normalize_sum1(a + noise)
            b = normalize_sum1(b + noise)
            stuck_count = 0

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f} M_a={M_a:.6f}")

    return best_a, best_score, iters


def spectral_init(n, rng):
    """Initialize with a spectrally-flattened sequence."""
    # Start with random phasing in frequency domain
    L = 1 << (2 * n - 1).bit_length()
    # Construct a spectrum with flat amplitude and random phase
    A = np.ones(L // 2 + 1) * (0.5 + 0.5 * rng.random(L // 2 + 1))
    A = A * np.exp(1j * rng.uniform(0, 2 * np.pi, L // 2 + 1))
    # Ensure Hermitian symmetry
    A_full = np.zeros(L, dtype=complex)
    A_full[:L // 2 + 1] = A
    A_full[L // 2 + 1:] = np.conj(A[1:L // 2][::-1])
    a_full = np.fft.ifft(A_full).real[:n]
    a = np.maximum(a_full, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback: use a cosine-bell window
        t = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
        s = a.sum()
    return a / s


def cosine_bell_init(n, peak_position=0.5, width=0.3, rng=None):
    """Initialize with a cosine bell at a specified position."""
    t = np.arange(n) / n
    center = peak_position
    spread = width
    a = np.exp(-0.5 * ((t - center) / spread) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def bimodal_init(n, rng):
    """Initialize with two peaks."""
    c1 = rng.uniform(0.1, 0.3)
    c2 = 1.0 - c1
    t = np.arange(n) / n
    a = np.exp(-0.5 * ((t - c1) / 0.08) ** 2) + np.exp(-0.5 * ((t - c2) / 0.08) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def ramp_init(n, direction=1):
    """Linear ramp initialization."""
    a = np.linspace(0.1, 1.0, n)
    if direction == -1:
        a = a[::-1]
    a = np.maximum(a, 0)
    return a / a.sum()


def uniform_init(n):
    return np.ones(n) / n


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def peak_split_refine(a0, time_limit, verbose=False):
    """
    Targeted peak splitting: identify peak-contributing pairs and
    redistribute mass to reduce the maximum convolution value.
    """
    n = len(a0)
    a = normalize_sum1(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_local(a)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find contributing indices
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[k_peak - idx]
        max_contrib = contribs.max()
        if max_contrib < 1e-15:
            break

        # Try reducing the top contributors
        top_idx = np.argsort(contribs)[::-1][:min(10, len(idx))]
        improved = False

        for ci in top_idx:
            i = int(idx[ci])
            j = k_peak - i
            if j < 0 or j >= n or a[i] < 1e-14:
                continue

            # Try reducing a[i] (and possibly a[j])
            for f1 in [0.92, 0.85, 0.75, 0.65]:
                a_trial = a.copy()
                a_trial[i] *= f1
                a_trial = normalize_sum1(a_trial)
                sc = score_local(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_a = a.copy()
                    best_score = sc
                    improved = True
                    break

            if not improved and i != j and 0 <= j < n and a[j] >= 1e-14:
                for f1 in [0.88, 0.8]:
                    for f2 in [0.88, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = normalize_sum1(a_trial)
                        sc = score_local(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_a = a.copy()
                            best_score = sc
                            improved = True
                            break
                    if improved:
                        break

            if improved:
                break

        if not improved:
            # Try adding mass to non-peak positions
            conv_vals = conv
            # Find positions that don't contribute much to any peak
            low_contrib = np.zeros(n)
            for k in range(n - 1, n + min(n - 1, len(conv) - n)):
                if 0 <= k < len(conv):
                    i_s = max(0, k - n + 1)
                    i_e = min(n - 1, k)
                    for i in range(i_s, i_e + 1):
                        low_contrib[i] += conv[k] * a[i] * a[k - i]

            # Try increasing positions with low contribution
            sorted_positions = np.argsort(low_contrib)[:min(20, n)]
            for pos in sorted_positions:
                if a[pos] < 0.001:
                    a_trial = a.copy()
                    a_trial[pos] += 0.01
                    a_trial = normalize_sum1(a_trial)
                    sc = score_local(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break

            if not improved and step > 50:
                break

    return best_a, best_score


def adaptive_coord_descent(a0, time_limit, verbose=False):
    """
    Adaptive coordinate descent with multiple step strategies and restarts.
    More aggressive than simple coordinate descent.
    """
    n = len(a0)
    a = normalize_sum1(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit:
        # Pick a random subset of coordinates to try
        n_try = min(n, max(20, n // 4))
        trial_idx = rng.choice(n, size=n_try, replace=False)

        for i in trial_idx:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting it to a small positive value
                for new_val in [0.001, 0.005, 0.01]:
                    a[i] = new_val
                    a = normalize_sum1(a)
                    sc = score_local(a)
                    if sc < best_score - 1e-12:
                        best_a = a.copy()
                        best_score = sc
                        no_improve = 0
                    else:
                        a[i] = 0
                        a = normalize_sum1(a)
                continue

            # Try multiplicative changes (geometric spacing)
            factors = np.geomspace(0.5, 2.0, 12)
            best_factor = 1.0
            best_sc = best_score
            for factor in factors:
                a[i] = old_val * factor
                a = normalize_sum1(a)
                sc = score_local(a)
                if sc < best_sc:
                    best_sc = sc
                    best_factor = factor

            if best_factor != 1.0:
                a[i] = old_val * best_factor
                a = normalize_sum1(a)
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a[i] = old_val
                    a = normalize_sum1(a)
            else:
                no_improve += 1

        if no_improve >= max_no_improve:
            # Restart with perturbation
            strength = min(0.15, 0.03 * (no_improve // max_no_improve))
            noise = rng.standard_normal(n) * strength
            a = normalize_sum1(best_a + noise)
            no_improve = 0
            max_no_improve = max(100, max_no_improve // 2)

    return best_a, best_score


def nesterov_subgrad(a0, time_limit, top_k=30, lr_init=0.005, verbose=False):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a0)
    a = normalize_sum1(np.asarray(a0, dtype=np.float64))
    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = score_local(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        # Compute soft subgradient using softmax weighting over peaks
        conv = fast_convolve(y)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                sg[j] += 2.0 * w[idx] * y[k - j]

        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        sc = score_local(y_projected)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0

        if no_improve > 400:
            noise = rng.standard_normal(n) * 0.02
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 1.5)
            no_improve = 0

        lr *= 0.9998

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluator and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def two_scale_refine(a_coarse, n_fine, time_limit, solver=None):
    """Use a coarse solution to initialize and refine at fine scale."""
    a_fine_init = interpolate_seq(a_coarse, n_fine)
    t0 = time.monotonic()
    t_fp = time_limit * 0.7
    t_cd = time_limit * 0.3

    a_fp, s_fp, _ = fictitious_play(n_fine, t_fp, initial=a_fine_init, solver=solver)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_cd, s_cd = adaptive_coord_descent(a_fp, remaining)
        if s_cd < s_fp:
            return a_cd, s_cd
    return a_fp, s_fp


def propose_candidate():
    """
    Main optimization function. Uses multi-phase approach:
    Phase 1: Diverse initialization + quick screening
    Phase 2: LP fictitious play at promising lengths
    Phase 3: Intensive local optimization (coord descent + peak splitting)
    Phase 4: Final polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(2024)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Deadline in 1780 seconds")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = score_local(parent)
            # Verify with actual evaluator
            p_score_eval = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score_eval):
                best_seq = parent.copy()
                best_score = p_score_eval
                print(f"Parent: n={len(parent)}, score={p_score_eval:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Diverse Initialization and Quick Screening ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t1_start = time.monotonic()
    t_phase1 = 120.0

    # Generate diverse candidates at several lengths
    n_values = [640, 768, 896, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []  # (score, sequence, n, label)

    for n in n_values:
        if time.monotonic() - t1_start > t_phase1:
            break
        inits = []
        labels = []

        if parent is not None:
            inits.append(interpolate_seq(parent, n))
            labels.append("parent_interp")
        inits.append(uniform_init(n))
        labels.append("uniform")
        inits.append(cosine_bell_init(n, 0.5, 0.35, rng))
        labels.append("cosine_bell")
        inits.append(bimodal_init(n, rng))
        labels.append("bimodal")
        inits.append(spectral_init(n, rng))
        labels.append("spectral")
        inits.append(ramp_init(n, 1))
        labels.append("ramp")

        for init, label in zip(inits, labels):
            if time.monotonic() - t1_start > t_phase1:
                break
            # Quick LP: just a few iterations
            t_quick = 8.0
            a_opt, s_opt, _ = fictitious_play(n, t_quick, initial=init, solver=solver)
            candidates.append((s_opt, a_opt, n, f"{label}_n{n}"))
            print(f"    {label} n={n}: score={s_opt:.6f}")

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 candidates:")
    for sc, a_c, n_c, lab in candidates[:5]:
        print(f"    {lab}: {sc:.6f}")

    # Update best
    for sc, a_c, n_c, lab in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, lab)

    # === Phase 2: LP Fictitious Play at Best Lengths ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 900.0

    # Select lengths for intensive LP
    fp_lengths = []
    if best_seq is not None:
        fp_lengths.append(len(best_seq))
    # Add nearby lengths
    if best_seq is not None:
        nb = len(best_seq)
        for alt in [nb - 64, nb + 64, nb - 128, nb + 128]:
            if 256 <= alt <= 2048 and alt not in fp_lengths:
                fp_lengths.append(alt)
    # Ensure we have at least 3 lengths
    for n in [768, 1024, 896]:
        if n not in fp_lengths:
            fp_lengths.append(n)
    fp_lengths = fp_lengths[:4]

    # Allocate time proportionally (more time to the best length)
    time_per_fp = t_phase2 / len(fp_lengths)

    for i, n_fp in enumerate(fp_lengths):
        rem = deadline - time.monotonic() - 350
        if rem < 60:
            break
        # Give more time to the first (best) length
        if i == 0:
            t_alloc = min(rem * 0.5, time_per_fp * 2)
        else:
            t_alloc = min(rem * 0.3, time_per_fp)
        if t_alloc < 30:
            continue

        print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

        # Use best available as initial
        if best_seq is not None:
            initial = interpolate_seq(best_seq, n_fp)
        else:
            # Use best candidate at this length if available
            initial = None
            for sc_c, a_c, n_c, lab in candidates:
                if n_c == n_fp:
                    initial = a_c
                    break
            if initial is None:
                initial = np.ones(n_fp) / n_fp

        a_fp, score_fp, iters = fictitious_play(
            n_fp, t_alloc, initial=initial, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

        # If this is the best length, do additional LP round
        if n_fp == (len(best_seq) if best_seq is not None else -1):
            rem = deadline - time.monotonic() - 300
            if rem > 60:
                t_alloc2 = min(rem * 0.4, 200.0)
                print(f"  FP2 n={n_fp}, budget={t_alloc2:.0f}s")
                a_fp2, score_fp2, _ = fictitious_play(
                    n_fp, t_alloc2, initial=best_seq, solver=solver, verbose=False
                )
                best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, f"fp2_n{n_fp}")

    # === Phase 3: Intensive Local Optimization ===
    print(f"\n=== Phase 3: Local Optimization (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # 3a: Peak splitting
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_ps = min(rem * 0.3, 120.0)
            print(f"  PeakSplit n={n_best}, budget={t_ps:.0f}s")
            a_ps, s_ps = peak_split_refine(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "peak_split")

        # 3b: Adaptive coordinate descent
        rem = deadline - time.monotonic() - 150
        if rem > 40:
            t_cd = min(rem * 0.5, 200.0)
            print(f"  CoordDescent n={n_best}, budget={t_cd:.0f}s")
            a_cd, s_cd = adaptive_coord_descent(best_seq, t_cd)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_descent")

        # 3c: Nesterov subgradient
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_nest = min(rem * 0.5, 100.0)
            print(f"  Nesterov n={n_best}, budget={t_nest:.0f}s")
            a_nt, s_nt = nesterov_subgrad(best_seq, t_nest, top_k=40, lr_init=0.004)
            best_seq, best_score = check_and_update(a_nt, best_seq, best_score, "nesterov")

        # 3d: Another round of peak splitting after gradient methods
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_ps2 = min(rem * 0.6, 60.0)
            a_ps2, s_ps2 = peak_split_refine(best_seq, t_ps2)
            best_seq, best_score = check_and_update(a_ps2, best_seq, best_score, "peak_split2")

    # === Phase 4: Try Alternative Strategy - Different Length ===
    print(f"\n=== Phase 4: Alternative Exploration (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        # Try a slightly different length with fresh initialization
        alt_n = n_best + int(n_best * 0.15)
        if 256 <= alt_n <= 2560:
            rem = deadline - time.monotonic() - 60
            if rem > 60:
                t_alt = min(rem * 0.5, 120.0)
                print(f"  Alt length n={alt_n}, budget={t_alt:.0f}s")
                # Use spectral init for diversity
                a_alt_init = spectral_init(alt_n, rng)
                a_alt, s_alt, _ = fictitious_play(alt_n, t_alt, initial=a_alt_init, solver=solver)
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

                # Refine with coord descent
                rem = deadline - time.monotonic() - 40
                if rem > 20 and s_alt < best_score + 0.001:
                    a_alt_cd, s_alt_cd = adaptive_coord_descent(a_alt, min(rem * 0.5, 40.0))
                    best_seq, best_score = check_and_update(a_alt_cd, best_seq, best_score, f"alt_cd_n{alt_n}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_final = len(best_seq)

        # Alternate between different local methods
        rem = deadline - time.monotonic() - 15
        methods_cycle = [
            ("cd", adaptive_coord_descent, {}),
            ("ps", peak_split_refine, {}),
            ("nest", nesterov_subgrad, {"top_k": 50, "lr_init": 0.002}),
        ]

        while rem > 30:
            for name, func, kwargs in methods_cycle:
                rem = deadline - time.monotonic() - 10
                if rem < 20:
                    break
                t_alloc = min(rem * 0.4, 50.0)
                print(f"  Final {name} budget={t_alloc:.0f}s")
                a_new, s_new = func(best_seq, t_alloc, **kwargs)
                best_seq, best_score = check_and_update(a_new, best_seq, best_score, f"final_{name}")
                rem = deadline - time.monotonic() - 10
                if rem < 20:
                    break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Elapsed: {time.monotonic() - (deadline - 1780.0):.1f}s")
    return [float(x) for x in best_seq]
