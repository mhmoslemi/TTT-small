"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization (uniform, cosine, Gaussian, random)
2. L-BFGS with increasing l_p power approximation (p=2,4,8,...,256)
3. LP-based fictitious play with Nesterov dual averaging
4. Subgradient descent with Nesterov momentum
5. Coordinate-wise exact 1D optimization
6. Peak shaving with pair coordinate updates
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that (T @ b)[k] = sum_i a[max(0,k-n+1):min(n,k+1)] ... """
    # Actually, for the LP: given fixed a, we want T @ b = conv(a, b)
    # (conv(a,b))[k] = sum_j a[j] * b[k-j]
    # So T[k, j] = a[j] if 0 <= k-j <= n-1, i.e., if max(0, k-n+1) <= j <= min(n-1, k)
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if hasattr(get_lp_solver, '_cached'):
        return get_lp_solver._cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    get_lp_solver._cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

    b = fft_convolve(a)
    b_safe = np.maximum(b, 1e-300)
    w = b_safe ** (p - 1)  # length 2n-1
    grad = np.zeros(n)
    for i in range(n):
        # sum_k w[k] * a[k-i] where 0 <= k-i <= n-1
        # k-i >= 0 => k >= i
        # k-i <= n-1 => k <= i+n-1
        k_min = max(i, 0)
        k_max = min(i + n - 1, 2 * n - 2)
        if k_min > k_max:
            continue
        # j = k - i, ranges from k_min - i to k_max - i
        j_min = k_min - i
        j_max = k_max - i
        grad[i] = np.dot(w[k_min:k_max + 1], a[j_min:j_max + 1])
    return 2.0 * p * grad


def lbgfs_lp_power(a0, time_limit, p_values=None, maxiter_per_p=None):
    """
    L-BFGS optimization using l_p power approximation with increasing p.
    Maximizes smoothness at low p, focuses on peak at high p.
    """
    if p_values is None:
        p_values = [2, 4, 8, 16, 32, 64, 128, 256]
    if maxiter_per_p is None:
        maxiter_per_p = 500
    
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    
    for p_val in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func_lpf(z_vec):
            a = softmax_from_z(z_vec)
            b = fft_convolve(a)
            b_safe = np.maximum(b, 1e-300)
            return float(np.sum(b_safe ** p_val))
        
        def jac_lpf(z_vec):
            a = softmax_from_z(z_vec)
            b = fft_convolve(a)
            b_safe = np.maximum(b, 1e-300)
            w = b_safe ** (p_val - 1)
            # grad w.r.t. a: 2*p * sum_k w_k * a[k-i]
            grad_a = np.zeros(n)
            for i in range(n):
                k_min = max(i, 0)
                k_max = min(i + n - 1, 2 * n - 2)
                if k_min <= k_max:
                    j_min = k_min - i
                    j_max = k_max - i
                    grad_a[i] = 2.0 * p_val * np.dot(w[k_min:k_max + 1], a[j_min:j_max + 1])
            # Convert to z-space: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func_lpf,
                x0=z,
                jac=jac_lpf,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_p, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxfun': maxiter_per_p * 100, 'disp': False}
            )
            z = res.x
            a = softmax_from_z(z)
            score = 2.0 * n * fft_convolve(a).max()
            print(f"    Lp p={p_val}: score={score:.6f}")
        except Exception as e:
            print(f"    Lp p={p_val}: error {e}")
    
    a_out = softmax_from_z(z)
    return a_out


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=10000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        # Weighted average for a
        a = ((it + 1) * a + (it + 2) * b_new) / (2 * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        score_a = 2.0 * n * fft_convolve(a).max()
        score_y = 2.0 * n * fft_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            print(f"    NesterovLP iter={it+1}: score={best_score:.6f}")
    
    return best_a, best_score


def subgradient_momentum(a0, time_limit, lr_init=0.01, momentum=0.9, n_steps=20000):
    """Subgradient descent with Nesterov momentum on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    # Nesterov: maintain y (lookahead) and v (velocity)
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fft_convolve(y)
        max_b = b.max()
        
        # Active set: indices where b is within 5% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: average of gradients at active points
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Step size (Polyak-like)
        t = step + 1
        lr = lr_init / np.sqrt(t)
        
        # Nesterov update
        v = momentum * v - lr * g
        y_new = y + v
        # Project to simplex
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            continue
        y_new = y_new / sy
        
        a = y_new
        y = y_new
        
        score = 2.0 * n * fft_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def coordinate_exact_optimize(a0, time_limit, n_passes=50):
    """
    For each coordinate i, find the optimal a[i] given all others fixed.
    The objective as a function of a[i] (with others fixed and sum=1) is:
    Let a[i] = t, others scaled to sum to 1-t.
    b_k = a[k-i](1-t) + t*a[k-i] + ... (complicated)
    
    Actually, let's parameterize: a[i] = t, a[j] = a0[j] * (1-t) / (1-a0[i]) for j != i.
    Then b_k = sum_{j != i} a[j] * a[k-j] + a[i]*a[k-i] (if k-i != i) 
              + 2 * a[i] * sum_{j != i, j != k-i} a[j] * delta... 
    
    This is a quadratic in t: b_k = A_k * t^2 + B_k * t + C_k
    And we want to minimize max_k b_k(t).
    
    This is a 1D problem: minimize max of quadratics. Can be solved by 
    finding the t where the max is minimized (bisection on the derivative).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    
    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Current value
            t0_val = a[i]
            others_sum = 1.0 - t0_val
            if others_sum < 1e-15:
                continue
            
            # Parameterize: a[i] = t, a[j] = a[j]_old * (1-t) / others_sum for j != i
            # b_k(t) = sum_j sum_l a[j](t) * a[k-j](t)
            # For j != i, l != i: a[j](t)*a[k-j](t) = a[j]*a[k-j] * ((1-t)/others_sum)^2
            # For j = i, l != i: a[i](t)*a[k-i](t) = t * a[k-i] * (1-t)/others_sum
            # For j != i, l = i: a[j](t)*a[k-j](t) = a[j] * (1-t)/others_sum * t
            # For j = i, l = i (k = 2i): t^2
            
            # So b_k(t) = C_k * ((1-t)/s)^2 + D_k * t*(1-t)/s + (1 if k==2i else 0) * t^2
            # where s = others_sum, C_k = sum_{j != i, k-j != i} a[j]*a[k-j]
            # D_k = a[k-i] (if k-i != i and k-i in range) + a[k-i] (if k != 2i)
            # Actually let me just compute it numerically.
            
            # Sampling approach: try several values of t
            # t ranges from 0 to 1 (since a[i] >= 0 and sum = 1)
            # But t should be close to t0_val for a good local search
            
            t_candidates = np.concatenate([
                np.linspace(0, 1, 30),
                np.array([t0_val * 0.5, t0_val * 0.7, t0_val * 0.9,
                          t0_val, t0_val * 1.1, t0_val * 1.3, t0_val * 1.5,
                          t0_val + 0.01, t0_val - 0.01,
                          t0_val + 0.05, t0_val - 0.05])
            ])
            t_candidates = np.clip(t_candidates, 0, 1)
            
            best_t = t0_val
            best_local_score = best_score
            
            for t in t
                if abs(t - t0_val) < 1e-10:
                    continue
                a_trial = np.empty(n)
                a_trial[i] = t
                if others_sum > 1e-15:
                    a_trial[np.arange(n) != i] = a[np.arange(n) != i] * (1 - t) / others_sum
                else:
                    a_trial[np.arange(n) != i] = 0
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = 2.0 * n * fft_convolve(a_trial).max()
                if sc < best_local_score:
                    best_local_score = sc
                    best_t = t
            
            if best_t != t0_val:
                a[i] = best_t
                if (1.0 - best_t) > 1e-15:
                    a[np.arange(n) != i] = a[np.arange(n) != i] * (1 - best_t) / (1.0 - t0_val)
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                sc = 2.0 * n * fft_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved and p > 3:
            break
    
    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, n_steps=10000):
    """Aggressive peak shaving with top-contributor analysis."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fft_convolve(a)
        # Find the peak (excluding endpoints which are always small)
        b_mid = b.copy()
        if n > 2:
            b_mid[0] = -np.inf
            b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(10, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fft_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair updates
            if len(top_contrib) >= 2:
                for ci1 in top_contrib[:4]:
                    for ci2 in top_contrib[:4]:
                        if ci1 == ci2:
                            continue
                        i1 = idx[ci1]
                        i2 = idx[ci2]
                        if a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        for f1 in [0.93, 0.88]:
                            for f2 in [0.93, 0.88]:
                                a_trial = a.copy()
                                a_trial[i1] *= f1
                                a_trial[i2] *= f2
                                a_trial = np.maximum(a_trial, 0)
                                st = a_trial.sum()
                                if st < 1e-15:
                                    continue
                                a_trial = a_trial / st
                                score = 2.0 * n * fft_convolve(a_trial).max()
                                if score < best_score - 1e-12:
                                    a = a_trial
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    break
                            if improved:
                                break
                        if improved:
                            break
                    if improved:
                        break
            
            if not improved:
                # No improvement possible with current strategy
                # Try random perturbation
                if step % 100 == 0:
                    perturbation = np.random.default_rng(step)
                    idx_pert = perturbation.integers(0, n, size=min(5, n))
                    a_trial = a.copy()
                    for ip in idx_pert:
                        a_trial[ip] *= (0.9 + 0.2 * perturbation.random())
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial / st
                        score = 2.0 * n * fft_convolve(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                if not improved and step > 500:
                    # Stuck, break
                    if step % 500 == 0 or step > 1500:
                        break
    
    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def cosine_window(n):
    """Raised cosine window."""
    j = np.arange(n)
    return np.cos(np.pi * (j + 0.5) / n) ** 2


def gaussian_window(n, sigma_frac=0.4):
    """Gaussian window."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((j - center) / sigma) ** 2)


def tent_window(n):
    """Triangular (tent) window."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    return np.maximum(0, 1 - np.abs(j - center) / center)
