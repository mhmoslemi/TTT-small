def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def lp_direction(f, delta=0.05, max_constraints=200):
    """
    Solve LP for direction: max sum(g) s.t. g>=0, (f*g)[k] <= M for peaks.
    Uses threshold-based peak selection: all k where b_k > M*(1-delta).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    threshold = M * (1.0 - delta)
    peak_idx = np.where(b >= threshold)[0]

    if len(peak_idx) > max_constraints:
        # Fall back to top K
        top_k = max_constraints
        peak_idx = np.argpartition(b, -top_k)[-top_k:]
    peak_idx = np.sort(peak_idx)
    K = len(peak_idx)

    if K == 0:
        return None, M

    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = peak_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=200000)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

    stuck = 0
    delta = 0.08

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, M = lp_direction(f, delta=delta, max_constraints=200)

        if g is None:
            stuck += 5
            delta = max(0.01, delta * 0.7)
            if stuck > 20:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 3
            if stuck > 15:
                break
            delta = max(0.01, delta * 0.8)
            continue

        h = g / S_g

        # Line search
        best_t = 0.1
        best_t_score = score_from_f(f)
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)

            f_trial /= s
            sc = score_from_f(f_trial)
            if sc < best_t_score:

                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_score = score_from_f(f)
        if cur_score < best_score - 1e-12:
            best_score = cur_score

def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_score = score_from_f(f)
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]
    rng = np.random.default_rng(777)

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                # Try setting to small positive
                f_trial = f.copy()
                f_trial[j] = 1e-4
                s = f_trial.sum()
                f_trial /= s
                sc = score_from_f(f_trial)
                if sc < best_score - 1e-12:
                    f = f_trial
                    best_score = sc
                    improved = True
                continue

            best_j_val = f[j]
            best_j_score = best_score
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0 or new_val > 10.0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:

                f_trial /= s
                sc = score_from_f(f_trial)
                if sc < best_j_score - 1e-12:
                    best_j_score = sc
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_score < best_score - 1e-12:
                    best_score = best_j_score
                    improved = True

    return f, best_score

def make_gaussian(n, sigma_frac=0.3):

    sigma = sigma_frac * n
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_blackman(n):
    """Blackman window."""
    a = np.zeros(n)
    for i in range(n):
        x = i / (n - 1) if n > 1 else 0
        a[i] = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, target_n):
    """Interpolate sequence to target length and renormalize."""
    n = len(f)
    if n == target_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-10:
        f_interp /= s
    else:
        f_interp = np.ones(target_n) / target_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-method approach combining soft-max L-BFGS,
    momentum simplex, LP direction finding, coordinate descent,
    and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = 2.0
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Soft-max L-BFGS annealing ===
    print(f"\n=== Phase 1: Soft-max L-BFGS (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, s1 = run_softmax_lbfgs(best_seq, time_limit=t1, label="sml1")
        if s1 < best_score:
            best_score = s1
            best_seq = f1.copy()
            print(f"  NEW BEST [sml1]: {best_score:.6f}")

    # === Phase 2: Momentum simplex ===
    print(f"\n=== Phase 2: Momentum simplex (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 350.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, s2 = run_momentum_simplex(best_seq, time_limit=t2, label="mom1")
        if s2 < best_score:
            best_score = s2
            best_seq = f2.copy()
            print(f"  NEW BEST [mom1]: {best_score:.6f}")

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, s3 = run_lp_phase(best_seq, time_limit=t3, label="lp1")
        if s3 < best_score:
            best_score = s3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp1]: {best_score:.6f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    if t4 > 15:
        print(f"  Budget: {t4:.0f}s")
        f4, s4 = coordinate_polish(best_seq, time_limit=t4, label="cd1")
        if s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [cd1]: {best_score:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.75, 0.85, 1.2, 1.5, 2.0]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t5 = min(rem * 0.25, 200.0)
        if t5 < 25:
            break

        f_interp = interpolate_sequence(best_seq, target_n)
        interp_score = score_from_f(f_interp)
        print(f"  n={target_n}, interp_score={interp_score:.6f}, budget={t5:.0f}s")

        # Quick soft-max optimization
        f5, s5 = run_softmax_lbfgs(f_interp, time_limit=min(t5 * 0.5, 80.0),
                                   label=f"sml_n{target_n}")
        cur_best = f5 if s5 < best_score else best_seq
        cur_score = min(s5, best_score)

        # Quick LP
        rem_sub = deadline - time.monotonic()
        if rem_sub > 20 and t5 * 0.4 > 15:
            f5b, s5b = run_lp_phase(cur_best, time_limit=min(rem_sub * 0.5, 80.0),
                                     label=f"lp_n{target_n}")
            if s5b < cur_score:
                cur_best = f5b
                cur_score = s5b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_best.copy()
            print(f"  NEW BEST [n{target_n}]: {best_score:.6f}")

    # === Phase 6: Diversity - new starting points ===
    print(f"\n=== Phase 6: Diversity (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_best = len(best_seq)
        starters = [
            ("raised_cos", make_raised_cosine(n_best)),
            ("gauss_02", make_gaussian(n_best, 0.2)),
            ("gauss_03", make_gaussian(n_best, 0.3)),
            ("gauss_04", make_gaussian(n_best, 0.4)),
            ("blackman", make_blackman(n_best)),
        ]

        per_starter = min(rem * 0.15, 80.0)
        for name, starter in starters[:3]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            st_score = score_from_f(starter)
            print(f"  {name}: initial_score={st_score:.6f}, budget={per_starter:.0f}s")

            # Quick optimization
            fs, ss = run_softmax_lbfgs(starter, time_limit=per_starter * 0.6,
                                       label=f"div_{name}")
            rem_sub = deadline - time.monotonic()
            if rem_sub > 15 and per_starter * 0.3 > 10:
                fs2, ss2 = run_lp_phase(fs, time_limit=min(rem_sub * 0.6, 40.0),
                                         label=f"lp_{name}")
                if ss2 < ss:
                    fs, ss = fs2, ss2

            if ss < best_score:
                best_score = ss
                best_seq = fs.copy()
                print(f"  NEW BEST [{name}]: {best_score:.6f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Short LP
        t7a = min(rem * 0.5, 60.0)
        if t7a > 10:
            f7, s7 = run_lp_phase(best_seq, time_limit=t7a, label="lp_final")
            if s7 < best_score:
                best_score = s7
                best_seq = f7.copy()

        # Coordinate descent
        rem = deadline - time.monotonic()
        t7b = min(rem * 0.5, 40.0)
        if t7b > 8:
            f7b, s7b = coordinate_polish(best_seq, time_limit=t7b, label="cd_final")
            if s7b < best_score:
                best_score = s7b
                best_seq = f7b.copy()

        # Short momentum
        rem = deadline - time.monotonic()
        t7c = min(rem * 0.5, 30.0)
        if t7c > 8:
            f7c, s7c = run_momentum_simplex(best_seq, time_limit=t7c, label="mom_final")
            if s7c < best_score:
                best_score = s7c
                best_seq = f7c.copy()

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
