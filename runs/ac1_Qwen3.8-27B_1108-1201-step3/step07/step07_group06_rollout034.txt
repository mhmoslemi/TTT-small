def fast_convolve(a):
    """Linear auto-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a, fft_size=None):
    """
    Compute cross-correlation: corr[i] = sum_j w[j+i] * a[j]
    w has length Lw, a has length La.
    Returns corr[0 : La] (the part we need for the gradient).
    """
    La = len(a)
    Lw = len(w)
    if fft_size is None:
        fft_size = 1 << (Lw + La - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return corr[:La]


def project_simplex(v, target=1.0):
    """
    Project v onto {x >= 0, sum(x) = target} (Euclidean projection).
    Uses the algorithm from Duchi et al. (2011).
    """
    n = len(v)
    if n == 0:
        return v
    # Scale to work with sum = 1, then rescale
    u = np.sort(v / target)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.zeros(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v / target - theta, 0.0) * target

    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = (u * idx > (cssv - target))
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_weights(b, T):
    """Compute soft-max weights from b with temperature T."""
    b_max = b.max()
    exp_b = np.exp(b / T - b_max / T)
    s = exp_b.sum()
    if s < 1e-300:
        return np.ones_like(b) / len(b)
    return exp_b / s


def get_score_fast(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def init_cosine(n):
    """Raised-cosine (Hann-window) shape."""
    i = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_gaussian(n, sigma_factor=0.4):
    """Gaussian bump centered in the sequence."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a + 1e-6
    return a / a.sum()


def init_super_gaussian(n, p=4.0):
    """Super-Gaussian: exp(-|x|^p)."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    x = (i - center) / (n / 2.0)
    a = np.exp(-np.abs(x) ** p)
    a = a + 1e-6
    return a / a.sum()


def init_sinusoidal(n, n_harmonics=5, rng=None):
    """Sum of sinusoids with random phases, made non-negative."""
    if rng is None:
        rng = np.random.default_rng(0)
    i = np.linspace(0, 2.0 * np.pi, n, endpoint=False)
    a = np.full(n, 1.0)
    for k in range(1, n_harmonics + 1):
        freq = k
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = 0.5 / k
        a += amp * np.sin(freq * i + phase)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def init_interpolated(parent, n, rng=None, noise=0.0):
    """Interpolate parent to length n with optional noise."""
    if len(parent) == n:
        a = parent.astype(np.float64).copy()
    else:
        old_x = np.linspace(0, 1, len(parent))
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, parent)
    if noise > 0:
        if rng is None:
            rng = np.random.default_rng(0)
        a = a + noise * rng.standard_normal(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
