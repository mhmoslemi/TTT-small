def fast_convolve(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    if s < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact 1D coordinate descent
with goldensection search, randomized order, and adaptive jumps.
Complements the previous Adam-based approach with a fundamentally different
optimization strategy that solves each 1D subproblem exactly.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def goldensection_minimize(f, a_lb, a_ub, max_iter=60, tol=1e-12):
    """1D minimization via golden section search."""
    gr = 0.6180339887498949  # (sqrt(5)-1)/2
    c = a_ub - gr * (a_ub - a_lb)
    d = a_lb + gr * (a_ub - a_lb)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if a_ub - a_lb < tol:
            break
        if fc < fd:
            a_ub = d
            d = c
            fd = fc
            c = a_ub - gr * (a_ub - a_lb)
            fc = f(c)
        else:
            a_lb = c
            c = d
            fc = fd
            d = a_lb + gr * (a_ub - a_lb)
            fd = f(d)
    x_opt = (a_lb + a_ub) * 0.5
    f_opt = min(fc, fd, f(x_opt))
    return x_opt, f_opt


def coord_sweep(a, c, order, n, S_total):
    """
    One pass of coordinate descent. Updates a and c in place.
    Returns updated S_total.
    """
    for j in order:
        x = a[j]
        k_min = j
        k_max = min(j + n - 1, 2 * n - 2)
        relevant_ks = np.arange(k_min, k_max + 1)
        offsets = relevant_ks - j
        mask_par = (offsets == j)
        mask_lin = ~mask_par
        
        # Compute S_relevant = c[relevant] - contribution of a[j]
        S_rel = c[relevant_ks].copy()
        if np.any(mask_lin):
            idx_lin = np.where(mask_lin)[0]
            S_rel[idx_lin] -= 2.0 * x * a[offsets[idx_lin]]
        if np.any(mask_par):
            idx_par = np.where(mask_par)[0]
            S_rel[idx_par] -= x * x
        
        # Constant part (max of c outside relevant range)
        C = 0.0
        if k_min > 0:
            C = max(C, c[:k_min].max())
        if k_max < 2 * n - 2:
            C = max(C, c[k_max + 1:].max())
        
        # Precompute coefficients
        coeffs = 2.0 * a[offsets].copy()
        coeffs[mask_par] = 0.0
        
        # Save for parabola update
        if np.any(mask_par):
            s_par = S_rel[mask_par].copy()
        else:
            s_par = None
        
        # Current score
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (S_total * S_total)
        
        # Objective function
        def obj(xp):
            vals = S_rel + coeffs * xp
            if s_par is not None:
                vals[mask_par] = s_par + xp * xp
            m = vals.max()
            if m < C:
                m = C
            ns = S_total - x + xp
            if ns < 1e-15:
                return 1e15
            return 2.0 * n * m / (ns * ns)
        
        # Search range
        x_max = min(3.0 * x + 0.001, 3.0 * a.max() + 0.01, 5.0)
        if x_max < 0.001:
            x_max = 0.01
        
        xp_opt, f_opt = goldensection_minimize(obj, 0.0, x_max, max_iter=50)
        
        if f_opt < cur_score - 1e-13:
            a[j] = xp_opt
            c[relevant_ks] = S_rel + coeffs * xp_opt
            if s_par is not None:
                c[relevant_ks[mask_par]] = s_par + xp_opt * xp_opt
            S_total = S_total - x + xp_opt
    
    return S_total
