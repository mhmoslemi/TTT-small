"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated smooth (log-sum-exp) gradient with temperature annealing
2. Nesterov momentum + per-coordinate Adam-like adaptive step sizes
3. LP-based fictitious play for polynomial-time refinement
4. Coordinate descent polishing on peak-contributing coordinates
5. Multi-length exploration with diverse initializations
Key innovation: O(n log n) gradient via FFT correlation (vs O(n*k) subgradient)

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def smooth_gradient(a, tau, fft_size=None):
    """
    Compute gradient of tau*log(sum(exp(b[k]/tau))) w.r.t. a.
    g[i] = 2 * sum_j w[i+j] * a[j] where w[k] = exp(b[k]/tau)/Z
    Computed via FFT: g[i] = 2 * convolve(a_rev, w)[i+n-1]
    """
    n = len(a)
    b = fast_convolve(a)
    # Softmax weights (shift to avoid overflow)
    b_shifted = b - b.max()
    w = np.exp(b_shifted / tau)
    Z = w.sum()
    if Z < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / Z
    # Gradient via correlation
    a_rev = a[::-1].copy()
    full_conv = np.convolve(a_rev, w, mode='full')
    g = 2.0 * full_conv[n - 1: 2 * n - 1]
    return g, b.max()

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def smooth_nesterov_adam(a0, time_limit, tau_init=0.05, tau_final=0.003,
                          lr_init=0.003, beta1=0.9, beta2=0.999, verbose=False):
    """
    Nesterov-accelerated smooth optimization with Adam-like per-coordinate
    adaptive step sizes and temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Nesterov state
    y = a.copy()
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    improving_iters = 0

    while time.monotonic() - t0 < time_limit:
        # Temperature schedule (exponential decay)
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_final / tau_init) ** frac
        tau = max(tau, tau_final)

        # Nesterov lookahead
        t_k = 1.0 + np.sqrt(1.0 + 4.0 * (t - 1) ** 2) / 2.0
        beta = (t_k - 1.0) / t_k

        # Compute smooth gradient at y
        g, peak_val = smooth_gradient(y, tau)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Step size with decay
        lr = lr_init / (1.0 + 0.5 * np.sqrt(t))

        # Update
        y_new = y - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        y_projected = simplex_project(y_new)

        # Nesterov momentum
        y = beta * y_projected + (1 - beta) * best_a

        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
            improving_iters += 1
        else:
            no_improve += 1

        t += 1

        # Restart logic
        if no_improve > 80:
            # Perturb best and restart
            sigma = 0.02 * (1.0 + no_improve / 200.0)
            noise = np.random.default_rng(t * 31 + 7).standard_normal(n) * sigma
            y = simplex_project(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.8
            no_improve = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SmoothNest t={t} ({elapsed:.0f}s): best={best_score:.6f} tau={tau:.5f} lr={lr:.6f}")

    return best_a, best_score

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver_cached
    if '_lp_solver_cached' in globals() and _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

                best_a = cand.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FP it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_cosine(n, rng):
    """Cosine-based initialization with random phase and width."""
    phase = rng.uniform(-np.pi, np.pi)
    width = rng.uniform(0.3, 1.5)
    t = np.arange(n, dtype=np.float64) / n
    a = np.cos(2 * np.pi * width * (t - 0.5) + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_gaussian_bump(n, rng):
    """Gaussian bump centered at random location."""
    center = rng.uniform(0.2, 0.8) * n
    sigma = rng.uniform(0.05, 0.25) * n
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-((t - center) ** 2) / (2 * sigma ** 2))
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

                best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, f"fp_alt_n{alt_n}")

    # === Phase 3: Coordinate Descent + Peak Shaving ===
    print(f"\n=== Phase 3: Coord Descent + Peak Shave (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        t_rem = deadline - time.monotonic() - 100
        if t_rem > 20:
            t_cd = min(t_rem * 0.4, 150.0)
            print(f"  CoordDescent n={n_best}, budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent(best_seq, t_cd, n_sweeps=200)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_desc")

        # Peak shaving
        t_rem = deadline - time.monotonic() - 60
        if t_rem > 15:
            t_ps = min(t_rem * 0.4, 80.0)
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "peak_shave")

        # Another smooth pass on the polished result
        t_rem = deadline - time.monotonic() - 30
        if t_rem > 20:
            t_sm = min(t_rem * 0.5, 60.0)
            a_sm, s_sm = smooth_nesterov_adam(
                best_seq, t_sm,
                tau_init=0.005, tau_final=0.001,
                lr_init=0.002
            )
            best_seq, best_score = check_and_update(a_sm, best_seq, best_score, "smooth_final")

        # Final coordinate descent
        t_rem = deadline - time.monotonic() - 15
        if t_rem > 10:
            a_fin, _ = coord_descent(best_seq, t_rem * 0.6, n_sweeps=50)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
