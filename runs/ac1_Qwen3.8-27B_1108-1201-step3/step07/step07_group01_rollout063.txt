```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_val(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(n-1, k)
        for j in range(js, je+1):
            A[k, j] = f[k-j]
    return A
```

    best_b = np.ones(n) / n
    for _ in range(max_rounds):
        A_active = np.array([f[np.arange(max(0, k - n + 1), min(n, k + 1))]
                             for k in active])
        # Fix the matrix properly
        A_active = np.zeros((len(active), n))
        for ii, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            A_active[ii, js:je] = f[k - (je - 1):k - js + 1][::-1] if je > js else 0

        # Actually rebuild properly
        A_active = np.zeros((len(active), n))
        for ii, k in enumerate(active):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A_active[ii, j] = f[k - j]

        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True})
        if not res.success:
            break
        b = np.maximum(res.x, 0.0)

        # Check all constraints
        conv_fb = cross_convolve(f, b)
        violated = list(np.where(conv_fb > M + 1e-12)[0])
        if not violated:
            return b
        active = list(set(active) | set(violated))
        best_b = b

    return best_b


def lp_optimize(f_init, time_limit, step_start=0.1, use_cg=False, verbose=True):
    """LP-based iterative optimization with adaptive step size."""
    n = len(f_init)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_score = score_val(f)
    t0 = time.monotonic()
    t = step_start
    stuck = 0
    active_set = None

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b_seq = fast_convolve(f)
        M = float(b_seq.max())

        if use_cg:
            b = solve_lp_cg(f, M, max_rounds=20, init_active=active_set)
        else:
            b = solve_lp_full(f, M)
        b = normalize(b)

        # Track active constraints for CG
        conv_fb = cross_convolve(f, b)
        active_set = list(np.where(conv_fb > M * 0.95)[0])

        # Try a few step sizes
        improved = False
        for trial_t in [t, 0.6 * t, 0.3 * t, 0.15]:
            f_trial = (1 - trial_t) * f + trial_t * b
            f_trial = normalize(f_trial)
            sc = score_val(f_trial)
            if sc < best_score - 1e-13:
                f = f_trial
                best_score = sc
                best_f = f.copy()
                improved = True
                t = min(t * 1.05, 0.4)
                stuck = 0
                break

        if not improved:
            stuck += 1
            t = max(t * 0.85, 0.02)
            if stuck > 40:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.005
                f = normalize(best_f + noise)
                stuck = 0
                t = 0.05

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={it} ({elapsed:.0f}s): cur={score_val(f):.7f} "
                  f"best={best_score:.7f} M={M:.8f} t={t:.4f} stuck={stuck}")

    return best_f, best_score


def mirror_descent(f_init, time_limit, T_start=0.01, T_end=0.0005,
                   eta_start=0.02, eta_end=0.0005, max_iters=20000):
    """Adaptive mirror descent with temperature annealing."""
    n = len(f_init)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_score = score_val(f)
    t0 = time.monotonic()
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(f)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()

        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        F = rfft(f, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)
        grad = 4.0 * n * correl[:n]

        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
        else:
            f = f / s

        if stuck > 300:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.015
            f = normalize(best_f + noise)
            stuck = 0
            eta = min(eta * 3.0, eta_start * 0.5)

        if it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} ({elapsed:.0f}s): T={T:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with proportional redistribution."""
    n = len(f_init)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_score = score_val(f)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 80
    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_score
            for fac in factors:
                f_trial = f.copy()
                f_trial[i] = f_trial[i] * fac
                f_trial = normalize(f_trial)
                sc = score_val(f_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    f = f_trial
                    best_f = f.copy()
                    improved_pass = True
                    break
        if not improved_pass:
            no_improve += 1
        if no_improve % 20 == 0:
            print(f"    CD: best={best_score:.7f} no_improve={no_improve}")

    return best_f, best_score


def peak_shave(f_init, time_limit):
    """Targeted peak shaving: reduce contributions to top peaks."""
    n = len(f_init)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_score = score_val(f)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        b_work = b.copy()
        b_work[0] = -1
        b_work[-1] = -1
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[peak_k - idx]
        top = np.argsort(contribs)[::-1][:6]

        improved = False
        for ci in top:
            i = int(idx[ci])
            if f[i] < 1e-14:
                continue
            for factor in [0.92, 0.85, 0.75, 0.6]:
                f_trial = f.copy()
                f_trial[i] *= factor
                f_trial = normalize(f_trial)
                sc = score_val(f_trial)
                if sc < best_score - 1e-14:
                    f = f_trial
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try two-coordinate reduction
            for ci in top[:3]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or f[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    f_trial = f.copy()
                    f_trial[i] *= f1
                    f_trial[j] *= f1
                    f_trial = normalize(f_trial)
                    sc = score_val(f_trial)
                    if sc < best_score - 1e-14:
                        f = f_trial
                        best_score = sc
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    return best_f, best_score


def lbgfs_log_space(f_init, time_limit, T=50, maxiter=5000):
    """L-BFGS optimization in log-space with softmax smoothing."""
    n = len(f_init)
    f = normalize(f_init)
    z0 = np.log(np.maximum(f, 1e-15))
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def log_func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        f_local = softmax_fn(z)
        b = fast_convolve(f_local)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse

    def log_grad(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        f_local = softmax_fn(z)
        b = fast_convolve(f_local)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        F = rfft(f_local, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)
        grad_a = 4.0 * n * correl[:n]
        mean_grad = np.dot(f_local, grad_a)
        return f_local * (grad_a - mean_grad)

    def softmax_fn(z):
        z_s = z - z.max()
        e = np.exp(z_s)
        s = e.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return e / s

    try:
        res = minimize(fun=log_func, x0=z0, jac=log_grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'disp': False})
        z_opt = res.x
        return softmax_fn(z_opt)
    except Exception:
        return f_init.copy()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def make_initializations(n, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(("parent", normalize(parent.copy())))

    # Linear ramp (decreasing)
    x = np.linspace(1, 0.1, n)
    inits.append(("ramp", normalize(x)))

    # Quadratic decreasing
    x = (np.linspace(1, 0.05, n)) ** 2
    inits.append(("quad", normalize(x)))

    # Exponential decay
    x = np.exp(-1.5 * np.linspace(0, 1, n))
    inits.append(("exp", normalize(x)))

    # Sine envelope
    x = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    x = np.maximum(x, 0)
    inits.append(("sine", normalize(x)))

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Triangular (peak in middle)
    x = 1.0 - 2.0 * np.abs(np.linspace(-0.5, 0.5, n))
    inits.append(("tri", normalize(x)))

    # Skewed: more mass on left
    x = (1.0 - np.linspace(0, 1, n)) ** 0.5
    inits.append(("sqrt_skew", normalize(x)))

    return inits
