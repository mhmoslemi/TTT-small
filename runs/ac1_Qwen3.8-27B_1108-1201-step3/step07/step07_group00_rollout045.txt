"""
Optimized step-function search using cutting-plane LP refinement, spectral flattening,
and multi-length exploration. Key improvements over prior approach:
- Cutting-plane constraint selection (adaptive, more efficient than fixed top-K)
- Spectral flattening as a complementary fast refinement
- Multi-length exploration with interpolation
- Aggressive line search with backtracking
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_auto_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0) * (target_sum / max(v[0], 1e-30))])
    # Project onto {x >= 0, sum(x) = 1} then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        proj = np.ones(n) / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s < 1e-30:
        proj = np.ones(n) / n
        s = 1.0
    return proj * (target_sum / s)


def solve_direction_lp(f, n, M, active_idx, solver='SCS', eps=1e-8):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_idx.
    Returns g (array of length n) or None on failure.
    """
    K = len(active_idx)
    if K == 0:
        return np.ones(n) / n

    # Build constraint matrix A (K x n)
    A = np.zeros((K, n))
    for row_i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver, max_iters=500000, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def cutting_plane_refine(f_init, time_limit, max_active=300, label="cp"):
    """
    Cutting-plane LP refinement. Start with few constraints, add violated ones.
    Returns (best_f, best_obj).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_auto_convolve(f).max()
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_auto_convolve(f)
        M = b.max()

        # Start with top-5 peaks
        top_k = min(5, len(b))
        active_idx = np.sort(np.argpartition(b, -top_k)[-top_k:])

        # Cutting plane: iteratively add violated constraints
        g = None
        for cp_iter in range(50):
            g = solve_direction_lp(f, n, M, active_idx)
            if g is None:
                break
            S_g = g.sum()
            if S_g <= 1.0 + 1e-6:
                # Can't improve with these constraints
                break

            # Check violations
            fg = fast_cross_convolve(f, g)
            violations = fg - M
            max_viol = violations.max()
            if max_viol < M * 1e-8 + 1e-12:
                break  # Feasible (within tolerance)

            # Add most violated constraint
            new_idx = int(np.argmax(violations))
            if new_idx not in active_idx:
                active_idx = np.append(active_idx, new_idx)
                active_idx = np.sort(active_idx)
                if len(active_idx) >= max_active:
                    # Keep only the most violated max_active
                    vals = fg[active_idx]
                    keep = np.argsort(vals)[-max_active:]
                    active_idx = np.sort(active_idx[keep])
            else:
                break

        if g is None or g.sum() <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            # Perturb to escape
            rng = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
            f = project_simplex(f + rng.standard_normal(n) * 0.002)
            continue

        # Line search
        h = g / g.sum()
        best_t = 0.0
        best_t_obj = best_obj
        cur_obj = M

        # Geometric line search
        t_candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        for t in t_candidates:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_auto_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 30:
                break
            # Try smaller steps
            for t in [0.005, 0.002, 0.001]:
                f_trial = (1.0 - t) * f + t * h
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_auto_convolve(f_trial).max()
                if obj < best_t_obj:
                    best_t_obj = obj
                    best_t = t
            if best_t == 0.0:
                continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} cp_iter_wall={elapsed:.0f}s")

    return best_f, best_obj


def spectral_flatten(f_init, time_limit, n_iter_max=5000, label="spec"):
    """
    Spectral flattening: iteratively flatten the power spectrum of f.
    Fast O(n log n) per iteration. Complements the LP approach.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_auto_convolve(f).max()
    fft_size = 1 << (2 * n - 2).bit_length()

    no_improve = 0
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT
        F = rfft(f, fft_size)
        # Flatten: replace amplitude with mean (or a softened version)
        amp = np.abs(F)
        phase = np.angle(F)
        # Use a power-law flattening: amp_new = amp^alpha * mean(amp)^(1-alpha)
        alpha = 0.5 + 0.3 * np.sin(it * 0.01)  # varies over time
        alpha = np.clip(alpha, 0.3, 0.8)
        mean_amp = amp.mean()
        new_amp = np.power(amp, alpha) * np.power(mean_amp, 1.0 - alpha)
        # Keep DC component
        new_amp[0] = amp[0]
        F_new = new_amp * np.exp(1j * phase)
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        obj = fast_auto_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
            # Blend with previous best
            f = 0.7 * best_f + 0.3 * f_new
            f = np.maximum(f, 0.0)
            f = f / f.sum()

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.002, label="sg"):
    """Projected subgradient with softmax peak weighting."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_auto_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_auto_convolve(f)
        M = b.max()

        # Softmax peak weights
        temp = 0.0002 + 0.0001 * (it % 10)
        scaled = (b - M) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        # Subgradient computation
        g = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for k in nz:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]

        # Momentum + projected gradient step
        v = 0.9 * v - lr * g
        f_new = project_simplex(f + v)

        obj = fast_auto_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.01)
        else:
            no_improve += 1
            f = 0.9 * f + 0.1 * f_new
            f = project_simplex(f)
            if no_improve > 20:
                lr *= 0.9
                no_improve = 0
            if no_improve > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent with geometric multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_auto_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 2.0]
    improved_any = True

    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        for j in range(0, n, max(1, n // 512)):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_auto_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved_any = True

    return f, best_obj
