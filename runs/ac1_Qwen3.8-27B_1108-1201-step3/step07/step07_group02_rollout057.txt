```python
from scipy.optimize import minimize

def fast_convolve_autocorr(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a, b):
    """Compute full linear convolution a * b using FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def log_sum_exp(x):
    """Stable log-sum-exp."""
    m = x.max()
    return m + np.log(np.exp(x - m).sum())

def smooth_obj_and_grad(z, T):
    """
    Compute the smoothed objective and its gradient w.r.t. z.
    a = softmax(z), obj = (1/T) * log(sum_k exp(T * conv(a,a)[k]))
    """
    a = softmax(z)
    n = len(a)
    b = fast_convolve_autocorr(a)
    Tb = T * b
    LSE = log_sum_exp(Tb)
    obj = LSE / T
    w = np.exp(Tb - LSE)  # softmax weights over convolution coefficients
    
    # Compute gradient of f_T w.r.t. a using FFT
    # grad_a[j] = conv(w, a)[j] + conv(flip(a), w)[j+n-1] for j=0..n-1
    c2 = fast_convolve(w, a)[:n]  # conv(w, a)[j] = sum_d w[j-d] a[d]
    c1_full = fast_convolve(a[::-1], w)
    c1 = c1_full[n-1:2*n-1]  # c1[j] = conv(flip(a), w)[j+n-1]
    
    grad_a = c1 + c2
    
    # Convert to z-gradient: grad_z = a * (grad_a - dot(a, grad_a))
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z
```
