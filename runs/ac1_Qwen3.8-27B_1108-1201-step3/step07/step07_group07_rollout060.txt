import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient_smoothed(a, p):
    """
    Compute exact gradient of log-sum-exp(tau*conv(a,a)) w.r.t. a.
    a: sequence of length n
    p: softmax weights of length 2n-1 (already normalized)
    Returns: gradient of length n

    # Term 1 (cross-correlation): sum_k p[k]*a[k-i]

    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())

def fista_optimize(a0, time_limit, tau_init=10.0, tau_min=50.0, seed=42):
    """
    FISTA (Feng, Li, Arason - accelerated proximal gradient) for minimizing
    max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    
    y = a.copy()  # FISTA momentum variable
    t_k = 1.0  # FISTA momentum parameter
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    tau = tau_init
    lr = 1.0 / tau  # Initial learning rate (Lipschitz constant of gradient is ~ tau * 2 * max(a))
    # Actually, the Lipschitz constant of the smoothed gradient depends on tau and the values of a
    # For safety, use a smaller lr initially
    
    # Estimate Lipschitz constant
    # The gradient involves convolutions, so L ~ tau * 2 * sum(a)^2 / n * something
    # Let's just use backtracking
    
    no_improve = 0
    it = 0
    tau_schedule_decay = 1.0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed objective and gradient
        b = fast_convolve(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Compute gradient
        grad = compute_gradient_smoothed(a, p)
        
        # Backtracking line search
        # Objective: f(a) = (1/tau) * log(sum(exp(tau * b)))
        # We want to ensure f(y - lr*grad) <= f(y) - 0.5*lr*||grad||^2
        # For simplicity, use a fixed-ish lr with occasional adjustment
        
        # The Hessian of the smoothed function involves tau^2 * p * (something)
        # A safe lr is 1 / (tau * 2 * max_conv_value_estimate)
        # Let's use: lr = 1 / (tau * (b_max + 1))
        lr = 1.0 / (tau * (b_max + 1.0) * 2.0)
        
        # FISTA step
        z = y - lr * grad
        a_new = project_simplex(z)
        
        # Evaluate
        obj_new = eval_max_b(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        t_k = t_new
        it += 1
        
        # Adaptive tau: decrease as we converge
        if no_improve > 100:
            tau = min(tau * 1.5, tau_min)
            t_k = 1.0  # Reset momentum
            y = a.copy()
            no_improve = 0
            
        if no_improve > 500:
            # Perturb and restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            y = a.copy()
            t_k = 1.0
            tau = max(tau / 2.0, tau_init)
            no_improve = 0
    
    return best_a, best_obj

def lp_refine(f, max_iters=50, time_limit=60.0, verbose=False):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend with current f to reduce max convolution.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Build LP constraint matrix using sparse-like approach
        # A[k, j] = a[j] if 0 <= k-j < n, else 0
        # For efficiency, build row by row
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]
        
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t
            
            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.3f}, max_b={best_obj:.12f}")
                else:
                    a = best_a.copy()
            else:
                break
                
        except Exception as e:
            if verbose:
                print(f"    LP error: {e}")
            break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0, 5.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                a_tr = a_tr / a_tr.sum()
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    
    return best_a, best_obj

def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_uniform_init(n):
    return np.ones(n) / n

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with adaptive log-sum-exp
smoothing (continuation method), exact FFT-based gradients, LP refinement,
and coordinate descent polish. Key improvements: exact gradient (not subgradient),
FISTA acceleration, continuation schedule, and aggressive LP + CD polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_smoothed(a, p):
    """
    Compute exact gradient of log-sum-exp(tau*conv(a,a)) w.r.t. a.
    grad[i] = sum_k p[k]*(a[k-i] + a[i-k])
    Uses two FFTs for O(n log n) computation.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    # Term 2: (a * p)[i] for i=0..n-1 (linear convolution)
    conv_ap = irfft(A * P, fft_size)
    term2 = conv_ap[:n].copy()
    # Term 1: sum_k p[k]*a[k-i] = (p_rev * a)[2n-2-i]
    P_rev = rfft(p[::-1], fft_size)
    conv_pa = irfft(P_rev * A, fft_size)
    # Index: for i=0..n-1, we need conv_pa[2n-2-i]
    term1 = conv_pa[2 * n - 2 - np.arange(n)]
    return term1 + term2


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def eval_smoothed(a, tau):
    """Compute smoothed objective (1/tau)*log(sum(exp(tau*c)))."""
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    s = exp_c.sum()
    if s < 1e-300:
        return c_max
    return c_max + np.log(s) / tau


def fista_optimize(a0, time_limit, tau_start=5.0, tau_end=200.0, seed=42):
    """
    FISTA with log-sum-exp smoothing and continuation method.
    Minimizes max(conv(a,a)) s.t. a in simplex.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0
    it = 0

    # Continuation: geometric schedule from tau_start to tau_end
    n_iterations_estimate = max(100, int(time_limit * 500))
    tau_ratios = np.geomspace(1.0, tau_end / tau_start, min(n_iterations_estimate, 20000))
    tau_idx = 0

    while time.monotonic() - t0 < time_limit:
        batch = min(50, len(tau_ratios) - tau_idx)
        if batch <= 0:
            break
        tau = tau_start * tau_ratios[tau_idx]
        tau_idx += 1

        b = fast_convolve(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        grad = compute_gradient_smoothed(a, p)

        # Lipschitz estimate: L = 2*tau (conservative but safe)
        lr = 1.0 / (2.0 * tau)

        # Backtracking (max 5)
        f_curr = eval_smoothed(y, tau)
        for bt in range(6):
            z = y - lr * grad
            a_new = project_simplex(z)
            f_new = eval_smoothed(a_new, tau)
            if f_new <= f_curr - 0.4 * lr * np.dot(grad, grad):
                break
            lr *= 0.5

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_k = t_new

        # Evaluate true objective periodically
        if it % 10 == 0:
            obj = eval_max_b(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        it += 1

        if no_improve > 200:
            # Restart with perturbation
            pert = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + pert)
            y = a.copy()
            t_k = 1.0
            no_improve = 0

    # Final check
    obj = eval_max_b(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def lp_refine(f, max_iters=40, time_limit=60.0, verbose=False):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend with current f to reduce max convolution.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Build constraint matrix
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]

        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Line search with more granular steps
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2,
                            0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.3f}, max_b={best_obj:.12f}")
                else:
                    a = best_a.copy()
            else:
                break

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex with FFT-evaluated objective."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95,
                                1.05, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                a_tr = a_tr / a_tr.sum()
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

"""

smoothing (continuation method), exact FFT-based gradients, LP refinement,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_smoothed(a, p):
    """
    Compute exact gradient of log-sum-exp(tau*conv(a,a)) w.r.t. a.
    grad[i] = sum_k p[k]*(a[k-i] + a[i-k])
    Uses two FFTs for O(n log n) computation.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    # Term 2: (a * p)[i] for i=0..n-1 (linear convolution)
    conv_ap = irfft(A * P, fft_size)
    term2 = conv_ap[:n].copy()
    # Term 1: sum_k p[k]*a[k-i] = (p_rev * a)[2n-2-i]
    P_rev = rfft(p[::-1], fft_size)
    conv_pa = irfft(P_rev * A, fft_size)
    term1 = conv_pa[2 * n - 2 - np.arange(n)]
    return term1 + term2


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def eval_smoothed(a, tau):
    """Compute smoothed objective (1/tau)*log(sum(exp(tau*c)))."""
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    s = exp_c.sum()
    if s < 1e-300:
        return c_max
    return c_max + np.log(s) / tau

def lp_refine(f, max_iters=40, time_limit=60.0, verbose=False):
