Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP-based fictitious play with mirror descent acceleration
2. L-BFGS refinement with adaptive temperature schedule
3. Coordinate descent with peak-focused updates

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b (nonneg, sum=1) minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def fictitious_play(n, time_limit, initial=None, solver=None):
    """LP-based fictitious play for auto-convolution minimization."""
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
    
    a_avg = a.copy().astype(np.float64)
    b_avg = a.copy().astype(np.float64)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0
    stagnation = 0
    
    while monotonic() - t0 < time_limit:
        iters += 1
        
        # Solve LP best response for b given a
        b_new = solve_lp_best_response(a, solver)
        if b_new is None:
            b_new = a  # fallback
        
        # Solve LP best response for a given b
        a_new = solve_lp_best_response(b_new, solver)
        if a_new is None:
            a_new = b_new  # fallback
        
        # Update running averages
        a_avg = (iters * a_avg + a_new) / (iters + 1)
        b_avg = (iters * b_avg + b_new) / (iters + 1)
        
        # Evaluate candidates
        candidates = [a_new, b_new, 0.5*(a_new+b_new), a_avg, 0.5*(a_avg+b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            max_conv = fast_convolve(cand).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        
        a = a_new
        b = b_new  # actually b_new
        
        # Perturbation to escape stagnation
        if stagnation > 300:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            stagnation = 0
        
        if iters % 50 == 0:
            elapsed = monotonic() - t0
            print(f"  FP iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, iters


def lbgfs_smooth(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    
    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()
        
        # Gradient computation
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        
        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 15, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    
    return a_out


def coordinate_descent(a0, time_limit, max_outer=100):
    """Coordinate descent on the non-smooth max-convolution objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    for outer in range(max_outer):
        if monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_val = b.max()
        
        # For each coordinate, try to reduce the peak
        improved = False
        for i in range(n):
            if monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try reducing a[i]
            for factor in [0.98, 0.95, 0.9, 0.8]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute the difference
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    old_val = new_val
                    improved = True
                    break
        
        if not improved:
            # Try small random perturbations
            for attempt in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if not improved:
                break
    
    return best_a, best_score


def peak_shave(a0, time_limit):
    """Aggressive peak shaving focused on top contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0
    
    while monotonic() - t0 < time_limit and failed < 100:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        top_contrib = np.argsort(contribs)[::-1][:5]
        improved = False
        
        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            failed += 1
        else:
            failed = 0
    
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_fast(a_fixed):

    c = np.zeros(n + 1)
    c[-1] = 1.0

    T = build_conv_matrix(a_fixed)
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

def fictitious_play(n, time_limit, initial=None, verbose=True):

    best_a = a.copy()
    iters = 0
    stagnation = 0

        iters += 1

        # Solve LP best response for b given a

            a_new = b_new.copy()

        w = 1.0 / (iters + 1)
        a_avg = (1 - w) * a_avg + w * a_new

def lbgfs_smooth(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max (log-sum-exp) smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            return 0.0
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            return np.zeros(n)
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        # Gradient of smoothed objective wrt a
        grad_a = np.zeros(n)
        num_w = len(w)
        for k in range(num_w):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.9):
    """Subgradient method with momentum, focused on top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    step_count = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Focus on top peaks with soft weighting
        top_k = max(10, n // 8)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.001 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Momentum update
        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        score = 2.0 * n * fast_convolve(a_new).max()
        step_count += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.002)
            no_improve = 0

    return best_a, best_score


def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: for each element, find optimal reduction factor
    that minimizes the peak convolution. Uses bisection on the 1D subproblem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    # Precompute which indices contribute to each convolution lag
    # For coordinate i, it affects conv[k] for k in [i, 2*i] (roughly)
    # We'll use a simple approach: try reducing each element and check improvement

    improved_total = 0
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        current_peak = b[peak_k]

        if current_peak <= best_score / (2.0 * n) + 1e-15:
            break

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        # Sort by contribution, try top ones
        top_idx = np.argsort(contribs)[::-1][:8]
        round_improved = False

        for ci in top_idx:
            if monotonic() - t0 > time_limit:
                break
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue

            # Try a range of reduction factors
            best_factor = 1.0
            best_trial_score = best_score
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.82, 0.75]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to maintain sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()

                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_factor = factor

            if best_factor < 1.0 and best_trial_score < best_score - 1e-13:
                # Apply the best factor
                new_val = old_val * best_factor
                diff = old_val - new_val
                a[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum > 1e-15:
                    a[mask] = others + diff * others / others_sum
                a = a / a.sum()
                best_score = best_trial_score
                best_a = a.copy()
                round_improved = True
                improved_total += 1

        # Also try pair reductions
        if not round_improved:
            for ci in top_idx[:4]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            round_improved = True
                            improved_total += 1
                            break
                    if round_improved:
                        break
                if round_improved:
                    break

        # Random perturbation if stuck
        if not round_improved:
            a_trial = a + np.random.standard_normal(n) * 0.0005
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    round_improved = True
                    improved_total += 1

        if not round_improved:
            break

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining fast LP fictitious play,
    L-BFGS smoothing, subgradient descent, coordinate descent,
    and peak shaving. Uses scipy HiGHS for fast LP solves.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Test which LP solver works
    test_a = np.ones(16) / 16
    t_test = monotonic()
    test_b = solve_lp(test_a)
    lp_time = monotonic() - t_test
    print(f"LP solve time (n=16): {lp_time*1000:.1f}ms")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_main = len(best_seq)

    # === Phase 1: Main LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play (n={n_main}) ===")
    rem = deadline - monotonic()
    t_fp1 = min(rem * 0.50, 750.0)
    print(f"  Budget: {t_fp1:.0f}s")
    a_fp, score_fp, iters = fictitious_play(n_main, t_fp1, initial=best_seq, verbose=True)
    best_seq, best_score = check_and_update(a_fp, f"FP1_it{iters}", best_seq, best_score)

    # === Phase 2: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.6f}) ===")
    for T in [20, 50, 150]:
        rem = deadline - monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.15, 60.0)
        print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_smooth(best_seq, t_alloc, T=T)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 3: Subgradient optimization ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 30:
        t_sg = min(rem * 0.20, 100.0)
        print(f"  Budget: {t_sg:.0f}s")
        a_sg, _ = subgradient_optimize(best_seq, t_sg)
        best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 20:
        t_cd = min(rem * 0.20, 100.0)
        print(f"  Budget: {t_cd:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t_cd)
        best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 5: Second FP round with refined start ===
    print(f"\n=== Phase 5: FP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 40:
        t_fp2 = min(rem * 0.30, 200.0)
        print(f"  Budget: {t_fp2:.0f}s")
        a_fp2, _, iters2 = fictitious_play(n_main, t_fp2, initial=best_seq, verbose=True)
        best_seq, best_score = check_and_update(a_fp2, f"FP2_it{iters2}", best_seq, best_score)

        # Quick L-BFGS after FP2
        rem = deadline - monotonic()
        if rem > 15:
            a_lb2 = lbgfs_smooth(best_seq, min(rem * 0.25, 40.0), T=50)
            best_seq, best_score = check_and_update(a_lb2, "lbgfs_T50_2", best_seq, best_score)

    # === Phase 6: Final coordinate descent + peak shave ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 15:
        t_cd2 = min(rem * 0.30, 50.0)
        a_cd2, _ = coordinate_descent(best_seq, t_cd2)
        best_seq, best_score = check_and_update(a_cd2, "coord_descent_2", best_seq, best_score)

        rem = deadline - monotonic()
        if rem > 10:
            a_lb3 = lbgfs_smooth(best_seq, min(rem * 0.40, 25.0), T=30)
            best_seq, best_score = check_and_update(a_lb3, "final_lbgfs", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 0.8, 1.5]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 3072 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic()
            if rem < 40:
                break
            t_alt = min(rem * 0.40, 120.0)
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play(alt_n, t_alt, initial=init_alt, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
