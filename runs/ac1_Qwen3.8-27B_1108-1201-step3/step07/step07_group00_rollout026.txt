```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(f, n):
    """Build the (2n-1) x n sparse constraint matrix."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_optimize_full(f, num_iters=200, solver=cp.PDLP):
    """LP-based optimization using all convolution constraints."""
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for it in range(num_iters):
        b = fast_convolve(f)
        M = b.max()
        
        if M < 1e-15:
            break
        
        # Build constraint matrix
        A = build_constraint_matrix(f, n)
        
        # Solve LP
        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M, g >= 0])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if g.value is None:
            break
        g_val = np.maximum(g.value, 0.0)
        S_g = g_val.sum()
        
        if S_g <= 1.0 + 1e-6:
            # No improvement possible
            break
        
        # Take a step
        h = g_val / S_g
        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1-t) * f + t * h
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        if best_t == 0:
            break
        
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        f /= f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f.copy()
    
    return best_f, best_obj
```

```python
"""
Optimization of step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses dual LP formulation for efficient direction finding, with multi-length
exploration, structured initialization, and coordinate descent polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(f):
    """Compute the objective value for a normalized sequence (sum=1)."""
    b = fast_convolve(f)
    return 2.0 * len(f) * b.max()


def solve_dual_lp(f, M):
    """
    Solve the dual LP:
    minimize M * sum(y)
    subject to: sum_{k in window(i)} y[k] >= 1/f[i] for each i with f[i] > 0
                 y >= 0
    
    Returns the optimal dual value and the dual variables.
    The optimal primal value equals the optimal dual value.
    """
    n = len(f)
    num_lags = 2 * n - 1
    
    # Only include constraints for f[i] > epsilon
    eps = 1e-15
    active = f > eps
    if not active.any():
        return 0.0, None
    
    active_idx = np.where(active)[0]
    num_constraints = len(active_idx)
    
    # Build sparse constraint matrix (num_constraints x num_lags)
    # Row i (for active_idx[i]): sum_{k in window} y[k] >= 1/f[active_idx[i]]
    # window for index j: k from j to j+n-1, clipped to [0, 2n-2]
    
    rows = []
    cols = []
    data = []
    
    for row_i, j in enumerate(active_idx):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        for k in range(k_start, k_end + 1):
            rows.append(row_i)
            cols.append(k)
            data.append(1.0)
    
    A_dual = csr_matrix((data, (rows, cols)), shape=(num_constraints, num_lags))
    b_dual = 1.0 / f[active_idx]
    
    # Solve: minimize M * 1^T y subject to A_dual @ y >= b_dual, y >= 0
    # Convert to standard form: minimize c^T y subject to -A_dual @ y <= -b_dual, y >= 0
    
    y = cp.Variable(num_lags, nonneg=True)
    c = M * np.ones(num_lags)
    objective = cp.Minimize(c @ y)
    constraints = [A_dual @ y >= b_dual]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.PDLP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None
    
    if y.value is None:
        return None, None
    
    obj_val = float(c @ y.value)
    return obj_val, np.maximum(y.value, 0.0)


def lp_direction(f, solver=cp.PDLP):
    """
    Find the LP direction: maximize sum(g) subject to conv constraints.
    Uses the dual for efficiency.
    Returns (g, improvement) where g is the direction (sum=1) or None.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    if M < 1e-15:
        return None, 0.0
    
    # Solve dual to get optimal value
    dual_obj, y_val = solve_dual_lp(f, M, solver)
    
    if dual_obj is None:
        return None, 0.0
    
    # The optimal primal objective is dual_obj (by strong duality)
    # If dual_obj <= 1.0, no improvement (since sum(f)=1 is feasible)
    improvement = dual_obj - 1.0
    
    if improvement < 1e-8:
        return None, improvement
    
    # To recover the primal solution, we need to solve the primal LP
    # Alternatively, use the dual to guide the update
    # For now, let's just use a simple geodesic toward reducing peaks
    
    # Actually, let's solve the primal to get g
    # Primal: maximize sum(g) subject to A @ g <= M, g >= 0
    # where A[k, i] = f[i] for valid (k, i)
    
    # Build A as sparse
    num_lags = 2 * n - 1
    rows = []
    cols = []
    data = []
    for k in range(num_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    
    A = csr_matrix((data, (rows, cols)), shape=(num_lags, n))
    
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, 0.0
    
    if g.value is None:
        return None, 0.0
    
    g_val = np.maximum(g.value, 0.0)
    S = g_val.sum()
    if S < 1e-10:
        return None, 0.0
    
    return g_val / S, S - 1.0


def run_lp_optimize(f_init, time_limit, label="lp", solver=cp.PDLP):
    """Main LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, improvement = lp_direction(f, solver)
        
        if g is None:
            stuck += 1
            if stuck > 20:
                break
            # Try with a slightly relaxed bound
            b = fast_convolve(f)
            M = b.max()
            # Add small epsilon to M
            f_relaxed = f.copy()
            # Just perturb slightly
            rng = np.random.default_rng(it)
            f_perturbed = f + rng.standard_normal(n) * 1e-5
            f_perturbed = np.maximum(f_perturbed, 0)
            f_perturbed /= f_perturbed.sum()
            f = f_perturbed
            continue
        
        # Line search
        best_t = 0.1
        best_t_obj = best_obj
        cur_obj = fast_convolve(f).max()
        
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        # Update
        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent: try small multiplicative changes to each element."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    # Multipliers to try
    multipliers = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = f[j]
            
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_val = f[j]
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = old_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                f[j] = old_val
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
        
        if time.monotonic() - t0 > time_limit:
            break
    
    if time.monotonic() - t0 < time_limit:
        print(f"    {label} final obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
              f"t={time.monotonic()-t0:.0f}s")
    
    return f, best_obj


def subgradient_polish(f_init, time_limit, label="sg"):
    """Subgradient descent with momentum for fine-tuning."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = 0.001
    momentum = 0.9
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Find top peaks
        n_top = max(10, n // 10)
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        
        # Compute subgradient: weight on top peaks
        # grad[m] = 2 * sum_{k in top} w_k * f[k-m]
        # Use softmax weights on top peaks
        top_vals = b[top_idx]
        weights = np.exp((top_vals - top_vals.max()) / (0.001 * M + 1e-15))
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            w = weights[idx]
            if w < 1e-12:
                continue
            # grad[m] += 2 * w * f[k-m] for valid m
            # f[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                grad[m] += 2.0 * w * f[k - m]
        
        # Update
        v = momentum * v - lr * grad
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()
        
        f = f_new
        obj = fast_convolve(f).max()
        
        if it % 50 == 0:
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    no_improve = 0
            
            if time.monotonic() - t0 % 200 < 5:
                print(f"    {label} iter={it} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")
            
            if no_improve > 15:
                # Restart with perturbation
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0)
                f /= f.sum()
                v = np.zeros(n)
                no_improve = 0
    
    return best_f, best_obj
```

```python
"""
Optimization of step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP direction finding (primal formulation with sparse constraints),
multi-length exploration, structured initialization, coordinate descent polish,
and stochastic perturbation for fine-tuning.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build sparse (2n-1) x n matrix where A[k,i] = f[i] if valid."""
    num_lags = 2 * n - 1
    rows = []
    cols = []
    data = []
    for k in range(num_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    return csr_matrix((data, (rows, cols)), shape=(num_lags, n))


def lp_solve_direction(f):
    """
    Solve: maximize sum(g) s.t. A@g <= M, g >= 0.
    Returns (g_normalized, improvement) or (None, 0).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, 0.0

    A = build_conv_matrix(f, n)

    # Solve minimize -sum(g) s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=M * np.ones(2 * n - 1),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})

    if not res.success:
        return None, 0.0

    g_val = np.maximum(res.x, 0.0)
    S = g_val.sum()
    if S < 1e-10:
        return None, 0.0

    improvement = S - 1.0
    if improvement < 1e-7:
        return None, improvement

    return g_val / S, improvement


def run_lp_optimize(f_init, time_limit, label="lp"):
    """Iterative LP-based optimization with line search and perturbation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()

    it = 0
    stuck = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, improvement = lp_solve_direction(f)

        if g is None:
            stuck += 1

            # Perturb to escape
            f_pert = f + rng.standard_normal(n) * (0.001 + 0.002 * stuck / 30)
            f_pert = np.maximum(f_pert, 0.0)
            s = f_pert.sum()
            if s > 1e-10:
                f = f_pert / s
            continue

        # Line search with finer grid
        best_t = 0.1
        best_t_obj = best_obj

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()

    pass_count = 0
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = f[j]
            old_val = f[j]

                new_val = f[j] * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = old_val
                    continue

                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                f[j] = old_val

                f[j] = best_j_val

                if s > 1e-10:
```
