Now, for the main optimization algorithm, I'll use:

1. **Soft-max gradient descent with momentum**:
   - At each step, compute the soft-max weights for the auto-correlation peaks.
   - Compute the gradient.
   - Project onto the tangent space of the simplex (subtract the mean to maintain sum = 1, and handle the non-negativity constraint via projection).
   - Take a step with backtracking line search.

2. **Adaptive temperature**: Start with a low beta (smooth approximation) and gradually increase it (sharper approximation of the true max).

3. **Periodic LP refinement**: Every N iterations, solve the LP to get an exact direction and take a larger step.

4. **Multi-length exploration**: Periodically try different sequence lengths.

"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with momentum, adaptive temperature, periodic LP refinement,
and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, weights):
    """
    Compute grad[j] = 2 * sum_m a[m] * weights[m+j]
    using FFT-based convolution: grad[j] = 2 * conv(reversed(a), weights)[n-1+j]
    """
    n = len(a)
    b = a[::-1]  # reversed
    # conv(b, weights) has length n + len(weights) - 1 = n + (2n-1) - 1 = 3n-2
    size = n + len(weights) - 1
    fft_size = 1 << (size - 1).bit_length()
    B = rfft(b, fft_size)
    W = rfft(weights, fft_size)
    conv_bw = irfft(B * W, fft_size)[:size]
    # grad[j] = 2 * conv_bw[n-1+j] for j = 0..n-1
    return 2.0 * conv_bw[n - 1 : 2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_weights(conv_vals, beta):
    """Compute softmax(beta * (conv_vals - max)) weights."""
    shifted = conv_vals - conv_vals.max()
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        w = np.zeros(len(conv_vals))
        w[conv_vals.argmax()] = 1.0
        return w
    return exp_vals / total


def objective_value(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def run_softmax_gradient(f_init, time_limit, beta_init=50.0, beta_max=5000.0,
                         lr_init=0.01, momentum=0.9, label="softmax"):
    """
    Soft-max gradient descent with momentum and adaptive parameters.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    # Normalize to sum = 1
    s = f.sum()
    if s > 1e-10:
        f /= s
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute auto-correlation and soft-max weights
        conv_vals = fast_convolve(f)
        current_max = conv_vals.max()
        weights = soft_max_weights(conv_vals, beta)
        
        # Compute gradient
        grad = compute_gradient_fft(f, weights)
        
        # Project gradient onto tangent space of simplex (subtract component in direction of 1)
        grad_tangent = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad_tangent
        
        # Take step and project onto simplex
        f_new = project_simplex(f + v)
        
        # Backtracking line search
        best_t = 1.0
        best_f_trial = f_new
        base_obj = fast_convolve(f).max()
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_trial = project_simplex((1-t)*f + t*f_new)
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < base_obj:
                best_t = t
                best_f_trial = f_trial
                break
        
        f = best_f_trial
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            # Slightly increase learning rate on improvement
            lr = min(lr * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
                if no_improve > 100:
                    # Restart with noise
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    v = np.zeros(n)
                    no_improve = 0
            
        # Adapt beta: increase it over time for sharper max approximation
        if it % 100 == 0:
            beta = min(beta * 1.2, beta_max)
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} beta={beta:.0f} lr={lr:.5f} t={elapsed:.0f}s")
        
        if no_improve > 200:
            break

    return best_f, best_obj


def run_lp_refinement(f_init, time_limit, top_k=100, label="lp"):

    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        K = len(top_idx)

        A = np.zeros((K, n))
        for row_i in range(K):
            k = top_idx[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            break

        if g.value is None:
            break
        g_val = np.maximum(g.value, 0.0)
        if np.any(np.isnan(g_val)):
            break

        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 5:
                break
            continue

        h = g_val / S_g
        best_t = 0.1
        best_t_obj = fast_convolve(f).max()

        for t in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def try_length(f_base, target_n):
    """Interpolate a sequence to a new length."""
    n_base = len(f_base)
    old_x = np.linspace(0.0, 1.0, n_base)
    new_x = np.linspace(0.0, 1.0, target_n)
    f = np.interp(new_x, old_x, f_base)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = np.ones(target_n) / target_n
    return f

"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines soft-max gradient descent with momentum, adaptive temperature, periodic LP
refinement, multi-length exploration, and annealing-based restarts.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, weights):
    """
    Compute grad[j] = 2 * sum_m a[m] * weights[m+j]
    via FFT: grad[j] = 2 * conv(reversed(a), weights)[n-1+j]
    """
    n = len(a)
    b = a[::-1]
    size = n + len(weights) - 1
    fft_size = 1 << (size - 1).bit_length()
    B = rfft(b, fft_size)
    W = rfft(weights, fft_size)
    conv_bw = irfft(B * W, fft_size)[:size]
    return 2.0 * conv_bw[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_weights(conv_vals, beta):
    """Compute softmax(beta * (conv_vals - max)) normalized weights."""
    shifted = conv_vals - conv_vals.max()
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        w = np.zeros(len(conv_vals))
        w[conv_vals.argmax()] = 1.0
        return w
    return exp_vals / total

    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, obj5 = run_lp_refinement(best_seq, time_limit=t5, top_k=150,
                                     label="lp5")
        s5 = f5.sum()
        if s5 > 1e-10:
            f5 /= s5
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp5]: {score5:.6f}")

    # === Phase 6: Final high-precision gradient polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.40, 300.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_softmax_gradient(best_seq, time_limit=t6, beta_init=1000,
                                        beta_max=50000, lr_init=0.0008,
                                        momentum=0.95, label="polish")
        s6 = f6.sum()
        if s6 > 1e-10:
            f6 /= s6
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [polish]: {score6:.6f}")

    # === Phase 7: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 7: Final LP burst (best={best_score:.6f}) ===")
        print(f"  Budget: {rem:.0f}s")
        f7, obj7 = run_lp_refinement(best_seq, time_limit=rem, top_k=200,
                                     label="lp7")
        s7 = f7.sum()
        if s7 > 1e-10:
            f7 /= s7
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp7]: {score7:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
