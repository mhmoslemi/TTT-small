def _interpolate(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = _normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    return _normalize(f_new)

def _init_fourier(n, rng):
    """Random Fourier-based initialization."""
    n_h = int(rng.integers(3, 20))
    amps = rng.random(n_h) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_h)
    freqs = np.arange(1, n_h + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_h):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    return _normalize(a)

def _init_random(n, rng):
    """Random initialization."""
    a = rng.random(n) + 0.1
    return _normalize(a)

def _init_block(n, rng):
    """Block-structured initialization."""
    n_b = int(rng.integers(4, 30))
    bs = max(2, n // n_b)
    amps = rng.random(n_b) + 0.2
    a = np.zeros(n)
    for b in range(n_b):
        s = b * bs
        e = min(s + bs, n)
        a[s:e] = amps[b]
    # Smooth
    k = min(5, bs)
    if k > 1:
        ker = np.ones(k) / k
        a = np.convolve(a, ker, mode='same')
    return _normalize(a)

Hybrid optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Sparse-matrix LP-based fictitious play with cached CSR structure
2. Multi-start diverse initialization with quick local search
3. Adaptive coordinate descent + peak shaving + simulated annealing
4. Multi-length exploration with interpolation
5. Final intensive polishing

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import operator


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _precompute_csr(n):
    """Precompute CSR structure and a-index array for convolution matrix."""
    rows = []
    cols = []
    a_idx = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            a_idx.append(i)
    rows_arr = np.array(rows, dtype=np.int32)
    cols_arr = np.array(cols, dtype=np.int32)
    a_idx_arr = np.array(a_idx, dtype=np.int32)
    T = csr_matrix((np.ones(len(rows_arr)), (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    return a_idx_arr, T.indptr.copy(), T.indices.copy()


def _make_conv_matrix(n, a, csr_cache):
    """Build sparse conv matrix quickly using precomputed CSR."""
    a_idx, indptr, indices = csr_cache
    data = np.asarray(a[a_idx], dtype=np.float64)
    return csr_matrix((data, indptr, indices), shape=(2 * n - 1, n))


def _get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _solver_found
    if '_solver_found' in globals() and _solver_found is not None:
        return _solver_found
    for name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_found = name
                return name
        except Exception:
            continue
    _solver_found = cp.ECOS
    return cp.ECOS


def _solve_lp(a_fixed, solver, csr_cache, n):
    """Solve LP: min M s.t. conv(a_fixed,b)_k <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _make_conv_matrix(n, a_fixed, csr_cache)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    success = False
    for sol in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                success = True
                break
        except Exception:
            continue
    if not success:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def _norm(a):
    """Clip non-negative and normalize to sum 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _fictitious_play(n, time_limit, initial, solver, verbose=False):
    """LP-based fictitious play with cached CSR structure."""
    csr_cache = _precompute_csr(n)
    t0 = time.monotonic()
    if initial is not None:
        a = _norm(initial)
    else:
        a = np.ones(n, dtype=np.float64) / n
    b = a.copy()
    best_a = a.copy()
    best_score = _score(a)
    a_avg = a.copy().astype(np.float64)
    iters = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = _solve_lp(a, solver, csr_cache, n)
        if b_new is not None:
            b = b_new
        a_new, _ = _solve_lp(b, solver, csr_cache, n)
        if a_new is not None:
            a = a_new
        a_avg = (iters * a_avg + a) / (iters + 1.0)

        improved = False
        candidates = [a, b, 0.5 * (a + b), a_avg, 0.7 * a + 0.3 * b, 0.3 * a + 0.7 * b]
        for cand in candidates:
            c = _norm(cand)
            sc = _score(c)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = c.copy()
                improved = True

        if not improved:
            stagnant += 1
        else:
            stagnant = 0

        if stagnant > 80:
            # Perturb to escape
            rng_p = np.random.default_rng(iters * 31 + 7)
            pert = best_a + rng_p.standard_normal(n) * 0.03
            a = _norm(pert)
            b = _norm(pert + rng_p.standard_normal(n) * 0.02)
            stagnant = 0

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f} stag={stagnant}")

    return best_a, best_score, iters


def _coord_descent(a0, time_limit, n_sweeps=400, verbose=False):
    """Coordinate descent with coarse-to-fine factor search."""
    n = len(a0)
    a = _norm(a0)
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    coarse = [0.4, 0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.6, 2.0]
    fine = [0.88, 0.94, 0.97, 0.99, 1.01, 1.03, 1.06, 1.12]
    very_fine = [0.96, 0.98, 0.99, 1.01, 1.02, 1.04]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        if sweep < 30:
            factors = coarse
        elif sweep < 100:
            factors = fine
        else:
            factors = very_fine

        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-13:
                continue
            for factor in factors:
                a[i] = old_val * factor
                a = _norm(a)
                sc = _score(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old_val
                    a = _norm(a)

        if not improved and sweep > 30:
            # Try random restarts within local neighborhood
            any_improve = False
            for _ in range(20):
                i = rng.integers(0, n)
                if a[i] < 1e-13:
                    continue
                for factor in very_fine:
                    a[i] = a[i] * factor
                    a = _norm(a)
                    sc = _score(a)
                    if sc < best_score - 1e-15:
                        best_score = sc
                        best_a = a.copy()
                        any_improve = True
                    else:
                        a[i] *= 1.0 / factor
                        a = _norm(a)
                if any_improve:
                    break
            if not any_improve and sweep > 60:
                break

    return best_a, best_score


def _peak_shave(a0, time_limit):
    """Target highest peaks and reduce contributing coordinates."""
    n = len(a0)
    a = _norm(a0)
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        # Find top 3 peaks
        top_k = min(3, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        any_improve = False

        for pk in peaks:
            pk = int(pk)
            i_lo = max(0, pk - n + 1)
            i_hi = min(n - 1, pk)
            idx = np.arange(i_lo, i_hi + 1)
            contribs = a[idx] * a[pk - idx]
            top_c = np.argsort(contribs)[::-1][:10]

            for ci in top_c:
                i = int(idx[ci])
                j = pk - i
                if a[i] < 1e-13:
                    continue
                for factor in [0.8, 0.65, 0.5]:
                    a_t = a.copy()
                    a_t[i] *= factor
                    a_t = _norm(a_t)
                    sc = _score(a_t)
                    if sc < best_score - 1e-14:
                        a = a_t
                        best_score = sc
                        best_a = a.copy()
                        any_improve = True
                        break
                if any_improve:
                    break

                if i != j and a[j] >= 1e-13:
                    for f1 in [0.8, 0.6]:
                        for f2 in [0.8, 0.6]:
                            a_t = a.copy()
                            a_t[i] *= f1
                            a_t[j] *= f2
                            a_t = _norm(a_t)
                            sc = _score(a_t)
                            if sc < best_score - 1e-14:
                                a = a_t
                                best_score = sc
                                best_a = a.copy()
                                any_improve = True
                                break
                        if any_improve:
                            break
                if any_improve:
                    break

        if not any_improve:
            break

    return best_a, best_score


def _interp(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    fn = _norm(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, fn)
    return _norm(out)


def _init_fourier(n, rng):
    """Fourier-based random initialization."""
    n_h = int(rng.integers(3, 25))
    amps = rng.random(n_h) * 0.45
    phases = rng.uniform(0, 2 * np.pi, n_h)
    freqs = np.arange(1, n_h + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_h):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    return _norm(a)


def _init_block(n, rng):
    """Block-structured initialization with smoothing."""
    n_b = int(rng.integers(5, 35))
    bs = max(2, n // n_b)
    amps = rng.random(n_b) + 0.15
    a = np.zeros(n)
    for b in range(n_b):
        s = b * bs
        e = min(s + bs, n)
        a[s:e] = amps[b]
    k = min(5, bs)
    if k > 1:
        ker = np.ones(k) / k
        a = np.convolve(a, ker, mode='same')
    return _norm(a)


def _init_random(n, rng):
    """Simple random initialization."""
    a = rng.random(n) + 0.1
    return _norm(a)


def _check(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = _norm(a)
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    return best_seq, best_score


def _simulated_anneal(a0, time_limit, verbose=False):
    """Simulated annealing with periodic local search."""
    n = len(a0)
    a = _norm(a0)
    best = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    T = 0.015
    last_ls = t0

    while time.monotonic() - t0 < time_limit:
        n_pert = int(rng.integers(2, 12))
        idx_p = rng.choice(n, size=n_pert, replace=False)
        a_t = a.copy()
        for i in idx_p:
            a_t[i] *= rng.uniform(0.4, 1.6)
        a_t = _norm(a_t)
        sc = _score(a_t)

        if sc < best_score - 1e-13:
            best_score = sc
            best = a_t.copy()
            a = a_t.copy()
            T = max(T * 0.995, 0.0005)
        elif rng.random() < np.exp(-(sc - best_score) / max(T, 1e-8)):
            a = a_t.copy()

        T *= 0.99985

        # Periodic local search
        if time.monotonic() - last_ls > 8.0:
            last_ls = time.monotonic()
            a_ls, s_ls = _coord_descent(best, 2.5, n_sweeps=30)
            if s_ls < best_score:
                best_score = s_ls
                best = a_ls.copy()
                a = a_ls.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 20 == 0 and int(elapsed) > 0:
                print(f"    SA ({elapsed:.0f}s): best={best_score:.8f} T={T:.6f}")

    return best, best_score


def propose_candidate():
    """
    Multi-phase optimizer: diverse init -> LP fictitious play ->
    coordinate descent -> peak shaving -> simulated annealing -> final polish.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(7)
    solver = _get_lp_solver()
    print(f"Solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            psc = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(psc):
                best_seq = parent.copy()
                best_score = psc
                print(f"Parent: n={len(parent)}, score={psc:.8f}")
    except Exception:
        pass

    # Phase 1: Diverse initialization (3 min)
    print("\n[P1] Diverse Init (3 min)")
    t1 = time.monotonic()
    n_list = [512, 640, 768, 896, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_list:
            n_list.insert(0, np_)

    cands = []
    for n in n_list:
        if time.monotonic() - t1 > 170:
            break
        inits = []
        if parent is not None:
            inits.append(_interp(parent, n))
        inits.append(_init_fourier(n, rng))
        inits.append(_init_block(n, rng))
        inits.append(_init_random(n, rng))

        for init in inits:
            a_o, s_o = _coord_descent(init, 3.0, n_sweeps=15)
            cands.append((s_o, a_o, n))

    cands.sort(key=operator.itemgetter(0))
    print(f"  {len(cands)} cands, best={cands[0][0]:.8f}")
    for s_c, a_c, n_c in cands[:6]:
        best_seq, best_score = _check(a_c, best_seq, best_score, f"p1_{n_c}")

    # Phase 2: LP Fictitious Play (11 min)
    print(f"\n[P2] FP (best={best_score:.8f})")
    t2 = time.monotonic()
    fp_lens = []
    if best_seq is not None:
        fp_lens.append(len(best_seq))
    for nx in [768, 1024, 640, 896]:
        if nx not in fp_lens:
            fp_lens.append(nx)
    fp_lens = fp_lens[:4]
    time_per_fp = 660.0 / len(fp_lens)

    for n_fp in fp_lens:
        rem = deadline - time.monotonic() - 500
        if rem < 60:
            break
        t_alloc = min(time_per_fp, rem * 0.9)
        if t_alloc < 30:
            continue
        print(f"  FP n={n_fp} budget={t_alloc:.0f}s")
        init_fp = _interp(best_seq, n_fp) if best_seq is not None else None
        a_fp, s_fp, it = _fictitious_play(n_fp, t_alloc, init_fp, solver, verbose=True)
        best_seq, best_score = _check(a_fp, best_seq, best_score, f"fp_{n_fp}")

        # Quick polish
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            a_p, s_p = _coord_descent(a_fp, min(rem * 0.3, 60.0), n_sweeps=100)
            best_seq, best_score = _check(a_p, best_seq, best_score, f"pol_{n_fp}")

    # Phase 3: Intensive local search (4 min)
    print(f"\n[P3] Local Search (best={best_score:.8f})")
    if best_seq is not None:
        # Coordinate descent
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_cd = min(rem * 0.35, 150.0)
            print(f"  CD budget={t_cd:.0f}s")
            a_cd, s_cd = _coord_descent(best_seq, t_cd, n_sweeps=500)
            best_seq, best_score = _check(a_cd, best_seq, best_score, "cd")

        # Peak shave
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_ps = min(rem * 0.35, 100.0)
            print(f"  PS budget={t_ps:.0f}s")
            a_ps, s_ps = _peak_shave(best_seq, t_ps)
            best_seq, best_score = _check(a_ps, best_seq, best_score, "ps")

        # More CD after peak shave
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            a_cd2, s_cd2 = _coord_descent(best_seq, min(rem * 0.4, 60.0), n_sweeps=200)
            best_seq, best_score = _check(a_cd2, best_seq, best_score, "cd2")

        # Simulated annealing
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_sa = min(rem * 0.6, 120.0)
            print(f"  SA budget={t_sa:.0f}s")
            a_sa, s_sa = _simulated_anneal(best_seq, t_sa, verbose=True)
            best_seq, best_score = _check(a_sa, best_seq, best_score, "sa")

            # Polish after SA
            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_sp, s_sp = _coord_descent(best_seq, min(rem * 0.5, 40.0), n_sweeps=100)
                best_seq, best_score = _check(a_sp, best_seq, best_score, "sa_pol")

    # Phase 4: Try alternative lengths
    print(f"\n[P4] Alt Lengths (best={best_score:.8f})")
    if best_seq is not None:
        n_b = len(best_seq)
        alt_ns = []
        for f in [0.75, 1.33, 1.5]:
            an = int(n_b * f)
            if 256 <= an <= 2560 and an != n_b:
                alt_ns.append(an)
        for an in alt_ns[:2]:
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            t_alt = min(rem * 0.5, 90.0)
            init_alt = _interp(best_seq, an)
            a_alt, s_alt, _ = _fictitious_play(an, t_alt, init_alt, solver, verbose=False)
            best_seq, best_score = _check(a_alt, best_seq, best_score, f"alt_{an}")

    # Phase 5: Final polish
    print(f"\n[P5] Final (best={best_score:.8f})")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 8
        if rem > 10:
            a_f, s_f = _coord_descent(best_seq, min(rem * 0.6, 40.0), n_sweeps=200)
            best_seq, best_score = _check(a_f, best_seq, best_score, "fin_cd")
        rem = deadline - time.monotonic() - 4
        if rem > 8:
            a_f2, _ = _peak_shave(best_seq, min(rem * 0.6, 20.0))
            best_seq, best_score = _check(a_f2, best_seq, best_score, "fin_ps")

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
