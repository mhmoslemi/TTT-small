"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature gradient descent
with FFT-based gradient computation, multiple diverse restarts, and peak-targeting
heuristics. Key improvements over previous:
1. Exact gradient via cross-correlation FFT (4n * (w ⋆ a))
2. Proper backtracking line search

4. Adaptive temperature with restarts
5. Peak-targeting moves to directly attack the largest autocorrelation peak
6. Final coordinate descent polish

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the non-negative simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Simple projection: clip and normalize (approximate but effective for small steps)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def project_simplex_exact(v):
    """Exact projection onto non-negative simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def grad_smoothed(a, conv, T):
    """
    Compute gradient of 2n * soft_T(conv) where soft_T = T*logsumexp(x/T).
    = 4n * (w ⋆ a) where w = softmax(conv/T)
    Using cross-correlation via FFT.
    """
    n = len(a)
    size = 2 * n - 1
    # softmax(conv / T)
    shifted = conv / T
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(size) / size
    else:
        w /= s

    # Cross-correlation: (w ⋆ a)[j] = sum_k w[k] * a[k-j]
    # FFT size >= (2n-1) + n - 1 = 3n-2
    P = 1 << (3 * n - 2).bit_length()
    W = rfft(w, P)
    A = rfft(a, P)
    cc = irfft(np.conj(W) * A, P)
    # We need cc[0:n]
    return 4.0 * n * cc[:n]


def smooth_obj_value(conv, T):
    """Compute 2n * T * logsumexp(conv/T)."""
    n = len(conv) // 2 + 1  # not used directly
    shifted = conv / T
    max_s = shifted.max()
    lse = max_s + np.log(np.exp(shifted - max_s).sum())
    # 2n * soft_T(conv) where soft_T = T * logsumexp(conv/T)
    # But we need to know n. Let's pass it.
    return lse  # we'll multiply by appropriate factor outside


def gradient_descent_optimize(a0, time_limit, T_init=50.0, T_min=1.0,
                              decrease_rate=0.999, patience=500, seed=0):
    """
    Adaptive-temperature gradient descent with backtracking line search.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    alpha = 1.0
    no_improve = 0
    rng = np.random.default_rng(seed)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        curr_score = compute_score(a)

        if curr_score < best_score - 1e-12:
            best_score = curr_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        grad = grad_smoothed(a, conv, T)

        # Backtracking line search
        accepted = False
        for backtrack in range(20):
            trial = a - alpha * grad
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                alpha *= 0.5
                continue
            trial = trial / ts
            trial_score = compute_score(trial)
            if trial_score < curr_score - 1e-14:
                a = trial
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            # Try a smaller step or perturb
            alpha = max(alpha * 0.5, 1e-8)
            # Add small noise to escape
            noise = rng.standard_normal(n) * 0.001
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        # Decrease temperature
        T = max(T_min, T * decrease_rate)

        # Periodic restart from best if stuck
        if no_improve > patience:
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            alpha = 1.0
            T = T_init * 0.5
            no_improve = 0

    return best_a, best_score


def peak_targeting(a0, time_limit, seed=0):
    """
    Directly target the largest peak by adjusting contributing coordinates.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))

        # Find contributing pairs for this peak
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Find top contributors
        order = np.argsort(contribs)[::-1]
        improved = False

        for ci in order[:10]:
            i = int(idx[ci])
            j = max_idx - i
            if i == j:
                # Diagonal: reduce a[i]
                for frac in [0.1, 0.3, 0.5]:
                    delta = a[i] * frac
                    trial = a.copy()
                    trial[i] -= delta
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    sc = compute_score(trial)
                    if sc < best_score - 1e-14:
                        a = trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce one of the pair
                for target in [i, j]:
                    other = j if target == i else i
                    if a[target] < 1e-14:
                        continue
                    for frac in [0.1, 0.3, 0.5, 0.7]:
                        delta = a[target] * frac
                        trial = a.copy()
                        trial[target] -= delta
                        # Redistribute to random other positions
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts < 1e-15:
                            continue
                        trial = trial / ts
                        sc = compute_score(trial)
                        if sc < best_score - 1e-14:
                            a = trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Try moving mass from a peak-contributing position to a random position
            # that doesn't contribute to the current peak
            peak_positions = set(int(x) for x in idx)
            # Find positions that don't contribute to this peak
            non_peak = [k for k in range(n) if k not in peak_positions]
            if non_peak:
                src = int(idx[order[0]])
                if a[src] > 1e-8:
                    dst = non_peak[rng.integers(0, len(non_peak))]
                    for frac in [0.05, 0.1, 0.2]:
                        delta = a[src] * frac
                        trial = a.copy()
                        trial[src] -= delta
                        trial[dst] += delta
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts < 1e-15:
                            continue
                        trial = trial / ts
                        sc = compute_score(trial)
                        if sc < best_score - 1e-14:
                            a = trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
            if not improved:
                # Small random perturbation
                a_pert = a + rng.standard_normal(n) * 0.002
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                sc = compute_score(a_pert)
                if sc < best_score:
                    a = a_pert
                    best_score = sc
                    best_a = a.copy()
                else:
                    # Keep trying but with smaller noise
                    pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=0):
    """
    Coordinate descent: for each coordinate, find optimal value (1D search).
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        # Visit coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            # Try adjusting a[i]: reduce it and redistribute
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.05, 0.2, 0.4]:
                delta = a[i] * frac
                # Redistribute to all other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                trial = a.copy()
                trial[i] -= delta
                trial[mask] = others + delta * others / os
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                sc = compute_score(trial)
                if sc < best_score - 1e-15:
                    a = trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                # Also try increasing a[i] (take from others)
                if a[i] < 0.5:
                    delta2 = os * 0.05
                    trial2 = a.copy()
                    trial2[i] += delta2
                    trial2[mask] = others - delta2 * others / os
                    trial2 = np.maximum(trial2, 0)
                    ts2 = trial2.sum()
                    if ts2 > 1e-15:
                        trial2 = trial2 / ts2
                        sc2 = compute_score(trial2)
                        if sc2 < best_score - 1e-15:
                            a = trial2
                            best_score = sc2
                            best_a = a.copy()
                            improved = True
                            break

        if not improved and iters > 10:
            # Perturb to escape
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def make_chirp_init(n, seed=0):
    """Create a chirp-like initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Chirp: frequency increases
    t = 2 * np.pi * (x + 0.5 * x * x) * n * 0.1
    a = 0.5 + 0.3 * np.cos(t)
    a = np.maximum(a, 0)
    # Add small random perturbation
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def make_random_init(n, seed=0):
    """Create a random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    return a


def make_structured_init(n, seed=0):
    """Create a structured initialization based on near-uniform with modulation."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Try different modulations
    k1 = rng.integers(1, 5)
    k2 = rng.integers(1, 5)
    phase1 = rng.random() * 2 * np.pi
    phase2 = rng.random() * 2 * np.pi
    a = 1.0 + 0.2 * np.cos(2 * np.pi * k1 * x + phase1) + 0.1 * np.cos(2 * np.pi * k2 * x + phase2)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def two_loop_optimize(a0, time_limit, seed=0):
    """
    Two-loop optimization: outer loop adjusts temperature, inner loop does GD steps.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 100.0
    alpha = 0.5
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        # Inner loop: fixed T, multiple GD steps
        inner_limit = min(30.0, time_limit - (time.monotonic() - t0))
        t_inner = time.monotonic()
        local_best_score = best_score
        local_best_a = best_a.copy()
        stagnant = 0

        while time.monotonic() - t_inner < inner_limit:
            total_iters += 1
            conv = fast_convolve(a)
            curr_score = compute_score(a)

            if curr_score < best_score - 1e-12:
                best_score = curr_score
                best_a = a.copy()
                stagnant = 0
            elif curr_score < local_best_score:
                local_best_score = curr_score
                local_best_a = a.copy()
            else:
                stagnant += 1

            if stagnant > 200:
                break

            grad = grad_smoothed(a, conv, T)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Line search
            for bt in range(15):
                trial = a - alpha * grad
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    alpha *= 0.5
                    continue
                trial = trial / ts
                sc = compute_score(trial)
                if sc < curr_score - 1e-15:
                    a = trial
                    break
                alpha *= 0.5
            else:
                # No improvement, perturb
                a = a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                alpha = max(alpha * 0.5, 1e-6)

        # Outer loop: decrease T
        if time.monotonic() - t0 > time_limit:
            break
        T = max(0.5, T * 0.7)
        alpha = min(2.0, alpha * 1.2)

        # Occasionally restart from best
        if rng.random() < 0.2:
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score

    for (name, init_a) in inits:
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(t_per_init, rem * 0.3)
        print(f"  GD [{name}], budget={t_alloc:.0f}s")
        # Use two_loop_optimize with different seeds
        a_res, sc_res = two_loop_optimize(init_a, t_alloc, seed=hash(name) % 10000)
        try_candidate(a_res, f"gd_{name}")

    # === Phase 2: Peak targeting on best ===
    print(f"\n=== Phase 2: Peak Targeting (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.3, 200.0)
        print(f"  Peak targeting, budget={t_alloc:.0f}s")
        a_pt, sc_pt = peak_targeting(best_seq, t_alloc, seed=123)
        try_candidate(a_pt, "peak_target")

    # === Phase 3: More GD from best with different params ===
    print(f"\n=== Phase 3: Refined GD (best={best_score:.8f}) ===")
    for T_init_val in [20.0, 5.0, 2.0]:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 120.0)
        print(f"  GD T_init={T_init_val}, budget={t_alloc:.0f}s")
        a_res, sc_res = two_loop_optimize(best_seq, t_alloc, seed=777)
        try_candidate(a_res, f"gd_T{T_init_val}")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.3, 150.0)
        print(f"  CD, budget={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent_polish(best_seq, t_alloc, seed=42)
        try_candidate(a_cd, "cd_polish")

    # === Phase 5: Peak targeting again ===
    print(f"\n=== Phase 5: Peak Targeting 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem * 0.3, 80.0)
        a_pt2, _ = peak_targeting(best_seq, t_alloc, seed=456)
        try_candidate(a_pt2, "peak_target2")

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    nearby_ns = [n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
                 n_parent - 4, n_parent + 4, n_parent - 8, n_parent + 8]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 20:
            break
        t_alloc = min(rem * 0.15, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  GD n={alt_n}, budget={talloc:.0f}s")
        a_res, _ = two_loop_optimize(init_alt, t_alloc, seed=alt_n)
        try_candidate(a_res, f"n{alt_n}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 30:
        # Alternate between GD and peak targeting
        t_chunk = rem / 4
        for i in range(4):
            rem2 = deadline - time.monotonic() - 10
            if rem2 < 5:
                break
            if i % 2 == 0:
                t_alloc = min(t_chunk, rem2 * 0.4)
                a_res, _ = two_loop_optimize(best_seq, t_alloc, seed=1000 + i)
                try_candidate(a_res, f"final_gd_{i}")
            else:
                t_alloc = min(t_chunk, rem2 * 0.4)
                a_res, _ = peak_targeting(best_seq, t_alloc, seed=2000 + i)
                try_candidate(a_res, f"final_pt_{i}")

    # === Phase 8: Final CD ===
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.5, 30.0)
        a_res, _ = coordinate_descent_polish(best_seq, t_alloc, seed=999)
        try_candidate(a_res, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        # Fallback
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
