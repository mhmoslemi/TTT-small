def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, T_matrix, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T_matrix @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def accelerated_fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """Accelerated fictitious play with momentum and adaptive perturbation."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:

        s = a.sum()

        else:
            a = a / s

    T_mat = build_conv_matrix(a)
    a_avg = a.copy().astype(np.float64)
    momentum_vec = np.zeros(n)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0
    stagnation = 0
    last_best_score = best_score
    momentum_decay = 0.85

    while monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best response for b given a

        # Update with momentum
        delta = b_new - a

def lbgfs_anneal(a0, time_limit, T_start=80.0, T_end=0.5, n_anneal_steps=8, maxiter_per_T=2000):
    """L-BFGS refinement with temperature annealing for log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    # Create geometric annealing schedule
    T_values = np.geomspace(T_start, T_end, n_anneal_steps)

    for T in T_values:
        if monotonic() - t0 > time_limit:
            break

        z0 = np.log(a) - np.log(a).mean()
        stopped = [False]

        def func(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            b = fast_convolve(a_local)
            scaled_b = b / T
            lse = T * (scaled_b.max() + np.log(np.exp(scaled_b - scaled_b.max()).sum()))
            return 2.0 * n * lse

        def jac(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros(n)
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            b = fast_convolve(a_local)
            scaled_b = b / T
            exp_b = np.exp(scaled_b - scaled_b.max())
            w = exp_b / exp_b.sum()

            # grad[i] = (1/T) * sum_k w[k] * a[k-i] for valid k
            # Vectorized: W[i,j] = w[i+j], grad = W @ a / T
            idx = np.arange(n)
            W = w[np.ix_(idx, idx + np.arange(n))]
            grad_a = W @ a_local / T
            grad_a *= 2.0 * n

            # Transform to z-space (softmax parameterization)
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - a_local * S)
            return grad_z

        try:
            t_sub = min(60.0, (time_limit - (monotonic() - t0)) / 2)
            res = minimize(
                fun=func, x0=z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_T, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': maxiter_per_T * 10, 'disp': False}
            )
            if not stopped[0]:
                z_final = res.x
                z_shifted = z_final - z_final.max()
                a_new = np.exp(z_shifted)
                a_new = a_new / a_new.sum()
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

        a = np.maximum(a, 1e-30)
        a = a / a.sum()

    return best_a, best_score

def peak_shave(a0, time_limit):
    """Aggressive local peak shaving with multiple strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        if peak_val <= best_score / (2.0 * n) + 1e-16:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False

        # Strategy 1: Reduce single contributor and redistribute
        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7, 0.6]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Reduce pair at peak
        if not improved:
            for ci in top_contrib[:5]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.85, 0.8]:
                    for f2 in [0.9, 0.85, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Move mass from peak contributors to low-contribution positions
        if not improved:
            for ci in top_contrib[:3]:
                i = int(idx[ci])
                if a[i] < 1e-10:
                    continue
                # Find positions that don't contribute to the peak
                non_peak = np.array([j for j in range(n) if j not in idx or a[idx].min() > a[j]])
                if len(non_peak) == 0:
                    continue
                target = non_peak[np.argmin(a[non_peak])]
                for frac in [0.1, 0.2, 0.3]:
                    amount = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[target] += amount
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 4: Small random perturbation
        if not improved:
            for attempt in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 100:
                break
        else:
            failed = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting optimization with n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Accelerated Fictitious Play ===
    print(f"\n=== Phase 1: Accelerated FP ===")
    rem = deadline - monotonic() - 600
    if rem > 60:
        t_alloc = min(max(rem * 0.4, 150.0), 450.0)
        print(f"  FP n={n_parent}, budget={t_alloc:.0f}s")
        a_fp, score_fp, iters = accelerated_fictitious_play(
            n_parent, t_alloc, initial=best_seq, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}_it{iters}", best_seq, best_score)

    # === Phase 2: L-BFGS with Temperature Annealing ===
    print(f"\n=== Phase 2: L-BFGS Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 450
        if rem > 30:
            t_alloc = min(max(rem * 0.4, 100.0), 350.0)
            print(f"  L-BFGS n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb, _ = lbgfs_anneal(best_seq, t_alloc, T_start=60.0, T_end=0.3, n_anneal_steps=8)
            best_seq, best_score = check_and_update(a_lb, "lbgfs_anneal", best_seq, best_score)

    # === Phase 3: Dual-Guided Search ===
    print(f"\n=== Phase 3: Dual-Guided Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 20:
            t_alloc = min(max(rem * 0.35, 60.0), 200.0)
            print(f"  DualGuided n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_dg, _ = dual_guided_search(best_seq, t_alloc, solver=solver)
            best_seq, best_score = check_and_update(a_dg, "dual_guided", best_seq, best_score)

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 200
        if rem > 15:
            t_alloc = min(max(rem * 0.4, 50.0), 150.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: Second FP round ===
    print(f"\n=== Phase 5: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic() - 150
        if rem > 40:
            t_alloc = min(max(rem * 0.4, 50.0), 150.0)
            print(f"  FP round 2: n={n_best}, budget={t_alloc:.0f}s")
            a_fp2, _, iters2 = accelerated_fictitious_play(
                n_best, t_alloc, initial=best_seq, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp2, f"FP2_n{n_best}", best_seq, best_score)

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5, 0.6]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 3000 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic() - 80
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = accelerated_fictitious_play(
                alt_n, t_alloc, initial=init_alt, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 20:
            t_final = min(rem * 0.5, 50.0)
            a_final = lbgfs_anneal(best_seq, t_final, T_start=20.0, T_end=0.2, n_anneal_steps=5)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - monotonic() - 5
        if rem > 10:
            t_ps = min(rem * 0.6, 20.0)
            a_ps_final, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
