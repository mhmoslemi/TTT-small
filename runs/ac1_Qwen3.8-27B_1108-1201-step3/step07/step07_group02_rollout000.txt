    # Start with flat spectrum (all ones in frequency domain)
    A = np.ones(n)  # frequency domain
    # Add some random phase to avoid trivial solution
    phase = np.random.default_rng(42).random(n) * 2 * np.pi
    A *= np.exp(1j * phase)

        a = np.fft.ifft(A).real
        # Project to non-negative
        a = np.maximum(a, 0)

def conv_subgradient(a, active_lags):

    for k in active_lags:

def simplex_project(v):

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def iaa_initialize(n, n_iter=5000, seed=42):
    """Iterative Amplitude Adjustment for flat-spectrum initialization."""
    rng = np.random.default_rng(seed)
    # Start with random phase in frequency domain
    A = np.exp(2j * np.pi * rng.random(n))
    
    a = np.ones(n) / n
    for it in range(n_iter):
        # Current spectrum
        A_cur = np.fft.fft(a)
        mag = np.abs(A_cur)
        
        # Target: flat magnitude spectrum
        target_mag = mag.mean()
        
        # New spectrum: keep phase, adjust magnitude toward flat
        phase = np.angle(A_cur)
        new_mag = target_mag + 0.1 * (mag - target_mag)  # gentle pull toward flat
        A_new = new_mag * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a

def lbgfs_optimize(a0, time_limit, T_start, T_end, maxiter):
    """L-BFGS with soft-max smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    def func_and_grad(z, T):
        a = softmax(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * lse
        
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return obj, grad_z
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = max(100, maxiter // n_stages)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=lambda z: func_and_grad(z, T)[0],
                x0=z,
                jac=lambda z: func_and_grad(z, T)[1],
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax(z)

def block_coordinate_descent(a0, time_limit, block_size=4):
    """Block coordinate descent with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    n_blocks = n // block_size
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick a random block
        bi = rng.integers(0, n_blocks)
        start = bi * block_size
        end = start + block_size
        if end > n:
            end = n
            start = end - block_size
        
        base = a.copy()
        base[start:end] = 0
        base_sum = base.sum()
        if base_sum < 1e-15:
            continue
        
        # Current block values
        block_val = a[start:end].sum()
        
        # Try redistributing the block mass
        # Current distribution within block
        block_dist = a[start:end] / max(block_val, 1e-30)
        
        # Try: uniform, shifted, random
        candidates = [
            np.ones(end - start) / (end - start),
            block_dist,
            rng.dirichlet(np.ones(end - start)),
            np.array([0.5, 0.5, 0, 0][:end-start]) if end-start >= 2 else block_dist,
        ]
        
        best_cand = block_dist
        best_local = best_score
        
        for cand in candidates:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            # Try different total mass for the block
            for mass_factor in [0.5, 0.8, 1.0, 1.2, 1.5]:
                total_mass = block_val * mass_factor
                if total_mass >= 1.0:
                    continue
                a_trial = base + total_mass * cand
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_cand = (cand, mass_factor)
        
        if best_local < best_score - 1e-14:
            if isinstance(best_cand, tuple):
                cand, mf = best_cand
            else:
                cand, mf = best_cand, 1.0
            a = base + block_val * mf * cand
            a = a / a.sum()
            best_score = best_local
            best_a = a.copy()
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD it={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def peak_shave_line_search(a0, time_limit):
    """Peak shaving with proper line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (excluding endpoints which are always small)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = int(idx[np.argmax(contribs)])
        
        # Line search: reduce a[top_i] and redistribute
        a0_val = a[top_i]
        if a0_val < 1e-12:
            break
        
        # Binary search for optimal reduction
        lo, hi = 0.0, 1.0  # fraction to reduce
        best_frac = 0.0
        best_sc = best_score
        
        for _ in range(20):
            mid = (lo + hi) / 2
            new_val = a0_val * (1 - mid)
            diff = a0_val - new_val
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a[mask]
            a_trial = a.copy()
            a_trial[top_i] = new_val
            if others.sum() > 1e-15:
                a_trial[mask] = others + diff * others / others.sum()
            a_trial = a_trial / a_trial.sum()
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_sc:
                best_sc = sc
                best_frac = mid
            if mid < 0.5:
                lo = mid
            else:
                hi = mid
        
        if best_frac > 0 and best_sc < best_score - 1e-14:
            new_val = a0_val * (1 - best_frac)
            diff = a0_val - new_val
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a[mask]
            a = a.copy()
            a[top_i] = new_val
            a[mask] = others + diff * others / others.sum()
            a = a / a.sum()
            best_score = best_sc
            best_a = a.copy()
        else:
            # Try reducing a different coordinate
            if len(idx) > 1:
                # Pick second-best contributor
                sorted_idx = np.argsort(contribs)[::-1]
                if len(sorted_idx) > 1:
                    alt_i = int(idx[sorted_idx[1]])
                    if alt_i != top_i and a[alt_i] > 1e-12:
                        for factor in [0.9, 0.8, 0.7]:
                            a_trial = a.copy()
                            a_trial[alt_i] *= factor
                            a_trial = a_trial / a_trial.sum()
                            sc = 2.0 * n * fast_convolve(a_trial).max()
                            if sc < best_score - 1e-14:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                break
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave it={step} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
