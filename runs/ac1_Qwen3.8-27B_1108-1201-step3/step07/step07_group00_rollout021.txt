"""
Multi-phase optimization: LP with PDLP + LSE-LBFGS + coordinate descent.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(a, w, factor=1.0):
    """
    Compute the gradient contribution: sum_k w[k] * d(a*a)[k]/da[j]
    d(a*a)[k]/da[j] = 2*a[k-j] if 0 <= k-j <= n-1 else 0
    Returns grad[j] = 2 * sum_k w[k] * a[k-j] (valid range)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += factor * wk * a[k - idx]
    return grad


def lp_direction(f, top_k, solver='PDLP', eps=1e-8):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M, active_idx) or (None, M, None).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M, None
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        if solver == 'PDLP':
            prob.solve(solver=cp.PDLP, max_iter=10000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=eps)
    except Exception as e:
        return None, M, None
    
    if g.value is None:
        return None, M, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() < 1e-10:
        return None, M, None
    return g_val, M, top_idx


def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=300,
                        solver='PDLP', label='lp'):
    """
    LP-based optimization loop with adaptive top_k and bisection line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive top_k
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)
        
        g, M, active_idx = lp_direction(f, top_k, solver=solver)
        
        if g is None:
            # Try with fewer constraints
            if top_k > 20:
                g, M, active_idx = lp_direction(f, max(20, top_k // 2), solver=solver)
            if g is None:
                # Try SCS as fallback
                g, M, active_idx = lp_direction(f, min(top_k, 100), solver='SCS')
            if g is None:
                stuck += 5
                if stuck > 20:
                    break
                continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                # Perturb and restart
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                g, M, _ = lp_direction(f, top_k, solver=solver)
                if g is None:
                    break
                S_g = g.sum()
                if S_g <= 1.0 + 1e-5:
                    break
            elif stuck % 5 == 0:
                top_k = min(top_k + 50, top_k_max)
            continue
        
        # Bisection line search
        h = g / S_g  # normalize h to have sum 1
        best_t = 0.1
        best_obj_trial = fast_convolve((1 - 0.1) * f + 0.1 * h).max()
        
        lo_t, hi_t = 0.0, 1.0
        for _ in range(12):  # 12 bisection steps
            mid_t = (lo_t + hi_t) / 2
            f_trial = (1 - mid_t) * f + mid_t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                hi_t = mid_t
                continue
            f_trial /= s
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_obj_trial:
                best_obj_trial = obj_trial
                best_t = mid_t
                lo_t = mid_t
            else:
                hi_t = mid_t
        
        # Update
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                # Try perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                stuck = 0
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} M={best_obj:.10f} score={score:.6f} "
                  f"top_k={top_k} t={elapsed:.0f}s")
    
    return best_f, best_obj


def run_lse_lbfgs(f_init, time_limit, t_init=0.05, t_min=0.001, label='lse'):
    """
    Log-sum-exp smoothed optimization using L-BFGS with simplex projection.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    temperature = t_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Compute softmax weights
        scaled = b / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_conv_gradient(f, w, factor=2.0 / temperature)
        
        # L-BFGS step (simplified: use a fixed number of L-BFGS steps)
        # Use scipy L-BFGS-B with simplex projection via callback
        def obj_and_grad(z):
            a = project_simplex(z)
            b_local = fast_convolve(a)
            scaled_local = b_local / temperature
            scaled_local -= scaled_local.max()
            w_local = np.exp(scaled_local)
            w_local_sum = w_local.sum()
            if w_local_sum < 1e-30:
                w_local = np.zeros(len(b_local))
                w_local[b_local.argmax()] = 1.0
            else:
                w_local /= w_local_sum
            obj_val = temperature * (scaled_local.max() + np.log(w_local_sum))
            # obj_val = np.log(w_local_sum) + b_local.max()  # same thing
            grad_local = compute_conv_gradient(a, w_local, factor=2.0 / temperature)
            return obj_val, grad_local
        
        # Run a few L-BFGS iterations
        result = minimize(obj_and_grad, f, method='L-BFGS-B',
                          options={'maxiter': 20, 'gtol': 1e-6})
        
        if result.success or result.fun < obj_and_grad(f)[0]:
            f_new = project_simplex(result.x)
            if f_new.sum() > 1e-10:
                f_new /= f_new.sum()
            f = f_new
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            # Decrease temperature
            temperature = max(temperature * 0.85, t_min)
        else:
            no_improve += 1
            if no_improve > 5:
                temperature = max(temperature * 0.9, t_min)
                if no_improve > 15:
                    break
        
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} M={best_obj:.10f} score={score:.6f} "
                  f"temp={temperature:.4f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label='cd'):
    """
    Careful coordinate descent with multiple multipliers.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    passes = 0
    improved = True
    
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = fast_convolve(f).max()
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
        if passes % 3 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} pass={passes} score={score:.6f} t={elapsed:.0f}s")
    
    return f, best_obj

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP with PDLP solver, LSE-smoothed projected gradient, and coordinate descent.
Key improvements over previous: PDLP solver for speed, bisection line search,
LSE-smoothed gradient polish, adaptive constraint tracking, and multi-phase strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, w, factor=1.0):
    """
    Compute grad[j] = factor * sum_k w[k] * 2*a[k-j] for valid k-j.
    This is the gradient of log-sum-exp((a*a)[k]/t) up to the factor.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += factor * wk * a[k - idx]
    return 2.0 * grad


def lp_direction(f, top_k, solver='PDLP', full_constraints=False):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for selected peaks.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if full_constraints:
        top_idx = np.arange(len(b))
    elif top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix efficiently
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        if solver == 'PDLP':
            prob.solve(solver=cp.PDLP, max_iter=20000)
        elif solver == 'SCS':
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        else:
            prob.solve(solver=cp.ECOS)
    except Exception:
        return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() < 1e-10:
        return None, M
    return g_val, M


def bisection_line_search(f, h, f_obj, max_iter=15):
    """
    Bisection line search: find t in [0,1] that minimizes f_obj((1-t)*f + t*h).
    h should have sum 1 (same normalization as f).
    """
    best_t = 0.1
    best_val = f_obj((1 - 0.1) * f + 0.1 * h)
    lo, hi = 0.0, 1.0

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        trial = (1 - mid) * f + mid * h
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-10:
            hi = mid
            continue
        trial /= s
        val = f_obj(trial)
        if val < best_val:
            best_val = val
            best_t = mid
            lo = mid
        else:
            hi = mid
    return best_t, best_val


def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=300,
                        solver='PDLP', label='lp', full_start=False):
    """
    LP-based optimization with adaptive top_k and bisection line search.
    """
    t0 = time.monotonic()

        fk = top_k if not use_full else (2 * n - 1)

def run_lse_projected_gradient(f_init, time_limit, t_init=0.05, t_min=0.001,
                                label='lse'):
    """
    LSE-smoothed optimization with projected subgradient and backtracking.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    temperature = t_init
    lr = 0.005
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        scaled = b / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        grad = compute_lse_gradient(f, w, factor=1.0 / temperature)

        # Projected gradient step with backtracking
        best_step = 0.01
        best_trial = fast_convolve(f).max()
        for step in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            f_trial = project_simplex(f - step * grad)
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_trial:
                best_trial = obj_trial
                best_step = step

        f = project_simplex(f - best_step * grad)
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            temperature = max(temperature * 0.9, t_min)
        else:
            no_improve += 1
            if no_improve > 10:
                temperature = max(temperature * 0.85, t_min)
                lr *= 0.9
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0

        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} M={best_obj:.10f} score={score:.8f} "
                  f"temp={temperature:.4f} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label='cd'):
    """
    Careful coordinate descent with logarithmic multipliers.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.4, 0.6, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.6, 2.0]
    passes = 0
    improved = True

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            current_obj = fast_convolve(f).max()
            best_j_val = f[j]
            best_j_obj = current_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    improved = True

        if passes % 5 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} pass={passes} score={score:.8f} t={elapsed:.0f}s")

    return f, best_obj


def pair_swap_polish(f_init, time_limit, label='pswap'):
    """
    Try swapping mass between pairs of coordinates.
    For each pair (i,j), try transferring a small amount from i to j and vice versa.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(999)
    improved = True
    it = 0

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        it += 1
        # Try random pairs
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j or f[i] < 1e-12 and f[j] < 1e-12:
                continue
            for frac in [0.1, 0.25, 0.5, 0.75]:
                amount = f[i] * frac
                if amount < 1e-14:
                    continue
                # Transfer from i to j
                f_trial = f.copy()
                f_trial[i] -= amount
                f_trial[j] += amount
                s = f_trial.sum()
                if s > 1e-10:
                    f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    f = f_trial.copy()
                    best_obj = obj
                    improved = True
                    break
    return f, best_obj
