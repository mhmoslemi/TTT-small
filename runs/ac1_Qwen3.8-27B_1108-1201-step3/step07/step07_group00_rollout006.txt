"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed optimization for warmup, LP with iterative
constraint addition for main optimization, multi-length exploration,
and coordinate descent polish. Key improvement over previous: iterative
constraint addition ensures LP directions respect ALL peak constraints,
not just a fixed top-k.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, b, tau):
    """Compute gradient of log-sum-exp smoothing of auto-convolution."""
    n = len(a)
    scaled = b / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        # j ranges where 0 <= k-j < n: j in [max(0,k-n+1), min(n-1,k)]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        # For each j in [j_lo, j_hi]: db[k]/da[j] involves a[k-j]
        # and if k==2j, there's an extra a[j]
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += w[k] * a[idx]
        # Handle the k==2j case (extra factor)
        if k % 2 == 0:
            j_mid = k // 2
            if j_lo <= j_mid <= j_hi:
                grad[j_mid] += w[k] * a[j_mid]

    return (2.0 / tau) * grad

        it += 1

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:

def lp_solve_with_constraints(f, n, constraint_indices, M, eps=1e-7):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f@g)[k] <= M for k in constraint_indices.
    Returns g or None.
    """
    K = len(constraint_indices)
    if K == 0:
        return np.ones(n) / n

    A = np.zeros((K, n))
    for row_i, k in enumerate(constraint_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=200000, abs_acc=eps, rel_acc=eps)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_iterative_direction(f, n, max_constraints=400, initial_k=20, eps=1e-7):
    """
    LP direction finding with iterative constraint addition.
    Starts with top-k peaks, solves LP, checks for new violations, adds them.
    Returns (g, M, num_constraints).
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M, 0

    # Start with top initial_k peaks
    top_idx = np.argpartition(b, -initial_k)[-initial_k:]
    constraint_set = set(top_idx.tolist())

    g_val = None
    for phase in range(30):
        # Solve LP with current constraints
        sorted_constraints = sorted(constraint_set)
        g_val = lp_solve_with_constraints(f, n, sorted_constraints, M, eps)

        if g_val is None:
            # Try with fewer constraints
            if len(constraint_set) > 10:
                # Keep only top half
                fg = fast_cross_convolve(f, g_val if g_val is not None else np.ones(n)/n)
                break
            return None, M, len(constraint_set)

        # Check for violations
        fg = fast_cross_convolve(f, g_val)
        # Find peaks that violate M and aren't already constrained
        all_idx = np.arange(len(fg))
        unconstrained = np.array([i for i in all_idx if i not in constraint_set])
        if len(unconstrained) == 0:
            break

        viol_mask = fg[unconstrained] > M + 1e-10
        if not viol_mask.any():
            break

        viol_idx = unconstrained[viol_mask]
        # Add the worst violators (up to 5 per phase)
        worst = viol_idx[np.argsort(fg[viol_idx])[-5:]]
        constraint_set.update(worst.tolist())

        if len(constraint_set) >= max_constraints:
            break

    if g_val is None:
        return None, M, 0

    # Final check: if the solution has sum <= 1, it means f is already optimal
    return g_val, M, len(constraint_set)

    t0 = time.monotonic()

        if S_g <= 1.0 + 1e-5:
            stuck += 1

                break
            # Perturb and try again
            rng = np.random.default_rng(it)
            f_pert = f + rng.standard_normal(n) * 0.001
            f = project_simplex(f_pert)
            continue

        # Line search with quadratic model
        h = g / S_g
        b_f = fast_convolve(f)

        # Evaluate at several step sizes
        best_t = 0.05
        best_t_obj = M

        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:

            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                # Try a larger step or restart
                f = project_simplex(best_f + np.random.default_rng(it * 31).standard_normal(n) * 0.002)

                if time.monotonic() - t0 > time_limit * 0.9:

                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_obj


def interpolate_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    old_x = np.linspace(0.0, 1.0, len(src))
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-10:
        result /= s
    else:
        result = np.ones(target_n) / target_n
    return result
