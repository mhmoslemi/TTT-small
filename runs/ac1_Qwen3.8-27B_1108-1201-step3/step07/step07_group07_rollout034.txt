            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed
gradient descent with beta annealing, FFT-accelerated gradients,
LP-based refinement, and coordinate descent polishing.

Key innovation: Uses the exact gradient of the log-sum-exp smoothing
of max(conv(a,a)), which is 2*cross_correlation(softmax(beta*b), a),
enabling efficient O(n log n) gradient steps with Nesterov momentum.
Beta is annealed from smooth (beta=1) to sharp (beta=200) to
navigate the optimization landscape effectively.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_logsumexp_gradient(a, beta):
    """
    Compute the gradient of (1/beta)*log(sum(exp(beta*b[k]))) w.r.t. a.
    Result: 2 * cross_correlation(softmax(beta*b_shifted), a)
    
    where b = conv(a,a) and cross_correlation(p,a)[j] = sum_i p[i+j]*a[i]
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    # Softmax weights (numerically stable)
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    # Cross-correlation via FFT
    # R[j] = sum_i p[i+j] * a[i] for j=0..n-1
    # R = IFFT(FFT(p) * conj(FFT(a)))[:n]
    fft_size = 1 << (3 * n - 2).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(P * np.conj(A), fft_size)[:n]
    
    return 2.0 * corr


def logsumexp_gd(a0, time_limit, beta_start=1.0, beta_end=200.0,
                 n_beta_steps=20, momentum=0.9, lr_init=0.01,
                 lr_decay=0.995, seed=42):
    """
    Log-sum-exp smoothed projected gradient descent with Nesterov momentum
    and beta annealing.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    best_a = a.copy()
    best_b = fast_convolve(a).max()
    
    # Beta schedule (exponential from beta_start to beta_end)
    if n_beta_steps > 1:
        log_betas = np.linspace(np.log(beta_start), np.log(beta_end), n_beta_steps)
        beta_schedule = np.exp(log_betas)
    else:
        beta_schedule = np.array([beta_end])
    
    iters_per_beta = max(50, int(time_limit / n_beta_steps / 0.001))
    # Actually, let me just run until time runs out, adjusting beta over time
    
    total_time = time_limit
    # We'll run in a loop, increasing beta over time
    
    cur_beta = beta_start
    lr = lr_init
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        # Gradually increase beta
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.7))  # reach beta_end at 70% of time
        log_beta = np.log(beta_start) + progress * (np.log(beta_end) - np.log(beta_start))
        cur_beta = np.exp(log_beta)
        
        # Also decay learning rate
        lr = lr_init * (lr_decay ** int(elapsed * 1000))  # decay per "thousand steps"
        # Better: decay based on iteration count
        # Let me use a different decay strategy
        lr = lr_init * (1.0 + elapsed / time_limit * (beta_end / beta_start)) ** (-0.5)
        
        # Compute gradient
        grad = compute_logsumexp_gradient(a, cur_beta)
        
        # Nesterov momentum step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate (only periodically to save time)
        if no_improve % 10 == 0:
            b = fast_convolve(a_new)
            obj = b.max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        a = a_new
        
        # Periodic restart
        if no_improve >= max_no_improve:
            kick = 0.005 * (1.0 - progress)
            a = project_simplex(best_a + rng.standard_normal(n) * kick)
            v = np.zeros(n)
            no_improve = 0
        
        # Adjust momentum: increase as beta increases (more confident)
        momentum = 0.9 + 0.08 * progress
    
    return best_a, best_b


def lp_refine(f, max_iters=20, time_limit=60.0):
    """LP-based refinement using sparse constraint structure."""
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Build constraint matrix (dense for simplicity, n up to ~2048)
        # Row k of A: A[k, j] = a[j] if 0 <= k-j < n, else 0
        # We want: A @ g <= M for all k, g >= 0
        # Maximize: sum(g)
        
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[k, lo:hi+1] = a[k - np.arange(lo, hi+1)]
        
        c = -np.ones(n)
        b_ub = np.full(num_constraints, M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend /= ss
                obj = fast_convolve(blend).max()
                if obj < best_blend_obj - 1e-14:
                    best_blend_obj = obj
                    best_t = t
            
            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a /= a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                # Try smaller t
                found = False
                for t in [0.02, 0.05, 0.08]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend /= blend.sum()
                    obj = fast_convolve(blend).max()
                    if obj < best_obj - 1e-14:
                        a = blend
                        best_obj = obj
                        best_a = a.copy()
                        found = True
                        break
                if not found:
                    break
        
        except Exception:
            break
    
    return best_a, best_obj

    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val

                new_val = old_val * mult

                if scale < -0.01 or scale > 20.0:

                a[mask] /= scale

    corr = irfft(P * np.conj(A), fft_size)[:n]
    return 2.0 * corr


def logsumexp_gd(a0, time_limit, beta_start=1.0, beta_end=200.0,
                 lr_init=0.01, seed=42):
    """Log-sum-exp smoothed projected GD with Nesterov and beta annealing."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_b = float(fast_convolve(a).max())
    lr = lr_init
    no_improve = 0
    it_count = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit * 0.7))
        cur_beta = np.exp(np.log(beta_start) + progress * (np.log(beta_end) - np.log(beta_start)))
        lr = lr_init * (1.0 + 20.0 * progress) ** (-1.0)
        momentum = 0.88 + 0.10 * progress
        grad = compute_logsumexp_gradient(a, cur_beta)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        it_count += 1
        if it_count % 5 == 0:
            obj = float(fast_convolve(a).max())
            if obj < best_b - 1e-15:
                best_b = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > 3000:
            kick = 0.003 * max(0.1, 1.0 - progress)
            a = project_simplex(best_a + rng.standard_normal(n) * kick)
            v = np.zeros(n)
            no_improve = 0
            it_count = 0
    return best_a, best_b


def logsumexp_gd_multi(a0, time_limit, seed=42):
    """Run logsumexp GD with multiple lr/beta configurations."""
    t0 = time.monotonic()
    n = len(a0)
    best_a = a0.copy()
    best_b = float(fast_convolve(a0 / a0.sum()).max())

    configs = [
        (1.0, 100.0, 0.01),
        (0.5, 50.0, 0.008),
        (2.0, 500.0, 0.015),
        (1.0, 150.0, 0.005),
        (0.3, 200.0, 0.012),
    ]
    t_per = time_limit / len(configs)
    for i, (bs, be, lr) in enumerate(configs):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        a_init = best_a.copy()
        a_opt, obj = logsumexp_gd(a_init, min(t_per, rem - 1),
                                  beta_start=bs, beta_end=be,
                                  lr_init=lr, seed=seed + i * 1000)
        if obj < best_b - 1e-14:
            best_b = obj
            best_a = a_opt.copy()
            print(f"    LSE-GD config {i}: max_b={obj:.10f}")
    return best_a, best_b


def lp_refine(f, max_iters=25, time_limit=90.0):
    """LP-based refinement: solve max sum(g) s.t. conv(f,g)<=M, g>=0."""
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        M = float(b.max())
        num_con = 2 * n - 1
        A = np.zeros((num_con, n))
        for k in range(num_con):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[k, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]
        c = -np.ones(n)
        b_ub = np.full(num_con, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend /= ss
                obj = float(fast_convolve(blend).max())
                if obj < best_blend_obj - 1e-14:
                    best_blend_obj = obj
                    best_t = t
            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a /= a.sum()
                obj = float(fast_convolve(a).max())
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                found = False
                for t in [0.01, 0.02, 0.03, 0.05]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    ss = blend.sum()
                    if ss < 1e-10:
                        continue
                    blend /= ss
                    obj = float(fast_convolve(blend).max())
                    if obj < best_obj - 1e-14:
                        a = blend
                        best_obj = obj
                        best_a = a.copy()
                        found = True
                        break
                if not found:
                    break
        except Exception:
            break
    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent: scale one element, rescale rest on simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            a_saved = a.copy()
            best_trial_obj = best_obj
            best_trial_val = old_val
            best_trial_scale = 1.0
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue
                a[j] = new_val
                a[mask] = a_saved[mask] * scale
                ss = a.sum()
                if ss < 1e-14:
                    a[:] = a_saved
                    continue
                a /= ss
                val = float(fast_convolve(a).max())
                if val < best_trial_obj - 1e-14:
                    best_trial_obj = val
                    best_trial_val = new_val
                    best_trial_scale = scale
            if best_trial_val != old_val:
                a[j] = best_trial_val
                a[mask] = a_saved[mask] * best_trial_scale
                a /= a.sum()
                best_obj = best_trial_obj
                best_a = a.copy()
                improved = True
            else:
                a[:] = a_saved
    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bimodal_init(n):
    """High at edges, low in center - helps reduce central peak."""
    x = np.linspace(0, 1, n)
    a = np.exp(-5.0 * (x - 0.5) ** 2) + 0.3
    a = 1.0 / (a + 0.1)  # invert to make bimodal
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_random_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_flat_center_init(n):
    """Flat in center, tapered at edges."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = n // 4
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.3 + 0.7 * t
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def run_optimization_pipeline(a_init, time_budget, n, seed=42):
    """Run full optimization pipeline: LSE-GD -> LP -> CD -> LP."""
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, float('inf')
    a = a / s
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    # Phase 1: Log-sum-exp GD
    t1 = min(time_budget * 0.50, time_budget - 30)
    if t1 > 5:
        a_gd, obj_gd = logsumexp_gd_multi(a, t1, seed=seed)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
        rem = time_budget - (time.monotonic() - t0)
    else:
        rem = time_budget

    # Phase 2: LP refinement
    t2 = min(rem * 0.30, 60.0)
    if t2 > 5:
        a_lp, obj_lp = lp_refine(best_a, max_iters=20, time_limit=t2)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"    LP: max_b={obj_lp:.10f}")
        rem = time_budget - (time.monotonic() - t0)

    # Phase 3: More LSE-GD from LP result
    t3 = min(rem * 0.30, 60.0)
    if t3 > 5:
        a_gd2, obj_gd2 = logsumexp_gd(best_a, t3,
                                       beta_start=50.0, beta_end=300.0,
                                       lr_init=0.003, seed=seed + 777)
        if obj_gd2 < best_obj:
            best_obj = obj_gd2
            best_a = a_gd2.copy()
        rem = time_budget - (time.monotonic() - t0)

    # Phase 4: Final LP
    t4 = min(rem * 0.40, 40.0)
    if t4 > 5:
        a_lp2, obj_lp2 = lp_refine(best_a, max_iters=15, time_limit=t4)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            print(f"    LP2: max_b={obj_lp2:.10f}")
        rem = time_budget - (time.monotonic() - t0)

    # Phase 5: Coordinate descent
    t5 = min(rem, 30.0)
    if t5 > 3:
        a_cd, obj_cd = coordinate_descent(best_a, time_limit=t5)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    return best_a, best_obj

            parent = p / p.sum()
