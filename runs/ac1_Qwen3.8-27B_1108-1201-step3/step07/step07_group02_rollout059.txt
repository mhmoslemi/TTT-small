"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. L-BFGS with LogSumExp smoothing and multi-stage temperature annealing
2. Spectral shaping (iterative power spectrum adjustment) to flatten auto-convolution
3. Aggressive peak shaving with redistribution to quiet regions
4. LP-based best response iteration
5. Multiple sequence lengths (768, 1024, 1536, 2048) with diverse initializations
6. Cosine-based diverse initializations to escape local minima
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_score(a):
    """Compute the objective score for a sequence (assumed non-negative)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return float(2.0 * len(a) * b.max())


def lbgfs_softmax(a0, time_limit, T_start=100.0, T_end=1.0, maxiter=5000):
    """L-BFGS refinement with soft-max (LogSumExp) smoothing and geometric temperature."""
    n = len(a0)
    z = z_from_a(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    stopped = [False]

    def func_factory(T_val):
        def func(zv):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def jac_factory(T_val):
        def jac(zv):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(zv)
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    # Geometric temperature schedule with more stages for finer control
    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit or stopped[0]:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=func_factory(stage_T),
                x0=z,
                jac=jac_factory(stage_T),
                method='L-BFGS-B',
                options={
                    'maxiter': iters_per_stage,
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxfun': iters_per_stage * 100,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_shaping(a0, time_limit, n_iters=500, alpha_init=0.5, alpha_decay=0.995):
    """
    Spectral shaping: iteratively modify the power spectrum to flatten auto-convolution.
    The idea: make |FFT(a)|^2 more concentrated at low frequencies to spread out
    the time-domain auto-convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    alpha = alpha_init
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT of current a (zero-padded for linear convolution)
        A = rfft(a, fft_size)
        P = A.real**2 + A.imag**2  # Power spectrum

        # Desired: boost low frequencies, suppress high frequencies
        freqs = np.arange(fft_size // 2 + 1)
        # Create a weighting that emphasizes low frequencies
        weight = 1.0 + alpha * np.exp(-freqs / (n * 0.3))
        # Also try to flatten: reduce peaks in the power spectrum
        P_max = P.max()
        if P_max > 0:
            # Compress the dynamic range
            P_new = P ** 0.95
            P_new = P_new / P_new.sum() * P.sum()

        # Combine: apply frequency weighting
        P_modified = P_new * weight
        P_modified = np.maximum(P_modified, 0)

        # New amplitude with original phase
        new_amp = np.sqrt(P_modified)
        phase_A = A / np.maximum(np.abs(A), 1e-300)
        A_new = new_amp * phase_A

        # Inverse FFT
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        # Decay alpha to reduce aggressiveness over time
        alpha *= alpha_decay

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralShape it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def peak_shave_advanced(a0, time_limit, n_steps=10000):
    """

    mass to 'quiet' regions (coordinates that contribute little to any peak).

    n = len(a0)

                            # Redistribute total mass to quiet region

                            # Find quiet coords

                                je2 = min(n - 1, pk)

                            if len(quiet_coords) > 0:

                    best_score = sc

        if (step + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0

def lp_refine(a0, time_limit, n_max_iters=2000):
    """
    LP-based best response iteration with Nesterov acceleration.
    Solves: min M s.t. conv(a_fixed, b)[k] <= M, b >= 0, sum(b) = 1
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    y = a.copy()  # virtual iterate
    t0 = time.monotonic()

    # Build convolution matrix once (it depends on the fixed a, which changes)
    # We'll rebuild it each iteration but use cached version when possible

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: min M s.t. T(y) @ b <= M, b >= 0, sum(b) = 1
        # Use scipy linprog or manual subgradient
        T = build_conv_matrix(y)
        m = 2 * n - 1

        # LP: variables are b[0..n-1] and M
        # Minimize M
        # Constraints: T[i] @ b - M <= 0 for all i, sum(b) = 1, b >= 0, M >= 0
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M

        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = T
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)

        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Bounds: b[i] >= 0, M >= 0
        bounds = [(0, None)] * n + [(0, None)]

        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                b_new = res.x[:n]
                b_new = np.maximum(b_new, 0)
                sb = b_new.sum()
                if sb > 1e-15:
                    b_new = b_new / sb
                else:
                    b_new = a.copy()
            else:
                continue
        except Exception:
            # Fallback: greedy best response
            b_new = a.copy()

        # Nesterov step
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Weighted average for actual iterate
        w_new = alpha * (it + 2)
        a = ((it + 1) * a + w_new * b_new) / (it + 1 + w_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Check both a and y
        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def cosine_init(n, freq=1, amplitude=1.0, rng=None):
    """Generate a cosine-based non-negative initialization."""
    if rng is None:
        rng = np.random.default_rng()
    j = np.arange(n)
    base = 1.0 + amplitude * np.cos(2 * np.pi * freq * j / n)
    base = np.maximum(base, 0)
    # Add small random perturbation
    noise = rng.normal(0, 0.01, n)
    base = base + noise
    base = np.maximum(base, 0)
    s = base.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return base / s


def random_init(n, rng):
    """Generate a random non-negative initialization."""
    a = rng.exponential(1.0, n)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining L-BFGS, spectral shaping,
    peak shaving, LP refinement, and diverse initializations across
    multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            # Verify with actual evaluator
            sc_verify = evaluate_safe(a)
            if np.isfinite(sc_verify) and sc_verify <= sc + 1e-9:
                best_score = sc_verify
                best_seq = a.copy()
                print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a)})")
            else:
                # Use the computed score for internal tracking
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        best_seq = cosine_init(n_init, freq=1, amplitude=0.5, rng=rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    print(f"Initial best: {best_score:.6f} (n={len(best_seq)})")

    # === Phase 1: L-BFGS on parent ===
    print(f"\n=== Phase 1: L-BFGS on parent n={len(best_seq)} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"  T: 100->1, budget={t1:.0f}s")
        a_r = lbgfs_softmax(best_seq, t1, T_start=100.0, T_end=1.0, maxiter=6000)
        try_update(a_r, "lbgfs_parent")

    # === Phase 2: Spectral shaping on best ===
    print(f"\n=== Phase 2: Spectral Shaping n={len(best_seq)} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 150.0)
    if t2 > 20:
        a_spec = spectral_shaping(best_seq, t2, n_iters=1000, alpha_init=1.0)
        try_update(a_spec, "spectral")
        # L-BFGS after spectral
        rem = deadline - time.monotonic()
        if rem > 30:
            t2b = min(rem * 0.10, 120.0)
            a_sr = lbgfs_softmax(best_seq, t2b, T_start=20.0, T_end=0.5, maxiter=4000)
            try_update(a_sr, "lbgfs_post_spectral")

    # === Phase 3: Try n=1024 ===
    print(f"\n=== Phase 3: n=1024 ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_try = 1024
        init_1024 = interpolate_sequence(best_seq, n_try)
        t3 = min(rem * 0.18, 280.0)
        if t3 > 30:
            print(f"  L-BFGS T: 50->1, budget={t3:.0f}s")
            a_1024 = lbgfs_softmax(init_1024, t3, T_start=50.0, T_end=1.0, maxiter=5000)
            try_update(a_1024, "n1024_lbgfs")

            rem = deadline - time.monotonic()
            if rem > 30 and len(best_seq) == n_try:
                t3b = min(rem * 0.08, 120.0)
                a_1024s = spectral_shaping(best_seq, t3b, n_iters=500)
                try_update(a_1024s, "n1024_spectral")

    # === Phase 4: Try n=1536 ===
    print(f"\n=== Phase 4: n=1536 ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_try = 1536
        init_1536 = interpolate_sequence(best_seq, n_try)
        t4 = min(rem * 0.15, 250.0)
        if t4 > 30:
            print(f"  L-BFGS T: 50->1, budget={t4:.0f}s")
            a_1536 = lbgfs_softmax(init_1536, t4, T_start=50.0, T_end=1.0, maxiter=5000)
            try_update(a_1536, "n1536_lbgfs")

    # === Phase 5: Try n=2048 ===
    print(f"\n=== Phase 5: n=2048 ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_try = 2048
        init_2048 = interpolate_sequence(best_seq, n_try)
        t5 = min(rem * 0.12, 200.0)
        if t5 > 30:
            print(f"  L-BFGS T: 50->2, budget={t5:.0f}s")
            a_2048 = lbgfs_softmax(init_2048, t5, T_start=50.0, T_end=2.0, maxiter=4000)
            try_update(a_2048, "n2048_lbgfs")

    # === Phase 6: LP refinement on best ===
    print(f"\n=== Phase 6: LP Refinement n={len(best_seq)} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 200.0)
    if t6 > 30:
        print(f"  budget={t6:.0f}s")
        a_lp = lp_refine(best_seq, t6, n_max_iters=1000)
        try_update(a_lp, "lp_refine")

    # === Phase 7: Diverse random restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    rem = deadline - time.monotonic()
    n_restarts = 5
    if rem > n_restarts * 30:
        n_best = len(best_seq)
        # Try different cosine frequencies and random inits
        n_init_list = [n_best, 1024, 1536]
        freqs = [1, 2, 3, n_best // 4, n_best // 3]

        t_per_restart = min(rem / (n_restarts + 1), 120.0)
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_r = min(t_per_restart, rem * 0.8)
            if t_r < 15:
                break

            # Choose initialization strategy
            if r % 3 == 0:
                # Cosine with random freq
                n_r = n_init_list[r % len(n_init_list)]
                f_r = freqs[r % len(freqs)]
                amp_r = 0.3 + 0.4 * rng.random()
                a_r_init = cosine_init(n_r, freq=f_r, amplitude=amp_r, rng=rng)
                label = f"cos_f{f_r}_n{n_r}"
            elif r % 3 == 1:
                # Interpolated parent + noise
                n_r = n_init_list[r % len(n_init_list)]
                a_r_init = interpolate_sequence(best_seq, n_r)
                noise = rng.normal(0, 0.1, n_r)
                a_r_init = a_r_init + noise
                a_r_init = simplex_project(a_r_init)
                label = f"parent_noise_n{n_r}"
            else:
                # Pure random
                n_r = n_init_list[r % len(n_init_list)]
                a_r_init = random_init(n_r, rng)
                label = f"random_n{n_r}"

            print(f"  Restart {r+1}: {label}, budget={t_r:.0f}s")
            a_opt = lbgfs_softmax(a_r_init, t_r, T_start=30.0, T_end=0.5, maxiter=3000)
            try_update(a_opt, label)

    # === Phase 8: Advanced peak shaving on best ===
    print(f"\n=== Phase 8: Advanced Peak Shaving n={len(best_seq)} ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.15, 200.0)
    if t8 > 20:
        print(f"  budget={t8:.0f}s")
        a_ps = peak_shave_advanced(best_seq, t8, n_steps=5000)
        try_update(a_ps, "peak_shave")

    # === Phase 9: Final L-BFGS polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.4, 120.0)
        if t9 > 15:
            print(f"  L-BFGS T: 5->0.3, budget={t9:.0f}s")
            a_final = lbgfs_softmax(best_seq, t9, T_start=5.0, T_end=0.3, maxiter=4000)
            try_update(a_final, "final_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 20:
            t9b = min(rem * 0.4, 60.0)
            if t9b > 10:
                print(f"  Peak shave, budget={t9b:.0f}s")
                a_ps2 = peak_shave_advanced(best_seq, t9b, n_steps=3000)
                try_update(a_ps2, "final_peakshave")

    # === Phase 10: LP on final best ===
    rem = deadline - time.monotonic()
    if rem > 30 and len(best_seq) <= 2048:
        t10 = min(rem * 0.5, 100.0)
        if t10 > 20:
            print(f"\n=== Phase 10: Final LP n={len(best_seq)} ===")
            a_lp2 = lp_refine(best_seq, t10, n_max_iters=500)
            try_update(a_lp2, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    if best_seq is None:
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
