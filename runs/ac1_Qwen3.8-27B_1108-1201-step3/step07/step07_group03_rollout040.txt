**Approach: Direct optimization of the "flatness" using a different parameterization**

Instead of optimizing `a` directly, optimize the "autocorrelation" `b` and then find an `a` that produces it.

The constraint is that `b` must be a valid autocorrelation of a non-negative sequence. This means:
1. `b >= 0`
2. `b` is symmetric: `b[k] = b[2n-2-k]`
3. The Toeplitz matrix `T(b)` must be PSD
4. There must exist a non-negative `a` with `conv(a,a) = b`

Condition 4 is the hardest. But if we relax it to just conditions 1-3, we get a convex feasible set for `b`, and we can find the "flattest" `b` in this set. Then we need to find an `a` that produces (approximately) this `b`.

Step 1: Find `b` minimizing `max(b)` subject to `b >= 0`, `b` symmetric, `T(b) PSD`, `sum(b) = 1`.
Step 2: Find `a >= 0` with `conv(a,a) ≈ b` and `sum(a) = 1`.

Step 1 is an SDP (semidefinite program). Step 2 is a feasibility problem (or least-squares).

The SDP in Step 1:
- Variables: `b[0], ..., b[n-1]` (by symmetry, `b[n], ..., b[2n-2]` are determined)
- Objective: minimize `t` (= `max(b)`)
- Constraints: `b[k] <= t` for all `k`, `b[k] >= 0`, `sum(b) = 1`, `T(b) PSD`

The Toeplitz matrix `T(b)` is `n x n` with `T[i,j] = b[|i-j|]`.

This is a standard SDP that can be solved by MOSEK, CVXOPT, or other SDP solvers. For `n = 768`, the SDP has a 768x768 PSD constraint, which might be too large for most solvers.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP iteration
with SDP-guided initialization, diverse starts, and multi-phase polishing.

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_direction_lp(a, active_peaks, solver=None, d_box=0.5):
    """
    Given current a (normalized, sum=1) and active peak indices,
    solve LP: minimize t s.t. b[k] + 2*G_k@d <= t, sum(d)=0, a+d>=0, |d|<=d_box
    Returns (d, t_val).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a)
    b = fast_convolve(a)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(d_var) == 0,
        a + d_var >= 0,
        d_var <= d_box,
        -d_var <= d_box,
        t_var >= 0
    ]
    
    for k in active_peaks:
        k = int(k)
        G_k = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            G_k[i] = 2.0 * a[k - i]
        constraints.append(G_k @ d_var <= t_var - b[k])
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    d_val = np.asarray(d_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    
    # Clean up
    d_val = np.maximum(d_val, -a * 0.99)  # Ensure a + d >= 0 strictly
    return d_val, t_val

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP iteration (linearized SQP),
diverse multi-start exploration, pairwise coordinate optimization, and Nesterov polishing.
Key innovation: solve a direction-finding LP at each iteration that linearizes the
peak constraint and finds the optimal perturbation direction, with backtracking.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def build_gradient_row(a, k):
    """Compute db[k]/da[j] for all j, where b = conv(a,a)."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    G = np.zeros(n)
    for j in range(n):
        coeff = 0.0
        if lo <= j <= hi:
            coeff += a[k - j]
        if lo <= (k - j) <= hi:
            coeff += a[k - j]
        G[j] = coeff
    return G

def solve_direction_lp(a, active_peaks, solver=None, d_box=0.3):
    """
    Solve LP: minimize t s.t. b[k] + G_k @ d <= t for k in active_peaks,
    sum(d)=0, a+d>=0, |d|<=d_box.
    Returns (d, t_val) or (None, None).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()

    n = len(a)
    b = fast_convolve(a)

    d_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.sum(d_var) == 0,
        a + d_var >= 0,
        d_var <= d_box,
        -d_var <= d_box,
        t_var >= 0
    ]

    for k in active_peaks:
        k = int(k)
        G_k = build_gradient_row(a, k)
        constraints.append(G_k @ d_var <= t_var - b[k])

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    d_val = np.asarray(d_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    d_val = np.clip(d_val, -a * 0.999, d_box)
    return d_val, t_val

    """
    Iteratively solve direction-finding LPs to reduce the autocorrelation peak.
    Uses backtracking line search and randomization when stuck.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    d_box = 0.25
    stuck_count = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        b = fast_convolve(a)
        current_peak = b.max()

        # Select active peaks (within 8% of max, capped at 60)
        threshold = current_peak * 0.92
        active = np.where(b >= threshold)[0]
        if len(active) > 60:
            b_act = b[active]
            order = np.argsort(b_act)[::-1][:60]
            active = active[order]
        elif len(active) < 3:
            # Always include top peaks
            top_k = min(10, len(b))
            active = np.argpartition(b, -top_k)[-top_k:]

        # Solve direction LP
        d, t_pred = solve_direction_lp(a, active, solver, d_box=d_box)
        if d is None:
            break

        # Skip if direction is negligible
        if np.linalg.norm(d) < 1e-12:
            d_box = min(1.0, d_box * 1.5)
            if d_box >= 1.0:
                break
            continue

        # Backtracking line search
        improved = False
        base_alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]
        for alpha in base_alphas:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score * (1.0 - 1e-11):
                a = a_trial
                best_a = a.copy()
                best_score = score
                improved = True
                stuck_count = 0
                d_box = min(0.35, d_box * 1.05)
                break

        if not improved:
            stuck_count += 1
            if stuck_count >= 4:
                # Random perturbation to escape
                noise_scale = 0.003 * (1.0 + 0.2 * min(stuck_count, 10))
                a = best_a + rng.standard_normal(n) * noise_scale
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s_t
                stuck_count = 0
                d_box = min(1.0, d_box * 1.3)

            if stuck_count > 80:
                a = best_a.copy()
                d_box = 0.1
                stuck_count = 0

        if verbose and total_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS-LP iter={total_iters} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score, total_iters

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_smoothed_subgrad(a, top_k=30):
    """Compute smoothed subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    temp = 0.004 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        G_k = build_gradient_row(a, k)
        sg += w[idx] * G_k
    return sg

def nesterov_optimize(a0, time_limit, top_k=30, lr_init=0.005):
    """Nesterov-accelerated projected subgradient."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        sg = compute_smoothed_subgrad(y, top_k=top_k)
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0

        if no_improve > 400:
            noise = rng.standard_normal(n) * 0.008
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.7, lr * 1.5)
            no_improve = 0

        lr *= 0.9998

    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=200):
    """Coordinate descent with scaling factors."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    factors_large = [0.7, 0.85, 0.92, 1.08, 1.15, 1.3]
    factors_small = [0.95, 0.97, 1.03, 1.05]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in factors_large:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s_t
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()

        if not improved:
            any_imp = False
            for i in rng.choice(n, size=min(n, 80), replace=False):
                old_val = a[i]
                if old_val < 1e-14:
                    continue
                for factor in factors_small:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t < 1e-15:
                        a[i] = old_val
                        continue
                    a = a / s_t
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        old_val = a[i]
                        any_imp = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()
            if not any_imp:
                break

    return best_a, best_score

def pairwise_transfer(a0, time_limit):
    """Transfer mass between pairs of coordinates to reduce peak contributions."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(5, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        improved = False
        for pk in peaks:
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            js = np.arange(lo, hi + 1)
            contribs = a[js] * a[pk - js]

            if contribs.max() < 1e-15:
                continue

            # Find pair with max contribution where values differ
            top_idx = np.argsort(contribs)[::-1][:8]
            for ti in top_idx:
                i = int(js[ti])
                j = pk - i
                if j < 0 or j >= n or i == j:
                    continue
                if abs(a[i] - a[j]) < 1e-15:
                    continue
                # Transfer mass from larger to smaller
                if a[i] > a[j]:
                    src, dst = i, j
                else:
                    src, dst = j, i
                for frac in [0.1, 0.2, 0.35, 0.5]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0 or a_trial[dst] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score
                        improved = True
                        break
            if improved:
                break

        if not improved:
            # Random perturbation
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            s_t = a.sum()
            if s_t > 1e-10:
                a = a / s_t
            else:
                a = best_a.copy()

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Fourier-based initialization."""
    n_harm = int(rng.integers(4, 20))
    amps = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_blocks(n, rng):
    """Block-based initialization."""
    n_blocks = int(rng.integers(5, 30))
    block_size = max(2, n // n_blocks)
    amps = rng.random(n_blocks) + 0.05
    a = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        a[start:end] = amps[b]
    ksize = min(5, block_size)
    if ksize > 1:
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
