   **What sequences actually minimize the max convolution?**

   The problem is equivalent to: find a probability distribution p on {0,...,n-1} that minimizes max_k (p * p)[k] where * is the (full) convolution.

   This is related to the "addition theorem" and the structure of the convolution.

   **Key insight**: The convolution (p*p)[k] represents the probability that X+Y = k where X and Y are independent random variables with distribution p. So we want to minimize the maximum point mass of X+Y.

   This is equivalent to making the distribution of X+Y as "flat" as possible!

   For a uniform distribution on {0,...,n-1}, X+Y has a triangular distribution (max mass at n-1).

   To make X+Y more uniform, we need to "anti-correlate" the distribution with itself in some sense.

   **Idea: Use a distribution that makes X+Y as uniform as possible on {0, ..., 2n-2}.**

   If p is uniform on {0,...,n-1}, then X+Y is triangular on {0,...,2n-2} with max = n/n^2 = 1/n.

   Can we do better? The minimum possible max mass of X+Y (over all distributions on {0,...,n-1}) is at least 1/(2n-1) (by pigeonhole, since X+Y can take 2n-1 values). For the uniform distribution, the max is 1/n which is larger than 1/(2n-1).

   So in principle, we can get the max down to 1/(2n-1), giving a score of 2n/(2n-1) ≈ 1 for large n. But achieving exactly 1/(2n-1) requires that X+Y is uniform on {0,...,2n-2}, which is impossible for a distribution supported on {0,...,n-1} (since P(X+Y=0) = p_0^2 and P(X+Y=2n-2) = p_{n-1}^2, and for uniformity these would need to equal 1/(2n-1), so p_0 = p_{n-1} = 1/sqrt(2n-1), but then P(X+Y=0) + P(X+Y=2n-2) = 2/(2n-1) which uses up 2/(2n-1) of the total probability mass of X+Y, leaving (2n-3)/(2n-1) for the remaining 2n-3 values, each needing 1/(2n-1), total (2n-3)/(2n-1). Total = 2/(2n-1) + (2n-3)/(2n-1) = 1. ✓ But we also need all intermediate values to be exactly 1/(2n-1), which imposes a system of quadratic equations that may not have a non-negative solution.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding combined
with multi-temperature L-BFGS in log-space. Key innovation: at each iteration, solves
an LP to find the optimal non-negative direction b that respects the current peak

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def solve_lp_direction(a, M):
    """
    Solve LP: max 1^T b s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns the optimal b vector (length n).
    """
    n = len(a)
    # Build constraint matrix: A[k, j] = a[k-j] if valid
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A[k, js] = a[k - js]
    b_ub = np.full(num_constraints, M)
    c = -np.ones(n)  # minimize -sum(b)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x, float(res.fun)
    except Exception:
        pass
    return a.copy(), -1.0


def lp_interpolation_step(a, time_limit, max_iters=2000, verbose=True, label=""):
    """
    LP-based optimization: at each step, solve LP for optimal direction,
    then interpolate between current a and LP solution.
    """
    n = len(a)
    a = np.maximum(a.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    alpha_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    current_alpha_idx = 0
    no_improve_count = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        M_current = fast_convolve(a).max()

        # Solve LP
        b_opt, lp_val = solve_lp_direction(a, M_current)
        if b_opt is None or not np.all(np.isfinite(b_opt)):
            b_opt = a.copy()
            lp_val = -1.0

        # Try interpolation
        alphas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        if no_improve_count > 50:
            alphas = [0.05, 0.02, 0.01, 0.005, 0.002]
        elif no_improve_count > 20:
            alphas = [0.2, 0.1, 0.05, 0.02]

        best_trial = None
        best_trial_score = best_score
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * b_opt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = score_normalized(a_trial)
            if sc < best_trial_score - 1e-14:
                best_trial_score = sc
                best_trial = a_trial

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve_count = 0
            # Reset alpha schedule if we made a big improvement
            if it % 10 == 0:
                current_alpha_idx = 0
        else:
            no_improve_count += 1
            # After many non-improvements, try a random perturbation
            if no_improve_count > 100:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve_count = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={it} ({elapsed:.0f}s): "
                  f"score={best_score:.7f} lp_val={lp_val:.4f} "

def smoothed_objective_and_grad(z, T, n):
    """Compute smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    # Smoothed max: T * log(sum exp(b/T))
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    obj = 2.0 * n * T * lse

    # Gradient
    w = np.exp(scaled - m)
    w = w / w.sum()
    w = w * (b / T)  # weight by b_k/T
    w = w / w.sum()

    # grad_a[k] = 4n * (w * a)[k] (cross-correlation)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 4.0 * n * correl

    # Project to tangent space
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return obj, grad_z


def lbfgs_log_space(a_init, time_limit, T_start=200, T_end=2, n_temps=8,
                    maxiter_per_T=2000, verbose=True, label=""):
    """L-BFGS with geometric temperature schedule."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_temps)
    best_a = a.copy()
    best_score = score_normalized(a)

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break

        call_count = [0]

        def func(z):
            call_count[0] += 1
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_z = softmax_from_z(z)
            b_z = fast_convolve(a_z)
            scaled = b_z / T
            m = scaled.max()
            lse = m + np.log(np.exp(scaled - m).sum())
            return 2.0 * n * T * lse

        def jac(z):
            call_count[0] += 1
            if time.monotonic() - t0 > time_limit:
                return np.zeros(n)
            a_z = softmax_from_z(z)
            b_z = fast_convolve(a_z)
            scaled = b_z / T
            m = scaled.max()
            w = np.exp(scaled - m)
            w = w / w.sum()
            w = w * (b_z / T)
            w = w / w.sum()
            fft_size = 1 << (3 * n - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a_z, fft_size)
            correl = irfft(np.conj(W) * A, fft_size)[:n]
            grad_a = 4.0 * n * correl
            mean_grad = np.dot(a_z, grad_a)
            return a_z * (grad_a - mean_grad)

        try:
            res = minimize(
                fun=func, x0=z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_T, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': maxiter_per_T * 20,
                         'disp': False}
            )
            z0 = res.x
            a_new = softmax_from_z(z0)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    {label} L-BFGS T={T:.1f} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def subgradient_descent(a_init, time_limit, eta_start=0.01, eta_end=0.0001,
                        max_iters=20000, verbose=True, label=""):
    """Subgradient method with peak targeting and simplex projection."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max_iters)
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        current_score = 2.0 * n * b[k_peak]

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        # Subgradient of C[k_peak] wrt a: 2 * B_k * a
        # B_k[i,j] = 1 if i+j = k_peak, 0 otherwise
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k_peak - idx]

        # Project to tangent space of simplex
        grad = grad - np.dot(a, grad)

        # Euclidean step + projection onto simplex
        a_new = a - eta * grad
        # Project onto simplex {x: sum x = 1, x >= 0}
        a_new = np.sort(a_new)[::-1]
        v = np.arange(1, n + 1)
        css = np.cumsum(a_new) - v
        rho = np.where(a_new - css / n > 0)[0]
        if len(rho) == 0:
            a_new = best_a.copy()
        else:
            theta = (a_new[rho[-1]] + 1.0 / (rho[-1] + 1))
            a_new = np.maximum(a - eta * grad - theta, 0)
            a_new = a_new / a_new.sum()

        a = a_new

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SG iter={it} ({elapsed:.0f}s): "
                  f"eta={eta:.6f} score={best_score:.7f}")

    return best_a, best_score


def peak_shave_refined(a_init, time_limit, verbose=True, label=""):
    """Refined peak shaving: identify top contributors and reduce them,
    redistributing to positions that least increase other peaks."""
    n = len(a) if 'a' in dir() else len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the peak (prefer interior peaks)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        if peak_val <= best_score / (2.0 * n) + 1e-14:
            break

        # Find top contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_ci = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_ci:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue

            # Try reducing a[i] and redistributing to random positions
            for factor in [0.95, 0.9, 0.85]:
                for dest_count in [3, 5, 10]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    if diff < 1e-15:
                        continue

                    # Redistribute to least-connected positions
                    # Positions that don't contribute much to the current peak
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    # Score each position by how much it contributes to peak
                    peak_contrib = np.zeros(n)
                    for idx2 in idx:
                        peak_contrib[idx2] += a_trial[idx2] * a_trial[peak_k - idx2]
                    peak_contrib[i] = np.inf  # Don't put mass back here
                    # Choose positions with low peak contribution
                    candidates = np.where(mask)[0]
                    candidates = candidates[np.argsort(peak_contrib[candidates])][:dest_count]
                    if len(candidates) == 0:
                        continue
                    # Distribute proportionally to current values
                    dest_vals = a_trial[candidates]
                    dest_sum = dest_vals.sum()
                    if dest_sum < 1e-15:
                        a_trial[candidates] = diff / len(candidates)
                    else:
                        a_trial[candidates] = dest_vals + diff * dest_vals / dest_sum

                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try reducing both i and j simultaneously
            for ci in top_ci[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f in [0.9, 0.8]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random restart in neighborhood
            rng = np.random.default_rng(step * 97 + 13)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            if step > 200:
                break

    return best_a, best_score


def fourier_init(n, seed=42):
    """Generate initialization using Fourier domain ideas.
    Try to make the power spectrum P^2 as flat as possible."""
    rng = np.random.default_rng(seed)
    # Start with random spectrum, make it flat-ish
    N = 1 << (2 * n - 1).bit_length()
    P = np.ones(n) + rng.standard_normal(n) * 0.3
    P = np.abs(P)  # Make positive
    P = P / P.sum() * n  # Normalize so that sum of time-domain = 1
    # Convert to time domain
    a = np.zeros(n)
    # Use the fact that if we set the DFT, we get the time domain
    P_full = np.zeros(N, dtype=complex)
    P_full[:n] = P
    a_full = np.real(ifft(P_full))[:n]
    a = np.maximum(a_full, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def random_simplex_init(n, seed=42):
    """Random point on the simplex."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return a / a.sum()


def structured_init(n, pattern="sin", seed=42):
    """Structured initialization based on trigonometric patterns."""
    x = np.arange(n) / n
    if pattern == "sin":
        a = np.abs(np.sin(2 * np.pi * x)) ** 0.7
    elif pattern == "cos_asym":
        a = (1 + np.cos(2 * np.pi * x)) ** 0.5 * np.exp(-x * 0.5)
    elif pattern == "bump":
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.5 * np.exp(-8 * (x - 0.7) ** 2)
    elif pattern == "sawtooth":
        a = x ** 0.5 + 0.1
    elif pattern == "arch":
        a = np.sin(np.pi * x) ** 0.3
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Generate diverse initializations (parent, random, structured, Fourier)
    2. LP-based direction finding (main method)
    3. Multi-temperature L-BFGS in log-space
    4. Subgradient descent with peak targeting
    5. Peak shaving refinement
    6. Multi-length exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    # Initialize best
    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))

    n_parent = len(best_seq)

    # =========================================================================
    # Phase 1: LP-based optimization from parent (main method)
    # =========================================================================
    print(f"\n=== Phase 1: LP-based from parent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem * 0.5, 400.0)
        print(f"  LP from parent: n={n_parent}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_interpolation_step(best_seq, t_alloc, label="LP_parent")
        best_seq, best_score = check_and_update(a_lp, "LP_parent", best_seq, best_score)

    # LP again from current best with finer parameters
    rem = deadline - time.monotonic() - 500
    if rem > 40:
        t_alloc = min(rem * 0.4, 300.0)
        print(f"  LP round 2: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp2, _ = lp_interpolation_step(best_seq, t_alloc, label="LP_r2")
        best_seq, best_score = check_and_update(a_lp2, "LP_r2", best_seq, best_score)

    # =========================================================================
    # Phase 2: Multi-temperature L-BFGS
    # =========================================================================
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.5, 250.0)
        print(f"  L-BFGS: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lb, _ = lbfgs_log_space(best_seq, t_alloc,
                                   T_start=500, T_end=1, n_temps=10,
                                   maxiter_per_T=3000, label="LB_main")
        best_seq, best_score = check_and_update(a_lb, "LB_main", best_seq, best_score)

    # =========================================================================
    # Phase 3: Subgradient descent
    # =========================================================================
    print(f"\n=== Phase 3: Subgradient (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"  SG: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_sg, _ = subgradient_descent(best_seq, t_alloc,
                                       eta_start=0.005, eta_end=0.00005,
                                       label="SG")
        best_seq, best_score = check_and_update(a_sg, "SG", best_seq, best_score)

    # =========================================================================
    # Phase 4: Peak shaving
    # =========================================================================
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_alloc = min(rem * 0.5, 100.0)
        print(f"  PS: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_refined(best_seq, t_alloc, label="PS")
        best_seq, best_score = check_and_update(a_ps, "PS", best_seq, best_score)

    # =========================================================================
    # Phase 5: Alternative initializations and lengths
    # =========================================================================
    print(f"\n=== Phase 5: Alternative inits/lengths (best={best_score:.7f}) ===")

    # Try different lengths
    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0, 0.5]:
        alt_n = int(n_parent * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_parent and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)
    # Also try some "natural" lengths
    for cand_n in [512, 640, 896, 1024, 1280]:
        if 256 <= cand_n <= 4096 and cand_n != n_parent and cand_n not in alt_lengths:
            alt_lengths.append(cand_n)

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 80.0)

        # Interpolate from best
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = lp_interpolation_step(init_alt, t_alloc * 0.5,
                                          label=f"LP_n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"LP_n{alt_n}",
                                                 best_seq, best_score)

        rem = deadline - time.monotonic() - 120
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            t_alloc2 = min(rem * 0.4, 40.0)
            a_lb_alt, _ = lbfgs_log_space(best_seq, t_alloc2,
                                           T_start=200, T_end=5, n_temps=5,
                                           maxiter_per_T=2000,
                                           label=f"LB_n{alt_n}")
            best_seq, best_score = check_and_update(a_lb_alt, f"LB_n{alt_n}",
                                                     best_seq, best_score)

    # Try structured initializations at parent length
    rem = deadline - time.monotonic() - 100
    if rem > 40:
        n_try = n_parent
        for pattern, plbl in [("sin", "sin"), ("cos_asym", "cosA"),
                              ("bump", "bump"), ("sawtooth", "saw")]:
            rem = deadline - time.monotonic() - 80
            if rem < 25:
                break
            t_alloc = min(rem * 0.3, 50.0)
            init_s = structured_init(n_try, pattern=pattern)
            a_s, _ = lp_interpolation_step(init_s, t_alloc,
                                            label=f"LP_{plbl}")
            best_seq, best_score = check_and_update(a_s, f"LP_{plbl}",
                                                     best_seq, best_score)

    # Random initializations
    rem = deadline - time.monotonic() - 60
    if rem > 30:
        for rs in [123, 456]:
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 30.0)
            init_r = random_simplex_init(n_parent, seed=rs)
            a_r, _ = lp_interpolation_step(init_r, t_alloc,
                                            label=f"rand_{rs}")
            best_seq, best_score = check_and_update(a_r, f"rand_{rs}",
                                                     best_seq, best_score)

    # Fourier init
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t_alloc = min(rem * 0.5, 30.0)
        init_f = fourier_init(n_parent, seed=789)
        a_f, _ = lbfgs_log_space(init_f, t_alloc, T_start=100, T_end=5,
                                  n_temps=4, maxiter_per_T=1500,
                                  label="Fourier")
        best_seq, best_score = check_and_update(a_f, "Fourier",
                                                 best_seq, best_score)

    # =========================================================================
    # Phase 6: Intensive refinement on current best
    # =========================================================================
    print(f"\n=== Phase 6: Intensive Refinement (best={best_score:.7f}) ===")
    n_best = len(best_seq)

    # LP round 3
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_alloc = min(rem * 0.4, 80.0)
        print(f"  LP round 3: n={n_best}, budget={t_alloc:.0f}s")
        a_lp3, _ = lp_interpolation_step(best_seq, t_alloc, label="LP_r3")
        best_seq, best_score = check_and_update(a_lp3, "LP_r3", best_seq, best_score)

    # Low-T L-BFGS
    rem = deadline - time.monotonic() - 40
    if rem > 15:
        t_alloc = min(rem * 0.5, 50.0)
        print(f"  L-BFGS low T: budget={t_alloc:.0f}s")
        a_lb_low, _ = lbfgs_log_space(best_seq, t_alloc,
                                       T_start=50, T_end=0.5, n_temps=6,
                                       maxiter_per_T=3000, label="LB_lowT")
        best_seq, best_score = check_and_update(a_lb_low, "LB_lowT",
                                                 best_seq, best_score)

    # Peak shave round 2
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_alloc = min(rem * 0.5, 40.0)
        a_ps2, _ = peak_shave_refined(best_seq, t_alloc, label="PS2")
        best_seq, best_score = check_and_update(a_ps2, "PS2", best_seq, best_score)

    # Subgradient round 2 (very low step size)
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.5, 30.0)
        a_sg2, _ = subgradient_descent(best_seq, t_alloc,
                                        eta_start=0.001, eta_end=0.00001,
                                        label="SG2")
        best_seq, best_score = check_and_update(a_sg2, "SG2", best_seq, best_score)

    # =========================================================================
    # Phase 7: Final single-shot attempts
    # =========================================================================
    print(f"\n=== Phase 7: Final (best={best_score:.7f}) ===")

    # Very low T L-BFGS
    rem = deadline - time.monotonic() - 10
    if rem > 8:
        t_alloc = min(rem * 0.5, 20.0)
        a_fin, _ = lbfgs_log_space(best_seq, t_alloc, T_start=10, T_end=0.1,
                                    n_temps=4, maxiter_per_T=5000,
                                    label="final_LB")
        best_seq, best_score = check_and_update(a_fin, "final_LB",
                                                 best_seq, best_score)

    # Final peak shave
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_ps_fin, _ = peak_shave_refined(best_seq, min(rem * 0.7, 10.0),
                                          label="final_PS")
        best_seq, best_score = check_and_update(a_ps_fin, "final_PS",
                                                 best_seq, best_score)

    # Final LP micro-steps
    rem = deadline - time.monotonic() - 3
    if rem > 5:
        a_lp_fin, _ = lp_interpolation_step(best_seq, min(rem * 0.7, 8.0),
                                             label="final_LP")
        best_seq, best_score = check_and_update(a_lp_fin, "final_LP",
                                                 best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
