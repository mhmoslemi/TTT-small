    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def _lp_solver_found():
    for s in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(2, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


_lp_solver = None

def get_solver():
    global _lp_solver
    if _lp_solver is None:
        _lp_solver = _lp_solver_found()
    return _lp_solver


def solve_best_response(a, solver=None):
    """Solve LP: min M s.t. conv(a,b)[k] <= M, b>=0, sum(b)=1."""
    if solver is None:
        solver = get_solver()
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    cons = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt is solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)


def softmax_normalize(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def log_normalize(a):
    return np.log(np.maximum(a, 1e-300))


def interp_sequence(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new /= ss
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def score_normalized(a):
    """Score for a normalized (sum=1) non-negative sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lbgfs_smoothed(a0, time_limit, T_start=50.0, T_end=1.0, maxiter=5000):
    """L-BFGS on softmax-parameterized sequence with log-sum-exp smoothing."""
    n = len(a0)
    t0 = time.monotonic()
    z = log_normalize(a0.copy())
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(maxiter // n_stages, 100)
    
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        
        def func(z_val):
            a = softmax_normalize(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            return (2.0 * n / T_val) * (m + np.log(np.exp(Tb - m).sum()))
        
        def jac(z_val):
            a = softmax_normalize(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w /= w.sum()
            # grad of (2n/T) * log(sum exp(Tb_k)) w.r.t. a = 2 * sum_k w_k * grad_a (a*a)[k]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * T_val * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-14,
                                   'gtol': 1e-8, 'maxfun': iters_per_stage * 30,
                                   'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax_normalize(z)


def lp_alternate_nesterov(a0, time_limit, solver=None, max_iters=2000):
    """Nesterov-accelerated LP alternating optimization."""
    if solver is None:
        solver = get_solver()
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a /= a.sum()
    y = a.copy()  # virtual iterate
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2.0)
        
        # Nesterov update
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y /= y.sum()
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        a /= a.sum()
        
        sc_a = score_normalized(a)
        sc_y = score_normalized(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            print(f"    LP-Alt iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def active_set_smoothed_optimize(a0, time_limit, T_start=20.0, T_end=0.5, maxiter=3000):
    """
    Optimize a weighted sum of the top active lags using L-BFGS.
    The weights are re-identified at each stage.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a /= a.sum()
    t0 = time.monotonic()
    
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(maxiter // n_stages, 100)
    
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Identify active lags
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Build weighted objective
        def func(z_val):
            a_local = softmax_normalize(z_val)
            b_local = fast_convolve(a_local)
            w = np.zeros(2 * n - 1)
            w[active] = 1.0 / len(active)
            Tb = T_val * (w * b_local)
            m = Tb.max()
            return (2.0 * n / T_val) * (m + np.log(np.exp(Tb - m).sum()))
        
        def jac(z_val):
            a_local = softmax_normalize(z_val)
            b_local = fast_convolve(a_local)
            w = np.zeros(2 * n - 1)
            w[active] = 1.0 / len(active)
            Tb = T_val * (w * b_local)
            m = Tb.max()
            exp_Tb = np.exp(Tb - m)
            ww = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in active:
                if ww[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * T_val * ww[k] * a_local[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)
        
        z = log_normalize(a)
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxfun': iters_per_stage * 30,
                                   'disp': False})
            a = softmax_normalize(res.x)
        except Exception:
            pass
    
    return a


def peak_shave_local(a0, time_limit, max_steps=5000):
    """Careful local peak shaving with line search."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all near-peak lags
        threshold = max_b * 0.99
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for pk in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Sort by contribution (descending)
            top = np.argsort(contribs)[::-1][:8]
            
            for ci in top:
                i = idx[ci]
                j = pk - i
                if a[i] < 1e-14:
                    continue
                
                # Try reducing a[i] and redistributing
                for factor in [0.999, 0.998, 0.997, 0.995, 0.99, 0.98]:
                    new_val = a[i] * factor
                    diff = a[i] - new_val
                    if diff < 1e-16:
                        continue
                    
                    # Redistribute to other elements proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial /= a_trial.sum()
                    
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Try a different strategy: reduce two elements simultaneously
                found = False
                for pk in peaks[:3]:
                    js = max(0, pk - n + 1)
                    je = min(n - 1, pk)
                    idx = np.arange(js, je + 1)
                    contribs = a[idx] * a[pk - idx]
                    top = np.argsort(contribs)[::-1][:4]
                    for ci in range(len(top) - 1):
                        i1 = idx[top[ci]]
                        i2 = idx[top[ci+1]]
                        if i1 == i2:
                            continue
                        if a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        for f1 in [0.995, 0.99, 0.98]:
                            for f2 in [0.995, 0.99, 0.98]:
                                a_trial = a.copy()
                                a_trial[i1] *= f1
                                a_trial[i2] *= f2
                                a_trial = np.maximum(a_trial, 0)
                                a_trial /= a_trial.sum()
                                sc = score_normalized(a_trial)
                                if sc < best_score - 1e-15:
                                    a = a_trial
                                    best_score = sc
                                    best_a = a.copy()
                                    found = True
                                    improved = True
                                    break
                            if improved:
                                break
                        if improved:
                            break
                    if improved:
                        break
                if not found:
                    if no_improve > 500:
                        break
            continue
        
        no_improve = 0
    
    return best_a, best_score
