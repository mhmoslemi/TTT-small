Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combined approach: diverse FP runs, soup mixing, parametric fitting,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic
import math

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def parametric_score(params, n, harmonics):
    """Compute score for a parametric sequence."""
    c = params[:2*harmonics:2]
    phi = params[1:2*harmonics:2]
    i = np.arange(n)

    s = a.sum()
    if s < 1e-15:
        return 1e15
    a = a / s
    return 2.0 * n * fast_convolve(a).max()

    try:

def local_polish(a0, time_limit):

    for outer in range(200):

                others_sum = a[mask].sum()
                if others_sum < 1e-15:

                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

            else:

def soup_mix(population, weights=None):
    """Mix sequences from a population."""
    n = len(population[0])
    if weights is None:
        weights = np.ones(len(population)) / len(population)
    soup = np.zeros(n)
    for w, seq in zip(weights, population):
        s = seq.sum()
        if s > 1e-15:
            soup += w * seq / s
    soup = np.maximum(soup, 0)
    s = soup.sum()
    if s > 1e-15:
        soup = soup / s
    return soup

            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None):
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    b = a.copy()
    a_avg = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0
    no_improve = 0
    
    while monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        candidates = [a, b, 0.5*(a+b), a_avg]
        cand_improved = False
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_n = cand / s
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand_n.copy()
                no_improve = 0
                cand_improved = True
        if not cand_improved:
            no_improve += 1
        
        if no_improve > 500:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = a + np.random.standard_normal(n) * 0.002
            b = np.maximum(b, 0)
            b = b / b.sum()
            a_avg = a.copy()
            no_improve = 0
    
    return best_a, best_score, iters

def damped_lp_refine(a0, time_limit, solver=None, schedule="standard"):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    it = 0
    no_improve = 0
    
    while monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        
        if schedule == "standard":
            alpha = 2.0 / (it + 2)
        elif schedule == "slow":
            alpha = 1.0 / (it + 1)
        elif schedule == "fast":
            alpha = min(1.0, 3.0 / (it + 1))
        else:  # "anneal"
            alpha = max(0.05, 1.0 * (0.995 ** it))
        
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        
        if no_improve > 2000:
            # Restart from best with perturbation
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_score

def _parametric_func(params, n, harmonics):
    c = params[0:2*harmonics:2]
    phi = params[1:2*harmonics:2]
    i = np.arange(n)
    a = np.ones(n) / n
    for k in range(1, harmonics + 1):
        a = a + c[k-1] * np.cos(2.0*np.pi*k*i/n + phi[k-1])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return 1e15
    a = a / s
    return 2.0 * n * fast_convolve(a).max()
