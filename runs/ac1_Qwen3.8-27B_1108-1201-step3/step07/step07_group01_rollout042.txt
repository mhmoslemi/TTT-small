Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterated projection,

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Cross-correlation/convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ b = convolve(a, b)."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        # conv(a,b)[k] = sum_j a[k-j] * b[j] for valid j
        # So A[k, j] = a[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A


def lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return a.copy()


def lp_optimize(a_init, time_limit, t0=0.5, verbose=True, label=""):
    """LP-based iterative optimization."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    M = fast_convolve(a).max()
    best_a = a.copy()
    best_M = M
    t_val = t0
    t0_start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0_start < time_limit:
        it += 1
        b_opt = lp_step(a, M)
        s_b = b_opt.sum()
        if s_b < 1e-15:
            b_opt = a.copy()
            s_b = 1.0
        b_norm = b_opt / s_b  # normalize to sum 1
        
        a_new = (1 - t_val) * a + t_val * b_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        M_new = fast_convolve(a_new).max()
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
            # Can increase step size

        else:

            if no_improve > 20:
                t_val *= 0.8
                if t_val < 0.01:
                    break

            M_new = best_M
        
        a = a_new
        M = M_new

            elapsed = time.monotonic() - t0_start
            print(f"    {label} LP iter={it} ({elapsed:.0f}s): M={M:.8f} best_M={best_M:.8f}")
    
    return best_a, best_M
