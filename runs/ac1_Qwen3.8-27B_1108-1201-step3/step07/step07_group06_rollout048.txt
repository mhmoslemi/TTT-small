import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_grad_conv_weights(a, w, n, fft_size):
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def adam_logspace_v2(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                     beta1=0.9, beta2=0.999, eps=1e-8, schedule='geo_plateau',
                     plateau_frac=0.3, use_variance=False, var_weight=0.5,
                     verbose=False, verbose_interval=5000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000
    lr_current = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'geo_plateau':
            if progress < plateau_frac:
                T = T_init
            else:
                p2 = (progress - plateau_frac) / max(1e-10, 1.0 - plateau_frac)
                T = T_init * (T_final / T_init) ** p2
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'cosine':
            T = T_final + 0.5 * (T_init - T_final) * (1 + np.cos(np.pi * progress))
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()

        if use_variance:
            b_clipped = b.copy()
            exp_b = np.exp(b / T - max_b / T)
            w = exp_b / exp_b.sum()
            var_component = var_weight * T * np.log(np.sum(exp_b * (b - np.sum(w * b)) ** 2 / np.sum(exp_b) + 1e-15))
        else:
            exp_b = np.exp(b / T - max_b / T)
            w = exp_b / exp_b.sum()
            var_component = 0.0

        grad_a = compute_grad_conv_weights(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr_current * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.008
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr_current = lr_current * 0.8

        lr_current = max(0.0001, lr_current * (1.0 - 0.00005))

        if verbose and t % verbose_interval == 0:
            print(f"      Adam v2 t={t} T={T:.5f} lr={lr_current:.6f} score={best_score:.7f}")

    return best_a, best_score


def subspace_search(a0, time_limit, n_modes=8, n_restarts=5):
    n = len(a0)
    base = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    base = base / base.sum()
    best_a = base.copy()
    best_score = get_internal_score(base)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    t = np.arange(n) / n
    cos_modes = []
    sin_modes = []
    for k in range(1, n_modes + 1):
        cos_modes.append(np.cos(2 * np.pi * k * t))
        sin_modes.append(np.sin(2 * np.pi * k * t))

    all_modes = cos_modes + sin_modes
    n_params = len(all_modes)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart == 0:
            params0 = np.zeros(n_params)
        else:
            params0 = rng.standard_normal(n_params) * 0.05

        def param_obj(params):
            a_trial = base.copy()
            for idx in range(n_params):
                a_trial += params[idx] * all_modes[idx]
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                return 1e10
            a_trial = a_trial / s
            conv = fast_convolve(a_trial)
            b = conv
            T = 0.01
            max_b = b.max()
            exp_b = np.exp(b / T - max_b / T)
            smax = T * (max_b / T + np.log(exp_b.sum()))
            return 2.0 * n * smax

        def param_grad(params):
            a_trial = base.copy()
            for idx in range(n_params):
                a_trial += params[idx] * all_modes[idx]
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                return np.zeros(n_params)
            a_trial = a_trial / s
            conv = fast_convolve(a_trial)
            T = 0.01
            max_b = conv.max()
            exp_b = np.exp(conv / T - max_b / T)
            w = exp_b / exp_b.sum()
            fft_size = 1 << (2 * n - 2).bit_length()
            grad_a = compute_grad_conv_weights(a_trial, w, n, fft_size)
            g = np.zeros(n_params)
            for idx in range(n_params):
                g[idx] = np.dot(grad_a, all_modes[idx]) - np.dot(grad_a, a_trial) * np.dot(all_modes[idx], a_trial)
            return g

        try:
            res = minimize(
                fun=param_obj,
                x0=params0,
                jac=param_grad,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'maxfun': 50000, 'ftol': 1e-15, 'gtol': 1e-12, 'disp': False}
            )
            params_opt = res.x
            a_trial = base.copy()
            for idx in range(n_params):
                a_trial += params_opt[idx] * all_modes[idx]
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                sc = get_internal_score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
        except Exception:
            pass

        if time.monotonic() - t0 > time_limit:
            break

        if restart > 0:
            base = best_a.copy()

    return best_a, best_score


def peak_aware_coordinate_descent(a0, time_limit, n_sweeps=50):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)

        for i in order:

            if a[i] < 1e-15:
                continue

            contrib = 0.0
            for k in range(n):
                j = i + k
                if j < n:
                    contrib += b[k] * 0  # placeholder
                j2 = i - k
                if j2 >= 0 and k <= n - 1:
                    pass

            factors = [0.8, 0.9, 1.1, 1.2]
            best_local = a[i]
            best_local_score = best_score

            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                diff = old_val - a_trial[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = get_internal_score(a_trial)
                if score < best_local_score:
                    best_local_score = score
                    best_local = a_trial[i]

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved and sweep > 5:
            break

    return best_a, best_score


def gradient_flow_refinement(a0, time_limit, n_steps=50000, lr_init=0.002, lr_final=0.0001):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_internal_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    T_current = 0.02
    momentum = np.zeros(n)
    beta_m = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        progress = step / max(1, n_steps)
        T = T_current * (0.001 / T_current) ** progress
        lr = lr_init * (lr_final / lr_init) ** progress

        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_grad_conv_weights(a, w, n, fft_size)
        S = np.dot(grad_a, a)
        grad = a * (grad_a - S)

        momentum = beta_m * momentum + (1 - beta_m) * grad
        a = a - lr * momentum
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        score = get_internal_score(a)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            ('cosine', 0.03, 0.001, 0.0),
            ('exp', 0.05, 0.003, 0.0),
        ]:
            rem = deadline - time.monotonic() - 350
            if rem < 15:
                break
            t_sub = min(t_alloc / 3.0, rem * 0.8)
            if t_sub < 10:
                continue

            a_res, score_res = adam_logspace_v2(
                init_a, t_sub, T_init=T_i, T_final=T_f,
                lr=0.006, schedule=sched, plateau_frac=plate,
                verbose=False
            )
            best_seq, best_score = evaluate_and_track(a_res, f"adam_{label}_{sched}", best_seq, best_score)

    print(f"  After Phase 1: best={best_score:.7f}")

    # === Phase 2: Subspace search (20% time) ===
    print(f"\n=== Phase 2: Subspace Search ===")
    t_p2 = (deadline - time.monotonic()) * 0.20
    a_sub, _ = subspace_search(best_seq, t_p2, n_modes=10, n_restarts=4)
    best_seq, best_score = evaluate_and_track(a_sub, "subspace", best_seq, best_score)

    # Run Adam from subspace result
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_sub = min(rem * 0.5, 120.0)
        a_res, _ = adam_logspace_v2(
            best_seq, t_sub, T_init=0.02, T_final=0.001,
            lr=0.004, schedule='geo_plateau', plateau_frac=0.3
        )
        best_seq, best_score = evaluate_and_track(a_res, "adam_after_sub", best_seq, best_score)

    print(f"  After Phase 2: best={best_score:.7f}")

    # === Phase 3: Intensive local refinement (25% time) ===
    print(f"\n=== Phase 3: Local Refinement ===")
    t_p3 = (deadline - time.monotonic()) * 0.25

    # Gradient flow
    t_sub = min(t_p3 * 0.4, 100.0)
    a_gf, _ = gradient_flow_refinement(best_seq, t_sub, n_steps=100000, lr_init=0.001, lr_final=0.00005)
    best_seq, best_score = evaluate_and_track(a_gf, "grad_flow", best_seq, best_score)

    # Coordinate descent
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_sub = min(rem * 0.4, 80.0)
        a_cd, _ = peak_aware_coordinate_descent(best_seq, t_sub, n_sweeps=30)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)

    # More Adam from improved point
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_sub = min(rem * 0.5, 80.0)
        a_res, _ = adam_logspace_v2(
            best_seq, t_sub, T_init=0.01, T_final=0.0005,
            lr=0.003, schedule='cosine', plateau_frac=0.0
        )
        best_seq, best_score = evaluate_and_track(a_res, "adam_polish1", best_seq, best_score)

    # Simulated annealing
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_sub = min(rem * 0.5, 40.0)
        a_sa, _ = simulate_annealing_local(best_seq, t_sub, n_iters=100000, T_start=0.0005, T_end=0.00001)
        best_seq, best_score = evaluate_and_track(a_sa, "sim_anneal", best_seq, best_score)

    print(f"  After Phase 3: best={best_score:.7f}")

    # === Phase 4: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    t_p4 = (deadline - time.monotonic()) * 0.15
    n_best = len(best_seq)

    alt_lengths = []
    for factor in [1.333, 0.75, 1.5, 1.25]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        t_sub = min(t_p4 / 3.0, rem * 0.8)
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = adam_logspace_v2(
            init_alt, t_sub, T_init=0.04, T_final=0.002,
            lr=0.007, schedule='geo_plateau', plateau_frac=0.25
        )
        best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
        print(f"  Alt n={alt_n}: {best_score:.7f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic() - 10

    if rem > 30:
        t_sub = min(rem * 0.4, 80.0)
        a_fin, _ = adam_logspace_v2(
            best_seq, t_sub, T_init=0.008, T_final=0.0003,
            lr=0.002, schedule='cosine', plateau_frac=0.0
        )
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 20:
        t_sub = min(rem * 0.6, 50.0)
        a_cd2, _ = peak_aware_coordinate_descent(best_seq, t_sub, n_sweeps=20)
        best_seq, best_score = evaluate_and_track(a_cd2, "final_cd", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.7, 30.0)
        a_gf2, _ = gradient_flow_refinement(best_seq, t_sub, n_steps=50000, lr_init=0.0005, lr_final=0.0001)
        best_seq, best_score = evaluate_and_track(a_gf2, "final_gf", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.7f}")

    return [float(x) for x in best_seq]
