"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using efficient FFT-based gradient
computation with multi-temperature L-BFGS annealing, active-set peak
equalization, and adaptive local search. Key innovation: O(N log N) gradient
via cross-correlation in frequency domain, enabling thousands of optimization
steps within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, T):
    """
    Compute gradient of the smoothed objective f(a) = 2n/T * log(sum exp(T*c[k]))
    using FFT-based cross-correlation. O(N log N) where N ~ 4n.
    Returns (gradient, smoothed_value, c, w)
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Soft-max weights
    max_c = c.max()
    scaled = T * (c - max_c)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp
    
    # Smoothed value: 2n/T * (max_c + log(mean(exp(T*(c - max_c)))))
    smoothed_val = 2.0 * n / T * (max_c + np.log(sum_exp / len(c)))
    
    # Gradient: 4n * (a *cross_corr* w)[:n]
    # (a cross_corr w)[i] = sum_m a[m] * w[m+i]
    N = 1 << (3 * n - 2).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    cross = np.real(ifft(A * np.conj(W)))[:n]
    
    grad = 4.0 * n * cross
    return grad, smoothed_val, c, w


def softmax_normalize(z):
    """Convert log-space to probability simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def lbgfs_fft(z0, T, maxiter=2000, time_budget=None, t_start=None, ftol=1e-14, gtol=1e-10):
    """
    L-BFGS in log-space with FFT-based gradient.
    z is the log of the sequence (unnormalized).
    """
    n = len(z0)
    if time_budget is None:
        time_budget = 600.0
    if t_start is None:
        t_start = monotonic()
    
    stopped = [False]
    call_count = [0]

    def func(z):
        if monotonic() - t_start > time_budget:
            stopped[0] = True
            return 1e15
        call_count[0] += 1
        a = softmax_normalize(z)
        grad, val, c, w = compute_gradient_fft(a, T)
        return val

    def jac(z):
        if stopped[0]:
            return np.zeros(n)
        a = softmax_normalize(z)
        grad_a, val, c, w = compute_gradient_fft(a, T)
        # Transform to z-space: grad_z = a * (grad_a - a * dot(grad_a, a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if stopped[0]:
            return z0, res
        return res.x, res
    except Exception:
        return z0, None

def subgradient_fft(a0, time_budget, lr_init=0.005, momentum=0.9, top_k=50):
    """
    Subgradient method with FFT-based gradient and momentum.
    Focuses on the top peaks for efficiency.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t_start = monotonic()
    no_improve = 0
    
    while monotonic() - t_start < time_budget:
        c = fast_convolve(a)
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        
        # Compute weights for top peaks
        pv = c[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w_peak = np.exp(scaled)
        w_peak = w_peak / w_peak.sum()
        
        # Build sparse weight vector
        w_full = np.zeros(len(c))
        w_full[peaks] = w_peak
        
        # Compute gradient using FFT cross-correlation
        N = 1 << (3 * n - 2).bit_length()
        A = fft(a, N)
        W = fft(w_full, N)
        cross = np.real(ifft(A * np.conj(W)))[:n]
        sg = 4.0 * n * cross
        
        v = momentum * v - lr * sg
        a_candidate = a + v
        
        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        if no_improve > 500:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.004)
            no_improve = 0
    
    return best_a, best_score


def peak_equalize(a0, time_budget, n_peaks=100, n_rounds=50):
    """
    Active-set peak equalization: iteratively identify top peaks and
    solve a least-squares system to find equalization directions.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = monotonic()
    
    for round_idx in range(n_rounds):
        if monotonic() - t_start > time_budget:
            break
        
        c = fast_convolve(a)
        M = c.max()
        
        # Find peaks near the maximum
        threshold = M - 0.05 * M
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > n_peaks:
            # Keep only the top n_peaks
            peak_indices = peak_indices[np.argsort(c[peak_indices])[::-1][:n_peaks]]
        
        if len(peak_indices) < 2:
            break
        
        # Build the system: find lambda such that sum_j lambda_j * a[k_j - i] = const
        # For i where a[i] > 0
        active_mask = a > 1e-10
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) < 2:
            break
        
        # Build matrix B: B[i, j] = a[k_j - active_idx[i]] (if valid)
        B = np.zeros((len(active_idx), len(peak_indices)))
        for j, k in enumerate(peak_indices):
            diff = k - active_idx
            valid = (diff >= 0) & (diff < n)
            B[valid, j] = a[diff[valid]]
        
        # Solve least squares: B @ lambda ≈ mu * 1
        # Normalize: minimize ||B @ lambda - mu * 1||^2
        # Use augmented system
        m = len(active_idx)
        p = len(peak_indices)
        A_aug = np.zeros((m + 1, p + 1))
        A_aug[:m, :p] = B
        A_aug[m, p] = 1.0
        b_aug = np.zeros(m + 1)
        b_aug[m] = 1.0
        
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_aug, b_aug, rcond=None)
            lambdas = np.maximum(sol[:p], 0)
            mu = sol[p]
        except Exception:
            break
        
        if lambdas.max() < 1e-15:
            break
        
        # Direction: d[i] = -2 * sum_j lambda_j * a[k_j - i]
        d = np.zeros(n)
        for j, k in enumerate(peak_indices):
            if lambdas[j] < 1e-15:
                continue
            diff = k - np.arange(n)
            valid = (diff >= 0) & (diff < n)
            d[valid] -= 2.0 * lambdas[j] * a[diff[valid]]
        
        # Normalize direction
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Line search
        best_alpha = 0
        best_s = best_score
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            s_val = 2.0 * n * fast_convolve(a_trial).max()
            if s_val < best_s - 1e-12:
                best_s = s_val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = a + best_alpha * d
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_s < best_score - 1e-12:
                best_score = best_s
                best_a = a.copy()
        else:
            # Try smaller steps
            found = False
            for alpha in [0.0001, 0.0005, 0.001]:
                a_trial = a + alpha * d
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                s_val = 2.0 * n * fast_convolve(a_trial).max()
                if s_val < best_score - 1e-12:
                    a = a_trial
                    best_score = s_val
                    best_a = a.copy()
                    found = True
                    break
            if not found:
                # Perturb and continue
                a = best_a + np.random.standard_normal(n) * 0.0005
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_score


def coordinate_descent(a0, time_budget, n_passes=200):
    """
    Coordinate descent: try adjusting individual elements and pairs.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = monotonic()
    rng = np.random.RandomState(123)
    
    for _ in range(n_passes):
        if monotonic() - t_start > time_budget:
            break
        
        improved = False
        
        # Try individual coordinate adjustments
        for _ in range(50):
            i = rng.randint(n)
            if a[i] < 1e-15:
                continue
            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                old = a[i]
                a_trial[i] = old * factor
                diff = a_trial[i] - old
                if abs(diff) < 1e-15:
                    continue
                # Redistribute to maintain sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Try pair adjustments
        for _ in range(30):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or a[i] < 1e-15 or a[j] < 1e-15:
                continue
            for f1 in [0.5, 0.7, 1.3, 1.5]:
                for f2 in [0.5, 0.7, 1.3, 1.5]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved and rng.random() < 0.1:
            # Random perturbation
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic splines."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(old_x, f_norm, bc_type='natural')
        f_interp = cs(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
