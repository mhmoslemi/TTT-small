g[j] = 2 * sum_k w[k] * h[k-j]

This is 2 * (w * h_rev)[n-1-j] where h_rev is the reversal of h... this is getting confusing.

Let me just use the direct computation with the understanding that for each active k, I add w[k] * 2 * f[k-j] to g[j] for valid j. Since the number of active peaks is small (maybe 50-200), and n is at most a few thousand, this is O(|S| * n) which is fine.

Or I can use FFT: define a vector W of length 2n-1 where W[k] = w[k] for active k, else 0. Then compute the cross-correlation of W with f.

Actually, let me just use the direct approach. It's simpler and fast enough for the sizes we're dealing with.

"""
Hybrid optimization: LP + Mirror Descent + Newton polish for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peak(f, k):
    """Compute the subgradient of b[k] = sum_i f[i]*f[k-i] w.r.t. f."""
    n = len(f)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * f[k - idx]
    return g


def compute_combined_subgradient(f, active_idx, weights):
    """Compute weighted sum of subgradients for active peaks."""
    n = len(f)
    g = np.zeros(n)
    for i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += weights[i] * 2.0 * f[k - idx]
    return g


def mirror_descent_step(f, b, eta, n_top, peak_tol=0.01):
    """
    One mirror descent step with entropy regularization.
    f: current iterate (on simplex)
    b: auto-convolution of f
    eta: step size
    n_top: number of peaks to consider
    peak_tol: tolerance for considering a peak "active"
    """
    n = len(f)
    b_max = b.max()
    
    # Select active peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    # Filter to those close to the max
    threshold = b_max * (1.0 - peak_tol)
    active_mask = b[top_idx] >= threshold
    active_idx = top_idx[active_mask]
    
    if len(active_idx) == 0:
        active_idx = np.array([top_idx[np.argmax(b[top_idx])]])
    
    # Weights: proportional to peak height (or uniform)
    w = b[active_idx]
    w = w / w.sum()
    
    # Compute subgradient
    g = compute_combined_subgradient(f, active_idx, w)
    
    # Mirror descent update: f_new[i] = f[i] * exp(-eta * g[i]) / Z
    # To avoid overflow, shift g
    g_shifted = g - g.max()
    log_f_new = np.log(f + 1e-30) + eta * g_shifted  # wait, sign...
    
    # Actually: f_new[i] = f[i] * exp(-eta * g[i]) / Z
    # log(f_new[i]) = log(f[i]) - eta * g[i] - log(Z)
    # To avoid underflow/overflow, shift:
    scores = np.log(f + 1e-30) - eta * g
    scores -= scores.max()
    f_new = np.exp(scores)
    f_new = f_new / f_new.sum()
    f_new = np.maximum(f_new, 0.0)
    
    return f_new, active_idx, w


def run_mirror_descent(f_init, time_limit, eta_init=0.5, eta_decay=0.9999, 
                       n_top=100, peak_tol=0.05, label="md"):
    """Mirror descent phase."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = eta_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        f_new, active_idx, w = mirror_descent_step(f, b, eta, n_top, peak_tol)
        
        # Line search: try several step sizes
        best_trial = f
        best_trial_obj = best_obj
        b = fast_convolve(f)
        
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            eta_trial = eta * scale
            f_trial, _, _ = mirror_descent_step(f, b, eta_trial, n_top, peak_tol)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_trial = f_trial
        
        f = best_trial
        
        cur_obj = best_trial_obj
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                eta *= eta_decay
                no_improve = 0
            
            if no_improve > 200:
                break
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "

    return best_f, best_obj


def lp_solve_reduced(f, n, top_k, eps=1e-8, solver=cp.SCS):
    """Solve reduced LP for best non-negative direction."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=solver, max_iters=200000, eps=eps)
    except Exception:
        return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp_phase(f_init, time_limit, top_k_init=30, top_k_max=150, label="lp"):
    """LP-based optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 40 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        g, M = lp_solve_reduced(f, n, top_k)
        if g is None:
            g, M = lp_solve_reduced(f, n, min(2 * top_k, 2 * n - 1))
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t = 0.1
        best_t_obj = fast_convolve(f).max()

        for t in [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def newton_polish(f_init, time_limit, active_tol=0.005, max_active=200):
    """
    Newton-type polish: solve linearized KKT system for active peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Find active set
        threshold = M * (1.0 - active_tol)
        active_idx = np.where(b >= threshold)[0]
        if len(active_idx) > max_active:
            # Keep only the largest
            top_from_active = np.argsort(b[active_idx])[::-1][:max_active]
            active_idx = active_idx[top_from_active]
        
        K = len(active_idx)
        
        # Build the linear system: 2 * A * delta = rhs, sum(delta) = 0
        # where A[i,j] = f[k_i - j] for valid j
        A = np.zeros((K, n))
        for i, k in enumerate(active_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        A = 2.0 * A
        rhs = M - b[active_idx]
        
        # Solve min ||delta||^2 s.t. A*delta = rhs, sum(delta) = 0
        # Using Lagrange multipliers: [A'; 1'] delta = [rhs; 0]
        # Minimum norm: delta = [A; 1']^+ [rhs; 0]
        # Or solve the KKT system:
        # A^T lambda + mu * 1 = 0  (from minimizing 0.5*||delta||^2)
        # A delta = rhs
        # 1^T delta = 0
        
        # Actually, let's use the projected approach:
        # delta = A^T * lambda + mu * 1
        # Substitute: A * A^T * lambda + A * mu * 1 = rhs
        #             1^T * A^T * lambda + n * mu = 0
        
        AA = A @ A.T  # K x K
        A1 = A @ np.ones(n)  # K-vector: A1[i] = sum_j A[i,j]
        n_vec = float(n)
        
        # System: [AA, A1; A1^T, 0] [lambda; mu] = [rhs; 0]
        Kkt = np.zeros((K+1, K+1))
        Kkt[:K, :K] = AA
        Kkt[:K, K] = A1
        Kkt[K, :K] = A1
        Kkt[K, K] = 0.0
        
        rhs_full = np.zeros(K+1)
        rhs_full[:K] = rhs
        
        try:
            sol = np.linalg.solve(Kkt, rhs_full)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(Kkt, rhs_full, rcond=None)[0]
            except:
                break
        
        lam = sol[:K]
        mu = sol[K]
        
        delta = A.T @ lam + mu * np.ones(n)
        
        # Line search
        best_alpha = 0.01
        best_alpha_obj = M
        for alpha in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1]:
            f_trial = f + alpha * delta
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_alpha_obj:
                best_alpha_obj = obj
                best_alpha = alpha
        
        if best_alpha_obj >= M:
            # No improvement, try smaller steps
            for alpha in [0.001, 0.002, 0.003]:
                f_trial = f + alpha * delta
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_alpha_obj:
                    best_alpha_obj = obj
                    best_alpha = alpha
        
        if best_alpha_obj >= M - 1e-14:
            # No improvement possible with this active set
            active_tol *= 2  # Be more inclusive
            if active_tol > 0.1:
                break
        
        f = f + best_alpha * delta
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            active_tol = max(active_tol * 0.9, 0.001)  # Tighten
        else:
            active_tol = min(active_tol * 1.5, 0.1)
        
        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    newton iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} K={K} tol={active_tol:.4f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.7, 0.85, 0.95, 1.05, 1.15, 1.3])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj

"""
Hybrid optimization: LP + Mirror Descent + Newton KKT polish + Coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses entropy-regularized mirror descent as a novel exploration mechanism
complementary to LP direction finding, plus a Newton-type KKT solver for final convergence.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, active_idx, weights):
    """Compute weighted sum of subgradients of b[k] for active peaks."""
    n = len(f)
    g = np.zeros(n)
    for i, k in enumerate(active_idx):
        w = weights[i]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w * 2.0 * f[k - idx]
    return g


def mirror_descent_step(f, b, eta, n_top, peak_tol):
    """
    One entropy-regularized mirror descent step.
    Returns (f_new, obj_new, n_active).
    """
    n = len(f)
    b_max = b.max()

    # Select active peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    threshold = b_max * (1.0 - peak_tol)
    active_mask = b[top_idx] >= threshold
    active_idx = top_idx[active_mask]

    if len(active_idx) == 0:
        best_peak = top_idx[np.argmax(b[top_idx])]
        active_idx = np.array([best_peak])

    # Weights proportional to peak height
    w = b[active_idx]
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active_idx)) / len(active_idx)
    else:
        w = w / w_sum

    # Compute subgradient
    g = compute_subgradient(f, active_idx, w)

    # Mirror descent: f_new[i] = f[i] * exp(-eta * g[i]) / Z
    log_f = np.where(f > 1e-30, np.log(f), -30.0)
    scores = log_f - eta * g
    scores -= scores.max()
    f_new = np.exp(scores)
    f_new_sum = f_new.sum()
    if f_new_sum < 1e-30:
        f_new = f.copy()
    else:
        f_new = f_new / f_new_sum
    f_new = np.maximum(f_new, 0.0)

    obj = fast_convolve(f_new).max()
    return f_new, obj, len(active_idx)


def run_mirror_descent(f_init, time_limit, eta_init=1.0, n_top=150, peak_tol=0.03, label="md"):
    """Mirror descent with backtracking line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = eta_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Backtracking line search on eta
        best_trial = f
        best_trial_obj = best_obj
        found = False

        for scale in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            eta_trial = eta * scale
            f_trial, obj_trial, _ = mirror_descent_step(f, b, eta_trial, n_top, peak_tol)
            if obj_trial < best_trial_obj - 1e-15:
                best_trial = f_trial
                best_trial_obj = obj_trial
                found = True
                break

        if not found:
            no_improve += 1
            if no_improve > 30:
                eta *= 0.7
                no_improve = 0
            if eta < 1e-4:
                break
            continue

        f = best_trial
        no_improve = 0

        if best_trial_obj < best_obj - 1e-14:
            best_obj = best_trial_obj
            best_f = f.copy()
            # Casually increase eta if improving consistently

        else:
            no_improve += 1
            if no_improve > 10:
                eta *= 0.8
                no_improve = 0

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} eta={eta:.4f} t={elapsed:.0f}s")

    return best_f, best_obj


def lp_solve_reduced(f, n, top_k, eps=1e-8):
    """Solve reduced LP: max sum(g) s.t. g>=0, A@g <= M."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=100000, abs_acceleration_limit=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp_phase(f_init, time_limit, top_k_init=30, top_k_max=150, label="lp"):
    """LP-based optimization with adaptive peak count and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 40 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        g, M = lp_solve_reduced(f, n, top_k)
        if g is None:
            g, M = lp_solve_reduced(f, n, min(2 * top_k, 2 * n - 1))
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g

        # Line search with more granular steps
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()
        for t in [0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 60:
                break

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def newton_kkt_polish(f_init, time_limit, active_tol=0.003, max_active=120, reg=1e-10):
    """
    Newton-type KKT polish: solve the linearized KKT system for active peaks.
    At the optimum, for active peaks k_i: 2*sum_j f[k_i-j]*delta[j] = M - b[k_i]
    with constraint sum(delta) = 0. Minimum-norm solution via KKT system.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Find active set
        threshold = M * (1.0 - active_tol)
        active_idx = np.where(b >= threshold)[0]
        if len(active_idx) > max_active:
            order = np.argsort(b[active_idx])[::-1][:max_active]
            active_idx = active_idx[order]

        K = len(active_idx)
        if K < 2:
            break

        # Build A: A[i, j] = f[k_i - j] for valid j (this is the "Jacobian/2")
        A = np.zeros((K, n))
        for i, k in enumerate(active_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi + 1] = f[lo:hi + 1]

        # Linearized system: 2*A*delta = rhs where rhs = M - b[active]
        rhs = M - b[active_idx]

        # Minimum-norm solution of 2*A*delta = rhs, 1^T*delta = 0
        # KKT: [2A; 1^T] delta = [rhs; 0]
        # Min-norm: delta = [2A; 1^T]^T * ([2A; 1^T][2A; 1^T]^T + reg*I)^{-1} * [rhs; 0]
        # Equivalent to solving:
        # (2A)(2A)^T * lam + (2A)*mu*1 = rhs
        # 1^T*(2A)^T * lam + n*mu = 0

        AA = (2.0 * A) @ (2.0 * A).T  # K x K
        A1 = (2.0 * A) @ np.ones(n)   # K-vector
        n_float = float(n)

        KKT = np.zeros((K + 1, K + 1))
        KKT[:K, :K] = AA + reg * np.eye(K)
        KKT[:K, K] = A1
        KKT[K, :K] = A1
        KKT[K, K] = 0.0

        rhs_full = np.zeros(K + 1)
        rhs_full[:K] = rhs

        try:
            sol = np.linalg.solve(KKT, rhs_full)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(KKT, rhs_full, rcond=None)[0]
            except Exception:
                break

        lam = sol[:K]
        mu = sol[K]

        delta = (2.0 * A).T @ lam + mu * np.ones(n)

        # Line search
        best_alpha = 0.0
        best_alpha_obj = M
        for alpha in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2]:
            f_trial = f + alpha * delta
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_alpha_obj:
                best_alpha_obj = obj
                best_alpha = alpha

        if best_alpha == 0.0:
            # Try very small steps
            for alpha in [0.001, 0.002, 0.003, 0.005]:
                f_trial = f + alpha * delta
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_alpha_obj:
                    best_alpha_obj = obj
                    best_alpha = alpha

        if best_alpha == 0.0:
            stuck_count += 1
            active_tol = min(active_tol * 1.5, 0.05)
            if stuck_count > 20 or active_tol > 0.05:
                # Try with regularization increase
                reg *= 10
                if reg > 1e-4:
                    break
            continue

        f = f + best_alpha * delta
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck_count = 0
            active_tol = max(active_tol * 0.85, 0.0005)
        else:
            stuck_count += 1
            active_tol = min(active_tol * 1.2, 0.05)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    newton iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} K={K} tol={active_tol:.5f} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements with golden-ratio multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.6, 0.8, 0.92, 0.97, 1.03, 1.08, 1.2, 1.5])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj


def soft_mirror_descent(f_init, time_limit, tau_init=0.001, n_peaks=200, label="smd"):
    """
    Smoothed mirror descent using log-sum-exp approximation of max.
    Uses Adam optimizer in the log-domain for better convergence.
    """
    t0 = time.monotonic()
    n = len(f_init)
    # Work in log-domain: u[i] = log(f[i]), f[i] = softmax(u)
    f = f_init.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    u = np.log(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    tau = tau_init
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        f = np.exp(u - u.max())
        f_sum = f.sum()
        f = f / f_sum

        b = fast_convolve(f)
        b_max = b.max()

        # Soft weights via log-sum-exp
        b_scaled = b / tau
        b_scaled -= b_scaled.max()
        p = np.exp(b_scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(b))
            p[b.argmax()] = 1.0
        else:
            p = p / p_sum

        # Focus on top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
            p_other = p.copy()
            p = np.zeros(len(b))
            p[top_idx] = p_other[top_idx]
            p = p / p.sum()

        # Gradient of log-sum-exp w.r.t. f: (1/tau) * sum_k p[k] * 2*f[k-i]
        grad_f = np.zeros(n)
        # Only process significant weights
        sig_mask = p > 1e-10
        sig_idx = np.where(sig_mask)[0]
        for k in sig_idx:
            wi = p[k] * 2.0 / tau
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad_f[idx] += wi * f[k - idx]

        # Convert to log-domain gradient: grad_u[i] = grad_f[i] * f[i] - (sum grad_f*f) * f[i]
        # Actually for softmax parameterization: d/d u[i] = f[i] * (grad_f[i] - f dot grad_f)
        grad_dot = np.dot(f, grad_f)
        grad_u = f * (grad_f - grad_dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        u = u - lr * m_hat / (np.sqrt(v_hat_hat) + 1e-8)

        # Convert back
        f_new = np.exp(u - u.max())
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
        else:
            # Revert if worse
            f_new = best_f.copy()
            u = np.log(np.maximum(f_new, 1e-15))

        f = f_new

        # Decay tau and lr
        if t % 50 == 0:
            tau = min(tau * 1.05, 0.05)
            lr *= 0.995

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={t} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} tau={tau:.5f} lr={lr:.4f} t={elapsed:.0f}s")

    return best_f, best_obj

            print(f"  NEW BEST [md]: {score2:.6f}")

    # === Phase 3: LP deep (more peaks) ===
    print(f"\n=== Phase 3: LP deep (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 300.0)
    if t3 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, top_k=60->200")
        f3, obj3 = run_lp_phase(best_seq, time_limit=t3, top_k_init=60,
                                top_k_max=200, label="lp_deep")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp_deep]: {score3:.6f}")

    # === Phase 4: Soft Mirror Descent (smoothed) ===
    print(f"\n=== Phase 4: Soft MD (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 200.0)
    if t4 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, obj4 = soft_mirror_descent(best_seq, time_limit=t4, tau_init=0.002,
                                       n_peaks=150, label="smd")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [smd]: {score4:.6f}")

    # === Phase 5: Newton KKT polish ===
    print(f"\n=== Phase 5: Newton KKT (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 250.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, obj5 = newton_kkt_polish(best_seq, time_limit=t5, active_tol=0.002,
                                     max_active=100, reg=1e-12)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [newton]: {score5:.6f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coord Descent (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 80.0)
    if t6 > 10:
        print(f"  Budget: {t6:.0f}s")
        f6, obj6 = coordinate_polish(best_seq, time_limit=t6)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {score6:.6f}")

    # === Phase 7: Final Newton + LP burst ===
    print(f"\n=== Phase 7: Final polish (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.4, 150.0)
    if t7 > 20:
        # Newton first
        t7a = t7 * 0.5
        f7a, obj7a = newton_kkt_polish(best_seq, time_limit=t7a, active_tol=0.001,
                                        max_active=80, reg=1e-12)
        score7a = float(evaluate_sequence(f7a.tolist()))
        if np.isfinite(score7a) and score7a < best_score:
            best_score = score7a
            best_seq = f7a.copy()
            print(f"  NEW BEST [newton2]: {score7a:.6f}")

        # Then LP
        t7b = t7 * 0.5
        f7b, obj7b = run_lp_phase(best_seq, time_limit=t7b, top_k_init=50,
                                   top_k_max=150, label="lp_final")
        score7b = float(evaluate_sequence(f7b.tolist()))
        if np.isfinite(score7b) and score7b < best_score:
            best_score = score7b
            best_seq = f7b.copy()
            print(f"  NEW BEST [lp_final]: {score7b:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
