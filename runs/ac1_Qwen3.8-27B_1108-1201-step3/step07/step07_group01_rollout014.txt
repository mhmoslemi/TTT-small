"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Frank-Wolfe (conditional gradient)
with exact LP directions, subgradient polishing, and multi-scale exploration.
Key insight: the problem is convex (max of bilinear forms), so FW with exact
LP directions converges globally. We combine this with subgradient averaging
and local search for the final polish.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix A such that A @ b = a * b (linear convolution)."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        k_end = min(2*n - 1, j + n)
        A[j:k_end, j] = a[:k_end - j]
    return A


def frank_wolfe_direction(a, M, verbose=False):
    """Solve LP: max 1'b s.t. Conv(a) @ b <= M, b >= 0.
    Returns the optimal b (unnormalized)."""
    n = len(a)
    A = build_conv_matrix(a)
    
    b_var = cp.Variable(n)
    constraints = [b_var >= 0, A @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.GLOP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None and np.isfinite(b_var.value).all():
                b_val = np.maximum(b_var.value, 0)
                if verbose:
                    print(f"    LP solved with {solver}, status={prob.status}, obj={prob.value:.6f}")
                return b_val, prob.status
        except Exception as e:
            if verbose:
                print(f"    LP solver {solver} failed: {e}")
            continue
    
    # Fallback: uniform
    return np.ones(n), "infeasible"


def fw_line_search(a, b_dir, f_a, max_iter=20):

    best_t = 0.0
    best_f = f_a

        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        b_conv = fast_convolve(a_trial)
        f_trial = b_conv.max()
        if f_trial < best_f:
            best_f = f_trial
            best_t = t

    if 0 < best_t < 1:
        for t in np.linspace(max(0, best_t - 0.1), min(1, best_t + 0.1), 20):
            a_trial = (1 - t) * a + t * b_norm
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            b_conv = fast_convolve(a_trial)
            f_trial = b_conv.max()
            if f_trial < best_f:
                best_f = f_trial
                best_t = t
    
    return best_t, best_f


def frank_wolfe_optimize(a_init, time_limit, verbose=True, label="FW"):
    """Frank-Wolfe algorithm with exact LP directions."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_conv = fast_convolve(a)
        current_peak = b_conv.max()
        M = current_peak  # Current peak value
        
        if current_peak < best_score - 1e-15:
            best_score = current_peak
            best_a = a.copy()
        
        # Get FW direction from LP
        b_dir, status = frank_wolfe_direction(a, M)
        b_sum = b_dir.sum()
        if b_sum < 1e-15:
            b_dir = np.ones(n) / n
        else:
            b_dir = b_dir / b_sum
        
        # Check if we've converged (direction is the same as current)
        if np.max(np.abs(b_dir - a)) < 1e-8:
            if verbose:
                print(f"    {label}: Converged at iter {it}, peak={best_score:.10f}")
            break
        
        # Line search for step size
        t_best, f_best = fw_line_search(a, b_dir, current_peak)
        
        if t_best <= 0:
            # Try smaller step
            t_best = 2.0 / (it + 2)
        
        a_new = (1 - t_best) * a + t_best * b_dir
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Verify improvement
        b_new = fast_convolve(a_new)
        f_new = b_new.max()
        
        if f_new <= current_peak + 1e-12:
            a = a_new
        else:
            # Halve the step
            t_best /= 2
            a_new = (1 - t_best) * a + t_best * b_dir
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.1f}s): peak={current_peak:.10f} "
                  f"best={best_score:.10f} t={t_best:.4f}")
    
    return best_a, best_score


def subgradient_optimize(a_init, time_limit, verbose=True, label="SG"):
    """Subgradient method with Polyak step size and iterate averaging."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    
    # Lower bound: 1/(2n-1) (average of convolution values)
    f_star_lb = 1.0 / (2 * n - 1)
    
    # Better lower bound from endpoint constraints
    # (a*a)_0 = a_0^2 <= f*, so a_0 <= sqrt(f*)
    # This gives a recursive bound but let's just use the basic one
    
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    count = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_conv = fast_convolve(a)
        f_a = b_conv.max()
        
        if f_a < best_score - 1e-15:
            best_score = f_a
            best_a = a.copy()
        
        # Active set: all k where (a*a)_k is close to the max
        threshold = f_a * 0.999
        active = np.where(b_conv >= threshold)[0]
        
        # Subgradient: average of gradients at active peaks
        # grad_k[i] = 2*a[k-i] (if valid)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Project gradient onto tangent space of simplex
        g = g - np.dot(a, g)
        
        # Polyak step size
        gap = f_a - f_star_lb
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        t = gap / g_norm_sq
        t = min(t, 1.0)  # Cap at 1
        
        # Take step
        a_new = a - t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        # Running average
        avg_a += a
        count += 1
        
        if verbose and it % 1000 == 0:
            avg_a_norm = avg_a / count
            avg_score = fast_convolve(avg_a_norm).max()
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.1f}s): f={f_a:.10f} "
                  f"best={best_score:.10f} avg={avg_score:.10f}")
    
    # Also check the average iterate
    avg_a_norm = avg_a / max(count, 1)
    avg_score = fast_convolve(avg_a_norm).max()
    if avg_score < best_score:
        best_a = avg_a_norm
        best_score = avg_score
    
    return best_a, best_score


def peak_shave_polish(a_init, time_limit, verbose=True, label="PS"):
    """Polish by targeting the specific peaks that are at maximum."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_conv = fast_convolve(a)
        current_peak = b_conv.max()
        
        if current_peak < best_score - 1e-15:
            best_score = current_peak
            best_a = a.copy()
        
        # Find all peaks at (or near) the maximum
        threshold = current_peak * 0.9999
        peak_indices = np.where(b_conv >= threshold)[0]
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find which pairs (i, j) with i+j=k contribute most
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # Try reducing the top contributors
            top_contributors = np.argsort(contribs)[::-1][:10]
            
            for ci in top_contributors:
                i = int(idx[ci])
                j = k - i
                
                for factor in [0.99, 0.98, 0.95, 0.9, 0.8]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    
                    # Redistribute to maintain sum=1
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    sc = fast_convolve(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                    elif sc < current_peak:
                        # Accept even if not better than best (descent)
                        a = a_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try adding mass to low-value positions
            # Find position i where adding mass least increases the peak
            b_conv = fast_convolve(a)
            
            best_gain = 0
            best_pos = -1
            for i in range(0, n, max(1, n // 200)):  # Sample positions
                a_trial = a.copy()
                a_trial[i] += 1e-4
                a_trial = a_trial / a_trial.sum()
                sc = fast_convolve(a_trial).max()
                gain = current_peak - sc
                if gain > best_gain:
                    best_gain = gain
                    best_pos = i
            
            if best_pos >= 0 and best_gain > 0:
                a_trial = a.copy()
                a_trial[best_pos] += 0.01
                a_trial = a_trial / a_trial.sum()
                sc = fast_convolve(a_trial).max()
                if sc < current_peak:
                    a = a_trial
                    improved = True
        
        if not improved and step > 100:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
