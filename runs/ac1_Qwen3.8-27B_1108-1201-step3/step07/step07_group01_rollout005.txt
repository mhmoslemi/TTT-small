"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative direction finding
combined with smoothed L-BFGS polishing and multi-start exploration. The LP approach
solves a linear program to find the best non-negative descent direction targeting
peak regions of the auto-convolution, with adaptive constraint relaxation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * float(b.max())


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def solve_lp_direction(a, b, M, peak_frac=0.1, slack=0.0):
    """Solve LP to find best non-negative correction direction g.
    Maximize sum(g) s.t. g >= 0, sum(g) <= 1,
    2*conv(a,g)[k] <= M + slack - b[k] for peaks within peak_frac of max.
    Returns (g, optimal_value) or (None, 0).
    """
    n = len(a)
    # Determine binding constraints: peaks within peak_frac of M
    threshold = M * (1.0 - peak_frac)
    binding_indices = np.where(b >= threshold)[0]
    
    if len(binding_indices) == 0:
        binding_indices = np.array([int(np.argmax(b))])
    
    m = len(binding_indices)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    for idx, k in enumerate(binding_indices):
        rhs = M + slack - b[k]
        if rhs < -1e-10:
            rhs = 0.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            coeff = 2.0 * a[i]
            if coeff > 1e-14:
                rows.append(idx)
                cols.append(j)
                vals.append(coeff)
        b_ub.append(rhs)
    
    # Add sum(g) <= 1
    sum_idx = m
    for j in range(n):
        rows.append(sum_idx)
        cols.append(j)
        vals.append(1.0)
    b_ub.append(1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m + 1, n))
    c = -np.ones(n)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.array(b_ub),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 5.0})
        if res.success:
            return res.x, -float(res.fun)
    except Exception:
        pass
    return None, 0.0

def smooth_objective(z, T, n):
    """Smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def smooth_grad(z, T, n):
    """Gradient of smoothed objective in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    w = np.exp(scaled - m)
    w = w / w.sum()
    
    # Gradient wrt a: 2n * cross_corr(w, a) * 2 (chain rule through softmax)
    # Actually: d/da_j [2n * sum_k w_k * b_k] = 4n * sum_k w_k * a[k-j]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 4.0 * n * correl
    
    # Project to tangent space of simplex: grad_z = a * (grad_a - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    return a * (grad_a - mean_grad)


def lbgfs_smooth(a_init, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smooth_objective(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return smooth_grad(z, T, n)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'disp': False})
        return softmax_from_z(res.x)
    except Exception:
        return a_init.copy()


def subgradient_optimize(a_init, time_limit, verbose=True, label=""):

    n = len(a_init)

            break
        
        if it % 1000 == 0:
            beta = 1.0 / np.sqrt(it)
        
        # Compute gradient at y
        b = fast_convolve(y)
        k_peak = int(np.argmax(b))
        
        # Subgradient: 2*a[k_peak - j]
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]
        
        # Project onto tangent space

        # Step size
        eta = 0.5 / (np.sqrt(it) + 1.0)

        # Nesterov: a_{k+1} = (1-beta)*a_k + beta*y_k - eta*grad
        a_new = (1.0 - beta) * a + beta * y - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        y = a_new.copy()
        a = a_new
        
        # Track best

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
