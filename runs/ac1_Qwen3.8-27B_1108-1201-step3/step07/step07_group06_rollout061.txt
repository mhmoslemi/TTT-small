LP-based fictitious play for minimizing the peak-to-average ratio of 
auto-convolution. Uses exact LP solves for best-response computation,
adaptive blending, and multiple phases of optimization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def solve_best_response(a, method='highs'):
    """
    Solve the LP: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) where g is the best response and t is the optimal peak.
    """
    n = len(a)
    # Variables: [g[0], ..., g[n-1], t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: (a*g)[k] - t <= 0
    # Row k: sum_j a[k-j]*g[j] - t <= 0
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0  # -t
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method=method)
    
    if res.status != 0:
        return None, None
    
    g = res.x[:n]
    t = res.x[n]
    return g, t


def evaluate_score(a):
    """Compute the objective score."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def lp_fictitious_play(a0, time_limit, initial_blend=0.5, min_blend=0.05,
                       blend_decay=0.999, verbose=True):
    """
    LP-based fictitious play with adaptive blending.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = evaluate_score(a)
    t_blend = initial_blend
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_convolve(a)
        M = b.max()
        
        g, t_opt = solve_best_response(a)
        if g is None:

        # Normalize g (should already sum to 1 from LP, but be safe)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Check improvement: t_opt should be < M for progress
        # Blend
        t_actual = min(1.0, t_blend * (t_opt / M) ** 0.5) if M > 1e-15 else 0.1
        a_new = (1 - t_actual) * a + t_actual * g
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-10:

        a_new = a_new / ss
        
        new_score = evaluate_score(a_new)

            best_a = a_new.copy()
            no_improve = 0
        else:

        t_blend = max(min_blend, t_blend * blend_decay)

            a = np.maximum(a, 0)
            a = a / a.sum()
            t_blend = initial_blend
            no_improve = 0

def interpolate_to_length(a, new_n):

        return np.ones(new_n) / new_n
    a_n = a / s
    old_x = np.linspace(0, 1, old_n)

    out = np.interp(new_x, old_x, a_n)
    out = np.maximum(out, 0)
    ss = out.sum()
    if ss > 1e-15:
        out /= ss
    return out

"""
LP-based fictitious play with adaptive blending, vectorized constraint building,
multi-length exploration, and coordinate descent polishing for minimizing
2*n*max(autoconv(a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_n = a / s
    return 2.0 * n * fast_convolve(a_n).max()


def build_constraint_matrix(a):
    """Build A_ub for the LP best-response problem (vectorized)."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j_idx = np.arange(n)[None, :].astype(np.int64)
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    safe_diff = np.where(valid, diff, 0)
    A_core = np.where(valid, a[safe_diff], 0.0)
    A_ub = np.hstack([A_core, -np.ones((2 * n - 1, 1))])
    return A_ub


def solve_lp_best_response(a, A_ub=None):
    """
    Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None) on failure.
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0

    if A_ub is None:
        A_ub = build_constraint_matrix(a)

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[n]

        gs = g.sum()
        if gs < 1e-12:

        b = fast_convolve(a)
        # Find the worst interior peak
        b_search = b.copy()
        if len(b_search) > 2:
            b_search[0] = -np.inf
            b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = idx[np.argsort(contribs)[::-1][:10]]

        for i in top:
            if a[i] < 1e-14:
                continue
            for f in [0.88, 0.92, 0.96, 1.04, 1.08, 1.12]:
                a_t = a.copy()
                a_t[i] *= f
                a_t = np.maximum(a_t, 0.0)
                ss = a_t.sum()
                if ss < 1e-12:
                    continue
                a_t = a_t / ss
                sc = compute_score(a_t)
                if sc < best_score - 1e-14:
                    a = a_t
                    best_score = sc
                    best_a = a_t.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbations
            found = False
            for _ in range(30):
                i = rng.integers(n)
                if a[i] < 1e-14:
                    continue
                f = rng.uniform(0.85, 1.15)
                a_t = a.copy()
                a_t[i] *= f
                a_t = np.maximum(a_t, 0.0)
                a_t = a_t / a_t.sum()
                sc = compute_score(a_t)
                if sc < best_score - 1e-14:
                    a = a_t
                    best_score = sc
                    best_a = a_t.copy()
                    found = True
                    break
            if not found:
                # Try dual peak (symmetric)
                if peak_k > 0 and peak_k < 2 * n - 1:
                    sym_k = 2 * n - 2 - peak_k
                    lo2 = max(0, sym_k - n + 1)
                    hi2 = min(n - 1, sym_k)
                    idx2 = np.arange(lo2, hi2 + 1)
                    contribs2 = a[idx2] * a[sym_k - idx2]
                    top2 = idx2[np.argsort(contribs2)[::-1][:5]]
                    for i in top2:
                        if a[i] < 1e-14:
                            continue
                        for f in [0.9, 0.95, 1.05, 1.1]:
                            a_t = a.copy()
                            a_t[i] *= f
                            a_t = np.maximum(a_t, 0.0)
                            a_t = a_t / a_t.sum()
                            sc = compute_score(a_t)
                            if sc < best_score - 1e-14:
                                a = a_t
                                best_score = sc
                                best_a = a_t.copy()
                                improved = True
                                break
                        if improved:
                            break
                if not improved:
                    break
    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_n = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, a_n)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss > 1e-15:
        out = out / ss
    return out


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_n = a / s
    try:
        sc = float(evaluate_sequence(a_n.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_n.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_n)})")
    return best_seq, best_score


def gmres_smooth(a, n_steps=50, alpha=0.1):
    """
    Spectral smoothing: apply (I - alpha*L) in Fourier domain to reduce
    high-frequency content, which often correlates with autocorr peaks.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    # Low-pass: dampen high frequencies
    damp = 1.0 - alpha * (np.abs(freqs) / 0.5) ** 2
    damp = np.clip(damp, 0.1, 1.0)
    A *= damp
    a_out = irfft(A, fft_size)[:n]
    a_out = np.maximum(a_out, 0.0)
    ss = a_out.sum()
    if ss > 1e-15:
        a_out = a_out / ss
    return a_out
