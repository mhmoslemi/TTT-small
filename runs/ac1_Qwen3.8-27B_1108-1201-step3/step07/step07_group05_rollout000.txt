import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def lp_best_response(f, M):
    """
    Solve: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i]
    # We need A_ub @ g <= b_ub
    
    num_constraints = 2 * n - 1
    # Use sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i  # index into g
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    b_ub = np.full(num_constraints, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    return res.x

def fictitious_play(f0, time_limit, initial_step=0.1, min_step=0.01, verbose=True):

    Fictitious play iteration:
    1. Compute M = max(f*f)
    2. Solve LP: max sum(g) s.t. f*g <= M, g >= 0
    3. Normalize g, blend: f <- (1-t)f + t*g
    """

    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    t = initial_step

        M = conv.max()
        
        # Solve LP for best response

            g_sum = g.sum()
            if g_sum < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / g_sum
        
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        score = compute_score(f_new)
