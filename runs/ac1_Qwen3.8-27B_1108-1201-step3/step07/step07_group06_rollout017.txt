import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def get_conv_weights(b, T):
    """Get soft-max weights from convolution values."""
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    s = exp_b.sum()
    if s < 1e-300:
        return np.ones_like(b) / len(b)
    return exp_b / s


def adam_logspace(a0, time_limit, T_init=0.04, T_final=0.001, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='cyclical',
                  n_cycles=3, verbose=False, verbose_interval=2000):
    """Adam optimizer in log-space with cyclical temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Cyclical temperature schedule
        if schedule == 'cyclical':
            cycle_progress = (progress * n_cycles) % 1.0
            base_progress = progress
            T = T_final + (T_init - T_final) * ((1 - base_progress) ** 0.5)
            T = T * (0.5 + 0.5 * np.cos(cycle_progress * np.pi * 2))
            T = max(T, T_final)
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        w = get_conv_weights(b, T)

        # Gradient via FFT cross-correlation
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        lr = max(0.0005, lr * 0.99995)

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=50, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * (T * (max_b / T + np.log(exp_b.sum())))

    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=soft_max_obj, x0=z0, jac=soft_max_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()
    return a_out


def peak_pair_swap(a0, time_limit, top_k_peaks=5, n_trials=10000):
    """
    Exact peak-pair swapping local search.
    For each near-peak lag, identify top contributing pairs and swap mass
    from high-pressure elements to low-pressure elements.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find top-K peaks
        peak_threshold = max_b * 0.95
        peak_lags = np.where(b >= peak_threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(b)])

        improved = False
        # Try each peak lag
        for k_star in peak_lags[:top_k_peaks]:
            k_star = int(k_star)
            # Find contributing pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            partners = k_star - idx
            contribs = a[idx] * a[partners]
            top_idx = np.argsort(contribs)[::-1][:8]

            for ci in top_idx:
                i = int(idx[ci])
                j = int(partners[ci])
                if a[i] < 1e-10:
                    continue

                # Compute "pressure" on each element: how much it contributes to all near-peaks
                # Find safe target elements (low contribution to all peaks)
                # Simple heuristic: target elements far from all peak partners
                target_candidates = []
                for t in range(n):
                    if t == i:
                        continue
                    if a[t] >= a[i] * 0.5:  # Don't target too-large elements
                        continue
                    # Check if t is a partner in any peak
                    t_pressure = 0.0
                    for pk in peak_lags[:top_k_peaks]:
                        pk = int(pk)
                        if 0 <= pk - t <= n - 1:
                            t_pressure += b[pk]
                    target_candidates.append((t_pressure, t))

                if not target_candidates:
                    continue
                target_candidates.sort()

                # Try moving mass from i to the safest target
                for factor in [0.85, 0.9, 0.95]:
                    amount = a[i] * (1.0 - factor)
                    for _tp, t_target in target_candidates[:3]:
                        a_trial = a.copy()
                        a_trial[i] -= amount
                        a_trial[t_target] += amount
                        if a_trial[i] < 0 or a_trial[t_target] < 0:
                            continue
                        score_trial = 2.0 * n * fast_convolve(a_trial).max()
                        if score_trial < best_score - 1e-13:
                            a = a_trial
                            best_score = score_trial
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            # Periodic random restart to escape local minima
            if trial % 50 == 0 and trial > 100:
                n_perturb = max(1, n // 100)
                idx_perturb = rng.choice(n, n_perturb, replace=False)
                a_trial = best_a + rng.standard_normal(n_perturb) * 0.01
                a_trial[idx_perturb] += 0
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_score * 1.02:
                    a = a_trial  # Accept even if slightly worse (exploration)

    return best_a, best_score


def simulated_annealing(a0, time_limit, T_start=0.01, T_end=0.0001,
                        step_scale=0.05, n_restarts=5):
    """
    Simulated annealing in the original (non-negative) space.
    Uses direct score evaluation (no smoothing).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    total_time = time_limit
    time_per_restart = total_time / n_restarts

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart == 0:
            a = best_a.copy()
        else:
            # Restart from perturbed best
            a = best_a + rng.standard_normal(n) * 0.02
            a = np.maximum(a, 0)
            a = a / a.sum()

        t_restart = time.monotonic()
        t_restart_limit = min(time_per_restart, time_limit - (t0 + t_restart - t0))

        current_score = 2.0 * n * fast_convolve(a).max()

        while time.monotonic() - t_restart < t_restart_limit:
            elapsed_r = time.monotonic() - t_restart
            progress = min(1.0, elapsed_r / max(t_restart_limit, 1e-10))
            T = T_start * (T_end / T_start) ** progress

            # Random perturbation
            n_perturb = max(1, min(5, n // 64))
            idx = rng.choice(n, n_perturb, replace=False)
            perturb = rng.standard_normal(n_perturb) * step_scale * (1.0 - progress * 0.5)
            a_trial = a.copy()
            a_trial[idx] += perturb
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial

            score_trial = 2.0 * n * fast_convolve(a_trial).max()

            if score_trial < current_score:
                a = a_trial
                current_score = score_trial
                if score_trial < best_score - 1e-13:
                    best_score = score_trial
                    best_a = a_trial.copy()
            else:
                # Accept with probability exp(-(delta)/T)
                delta = score_trial - current_score
                if delta < T * 10:
                    accept_prob = np.exp(-delta / max(T, 1e-15))
                    if rng.random() < accept_prob:
                        a = a_trial
                        current_score = score_trial

    return best_a, best_score


def coord_descent_momentum(a0, time_limit, n_sweeps=500):
    """
    Coordinate descent with momentum and adaptive step sizes.
    Tries both increases and decreases for each element.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    # Momentum per coordinate
    momentum = np.zeros(n)
    step_sizes = np.full(n, 0.05)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            best_factor = 1.0
            best_local_score = best_score

            # Adaptive factor range
            lo = max(0.5, 1.0 - step_sizes[i] * 2)
            hi = min(2.0, 1.0 + step_sizes[i] * 2)
            factors = np.linspace(lo, hi, 7)

            for factor in factors:
                a_trial = a.copy()
                a_trial[i] *= factor
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_factor = factor

            if best_factor != 1.0:
                # Update momentum
                delta = best_factor - 1.0
                momentum[i] = 0.9 * momentum[i] + 0.1 * delta
                # Keep momentum small to avoid oscillation
                if abs(momentum[i]) > 0.15:
                    momentum[i] = np.sign(momentum[i]) * 0.15

                a[i] *= best_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
                    # Reduce step size on success
                    step_sizes[i] *= 0.9
                else:
                    step_sizes[i] *= 1.1
            else:
                # No improvement - increase step size slightly
                step_sizes[i] = min(0.3, step_sizes[i] * 1.05)

        if not sweep_improved and sweep > 50:
            # Stuck - try random perturbation
            a = best_a + np.random.standard_normal(n) * 0.02
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score_check = 2.0 * n * fast_convolve(a).max()
            if best_score_check < best_score:
                best_score = best_score_check
                best_a = a.copy()

    return best_a, best_score


def multi_peak_flatten(a0, time_limit, n_iterations=5000):
    """
    Multi-peak flattening: iteratively reduce the spread of the convolution.
    Uses the idea that we want all convolution values to be as equal as possible.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    centroid = np.mean(a)  # = 1/n

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        mean_b = b.mean()
        spread = max_b - mean_b

        if spread < 1e-8:
            break

        # Element with maximum "excess" contribution to peak
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        partners = peak_k - idx
        contribs = a[idx] * a[partners]

        if len(idx) == 0:
            continue

        # Find element with highest contribution that's above centroid
        above_center = (a[idx] > centroid) & (a[partners] > centroid)
        if np.any(above_center):
            candidates = idx[above_center]
            cands_contrib = contribs[np.where(above_center)[0]]
            i = candidates[np.argmax(cands_contrib)]
        else:
            continue

        # Find safe target: element with low value that doesn't contribute to peak
        target_mask = (np.arange(n) != i) & (np.arange(n) != (peak_k - i))
        # Also avoid being a partner in the peak
        for pk in range(peak_k - 1, peak_k + 2):
            if 0 <= pk < 2 * n - 1:
                target_mask[peak_k - idx] = False

        safe_targets = np.where(target_mask & (a < centroid))[0]
        if len(safe_targets) == 0:
            safe_targets = np.where(target_mask & (a < a[i] * 0.8))[0]
        if len(safe_targets) == 0:
            continue

        # Move a small amount from i to the safest target
        amount = a[i] * 0.02
        t_target = safe_targets[0]

        a[i] -= amount
        a[t_target] += amount

        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        elif score > best_score * 1.01:
            # Revert if too bad
            a = best_a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_initializations(parent, n, rng):
    """Generate diverse starting points."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(p.copy())
            for scale in [0.005, 0.01, 0.02, 0.05]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append(pert)
            # Scale-preserving perturbation
            for scale in [0.01, 0.03]:
                delta = rng.standard_normal(n) * scale
                delta -= delta.mean() * p  # Preserve relative structure
                pert = p * (1 + delta)
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append(pert)
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(interp)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Triangular
    tri = np.linspace(0, 1, n) + np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    candidates.append(tri / tri.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 3)
    candidates.append(gau / gau.sum())

    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    bi[max(0, mid - n // 6):mid + n // 6] = 1.0
    bi[max(0, mid - n // 4):mid + n // 4] = 0.3
    bi = np.maximum(bi, 0.001)
    candidates.append(bi / bi.sum())

    # Cosine window
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    cos_w = 0.5 + 0.5 * np.cos(t)
    cos_w = np.maximum(cos_w, 0.01)
    candidates.append(cos_w / cos_w.sum())

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    candidates.append(rand / rand.sum())

    return candidates


def evaluate_and_track(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Adam log-space with cyclical annealing (35% time)
    Phase 2: Peak-pair swap local search (20% time)
    Phase 3: Simulated annealing (15% time)
    Phase 4: Coordinate descent + multi-peak flatten (15% time)
    Phase 5: Final polish with L-BFGS + exact search (10% time)
    Phase 6: Multi-length exploration (5% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Multi-start Adam with cyclical annealing ===
    print(f"\n=== Phase 1: Adam Cyclical (n={n_parent}) ===")
    t_p1_end = min(630.0, (deadline - time.monotonic()) * 0.35)
    t_p1_start = time.monotonic()

    inits = diverse_initializations(parent if parent is not None else best_seq,
                                    n_parent, rng)

    # Score initializations
    inits_scored = []
    for init_a in inits:
        s = 2.0 * n_parent * fast_convolve(init_a).max()
        inits_scored.append((s, init_a))
    inits_scored.sort(key=lambda x: x[0])

    n_starts = min(5, len(inits_scored))
    time_per_start = t_p1_end / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(time_per_start, rem * 0.9)
        if t_alloc < 30:
            continue

        init_a = inits_scored[idx][1]
        print(f"  Start {idx+1}/{n_starts}: score={inits_scored[idx][0]:.6f}, budget={t_alloc:.0f}s")

        # Try different schedules
        schedules = [
            ('cyclical', 0.04, 0.001, 0.008, 3),
            ('exp', 0.03, 0.001, 0.006, 1),
            ('hold_anneal', 0.025, 0.0005, 0.005, 1),
        ]

        for sched_name, T_i, T_f, lr, n_cycles in schedules[:2]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_sub = min(t_alloc / 2.0, rem * 0.8)
            if t_sub < 15:
                continue

            a_res, score_res = adam_logspace(
                init_a, t_sub, T_init=T_i, T_final=T_f,
                lr=lr, schedule=sched_name, n_cycles=n_cycles,
                verbose=False
            )
            best_seq, best_score = evaluate_and_track(
                a_res, f"adam{idx}_{sched_name}", best_seq, best_score)

    print(f"  After Phase 1: best={best_score:.6f}")

    # === Phase 2: Peak-pair swap local search ===
    print(f"\n=== Phase 2: Peak-Pair Swap ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sub = min(rem * 0.25, 120.0)
        a_ps, _ = peak_pair_swap(best_seq, t_sub, top_k_peaks=5, n_trials=15000)
        best_seq, best_score = evaluate_and_track(a_ps, "pair_swap", best_seq, best_score)
        print(f"  After pair swap: best={best_score:.6f}")

    # === Phase 3: Simulated annealing ===
    print(f"\n=== Phase 3: Simulated Annealing ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sub = min(rem * 0.20, 100.0)
        a_sa, _ = simulated_annealing(best_seq, t_sub, T_start=0.02, T_end=0.0001,
                                       step_scale=0.05, n_restarts=4)
        best_seq, best_score = evaluate_and_track(a_sa, "sim_anneal", best_seq, best_score)
        print(f"  After sim anneal: best={best_score:.6f}")

    # === Phase 4: Coordinate descent + multi-peak flatten ===
    print(f"\n=== Phase 4: Coord Descent + Flatten ===")
    rem = deadline - time.monotonic() - 150

    if rem > 20:
        t_sub = min(rem * 0.3, 80.0)
        a_cd, _ = coord_descent_momentum(best_seq, t_sub, n_sweeps=300)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: best={best_score:.6f}")

    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_sub = min(rem * 0.3, 60.0)
        a_mp, _ = multi_peak_flatten(best_seq, t_sub, n_iterations=8000)
        best_seq, best_score = evaluate_and_track(a_mp, "multi_flat", best_seq, best_score)
        print(f"  After multi-flatten: best={best_score:.6f}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic() - 60

    if rem > 15:
        # L-BFGS at different temperatures
        for T in [30, 10, 5]:
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            t_sub = min(rem * 0.3, 40.0)
            a_lb = lbgfs_polish(best_seq, t_sub, T=T, maxiter=3000)
            best_seq, best_score = evaluate_and_track(a_lb, f"lbgfs_T{T}", best_seq, best_score)
        print(f"  After L-BFGS: best={best_score:.6f}")

    rem = deadline - time.monotonic() - 30
    if rem > 10:
        # Another round of pair swap
        t_sub = min(rem * 0.5, 30.0)
        a_ps2, _ = peak_pair_swap(best_seq, t_sub, top_k_peaks=3, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_ps2, "pair_swap_2", best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    rem = deadline - time.monotonic() - 20
    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 2.0, 0.75, 1.1]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 3000 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 15
            if rem < 30:
                break
            t_sub = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = adam_logspace(
                init_alt, t_sub, T_init=0.03, T_final=0.001,
                lr=0.006, schedule='cyclical', n_cycles=2, verbose=False
            )
            best_seq, best_score = evaluate_and_track(
                a_alt, f"alt_n{alt_n}", best_seq, best_score)
            print(f"  Alt n={alt_n}: best={best_score:.6f}")

    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
