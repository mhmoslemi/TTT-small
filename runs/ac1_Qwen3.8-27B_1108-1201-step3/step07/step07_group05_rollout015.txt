```python
LP-based Fictitious Play for minimizing the peak auto-convolution ratio.
Uses the LP best-response oracle to drive the auto-correlation toward flatness,
combined with smooth gradient refinement and peak equalization.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max(conv)/sum^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def build_conv_matrix(f):
    """Build (2n-1) x n matrix C where (C@b) = cross-convolution of f and b."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for i in range(n):
        C[i:i+n, i] = f
    return C

def solve_lp_best_response(f, M, time_limit=5.0):
    """Solve LP: max 1^T b s.t. C@b <= M, b >= 0. Returns b."""
    n = len(f)
    C = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lp_fictitious_play(f0, time_limit, verbose=True):
    """LP-based fictitious play optimization."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)

        return f0, compute_score(f0)
    f = f / s

    best_score = compute_score(f)
    t0 = time.monotonic()

    avg = f.copy()
    n_avg = 0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        b = solve_lp_best_response(f, M)
        if b is None:
            # Fallback: use a simple perturbation
            b = f + 0.01 * np.random.randn(n)
            b = np.maximum(b, 0)
            b = b / b.sum()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Adaptive step size
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.8))
        t = 0.3 * (1 - progress) + 0.03 * progress
        
        f_new = (1 - t) * f + t * b
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:

        f_new = f_new / ss
```

```python
def fast_convolve(a):

    A = rfft(a, fft_size)
```

```python
def subgradient_descent(a0, time_limit, lr_init=0.01, verbose=True):
    """Subgradient descent on the non-smooth objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find active set (peaks within 1% of max)
        active = conv >= max_val * 0.999
        n_active = active.sum()
        w = active.astype(float) / n_active
        
        # Subgradient: grad_a_i = 2n * sum_k w_k * a[k-i]
        # But since we're on the sum=1 simplex, we need the projected version
        # Actually, for the non-normalized objective 2n*max(conv), the subgradient is:
        # g_i = 2n * sum_k w_k * a[k-i]
        # For the normalized objective (sum=1), it's more complex.
        # Let me just use the raw subgradient and project.
        
        fft_size = 1 << (2 * n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g = 2.0 * n * irfft(W * np.conj(A), fft_size)[:n]
        
        # Projected gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        lr = lr_init * (1.0 / (1.0 + 0.01 * it))
        
        if verbose and it % 100 == 0 and it > 0:
            print(f"  SubGrad iter={it}, score={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score
```
