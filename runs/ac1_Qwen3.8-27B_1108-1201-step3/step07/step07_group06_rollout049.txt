    Refine a sequence using KKT-inspired iterations.
    At each step:
    1. Identify active (peak) lags
    2. Compute the "pressure" at each position (weighted correlation)
    3. Redistribute mass to equalize pressure
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum() * target_sum

    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    N_corr = 1 << ((3 * n - 2) - 1).bit_length()
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        peak_val = b.max()

            best_a = a.copy()

        threshold = peak_val * 0.999
        active_lags = np.where(b >= threshold)[0]

        w = np.zeros(2 * n - 1)
        w[active_lags] = b[active_lags]  # or 1/len(active_lags)
        w = w / w.sum()
        
        # Compute pressure: p[j] = sum_k w[k] * 2 * a[k-j]
        W = rfft(w, N_corr)

        # Strategy: reduce a[j] where pressure > lambda, increase where pressure < lambda

def sca_refinement(a0, time_limit, K=5, verbose=False):
    """
    SCA (Successive Convex Approximation) refinement.
    At each step, linearize the top-K autocorrelation peak constraints
    and solve a convex QP.
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum() * target_sum
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    
    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Get top K lags
        top_k = np.argsort(b)[::-1][:K]
        peak_val = b.max()
        
        # Linearize constraints at current a:
        # a_new^T S_k a_new ≈ 2 * a^T S_k a_new - a^T S_k a
        # Constraint: 2 * (S_k @ a) @ a_new <= b[k] - delta
        # where delta is a small margin
        
        # (S_k @ a)[j] = sum_l S_k[j,l] * a[l] = a[k-j] (with bounds)
        # So (S_k @ a) is just a reversed/shifted version of a
        # Specifically: (S_k @ a)[j] = a[k-j] if 0 <= k-j < n else 0
        
        # The linearized constraint for lag k becomes:
        # 2 * sum_j a[k-j] * a_new[j] <= b[k] - delta
        # = 2 * dot(flipped_a_k, a_new) <= b[k] - delta
        
        A_constraints = []
        b_constraints = []
        delta = 0.001 * peak_val  # small reduction target
        
        for k in top_k:
            # (S_k @ a)[j] = a[k-j] for valid j
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]
            A_constraints.append(2.0 * row)
            b_constraints.append(b[k] - delta)
        
        # Also add constraint: sum(a_new) = target_sum
        # And a_new >= 0
        
        # Solve QP: min ||a_new - a||^2 s.t. A @ a_new <= b_vec, sum(a_new) = target_sum, a_new >= 0
        # Using scipy minimize with SLSQP or trust-constr
        
        from scipy.optimize import minimize as sp_minimize
        
        def obj(x):
            return np.sum((x - a) ** 2)
        
        def grad_obj(x):
            return 2.0 * (x - a)
        
        constraints = []
        for i in range(len(top_k)):
            A_row = A_constraints[i]
            b_val = b_constraints[i]
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, A_r=A_row, b_v=b_val: b_v - A_r @ x,
                'jac': lambda x, A_r=A_row: -A_r
            })
        
        constraints.append({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)
        })
        
        bounds = [(0, None)] * n
        
        try:
            res = sp_minimize(
                obj, a, jac=grad_obj, method='SLSQP',
