Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided optimization with
coordinate descent polishing and multi-phase refinement. Key innovation: uses
linear programming to find optimal update directions that explicitly respect
the convolution peak constraint, combined with aggressive coordinate descent

import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def lp_update(a, M, solver=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    a_arr = np.asarray(a, dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        coeffs = a_arr[js:je+1]
        g_idx = np.arange(k - je, k - js + 1)
        constraints.append(coeffs @ g[g_idx] <= M)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return np.maximum(g.value, 0)
    except Exception as e:
        print(f"    LP error: {e}")
    return None

def lp_iteration(a, t0, max_time, M_factor=1.0, t_step=0.1, solver=None):

def penalty_optimization(a, t0, max_time, beta_start=1.0, beta_end=100.0, 
                          lr_start=0.01, lr_end=0.0001, max_iters=5000):
    """Penalty-based optimization: minimize max(b) + beta * sum((b - threshold)_+^2)."""
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > max_time:
            break
        
        progress = min(1.0, it / max_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        b = fast_convolve(a)
        threshold = b.max() * 0.99
        
        # Gradient of penalty
        # d/d(a_i) [max(b)] = 2*a[peak-i] for the peak
        # d/d(a_i) [beta * sum((b_k - th)_+^2)] = 2*beta * sum_k (b_k - th)_+ * db_k/da_i
        # db_k/da_i = 2*a[k-i] (for valid indices)
        
        peak_k = int(np.argmax(b))
        grad_max = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        grad_max[idx] = 2.0 * a[peak_k - idx]
        
        # Penalty gradient
        grad_pen = np.zeros(n)
        excess = np.maximum(b - threshold, 0)
        if excess.sum() > 1e-15:
            # This requires computing 2*beta * sum_k excess[k] * 2*a[k-i] for each i
            # which is a convolution
            fft_size = 1 << (3*n-1).bit_length()
            E = rfft(excess, fft_size)
            A = rfft(a, fft_size)
            pen_grad = 4.0 * beta * np.real(irfft(E * A, fft_size))[:n]
            grad_pen = pen_grad
        
        grad = 2*n * (grad_max + grad_pen)
        # Project to tangent space
        grad = grad - np.dot(a, grad)
        
        # Update
        a = a - lr * grad
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        sc = score_from_a(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided optimization with
constraint generation, aggressive coordinate descent with pair updates,
penalty-based smoothing, and multi-length exploration. Key innovation: uses
linear programming (with constraint generation for efficiency) to find optimal
update directions that explicitly respect the convolution peak constraint,
combined with structured coordinate descent for local refinement.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score

    s = a.sum()
    if s < 0.01:

    try:

    except Exception:

    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")

def lp_update_cg(a, M, max_rounds=30, top_k=150, add_k=30):

    n = len(a)
    a_arr = np.asarray(a, dtype=np.float64)

    b = fast_convolve(a_arr)
    top_indices = np.argsort(b)[::-1][:top_k]
    active_set = list(top_indices.tolist())

    g_val = None
    for rnd in range(max_rounds):
        g = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_set:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            coeffs = a_arr[js:je + 1]
            g_idx = np.arange(k - je, k - js + 1)
            constraints.append(coeffs @ g[g_idx] <= M)

        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve()
        except Exception:
            break

        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break

        g_val = np.maximum(g.value, 0)
        if g_val.sum() < 1e-10:
            return None

        # Check for violated constraints
        b_g = fast_convolve(g_val)
        # Actually need conv(a, g) not conv(g, g)
        # conv(a,g)[k] = sum_j a[j]*g[k-j]

def lp_iteration(a, t0, max_time, M_factor=1.0, t_initial=0.1, verbose=True):
    """LP-guided iterations with line search."""
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t_step = t_initial
    stall_count = 0

    for it in range(5000):
        if time.monotonic() - t0 > max_time:
            break

        b = fast_convolve(a)
        M = b.max() * M_factor

        g = lp_update_cg(a, M, max_rounds=20, top_k=120, add_k=25)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Line search
        best_t = 0.0
        best_s = best_score
        for t in [t_step, t_step * 0.5, t_step * 1.5, t_step * 0.3, t_step * 2.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            sc = score_from_a(a_trial)
            if sc < best_s - 1e-14:
                best_s = sc
                best_t = t

        if best_t == 0.0:
            stall_count += 1
            t_step *= 0.8
            if stall_count > 30:
                break
            continue

        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a = a / a.sum()

        current_score = score_from_a(a)
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        t_step = max(t_step * 0.995, 0.001)

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it} ({elapsed:.0f}s): score={current_score:.7f} "
                  f"best={best_score:.7f} t={best_t:.4f} stall={stall_count}")

    return best_a, best_score


def penalty_optimization(a, t0, max_time, beta_start=5.0, beta_end=200.0,
                         lr_start=0.003, lr_end=0.00005, max_iters=5000):
    """Penalty-based optimization with soft peak smoothing."""
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > max_time:
            break

        progress = min(1.0, it / max_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        threshold = b.max() * 0.98

        # Gradient of max term
        peak_k = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[peak_k - idx]

        # Penalty gradient: 2*beta * sum_k max(0, b[k]-th) * 2*a[k-i]
        excess = np.maximum(b - threshold, 0)
        if excess.sum() > 1e-15:
            fft_size = 1 << (3 * n - 1).bit_length()
            E = rfft(excess, fft_size)
            A = rfft(a, fft_size)
            pen_grad = 4.0 * beta * np.real(irfft(E * A, fft_size))[:n]
            grad += pen_grad

        grad = 2.0 * n * grad
        # Project to tangent space of simplex
        grad = grad - np.dot(a, grad)

        a = a - lr * grad
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        sc = score_from_a(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    return best_a, best_score


def aggressive_coord_descent(a, t0, max_time, verbose=True):
    """Aggressive coordinate descent with single and pair updates targeting peaks."""
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)

    factors_single = [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3]
    factors_pair = [0.5, 0.7, 0.85, 1.15, 1.3]

    no_improve = 0
    max_no_improve = 100
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < max_time and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        order = rng.permutation(n)

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        for i in order:
            if time.monotonic() - t0 > max_time:
                break

            # Single coordinate update
            for f in factors_single:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = old * f
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            # Pair update with peak-contributing coordinate
            if js <= i <= je:
                j = peak_k - i
                if 0 <= j < n and j != i:
                    for f1 in factors_pair[:3]:
                        if improved:
                            break
                        for f2 in factors_pair[:3]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            ss = a_trial.sum()
                            if ss < 1e-15:
                                continue
                            a_trial = a_trial / ss
                            sc = score_from_a(a_trial)
                            if sc < best_score - 1e-13:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break

        if not improved:
            no_improve += 1
        elif verbose:
            elapsed = time.monotonic() - t0
            if it % 10 == 0:
                print(f"      CD ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def mirror_descent_step(a, t0, max_time, T=0.01, eta=0.01, max_iters=5000):
    """Mirror descent with soft-max smoothing of the max function."""
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > max_time:
            break

        progress = min(1.0, it / max_iters)
        T_cur = T * (0.01) ** progress
        eta_cur = eta * (0.1) ** progress

        b = fast_convolve(a)
        scaled = b / T_cur
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient: 4n * cross_corr(w, a)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = np.real(irfft(np.conj(W) * A, fft_size))[:n]
        grad_a = 4.0 * n * correl

        # Multiplicative update
        log_update = -eta_cur * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        sc = score_from_a(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-guided optimization with constraint generation (primary)
    2. Penalty-based smoothing
    3. Mirror descent
    4. Aggressive coordinate descent
    5. Second LP round
    6. Final polish
    7. Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 18.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initialized uniform: n={len(best_seq)}, score={best_score:.7f}")

    # === Phase 1: LP-guided optimization (primary) ===
    print(f"\n=== Phase 1: LP-Guided (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 700
        if rem > 60:
            t_alloc = min(rem * 0.5, 400.0)
            n_cur = len(best_seq)
            print(f"  LP: n={n_cur}, budget={t_alloc:.0f}s")
            a_lp, _ = lp_iteration(best_seq, time.monotonic(), t_alloc,
                                   M_factor=1.0, t_initial=0.1)
            best_seq, best_score = check_and_update(a_lp, "LP1", best_seq, best_score)

    # === Phase 2: Penalty-based optimization ===
    print(f"\n=== Phase 2: Penalty (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 550
        if rem > 60:
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  Penalty: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_pen, _ = penalty_optimization(best_seq, time.monotonic(), t_alloc,
                                            beta_start=5.0, beta_end=100.0,
                                            lr_start=0.003, lr_end=0.00003,
                                            max_iters=4000)
            best_seq, best_score = check_and_update(a_pen, "penalty", best_seq, best_score)

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 450
        if rem > 60:
            t_alloc = min(rem * 0.4, 180.0)
            print(f"  MD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent_step(best_seq, time.monotonic(), t_alloc,
                                          T=0.02, eta=0.01, max_iters=5000)
            best_seq, best_score = check_and_update(a_md, "MD", best_seq, best_score)

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 60:
            t_alloc = min(rem * 0.5, 250.0)
            print(f"  LP2: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp2, _ = lp_iteration(best_seq, time.monotonic(), t_alloc,
                                    M_factor=1.0, t_initial=0.05)
            best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 5: Aggressive coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  CD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd, _ = aggressive_coord_descent(best_seq, time.monotonic(), t_alloc)
            best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    # === Phase 6: LP polish ===
    print(f"\n=== Phase 6: LP Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  LP3: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp3, _ = lp_iteration(best_seq, time.monotonic(), t_alloc,
                                    M_factor=1.0, t_initial=0.02)
            best_seq, best_score = check_and_update(a_lp3, "LP3", best_seq, best_score)

    # === Phase 7: Final coordinate descent ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem * 0.6, 60.0)
            print(f"  CD2: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd2, _ = aggressive_coord_descent(best_seq, time.monotonic(), t_alloc)
            best_seq, best_score = check_and_update(a_cd2, "CD2", best_seq, best_score)

    # === Phase 8: Alternative length exploration ===
    print(f"\n=== Phase 8: Alt Length (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.8, 2.0, 0.7]:
            alt_n = int(n_cur * factor)
            if 256 <= alt_n <= 2048 and alt_n != n_cur and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 30
            if rem < 80:
                break
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = lp_iteration(init_alt, time.monotonic(), t_alloc * 0.55,
                                    M_factor=1.0, t_initial=0.1)
            a_alt, _ = aggressive_coord_descent(a_alt, time.monotonic(), t_alloc * 0.45)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
