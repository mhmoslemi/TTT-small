    A = rfft(a, fft_size)

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_weights(b, T):
    """Compute softmax weights exp(T*b) / sum(exp(T*b))."""
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return exp_Tb / exp_Tb.sum()


def fast_corr(w, a):
    """
    Compute cross-correlation corr[i] = sum_k w[k] * a[k-i] for i=0..n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    L = 1 << (3 * n - 3).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    corr = irfft(W * np.conj(A), L)[:n]
    return corr


def smooth_obj_and_grad_z(z, T):
    """
    Compute smoothed objective and gradient in log-space.
    z: log-domain variable (a = softmax(z))
    Returns (obj_value, grad_z).

    s = a.sum()
    if s < 1e-300:

        # Compute subgradient: average of 2*a[k-i] for active k
        grad_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(active)

        # Mirror descent step in log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        z = z - lr * grad_z

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.85
            no_improve = 0
        lr = max(lr, 1e-8)
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc_cur = 2.0 * n * fast_convolve(a_cur).max()
            print(f"    PeakShave step={step} ({elapsed:.0f}s): best={best_score:.7f} cur={sc_cur:.7f} lr={lr:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, n_passes=200):
    """
    Fast coordinate descent: for each coordinate, try scaling it
    and redistributing the difference proportionally.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Find the peak position to focus on
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Focus on coordinates that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = idx[np.argsort(contribs)[::-1][:min(20, len(idx))]]

        for i in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            # Try reducing a[i] by various amounts
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute diff proportionally to other elements
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / osum
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if p % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def multi_start_lbgfs(a0, time_limit, n_starts=3, verbose=True):
    """
    Try multiple perturbed starting points and keep the best.
    """
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    b = fast_convolve(a0)
    best_score = 2.0 * n * b.max() / (a0.sum() ** 2)

    if n_starts <= 1:
        return best_a, best_score

    time_per_start = time_limit / n_starts
    rng = np.random.default_rng(12345)

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        alloc = min(time_per_start, remaining)
        if alloc < 20:
            break

        # Create perturbation
        if s == 0:
            a_start = a0.copy()
        else:
            # Multiplicative perturbation in log-space
            noise_scale = 0.1 * (s + 1)
            z = np.log(np.maximum(a0, 1e-300)) + noise_scale * rng.standard_normal(n)
            a_start = np.exp(z - z.max())
            a_start = a_start / a_start.sum()

        T_sched = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
        a_new, sc_new = lbgfs_temperature_ladder(a_start, alloc, T_schedule=T_sched, verbose=False)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            if verbose:
                print(f"    Multi-start {s+1}: NEW BEST {sc_new:.7f}")

    return best_a, best_score


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        sc = float(evaluate_sequence(a_list))
        return sc
    except Exception:
        return float('inf')


def try_update(a, label, best_state, verbose=True):
    """Try to update best state with new candidate a."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state[1]:
        best_state[0] = a.copy()
        best_state[1] = sc
        if verbose:
            print(f"  NEW BEST [{label}]: {sc:.7f} (n={len(a)})")
    return best_state


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Fast L-BFGS with FFT gradients (main workhorse)
    2. Peak shaving with exact subgradient
    3. Coordinate descent polish
    4. Multi-start exploration
    5. Final ultra-low temperature L-BFGS
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    best_state = [None, float('inf')]

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_state = [parent.copy(), p_score]
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_score = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_score:.7f}")
        best_state = [parent.copy(), best_score]

    n_best = len(best_state[0])
    print(f"Starting optimization: n={n_best}, initial score={best_score:.7f}")

    # === Phase 1: Main L-BFGS with aggressive temperature ladder (55% time) ===
    print(f"\n=== Phase 1: L-BFGS Temperature Ladder ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.55, 1000.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, n={n_best}")
        T_ladder = [80.0, 40.0, 20.0, 10.0, 5.0, 2.5, 1.2, 0.6, 0.3, 0.15, 0.08]
        a_lb, sc_lb = lbgfs_temperature_ladder(best_state[0], t1, T_schedule=T_ladder)
        best_state = try_update(a_lb, "lbgfs_main", best_state)
        # Re-evaluate with actual score
        best_state[1] = evaluate_safe(best_state[0])
        print(f"  After Phase 1: {best_state[1]:.7f}")

    # === Phase 2: Peak shaving with subgradient (20% time) ===
    print(f"\n=== Phase 2: Peak Shaving ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 400.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")

        best_state = try_update(a_ps, "peak_shave", best_state)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"  After Phase 2: {best_state[1]:.7f}")

    # === Phase 3: Coordinate descent (10% time) ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 200.0)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a_cd, sc_cd = coordinate_descent_fast(best_state[0], t3, n_passes=500)
        best_state = try_update(a_cd, "coord_desc", best_state)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"  After Phase 3: {best_state[1]:.7f}")

    # === Phase 4: Second L-BFGS round with very low T (10% time) ===
    print(f"\n=== Phase 4: Low-T L-BFGS Refinement ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 250.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        T_low = [1.0, 0.5, 0.25, 0.1, 0.05]
        a_lb2, sc_lb2 = lbgfs_temperature_ladder(best_state[0], t4, T_schedule=T_low)
        best_state = try_update(a_lb2, "lbgfs_lowT", best_state)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"  After Phase 4: {best_state[1]:.7f}")

    # === Phase 5: Multi-start exploration (5% time) ===
    print(f"\n=== Phase 5: Multi-Start ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.05, 120.0)
    if t5 > 30:
        print(f"  Budget: {t5:.0f}s")
        a_ms, sc_ms = multi_start_lbgfs(best_state[0], t5, n_starts=3)
        best_state = try_update(a_ms, "multi_start", best_state)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"  After Phase 5: {best_state[1]:.7f}")

    # === Phase 6: Final polish (5% time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t6a = min(rem * 0.4, 60.0)
        t6b = min(rem * 0.4, 60.0)
        if t6a > 10:
            a_f1, _ = peak_shave_subgradient(best_state[0], t6a, n_steps=50000, lr_init=0.002)
            best_state = try_update(a_f1, "final_shave", best_state)
            best_state[1] = evaluate_safe(best_state[0])
        rem = deadline - time.monotonic()
        if rem > 10:

                                                T_schedule=[0.3, 0.1, 0.05], verbose=False)
            best_state = try_update(a_f2, "final_lbgfs", best_state)
            best_state[1] = evaluate_safe(best_state[0])

    # Final verification
    final_score = evaluate_safe(best_state[0])
    print(f"\nFINAL SCORE: {final_score:.7f} (n={len(best_state[0])})")
    return [float(x) for x in best_state[0]]
