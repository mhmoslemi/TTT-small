import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False, damping=None):
    """Alternating LP best responses with adaptive damping."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            if damping is None:
                alpha = min(1.0, 5.0 / (iters + 1))
            else:
                alpha = damping(iters)
            b = (1 - alpha) * b + alpha * b_new
            b = np.maximum(b, 0)
            s = b.sum()
            if s > 1e-15:
                b = b / s

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = (1 - alpha) * a + alpha * a_new
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s

        # Check candidates
        for cand in [a, b, 0.5 * (a + b)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def coord_descent_exact(a0, time_limit, n_sweeps=500):
    """
    Coordinate descent exploiting the structure of the objective.
    For each coordinate j, the optimal value is found analytically
    or via a small grid search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try small positive values
                for trial in [0.01/n, 0.05/n, 0.1/n]:
                    a[i] = trial
                    s = a.sum()
                    a_norm = a / s
                    score = 2.0 * n * fast_convolve(a_norm).max()
                    if score < best_score - 1e-13:
                        best_score = score
                        best_a = a.copy()
                        old_val = trial
                        improved = True
                        break
                if old_val < 1e-15:
                    a[i] = 0.0
                continue

            # Try a range of values around old_val
            s_rest = s - old_val
            # Analytical candidate: the minimum of max_k (slope_k * x + intercept_k) / (x + s_rest)^2
            # Try log-spaced values
            candidates = [old_val * f for f in [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.1, 1.3, 1.6, 2.5, 4.0]]
            candidates = [c for c in candidates if c >= 0]
            # Also try the "flat" value
            candidates.append(s / n)

            best_trial = old_val
            best_trial_score = best_score + 1
            for trial in candidates:
                a[i] = trial
                s_new = a.sum()
                if s_new < 1e-15:
                    continue
                a_norm = a / s_new
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = trial

            if best_trial_score < best_score - 1e-14:
                a[i] = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val

        if not improved and sweep > 10:
            break

    return best_a, best_score

def coord_descent_fine(a0, time_limit):
    """Fine coordinate descent with golden-section-like search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue

            # Golden section around old_val
            lo = old_val * 0.7
            hi = old_val * 1.3
            if lo < 0:
                lo = 0.0

            best_x = old_val
            best_s = best_score
            for _ in range(8):
                m1 = lo + 0.382 * (hi - lo)
                m2 = lo + 0.618 * (hi - lo)

                a[i] = m1
                s1 = a.sum()
                sc1 = 2.0 * n * fast_convolve(a / s1).max() if s1 > 0 else 999

                a[i] = m2
                s2 = a.sum()
                sc2 = 2.0 * n * fast_convolve(a / s2).max() if s2 > 0 else 999

                if sc1 < sc2:
                    hi = m2
                    if sc2 < best_s:
                        pass
                else:
                    lo = m1
                    if sc1 < best_s:
                        pass

                # Track best
                if sc1 < best_s:
                    best_s = sc1
                    best_x = m1
                if sc2 < best_s:
                    best_s = sc2
                    best_x = m2

            if best_s < best_score - 1e-14:
                a[i] = best_x
                best_score = best_s
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val
                s = a.sum()

        if not improved and sweep > 5:
            break

    return best_a, best_score

def peak_shave(a0, time_limit):
    """Aggressively shave peaks by adjusting contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        # Find top 3 peaks
        top_k = min(3, len(c))
        peaks = np.argpartition(c, -top_k)[-top_k:]
        peaks = peaks[np.argsort(c[peaks])[::-1]]

        improved = False
        for pk in peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = np.array([a[i] * a[pk - i] for i in idx])

            if contribs.max() < best_score * 0.1:
                continue

            # Sort by contribution
            order = np.argsort(contribs)[::-1]
            for ci in order[:8]:
                i = int(idx[ci])
                j = pk - i
                if a[i] < 1e-14:
                    continue

                # Try reducing a[i]
                for factor in [0.7, 0.8, 0.85, 0.9]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

                if improved:
                    break

                # Try reducing a[j] if different
                if j != i and a[j] >= 1e-14:
                    for factor in [0.7, 0.8, 0.85, 0.9]:
                        a_trial = a.copy()
                        a_trial[j] *= factor
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

            if improved:
                break

        if not improved:
            break

    return best_a, best_score

def nesterov_optimize(a0, time_limit, top_k=30, lr_init=0.005):
    """Nesterov-accelerated projected subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        # Compute soft subgradient
        c = fast_convolve(y)
        k_act = min(top_k, len(c))
        peaks = np.argpartition(c, -k_act)[-k_act:]
        pv = c[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                sg[j] += 2.0 * w[idx] * y[k - j]

        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        # Project onto simplex
        y_projected = simplex_project(y_new)

        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0

        if no_improve > 400:
            noise = rng.standard_normal(n) * 0.02
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0

        lr *= 0.9995

    return best_a, best_score

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, alpha, rng):
    """Create an asymmetric (slightly linear) initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * (2.0 * t / (n - 1) - 1.0)
    a = np.maximum(a, 0.01)
    s = a.sum()
    a = a / s
    return a

def init_fourier(n, rng):
    """Random Fourier superposition initialization."""
    n_harm = int(rng.integers(3, 15))
    amps = rng.random(n_harm) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 1.0)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
