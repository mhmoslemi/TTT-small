```python
def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g (unnormalized)."""
    n = len(f)
    # Build constraint matrix
    # Constraint k: sum_j f[k-j] * g[j] <= M for valid j
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]

    b = np.full(2*n-1, M)

    if res.success:
        return res.x
    else:
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based majorization-minimization 
with subgradient polishing. Core idea: at each step, solve an LP to find the 
optimal direction g (maximizing sum(g) subject to conv(f,g) <= M), then update 
f toward g via line search. Combined with multi-start, spectral initialization, 
and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) is already 1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_lp_indices(n):
    """Precompute (row, col) indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, M, rows, cols, n):
    """Solve LP: max sum(g) s.t. A@g <= M*ones, g >= 0.
    Returns normalized g (sum=1) or None if infeasible."""
    values = f[rows - cols]  # f[k-j] for each (k,j) pair
    A = csr_matrix((values, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if res.success:
            g = res.x
            s = g.sum()
            if s < 1e-15:
                return None
            return g / s
    except Exception:
        pass
    return None


def line_search(f, g, n, time_check=None, deadline=None):
    """Find best convex combination t of f and g."""
    best_t = 0.0
    best_score = score_normalized(f)
    grid = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for t in grid:
        if time_check is not None and time_check() > deadline:
            break
        a = (1 - t) * f + t * g
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = score_normalized(a)
        if sc < best_score - 1e-15:
            best_score = sc
            best_t = t
    # Refine around best_t
    if best_t > 0:
        lo = max(0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        for _ in range(10):
            if time_check is not None and time_check() > deadline:
                break
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            a1 = (1 - m1) * f + m1 * g
            a1 = a1 / a1.sum()
            a2 = (1 - m2) * f + m2 * g
            a2 = a2 / a2.sum()
            s1 = score_normalized(a1)
            s2 = score_normalized(a2)
            if s1 < s2:
                hi = m2
            else:
                lo = m1
            t_mid = (lo + hi) / 2
            a_mid = (1 - t_mid) * f + t_mid * g
            a_mid = a_mid / a_mid.sum()
            smid = score_normalized(a_mid)
            if smid < best_score - 1e-15:
                best_score = smid
                best_t = t_mid
    return best_t, best_score


def lp_iteration(f_init, time_limit, deadline, verbose=True, label=""):
    """LP-based majorization-minimization iteration."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    rows, cols = build_lp_indices(n)
    t0 = time.monotonic()
    stale_count = 0
    best_stale_threshold = 50

    for it in range(1, 100000):
        if time.monotonic() > deadline or time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(f).max()
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_f = f.copy()
            stale_count = 0
        else:
            stale_count += 1

        # Solve LP
        g = solve_lp_direction(f, M, rows, cols, n)
        if g is None:
            break

        # Check if g is essentially the same as f (convergence)
        ratio = g[best_f > 1e-10] / (best_f[best_f > 1e-10] + 1e-30)
        if np.max(np.abs(ratio - 1)) < 0.005:
            stale_count += 20

        # Line search
        t, sc = line_search(f, g, n, time_check=time.monotonic, deadline=deadline)
        if t > 0:
            f = (1 - t) * f + t * g
            s = f.sum()
            f = f / s
            f = np.maximum(f, 0.0)
            f = f / f.sum()
        else:
            # No improvement from LP direction, try smaller M
            M_tight = M * 0.99
            g = solve_lp_direction(f, M_tight, rows, cols, n)
            if g is not None:
                t, sc = line_search(f, g, n, time_check=time.monotonic, deadline=deadline)
                if t > 0:
                    f = (1 - t) * f + t * g
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()

        # Stale detection: random perturbation
        if stale_count > best_stale_threshold:
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.05 * (stale_count / best_stale_threshold)
            f = best_f + noise_scale * rng.standard_normal(n) * best_f
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            stale_count = 0
            best_stale_threshold = int(best_stale_threshold * 1.3)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f} stale={stale_count}")

    return best_f, best_score


def subgradient_polish(f_init, time_limit, deadline, verbose=True, label=""):
    """Subgradient-based polishing using soft-max over peaks."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()
    T = 0.005
    eta = 0.005

    for it in range(1, 50000):
        if time.monotonic() > deadline or time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_f = f.copy()

        # Soft-max weights over b
        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        # Gradient: grad[i] = sum_k w[k] * d(conv(f,f))[k]/d(f[i])
        # d(conv(f,f))[k]/d(f[i]) = f[k-i]*(1+[i!=k-i]) for valid k-i
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                coeff = f[k - j]
                if j != k - j:
                    coeff *= 2.0
                grad[j] += w[k] * coeff

        # Project to tangent space
        grad = grad - np.dot(f, grad)

        # Multiplicative update
        log_up = -eta * grad
        log_up = np.clip(log_up, -2.0, 2.0)
        f = f * np.exp(log_up)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
        else:
            f = f / s

        # Anneal
        T *= 0.999
        eta *= 0.9995

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SG iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f} T={T:.6f}")

    return best_f, best_score


def peak_shave_polish(f_init, time_limit, deadline, verbose=True, label=""):
    """Targeted peak shaving: reduce top contributing pairs."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()

    for step in range(50000):
        if time.monotonic() > deadline or time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        # Find top interior peaks
        b_work = b.copy()
        b_work[0] = -1
        b_work[-1] = -1
        peak_k = int(np.argmax(b_work))

        if b[peak_k] * 2.0 * n <= best_score + 1e-14:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[peak_k - idx]
        top_count = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            for factor in [0.97, 0.95, 0.92, 0.88, 0.82, 0.75]:
                a_trial = f.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    f = a_trial
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ci in top_idx[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or f[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = f.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            f = a_trial
                            best_score = sc
                            best_f = f.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} PS step={step} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_f, best_score


def coordinate_descent_polish(f_init, time_limit, deadline, verbose=True, label=""):
    """Coordinate descent with factor search."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    best_f = f.copy()
    best_score = score_normalized(f)
    t0 = time.monotonic()

    factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]
    no_improve = 0

    while time.monotonic() < deadline and time.monotonic() - t0 < time_limit and no_improve < 30:
        no_improve = 0
        improved_this_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline or time.monotonic() - t0 > time_limit:
                break
            for factor in factors:
                a_trial = f.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    f = a_trial
                    best_score = sc
                    best_f = f.copy()
                    improved_this_pass = True
                    break
        if not improved_this_pass:
            no_improve += 1
        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) > int((time.monotonic() - t0) - 5):
                print(f"    {label} CD ({elapsed:.0f}s): best={best_score:.7f}")

    return best_f, best_score


def make_gaussian_init(n, sigma=None):
    """Create a Gaussian-like initialization."""
    if sigma is None:
        sigma = n / 4.0
    x = np.arange(n) - n / 2.0
    g = np.exp(-x**2 / (2 * sigma**2))
    g = g / g.sum()
    return g


def make_u_shaped_init(n, sharpness=2.0):
    """Create a U-shaped initialization (high at edges)."""
    x = np.linspace(-1, 1, n)
    u = np.abs(x) ** sharpness
    s = u.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return u / s


def make_bimodal_init(n, center_frac=0.3):
    """Create a bimodal initialization."""
    c = int(n * center_frac)
    sigma = n / 8.0
    x = np.arange(n)
    b = np.exp(-((x - c)**2) / (2 * sigma**2)) + np.exp(-((x - (n - 1 - c))**2) / (2 * sigma**2))
    b = b / b.sum()
    return b


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate using evaluate_sequence and update if better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-based MM from parent (primary)
    2. Subgradient polishing
    3. Peak shaving
    4. Multi-start with different initializations
    5. Coordinate descent final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback initialization
        n_fallback = 768
        parent = make_gaussian_init(n_fallback)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Fallback init: n={len(parent)}, score={best_score:.7f}")

    n_best = len(best_seq)

    # === Phase 1: LP-based MM from parent ===
    print(f"\n=== Phase 1: LP-based MM (best={best_score:.7f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 800.0)
    if t_alloc > 30:
        print(f"  LP-MM: budget={t_alloc:.0f}s")
        a_lp, sc_lp = lp_iteration(best_seq, t_alloc, deadline, label="LP_main")
        best_seq, best_score = check_and_update(a_lp, "LP_main", best_seq, best_score)

    # === Phase 2: Subgradient polishing ===
    print(f"\n=== Phase 2: Subgradient Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.2, 250.0)
    if t_alloc > 20:
        print(f"  SG: budget={t_alloc:.0f}s")
        a_sg, sc_sg = subgradient_polish(best_seq, t_alloc, deadline, label="SG")
        best_seq, best_score = check_and_update(a_sg, "SG", best_seq, best_score)

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  PS: budget={t_alloc:.0f}s")
        a_ps, sc_ps = peak_shave_polish(best_seq, t_alloc, deadline, label="PS")
        best_seq, best_score = check_and_update(a_ps, "PS", best_seq, best_score)

    # === Phase 4: More LP from current best ===
    print(f"\n=== Phase 4: LP Refinement (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.2, 300.0)
    if t_alloc > 20:
        print(f"  LP2: budget={t_alloc:.0f}s")
        a_lp2, sc_lp2 = lp_iteration(best_seq, t_alloc, deadline, label="LP2")
        best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 5: Multi-start with different initializations ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.7f}) ===")
    n_curr = len(best_seq)
    inits = []

    # Gaussian
    g_init = make_gaussian_init(n_curr, sigma=n_curr / 3.5)
    inits.append(("gauss", g_init))

    # U-shaped
    u_init = make_u_shaped_init(n_curr, sharpness=1.5)
    inits.append(("u_shape", u_init))

    # Bimodal
    b_init = make_bimodal_init(n_curr, center_frac=0.25)
    inits.append(("bimodal", b_init))

    # Perturbed parent
    pert = best_seq + 0.1 * rng.standard_normal(n_curr) * best_seq
    pert = np.maximum(pert, 1e-10)
    pert = pert / pert.sum()
    inits.append(("perturbed", pert))

    for lbl, init in inits:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.2, 150.0)
        if t_alloc < 20:
            break
        print(f"  Multi-start [{lbl}]: budget={t_alloc:.0f}s")
        # Run LP from this init
        a_ms, sc_ms = lp_iteration(init, t_alloc, deadline, label=f"MS_{lbl}")
        best_seq, best_score = check_and_update(a_ms, f"MS_{lbl}", best_seq, best_score)

        # Quick subgradient polish
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sg = min(rem * 0.3, 60.0)
            a_ms2, _ = subgradient_polish(a_ms, t_sg, deadline, label=f"MS_{lbl}_sg")
            best_seq, best_score = check_and_update(a_ms2, f"MS_{lbl}_sg", best_seq, best_score)

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.7f}) ===")
    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0]:
        alt_n = int(n_curr * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_curr and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.25, 200.0)
        if t_alloc < 30:
            break
        print(f"  Alt length n={alt_n}: budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = lp_iteration(init_alt, t_alloc, deadline, label=f"alt_{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"alt_{alt_n}", best_seq, best_score)

        # Polish
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            a_alt2, _ = peak_shave_polish(best_seq, min(rem * 0.3, 60.0), deadline, label=f"alt_{alt_n}_ps")
            best_seq, best_score = check_and_update(a_alt2, f"alt_{alt_n}_ps", best_seq, best_score)

    # === Phase 7: Final coordinate descent polish ===
    print(f"\n=== Phase 7: Final CD Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.5, 80.0)
        if t_alloc > 5:
            print(f"  Final CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent_polish(best_seq, t_alloc, deadline, label="CD_final")
            best_seq, best_score = check_and_update(a_cd, "CD_final", best_seq, best_score)

    # === Phase 8: Final LP + peak shave ===
    print(f"\n=== Phase 8: Final Refinement (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.5, 60.0)
        if t_alloc > 10:
            a_fin, _ = lp_iteration(best_seq, t_alloc, deadline, label="LP_fin")
            best_seq, best_score = check_and_update(a_fin, "LP_fin", best_seq, best_score)

        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin2, _ = peak_shave_polish(best_seq, min(rem * 0.5, 30.0), deadline, label="PS_fin")
            best_seq, best_score = check_and_update(a_fin2, "PS_fin", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent if parent is not None else np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
