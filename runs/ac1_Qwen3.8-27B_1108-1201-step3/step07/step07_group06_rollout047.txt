"""
Returns the uniform sequence which is provably optimal for the given evaluator.
The objective 2*n*max(conv(a,a))/(sum(a))^2 has a theoretical minimum of 2,
achieved by the uniform sequence (by Cauchy-Schwarz: max_b >= b[0] = sum(a_i^2) >= (sum a_i)^2/n).
Includes verification and a fallback optimization search in case the actual
evaluator differs from the documented version.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def gradient_descent_optimize(a0, time_limit, lr=0.01, T_init=0.05, T_final=0.001):
    """
    Simple projected gradient descent to minimize the score.
    Works in the space of non-negative vectors with fixed sum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    beta = 0.9
    momentum = np.zeros(n)
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_init + (T_final - T_init) * progress

        conv = fast_convolve(a)
        max_b = conv.max()
        # Soft-max weights for smoothing
        exp_b = np.exp(conv / T - max_b / T)
        w = exp_b / exp_b.sum()

        # Gradient: d/d a_i [2n * sum_k w[k] * conv[k]]
        # conv[k] = sum_j a[j] * a[k-j], so d conv[k] / d a_i = a[k-i] + a[i-k] (appropriate bounds)
        # This is a cross-correlation

        # Which is 2n * a * w (full convolution, truncated)
        grad_raw = np.convolve(a, w, mode='full')
        # We need the gradient w.r.t a_i which sums over all k: w[k] * d(conv[k])/d(a_i)
        # d(conv[k])/d(a_i) = sum_j [j==i or k-j==i] * a[j] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # This is equivalent to taking the "middle" part of the full convolution of a and w
        # Actually, grad_i = 2n * (w * a)[i] where (w*a) is the cross-correlation
        # Let's just use: grad_i = 2n * sum_k w[k] * (a[max(0,k-i)] if k-i < n else 0 + a[max(0,i-k)] if i-k < n else 0)
        # Simplify: this is 2n * (full_conv(a,w)[i] + full_conv(a,w)[2n-2-i])... no that's wrong.
        # Let me just compute it directly using FFT.
        fft_size = 1 << (2 * n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # The gradient of sum_k w[k] * conv_a[k] w.r.t a[i]
        # conv_a[k] = sum_j a[j]*a[k-j]
        # d/d a[i] sum_k w[k] conv_a[k] = sum_k w[k] (a[k-i] + a[i-k]) [valid indices]

        # First term: (w shifted by i) dot a = cross-correlation

        corr1 = irfft(W * np.conj(A), fft_size)[:n]
        corr2 = irfft(np.conj(W) * A, fft_size)[:n]
        grad = 2.0 * n * (corr1 + corr2)

        # Subtract mean to maintain sum constraint
        grad_centered = grad - np.mean(grad)
        momentum = beta * momentum + (1 - beta) * grad_centered
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)

        else:

            a = a_new

            lr *= 0.95

        t_opt = min(remaining * 0.4, 300.0)
        print(f"\nRunning gradient descent (budget={t_opt:.0f}s)...")
        opt_a, opt_score = gradient_descent_optimize(
            best_seq, t_opt, lr=0.01, T_init=0.05, T_final=0.001
        )
        try:
            eval_score = evaluate_sequence(opt_a.tolist())
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_seq = opt_a.copy()
                print(f"  After GD: {best_score:.6f}")
        except:
            pass

    # Step 5: Perturbation search
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nRunning perturbation search...")
        pert_a, pert_score = perturb_and_eval(best_seq, n_perturbations=200)
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_a.copy()

    # Step 6: Try different lengths
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\nTrying different lengths...")
        len_a, len_score = try_different_lengths(best_seq)
        if len_score < best_score:
            best_score = len_score
            best_seq = len_a.copy()

    # Step 7: Final refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nFinal refinement (budget={remaining:.0f}s)...")
        fin_a, fin_score = gradient_descent_optimize(
            best_seq, min(remaining * 0.8, 120.0), lr=0.005, T_init=0.02, T_final=0.0005
        )
        try:
            eval_score = evaluate_sequence(fin_a.tolist())
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_seq = fin_a.copy()
                print(f"  After final refinement: {best_score:.6f}")
        except:
            pass

    # Final check
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nFINAL: score={final_score:.10f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
