def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def simplex_project(x):
    """Project x onto the probability simplex {p >= 0, sum p = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def subgradient_at(p, k_star):
    """Compute subgradient of max_k (p*p)[k] at p, given active lag k_star."""
    n = len(p)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idx = np.arange(js, je + 1)
    g[idx] = p[k_star - idx]
    return g


def nesterov_subgradient(p0, time_limit, lower_bound=None):
    """
    Nesterov accelerated projected subgradient method for minimizing
    max_k (p*p)[k] on the simplex.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0)
    s = p.sum()
    p = p / s
    
    if lower_bound is None:
        lower_bound = 1.0 / (2 * n - 1)  # theoretical lower bound
    
    x = p.copy()  # current iterate
    y = p.copy()  # Nesterov look-ahead
    t = 1.0
    
    best_p = p.copy()
    best_val = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    
    # Track best for Polyak step size
    f_opt_est = lower_bound  # running estimate of optimal value
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        b = fast_convolve(y)
        max_b = b.max()
        val = 2.0 * n * max_b
        
        if val < best_val:
            best_val = val
            best_p = y.copy()
        
        # Update f_opt estimate
        if val < f_opt_est + 1e-10:
            f_opt_est = val
        
        # Active set: all lags within 5% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Average subgradient over active set
        g = np.zeros(n)
        for k in active:
            g += subgradient_at(y, k)
        g /= len(active)
        
        # Polyak step size
        gap = 2.0 * n * max_b - 2.0 * n * f_opt_est
        if gap < 1e-15:
            gap = 1e-10
        eta = gap / (2.0 * np.dot(g, g) + 1e-15)
        eta = min(eta, 0.5)  # cap step size
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        
        x_new = simplex_project(y - eta * g)
        y = x_new + beta * (x_new - x)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x_new.copy()
        else:
            y = y / sy
        x = x_new
        t = t_new
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it+1} ({elapsed:.1f}s): best={best_val:.6f}")
    
    return best_p


def lbgfs_logsumexp(p0, time_limit, T_start=50.0, T_end=1.0, maxiter=5000):
    """
    L-BFGS optimization using log-sum-exp smoothing of the max-conv objective.
    Uses softmax parameterization.
    """
    n = len(p0)
    z = z_from_a(p0)
    t0 = time.monotonic()
    
    def make_func_and_jac(T_val):
        def func(zv):
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return 2.0 * n * lse / T_val
        
        def jac(zv):
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            # Convert to z-space
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        return func, jac
    
    # Multi-stage with restarts
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z_cur = z.copy()
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        iters = max(200, int(maxiter / n_stages))
        func, jac = make_func_and_jac(T_val)
        
        try:
            res = minimize(
                fun=func,
                x0=z_cur,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': iters * 50, 'disp': False}
            )
            z_cur = res.x
        except Exception:
            pass
        
        # Evaluate and potentially restart
        a_eval = softmax_from_z(z_cur)
    
    p_out = softmax_from_z(z_cur)
    return p_out


def block_coordinate_descent(p0, time_limit, block_size=2):
    """
    Block coordinate descent: optimize blocks of 2 coordinates at a time.
    For a block {i,j}, the objective max_k (p*p)[k] depends on p[i] and p[j]
    through the lags that involve i or j.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0)

    t0 = time.monotonic()
    
    # Precompute which lags are affected by each coordinate
    # Coordinate i affects lags k where p[i] appears, i.e., k = i+j for some j,
    # and k = j+i for some j. So lags in [i, i+n-1] intersect [0, 2n-2].
    
    n_passes = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            if p[i] < 1e-14:
                continue
            
            # Try cooperating with the next coordinate
            j = (i + 1) % n
            if p[j] < 1e-14 and j != i:
                j = (i + 2) % n
            
            if j == i:
                continue
            
            # Current values
            pi = p[i]
            pj = p[j]
            
            if pi + pj < 1e-14:
                continue
            
            # Remaining mass
            rest = 1.0 - pi - pj
            if rest < -1e-10:
                continue
            
            # Try different splits of the mass (pi + pj)
            total_ij = pi + pj
            best_frac = pi / total_ij if total_ij > 1e-14 else 0.5
            best_local = best_score
            
            for frac in np.linspace(0.0, 1.0, 21):
                new_pi = frac * total_ij
                new_pj = (1.0 - frac) * total_ij
                if new_pi < 0 or new_pj < 0:
                    continue
                if rest < -1e-10:
                    continue
                
                p_trial = p.copy()
                p_trial[i] = new_pi
                p_trial[j] = new_pj
                sc = 2.0 * n * fast_convolve(p_trial).max()
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_frac = frac
            
            if best_local < best_score - 1e-14:
                new_pi = best_frac * total_ij
                new_pj = (1.0 - best_frac) * total_ij
                p[i] = new_pi
                p[j] = new_pj
                best_score = best_local
                best_p = p.copy()
                improved = True
        
        if not improved:
            break
    
    return best_p

def generate_wave_init(n, rng):
    """Generate a wave-superposition initialization."""
    # Use 3-5 sinusoidal components with random frequencies and phases
    n_components = rng.integers(3, 7)
    freqs = rng.integers(1, n // 2, n_components)
    phases = rng.uniform(0, 2 * np.pi, n_components)
    amplitudes = rng.uniform(0.3, 1.0, n_components)
    
    x = np.arange(n)
    p = np.ones(n)
    for f, phi, amp in zip(freqs, phases, amplitudes):
        p += amp * np.cos(2 * np.pi * f * x / n + phi)
    
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s
    return p


def thue_morse_init(n):
    """Generate a Thue-Morse inspired initialization."""
    # Thue-Morse sequence: t(i) = popcount(i) mod 2

    # Map to [0, 1]

    # Add a small smoothing

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def softmax_from_z(z):
    """Convert log-domain z to probability via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def nesterov_subgradient(p0, time_limit, lower_bound_frac=0.51):
    """
    Nesterov accelerated projected subgradient method.
    Minimizes max_k (p*p)[k] on the simplex with O(1/t^2) convergence rate.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0)
    s = p.sum()
    p = p / s

    # Use a running lower bound estimate
    lb = lower_bound_frac / (2.0 * n - 1) * 2.0 * n  # in the 2n*max_b scale

    x = p.copy()
    y = p.copy()
    t = 1.0

    best_p = p.copy()
    best_val = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()

    f_best = best_val  # running best for Polyak step

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        val = 2.0 * n * max_b

        if val < best_val:
            best_val = val
            best_p = y.copy()
            f_best = val

        # Active set: lags within 10% of max
        threshold = max_b * 0.90
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Average subgradient over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(active)

        # Polyak step size
        gap = val - 2.0 * n * lb
        if gap < 1e-12:
            gap = 1e-10
        gg = np.dot(g, g)
        if gg < 1e-15:
            gg = 1e-15
        eta = gap / (2.0 * gg)
        eta = min(eta, 1.0)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new

        x_new = simplex_project(y - eta * g)
        y = x_new + beta * (x_new - x)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x_new.copy()
        else:
            y = y / sy
        x = x_new
        t = t_new

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it+1} ({elapsed:.1f}s): best={best_val:.6f}")

    return best_p


def lbgfs_logsumexp(p0, time_limit, T_start=50.0, T_end=1.0, maxiter=5000):
    """
    L-BFGS with log-sum-exp smoothing. Multi-stage temperature with restart.
    """
    n = len(p0)
    z = z_from_a(p0)
    t0 = time.monotonic()

    n_stages = max(4, int(maxiter / 1000))
    temps = np.geomspace(T_start, T_end, n_stages)
    z_cur = z.copy()

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = max(100, int(maxiter / n_stages))

        def func(zv):
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return 2.0 * n * lse / T_val

        def jac(zv):
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z_cur,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters * 60, 'disp': False}
            )
            z_cur = res.x
        except Exception:
            pass

        if (si + 1) % 2 == 0:
            elapsed = time.monotonic() - t0
            a_eval = softmax_from_z(z_cur)
            sc = 2.0 * n * fast_convolve(a_eval).max()
            print(f"    L-BFGS stage {si+1}/{n_stages} T={T_val:.1f} ({elapsed:.1f}s): score={sc:.6f}")

    p_out = softmax_from_z(z_cur)
    return p_out


def block_coord_descent(p0, time_limit, n_grid=31):
    """
    Block coordinate descent on pairs of coordinates.
    For pair (i,j), sweep the fraction of mass allocated to i vs j.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0)
    s = p.sum()
    p = p / s

    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for i in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break

            j = i + 1
            if j >= n:
                continue

            pi = p[i]
            pj = p[j]
            total_ij = pi + pj

            if total_ij < 1e-14:
                continue

            best_frac = pi / total_ij
            best_local = best_score

            # Fine grid sweep
            fracs = np.linspace(0.0, 1.0, n_grid)
            for frac in fracs:
                new_pi = frac * total_ij
                new_pj = (1.0 - frac) * total_ij
                p[i] = new_pi
                p[j] = new_pj
                sc = 2.0 * n * fast_convolve(p).max()
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_frac = frac

            # Restore best
            p[i] = best_frac * total_ij
            p[j] = (1.0 - best_frac) * total_ij

            if best_local < best_score - 1e-14:
                best_score = best_local
                best_p = p.copy()
                improved = True

        if not improved:
            break

        if n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD pass={n_passes} ({elapsed:.1f}s): score={best_score:.6f}")

    return best_p


def peak_shave_advanced(p0, time_limit):
    """
    Advanced peak shaving with gradient-guided mass redistribution.
    Tries multiple strategies: single coordinate reduction, pair reduction,
    and random perturbation.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0)
    s = p.sum()
    p = p / s

    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    no_improve = 0
    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(p)
        max_b = b.max()

        # Find top peak lags
        sorted_lags = np.argsort(b)[::-1]
        peak_k = int(sorted_lags[0])

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = p[idx] * p[peak_k - idx]
        order = np.argsort(contribs)[::-1]

        improved = False

        # Strategy 1: Reduce top contributor, add to a "safe" coordinate
        for ci in order[:5]:
            i = idx[ci]
            if p[i] < 1e-13:
                continue
            for factor in [0.96, 0.92, 0.87, 0.82]:
                new_val = p[i] * factor
                diff = p[i] - new_val
                if diff < 1e-15:
                    continue

                # Find safe target: coordinate not in any top-3 peak
                peak_coords_1 = set(idx.tolist())
                if len(sorted_lags) > 1:
                    pk2 = int(sorted_lags[1])
                    js2 = max(0, pk2 - n + 1)
                    je2 = min(n - 1, pk2)
                    peak_coords_1.update(range(js2, je2 + 1))

                candidates = [j for j in range(n) if j != i and j not in peak_coords_1]
                if not candidates:
                    candidates = [j for j in range(n) if j != i]

                if not candidates:
                    continue

                # Pick coordinate with smallest value (least disruption)
                min_cand = min(candidates, key=lambda j: p[j])

                p[i] = new_val
                p[min_cand] += diff
                sc = 2.0 * n * fast_convolve(p).max()

                if sc < best_score - 1e-14:
                    best_score = sc
                    best_p = p.copy()
                    improved = True
                    break
                else:
                    # Undo
                    p[i] = new_val + diff
                    p[min_cand] -= diff

            if improved:
                break

        # Strategy 2: Reduce pair of contributors
        if not improved and len(order) >= 2:
            for ci1 in order[:3]:
                for ci2 in order[:3]:
                    if ci1 == ci2:
                        continue
                    i1 = idx[ci1]
                    i2 = idx[ci2]
                    if p[i1] < 1e-13 or p[i2] < 1e-13:
                        continue
                    for f1 in [0.94, 0.90]:
                        for f2 in [0.94, 0.90]:
                            d1 = p[i1] * (1 - f1)
                            d2 = p[i2] * (1 - f2)
                            p[i1] *= f1
                            p[i2] *= f2
                            # Redistribute to lowest non-peak coordinate
                            total_diff = d1 + d2
                            peak_set = set(idx.tolist())
                            safe = [j for j in range(n) if j not in peak_set]
                            if safe:
                                target = min(safe, key=lambda j: p[j])
                                p[target] += total_diff
                            sc = 2.0 * n * fast_convolve(p).max()
                            if sc < best_score - 1e-14:
                                best_score = sc
                                best_p = p.copy()
                                improved = True
                                break
                            else:
                                p[i1] /= f1
                                p[i2] /= f2
                                if safe:
                                    p[target] -= total_diff
                        if improved:
                            break
                    if improved:
                        break

        # Strategy 3: Random perturbation (escape local minima)
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0 or no_improve > 100:
                for _ in range(20):
                    i, j = rng.integers(0, n, 2)
                    if i == j:
                        continue
                    mass = min(p[i], p[j]) * rng.uniform(0.02, 0.08)
                    p[i] -= mass
                    p[j] += mass
                    sc = 2.0 * n * fast_convolve(p).max()
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_p = p.copy()
                        improved = True
                        no_improve = 0
                        break
                    else:
                        p[i] += mass
                        p[j] -= mass
                if no_improve > 500:
                    break

        if improved:
            no_improve = 0

    return best_p


def generate_wave_init(n, rng):
    """Generate wave-superposition initialization with random structure."""
    n_comp = int(rng.integers(2, 6))
    x = np.arange(n, dtype=np.float64)
    p = np.ones(n, dtype=np.float64)
    for _ in range(n_comp):
        f = float(rng.integers(1, max(2, n // 2)))
        phi = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.8)
        p += amp * np.cos(2.0 * np.pi * f * x / n + phi)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return p / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate():
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines Nesterov accelerated subgradient, multi-temperature L-BFGS,
    block coordinate descent, and gradient-guided peak shaving.
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + 1780.0  # Leave 20s buffer

    best_seq = None
    best_score = float('inf')

    # === Load parent or generate initial ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = generate_wave_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Generated init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Nesterov accelerated subgradient (200s) ===
    print(f"\n=== Phase 1: Nesterov SG (best={best_score:.6f}) ===")
    t1 = 200.0
    if time.monotonic() + t1 < deadline:
        a_ns = nesterov_subgradient(best_seq, t1)
        try_update(a_ns, "nesterov_sg")

    # === Phase 2: Multi-start L-BFGS (700s) ===
    print(f"\n=== Phase 2: Multi-Start L-BFGS (best={best_score:.6f}) ===")
    t2 = min(700.0, deadline - time.monotonic() - 500.0)
    if t2 > 60:
        # Start 0: direct from current best
        print(f"  L-BFGS start 0: T=80->1, budget={t2*0.5:.0f}s")
        a_s0 = lbgfs_logsumexp(best_seq, t2 * 0.5, T_start=80.0, T_end=1.0, maxiter=6000)
        try_update(a_s0, "lbgfs_s0")

        # Start 1: perturbed
        rem = deadline - time.monotonic() - 400.0
        if rem > 40:
            noise = rng.standard_normal(n_best) * 0.05 / np.sqrt(n_best)
            p_pert = best_seq + noise
            p_pert = np.maximum(p_pert, 0)
            p_pert = p_pert / p_pert.sum()
            print(f"  L-BFGS start 1: T=20->0.5, budget={rem:.0f}s")
            a_s1 = lbgfs_logsumexp(p_pert, rem, T_start=20.0, T_end=0.5, maxiter=5000)
            try_update(a_s1, "lbgfs_s1")

        # Start 2: different temperature profile
        rem = deadline - time.monotonic() - 300.0
        if rem > 30:
            print(f"  L-BFGS start 2: T=10->0.3, budget={rem:.0f}s")
            a_s2 = lbgfs_logsumexp(best_seq, rem, T_start=10.0, T_end=0.3, maxiter=4000)
            try_update(a_s2, "lbgfs_s2")

    # === Phase 3: Block coordinate descent (300s) ===
    print(f"\n=== Phase 3: Block Coord Descent (best={best_score:.6f}) ===")
    t3 = min(300.0, deadline - time.monotonic() - 200.0)
    if t3 > 20:
        a_bd = block_coord_descent(best_seq, t3, n_grid=21)
        try_update(a_bd, "block_cd")

    # === Phase 4: Peak shaving (200s) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    t4 = min(200.0, deadline - time.monotonic() - 100.0)
    if t4 > 15:
        a_ps = peak_shave_advanced(best_seq, t4)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Final L-BFGS polish (100s) ===
    print(f"\n=== Phase 5: Final Polish L-BFGS (best={best_score:.6f}) ===")
    t5 = min(100.0, deadline - time.monotonic() - 30.0)
    if t5 > 15:
        a_fin = lbgfs_logsumexp(best_seq, t5, T_start=5.0, T_end=0.2, maxiter=3000)
        try_update(a_fin, "final_lbgfs")

    # === Phase 6: Final peak shave ===
    t6 = min(60.0, deadline - time.monotonic() - 5.0)
    if t6 > 10:
        a_ps2 = peak_shave_advanced(best_seq, t6)
        try_update(a_ps2, "final_ps")

    # === Phase 7: Try alternative lengths if time remains ===
    rem = deadline - time.monotonic()
    if rem > 80:
        alt_n = int(n_best * 1.2)
        print(f"\n=== Phase 7: Alt length n={alt_n} ===")
        init_alt = interpolate_sequence(best_seq, alt_n)
        t_alt = min(rem * 0.6, 120.0)
        a_alt = lbgfs_logsumexp(init_alt, t_alt, T_start=30.0, T_end=1.0, maxiter=3000)
        try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None:
            a_alt_ps = peak_shave_advanced(best_seq, min(rem - 5, 40.0))
            try_update(a_alt_ps, f"alt_ps")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
