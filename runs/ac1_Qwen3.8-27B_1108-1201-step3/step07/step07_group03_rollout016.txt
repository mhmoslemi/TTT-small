def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)

        theta = (cssv[rho_val - 1] - 1.0) / rho_val
        return np.maximum(v - theta, 0.0)

    if old_rest > 1e-15:
        a2[np.arange(n) != k] = a[np.arange(n) != k] * (rest / old_rest)
    else:
        mask = np.arange(n) != k
        a2[mask] = rest / (n - 1 if n > 1 else 1)
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a2 / s


def softmax_gradient_descent(a0, time_limit, T_init=None, T_min=0.0001,
                             T_decay=0.9995, alpha_init=0.5,
                             verbose=False, stop_on_stuck=True):
    """Adaptive softmax-smoothed projected gradient descent."""
    n = len(a0)
    a = normalize(a0)
    if T_init is None:
        T_init = compute_score(a) / (2.0 * n) * 0.04
    T = max(T_init, 0.00005)
    alpha = alpha_init
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stuck_counter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autoconv(a)
        w = softmax_stable(b, T)
        g = 2.0 * hankel_matvec(w, a)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break

        accepted = False
        for _ in range(20):
            a_trial = simplex_project(a - alpha * g)
            sc = compute_score(a_trial)
            if sc < best_score - 1e-16:
                a = a_trial
                best_score = sc
                best_a = a_trial.copy()
                no_improve = 0
                stuck_counter = 0
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            no_improve += 1
            stuck_counter += 1
            alpha = max(alpha_init * 0.3, alpha * 2.0)
            if no_improve > 400:
                # Reset
                alpha = alpha_init
                no_improve = 0
                stuck_counter = 0
            if stop_on_stuck and stuck_counter > 600:
                break

        T = max(T_decay * T, T_min)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.1f}s): score={best_score:.8f} T={T:.6f}")

    return best_a, best_score, it


def direct_linearization(a0, time_limit, verbose=False):
    """Direct auto-convolution linearization (LP-free local optimization)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autoconv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]

        # Subgradient at peak k_star: g[m] = 2*a[k_star - m] if valid
        g = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                g[m] = 2.0 * a[j]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break

        accepted = False
        alpha = min(0.5, 0.1 * M / (g_norm * g_norm))
        alpha = max(alpha, 0.001)
        for _ in range(30):
            a_trial = simplex_project(a - alpha * g)
            sc = compute_score(a_trial)
            if sc < best_score - 1e-16:
                a = a_trial
                best_score = sc
                best_a = a_trial.copy()
                no_improve = 0
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            no_improve += 1
            if no_improve > 300:
                break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Lin it={it} ({elapsed:.1f}s): score={best_score:.8f}")

    return best_a, best_score, it


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with golden-section 1D line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        perm = rng.permutation(n)
        for k in perm:
            if time.monotonic() - t0 > time_limit:
                break
            uk = a[k]
            if uk < 1e-14:
                continue
            # 1D search: new value for a[k]
            lo = max(0.0, uk * 0.2)
            hi = min(0.9, uk * 3.0)
            if hi - lo < 1e-12:
                continue
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            a_c = adjust_to_sum1(a, k, c - uk)
            s_c = compute_score(a_c)
            a_d = adjust_to_sum1(a, k, d - uk)
            s_d = compute_score(a_d)
            for _ in range(25):
                if hi - lo < 1e-12:
                    break
                if s_c < s_d:
                    hi = d
                    d = c
                    s_d = s_c
                    c = hi - gr * (hi - lo)
                    a_c = adjust_to_sum1(a, k, c - uk)
                    s_c = compute_score(a_c)
                else:
                    lo = c
                    c = d
                    s_c = s_d
                    d = lo + gr * (hi - lo)
                    a_d = adjust_to_sum1(a, k, d - uk)
                    s_d = compute_score(a_d)
            best_t = (lo + hi) / 2.0
            a_b = adjust_to_sum1(a, k, best_t - uk)
            s_b = compute_score(a_b)
            if s_b < best_score - 1e-14:
                a = a_b
                best_score = s_b
                best_a = a_b.copy()
                improved = True

        if not improved:
            break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): score={best_score:.8f}")

    return best_a, best_score, sweep


def dummy_lp_iter(a, b):
    """Single fictitious play iteration - solve min max_k (a*b)_k via subgradient on simplex."""
    n = len(a)
    result = a.copy()
    best_score = compute_score(a)
    for it in range(50):
        ab = fast_autoconv(a * b)
        k = int(np.argmax(ab))
        M_val = ab[k]
        g = np.zeros(n)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                g[m] = 2.0 * a[j]
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break
        alpha = min(0.5, 0.1 * M_val / (g_norm * g_norm))
        b_new = simplex_project(b - alpha * g)
        sc = compute_score(b_new)
        if sc < best_score - 1e-15:
            result = b_new.copy()
            best_score = sc
        b = b_new
    return result, best_score


def fp_round(a0, time_limit, verbose=False):
    """Fictitious play round using direct subgradient (no LP solver)."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        a_new, sc_a = dummy_lp_iter(b, a)
        if sc_a < best_score - 1e-15:
            best_score = sc_a
            best_a = a_new.copy()
        a = a_new
        if time.monotonic() - t0 > time_limit:
            break
        b_new, sc_b = dummy_lp_iter(a, b)
        if sc_b < best_score - 1e-15:
            best_score = sc_b
            best_a = b_new.copy()
        b = b_new
        avg = 0.5 * (a + b)
        sc_avg = compute_score(avg)
        if sc_avg < best_score - 1e-15:
            best_score = sc_avg
            best_a = avg.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FP ({elapsed:.1f}s): score={best_score:.8f}")

    return best_a, best_score


def init_fourier(n, rng):
    """Fourier-based initialization."""
    n_harm = int(rng.integers(4, 20))
    amps = rng.random(n_harm) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    return normalize(a)


def init_blocks(n, rng):
    """Block-based initialization."""
    n_blocks = int(rng.integers(4, 20))
    block_size = max(2, n // n_blocks)
    amps = rng.random(n_blocks) + 0.1
    a = np.zeros(n)
    for bi in range(n_blocks):
        s = bi * block_size
        e = min(s + block_size, n)
        a[s:e] = amps[bi]
    a = np.convolve(a, np.ones(5) / 5.0, mode='same')
    return normalize(a)


def init_noise(n, base, rng, sigma=0.08):
    """Noisy perturbation of base."""
    a = base.copy() + rng.standard_normal(n) * sigma
    return normalize(a)


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, maintain sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    return normalize(fi)


def check_eval(a, label=""):
    """Evaluate with the official evaluator."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return a, float('inf')
    try:
        sc = float(evaluate_sequence(a.tolist()))
    except Exception:
        return a, float('inf')
    if np.isfinite(sc) and sc < 2.0:
        print(f"  [{label}] n={len(a)} eval_score={sc:.8f}")
    return a, sc


def propose_candidate():
    """
    Multi-phase optimizer:
    - Direct auto-conv linearization (fast, no LP solver)
    - Adaptive softmax gradient descent with annealing
    - Coordinate descent with golden-section search
    - Multi-length exploration and diverse initialization
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(20250615)

    best_seq = None
    best_eval = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 0.01:
            parent = p / p.sum()
            _, pe = check_eval(parent, "parent")
            if pe < best_eval:
                best_eval = pe
                best_seq = parent.copy()
            print(f"Parent loaded: n={len(parent)}, score={pe:.8f}")
    except Exception:
        pass

    # ========== PHASE 0: Quick direct linearization on parent ==========
    if parent is not None:
        rem = deadline - time.monotonic()
        t0_phase = min(30.0, rem * 0.05)
        if t0_phase > 5:
            print(f"\n=== Phase 0: Quick Direct Linearization ({t0_phase:.0f}s) ===")
            a_lin, sc_lin, _ = direct_linearization(parent, t0_phase, verbose=False)
            a_lin, ev_lin = check_eval(a_lin, "direct_lin")
            if ev_lin < best_eval:
                best_eval = ev_lin
                best_seq = a_lin.copy()

    # ========== PHASE 1: Diverse initializations ==========
    print(f"\n=== Phase 1: Diverse Init (best_eval={best_eval:.8f}) ===")
    t1_budget = min(60.0, (deadline - time.monotonic()) * 0.06)

    n_values = []
    if parent is not None:
        n_p = len(parent)
        n_values.append(n_p)
        for factor in [0.75, 0.85, 1.15, 1.25, 1.5]:
            alt_n = int(n_p * factor)
            if 256 <= alt_n <= 2560 and alt_n not in n_values:
                n_values.append(alt_n)
    else:
        n_values = [512, 640, 768, 896, 1024, 1280]

    n_values = sorted(set(n_values))[:6]
    print(f"  Lengths to try: {n_values}")

    candidates = []
    for n in n_values:
        if time.monotonic() > deadline - t1_budget - 300:
            break
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_fourier(n, rng))
        inits.append(init_blocks(n, rng))
        if parent is not None:
            inits.append(init_noise(n, interpolate_seq(parent, n), rng, 0.1))

        for a_init in inits:
            # Quick gradient descent
            a_opt, sc_opt, _ = softmax_gradient_descent(a_init, 3.0, stop_on_stuck=True)
            candidates.append((sc_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best local: {candidates[0][0]:.8f}")

    for sc_c, a_c, n_c in candidates[:6]:
        a_ev, ev = check_eval(a_c, f"p1_n{n_c}")
        if ev < best_eval:
            best_eval = ev
            best_seq = a_ev.copy()

    # ========== PHASE 2: Main gradient descent on best ==========
    print(f"\n=== Phase 2: Main SGD (best_eval={best_eval:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t2 = min(300.0, rem * 0.4)
        if t2 > 10:
            print(f"  Budget: {t2:.0f}s")
            a_sg, sc_sg, iters = softmax_gradient_descent(best_seq, t2, verbose=True)
            a_ev, ev = check_eval(a_sg, f"sgd_n{len(a_sg)}_it{iters}")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

            # Follow with direct linearization
            rem = deadline - time.monotonic() - 300
            t2b = min(100.0, rem * 0.3)
            if t2b > 10:
                a_dl, sc_dl, _ = direct_linearization(a_sg, t2b, verbose=False)
                a_ev, ev = check_eval(a_dl, "after_sgd_lin")
                if ev < best_eval:
                    best_eval = ev
                    best_seq = a_ev.copy()

    # ========== PHASE 3: Coordinate descent polishing ==========
    print(f"\n=== Phase 3: Coordinate Descent (best_eval={best_eval:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t3 = min(200.0, rem * 0.4)
        if t3 > 10:
            print(f"  Budget: {t3:.0f}s")
            a_cd, sc_cd, sweeps = coordinate_descent(best_seq, t3, verbose=True)
            a_ev, ev = check_eval(a_cd, f"cd_sweeps{sweeps}")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

    # ========== PHASE 4: FP-style alternating + refine ==========
    print(f"\n=== Phase 4: FP Alternating (best_eval={best_eval:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        t4 = min(150.0, rem * 0.5)
        if t4 > 15:
            print(f"  Budget: {t4:.0f}s")
            a_fp, sc_fp = fp_round(best_seq, t4, verbose=True)
            a_ev, ev = check_eval(a_fp, "fp")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

            # Polish with SGD
            rem = deadline - time.monotonic() - 100
            t4b = min(60.0, rem * 0.4)
            if t4b > 10:
                a_sg2, sc_sg2, _ = softmax_gradient_descent(a_fp, t4b, T_init=0.0005,
                                                             T_min=0.00005, T_decay=0.9997)
                a_ev, ev = check_eval(a_sg2, "sgd_after_fp")
                if ev < best_eval:
                    best_eval = ev
                    best_seq = a_ev.copy()

    # ========== PHASE 5: Multi-length exploration ==========
    print(f"\n=== Phase 5: Multi-Length (best_eval={best_eval:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.8, 0.9, 1.1, 1.25]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2560 and alt_n != n_best:
                alt_lengths.append(alt_n)
        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alt = min(50.0, rem * 0.4)
            if t_alt < 10:
                continue
            a_interp = interpolate_seq(best_seq, alt_n)
            a_opt, sc_opt, _ = softmax_gradient_descent(a_interp, t_alt, verbose=False)
            a_cd2, sc_cd2, _ = coordinate_descent(a_opt, min(20.0, t_alt * 0.4))
            a_ev, ev = check_eval(a_cd2, f"alt_n{alt_n}")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

    # ========== PHASE 6: Final intensive polishing ==========
    print(f"\n=== Phase 6: Final Polish (best_eval={best_eval:.8f}) ===")
    if best_seq is not None:
        # Direct linearization
        rem = deadline - time.monotonic() - 20
        t6a = min(60.0, rem * 0.3)
        if t6a > 5:
            a_lin2, sc_lin2, _ = direct_linearization(best_seq, t6a, verbose=False)
            a_ev, ev = check_eval(a_lin2, "final_lin")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

        # Coordinate descent
        rem = deadline - time.monotonic() - 15
        t6b = min(50.0, rem * 0.4)
        if t6b > 5:
            a_cd3, sc_cd3, _ = coordinate_descent(best_seq, t6b, verbose=False)
            a_ev, ev = check_eval(a_cd3, "final_cd")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

        # Very fine SGD
        rem = deadline - time.monotonic() - 5
        t6c = min(20.0, rem * 0.5)
        if t6c > 3:
            a_sg3, sc_sg3, _ = softmax_gradient_descent(best_seq, t6c, T_init=0.0002,
                                                         T_min=0.00005, T_decay=0.9998,
                                                         alpha_init=0.2)
            a_ev, ev = check_eval(a_sg3, "final_sgd")
            if ev < best_eval:
                best_eval = ev
                best_seq = a_ev.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_eval = float(evaluate_sequence(best_seq.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_eval = final_score

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_eval:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
