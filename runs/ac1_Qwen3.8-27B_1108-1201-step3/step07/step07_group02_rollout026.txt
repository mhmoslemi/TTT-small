```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated L-BFGS with multi-stage temperature annealing
2. Multi-peak subgradient method with backtracking line search
3. LP best-response iteration with exponential averaging
4. Parabolic peak shaving with exact 1D search
5. Frequency-domain diversification to escape local minima
6. Exploration of alternative sequence lengths (n=512, 1024)
"""
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def eval_score(a):
    """Compute the objective for a non-negative sequence with sum=1."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()


def softmax_of_z(z):
    """Convert log-domain z to probability vector."""
    z_sh = z - z.max()
    e = np.exp(z_sh)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def z_of_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def simplex_project(x):
    """Project onto probability simplex (sum=1, x>=0)."""
    n = len(x)
    u = np.sort(x)[::-1]
    cv_ = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv_ - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cv_[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def soft_max_obj_val(a, n, T):
    """Compute soft-max smoothed objective."""
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    return (2.0 * n / T) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum()))


def soft_max_grad_a(a, n, T):
    """Compute gradient of soft-max objective w.r.t. a (before projection)."""
    b = fast_autocorr(a)
    Tb = T * b
    w = np.exp(Tb - Tb.max())
    w /= w.sum()
    # grad_a[i] = sum_k w[k] * a[k-i] for valid pairs
    # This is the cross-correlation of w and a
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = np.dot(w[i:i + n], a)
    return grad_a


def make_soft_max_func(a, n, T):
    """Create objective function in z-space."""
    def func(z):
        a_val = softmax_of_z(z)
        return soft_max_obj_val(a_val, n, T)
    return func


def make_soft_max_jac(a, n, T):
    """Create gradient function in z-space."""
    def jac(z):
        a_val = softmax_of_z(z)
        grad_a = soft_max_grad_a(a_val, n, T)
        S = np.dot(grad_a, a_val)
        return a_val * (grad_a - S)
    return jac


def lbgfs_multi_stage(a0, time_limit, T_stages, iters_per_stage):
    """L-BFGS with multiple temperature stages."""
    n = len(a0)
    z = z_of_a(a0)
    t0 = time.monotonic()
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=make_soft_max_func(a0, n, T),
                x0=z,
                jac=make_soft_max_jac(a0, n, T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 30,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_of_z(z)


def multi_peak_subgrad(a0, time_limit, n_peaks=5, lr_init=0.01):
    """Multi-peak subgradient descent with backtracking."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_autocorr(a)
        b_max = b.max()
        score = 2.0 * n * b_max

        # Find top-K peaks
        threshold = b_max - 0.05 * b_max
        active = np.where(b >= threshold)[0]
        if len(active) > n_peaks:
            top_idx = np.argsort(b)[::-1][:n_peaks]
            active = top_idx
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                g[i] += a[k - i]
        g /= len(active)

        # Backtracking line search
        old_score = score
        accepts = False
        for attempt in range(20):
            trial_lr = lr * (0.5 ** attempt)
            a_new = simplex_project(a - trial_lr * g)
            new_score = eval_score(a_new)
            if new_score < old_score - 1e-14:
                a = a_new
                accepts = True
                break
        if accepts:
            lr = min(lr * 1.05, 0.5)
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr *= 0.7

        if no_improve > 50:
            lr = max(lr * 0.5, 1e-8)
            no_improve = 0

        if step % 2000 == 0:
            print(f"    MPS {time.monotonic()-t0:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

def solve_lp_br(a_fixed, solver):
    """Solve LP: min M s.t. conv(a,b)_k <= M, b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint matrix: row k has a_fixed[j] at column (k-j) for valid j
    # conv(a,b)[k] = sum_j a[j]*b[k-j]
    # So the matrix A has A[k, col] = a_fixed[k-col] if 0<=col<n and 0<=k-col<n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            col = k - j
            if 0 <= col < n:
                A[k, col] = a_fixed[j]
    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_refine_exp_avg(a0, time_limit, solver, n_iters=2000):
    """LP best-response with exponential moving average."""
    n = len(a0)
    a = a0.copy()
    ema = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    beta = 0.1  # EMA coefficient

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_br(ema, solver)
        if b_new is None:
            continue
        # Update EMA
        ema = (1 - beta) * ema + beta * b_new
        ema = np.maximum(ema, 0)
        ema = ema / ema.sum()
        # Also update a
        a = (1 - beta) * a + beta * b_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        # Evaluate both
        for cand in [a, ema]:
            sc = eval_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if (it + 1) % 100 == 0:
            # Decay beta over time
            beta = max(0.01, beta * 0.995)
            print(f"    LP {time.monotonic()-t0:.0f}s: score={best_score:.8f} it={it+1}")

    return best_a, best_score


def peak_shave_parabolic(a0, time_limit, n_steps=5000):
    """Peak shaving with parabolic interpolation for step size."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_autocorr(a)
        # Find peaks excluding edges
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peaks = np.argsort(b_inner)[::-1][:3]

        improved = False
        for pk in peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Find top contributor
            top_c = np.argmax(contribs)
            i_main = idx[top_c]
            j_main = pk - i_main
            if a[i_main] < 1e-14:
                continue

            # Parabolic search for optimal reduction factor
            best_f = 1.0
            best_sc = best_score
            for factor in [0.98, 0.95, 0.92, 0.88, 0.82, 0.75]:
                a_trial = a.copy()
                a_trial[i_main] *= factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                sc = eval_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_f = factor
            if best_f < 1.0:
                a[i_main] *= best_f
                a = np.maximum(a, 0)
                a /= a.sum()
                best_score = best_sc
                best_a = a.copy()
                improved = True

            # Also try reducing j_main
            if a[j_main] > 1e-14 and j_main != i_main:
                for factor in [0.98, 0.95, 0.9]:
                    a_trial = a.copy()
                    a_trial[j_main] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    sc = eval_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            # Try simultaneous reduction of top 2 contributors at main peak
            pk = peaks[0]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top2 = np.argsort(contribs)[::-1][:2]
            i1, i2 = idx[top2[0]], idx[top2[1]]
            done = False
            for f1 in [0.97, 0.93, 0.88]:
                for f2 in [0.97, 0.93, 0.88]:
                    a_trial = a.copy()
                    a_trial[i1] *= f1
                    a_trial[i2] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s
                    sc = eval_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        done = True
                        break
                if done:
                    break
            if not improved:
                break

    return best_a, best_score


def freq_diversify(a0, rng, n_trials=50):
    """Frequency domain perturbation to escape local minima."""
    n = len(a0)
    best = a0.copy()
    best_sc = eval_score(a0)
    N = 1 << (2 * n - 1).bit_length()

    for _ in range(n_trials):
        A = rfft(a0, N)
        # Random phase perturbation (preserve DC)
        phase_pert = rng.random(N // 2 + 1) * 0.3
        A_pert = A * np.exp(1j * phase_pert)
        A_pert[0] = A[0]  # Preserve DC
        a_pert = irfft(A_pert, N)[:n]
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert /= s
        sc = eval_score(a_pert)
        if sc < best_sc:
            best_sc = sc
            best = a_pert.copy()
    return best, best_sc


def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new /= ss
    return a_new


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def safe_evaluate(a):
    """Safely evaluate a sequence using the global evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        result = evaluate_sequence(a_list)
        return float(result)
    except Exception:
        return float('inf')
```
