def simulate_annealing(a0, time_limit, T_init=0.01, T_final=0.0001, max_iters=100000):
    """Simulated annealing with Metropolis acceptance."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        if monotonic() - t0 > time_limit:
            break
        if (it+1) % 1000 == 0 and (it+1) // 1000 > n_iters_estimate // 1000:
            break

        progress = min(1.0, (it+1) / max_iters)
        T = T_init * (T_final / T_init) ** progress

        # Mix of small local moves and occasional larger moves
        if rng.rand() < 0.1:
            # Larger move: perturb a random subset
            n_perturb = max(1, n // 10)
            idx = rng.choice(n, n_perturb, replace=False)
            a_new = a.copy()
            a_new[idx] *= np.exp(rng.standard_normal(n_perturb) * 0.3)
        else:
            # Small move
            a_new = a + rng.standard_normal(n) * 0.005
        
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        new_score = score_normalized(a_new)
        score_diff = new_score - best_score
        
        # Metropolis acceptance
        if new_score < a_score or rng.rand() < np.exp(-score_diff / max(T, 1e-10)):
            a = a_new
            a_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
        else:
            a_score = score_normalized(a)  # recompute if needed
    
    return best_a, best_score

So max_k(C_k + d_k*t + e_k*t^2) = max(
    max_{k in const} C_k,
    max_{k in linear} (C_k + d_k*t),
    C_{2j} + t^2
)

The "const" max is just max of b[k] over k not in [j, j+n-1]. This is O(n) to compute but is the same for all t (for a given j).

The "linear" max is max over at most n-1 values of (C_k + d_k*t). This is the max of linear functions, which is a convex piecewise-linear function of t.

So f(t) = 2n * max(M_const, M_linear(t), C_{2j} + t^2) / (S' + t)^2

where S' = S_old - a[j].

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with golden section search,
combined with L-BFGS smoothing and multi-length exploration.

Key innovations over previous approach:
1. Exact 1D coordinate optimization (no LP overhead per step)
2. Golden section search for each coordinate with analytic structure exploitation
3. Multi-scale length exploration with adaptive time allocation
4. Smooth L-BFGS polish using log-sum-exp
5. Strategic restarts with structured perturbations
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the evaluation score for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def logsumexp_smooth(a, T):
    """Smooth approximation of max using log-sum-exp."""
    b = fast_convolve(a)
    b_max = b.max()
    return b_max + T * np.log(np.exp((b - b_max) / T).sum())


def lbfgs_smooth_obj(z, T, n):
    """Objective and gradient for L-BFGS in log-space."""
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    lse = b_max + T * np.log(exp_scaled.sum())
    value = 2.0 * n * lse

    # Gradient computation
    # db_k/da_j = 2*a_{k-j} if max(0,k-n+1) <= j <= min(k,n-1)
    # df/da_j = 2n * sum_k w_k * 2*a_{k-j} * indicator = 4n * sum_{m=0}^{n-1} w_{j+m} * a[m]
    size = 2 * n - 1
    a_padded = np.zeros(size)
    a_padded[:n] = a
    a_flip = a_padded[::-1]
    conv_size = 2 * size - 1
    cfs = 1 << (conv_size - 1).bit_length()
    conv_result = irfft(rfft(w, cfs) * rfft(a_flip, cfs), cfs)[:conv_size]
    grad_a = 4.0 * n * conv_result[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return value, grad_z


def lbfgs_refine(a0, time_limit, T=20, maxiter=2000):
    """L-BFGS refinement in log-space with log-sum-exp smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe) - np.log(a_safe).mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        v, _ = lbfgs_smooth_obj(z, T, n)
        return v

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(len(z))
        _, g = lbfgs_smooth_obj(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            a_out = np.exp(z_final - z_final.max())
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0
    return a_out

def peak_shave_local(a0, time_limit, max_steps=2000):

        peak_val = b[peak_k]
        target_val = best_score * S * S / (2.0 * n)

        if peak_val <= target_val * 1.001:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_n = min(8, len(contribs))
        top_contrib_idx = np.argsort(contribs)[::-1][:top_n]

        improved = False
        for ci in top_contrib_idx:
            i = int(idx[ci])
            j_comp = peak_k - i
            if j_comp < 0 or j_comp >= n or j_comp == i:
                continue
            if a[i] < 1e-14 or a[j_comp] < 1e-14:
                continue

            old_i = a[i]
            old_j = a[j_comp]

            for f1 in [0.95, 0.9, 0.85, 0.8]:
                for f2 in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    a_trial[i] = old_i * f1
                    a_trial[j_comp] = old_j * f2
                    # Redistribute the difference
                    diff = (old_i - a_trial[i]) + (old_j - a_trial[j_comp])
                    if diff < 1e-15:

                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j_comp] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:

                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try small random perturbation
            for attempt in range(3):
                a_trial = a + rng.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                sc = score_of(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 30:
                break
        else:
            failed = 0

        if (step + 1) % 200 == 0:
            print(f"    PeakShave step={step+1} ({monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def hann_window_init(n):
    """Create a Hann-window initialized sequence."""
    t = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * t / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def raised_cosine_init(n):
    """Raised cosine (sin^2) initialization."""
    t = np.arange(n)
    a = np.sin(np.pi * (t + 0.5) / n) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate():
    """
    Multi-phase optimization:
    1. Coordinate descent (main workhorse) at multiple lengths
    2. L-BFGS smooth refinement
    3. Peak shaving local search
    4. Multi-length exploration
    """
    np.random.seed(42)
    deadline = monotonic() + 1780.0  # leave 20s buffer

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = hann_window_init(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial (Hann): n=512, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Coordinate Descent at parent length ===
    print(f"\n=== Phase 1: Coordinate Descent n={n_parent} ===")
    rem = deadline - monotonic() - 600
    t_cd1 = max(120, min(rem * 0.6, 800))
    print(f"  Budget: {t_cd1:.0f}s")
    a_cd, sc_cd = coordinate_descent(best_seq, t_cd1, verbose=True, seed=42)
    best_seq, best_score = check_and_update(a_cd, f"CD_n{n_parent}", best_seq, best_score)

    # === Phase 2: L-BFGS polish ===
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.6f}) ===")
    for T in [15, 5, 2]:
        rem = deadline - monotonic() - 300
        if rem < 20:
            break
        t_lb = min(rem * 0.25, 100)
        print(f"  L-BFGS T={T}, budget={t_lb:.0f}s")
        a_lb = lbfgs_refine(best_seq, t_lb, T=T, maxiter=1500)
        best_seq, best_score = check_and_update(a_lb, f"LBF_T{T}_n{len(best_seq)}", best_seq, best_score)

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 250
    if rem > 20:
        t_ps = min(rem * 0.3, 150)
        print(f"  Budget: {t_ps:.0f}s")
        a_ps, _ = peak_shave_local(best_seq, t_ps)
        best_seq, best_score = check_and_update(a_ps, f"PS_n{len(best_seq)}", best_seq, best_score)

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for factor in [0.75, 1.0, 1.25, 1.5, 2.0]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n not in candidate_lengths:
            candidate_lengths.append(alt_n)
    # Also try some standard lengths
    for std_n in [512, 768, 1024, 1536, 2048]:
        if 256 <= std_n <= 4096 and std_n not in candidate_lengths:
            candidate_lengths.append(std_n)

    candidate_lengths.sort(key=lambda x: abs(x - n_best))
    candidate_lengths = candidate_lengths[:4]

    for alt_n in candidate_lengths:
        rem = deadline - monotonic() - 100
        if rem < 60:
            break
        if alt_n == n_best:
            # Already tried, do a second CD round with different seed
            t_alloc = min(rem * 0.3, 200)
            print(f"  CD round 2: n={alt_n}, budget={t_alloc:.0f}s")
            a_cd2, _ = coordinate_descent(best_seq, t_alloc, verbose=False, seed=1234)
            best_seq, best_score = check_and_update(a_cd2, f"CD2_n{alt_n}", best_seq, best_score)
        else:
            t_alloc = min(rem * 0.3, 250)
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Try different inits
            inits = [init_alt, hann_window_init(alt_n), raised_cosine_init(alt_n)]
            best_for_n = init_alt
            best_for_n_score = score_of(init_alt)
            for init_i, init in enumerate(inits):
                t_sub = t_alloc / len(inits)
                a_cd_n, sc_n = coordinate_descent(init, t_sub, verbose=False, seed=100 + init_i)
                if sc_n < best_for_n_score:
                    best_for_n_score = sc_n
                    best_for_n = a_cd_n
            best_seq, best_score = check_and_update(best_for_n, f"CD_n{alt_n}", best_seq, best_score)
            n_best = len(best_seq)

    # === Phase 5: Final L-BFGS + peak shaving ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 20
    if rem > 15:
        t_final_lb = min(rem * 0.4, 80)
        a_final = lbfgs_refine(best_seq, t_final_lb, T=3, maxiter=1000)
        best_seq, best_score = check_and_update(a_final, f"FinalLBF_n{len(best_seq)}", best_seq, best_score)

    rem = deadline - monotonic() - 10
    if rem > 10:
        t_final_ps = min(rem * 0.5, 40)
        a_fp, _ = peak_shave_local(best_seq, t_final_ps)
        best_seq, best_score = check_and_update(a_fp, f"FinalPS_n{len(best_seq)}", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = hann_window_init(512)
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
