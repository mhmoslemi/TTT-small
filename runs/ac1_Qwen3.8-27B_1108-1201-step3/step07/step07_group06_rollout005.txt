"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated soft-max
gradient descent in log-space with adaptive temperature, active-set L-BFGS
refinement, multi-length exploration, and simulated annealing restarts.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def nesterov_logspace(a0, time_limit, T_init=0.05, T_final=0.001,
                      lr=0.01, mu=0.9, schedule='exp', verbose=False):
    """
    Nesterov momentum in log-space with soft-max temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    z = z_from_a(a)
    z_prev = z.copy()
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature
        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        else:  # 'plateau'
            if progress < 0.2:
                T = T_init
            elif progress < 0.5:
                T = T_init * 0.5
            else:
                p2 = (progress - 0.5) / 0.5
                T = T_init * 0.5 * (T_final / (T_init * 0.5)) ** p2
        
        # Nesterov lookahead
        z_nest = z + mu * (z - z_prev)
        a_curr = softmax_from_z(z_nest)
        
        # Soft-max weights
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        # Gradient in a-space: grad_a[i] = 2n * sum_k w[k] * a[k-i] (valid k-i)
        # This is 2n * cross-correlation of w and a
        # Using FFT: the cross-correlation can be computed as:
        # corr[i] = sum_k w[k] * a[k-i] = (w * a_reversed)[i + n - 1]
        # Or more simply: it's the convolution of w with a, evaluated at specific indices.
        
        # Actually, let's use the formula:
        # grad_a[i] = 2n * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # Let j = k-i, so k = j+i:
        # grad_a[i] = 2n * sum_{j: 0<=j<n, 0<=j+i<2n-1} w[j+i] * a[j]
        # = 2n * (w_shifted * a)[i] where w_shifted is w padded
        # This is the cross-correlation of w and a, which equals:
        # = 2n * ifft(conj(FFT(a)) * FFT(w)) ... but w has length 2n-1 and a has length n
        
        # Simpler approach: compute via direct FFT
        # The gradient is: grad_a = 2n * (w * a) where * denotes cross-correlation
        # In FFT terms: corr = ifft(FFT(w_padded) * conj(FFT(a_padded)))
        
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        # Cross-correlation: corr[i] = sum_j w[j+i] * a[j]
        # This equals the linear convolution of w with flipped a, shifted
        # Alternatively: corr = irfft(conj(W) * A) gives sum_j w[j] * a[j+i] ... 
        # Let me just use the relationship:
        # sum_k w[k] * a[k-i] for k from i to i+2n-2 (but a is only defined for 0..n-1)
        # = sum_{j=0}^{n-1} w[j+i] * a[j] (where w[j+i] = 0 if j+i >= 2n-1)
        # This is the cross-correlation (w ⊛ a)[i]
        # In FFT: cross_corr = ifft(conj(FFT(a)) * FFT(w))
        # but we need to be careful about lengths
        
        # Actually the simplest: the gradient of (a*a)_k w.r.t. a is 2*a[k-i] (reversed)
        # So grad_a[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
        # = 2 * (w * a)[i] where * is cross-correlation
        # Using FFT with appropriate padding:
        
        # Let's use: fft(w, fft_size) * conj(fft(a, fft_size)) -> ifft gives cross-correlation
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr
        
        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Nesterov update
        z_prev = z.copy()
        z = z - lr * grad_z
        
        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        if no_improve > 5000:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            z_prev = z.copy()
            no_improve = 0
            mu = max(0.5, mu * 0.9)
        
        if verbose and t % 5000 == 0:
            print(f"      Nesterov t={t} T={T:.5f} mu={mu:.3f} score={best_score:.6f}")
    
    return best_a, best_score


def active_set_lbfgs(a0, time_limit, max_lags=10, maxiter=2000):
    """
    L-BFGS optimization where the objective is the max over the top `max_lags`
    lags of the auto-convolution (approximating the full max with a smooth max of top lags).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    
    # Find top lags
    b = fast_convolve(a)
    top_lags = np.argsort(b)[::-1][:max_lags]
    
    def objective(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        aa = softmax_from_z(z)
        bb = fast_convolve(aa)
        # Smooth max over top lags
        vals = bb[top_lags]
        M = vals.max()
        T_smooth = 0.01
        return 2.0 * n * T_smooth * (M / T_smooth + np.log(np.exp(vals / T_smooth - M / T_smooth).sum()))
    
    def gradient(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        aa = softmax_from_z(z)
        bb = fast_convolve(aa)
        vals = bb[top_lags]
        M = vals.max()
        T_smooth = 0.01
        exp_v = np.exp(vals / T_smooth - M / T_smooth)
        w_top = exp_v / exp_v.sum()
        
        # Gradient of smooth max of top lags
        W_full = np.zeros(2 * n - 1)
        W_full[top_lags] = w_top
        W = rfft(W_full, fft_size)
        A = rfft(aa, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr
        S = np.dot(grad_a, aa)
        return aa * (grad_a - S)
    
    z0 = z_from_a(a)
    try:
        res = minimize(
            fun=objective, x0=z0, jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a.copy()
    
    return a_out


def peak_equalize(a0, time_limit, n_trials=5000):
    """
    Peak equalization: find the top peaks and try to equalize them by
    adjusting the elements that contribute most to each peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (exclude lag 0 and lag 2n-2 as they are inherently small)
        b_masked = b.copy()
        b_masked[0] = -np.inf
        b_masked[-1] = -np.inf
        peak_k = int(np.argmax(b_masked))
        
        # Try to reduce this peak
        # Elements contributing to peak_k: a[i] * a[peak_k - i] for valid i
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_i = idx[np.argmax(contribs)]
        
        improved = False
        for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
            a_trial = a.copy()
            a_trial[top_i] *= factor
            # Redistribute
            diff = a[top_i] - a_trial[top_i]
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a_trial[mask]
            if others.sum() > 1e-15:
                a_trial[mask] += diff * others / others.sum()
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a_trial.copy()
                improved = True
                break
        
        if not improved:
            # Try increasing a non-contributing element
            non_contrib = np.setdiff1d(np.arange(n), idx)
            if len(non_contrib) > 0:
                rand_i = rng.choice(non_contrib)
                for factor in [1.1, 1.2, 1.5]:
                    a_trial = a.copy()
                    a_trial[rand_i] *= factor
                    s = a_trial.sum()
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
        
        if not improved and trial % 200 == 199:
            # Random perturbation
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Triangular
    tri = np.linspace(0, 1, n)
    tri += np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    tri = tri / tri.sum()
    inits.append(('triangular', tri))
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x**2 / 3)
    gau = np.maximum(gau, 1e-8)
    gau = gau / gau.sum()
    inits.append(('gaussian', gau))
    
    # Bimodal (peaks at 1/3 and 2/3)
    x = np.linspace(0, 1, n)
    bi = np.exp(-((x - 1/3)**2) / 0.02) + np.exp(-((x - 2/3)**2) / 0.02)
    bi = np.maximum(bi, 1e-8)
    bi = bi / bi.sum()
    inits.append(('bimodal', bi))
    
    # Cosine window
    cos_w = np.cos(np.pi * x)**2
    cos_w = np.maximum(cos_w, 0.001)
    cos_w = cos_w / cos_w.sum()
    inits.append(('cosine', cos_w))
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 0.001)
    rc = rc / rc.sum()
    inits.append(('raised_cos', rc))
    
    # Random smooth
    freq = rng.integers(1, 5)
    phase = rng.random() * 2 * np.pi
    smooth = 1 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    smooth = np.maximum(smooth, 0.01)
    smooth = smooth / smooth.sum()
    inits.append(('random_smooth', smooth))
    
    # Exp decay from center
    center = n // 2
    dist = np.abs(np.arange(n) - center)
    expo = np.exp(-dist / (n / 4))
    expo = np.maximum(expo, 1e-8)
    expo = expo / expo.sum()
    inits.append(('center_exp', expo))
    
    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    inits.append(('random', rand))
    
    # Flat top (highpass-like in freq domain)
    ft = np.ones(n)
    ft[:n//8] *= 0.5
    ft[-n//8:] *= 0.5
    ft = ft / ft.sum()
    inits.append(('flat_top', ft))
    
    return inits


def evaluate_and_track(a, label, best_seq, best_score):
