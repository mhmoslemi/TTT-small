def remez_equality(n, active_set, a_init, max_iters=100):
    """
    Solve the system q_{k_1}(a) = q_{k_2}(a) = ... = q_{k_{n-1}}(a), sum(a) = 1
    using Newton's method.
    
    active_set: list of n-1 indices k_1, ..., k_{n-1}
    a_init: initial guess for a
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    k_ref = active_set[0]
    k_others = active_set[1:]
    
    for it in range(max_iters):
        # Compute residuals
        conv = fast_convolve(a)
        residual = np.zeros(n)
        for idx, k in enumerate(k_others):
            residual[idx] = conv[k] - conv[k_ref]
        residual[-1] = a.sum() - 1.0
        
        if np.max(np.abs(residual)) < 1e-14:
            break
        
        # Compute Jacobian
        # J[i][j] = 2*a[k_others[i] - j] - 2*a[k_ref - j] for i=0,...,n-3
        # J[n-2][j] = 1
        J = np.zeros((n, n))
        for i, k in enumerate(k_others):
            # Gradient of q_k: 2*a[k-j] for j=0,...,n-1 (with a out of range = 0)
            grad_k = np.zeros(n)
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    grad_k[j] = 2.0 * a[ki]
            grad_ref = np.zeros(n)
            for j in range(n):
                ki = k_ref - j
                if 0 <= ki < n:
                    grad_ref[j] = 2.0 * a[ki]
            J[i] = grad_k - grad_ref
        J[-1] = 1.0
        
        # Solve J * da = -residual
        try:
            da = np.linalg.solve(J, -residual)
        except np.linalg.LinAlgError:
            da = np.linalg.lstsq(J, -residual, rcond=None)[0]
        
        # Line search
        alpha = 1.0
        for ls in range(30):
            a_new = a + alpha * da
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:

            for idx, k in enumerate(k_others):

                a = a_new
                break
            alpha *= 0.5

import numpy as np
from scipy.fft import rfft, irfft
import time

# ==================== Core Utilities ====================

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver via cvxpy."""
    import cvxpy as cp
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

# ==================== LP Best Response ====================

def build_conv_matrix_rows(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None, warm_start=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix
    T = build_conv_matrix_rows(a_fixed)
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solve_kwargs = {}
    if warm_start is not None:
        try:
            b_var.value = warm_start
        except Exception:
            pass
    
    for attempt in range(3):
        try:
            prob.solve(solver=solver, **solve_kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try fallback solvers
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    # Get dual variables if available
    duals = None
    try:
        if constraints[0].dual_value is not None:
            duals = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, float(M_var.value), duals

def solve_lp_weighted(a_fixed, weights, solver=None):
    """Minimize max_k weights[k] * (T@a)[k] subject to sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed)
    # Scale the constraints
    scaled_T = T * weights[:, np.newaxis]
    constraints = [
        scaled_T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

# ==================== Recursive Ideal Initialization ====================

def recursive_ideal_init(n):
    """
    Construct a sequence that (approximately) equalizes the auto-convolution.
    Uses the recurrence: conv[k] = 1/(2n-1) for all k.
    a[0] = 1/sqrt(2n-1)
    a[k] = (target - 2*sum_{j=1}^{k-1} a[j]*a[k-j] - (a[k/2]^2 if k even)) / (2*a[0])
    """
    target = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(target)
    
    for k in range(1, n):
        # conv[k] = sum_{j=0}^{k} a[j] * a[k-j] = target
        # = 2*a[0]*a[k] + 2*sum_{j=1}^{k-1} a[j]*a[k-j] (with middle term adjusted)
        # For k < n, all indices are valid (0 <= k-j < n since j >= 0 and k < n)
        s = 0.0
        for j in range(1, k):
            s += a[j] * a[k - j]
        if k % 2 == 0:
            s += a[k // 2] ** 2  # middle term counted once in the sum above? No.
            # Actually: conv[k] = a[0]*a[k] + a[1]*a[k-1] + ... + a[k]*a[0]
            # = 2*(a[0]*a[k] + a[1]*a[k-1] + ... + a[k/2-1]*a[k/2+1]) + a[k/2]^2
            # So: 2*a[0]*a[k] + 2*sum_{j=1}^{k/2-1} a[j]*a[k-j] + a[k/2]^2 = target
            # Wait, I already computed s = sum_{j=1}^{k-1} a[j]*a[k-j]
            # This includes a[k/2]^2 if k is even.
            # conv[k] = 2*a[0]*a[k] + 2*sum_{j=1}^{k-1} a[j]*a[k-j] - a[k/2]^2 (if k even)
            # No wait: conv[k] = sum_{j=0}^{k} a[j]*a[k-j]
            # = a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j] + a[k]*a[0]
            # = 2*a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j]
            # So: a[k] = (target - sum_{j=1}^{k-1} a[j]*a[k-j]) / (2*a[0])
            pass
        # General formula: a[k] = (target - sum_{j=1}^{k-1} a[j]*a[k-j]) / (2*a[0])
        # Wait, let me redo:
        # conv[k] = sum_{j=0}^{k} a[j]*a[k-j] = 2*a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j]
        # So: a[k] = (target - sum_{j=1}^{k-1} a[j]*a[k-j]) / (2*a[0])
        s = 0.0
        for j in range(1, k):
            s += a[j] * a[k - j]
        a[k] = (target - s) / (2.0 * a[0])
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def recursive_ideal_init_v2(n):
    """
    Alternative recursive construction: enforce equal convolution values.
    More numerically stable version.
    """
    target = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(target)
    
    for k in range(1, min(n, 200)):  # Only compute first 200 to handle numerical issues
        s = np.dot(a[1:k], a[k-1:0:-1]) if k > 1 else 0.0
        val = (target - s) / (2.0 * a[0])
        if val > 0:
            a[k] = val
        else:
            # Pad the rest with a smooth taper
            remaining = target * (n - k) / max(a[:k].sum(), 1e-15)
            a[k:] = remaining / (n - k) if n > k else 0
            break
    else:
        # If we got through all n, good
        pass
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

# ==================== Fictitious Play (Inertial) ====================

def inertial_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    Inertial fictitious play: alternating LP best responses with running averages.
    Uses the fact that the running average converges to the equilibrium.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = compute_score(a)
    iters = 0
    warm_start_a = a.copy()
    warm_start_b = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Best response for a (given b)
        b_new, M_b, _ = solve_lp_best_response(b, solver, warm_start=warm_start_b)
        if b_new is not None:
            b = b_new
            warm_start_b = b.copy()
        
        # Best response for b (given a)
        a_new, M_a, _ = solve_lp_best_response(a, solver, warm_start=warm_start_a)
        if a_new is not None:
            a = a_new
            warm_start_a = a.copy()
        
        # Update running averages (inertial fictitious play)
        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)
        
        # Evaluate candidates
        candidates = [a, b, a_avg, b_avg, 0.5*(a+b), 0.5*(a_avg+b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = compute_score(cand)
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    IFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f} M_a={M_a:.6f} M_b={M_b:.6f}")

    return best_a, best_score, iters

# ==================== Subgradient Methods ====================

def compute_softmax_weights(conv_vals, temperature):
    """Compute softmax weights over convolution values."""
    shifted = conv_vals - conv_vals.max()
    weights = np.exp(shifted / temperature)
    wsum = weights.sum()
    if wsum < 1e-30:
        w = np.zeros(len(conv_vals))
        w[np.argmax(conv_vals)] = 1.0
        return w
    return weights / wsum

def compute_subgradient(a, weights):
    """
    Compute the subgradient of max_k q_k(a) using given weights on positions.
    grad[i] = 2 * sum_k weights[k] * a[k-i] (with a_j = 0 out of range)
    This is 2 * conv(a, w_reversed) appropriately indexed.
    """
    n = len(a)
    # The gradient of q_k at a is: grad_k[i] = 2*a[k-i] for valid k-i
    # Weighted sum: grad[i] = 2 * sum_k w_k * a[k-i]
    # This is 2 * (conv(a, w))[i] where w is reversed... let me think.
    # conv(a, w)[i] = sum_j a[j] * w[i-j]
    # We want: sum_k w[k] * a[k-i] for each i
    # Let m = k - i, then k = m + i: sum_m w[m+i] * a[m] = conv(a, w_shifted)[i]
    # Actually: sum_k w[k]*a[k-i] where k ranges over valid indices
    # = sum_m a[m] * w[m+i] where m+i is valid
    # This is the cross-correlation of a with w at lag i.
    # = (a * w_flipped)[n-1+i] ... this is getting confusing.
    # Let me just compute it directly using the convolution.
    # grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # Let j = k - i: grad[i] = 2 * sum_j w[j+i] * a[j]
    # = 2 * (a conv w)[i] where conv is defined as (a conv w)[i] = sum_j a[j]*w[j+i]
    # This is the "correlation" not "convolution". 
    # a conv w (full) has length 2n-1, and (a conv w)[i] = sum_j a[j]*w[i-j]
    # What we want is sum_j a[j]*w[j+i] = (a conv w_rev)[2n-2-i] ... 
    # Simplest: just compute via FFT.
    # grad = 2 * ifft(fft(a) * conj(fft(w))) ... no that's cross-correlation.
    # Let me just do it directly for clarity (n is at most ~2000).
    
    # Use the fast approach: grad[i] = 2 * sum_j a[j] * w[i+j] (with w out of range = 0)
    # This is the "anti-convolution" or "cross-correlation"
    # corr[i] = sum_j a[j] * w[i+j]
    # = (a * flip(w))[n-1-i] ... 
    
    # Simplest correct approach using FFT:
    # We want g[i] = sum_j a[j] * w[i+j] for i = 0, ..., n-1
    # Pad w to length 2n-1 (it already is)
    # g = ifft(fft(a padded) * fft(w reversed padded)) ... 
    
    # Actually the cleanest: g[i] = sum_j a[j] w[i+j]
    # Reverse a: ar[j] = a[n-1-j]
    # then g[i] = sum_j ar[n-1-j] * w[i+j]
    # Let k = n-1-j: g[i] = sum_k ar[k] * w[i+n-1-k]
    # = (ar * w)[i + n - 1] (standard convolution)
    
    ar = a[::-1]
    # conv(ar, w) has length 2n-1 (since both have length n... wait, w has length 2n-1)
    # conv(ar, w)[m] = sum_k ar[k] * w[m-k]
    # We want m = i + n - 1: sum_k ar[k] * w[i+n-1-k]
    # = sum_k a[n-1-k] * w[i+n-1-k]
    # Let j = n-1-k: sum_j a[j] * w[i+j]  ✓
    
    size = 1 << (n + len(w) - 1 - 1).bit_length()
    Ar = rfft(ar, size)
    Wr = rfft(w, size)
    conv_ar_w = irfft(Ar * Wr, size)
    g = conv_ar_w[n-1:n-1+n]  # positions i+n-1 for i=0,...,n-1
    
    return 2.0 * g

def nesterov_subgrad(a0, time_limit, top_k=30, lr_init=0.005, temp_init=0.005, verbose=False):
    """
    Nesterov-accelerated projected subgradient with adaptive softmax temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        # Compute softmax weights on the top peaks
        conv_y = fast_convolve(y)
        # Use top-k peaks for the subgradient
        k_actual = min(top_k, len(conv_y))
        peaks_idx = np.argpartition(conv_y, -k_actual)[-k_actual:]
        
        # Compute softmax over all positions with focus on peaks
        # Use a temperature that adapts
        max_val = conv_y.max()
        min_val = conv_y[peaks_idx].min()
        temp = max(0.001, (max_val - min_val) * 0.1)
        
        w = compute_softmax_weights(conv_y, temp)
        
        # Compute subgradient
        sg = compute_subgradient(y, w)
        
        # Nesterov update
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        score = compute_score(y_projected)
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
            temp *= 0.5  # Sharpen the softmax focus
        
        if no_improve > 400:
            # Random restart
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * 0.05
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            temp = temp_init
            no_improve = 0

        lr *= 0.9998
        temp *= 0.9995  # Gradually sharpen

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nest it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f} T={temp:.6f}")

    return best_a, best_score

# ==================== Conjugate Gradient on Smoothed Objective ====================

def smoothed_objective_and_grad(a, temperature):
    """
    Compute the smoothed objective F(a) = n * log(sum_k exp(q_k(a)/T)) / (2*T * sum(a)^2)
    and its gradient. (Simplified: just the numerator part for the subgradient.)
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax-smoothed max
    shifted = conv - conv.max()
    log_sum_exp = conv.max() + np.log(np.sum(np.exp(shifted / temperature)) / len(conv)) * temperature
    # Actually for the subgradient we just need the weights
    w = np.exp(shifted / temperature)
    wsum = w.sum()
    w = w / wsum
    grad = compute_subgradient(a, w)
    return log_sum_exp, grad, w

def fft_conjugate_gradient(a0, time_limit, temp=0.01, verbose=False):
    """
    Conjugate gradient method on the smoothed objective, projected to simplex.
    Uses FFT-based Hessian-vector products.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at current point
        conv = fast_convolve(a)
        w = compute_softmax_weights(conv, temp)
        grad = compute_subgradient(a, w)
        
        # Project gradient to simplex tangent space
        grad_proj = grad - np.mean(grad)  # Project to sum=0 hyperplane
        
        # Take a step
        alpha = 0.01 / (1 + it * 0.01)
        a_new = simplex_project(a - alpha * grad_proj)
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            alpha *= 0.5
            a_new = simplex_project(a - alpha * grad_proj)
            score = compute_score(a_new)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new
        
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    CG it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

# ==================== Coordinate Descent ====================

def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """
    Coordinate descent: try scaling each coordinate to reduce the peak.
    Uses finer steps and focuses on peak-contributing coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find peak-contributing coordinates
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Determine which coordinates contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        peak_coords = set(range(js, je + 1))
        
        # Process peak coordinates first, then random
        if peak_coords:
            peak_list = list(peak_coords)
            rng.shuffle(peak_list)
            other_list = [i for i in range(n) if i not in peak_coords]
            rng.shuffle(other_list)
            order = peak_list + other_list
        else:
            order = rng.permutation(n).tolist()
        
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Try a range of scaling factors
            factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3, 1.5]
            best_factor = 1.0
            best_factor_score = best_score
            
            for factor in factors:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                score = compute_score(a)
                if score < best_factor_score:
                    best_factor_score = score
                    best_factor = factor
                a[i] = old_val
                a = a / a.sum()
            
            if best_factor != 1.0 and best_factor_score < best_score - 1e-13:
                a[i] = old_val * best_factor
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_factor_score
                best_a = a.copy()
                improved = True

        if not improved and sweep > 10:
            # Try very fine steps on random coordinates
            any_improve = False
            for i in rng.choice(n, size=min(n, 200), replace=False):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-15:
                    continue
                for factor in [0.97, 0.98, 0.99, 1.01, 1.02, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        continue
                    a = a / s
                    score = compute_score(a)
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        any_improve = True
                        break
                    else:
                        a[i] = old_val
                        a = a / a.sum()
                if any_improve:
                    break
            if not any_improve and sweep > 30:
                break

    return best_a, best_score

def pair_coord_descent(a0, time_limit):
    """
    2D coordinate descent: adjust pairs of coordinates simultaneously.
    Focus on pairs that jointly contribute to the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Find top contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        top_pairs = np.argsort(contribs)[::-1][:8]
        
        improved = False
        for pi in top_pairs:
            i = int(idx[pi])
            j = peak_k - i
            if i == j or i not in range(n) or j not in range(n):
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Try scaling both
            for f1 in [0.8, 0.7, 0.6]:
                for f2 in [0.8, 0.9, 1.0, 1.1]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = compute_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved and step > 50:
            break

    return best_a, best_score

# ==================== Damped LP Refinement ====================

def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP refinement with decreasing step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, _, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

# ==================== Initialization Methods ====================

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
