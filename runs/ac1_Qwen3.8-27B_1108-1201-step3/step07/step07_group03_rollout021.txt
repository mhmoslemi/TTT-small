Given $a$ with $\sum a = 1$, define:
$$f_\lambda(a) = \frac{1}{\lambda} \log \sum_{k=0}^{2n-2} \exp(\lambda (a*a)[k])$$

As $\lambda \to \infty$, $f_\lambda(a) \to \max_k (a*a)[k]$.

The gradient:
$$\nabla f_\lambda(a)_j = \frac{2}{\lambda} \sum_k p_k(a) \cdot a_{k-j} \cdot \mathbf{1}[0 \leq k-j \leq n-1]$$

where $p_k(a) = \exp(\lambda (a*a)[k]) / \sum_j \exp(\lambda (a*a)[j])$.

(The factor of 2 comes from the symmetry of the autocorrelation, and we ignore the $\delta$ term for simplicity.)

We can compute this efficiently:
- Compute $b = a * a$ via FFT
- Compute $p_k = \text{softmax}(\lambda b)$
- Then $\nabla f_j \propto \sum_k p_k a_{k-j}$ which is $(p * a)[j]$ (cross-correlation!)

This is very efficient to compute!

**Algorithm:**
1. Start with a good initial $a$ (from parent or LP)
2. For $\lambda$ = small to large:
   - Compute $b = a * a$
   - Compute $p = \text{softmax}(\lambda b)$
   - Compute gradient $g = p * a$ (via FFT)
   - Project gradient descent step onto the simplex
3. Continue until $\lambda$ is large enough that $f_\lambda \approx \max b$

This is essentially a "soft-to-hard" annealing approach.

Let me also think about **mirror descent** on the simplex, which might be better than Euclidean projection:

Mirror descent with entropy regularization:
$$a_{new} = \arg\min_{a \geq 0, \sum a = 1} \langle g, a \rangle + \frac{1}{\eta} D_{KL}(a \| a_{current})$$

The closed form is: $a_{new}[j] = a_{current}[j] \exp(-\eta g_j) / \sum_k a_{current}[k] \exp(-\eta g_k)$

This is the "exponential weight" update, which is great for simplex constraints.

import numpy as np
from scipy.fft import rfft, irfft
import time

# Global state
_lp_solver_name = None
_best_fft_size = {}

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_fft_size(n):
    """Get appropriate FFT size for n."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def get_gradient_fft(a, p, fft_size=None):
    """Compute g[j] = sum_i a[i] * p[i+j] for j=0..n-1 via FFT."""
    n = len(a)
    if fft_size is None:
        # Need size >= 3n-2
        needed = 3 * n - 2
        fft_size = 1 << (needed - 1).bit_length()
    p_rev = p[::-1]
    A = rfft(a, fft_size)
    P = rfft(p_rev, fft_size)
    conv = irfft(A * P, fft_size)
    # g[j] = conv[2n-2-j] for j=0..n-1
    g = conv[n-1 : 2*n-1][::-1]
    return g

def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLPK, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, active_constraints=None, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Optionally use only active_constraints (indices of tight rows)."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if active_constraints is not None and len(active_constraints) < 2 * n - 1:
        T = np.zeros((len(active_constraints), n))
        for row_idx, k in enumerate(active_constraints):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            T[row_idx, j] = a_fixed[k - j]
    else:
        T = build_conv_matrix(a_fixed)
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS, cp.CBC, cp.GUROBI, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """LP fictitious play with adaptive active constraints."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    
    # Track active constraints
    active_set = None
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Solve LP for b given a
        b_new, _ = solve_lp_best_response(a, active_set, solver)
        if b_new is not None:
            b = b_new
        
        # Update active set based on current b
        ab = fast_convolve(b)
        threshold = np.percentile(ab, 95)  # top 5% are "active"
        active_set = np.where(ab >= threshold)[0].tolist()
        
        # Solve LP for a given b
        a_new, _ = solve_lp_best_response(b, active_set, solver)
        if a_new is not None:
            a = a_new
        
        # Polyak averaging
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        # Check candidates
        candidates = [a, b, 0.5 * (a + b), a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()
        
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, iters

def logsumexp_mirror_descent(a0, time_limit, verbose=False):
    """
    Log-sum-exp smoothed optimization with mirror descent on the simplex.
    Anneals temperature lambda from low to high.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    # FFT size for gradient computation
    needed = 3 * n - 2
    fft_size = 1 << (needed - 1).bit_length()
    
    # Annealing parameters
    lambda_min = 3.0
    lambda_max = 200.0
    total_anneal_steps = 300  # steps to go from lambda_min to lambda_max
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Current lambda
        if it <= total_anneal_steps:
            frac = it / total_anneal_steps
            lam = lambda_min * (lambda_max / lambda_min) ** frac
        else:
            lam = lambda_max
        
        # Compute b = a * a
        b = fast_convolve(a)
        b_max = b.max()
        
        # Softmax weights
        log_p = lam * (b - b_max)
        log_p -= log_p.max()  # numerical stability (should be mostly 0)
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(b))
            p[np.argmax(b)] = 1.0
        else:
            p /= p_sum
        
        # Gradient: g[j] = sum_i a[i] * p[i+j]
        g = get_gradient_fft(a, p, fft_size)
        
        # Mirror descent step (exponential weights)
        # a_new[j] proportional to a[j] * exp(-eta * g[j])
        g_max = g.max()
        if g_max < 1e-15:
            eta = 0.01
        else:
            eta = 0.5 / g_max
        
        # Cap eta to avoid catastrophic steps
        eta = min(eta, 1.0)
        
        log_ratio = -eta * (g - g.mean())  # center for stability
        log_ratio -= log_ratio.max()
        ratio = np.exp(log_ratio)
        
        a_new = a * ratio
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        # Evaluate
        b_new = fast_convolve(a_new)
        score = 2.0 * n * b_new.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} ({elapsed:.0f}s): lam={lam:.1f} best={best_score:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit, verbose=False):
    """Target the highest peaks by reducing contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    a = a / s
    
    best_a = a.copy()

    t0 = time.monotonic()
    
    for step in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peak_k = int(peaks[0])
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
        
        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in top_contrib:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx[ci])
            j = peak_k - i
            
            # Try reducing coordinate i
            for factor in [0.9, 0.8, 0.7]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            # Try reducing both i and j (if different)
            if not improved and i != j and a[j] >= 1e-12:
                for f1 in [0.85, 0.75]:
                    for f2 in [0.85, 0.75]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score

                            improved = True
                            break
                    if improved:
                        break
            
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
