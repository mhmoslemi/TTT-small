def compute_subgrad(a, top_k=30):
    """Compute smoothed subgradient of max(conv(a,a)) at point a."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    temp = 0.005 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * a[k - j]
    return sg

            no_improve = 0

        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.008
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 1.5)
            no_improve = 0

        lr *= 0.9998

        if verbose and it % 100 == 0:
            print(f"    Nesterov it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score

def adapt

ive_coord_descent(a0, time_limit, verbose=False):
    """
    Powerful coordinate descent with adaptive search:
    - Coarse grid scan per coordinate
    - Zoom-in refinement around best point
    - Pairwise updates for nearby coordinates
    - Adaptive stopping
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-14:
                # Zero coordinate: try small activations
                candidates = [0.5 / n, 1.0 / n, 2.0 / n, 3.0 / n]
            else:
                # Coarse grid around current value
                scale = max(old_val, 1.0 / n)
                candidates = [0.0, 0.4 * scale, 0.7 * scale, 0.85 * scale,
                              scale, 1.15 * scale, 1.3 * scale, 1.6 * scale, 2.0 * scale]

            best_t = old_val
            best_local = best_score

            for t in candidates:
                a[i] = t
                a = np.maximum(a, 0)
                s_tmp = a.sum()
                if s_tmp < 1e-15:
                    a[i] = old_val
                    s_tmp = a.sum()
                    continue
                a = a / s_tmp
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_t = t

            # Zoom-in: refine around best_t
            if best_t > 1e-12 and best_local < best_score:
                for _ in range(3):
                    step = max(0.1 * best_t, 1e-4 / n)
                    best_t_zoom = best_t
                    best_local_zoom = best_local
                    for frac in [-1.0, -0.5, -0.25, 0.25, 0.5, 1.0]:
                        t_fine = best_t + frac * step
                        if t_fine < 0:
                            continue
                        a[i] = t_fine
                        a = np.maximum(a, 0)
                        s_tmp = a.sum()
                        if s_tmp < 1e-15:
                            a[i] = best_t
                            continue
                        a = a / s_tmp
                        sc = compute_score(a)
                        if sc < best_local_zoom:
                            best_local_zoom = sc
                            best_t_zoom = t_fine
                    if best_t_zoom == best_t:
                        break
                    best_t = best_t_zoom
                    best_local = best_local_zoom

            # Apply best
            a[i] = best_t
            a = np.maximum(a, 0)
            s_tmp = a.sum()
            if s_tmp > 1e-15:
                a = a / s_tmp

            if best_local < best_score - 1e-15:
                best_score = best_local
                best_a = a.copy()
                improved = True

        # Pairwise refinement: try adjusting pairs of nearby coordinates
        if sweep % 3 == 0:
            for _ in range(min(n // 4, 50)):
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 == i2 or a[i1] < 1e-14 or a[i2] < 1e-14:
                    continue
                v1, v2 = a[i1], a[i2]
                for f1 in [0.7, 0.85, 1.0, 1.15, 1.3]:
                    for f2 in [0.7, 0.85, 1.0, 1.15, 1.3]:
                        a[i1] = v1 * f1
                        a[i2] = v2 * f2
                        a = np.maximum(a, 0)
                        s_tmp = a.sum()
                        if s_tmp < 1e-15:
                            a[i1] = v1
                            a[i2] = v2
                            continue
                        a = a / s_tmp
                        sc = compute_score(a)
                        if sc < best_score - 1e-15:
                            best_score = sc
                            best_a = a.copy()
                            improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 8:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweep % 20 == 0:
            print(f"  CD sweep={sweep} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def simulated_annealing(a0, time_limit, verbose=False):
    """Simulated annealing with adaptive temperature and perturbation size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    T = 0.002
    it = 0
    sc_current = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Perturb a random subset
        n_change = rng.integers(1, max(2, n // 80))
        idx = rng.choice(n, size=n_change, replace=False)
        a_new = a.copy()
        for j in idx:
            factor = rng.lognormal(0, 0.4)
            a_new[j] = max(0, a_new[j] * factor)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        sc_new = compute_score(a_new)
        delta = sc_new - sc_current
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-8)):
            a = a_new
            sc_current = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
                T = max(0.0005, T * 0.95)
        else:
            T = max(0.0005, T * 0.998)

        # Periodic restart from best
        if it % 500 == 0:
            a = best_a.copy()
            sc_current = best_score
            T = 0.002

        if verbose and it % 200 == 0:
            print(f"    SA it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} T={T:.6f}")

    return best_a, best_score

def dual_subgradient(a0, time_limit, verbose=False):
    """
    Mirror-descent on the dual: maintain a distribution q over convolution
    positions, update a to minimize weighted sum, update q to emphasize peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    q = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute weighted gradient: grad_a = 2 * Q @ a where Q[i,j] = q[i+j]
        # (Q @ a)[i] = sum_j q[i+j] * a[j]
        # This can be computed via convolution with a reversed/padded q
        conv_vals = fast_convolve(a)
        # Weighted sum
        weighted = np.dot(q, conv_vals)

        # Gradient of weighted objective w.r.t. a
        # d/da[i] sum_k q[k] * conv(a,a)[k] = 2 * sum_k q[k] * a[k-i] (for valid k-i)
        # = 2 * (q_restricted * a)[i] where the convolution is one-sided
        # More precisely: grad[i] = 2 * sum_{j=0}^{n-1} q[i+j] * a[j] (for i+j < 2n-1)
        # We can compute this via: pad a to length 2n-1, convolve with q, take first n
        a_pad = np.zeros(2 * n - 1)
        a_pad[:n] = a
        # grad[i] = 2 * sum_j a[j] * q[i+j]
        # = 2 * (correlation of a_pad with q at shift i)... let's just do it directly
        # Actually: grad[i] = 2 * sum_{j} a[j] * q[i+j]
        # Let's compute via FFT: define r[k] = q[k] for k=0..2n-2
        # Then grad[i] = 2 * sum_j a[j] * r[i+j]
        # This is not a standard convolution. Let's compute it as:
        # grad = 2 * (a * r)[0:n] reversed... hmm
        # Actually: sum_j a[j] * q[i+j] for i=0..n-1
        # Let's define b[i] = sum_j a[j] * q[i+j]
        # b[i] = sum_{j=0}^{n-1} a[j] * q[i+j]
        # This is like a "cross-correlation" but with a one-sided window on q
        # We can compute: pad a to [a, 0, 0, ..., 0] (length 2n-1)
        # Then b = [a[0]*q[i] + a[1]*q[i+1] + ... for i=0..n-1]
        # = The first n elements of the "anti-convolution" of a (padded) with q
        
        # Simpler: just compute directly for the gradient (O(n^2) but n is at most ~2000)
        if n <= 2000:
            grad = np.zeros(n)
            for i in range(n):
                j_start = 0
                j_end = min(n - 1, 2 * n - 2 - i)
                grad[i] = 2.0 * np.dot(a[j_start:j_end + 1], q[i:i + j_end - j_start + 1])
        else:
            # Use a simpler subgradient based on peaks
            grad = compute_subgrad(a, top_k=30)

        # Gradient step with projection
        y_new = a - lr * grad
        a_projected = simplex_project(y_new)

        sc = compute_score(a_projected)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_projected

        # Update q: emphasize peaks
        conv_updated = fast_convolve(a)
        q_p = conv_updated ** 2  # Square to emphasize peaks more
        q_sum = q_p.sum()
        if q_sum > 1e-30:
            q_new = q_p / q_sum
            # Smooth: blend with old q
            q = 0.7 * q + 0.3 * q_new
            q = q / q.sum()

        # Adaptive learning rate
        if no_improve > 80:
            lr *= 0.8
            no_improve = 0
        lr = max(lr, 1e-5)

        if verbose and it % 50 == 0:
            print(f"    DualSG it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")
    return best_seq, best_score
