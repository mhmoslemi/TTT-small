"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with softmax subgradient as the main driver,
combined with LP refinement, coordinate descent, and multi-length search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w, n_top=None):
    """Compute subgradient of sum_k w[k] * (f*f)[k]."""
    n = len(f)
    g = np.zeros(n)
    if n_top is not None and n_top < len(w):
        top_idx = np.argpartition(w, -n_top)[-n_top:]
    else:
        top_idx = np.where(w > 1e-15)[0]
    
    for k in top_idx:
        wk = 2.0 * w[k]
        if wk < 1e-15:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        u_lo = k - j_hi
        u_hi = k - j_lo
        g[j_lo:j_hi + 1] += wk * f[u_lo:u_hi + 1][::-1]
    return g


def mirror_descent(f, eta, tau, n_top=None):
    """One mirror descent step with softmax subgradient."""
    n = len(f)
    b = fast_convolve(f)
    
    # Softmax weights
    shifted = (b - b.max()) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Subgradient
    g = compute_subgradient(f, w, n_top)
    
    # Multiplicative update
    f_new = f * np.exp(-eta * g)
    s = f_new.sum()
    if s < 1e-30:
        f_new = np.ones(n) / n
    else:
        f_new /= s
    return f_new, b.max()


def run_mirror_descent(f_init, time_limit, eta_init=0.01, tau_init=0.001, label="md"):
    """Mirror descent with adaptive learning rate and temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = eta_init
    tau = tau_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f, cur_obj = mirror_descent(f, eta, tau)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.5)
            tau = max(tau * 0.995, 1e-5)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                eta *= 0.9
                tau = min(tau * 1.1, 1.0)
            if no_improve > 200:
                # Restart with perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                eta = eta_init
                tau = tau_init
                no_improve = 0
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} obj={best_obj:.10f} score={score:.6f} eta={eta:.4f} tau={tau:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def lp_solve(f, top_k, eps=1e-8):
    """Solve the LP for direction finding."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

        if S_g <= 1.0 + 1e-5:

def coordinate_polish(f_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj

def block_coordinate_step(f, B, n_top):
    """One pass of block coordinate descent."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    # For each block, compute the subgradient restricted to the block
    # and take a step
    for start in range(0, n, B):
        end = min(start + B, n)
        # Compute subgradient for this block
        g_block = np.zeros(end - start)
        for k in top_idx:
            wk = 1.0  # or use softmax weight
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            # Overlap with block [start, end)
            b_lo = max(start, j_lo)
            b_hi = min(end - 1, j_hi)
            if b_lo > b_hi:
                continue
            u_lo = k - b_hi
            u_hi = k - b_lo
            g_block[b_lo - start : b_hi - start + 1] += 2.0 * f[u_lo:u_hi + 1][::-1]
        
        # Try to reduce the peak by moving in -g_block direction
        # f_block_new = f_block - eta * g_block (clamped to 0)
        f_block = f[start:end]
        for eta in [0.001, 0.005, 0.01, 0.05, 0.1]:
            f_trial = f.copy()
            f_trial[start:end] = np.maximum(f_block - eta * g_block, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-12:
                f = f_trial
                # Recompute b and M
                b = fast_convolve(f)
                M = b.max()
                break
    
    return f, M

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w, n_top=None):
    """Compute subgradient of sum_k w[k] * (f*f)[k] w.r.t. f."""
    n = len(f)
    g = np.zeros(n)
    if n_top is not None and n_top < len(w):
        top_idx = np.argpartition(w, -n_top)[-n_top:]
    else:
        top_idx = np.where(w > 1e-14)[0]

    for k in top_idx:
        wk = 2.0 * w[k]
        if wk < 1e-14:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        u_lo = k - j_hi
        u_hi = k - j_lo
        g[j_lo:j_hi + 1] += wk * f[u_lo:u_hi + 1][::-1]
    return g


def md_step(f, eta, tau, n_top=None):
    """One mirror descent step with softmax subgradient."""
    n = len(f)
    b = fast_convolve(f)

    shifted = (b - b.max()) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    g = compute_subgradient(f, w, n_top)
    f_new = f * np.exp(-eta * g)
    s = f_new.sum()
    if s < 1e-30:
        f_new = np.ones(n) / n
    else:
        f_new /= s
    return f_new, b.max()

def lp_solve(f, top_k, eps=1e-8):
    """LP for best non-negative direction given top-K peak constraints."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def coordinate_polish(f_init, time_limit):

            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj


def block_polish(f_init, time_limit, B=64):

    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()

    n_top = max(20, n // 10)

        else:
            top_idx = np.arange(len(b))

        improved_pass = False
        for start in range(0, n, B):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            end = min(start + B, n)
            f_block = f[start:end].copy()

            # Compute subgradient for this block
            g_block = np.zeros(end - start)
            for k in top_idx:
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                b_lo = max(start, j_lo)
                b_hi = min(end - 1, j_hi)
                if b_lo > b_hi:
                    continue
                u_lo = k - b_hi
                u_hi = k - b_lo
                g_block[b_lo - start:b_hi - start + 1] += 2.0 * f[u_lo:u_hi + 1][::-1]

            # Try multiplicative update on this block
            for eta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                new_block = f_block * np.exp(-eta * g_block)

                if obj < M - 1e-12:
                    f = f_trial
                    M = obj
                    improved_pass = True
                    if M < best_obj - 1e-13:
                        best_obj = M
                    break

        if not improved_pass:
            break

    return f, best_obj

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    n_base = len(parent) if parent is not None else 1152

    # === Phase 1: Mirror Descent from multiple starts (~600s) ===
    print(f"\n=== Phase 1: Mirror Descent multi-start ===")
    inits = make_initializations(n_base, rng)
    if parent is not None:
        inits.insert(0, ("parent", parent.copy()))

    rem = deadline - time.monotonic()
    t_total = rem * 0.35
    t_per = t_total / len(inits)

    for name, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t1 = min(t_per, rem * 0.35)
        if t1 < 10:
            break
        print(f"  MD '{name}' n={len(f_init)} budget={t1:.0f}s")
        f1, obj1 = run_mirror_descent(f_init, time_limit=t1,
                                      eta_init=0.02, tau_init=0.002,
                                      label=f"md_{name}")
        score1 = 2.0 * len(f1) * obj1
        if score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [md_{name}]: {score1:.6f}")

    # === Phase 2: LP Refinement (~450s) ===
    print(f"\n=== Phase 2: LP (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = run_lp_phase(best_seq, time_limit=t2,
                                 top_k_init=40, top_k_max=200, label="lp_main")
        score2 = 2.0 * len(f2) * obj2
        if score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp_main]: {score2:.6f}")

    # === Phase 3: Block polish (~150s) ===
    print(f"\n=== Phase 3: Block polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 15:
        f3, obj3 = block_polish(best_seq, time_limit=t3, B=48)
        score3 = 2.0 * len(f3) * obj3
        if score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [block]: {score3:.6f}")

    # === Phase 4: MD continuation on best (~150s) ===
    print(f"\n=== Phase 4: MD continuation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 15:
        f4, obj4 = run_mirror_descent(best_seq, time_limit=t4,
                                       eta_init=0.005, tau_init=0.0005,
                                       label="md_cont")
        score4 = 2.0 * len(f4) * obj4
        if score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [md_cont]: {score4:.6f}")

    # === Phase 5: Multi-length exploration (~200s) ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    target_lengths = [int(n_cur * f) for f in [0.7, 0.85, 1.1, 1.25, 1.5, 2.0]]
    target_lengths = [max(512, min(4096, tl)) for tl in target_lengths]
    target_lengths = [tl for tl in target_lengths if tl != n_cur]
    seen = set()
    unique_lens = []
    for tl in target_lengths:
        if tl not in seen:
            seen.add(tl)
            unique_lens.append(tl)
    unique_lens.sort(key=lambda x: abs(x - n_cur))

    for target_n in unique_lens[:4]:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t5 = min(rem * 0.20, 100.0)
        if t5 < 15:
            break

        old_x = np.linspace(0.0, 1.0, n_cur)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  MD+LP n={target_n}, budget={t5:.0f}s")
        # Short MD then short LP
        t_md = t5 * 0.4
        t_lp = t5 * 0.6
        f_m, _ = run_mirror_descent(f_interp, time_limit=t_md,
                                     eta_init=0.01, tau_init=0.001,
                                     label=f"md_n{target_n}")
        f5, obj5 = run_lp_phase(f_m, time_limit=t_lp,
                                 top_k_init=30, top_k_max=100,
                                 label=f"lp_n{target_n}")
        score5 = 2.0 * target_n * obj5
        if score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.6f}")

    # === Phase 6: Final LP burst + CD (~remaining) ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6a = min(rem * 0.5, 80.0)
    t6b = min(rem * 0.3, 40.0)

    if t6a > 10:
        n_b = len(best_seq)
        f6, obj6 = run_lp_phase(best_seq, time_limit=t6a,
                                 top_k_init=100, top_k_max=250, label="lp_final")
        score6 = 2.0 * n_b * obj6
        if score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

    if t6b > 10:
        f7, obj7 = coordinate_polish(best_seq, time_limit=t6b)
        score7 = 2.0 * len(f7) * obj7
        if score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {score7:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        if parent is not None:
            best_seq = parent.copy()
            best_score = p_score
        else:
            best_seq = np.ones(n_base) / n_base
            best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
