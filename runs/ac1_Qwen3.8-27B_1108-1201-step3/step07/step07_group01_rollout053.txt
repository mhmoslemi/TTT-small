def mirror_descent_topk(a_init, time_limit, T_start=0.03, T_end=0.001,
                        eta_start=0.03, eta_end=0.0005, top_k=8,
                        verbose=True, label=""):
    """
    Mirror descent with top-k peak subgradient.
    At each step, identify top-k peaks and use their subgradient.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    stuck = 0
    
    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / 10000.0)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute subgradient from top-k peaks
        # Use weighted combination of peak subgradients
        s = b / T
        m = s.max()
        w_peaks = np.exp(s - m)
        w_peaks = w_peaks / w_peaks.sum()
        
        # Gradient of b[k] w.r.t. a[j] is: a[k-j] + a[j-k] (for valid indices)
        # Weighted gradient: sum_k w_peaks[k] * (a[k-j] + a[j-k])
        # We only need the top contributing peaks for efficiency
        top_idx = np.argsort(w_peaks)[::-1][:top_k]
        
        grad = np.zeros(n)
        for k in top_idx:
            wk = w_peaks[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # b[k] = sum_i a[i]*a[k-i], db[k]/da[j] = a[k-j] + a[j-k]
            # For j in [js, je]: a[k-j] is valid (k-j in [0, n-1])
            # a[j-k] is valid only if j >= k, i.e., j-k in [0, n-1]
            contrib1 = a[k - idx]
            # For a[j-k]: j-k >= 0 means j >= k, j-k < n means j < k+n (always true)
            valid2 = idx >= k
            contrib2 = np.zeros(n)
            contrib2[valid2] = a[idx[valid2] - k]
            grad += wk * (contrib1 + contrib2)
        
        # Project to tangent space
        grad = grad - np.dot(a, grad)
        
        # Mirror (multiplicative) update
        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s_val = a.sum()
        if s_val < 1e-15:
            a = best_a.copy()
        else:
            a = a / s_val
        
        # Stuck detection
        if stuck > 500:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            eta = min(eta * 5.0, 0.05)
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score

    if s < 1e-15:

    old_x = np.linspace(0, 1, old_n)

                  f"score={current_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def dual_ascent_optimize(a_init, time_limit, T_start=0.05, T_end=0.001,
                         verbose=True, label=""):
    """
    Dual-ascent with peak equalization.
    Maintains soft dual weights on peaks, uses water-filling primal update.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / 5000.0)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
        
        # Soft dual weights (temperature-controlled)
        s = b / T
        m = s.max()
        lam = np.exp(s - m)
        lam = lam / lam.sum()
        
        # Compute potential V[j]
        V = cross_corr_weighted(lam, a)
        
        # Water-filling: a_new proportional to exp(V / tau)
        tau = T
        V_shifted = V - V.max()
        a_new = np.exp(V_shifted / tau)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        # Blend with current (higher alpha later for convergence)
        alpha = 0.2 + 0.6 * progress
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} DA iter={it} ({elapsed:.0f}s): "
                  f"score={current_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def newton_peak_reduce(a_init, time_limit, verbose=True, label=""):
    """
    Newton-style peak reduction: target the peak's top contributors,
    reduce them with proportional redistribution.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
        
        # Find peak (prefer interior)
        b_work = b.copy()
        b_work[0] = b.max() * 0.999
        b_work[-1] = b.max() * 0.999
        k_peak = int(np.argmax(b_work))
        
        # Identify contributors
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[k_peak - idx]
        
        improved = False
        # Try top contributors
        top_ci = np.argsort(contribs)[::-1][:5]
        for ci in top_ci:
            i = int(idx[ci])
            j = k_peak - i
            if a[i] < 1e-15:
                continue
            for factor in [0.995, 0.99, 0.98, 0.97, 0.95, 0.93, 0.9]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_norm(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # If single reduction didn't work, try paired reduction
        if not improved:
            for ci in top_ci[:3]:
                i = int(idx[ci])
                j = k_peak - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.99, 0.97, 0.95]:
                    for f2 in [0.99, 0.97, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = score_norm(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Try broader: reduce all top-n contributors slightly
            n_broad = min(10, len(idx))
            top_broad = np.argsort(contribs)[::-1][:n_broad]
            a_trial = a.copy()
            for ci in top_broad:
                i = int(idx[ci])
                a_trial[i] *= 0.99
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                sc = score_norm(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
            if not improved:
                break
    
    return best_a, best_score


def coordinate_descent_fast(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    
    factors = np.array([0.7, 0.85, 0.95, 1.05, 1.15, 1.3])
    no_improve = 0
    max_no_improve = 30
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve += 1
        improved_pass = False
        rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
        order = rng.permutation(n)
        
        for i in order:
            if no_improve >= max_no_improve:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for f in factors:
                a_trial = a.copy()
                a_trial[i] = old_val * f
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
