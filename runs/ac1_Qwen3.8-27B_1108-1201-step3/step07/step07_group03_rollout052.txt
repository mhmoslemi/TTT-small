"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP-based fictitious play with damped updates at multiple sequence lengths
2. Adam optimizer with log-sum-exp smoothed gradient
3. Golden-section coordinate descent polishing
4. Multi-scale exploration and iterative refinement
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.CVXOPT]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play_damped(n, time_limit, initial=None, solver=None, verbose=False):
    """Damped alternating LP best responses with running average."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Damping factor
        alpha = min(1.0, 2.0 / (iters + 1))
        
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = (1 - alpha) * b + alpha * b_new
        
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = (1 - alpha) * a + alpha * a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)

        # Evaluate candidates
        candidates = [a, b, 0.5 * (a + b), a_avg]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def adam_smooth_optimize(a0, time_limit, verbose=False):
    """Adam optimizer with log-sum-exp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    temperature = 0.02  # Log-sum-exp temperature
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        
        # Smoothed max via log-sum-exp
        temp = temperature
        b_max = b.max()
        logsumexp = b_max + temp * np.log(np.sum(np.exp((b - b_max) / temp)))
        # Gradient of logsumexp w.r.t. b
        w = np.exp((b - b_max) / temp)
        w /= w.sum()
        
        # Gradient of conv(a,a) w.r.t. a[i] is 2 * (a * reverse(a)) [i]... 
        # Actually, d/da[i] conv(a,a)[k] = a[k-i] + a[i-k] (for valid indices)
        # So gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a[i] = 2 * sum_k w[k] * (contribution)
        
        # Compute gradient efficiently
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            wk = 2.0 * w[k] / n  # The factor 2/n comes from the objective 2n*max/(sum)^2, but since we're optimizing with sum=1, the gradient of 2n*conv is just 2n * dconv
            # Actually, let's just compute d(dconv)/da[i]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                grad[j] += 2.0 * w[k] * a[k - j]
        
        # Scale gradient
        grad = grad / n  # Normalize
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        y = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(y)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Anneal temperature
        if it % 100 == 0:
            temperature = max(0.005, temperature * 0.95)
            lr = max(0.001, lr * 0.99)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam it={it} ({elapsed:.0f}s): best={best_score:.6f} temp={temperature:.4f}")

    return best_a, best_score

def golden_section_coord_descent(a0, time_limit, n_sweeps=100):
    """Coordinate descent with golden section search for optimal value of each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    golden = (np.sqrt(5) - 1) / 2  # Golden section ratio
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try small positive values
                for trial_val in [0.001, 0.005, 0.01]:
                    a[i] = trial_val
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-12:
                        best_score = score
                        best_a = a.copy()
                        old_val = trial_val
                        improved = True
                    else:
                        a[i] = 0.0
                        a = a / a.sum()
                continue
            
            # Golden section search around old_val
            # Search in [old_val * 0.5, old_val * 2.0]
            lo = old_val * 0.5
            hi = old_val * 2.0
            c = hi - golden * (hi - lo)
            d_val = lo + golden * (hi - lo)
            
            # Evaluate at c and d
            a[i] = c
            a = np.maximum(a, 0)
            a = a / a.sum()
            score_c = 2.0 * n * fast_convolve(a).max()
            
            a[i] = d_val
            a = np.maximum(a, 0)
            a = a / a.sum()
            score_d = 2.0 * n * fast_convolve(a).max()
            
            # Also check the original value
            a[i] = old_val
            a = a / a.sum()
            score_orig = 2.0 * n * fast_convolve(a).max()
            
            # Narrow the search
            for _ in range(8):
                if score_c < score_d:
                    hi = d_val
                    d_val = c
                    score_d = score_c
                    c = hi - golden * (hi - lo)
                    a[i] = c
                    a = a / a.sum()
                    score_c = 2.0 * n * fast_convolve(a).max()
                else:
                    lo = c
                    c = d_val
                    score_c = score_d
                    d_val = lo + golden * (hi - lo)
                    a[i] = d_val
                    a = a / a.sum()
                    score_d = 2.0 * n * fast_convolve(a).max()
            
            # Best of c, d, original
            best_local = min(score_c, score_d, score_orig)
            if best_local < best_score - 1e-12:
                if best_local == score_c:
                    a[i] = c
                elif best_local == score_d:
                    a[i] = d_val
                else:
                    a[i] = old_val
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val
                a = a / a.sum()
        
        if not improved and sweep > 5:
            break
    
    return best_a, best_score

def pairwise_coord_descent(a0, time_limit, n_sweeps=50):
    """Try updating pairs of coordinates simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for _ in range(n // 2):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.choice(n, size=2, replace=False)
            old_i, old_j = a[i], a[j]
            
            # Try various joint adjustments
            for fi, fj in [(0.8, 1.2), (1.2, 0.8), (0.7, 1.3), (1.3, 0.7),
                          (0.9, 1.1), (1.1, 0.9), (0.6, 1.4), (1.4, 0.6)]:
                a[i] = old_i * fi
                a[j] = old_j * fj
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t < 1e-15:
                    a[i], a[j] = old_i, old_j
                    continue
                a = a / s_t
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i], a[j] = old_i, old_j
                    a = a / a.sum()
            else:
                continue
            break  # Inner break triggered
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def random_fourier_init(n, rng, n_harm=10):
    """Random Fourier superposition initialization."""
    freqs = rng.integers(1, 20, n_harm)
    amplitudes = rng.random(n_harm) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    return best_seq, best_score
