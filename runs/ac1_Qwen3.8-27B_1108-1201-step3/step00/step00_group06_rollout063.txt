import numpy as np
import time

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

    result = np.fft.irfft(A * B, size)
    return result[:n]


def _fft_correlate(p, a):
    """Compute grad[i] = sum_j p[i+j]*a[j] for i=0..len(a)-1 using FFT."""
    # This is the cross-correlation in 'valid' mode
    n_p = len(p)
    n_a = len(a)
    n_out = n_p - n_a + 1
    size = 1 << (n_p + n_a - 1 - 1).bit_length()
    P = np.fft.rfft(p, size)
    Ar = np.fft.rfft(a[::-1], size)
    corr = np.fft.irfft(P * Ar, size)
    # corr[k] = sum_j a[::-1][j] * p[k-j] = sum_j a[n_a-1-j] * p[k-j]
    # We want result[i] = sum_j p[i+j] * a[j]
    # Let me verify: corr_full = ifft(fft(p, size) * conj(fft(a, size)))
    # corr_full[m] = sum_k p[k] * conj(a[k - m + offset])
    # This is getting confusing. Let me just use the direct formula via FFT:
    # result[i] = sum_{j=0}^{n_a-1} p[i+j] * a[j]
    # = cross-correlation of p and a at lag i
    # Using: result = ifft(fft(p, N) * conj(fft(a, N)))  [circular correlation]
    # For N >= n_p + n_a - 1, the circular = linear
    # corr[m] = sum_k p[k] * a[k - m] (circular)
    # We want result[i] = sum_j p[i+j] * a[j] = sum_j p[j+i] * a[j]
    # = cross_corr(i) where cross_corr[i] = sum_j a[j] * p[j+i]
    # In the standard definition: (a * p')[m] = sum_j a[j] p'[m-j] where p' = p reversed
    # p'[m-j] = p[len(p)-1-(m-j)] ... this is getting too convoluted.
    # Let me just use numpy for this (it's one call, should be fine)
    return np.correlate(p, a, mode='valid')


def _optimize_from(a_init, n, rng, deadline, best_score_global, best_seq_global):
    """Run projected gradient descent from a_init, return (best_score, best_seq)."""
    best = a_init.copy()
    best_score = 2.0 * n * np.max(_convolve_full(best, best))
    
    lam = 30.0
    lr = 0.05
    mu = 0.85
    v = np.zeros(n)
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = _convolve_full(a_init, a_init)
        max_c = np.max(c)
        current_score = 2.0 * n * max_c
        
        # Softmax weights
        shifted = c - max_c
        log_w = lam * shifted
        log_w = np.clip(log_w, -400.0, 400.0)
        w = np.exp(log_w)
        wsum = w.sum()
        if wsum > 0 and np.isfinite(wsum):
            p = w / wsum
        else:
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        
        # Gradient
        grad = 2.0 * _fft_correlate(p, a_init)
        
        # Momentum update
        v = mu * v + (1.0 - mu) * grad
        a_new = _project_simplex(a_init - lr * v)
        
        if not np.isfinite(a_new).all():
            a_new = best.copy()
            v = np.zeros(n)
        
        a_init = a_new
        total_iters += 1
        
        # Track best (check every few iterations to save time)
        if total_iters % 5 == 0:
            c_check = _convolve_full(a_init, a_init)
            score_check = 2.0 * n * np.max(c_check)
            if score_check < best_score:
                best = a_init.copy()
                best_score = score_check
                no_improve = 0
            else:
                no_improve += 5
        
        # Restart if stuck
        if no_improve > 500:
            a_init = best.copy()
            v = np.zeros(n)
            noise_scale = 0.02 + 0.05 * rng.random()
            noise = rng.normal(0, noise_scale, size=n)
            a_init = _project_simplex(a_init + noise)
            no_improve = 0
            lam = min(lam * 1.5, 300.0)
            lr = max(lr * 0.75, 0.001)
        
        # Adaptive parameters
        if total_iters % 3000 == 0:
            lam = min(lam * 1.15, 300.0)
        if total_iters % 10000 == 0:
            lr = max(lr * 0.92, 0.001)
    
    # Final check
    c_final = _convolve_full(best, best)
    best_score = 2.0 * n * np.max(c_final)
    
    # Update global best
    if best_score < best_score_global:
        best_score_global = best_score
        best_seq_global = best.copy()
    
    return best_score, best_seq_global


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
    with log-sum-exp smoothing, momentum, and multiple random restarts.
    The problem is scale-invariant, so we normalize sum(a)=1 and minimize 2n*max(a*a).
    Uses FFT for fast convolution and simplex projection for the constraint.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    
    # Get initial sequence
    try:
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    except (NameError, AttributeError):
        a0 = np.ones(1024, dtype=np.float64)
    
    if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all() or np.sum(a0) < 1e-10:
        a0 = np.ones(1024, dtype=np.float64)
    
    n = len(a0)
    print(f"n = {n}")
    
    # Generate diverse initializations
    inits = []
    
    # 1. Original sequence (normalized)
    s = np.sum(a0)
    inits.append(a0 / s if s > 1e-10 else np.ones(n) / n)
    
    # 2. Uniform
    inits.append(np.ones(n) / n)
    
    # 3. Gaussian-like (centered bump)
    centers = np.arange(n)
    for width_factor in [0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((centers - n / 2) / (width_factor * n)) ** 2)
        g /= g.sum()
        inits.append(g)
    
    # 4. Two-block (asymmetric)
    ratio = 0.3
    two_block = np.zeros(n)
    two_block[:int(n * ratio)] = 1.0
    two_block[int(n * ratio):] = 0.3
    two_block /= two_block.sum()
    inits.append(two_block)
    
    # 5. Random exponential
    rand_exp = rng.exponential(1.0, size=n)
    rand_exp /= rand_exp.sum()
    inits.append(rand_exp)
    
    # 6. Cosine-modulated
    cos_mod = 0.5 + 0.5 * np.cos(2 * np.pi * centers / max(1, n // 4))
    cos_mod = np.maximum(cos_mod, 0.01)
    cos_mod /= cos_mod.sum()
    inits.append(cos_mod)
    
    # 7. Ramp (linearly increasing)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits.append(ramp)
    
    print(f"Number of initializations: {len(inits)}")
    
    # Track global best
    best_score_global = np.inf
    best_seq_global = None
    
    # Evaluate all initializations
    for i, init in enumerate(inits):
        c = _convolve_full(init, init)
        score = 2.0 * n * np.max(c)
        if score < best_score_global:
            best_score_global = score
            best_seq_global = init.copy()
        print(f"  Init {i}: score = {score:.6f}")
    
    # Divide time among restarts (give more time to better initializations)
    # Strategy: run each init for a portion of the time, but stop early if time is up
    n_inits = len(inits)
    # Allocate time proportionally: more time for promising starts
    init_scores = np.array([2.0 * n * np.max(_convolve_full(init, init)) for init in inits])
    # Lower score = better = more time
    weights = 1.0 / (init_scores - best_score_global + 0.1)
    weights /= weights.sum()
    
    total_time_available = max(0.0, total_deadline - time.monotonic())
    print(f"Total time for optimization: {total_time_available:.1f}s")
    
    for i, init in enumerate(inits):
        if time.monotonic() >= total_deadline:
            break
        # Allocate time for this init
        alloc_time = total_time_available * weights[i]
        # Ensure at least 10 seconds per init
        alloc_time = max(10.0, alloc_time)
        local_deadline = min(time.monotonic() + alloc_time, total_deadline)
        
        print(f"Optimizing init {i} (score={init_scores[i]:.6f}), time budget={alloc_time:.1f}s")
        best_score_global, best_seq_global = _optimize_from(
            init.copy(), n, rng, local_deadline, best_score_global, best_seq_global
        )
        print(f"  After init {i}: best_score = {best_score_global:.6f}")
    
    # Final polish: run a short optimization from the global best with fine parameters
    if time.monotonic() < total_deadline - 30:
        polish_deadline = total_deadline
        a Polish = best_seq_global.copy()
        best_score_global, best_seq_global = _optimize_from(
            best_seq_global.copy(), n, np.random.default_rng(seed + 999),
            polish_deadline, best_score_global, best_seq_global
        )
    
    # Official evaluation
    official_score = float(evaluate_sequence(best_seq_global.tolist()))
    print(f"\nFinal official score: {official_score:.6f}")
    print(f"Final direct score: {best_score_global:.6f}")
    
    if np.isfinite(official_score) and official_score < best_score_global:
        best_score_global = official_score
    
    return [float(x) for x in best_seq_global]
