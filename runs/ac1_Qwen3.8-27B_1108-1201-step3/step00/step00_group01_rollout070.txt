import numpy as np
from scipy import fft as spfft
import time

"""
Minimize 2n*max(auto_convolution) / (sum_a)^2 using gradient-based optimization
with smooth-max (log-sum-exp) approximation and FFT-based gradients.
Explores multiple initializations (geometric, power-law, given sequence)
and multiple sequence lengths, with backtracking line search and
non-negativity projection.
"""


def next_pow2(n):
    x = 1
    while x < n:
        x *= 2
    return x


def auto_convolve_fft(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    A = spfft.fft(a, size)
    c = spfft.ifft(A * A).real[:2 * n - 1]
    return c


def smooth_max_and_weights(c, beta):
    c_max = c.max()
    shifted = np.clip(beta * (c - c_max), -500, 500)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    m = c_max + np.log(total) / beta
    return m, weights


def grad_smooth_max(a, weights):
    n = len(a)
    size = next_pow2(3 * n - 1)
    a_fft = spfft.fft(a, size)
    w_fft = spfft.fft(weights, size)
    aw = spfft.ifft(a_fft * w_fft).real[:3 * n - 1]
    a_rev = a[::-1]
    ar_fft = spfft.fft(a_rev, size)
    awr = spfft.ifft(ar_fft * w_fft).real[:3 * n - 1]
    term2 = aw[:n]
    term1 = awr[n - 1:2 * n - 1]
    return term1 + term2


def exact_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = auto_convolve_fft(a)
    return float(2.0 * n * c.max() / (S * S))


def geometric_init(n, c_param=1.0):
    r = 1.0 + c_param / n
    return np.power(r, np.arange(n, dtype=np.float64))


def power_law_init(n, p=1.0):
    return np.power(np.arange(1, n + 1, dtype=np.float64), p)


def optimize_sequence(a0, beta_start=5.0, beta_end=80.0, max_iters=3000,
                      deadline=None, verbose=False):
    """Run gradient descent on a given initialization. Returns (best_a, best_score)."""
    n = len(a0)
    a = np.maximum(0, a0).astype(np.float64)
    S = a.sum()
    if S < 0.01:
        return a, np.inf
    # Normalize to mean=1 for numerical stability
    a = a / S * n
    best_a = a.copy()
    best_score = exact_objective(a)
    if not np.isfinite(best_score):
        return best_a, best_score

    lr = 0.5
    beta = beta_start

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = auto_convolve_fft(a)
        m_smooth, weights = smooth_max_and_weights(c, beta)
        grad_m = grad_smooth_max(a, weights)

        S = a.sum()
        M_exact = c.max()
        grad_obj = (2.0 * n / (S * S)) * grad_m - (2.0 * n * M_exact / (S ** 3))

        # Backtracking line search (limit to 8 halvings)
        step = lr
        improved = False
        for bt in range(8):
            a_try = np.maximum(0, a - step * grad_obj)
            S_try = a_try.sum()
            if S_try < 0.01:
                step *= 0.5
                continue
            # Quick eval using FFT
            c_try = auto_convolve_fft(a_try)
            score_try = 2.0 * n * c_try.max() / (S_try * S_try)
            if np.isfinite(score_try) and score_try < best_score:
                a = a_try
                best_score = score_try
                best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if improved:
            # Occasionally increase beta (sharpen the max approximation)
            if it % 30 == 29 and beta < beta_end:
                beta = min(beta * 1.4, beta_end)
            # Occasionally reset lr upward
            if it % 50 == 49:
                lr = min(lr * 1.1, 2.0)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break

    # Final block perturbation refinement
    rng_local = np.random.default_rng(123 + n)
    for it in range(100):
        if deadline is not None and time.monotonic() > deadline:
            break
        width = int(rng_local.integers(max(1, n // 80), max(2, n // 8)))
        start = int(rng_local.integers(0, n - width + 1))
        a_try = best_a.copy()
        a_try[start:start + width] *= float(np.exp(rng_local.normal(0, 0.03)))
        a_try = np.maximum(0, a_try)
        S_try = a_try.sum()
        if S_try < 0.01:
            continue
        c_try = auto_convolve_fft(a_try)
        score_try = 2.0 * n * c_try.max() / (S_try * S_try)
        if np.isfinite(score_try) and score_try < best_score:
            best_a = a_try
            best_score = score_try

    return best_a, best_score
