import numpy as np
import time
from math import lgamma, exp


def compute_score_fast(a):
    """Fast score computation: 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a, mode='full')
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (sum_a * sum_a))


def expected_asymptotic_score(p):
    """Asymptotic score for a[j]=j^p: 2*B(p+1,p+1)*(p+1)^2."""
    logB = 2.0 * lgamma(p + 1) - lgamma(2 * p + 2)
    return 2.0 * exp(logB) * (p + 1) ** 2


def logbeta_p(p):
    """log(B(p+1, p+1)) = 2*lgamma(p+1) - lgamma(2p+2)."""
    return 2.0 * lgamma(p + 1) - lgamma(2 * p + 2)


def build_power_sequence(n, p):
    """Build a[j] = (j/(n-1))^p, values in [0,1]."""
    base = np.arange(n, dtype=np.float64) / float(n - 1)
    return np.power(base, float(p))
