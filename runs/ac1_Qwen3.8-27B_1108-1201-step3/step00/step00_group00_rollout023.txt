```python
import numpy as np
import time
from scipy import fftpack

def fast_convolve(a, b):
    """Compute full convolution using FFT."""
    n = len(a) + len(b) - 1
    f = np.fft.fft(a, n=2*n)  # pad to avoid circular convolution issues
    g = np.fft.fft(b, n=2*n)
    return np.fft.ifft(f * g)[:n].real

def compute_objective(a):
    """Compute the objective value without calling evaluate_sequence."""
    n = len(a)
    c = fast_convolve(a, a)
    max_c = np.max(c)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def gradient_descent(a, max_time=60.0, lr=0.01):
    """Run gradient descent to minimize the objective."""
    a = a.copy()
    n = len(a)
    start = time.monotonic()
    best = a.copy()
    best_score = compute_objective(a)
    
    while time.monotonic() - start < max_time:
        c = fast_convolve(a, a)
        k_star = np.argmax(c)
        s = np.sum(a)
        
        if s < 0.01:
            break
        
        # Gradient: 4*n/S^2 * (a[k*-j] * 1[0<=k*-j<=n-1] - C[k*]/S)
        mirror = np.zeros(n)
        # For each j, mirror[j] = a[k*-j] if 0 <= k*-j <= n-1
        valid_start = max(0, k_star - (n - 1))
        valid_end = min(n - 1, k_star)
        for j in range(valid_start, valid_end + 1):
            mirror[j] = a[k_star - j]
        
        r = c[k_star] / s
        grad = 4.0 * n / (s * s) * (mirror - r)
        
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, 1000)
        
        score = compute_objective(a_new)
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new
        else:
            lr *= 0.9
    
    return best, best_score
```

"""
Gradient descent on probability simplex to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based convolution for speed, projected gradient with adaptive learning rate,
multi-start with different sequence lengths, and parametric initialization.
"""

import numpy as np
import time


def _fft_convolve_full(a):
    """Compute full autocorrelation convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    nfft = 1
    while nfft < 2 * size:
        nfft <<= 1
    f = np.fft.rfft(a, n=nfft)
    result = np.fft.irfft(f * f, n=nfft)[:size]
    # Clamp tiny negative values from FFT roundoff
    np.maximum(result, 0, out=result)
    return result


def _fast_objective(a):
    """Compute objective: 2*n*max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve_full(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _gradient(a):
    """Compute gradient of 2*n*C(k*)/S^2 assuming k*=argmax is fixed."""
    n = len(a)
    s = float(a.sum())
    c = _fft_convolve_full(a)
    k_star = int(c.argmax())
    c_max = float(c[k_star])

    # mirror[j] = a[k_star - j] if 0 <= k_star - j < n, else 0
    mirror = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        # mirror[lo..hi] = a[k_star-hi .. k_star-lo] reversed
        seg = a[k_star - hi: k_star - lo + 1]
        mirror[lo:hi + 1] = seg[::-1]

    r = c_max / s
    grad = 4.0 * n / (s * s) * (mirror - r)
    return grad


def _project_to_simplex(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def _run_gd(a_init, time_budget, lr_init=0.05, seed=0):
    """Run gradient descent. Returns (best_a, best_score, n_iterations)."""
    rng = np.random.default_rng(seed)
    a = _project_to_simplex(a_init.copy().astype(np.float64))
    n = len(a)
    best = a.copy()
    best_score = _fast_objective(best)
    lr = lr_init
    stagnant = 0
    start_t = time.monotonic()
    iters = 0

    while time.monotonic() - start_t < time_budget:
        grad = _gradient(a)
        # Project gradient to tangent space (sum=0)
        grad = grad - grad.mean()
        a_new = a + lr * grad  # Move in negative gradient direction
        a_new = _project_to_simplex(a_new)
        iters += 1

        score = _fast_objective(a_new)
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            stagnant = 0
            # Slightly increase lr if making progress

        else:

            lr *= 0.9

                a = _project_to_simplex(best * (1.0 + noise))
                lr = lr_init
                stagnant = 0
                continue
        a = a_new

    return best, best_score, iters

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Family 5: inverse (low in middle, high at edges)
            a = 1.0 + t * (1.0 - (2.0 * x - 1.0) ** 2)
            a = np.maximum(a, 0.01)  # Keep positive
            score = _fast_objective(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    if best_a is None:
        best_a = np.ones(512)
        best_score = 2.0
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses:
    1. Parametric family search for initialization.
    2. Gradient descent on the probability simplex with adaptive LR.
    3. Multi-start with different sequence lengths and perturbations.
    4. Final verification with evaluate_sequence.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    # Phase 1: Parametric search (30 seconds)
    p1_budget = min(30.0, (deadline - time.monotonic()) * 0.1)
    init_a, init_score = _parametric_search(p1_budget, seed=seed)
    print(f"Phase 1 done: best_score={init_score:.6f}, n={len(init_a)}")

    best_a = init_a.copy()
    best_score = init_score

    # Phase 2: Gradient descent refinement
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        # Run GD on the best parametric result
        gd_budget = remaining * 0.4
        a_gd, score_gd, iters = _run_gd(best_a, gd_budget, lr_init=0.05, seed=seed)
        print(f"Phase 2a: score={score_gd:.6f} (iters={iters})")
        if score_gd < best_score:
            best_a = a_gd
            best_score = score_gd

    # Phase 3: Multi-start with different lengths and perturbations
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        lengths_to_try = [256, 512, 1024, 2048, 4806, 4096, 8192]
        # Include the length that gave best so far
        lengths_to_try = list(set(lengths_to_try + [len(best_a)]))
        lengths_to_try.sort()

        per_length_budget = max(5.0, remaining * 0.6 / len(lengths_to_try))

        for n_try in lengths_to_try:
            if time.monotonic() >= deadline:
                break
            actual_budget = min(per_length_budget, deadline - time.monotonic())
            if actual_budget < 5.0:
                break

            # Try multiple initializations for this length
            n_starts = max(1, int(actual_budget / 10.0) // 2)
            n_starts = min(n_starts, 4)

            for start_i in range(n_starts):
                if time.monotonic() >= deadline:
                    break
                sub_budget = max(5.0, (deadline - time.monotonic()) / max(1, n_starts - start_i))
                sub_budget = min(sub_budget, actual_budget)

                if start_i == 0:
                    # Start from current best (resized or as-is)
                    a_init = best_a.copy()
                    if len(a_init) != n_try:
                        # Resample to new length
                        old_x = np.linspace(0, 1, len(a_init))
                        new_x = np.linspace(0, 1, n_try)
                        a_init = np.interp(new_x, old_x, a_init)
                else:
                    # Random perturbations of various types
                    base = np.ones(n_try)
                    pert_type = start_i % 4
                    if pert_type == 1:
                        # Random smooth perturbation
                        freqs = rng.integers(1, 5, size=3)
                        amps = rng.uniform(0.1, 0.5, size=3)
                        phases = rng.uniform(0, 2 * np.pi, size=3)
                        x = np.arange(n_try) / n_try
                        for f, amp, ph in zip(freqs, amps, phases):
                            base += amp * np.cos(2 * np.pi * f * x + ph)
                    elif pert_type == 2:
                        # Edge boost
                        x = np.arange(n_try) / max(1, n_try - 1)
                        base += 0.5 * (2 * x - 1) ** 2
                    elif pert_type == 3:
                        # Middle dip
                        x = np.arange(n_try) / max(1, n_try - 1)
                        base -= 0.3 * np.exp(-((2 * x - 1) / 0.3) ** 2)
                    base = np.maximum(base, 0.01)

                    a_init = base

                a_gd, score_gd, _ = _run_gd(a_init, sub_budget,
                                             lr_init=0.05,
                                             seed=seed + start_i * 1000 + n_try)
                if score_gd < best_score:
                    best_score = score_gd
                    best_a = a_gd.copy()
                    print(f"  n={n_try}, start={start_i}: improved to {best_score:.6f}")

    # Phase 4: Final polish on the best sequence
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        a_gd, score_gd, _ = _run_gd(best_a, remaining, lr_init=0.01, seed=seed + 99999)
        if score_gd < best_score:
            best_score = score_gd
            best_a = a_gd.copy()
            print(f"Phase 4 polish: {best_score:.6f}")

    # Verify with the official evaluator
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    print(f"Final verified score: {final_score:.6f} (n={len(best_a)})")

    return [float(x) for x in best_a]
