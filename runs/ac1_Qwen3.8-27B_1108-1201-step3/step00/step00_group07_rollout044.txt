import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

def compute_score(a):
    """Compute the score for a normalized sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return float('inf')
    conv = fft_convolve(a, a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)

    """Run gradient-based optimization on sequence a."""
    deadline = time.monotonic() + budget_s
    n = len(a)
    a = a.copy().astype(np.float64)
    best = a.copy()
    best_score = compute_score(a)
    
    lr = 0.1
    lr_decay = 0.999
    step = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        S = np.sum(a)
        if S < 1e-10:
            break
        conv = fft_convolve(a, a)
        M = np.max(conv)
        score = 2.0 * n * M / (S * S)
        
        if score < best_score:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        # Find max positions (use a small set for efficiency)
        threshold = M * 0.999
        max_mask = conv >= threshold
        max_pos = np.where(max_mask)[0]
        
        # Limit to at most a few positions
        if len(max_pos) > 50:
            top_idx = np.argsort(conv[max_pos])[-50:]
            max_pos = max_pos[top_idx]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in max_pos:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                i_vals = np.arange(lo, hi + 1)
                # grad[i] += 2 * a[k - i]
                # k - i ranges from k-hi to k-lo
                a_slice = a[k - hi : k - lo + 1][::-1]
                grad[lo:hi+1] += 2.0 * a_slice
        
        # Score gradient
        score_grad = 2.0 * n * (grad * S - 2.0 * M) / (S ** 3)
        
        # Multiplicative update
        a = a * np.exp(-lr * score_grad)
        a = np.maximum(a, 1e-15)
        
        # Adaptive learning rate
        lr *= lr_decay
        if lr < 1e-6:
            lr = 0.05
            # Try a random restart from current best
            if no_improve > 1000:
                a = best * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                no_improve = 0
        
        step += 1
    
    return best, best_score

Let f(a) = 2n * M(a) / S(a)^2, where M(a) = max_k (a*a)_k and S(a) = sum(a).

∂f/∂a_i = 2n * [∂M/∂a_i * S^2 - M * 2S * ∂S/∂a_i] / S^4
= 2n * [∂M/∂a_i * S - 2M] / S^3

where ∂M/∂a_i is a subgradient of M at a.

If M is achieved at position k*, then:
∂M/∂a_i = ∂(a*a)_{k*}/∂a_i = 2 * a[k*-i] if 0 <= k*-i < n, else 0.

So the subgradient of f is:
∂f/∂a_i = 2n * [2*a[k*-i]*[0<=k*-i<n] * S - 2M] / S^3

Let me simplify: let c_i = 2*a[k*-i]*[0<=k*-i<n]. Then:
∂f/∂a_i = 2n * (c_i * S - 2M) / S^3 = 4n * (c_i * S - 2M) / (2*S^3)

For a uniform sequence (a_i = 1/S for all i, S=1 so a_i = 1/n... wait, let me use a_i = c for all i, S = nc):

- c_i = 2*c for all valid i (all i are valid when k* = n-1)
