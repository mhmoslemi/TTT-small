```python
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import time

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    # Build the constraint matrix A (2n-1 x n)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] if k-n+1 <= j <= k and 0 <= j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n - 1, M)
    c = -np.ones(n)  # minimize -sum(g)
    
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```
