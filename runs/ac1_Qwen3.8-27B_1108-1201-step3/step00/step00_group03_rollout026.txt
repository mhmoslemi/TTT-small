def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    mu = np.sort(v)[::-1]
    rho = np.nonzero(mu * np.arange(1, n+1) > (mu.sum() - 1))[0][-1]
    theta = (mu[:rho+1].sum() - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

from scipy.sparse import csr_matrix

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2011
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, k):
    """Compute gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad

def gradient_descent_step(a, k, v, lr, momentum):
    """One step of gradient descent with momentum on the simplex."""
    grad = compute_gradient(a, k)
    v = momentum * v - lr * grad
    a_new = a + v
    return simplex_project(a_new), v

def solve_lp_step(a, M):
    """Solve the LP to find a good direction."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            g_indices = k - indices
            rows_list.extend(np.full(len(indices), k).tolist())
            cols_list.extend(g_indices.tolist())
            vals_list.extend(a[indices].tolist())
    
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))

    n = len(a)

    if n == 0:

    return grad


def compute_gradient_topk(a, c, top_k=3):

                        if a_trial.sum() < 1e-10:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        trial_s = score_of(a_trial, n)
                        if trial_s < best_score:
                            best_score = trial_s
                            best_a = a_trial.copy()
                            a = a_trial
                            stagnant = 0
                            print(f"  LP step: score={best_score:.6f} (t={t_b})", flush=True)
                            break
                    else:
                        a = (1 - 0.05) * a + 0.05 * g
                        a = np.clip(a, 0, None)
                        if a.sum() > 1e-10:
                            a = a / a.sum()
                stagnant = max(0, stagnant - 500)
        
        total_iters += 1
        
        if total_iters % 20000 == 0:
            print(f"Iter {total_iters}: best={best_score:.6f}, cur={current_score:.6f}, lr={lr:.5f}, stag={stagnant}", flush=True)
    
    return best_a, best_score, total_iters


def propose_candidate(budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses gradient descent with momentum, top-k active constraints, LP refinement,
    simulated annealing, and multiple restart strategies.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    
    # --- Get initial sequence ---
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        base = np.array(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, 1000.0)
        if base.size > 0 and np.all(np.isfinite(base)) and base.sum() > 0.01:
            n = base.size
        else:
            n = 2535
            base = np.ones(n)
    else:
        n = 2535
        base = np.ones(n)
    
    rng = np.random.default_rng(42)
    
    best_a = base / base.sum()
    best_score = score_of(best_a, n)
    print(f"Starting: n={n}, initial_score={best_score:.6f}", flush=True)
    
    # --- Multi-start optimization ---
    # Try several different initial noise patterns
    start_patterns = [
        ("gaussian", lambda x: rng.normal(0, 0.05, n)),
        ("sine3", lambda x: 0.05 * np.sin(6 * np.pi * x / n)),
        ("sine7", lambda x: 0.04 * np.sin(14 * np.pi * x / n)),
        ("random_block", lambda x: np.repeat(rng.normal(0, 0.03, n // 10 + 1), 10)[:n]),
        ("edge_boost", lambda x: 0.03 * (np.arange(n) / n - 0.5) ** 2 * 4),
    ]
    
    time_per_start = (deadline - time.monotonic()) / (len(start_patterns) + 1)
    
    for i, (name, pattern_fn) in enumerate(start_patterns):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        start_deadline = time.monotonic() + min(time_per_start, remaining * 0.4)
        
        a_init = best_a + pattern_fn(np.arange(n)) + rng.normal(0, 0.02, n)
        a_init = np.clip(a_init, 0, None)
        if a_init.sum() < 1e-10:
            a_init = best_a + rng.normal(0, 0.05, n)
            a_init = np.clip(a_init, 0, None)
        a_init = a_init / a_init.sum()
        
        print(f"Start {i+1}/{len(start_patterns)} ({name}): starting_score={score_of(a_init, n):.6f}", flush=True)
        
        local_best, local_score, iters = optimize_sequence(a_init, n, start_deadline, rng)
        
        if local_score < best_score:
            best_score = local_score
            best_a = local_best
            print(f"  -> New best: {best_score:.6f} after {iters} iters", flush=True)
    
    # --- Final polish with remaining time ---
    if time.monotonic() < deadline - 3:
        print("Final polish phase...", flush=True)
        a = best_a.copy()
        v = np.zeros(n)
        lr = 0.01
        momentum = 0.9
        stagnant = 0
        
        while time.monotonic() < deadline - 3:
            c = np.convolve(a, a)
            k = int(np.argmax(c))
            max_c = c[k]
            current_score = 2.0 * n * max_c
            
            if current_score < best_score - 1e-13:
                best_score = current_score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            grad = compute_gradient_single(a, k)
            v = momentum * v - lr * grad
            a = simplex_project(a + v)
            
            if stagnant > 1000:
                lr = max(0.002, lr * 0.9)
                if lr < 0.002:
                    a = best_a + rng.normal(0, 0.01, n)
                    a = np.clip(a, 0, None)
                    a = a / a.sum()
                    v = np.zeros(n)
                    lr = 0.01
                    stagnant = 0
    
    final_score = score_of(best_a, n)
    print(f"FINAL: n={n}, score={final_score:.6f}", flush=True)
    return [float(x) for x in best_a]
