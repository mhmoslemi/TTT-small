import numpy as np
import time
import math

"""
Algorithm summary
-----------------
We minimize  f(a) = 2*n*max_k (a*a)[k] / (sum a)^2  over non-negative sequences.
The objective is scale-invariant, so we work internally with unit-sum vectors and
return a unit-sum list (the score is unchanged by rescaling).

Key constructions / ideas
  * A wide, smooth, symmetric (Gaussian-like) shape already reduces the
    peak/average of the self-convolution from 2.0 (flat) to ~1.5, because
    it spreads mass so the convolution is flatter.  We sweep a few Gaussian /
    cosine-eased widths to obtain a strong starting point.
  * Local refinement via a subgradient method for max_k c[k] / s^2:
        direction_i = (h_i * s - 2*max_c) / s^3,
    where h_i is the average, over the (top) active peaks k with c[k] near the
    maximum, of a[k-i].  We use Nesterov-style momentum, periodic symmetrization
    and a reject-and-halve learning-rate rule for stability.
  * Several restarts from different structured inits protect against local minima.
Convolution is done with FFT (O(n log n)).  The authoritative evaluator
(evaluate_sequence) is used for the final sanity check on the returned sequence.
"""


def fft_convolve(a):
    """Full self-convolution of a (length n) via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    # remove tiny negative round-off without altering the maximum
    if c.min() < 0.0:
        c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    """2*n*max(self-conv) / sum^2  (== evaluate_sequence for valid input)."""
    n = len(a)
    s = float(a.sum())
    if n == 0 or s < 0.01:
        return float("inf")
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def active_peak_gradient(a, peak_tol=0.01, max_peaks=25):
    """
    h_i  =  mean over active peaks k  of  a[k-i]      (valid indices only)
    (h, max_c, c)
    This is (1/|A|) * subgradient of max_k c[k].
    Vectorized per active peak; O(n * |A|) total.
    """
    n = len(a)
    c = fft_convolve(a)
    max_c = float(c.max())
    if max_c <= 0.0:
        return np.zeros(n, dtype=np.float64), max_c, c
    active = np.where(c >= max_c - peak_tol * max_c)[0]
    if active.size > max_peaks:
        order = np.argsort(c[active])[::-1][:max_peaks]
        active = active[order]
    h = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # h[lo:hi+1] += reversed slice of a
            h[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    h *= 1.0 / max(1, active.size)
    return h, max_c, c


def subgradient_optimize(a_init, deadline, peak_tol=0.01, max_peaks=25,
                         label="", verbose=False):
    """Momentum subgradient descent with reject-and-halve rule + symmetrization."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-12:
        a = np.ones_like(a)
    a = normalize(a)

    best = a.copy()
    best_score = compute_score(best)
    n = a.size

    lr = 0.5 / max(1.0, float(np.sqrt(n)))
    v = np.zeros(n, dtype=np.float64)
    beta = 0.9
    no_improve = 0
    max_no_improve = 250
    it = 0

    while time.monotonic() < deadline:
        it += 1
        s = float(a.sum())
        if s < 1e-12:
            break
        h, max_c, _ = active_peak_gradient(a, peak_tol, max_peaks)
        direction = (h * s - 2.0 * max_c) / (s * s * s)

        v = beta * v + (1.0 - beta) * direction
        cand = a - lr * v
        cand = np.clip(cand, 0.0, 1000.0)
        cand = normalize(cand)

        score = compute_score(cand)
        if score < best_score - 1e-12:
            best = cand
            best_score = score
            no_improve = 0
            a = cand
            if verbose and (it % 500 == 0 or best_score < 2.0 - 1e-9):
                print("  %s it=%d score=%.6f" % (label, it, best_score))
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            lr *= 0.5
            beta = max(0.4, beta * 0.93)
            no_improve = 0
            peak_tol = max(0.002, peak_tol * 0.75)
            if lr < 1e-9:
                break

        if it % 80 == 0:
            a = 0.5 * (a + a[::-1])
            a = normalize(np.clip(a, 0.0, 1000.0))

    return best, best_score


def make_structured_inits(n, rng):

    c = (n - 1) / 2.0
    inits = []

    return [np.clip(i, 0.0, 1000.0) for i in inits]
