"""
Minimizes 2*n*max(f*f)/(sum(f))^2 using:
1. Projected subgradient descent from diverse non-uniform starting points
2. LP-based iteration: solve max sum(g) s.t. (f*g)(k)<=M, g>=0, then blend
3. Adaptive block-perturbation local search
All operations use numpy vectorization; LP uses scipy HiGHS via sparse matrices.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def project_nonneg_sum(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} via clip-and-scale."""
    v = np.asarray(v, dtype=np.float64)
    v = np.clip(v, 0.0, None)
    s = np.sum(v)
    if s < 1e-15:
        return np.full(v.shape, target_sum / len(v))
    return v * (target_sum / s)


def compute_score_fast(f):
    """Compute 2*n*max(f*f)/(sum(f))^2 without calling evaluate_sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(f))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_sparse(f, top_k=None):
    """Build sparse convolution matrix A[k,i] = f[k-i] if 0<=k-i<n.
    If top_k given, only keep rows near the peak region."""
    n = len(f)
    # Each column i has nonzeros at rows i..i+n-1 with values f[0]..f[n-1]
    # Total: n columns × n nonzeros each = n^2 nonzeros
    rows = []
    cols = []
    vals = []
    for i in range(n):
        r = np.arange(i, i + n)
        rows.append(r)
        cols.append(np.full(n, i))
        vals.append(f)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_for_g(f, M, time_limit=45.0):
    """Solve max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns optimal g vector or None if failed."""
    n = len(f)
    A = build_conv_sparse(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def subgradient_descent(f, S, max_time, seed=0):
    """Projected subgradient descent minimizing max(f*f) with sum(f)=S fixed.
    Returns (best_f, best_M)."""
    f = f.astype(np.float64).copy()
    S = float(np.sum(f))
    n = len(f)
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()

    best_f = f.copy()
    best_M = compute_score_fast(f)  # use score as proxy (scale-invariant)
    eta = S / n * 0.5
    decay = 0.99995
    it = 0
    stagnation = 0

    while time.monotonic() - t_start < max_time:
        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))
        M_cur = conv[k_star]

        score_cur = 2.0 * n * M_cur / (S * S)
        if score_cur < best_M - 1e-12:
            best_M = score_cur
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Subgradient of (f*f)(k_w)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * f[k_star - idx]

        # Step
        f = f - eta * grad
        f = project_nonneg_sum(f, S)

        # Adaptive step size with occasional restart
        eta *= decay
        if stagnation > 2000:
            # Add small random perturbation to escape

            stagnation = 0

        # Occasionally try a larger step

        it += 1

        if it % 50000 == 0 and it > 0:
            elapsed = time.monotonic() - t_start
            print(f"      SGD it={it}, elapsed={elapsed:.1f}s, best_M={best_M:.6f}")

    return best_f, best_M

    """LP-based iteration: repeatedly solve for optimal g and blend with f.
    Returns (best_f, best_score)."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = float(np.sum(f))
    best_f = f.copy()

    it = 0
    stall = 0

    print(f"    LP phase start: n={n}, score={best_score:.6f}")

    while time.monotonic() - t_start < max_time:
        conv = np.convolve(f, f)
        M = float(np.max(conv))

        t_lp0 = time.monotonic()

        if g is None:
            print(f"    LP solve failed at iter {it}")
            break

        sum_g = float(np.sum(g))
        if sum_g <= S * (1 + 1e-8):
            print(f"    LP converged: sum_g={sum_g:.2f} <= S={S:.2f}")
            break

        # Rescale g to have same sum as f
        g_r = g * (S / sum_g)

        # Try multiple blend parameters
        improved = False
        best_t = 0.0
        best_s = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = compute_score_fast(new_f)
            if ns < best_s:
                best_s = ns
                best_t = t

        if best_t > 0.0:
            f = (1.0 - best_t) * f + best_t * g_r
            f = np.clip(f, 0.0, 1000.0)
            f = project_nonneg_sum(f, S)
            cur_score = compute_score_fast(f)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                stall = 0
            else:
                f = best_f.copy()
                stall += 1
            it += 1

            stall += 1
            if stall > 5:
                break
            # Try a smaller perturbation to f before next LP
            rng2 = np.random.default_rng(seed + it)
            f = project_nonneg_sum(f + rng2.standard_normal(n) * 0.001 * S / n, S)

    return best_f, best_score


def block_local_search(f, max_time, seed=42):
    """Adaptive block multiplicative perturbation local search."""
    n = len(f)
    S = float(np.sum(f))
    f = f.astype(np.float64).copy()
    best_score = compute_score_fast(f)
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    count = 0
    noise_scale = 0.05
    no_improve = 0

    while time.monotonic() - t_start < max_time:
        # Adaptive block width
        width = int(rng.integers(max(2, n // 1000), max(3, n // 50)))
        start = int(rng.integers(0, max(1, n - width)))

        candidate = f.copy()
        noise = np.exp(rng.normal(0.0, noise_scale, width))
        candidate[start:start + width] *= noise
        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = project_nonneg_sum(candidate, S)

        score = compute_score_fast(candidate)
        count += 1

        if score < best_score - 1e-12:
            f = candidate
            best_score = score
            no_improve = 0
            noise_scale = min(0.15, noise_scale * 1.02)
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                noise_scale = max(0.01, noise_scale * 0.95)
            if no_improve > 5000:
                # Global restart perturbation
                wide = int(rng.integers(n // 10, n // 3))
                wstart = int(rng.integers(0, n - wide))
                f[wstart:wstart + wide] *= np.exp(rng.normal(0, 0.1, wide))
                f = project_nonneg_sum(f, S)
                best_score = compute_score_fast(f)
                no_improve = 0

        if count % 10000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      LS it={count}, elapsed={elapsed:.1f}s, score={best_score:.6f}")

    return f, best_score
