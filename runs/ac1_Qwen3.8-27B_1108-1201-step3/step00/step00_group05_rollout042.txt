"""
Minimizes 2n*max(autocorrelation)/(sum)^2 for non-negative sequences.
Uses soft-max smoothed gradient descent with FFT-based convolution,
adaptive annealing, and multiple restarts.
"""

import numpy as np
import time

def _fft_convolve(x, y):
    """Fast linear convolution using real FFT."""
    size = len(x) + len(y) - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]

def _score(a):
    """Compute the objective value (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = _fft_convolve(a, a)
    return 2.0 * n * b.max() / (s * s)

def _soft_descent(a, deadline, t_init, verbose=False):
    """Run soft-max gradient descent on sequence a."""
    n = len(a)
    K = 2 * n - 1
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    t = t_init
    eta = 2.0 / n  # initial step size
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = _fft_convolve(a, a)
        bm = b.max()
        if bm < 1e-18:
            break
        
        # Soft weights (normalized by max for stability)
        w = np.exp(t * (b - bm))
        
        # Gradient: grad[m] = (2t/K) * sum_j a[j]*w[j+m]
        # = (2t/K) * convolve(a_rev, w)[m + n - 1]
        conv = _fft_convolve(a[::-1], w)
        grad = (2.0 * t / K) * conv[n - 1: 2 * n - 1]
        
        # Center to preserve sum
        grad -= grad.mean()
        
        # Gradient norm for normalization
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        
        # Backtracking line search
        alpha = eta
        improved = False
        for _ in range(8):
            a_try = a - alpha * grad / gn
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / st
            sc_try = _score(a_try)
            if sc_try < best_sc:
                a = a_try
                best_sc = sc_try
                best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Restart with perturbation
                a = best_a + 0.05 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                t = t_init
                eta = 2.0 / n
                stall = 0
        
        # Anneal
        if it % 100 == 0:
            t *= 1.3
            eta *= 0.95
            if verbose and it % 500 == 0:
                print(f"  iter {it}: score={best_sc:.6f} t={t:.1f}")
    
    return best_a, best_sc

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _fft_convolve(x, y):
    size = len(x) + len(y) - 1
    fs = _next_pow2(size)
    return np.fft.irfft(np.fft.rfft(x, fs) * np.fft.rfft(y, fs), fs)[:size]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    bm = _fft_convolve(a, a).max()
    return 2.0 * n * bm / (s * s)
