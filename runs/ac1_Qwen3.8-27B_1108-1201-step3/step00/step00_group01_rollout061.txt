import time

def solve_lp_direction(f, M):
    """Solve LP to find direction g that keeps (f*g) <= M."""
    n = len(f)
    g = cp.Variable(n)
    # Constraints: (f*g)[k] <= M for all k
    constraints = [g >= 0]
    for k in range(2*n - 1):
        # (f*g)[k] = sum_i f[i] * g[k-i] for valid i
        # = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i]
        # Let j = k-i, so i = k-j
        # j ranges from max(0, k-n+1) to min(k, n-1)
        expr = 0
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            i = k - j
            expr += f[i] * g[j]
        constraints.append(expr <= M)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or CLARABEL, SCS
    return g.value

def solve_lp(f, active_lags, M):
    n = len(f)
    K = len(active_lags)
    
    # Build constraint matrix A (K x n)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)

        A[idx, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    
    b = np.full(K, M)
    c = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

"""
LP-guided optimization for step function with minimized peak auto-correlation ratio.
Algorithm: Iteratively solve active-set LPs to find descent directions,
with line search and random restarts to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _active_lags(c, frac=0.85, max_k=100):
    """Find indices where auto-correlation is close to its maximum."""
    M = float(c.max())
    thr = frac * M
    idx = np.where(c >= thr)[0]
    if len(idx) > max_k:
        order = np.argsort(c[idx])[::-1]
        idx = idx[order[:max_k]]
    return idx, M


def _lp_solve(f, active, M):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in active, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    K = len(active)
    # Build constraint matrix
    A = np.zeros((K, n))
    for idx, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[idx, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    
    b = np.full(K, M)
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, 
                      method='highs', options={"presolve": True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

2. Solve LP, get direction g
3. Line search: find best t
4. Update f

6. If any inactive lag now has c[k] > M_old (or close to the new max), add it to the active set
7. Repeat

"""
Step function optimization via active-set LP direction finding.
Minimizes 2n*max(f*f)/(sum f)^2 using iterative LPs to find descent directions.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _get_active(c, max_k=100, frac=0.9):
    """Get active lags: top max_k by value, or those above frac*max."""
    M = c.max()
    idx = np.argsort(c)[::-1][:max_k]
    # Also ensure we include all lags above threshold
    thr = frac * M
    extra = np.where(c >= thr)[0]
    active = np.unique(np.concatenate([idx, extra]))
    if len(active) > max_k + 50:
        # Keep only top max_k+50
        active = active[np.argsort(c[active])[::-1]][:max_k + 50]
    return active, M


def _lp_dir(f, active, M):
    """Solve LP: max sum(g) s.t. (f*g)[k]<=M for k in active, g>=0."""
    n = len(f)
    K = len(active)
    A = np.zeros((K, n))
    for ii, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[ii, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    b = np.full(K, M)

        r = linprog(co, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if r.success:
            return np.maximum(r.x, 0)
    except:
        pass
    return None
