"""
Multi-start LP-guided optimization for minimizing 2*n*max(f*f)/(sum(f))^2.
Uses sparse LP with top-K convolution constraints, diverse initializers,
and adaptive step sizes to escape the uniform (score=2) fixed point.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_score(f):
    """Compute 2*n*max(convolve(f,f))/(sum(f))^2 quickly."""
    n = len(f)
    if n == 0:
        return np.inf
    s = f.sum()
    if s < 0.01:
        return np.inf
    h = np.convolve(f, f)
    return float(2.0 * n * h.max() / (s * s))


def build_A(f, k_positions):
    """Build sparse matrix: A[i, j] = f[k_i - j] for valid j."""
    n = len(f)
    n_c = len(k_positions)
    all_rows, all_cols, all_vals = [], [], []
    for i, k in enumerate(k_positions):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            js = np.arange(lo, hi + 1, dtype=np.int32)
            all_rows.append(np.full(len(js), i, dtype=np.int32))
            all_cols.append(js)
            all_vals.append(f[k - js].astype(np.float64))
    if not all_rows:
        return csr_matrix((n_c, n))
    return csr_matrix(
        (np.concatenate(all_vals),
         (np.concatenate(all_rows), np.concatenate(all_cols))),
        shape=(n_c, n))


def lp_direction(f, M, k_pos, tlimit=8):
    """LP: max sum(g) s.t. A(g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_A(f, k_pos)
    c = -np.ones(n)
    b = np.full(len(k_pos), M)
    try:
        r = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                    method='highs',
                    options={'time_limit': tlimit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.clip(r.x, 0.0, 1e6)
            if g.sum() > 1e-12:
                return g / g.sum()
    except Exception:
        pass
    return None


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-12:
        return np.ones(n) / n
    v = v.copy()
    lam = 0.0
    for _ in range(50):
        u = np.maximum(v - lam, 0.0)
        s = u.sum()
        if abs(s - 1.0) < 1e-12:
            break
        lam += (s - 1.0) / np.count_nonzero(u > 0)
    return np.maximum(v - lam, 0.0)


def subgradient_step(f, eta):
    """One subgradient step on the peak of f*f, projected to simplex."""
    h = np.convolve(f, f)
    k_star = int(np.argmax(h))
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    js = np.arange(lo, hi + 1)
    grad[js] = 2.0 * f[k_star - js]
    f_new = f - eta * grad
    return simplex_project(f_new)


def local_optimize(f, rng, deadline, K=30, use_subgrad=False):
    """LP-guided local optimization from starting point f."""
    f = np.clip(f.copy().astype(np.float64), 0.0, 1e6)
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    best = f.copy()
    best_s = fast_score(f)
    step = 0
    no_imp = 0
    K_eff = K

    while time.monotonic() < deadline:
        step += 1
        t0 = time.monotonic()

        h = np.convolve(f, f)
        M = float(h.max())
        tk = np.sort(np.argsort(h)[-K_eff:])

        g = lp_direction(f, M, tk, tlimit=6)
        improved = False
        if g is not None:
            for t in [0.5, 0.3, 0.15, 0.05]:
                fn = np.clip((1.0 - t) * f + t * g, 0.0, 1e6)
                sn = fast_score(fn)
                if sn < best_s:
                    best, best_s, f = fn, sn, fn
                    no_imp = 0
                    improved = True
                    break
        if not improved:
            # Try subgradient as alternative
            if use_subgrad or no_imp > 3:
                eta = 0.01 / max(1, step // 50)
                fn = subgradient_step(f, eta)
                sn = fast_score(fn)
                if sn < best_s:
                    best, best_s, f = fn, sn, fn
                    no_imp = 0
                    improved = True
        if not improved:
            no_imp += 1

        # Perturbation to escape local minima
        if no_imp >= 12:
            sigma = min(0.4, 0.15 + 0.02 * no_imp)
            f = np.clip(f * np.exp(rng.normal(0.0, sigma, f.size)), 0.01, 1e6)
            f = f / f.sum()
            no_imp = 0

        # Adapt K
        if no_imp >= 30:
            K_eff = min(K_eff + 10, 80) if K_eff < 60 else max(15, K_eff - 15)
            no_imp = 0

        # Reduce K if iteration too slow
        if time.monotonic() - t0 > 30 and K_eff > 15:
            K_eff = max(15, K_eff - 5)

    return best, best_s
