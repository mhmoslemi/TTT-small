"""
Minimizes 2n*max(a*a)/(sum(a))^2 using:
1. Gradient descent on log-sum-exp approximation with softmax parametrization
2. Beta annealing for progressively sharper max approximation
3. Multiple diverse starting points
4. Exact-evaluation local search for fine-tuning
"""

def softmax(y):
    y = y - np.max(y)
    e = np.exp(y)
    return e / np.sum(e)

def compute_score_fast(a):
    n = len(a)
    c = np.convolve(a, a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def gradient_descent_softmax(y, beta_schedule, step_schedule, deadline, best_a, best_score, label, n_report=200):
    """
    Gradient descent on logsumexp(beta * conv(x,x)) where x = softmax(y).
    Returns updated y, best_a, best_score.
    """
    n = len(y)
    it = 0
    while time.monotonic() < deadline:
        x = softmax(y)
        
        c = np.convolve(x, x)
        beta = beta_schedule(it, n)
        step = step_schedule(it, n)
        
        mc = beta * c
        mc_max = np.max(mc)
        exp_vals = np.exp(mc - mc_max)
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient wrt x
        gx = 2.0 * beta * np.correlate(p, x, mode='valid')
        
        # Gradient wrt y (softmax reparametrization)
        gy = x * (gx - np.dot(gx, x))
        
        # Step
        y = y - step * gy
        
        it += 1
        if it % n_report == 0:
            score = compute_score_fast(x)
            if score < best_score * 1.01:
                exact = float(evaluate_sequence(x.tolist()))
                if exact < best_score:
                    best_a[:] = x
                    best_score = exact
                    print(f"  {label} it={it} beta={beta:.0f}: score={best_score:.6f}")
    
    # Final check
    x = softmax(y)
    score = compute_score_fast(x)

        if exact < best_score:

            best_score = exact
