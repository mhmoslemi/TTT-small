def fast_score_array(a):
    n=a.size
    if n==0: return inf
    if not np.all(np.isfinite(a)): return inf
    a=np.clip(a,0,1000)

    b=a/s
    need=2*n-1
    N=1

    c=np.fft.irfft(F*F,N)[:need]
    maxc=np.max(c)
    if maxc <0: maxc=0.0

```python
"""
Search for low-score non-negative step functions.

The evaluator minimizes 2*n*max(convolve(a,a))/(sum(a))^2.  We exploit
scale invariance and work with sequences normalized to sum 1.  A strong
deterministic construction is a left-heavy linear ramp
[a_0 > a_1 > ... > a_{n-1}=0], whose score is asymptotically 4/3.
The search briefly compares FFT-scored ramp families (power ramps, affine
ramps, exponential ramps) and exact-evaluates a small set of finalists.
The guaranteed fallback is a normalized linear decreasing ramp.
"""

import numpy as np
import time


def fallback_sequence(n=4096):
    """Normalized left-heavy linear ramp: [n-1, n-2, ..., 0]."""
    n = int(max(2, n))
    x = np.arange(n, dtype=np.float64)
    a = (float(n - 1) - x)
    s = np.sum(a)
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a / s
    return a


def make_power_ramp(n, power):
    """Normalized ramp proportional to (1 - x)^power, x in [0, 1]."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if power <= 0.0:
        base = np.ones(n, dtype=np.float64)
    else:
        t = np.maximum(0.0, 1.0 - x)
        base = np.power(t, float(power))

    base = np.nan_to_num(base, nan=0.0, posinf=1000.0, neginf=0.0)
    base = np.clip(base, 0.0, 1000.0)
    s = np.sum(base)
    if s <= 1e-15:
        base = np.ones(n, dtype=np.float64)
        s = float(n)
    return base / s


def make_affine_ramp(n, ratio):
    """Normalized ramp proportional to max(0, 1 - ratio*x)."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    base = 1.0 - float(ratio) * x
    base = np.maximum(base, 0.0)
    base = np.nan_to_num(base, nan=0.0, posinf=1000.0, neginf=0.0)
    base = np.clip(base, 0.0, 1000.0)

    s = np.sum(base)
    if s <= 1e-15:
        base = np.ones(n, dtype=np.float64)
        s = float(n)
    return base / s


def make_exp_ramp(n, alpha):
    """Normalized exponential ramp proportional to exp(-alpha*x)."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    base = np.exp(-float(alpha) * x)
    base = np.maximum(base, 0.0)
    base = np.nan_to_num(base, nan=0.0, posinf=1000.0, neginf=0.0)
    base = np.clip(base, 0.0, 1000.0)

    s = np.sum(base)
    if s <= 1e-15:
        base = np.ones(n, dtype=np.float64)
        s = float(n)
    return base / s


def fast_score(a):
    """
    Fast score using FFT convolution.

    The objective is scale-invariant, so normalize to sum = 1 first.
    For normalized a, score = 2*n*max(convolve(a,a)).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(a)):
        return np.inf

    a = np.clip(a, 0.0, 1000.0)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf

    b = a / s
    need = 2 * n - 1

    if n <= 300:
        c = np.convolve(b, b, mode="full")
        max_c = float(np.max(c))
    else:
        N = 1
        while N < need:
            N <<= 1
        F = np.fft.rfft(b, N)
        c = np.fft.irfft(F * F, N)[:need]
        max_c = float(np.max(c))

    if max_c < 0.0:
        max_c = 0.0
    return float(2.0 * n * max_c)


def add_candidate(candidates, a):
    """Add a candidate if its FFT score is finite."""
    sc = fast_score(a)
    if np.isfinite(sc):
        candidates.append((sc, a))


def exact_score(a):
    """Score using the authoritative global evaluate_sequence function."""
    try:
        vals = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf

                if np.isfinite(ex) and ex < best_score:
                    best_score = ex
                    best_arr = arr.copy()
                top_count += 2

    # Final safety cleanup.
    final_arr = np.asarray(best_arr, dtype=np.float64).ravel()
    if final_arr.size == 0:
        final_arr = fallback

    final_arr = np.nan_to_num(final_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    final_arr = np.clip(final_arr, 0.0, 1000.0)

    if final_arr.size == 0 or np.sum(final_arr) < 0.01:
        final_arr = fallback

    if np.isfinite(best_score):
        print("final exact score:", best_score)
    else:
        fs = fast_score(final_arr)
        print("final FFT score (exact not run):", fs)
    print("final length:", final_arr.size)

    return [float(x) for x in final_arr]
```
