import numpy as np
import time


def _softmax(theta):
    """Convert a log-vector to a strictly positive simplex vector."""
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(theta)):
        theta = np.where(np.isfinite(theta), theta, 0.0)
    t = theta - float(np.max(theta))
    t = np.clip(t, -40.0, 0.0)
    e = np.exp(t)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        e = np.zeros_like(theta)
        e[-1] = 1.0
        s = 1.0
    return e / s


def _normalize(a):
    """Clip to a valid non-negative vector and normalize sum to one."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def _conv_full(a):
    """Full linear self-convolution.  Uses FFT for larger n."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n <= 96:
        c = np.convolve(a, a, mode="full")
    else:
        full_len = n + n - 1
        L = 1 << (full_len - 1).bit_length()
        fa = np.fft.rfft(a, n=L)
        c = np.fft.irfft(fa * fa, n=L)[:full_len].real
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.where(c < 0.0, 0.0, c)
    return c


def _score_from_R(R, a):
    """Evaluate the objective from a precomputed convolution R."""
    n = a.size
    s = float(np.sum(a))
    if n <= 0 or not np.isfinite(s) or s <= 1e-12:
        return np.inf
    if R.size == 0:
        return np.inf
    m = float(np.max(R))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def _select_top_weights(R, K, tau_rel):
    """Choose a weighted active set of near-maximum convolution entries."""
    R = np.asarray(R, dtype=np.float64)
    m = R.size
    K = int(K)
    if K <= 0:
        K = 1
    if K >= m:
        idx = np.arange(m, dtype=np.int64)
        vals = R
    else:
        if K == 1:
            idx = np.array([int(np.argmax(R))], dtype=np.int64)
            vals = R[idx]
        else:
            idx = np.argpartition(R, m - K)[-K:]
            vals = R[idx]
    rmax = float(np.max(vals))
    denom = max(float(tau_rel) * rmax, 1e-300)
    z = (vals - rmax) / denom
    z = np.clip(z, -30.0, 0.0)
    w = np.exp(z)
    ws = float(np.sum(w))
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros_like(vals)
        w[int(np.argmax(vals))] = 1.0
    else:
        w = w / ws
    return idx, w


def _active_gradient(a, idx, weights):
    """
    Gradient of sum_k weights[k] * (a*a)[k] with respect to a.

    For a fixed lag k, d/d a_i = 2*a_{k-i} whenever 0 <= k-i < n.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for k, w in zip(idx, weights):
        if w <= 0.0:
            continue
        k = int(k)
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if hi < lo:
            continue
        min_i = k - hi
        max_i = k - lo
        if max_i >= min_i:
            if lo == 0:
                segment = a[hi::-1]
            else:
                segment = a[hi:lo - 1:-1]
            g[min_i:max_i + 1] += 2.0 * float(w) * segment
    return g


def _init_step(n, split, hi, lo):
    """Two-level step initialization."""
    a = np.full(max(1, int(n)), float(lo), dtype=np.float64)
    n = int(a.size)
    if n == 1:
        a[:] = 1.0
        return _normalize(a)
    k = int(round(n * float(split)))
    k = min(max(1, k), n - 1)
    a[:k] = float(hi)
    a[k:] = float(lo)
    return _normalize(a)


def _init_levels(n, fractions, values):
    """Piecewise constant initialization."""
    n = max(1, int(n))
    a = np.zeros(n, dtype=np.float64)
    start = 0
    for frac, val in zip(fractions, values):

        val = float(val)

        if end <= start:
            end = start + 1
        if end > n:
            end = n
        a[start:end] = val
        start = end
    if start < n:
        a[start:] = float(values[-1])
    return _normalize(a)


def _make_sine_fixed(n, variant):
    """Smooth positive Fourier-mode initializations."""
    n = max(1, int(n))
    x = np.arange(n, dtype=np.float64) / float(n)
    tpi = 2.0 * np.pi
    if variant == 0:
        theta = 0.8 * np.sin(tpi * 3.0 * x) + 0.4 * np.sin(tpi * 2.0 * x)
    elif variant == 1:
        theta = 0.8 * np.sin(tpi * 3.0 * x + 0.9) + 0.5 * np.sin(tpi * 4.0 * x)
    elif variant == 2:
        theta = 0.7 * np.sin(tpi * 2.0 * x - 0.5) + 0.7 * np.sin(tpi * 5.0 * x + 0.2)
    elif variant == 3:
        theta = 0.6 * np.sin(tpi * 3.0 * x + 1.3) + 0.4 * np.sin(tpi * 6.0 * x)
    else:
        theta = np.zeros(n, dtype=np.float64)
    return _softmax(theta)


def _make_sine_random(n, rng, amp=0.75):
    """Random smooth Fourier initialization."""
    n = max(1, int(n))
    x = np.arange(n, dtype=np.float64) / float(n)
    tpi = 2.0 * np.pi
    theta = np.zeros(n, dtype=np.float64)
    for _ in range(3):
        m = int(rng.integers(2, 9))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        coeff = float(rng.normal(0.0, max(0.0, amp) / 2.0))
        theta += coeff * np.sin(tpi * float(m) * x + phase)
    return _softmax(theta)


def _resample_sequence(seq, n):
    """Linearly resample a valid sequence to a new length, preserving shape."""
    n = max(1, int(n))
    seq = np.asarray(seq, dtype=np.float64).ravel()
    if seq.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(np.sum(seq))
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    seq = seq / s
    m = seq.size
    if m == n:
        return seq.copy()
    x_old = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    x_new = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.interp(x_new, x_old, seq)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    ss = float(np.sum(a))
    if not np.isfinite(ss) or ss <= 1e-12:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / ss
    return a


def _random_perturb_log(n, rng, amp):
    """Random smooth/noisy perturbation in log-space."""
    n = max(1, int(n))
    if amp <= 0.0:
        return np.zeros(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n)
    tpi = 2.0 * np.pi
    out = np.zeros(n, dtype=np.float64)
    for _ in range(3):
        m = int(rng.integers(1, 9))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        coeff = float(rng.normal(0.0, amp / 3.0))
        out += coeff * np.sin(tpi * float(m) * x + phase)
    if rng.random() < 0.5:
        w = int(rng.integers(2, max(3, n // 8)))
        if w < n:
            st = int(rng.integers(0, n - w))
            out[st:st + w] += float(rng.normal(0.0, amp / 4.0))
    return out


def _optimizer(n, a0, time_limit, rng):
    """
    Natural-gradient active-set search in log-space.

    Minimizes the maximum coefficient of the self-convolution under a_i >= 0
    and sum(a_i)=1.  The objective returned is exactly 2*n*max(a*a).
    """

    time_limit = max(0.0, float(time_limit))
    a0 = _normalize(a0)
    if a0.size != n:
        a0 = _resample_sequence(a0, n)

        R = _conv_full(a0)
        return a0, _score_from_R(R, a0)

    theta = np.log(a0 + 1e-300)
    theta = theta - float(np.max(theta))
    cur_a = a0.copy()
    cur_theta = theta.copy()
    R = _conv_full(cur_a)
    cur_score = _score_from_R(R, cur_a)

    best_a = cur_a.copy()
    best_score = cur_score
    best_theta = theta.copy()
    if not np.isfinite(best_score):
        best_score = np.inf

    if n <= 64:
        k_base = 8
    elif n <= 128:
        k_base = 12
    elif n <= 256:
        k_base = 18
    elif n <= 512:
        k_base = 28
    else:
        k_base = 24

    eta = 0.08
    tau_rel = 0.45
    stall = 0
    amp = 0.05
    it = 0

    start = time.monotonic()
    deadline = start + time_limit

    while time.monotonic() < deadline:
        if not np.isfinite(cur_score) or not np.all(np.isfinite(cur_a)):
            cur_a = best_a
            cur_theta = best_theta
            cur_score = best_score

        if np.isfinite(best_score) and np.isfinite(cur_score):
            if cur_score > best_score * 1.25 + 1e-8:
                cur_a = best_a
                cur_theta = best_theta
                cur_score = best_score

        R = _conv_full(cur_a)
        score = _score_from_R(R, cur_a)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = cur_a.copy()
            best_theta = cur_theta.copy()

        it += 1

        if it < 600:
            k_use = k_base
        elif it < 2500:
            k_use = max(8, k_base // 2)
        else:
            k_use = max(4, k_base // 4)

        tau_rel = max(0.06, tau_rel * 0.9995)

        idx, weights = _select_top_weights(R, k_use, tau_rel)
        g = _active_gradient(cur_a, idx, weights)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        G = float(np.dot(cur_a, g))
        d = G - g
        scale = float(np.max(np.abs(d)))

        accepted = False
        if scale > 1e-18 and np.isfinite(score):
            for mult in (1.0, 0.5, 0.25):
                step = eta * mult
                if step <= 0.0:
                    break
                theta_try = cur_theta + step * (d / scale)
                theta_try = theta_try - float(np.max(theta_try))
                a_try = _softmax(theta_try)
                R_try = _conv_full(a_try)
                score_try = _score_from_R(R_try, a_try)
                if np.isfinite(score_try) and score_try < score * (1.0 + 1e-6) + 1e-12:
                    cur_theta = theta_try
                    cur_a = a_try
                    cur_score = score_try
                    accepted = True
                    break

        if accepted:
            eta = min(0.35, eta * 1.04)
            stall = 0
            if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur_a.copy()
                best_theta = cur_theta.copy()
        else:
            eta = max(0.004, eta * 0.65)
            stall += 1

        if stall > 25:
            if np.isfinite(best_score) and np.isfinite(cur_score):
                if cur_score > best_score * 1.01 + 1e-9:
                    cur_a = best_a
                    cur_theta = best_theta
                    cur_score = best_score

            pert = _random_perturb_log(n, rng, amp)
            theta_try = cur_theta + pert
            theta_try = theta_try - float(np.max(theta_try))
            a_try = _softmax(theta_try)
            R_try = _conv_full(a_try)
            score_try = _score_from_R(R_try, a_try)

            if np.isfinite(score_try):
                if score_try < best_score * 1.10 + 1e-9 or score_try < cur_score:
                    cur_theta = theta_try
                    cur_a = a_try
                    cur_score = score_try
                elif score_try < np.inf and best_score < np.inf and score_try < best_score * 1.45:
                    cur_theta = theta_try
                    cur_a = a_try
                    cur_score = score_try

            eta = 0.06
            tau_rel = min(0.8, tau_rel * 1.2)
            amp = min(0.9, amp * 1.15)
            stall = 0

        if it % 2500 == 0:
            print("opt n", n, "it", it, "best", float(best_score), "cur", float(cur_score) if np.isfinite(cur_score) else np.inf)

    return best_a, float(best_score)


def _optimize_multiple(n, inits, time_limit, rng):
    """Run several initializations for one n, allocating time by initial score."""
    n = max(1, int(n))
    time_limit = max(0.0, float(time_limit))
    if time_limit <= 0.0 or len(inits) == 0:
        return None, np.inf

    scored = []
    for i, a0 in enumerate(inits):
        a0 = _normalize(a0)
        if a0.size != n:
            a0 = _resample_sequence(a0, n)
        R = _conv_full(a0)
        sc = _score_from_R(R, a0)
        if np.isfinite(sc):
            scored.append((float(sc), i, a0))

    if not scored:
        a0 = np.ones(n, dtype=np.float64) / float(n)
        R = _conv_full(a0)
        return a0, _score_from_R(R, a0)

    scored.sort()

    if len(scored) == 1:
        fracs = [1.0]
    elif len(scored) == 2:
        fracs = [0.60, 0.40]
    elif len(scored) == 3:
        fracs = [0.50, 0.30, 0.20]
    else:
        fracs = [0.45, 0.25, 0.18, 0.12]

    start = time.monotonic()
    deadline = start + time_limit
    best_a = None
    best_sc = np.inf

    for j in range(min(len(scored), len(fracs))):
        remaining = deadline - time.monotonic()
        if remaining <= 0.0:
            break
        alloc = min(remaining, float(time_limit) * fracs[j])
        if alloc < 2.0:
            alloc = min(remaining, max(1.0, remaining * 0.5))
        if alloc <= 0.0:
            break

        sc0, _, a0 = scored[j]
        if sc0 < best_sc:
            best_sc = sc0
            best_a = a0

        res_a, res_sc = _optimizer(n, a0, alloc, rng)
        if res_a is not None and np.isfinite(res_sc) and res_sc < best_sc:
            best_sc = float(res_sc)
            best_a = res_a

    if best_a is None:
        best_a = np.ones(n, dtype=np.float64) / float(n)
        best_sc = _score_from_R(_conv_full(best_a), best_a)

    return best_a, float(best_sc)


def _make_inits_for_n(n, rng, previous_best, parent):
    """Build candidate initial sequences for a given length."""
    n = max(1, int(n))
    inits = []

    if previous_best is not None:
        inits.append(_resample_sequence(previous_best, n))

    inits.append(_init_step(n, 0.50, 2.0, 1.0))
    inits.append(_init_levels(n, (1.0 / 3.0, 1.0 / 3.0, 1.0 / 3.0), (1.5, 1.0, 0.5)))
    inits.append(_make_sine_fixed(n, 0))

    if parent is not None:
        inits.append(_resample_sequence(parent, n))

    inits.append(_make_sine_random(n, rng))

    if n <= 512:
        max_inits = 4
    else:
        max_inits = 4

    if len(inits) > max_inits:
        inits = inits[:max_inits]
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-scale natural-gradient / active-set search for low-peak self-convolution.

    The objective is scale-invariant, so each candidate is kept on the simplex.
    The final returned vector is non-negative, finite, and has sum one.
    """
    start = time.monotonic()
    total_budget = max(120.0, float(budget_s) - 25.0)
    deadline = start + total_budget
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = _normalize(p)
        if p.size > 0:
            R = _conv_full(p)
            ps = _score_from_R(R, p)
            if np.isfinite(ps):
                parent = p
    except Exception:
        parent = None

    best_seq = None
    best_score = np.inf

    configs = (
        (64, 0.020),
        (128, 0.055),
        (256, 0.190),
        (512, 0.430),
        (1024, 0.270),
    )

    for n, frac in configs:
        remaining = deadline - time.monotonic()
        if remaining <= 5.0:
            break
        alloc = min(remaining, float(total_budget) * float(frac))
        if alloc < 3.0:
            alloc = min(remaining, max(1.0, remaining * 0.25))
        if alloc <= 0.0:
            break

        inits = _make_inits_for_n(int(n), rng, best_seq, parent)
        try:
            res_a, res_s = _optimize_multiple(int(n), inits, float(alloc), rng)
        except Exception:
            res_a, res_s = None, np.inf

        if res_a is not None and np.isfinite(res_s) and res_s < best_score:
            best_score = float(res_s)
            best_seq = res_a
            print("new best n", int(n), "local score", best_score)

    # Final polish if any time remains.
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 15.0:
        n = int(best_seq.size)
        try:
            res_a, res_s = _optimizer(n, best_seq, min(remaining, 0.04 * total_budget), rng)
            if res_a is not None and np.isfinite(res_s) and res_s < best_score:
                best_score = float(res_s)
                best_seq = res_a
                print("polish best n", n, "local score", best_score)
        except Exception:
            pass

    if best_seq is None:
        best_seq = _init_step(64, 0.5, 2.0, 1.0)
    else:
        best_seq = _normalize(best_seq)

    # Safeguard: if the exact evaluator is available, validate / choose among
    # the optimized best and a few robust fallbacks only when needed.
    try:
        exact_best = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(exact_best):
            raise ValueError("bad exact")
        fallbacks = (
            _init_step(64, 0.5, 2.0, 1.0),
            _init_step(128, 0.45, 2.2, 1.0),
            _init_levels(128, (1.0 / 3.0, 1.0 / 3.0, 1.0 / 3.0), (1.5, 1.0, 0.5)),
            _make_sine_fixed(128, 0),
        )
        saw_finite = False
        for cand in fallbacks:
            cand = _normalize(cand)
            try:
                sc = float(evaluate_sequence([float(x) for x in cand]))
            except Exception:
                sc = np.inf
            if np.isfinite(sc):
                saw_finite = True
                if sc < exact_best:
                    exact_best = sc
                    best_seq = cand
        if not saw_finite:
            pass
        print("final exact/or local score", exact_best)
    except Exception:
        final_local = _score_from_R(_conv_full(best_seq), best_seq)
        print("final local score", float(final_local) if np.isfinite(final_local) else np.inf)

    out = [float(x) for x in best_seq]
    # Final hard safety: replace any unexpected non-finite or negative value.
    fixed = []
    for x in out:
        if not np.isfinite(x) or x < 0.0:
            fixed.append(0.0)
        else:
            fixed.append(float(x))
    if not fixed or float(np.sum(fixed)) < 0.01:
        fixed = [0.5, 0.5]
    else:
        s = float(np.sum(fixed))
        fixed = [x / s for x in fixed]
    return fixed
