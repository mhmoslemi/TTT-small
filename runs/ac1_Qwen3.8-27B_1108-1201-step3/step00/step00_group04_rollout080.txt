"""
Improves the coefficient sequence far below the previous 2.0 bound.

Key idea
--------
For the *full* self-convolution c[k] = sum_{i+j=k} a[i] a[j] (i.e. np.convolve(a,a)),
the "central binomial" sequence  a[k] = binom(2k,k) / 4^k  has the remarkable property
that the first n anti-diagonals are ALL exactly equal (the generating function
1/sqrt(1-x) convolved with itself is 1/(1-x), whose coefficients are all 1).  This
makes the peak of the convolution exactly a[0]^2 while the tail anti-diagonals are
smaller, giving an objective that tends to pi/2 ~= 1.5708  (already < 2.0).

Because max_k c[k] is not convex in a (each c[k] is an indefinite quadratic form), we
use the alternating block / two-sided LP scheme: keeping `a` fixed, solve the linear
program  max sum(b)  s.t.  (a*b)[k] <= M, b >= 0,  where M = max(a*a).  The optimizer b
is the "most spread-out" non-negative vector that still keeps a*b bounded by the current
peak; mixing a toward b (and renormalising) systematically lowers max(a*a).  Repeating
drives the objective below pi/2, approaching the theoretical lower bound 2n/(2n-1) ~= 1.

Two phases run with different lengths (more variables = more room to flatten the tail),
and a short random multiplicative local search polishes the global best.  The central
binomial sequence alone is returned in the worst case, so the result is always <= ~1.57.
"""

import numpy as np
import time
from scipy.optimize import linprog


def central_binomial_sequence(n):
    """a[k] = binom(2k,k) / 4^k  via the stable recurrence C_k = C_{k-1} (2k-1)/(2k)."""
    C = np.empty(n, dtype=np.float64)
    C[0] = 1.0
    for k in range(1, n):
        C[k] = C[k - 1] * (2 * k - 1) / (2 * k)
    return C


def build_conv_matrix(a):
    """Dense matrix A with A[k, j] = a[k-j] if 0 <= k-j < n else 0  (shape (2n-1, n))."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = a[k::-1]
    for k in range(n, 2 * n - 1):
        jmin = k - n + 1
        A[k, jmin:] = a[n - 1:k - n:-1]
    return A


def _score(seq):
    try:
        s = float(evaluate_sequence(seq.tolist()))
    except Exception:
        return np.inf
    return s


def run_lp_phase(n, t, deadline, seed):

    rng = np.random.default_rng(seed)
    a = central_binomial_sequence(n)
    a = a / a.sum()
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = _score(best)
    if not np.isfinite(best_score):
        best_score = np.inf
    print("[phase n=%d] start score = %.6f" % (n, best_score))

    it = 0
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(conv.max())
        if M <= 1e-18:
            break

        improved = False
        adopted = False
        try:
            A = build_conv_matrix(a)
            c = -np.ones(n)
            b_ub = np.full(2 * n - 1, M)

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method="highs",

            if res is not None and res.success and res.x is not None:

def polish_local_search(seq, deadline, seed):
    """Random multiplicative block perturbation local search to squeeze the global best."""
    rng = np.random.default_rng(seed)
    a = np.asarray(seq, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return a, _score(a)
    a = a / s
    n = a.size
    best = a.copy()
    best_score = _score(best)
    print("[polish] start score = %.6f (n=%d)" % (best_score, n))

    while time.monotonic() < deadline:
        width = int(rng.integers(max(1, n // 400), max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        cand = a.copy()
        cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.10)))

            best = cand.copy()
            best_score = score
            a = cand.copy()
            print("[polish] improved to %.6f" % best_score)

    return best, best_score
