    return a_new


def _subgrad_step(a, lr, Nf):
    """One projected subgradient step on max(conv(a,a))."""
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, Nf)
    B = A * A
    b = np.fft.irfft(B, Nf)[:N]
    max_b = b.max()
    if max_b <= 0:
        return a
    thresh = max_b * 0.9
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > 20:
        pv = b[peaks]
        peaks = peaks[np.argsort(pv)[-20:]]
    g = np.zeros(n)
    for k in peaks:
        idx = k - np.arange(n)
        ok = (idx >= 0) & (idx < n)
        g[ok] += 2.0 * a[idx[ok]]
    g /= len(peaks)
    a_new = a - lr * g
    np.maximum(a_new, 0.0, out=a_new)
    s = a_new.sum()
    if s > 0.01:
        a_new /= s
    else:
        a_new.fill(1.0 / n)
    return a_new


def _run_gs_phase(a, best, best_score, clip, rng, Nf, max_steps=5000):
    """Run GS iterations until stuck or max_steps reached."""
    n = len(a)
    steps_done = 0
    local_best = a.copy()
    local_best_score = best_score
    stall = 0
    
    for step in range(max_steps):
        alpha = 0.05
        a = _gs_step(a, clip, alpha, Nf)
        steps_done += 1
        
        # Evaluate every 500 steps
        if (step + 1) % 500 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < local_best_score:
                local_best = a.copy()
                local_best_score = score
                stall = 0
            else:
                stall += 1
            if stall > 4:
                break
    
    # Final evaluation
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < local_best_score:
        local_best = a.copy()
        local_best_score = score
    
    return local_best, local_best_score, steps_done


def _explore(a_best, n, rng, Nf, clip):
    """Generate a new candidate via random perturbation or restart."""
    r = rng.random()
    if r < 0.2:
        # Small multiplicative perturbation
        a = a_best * np.exp(rng.normal(0.0, 0.02, n))
    elif r < 0.4:
        # Small additive perturbation
        a = a_best + rng.normal(0.0, 0.001, n)
        np.maximum(a, 0.0, out=a)
    elif r < 0.55:
        # Partial randomization
        a = a_best.copy()
        m = rng.random(n) < 0.1
        a[m] = rng.uniform(0.0, 4.0 / n, int(m.sum()))
    elif r < 0.7:
        # GS from perturbed start
        a = a_best.copy()
        a[rng.integers(0, n)] *= rng.uniform(0.1, 3.0)
        for _ in range(50):
            a = _gs_step(a, clip, 0.1, Nf)
    elif r < 0.85:
        # Fresh Dirichlet + GS warmup
        a = rng.dirichlet(np.full(n, 0.2))
        for _ in range(200):
            a = _gs_step(a, None, 0.3
