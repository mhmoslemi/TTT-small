import numpy as np
import time

"""
Algorithm: Optimize non-negative sequence a to minimize f(a) = 2n*max(a*a)/(sum(a))^2.
Uses geometric sequence initialization (a_i = r^i with r = exp(c/n)), followed by
projected gradient descent on the convex inner problem (minimize max auto-conv given sum=1),
with simulated annealing and multiple restarts.
"""


def fast_score(a):
    """Compute score directly for speed (no evaluate_sequence call)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def geometric_init(n, c):
    """Create normalized geometric sequence with ratio r = exp(c/n)."""
    a = np.exp((c / n) * np.arange(n))
    a = a / np.sum(a)
    return a


def compute_gradient(a, k_star):
    """Compute gradient of (a*a)[k_star] w.r.t. a. Result: 2*a[k_star - j] with bounds."""
    n = len(a)
    g = np.zeros(n)
    # For each j, contribution is 2*a[k_star - j] if 0 <= k_star - j < n
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        g[indices] = 2.0 * a[k_star - indices]
    # Handle the diagonal case: when k_star = 2*j, the term a_j^2 has derivative 2*a_j
    # This is already handled since a[k_star - j] = a[j] when k_star = 2*j
    # But we need to add the second contribution from the symmetric term
    # Actually, the derivative of (a*a)_k w.r.t. a_j is:
    #   a[k-j] (from i=j term) + a[k-j] (from i=k-j term) = 2*a[k-j] if k != 2j
    #   2*a[j] (from i=j=k-j term, derivative of a_j^2) if k = 2j
    # Both cases give 2*a[k-j], so the formula above is correct.
    return g


def projected_step(a, grad, eta):
    """Project a - eta*grad onto {x >= 0, sum(x) = 1}."""
    v = a - eta * grad
    # Find tau such that sum(max(v - tau, 0)) =
    # using bisection
    vmin = np.min(v)
    vmax = np.max(v)
    lo = vmin - 1.0
    hi = vmax + 1.0
    target = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    a_new = np.maximum(v - tau, 0.0)
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones_like(a) / len(a)
    return a_new


def local_search(a, score, deadline, rng, verbose=True):
    """Iterative local search with gradient steps and random perturbations."""
    best = a.copy()
    best_score = score
    n = len(a)
    no_improve_count = 0
    max_no_improve = 500

    while time.monotonic() < deadline:
        # Compute auto-conv
        b = np.convolve(best, best)
        k_star = int(np.argmax(b))
        max_b = b[k_star]
        sum_a = np.sum(best)

        # Compute gradient
        g = compute_gradient(best, k_star)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            g_norm = 1.0

        # Try multiple step sizes
        improved = False
        for eta_factor in [0.05, 0.01, 0.001, 0.5, 0.1]:
            eta = eta_factor / g_norm
            candidate = projected_step(best, g, eta)
            cand_score = fast_score(candidate)
            if cand_score < best_score - 1e-12:
                best = candidate
                best_score = cand_score
                improved = True
                break

        if not improved:
            # Try random perturbation
            n_perturb = rng.integers(1, max(2, n // 50))
            indices = rng.choice(n, size=n_perturb, replace=False)
            candidate = best.copy()
            multipliers = np.exp(rng.normal(0, 0.3, size=n_perturb))
            candidate[indices] *= multipliers
            candidate = np.maximum(candidate, 0)
            s = np.sum(candidate)
            if s > 0.01:
                candidate = candidate / s
                cand_score = fast_score(candidate)
                if cand_score < best_score - 1e-12:
                    best = candidate
                    best_score = cand_score
                    improved = True

        if not improved:
            # Try gradient step targeting a random position (not just the max)
            k_rand = int(rng.integers(0, 2 * n - 1))
            g_rand = compute_gradient(best, k_rand)
            gr_norm = np.linalg.norm(g_rand)
            if gr_norm > 1e-15:
                eta = 0.02 / gr_norm
                candidate = projected_step(best, g_rand, eta)
                cand_score = fast_score(candidate)
                if cand_score < best_score - 1e-12:
                    best = candidate
                    best_score = cand_score
                    improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                # Restart with perturbation
                candidate = best.copy()
                block_len = int(rng.integers(n // 10, n // 3))
                start = int(rng.integers(0, n - block_len))
                candidate[start:start + block_len] *= np.exp(rng.normal(0, 0.5, size=block_len))
                candidate = np.maximum(candidate, 0)
                s = np.sum(candidate)

                    candidate = candidate / s
                    cand_score = fast_score(candidate)

                        best_score = cand_score
                no_improve_count = 0

            no_improve_count = 0

        if verbose and no_improve_count % 500 == 0 and no_improve_count > 0:
            print(f"  Local search: score={best_score:.6f}, no_improve={no_improve_count}")

    return best, best_score
