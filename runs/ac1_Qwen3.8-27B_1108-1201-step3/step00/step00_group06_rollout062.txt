```python
import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0"""
    n = len(a)
    # The convolution matrix: C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # We need C @ g <= M * ones(2n-1)
    # Objective: maximize sum(g) = minimize -sum(g)
    
    # Build the constraint matrix (2n-1) x n
    # C[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # But we only need C[k,j] where a[k-j] > 0 (sparse)
    
    # For efficiency, let's only include constraints where M is "active"
    # Actually, let's include all constraints but use sparse matrix
    
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n and a[i] > 0:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    C = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # max sum(g) s.t. C@g <= M*ones, g >= 0
    # linprog: min c@g s.t. A_ub@g <= b_ub, g >= 0
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
def fft_convolve(a, b):

    n = len(a) + len(b) - 1
```

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2
using FFT-accelerated subgradient descent with projection, multiple restarts,
and adaptive step sizing.
"""

import numpy as np
import time


def fft_convolve_self(a):
    """Compute a*a (self-convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgradient(a, b, top_frac=0.1):
    """
    Compute subgradient of max(b) w.r.t. a using active-set weighting.
    b = a * a (self-convolution).
    Returns grad of same length as a.
    """
    n = len(a)
    k_active = max(1, int(top_frac * (2 * n - 1)))
    # Get indices of top k_active values in b
    # Use argpartition for efficiency
    if k_active >= 2 * n - 1:
        active_mask = np.ones(2 * n - 1, dtype=bool)
    else:
        idx = np.argpartition(b, -k_active)[-k_active:]
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        active_mask[idx] = True
    
    # Weights: use the b values themselves as weights (normalized)
    w = np.zeros(2 * n - 1)
    w[active_mask] = b[active_mask]
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    
    # grad[i] = sum_k w[k] * a[k-i] (cross-correlation)
    # = (w * a_rev)[i + n - 1]
    a_rev = a[::-1]
    conv = fft_convolve_general(w, a_rev)
    grad = conv[n - 1 : n - 1 + n]
    
    return grad


def fft_convolve_general(a, b):
    """Compute linear convolution of a and b using FFT."""
    n_out = len(a) + len(b) - 1
    fft_size = 1 << (n_out - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n_out]

    """Project a_new onto {x: sum(x) = S, x >= 0}."""
    # Find lambda such that sum(max(0, a_new - lambda)) = S
    if a_new.max() <= 0:
        return np.full_like(a_new, S / len(a_new))
    
    # Bisection
    lo = 0.0
    hi = a_new.max()
    for _ in range(60):

        s = np.sum(np.maximum(0, a_new - mid))
        if s > S:
            lo = mid
        else:
            hi = mid
    lambda_ = (lo + hi) / 2
    result = np.maximum(0, a_new - lambda_)
    # Fix numerical issues
    total = result.sum()
    if total > 0:
        result *= S / total
    return result

    """Compute the score without calling evaluate_sequence (faster)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (S * S)


def run_subgradient(a, S, deadline, rng, verbose=False):
    """Run subgradient descent on a, maintaining sum(a) = S."""
    n = len(a)
    eta = 1.0  # Initial step size
    best_local = a.copy()
    best_local_score = score_of(a)
    no_improve_count = 0
    max_no_improve = 500
    top_frac = 0.05  # Use top 5% of lags
    
    iteration = 0
    while time.monotonic() < deadline:
        b = fft_convolve_self(a)
        M = b.max()
        current_score = 2.0 * n * M / (S * S)
        
        if current_score < best_local_score:
            best_local_score = current_score
            best_local = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Compute subgradient
        grad = compute_subgradient(a, b, top_frac)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        # Scale step size: eta should be relative to the "scale" of the problem
        # a has sum S, so typical a[i] ~ S/n
        # grad has typical magnitude ~ S/n as well
        # We want eta * grad to be a meaningful fraction of a
        step = eta * (S / n) / grad_norm  # This makes eta * grad ~ S/n in magnitude
        
        a_new = a - step * grad
        a = project_simplex_with_sum(a_new, S)

            no_improve_count = 0
```

```python
import numpy as np

def fft_conv_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    M = 1 << (size - 1).bit_length()
    F = np.fft.rfft(a, M)
    return np.fft.irfft(F * F, M)[:size]

def fft_conv(a, b):
    """General linear convolution via FFT."""
    n = len(a) + len(b) - 1
    M = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, M) * np.fft.rfft(b, M), M)[:n]

def subgrad(a, top_k):
    """Compute subgradient of max(a*a) w.r.t. a."""
    n = len(a)
    b = fft_conv_self(a)
    # Use top_k lags
    if top_k >= 2*n-1:
        w = b / b.sum()
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
        w = np.zeros(2*n-1)
        w[idx] = b[idx]
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w[:] = 1.0/(2*n-1)
    # grad = cross-correlation of w and a
    # grad[m] = sum_k w[k] * a[k-m] = (w * a_rev)[m + n - 1]
    a_rev = a[::-1]
    conv = fft_conv(w, a_rev)
    grad = conv[n-1 : 2*n-1]  # length n
    return grad, b

def project_sum(a_new, S):
    """Project onto {x >= 0, sum(x) = S}."""
    if a_new.max() <= 0:
        return np.full(len(a_new), S / len(a_new))
    # Bisection for lambda
    lo, hi = 0.0, a_new.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(0, a_new - mid).sum() > S:
            lo = mid
        else:
            hi = mid
    x = np.maximum(0, a_new - (lo+hi)*0.5)
    t = x.sum()
    if t > 0:

    return x
```

```python
import numpy as np
import time


def fft_conv_self(a):
    """Compute a * a (self-convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    M = 1 << (size - 1).bit_length()
    F = np.fft.rfft(a, M)

def fft_conv(a, b):
    """Compute linear convolution of a and b using FFT."""
    n_out = len(a) + len(b) - 1
    M = 1 << (n_out - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, M) * np.fft.rfft(b, M), M)[:n_out]


def compute_subgradient(a, top_k=None):
    """
    Compute subgradient of max(a*a) w.r.t. a using active-set weighting.
    Returns (grad, b) where b = a*a.
    """
    n = len(a)
    b = fft_conv_self(a)
    if top_k is None or top_k >= 2 * n - 1:
        w = b.copy()
    else:

def project_sum(x, S):

    if x.max() <= 0:

    # Phase 1: Subgradient on initial sequence (40% of time)
    phase_deadline = time.monotonic() + (total_deadline - time.monotonic()) * 0.35
    a_result, local_score = run_subgradient(a_init.copy(), a_init.sum(), phase_deadline, rng, verbose=True)
    if local_score < best_score:
        best_score = local_score
        best_a = a_result.copy()
        S_best = best_a.sum()
    print(f"After Phase 1 (subgrad): score={best_score:.6f}")

    # Phase 2: Local polish (15% of time)
    phase_deadline = time.monotonic() + (total_deadline - time.monotonic()) * 0.35
    a_result, local_score = local_polish(best_a.copy(), best_a.sum(), phase_deadline, rng)
    if local_score < best_score:
        best_score = local_score
        best_a = a_result.copy()
        S_best = best_a.sum()
    print(f"After Phase 2 (polish): score={best_score:.6f}")

    # Phase 3: Multiple restarts with different strategies (35% of time)
    remaining = total_deadline - time.monotonic()
    n_restarts = 6
    restart_budget = remaining / n_restarts

    for r in range(n_restarts):
        rd = time.monotonic() + restart_budget
        if time.monotonic() >= rd:
            break
        n = len(best_a)
        strategy = r % 3

        if strategy == 0:
            # Gaussian perturbation of best
            a_try = best_a + rng.normal(0, 0.08 * best_a.mean(), n)
        elif strategy == 1:
            # Block modulation
            a_try = best_a.copy()
            nb = max(1, n // 8)
            for _ in range(4):
                s = rng.integers(0, n - nb)
                a_try[s:s + nb] *= rng.uniform(0.3, 2.5)
        else:
            # Structured reinitialization (same length)
            a_try = make_structured_init(n, rng)
            a_try *= best_a.sum()

        a_try = np.maximum(0.0, a_try)
        ts = a_try.sum()
        if ts < 0.01:
            continue
        a_try *= S_best / ts

        # Short subgradient run
        a_result, local_score = run_subgradient(a_try.copy(), a_try.sum(),
                                                min(rd, time.monotonic() + restart_budget * 0.7), rng, verbose=False)
        if local_score < best_score:
            best_score = local_score
            best_a = a_result.copy()
            S_best = best_a.sum()
            print(f"  Restart {r} (strategy {strategy}): score={best_score:.6f}")

    # Phase 4: Final polish (remaining time)
    if time.monotonic() < total_deadline - 3:
        a_result, local_score = local_polish(best_a.copy(), best_a.sum(), total_deadline - 5, rng)
        if local_score < best_score:
            best_score = local_score
            best_a = a_result.copy()
            S_best = best_a.sum()
        print(f"After Phase 4 (final polish): score={best_score:.6f}")

    # Final verification with the actual evaluator
    final_score = float(evaluate_sequence(best_a.tolist()))
    if final_score < best_score:
        best_score = final_score
    print(f"FINAL: score={final_score:.6f}, n={len(best_a)}")

    return [float(x) for x in best_a]
```
