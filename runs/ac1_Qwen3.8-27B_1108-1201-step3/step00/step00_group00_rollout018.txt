import time

    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]

    """Compute the evaluation score."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    c = fft_convolve(f, f)
    M = np.max(c)
    return 2.0 * n * M / (S * S)

def compute_grad(f, tau=5.0):
    """Compute gradient of the score using smooth max."""
    n = len(f)
    S = np.sum(f)
    c = fft_convolve(f, f)
    M = np.max(c)
    
    # Softmax weights
    c_rel = c - M
    weights = np.exp(tau * c_rel)
    weights /= np.sum(weights)
    
    # Gradient of smooth max w.r.t. f
    f_rev = f[::-1].copy()

    grad = 2.0 * n * grad_M / (S * S) - 4.0 * n * M / (S ** 3)
    return grad, M, S

    S_new = np.sum(f_new)
    if S_new > 0.01:
        f_new = f_new * S / S_new
    return f_new
