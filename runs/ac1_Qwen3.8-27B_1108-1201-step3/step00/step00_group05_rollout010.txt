import numpy as np
import time

def linear_convolution(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    return conv

def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = linear_convolution(a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a ** 2))

def compute_subgradient(a, k_star):
    """Compute subgradient of max convolution at a, using the active lag k_star."""
    n = len(a)
    g = np.zeros(n)
    # j ranges from max(0, k_star - n + 1) to min(n-1, k_star)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        g[indices] = 2.0 * a[k_star - indices]
        # Fix the diagonal (where k_star = 2*j)
        if k_star % 2 == 0:
            j_diag = k_star // 2
            if j_min <= j_diag <= j_max:
                g[j_diag] = a[j_diag]
    return g

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    best_a = a.copy()
    best_score = compute_score(a)

    while time.monotonic() < deadline:
        conv = linear_convolution(a)
        max_val = np.max(conv)

        if top_k > 1:

            active_lags = np.where(conv >= threshold)[0]
        else:

        g = np.zeros(n)
        for k in active_lags:

        elif rng.random() < np.exp(-(current_score - score) * n / (temp * 10)):

            no_improve_count += 1

            temp = max(temp, 0.01)
            eta = max(eta, 0.005)

def compute_score_fast(a):
    """Compute J(a) = 2*n*max(conv)/(sum)^2. Assumes a is non-negative."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = linear_conv_self(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def subgrad_at_lag(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min > j_max:
        return g
    idx = np.arange(j_min, j_max + 1)
    g[idx] = 2.0 * a[k - idx]
    if k % 2 == 0:
        jd = k // 2
        if j_min <= jd <= j_max:
            g[jd] = a[jd]
    return g


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def init_flat_spectrum(n, rng):
    """Create a sequence with approximately flat power spectrum."""
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    X = np.exp(1j * phases).astype(np.complex128)
    # Enforce conjugate symmetry for real output
    Xr = np.zeros(fft_size, dtype=np.complex128)
    Xr[:fft_size // 2 + 1] = X
    if fft_size % 2 == 0:
        Xr[fft_size // 2 + 1:] = np.conj(Xr[1:fft_size // 2][::-1])
    else:
        Xr[(fft_size // 2 + 1):] = np.conj(Xr[1:fft_size // 2][::-1])
    a_full = np.fft.irfft(Xr, fft_size)[:n]
    a_full = np.maximum(a_full, 0.0)
    if a_full.sum() < 1e-10:
        a_full = np.ones(n)
    return a_full / a_full.sum()
