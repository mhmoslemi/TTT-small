def fft_self_conv(a):
    n = len(a)
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    result = np.fft.irfft(fa * fa, m)[:size]
    return np.maximum(result, 0)

    s = a.sum()
    if s < 0.01:

import numpy as np
import time


def fft_self_conv(a):
    """Compute full self-convolution of a using FFT, returning nonnegative result."""
    n = len(a)
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    result = np.fft.irfft(fa * fa, m)[:size]
    return np.maximum(result, 0.0)


def fast_score(a):
    """Fast score computation without type-checking overhead."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_self_conv(a)
    return 2.0 * n * c.max() / (s * s)


def compute_pressure(a, c, top_k):
    """Compute the pressure vector from the top lags of the convolution."""
    n = len(a)
    pressure = np.zeros(n)
    c_max = c.max()
    if c_max == 0:
        return pressure
    for k in top_k:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            mirror = k - idx
            pressure[lo:hi + 1] += a[mirror] * (c[k] / c_max)
    return pressure


def lp_direction(a, M, n):
    """
    Solve the LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns the optimal b vector, or None if the LP fails.
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None

    # Build constraint matrix A such that A @ b <= M
    # (a*b)[k] = sum_i a[i] * b[k-i] for valid k-i
    # This is a sparse-ish problem. For n=5582, we have 2n-1 = 11163 rows.
    # Each row k has at most n nonzeros.
    # This is too large for dense LP. Use a banded structure.
    # Actually, let's just build it as a list of constraints.
    # For efficiency, only include constraints where M is binding (c[k] close to max).
    # This is an approximation but should work well.

    n_vars = n
    n_cons = 2 * n - 1

    # Build A as a list of arrays (CSR format for linprog)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []

    for k in range(n_cons):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            rows.append(k)
            cols.append(i)
            vals.append(a[j])

    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n_vars))

    c_obj = -np.ones(n_vars)  # minimize -sum(b) = maximize sum(b)

    res = linprog(c_obj, A_ub=A, b_ub=np.full(n_cons, M),
                  bounds=[(0, None)] * n_vars, method='highs',
                  options={'time_limit': 30})

    if res.status == 0:
        return res.x
    return None

    inits.append(np.exp(-6.0 * (n - 1 - np.arange(n, dtype=np.float64)) / n))
    # Shifted Gaussians
    for frac in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        center = frac * n
        inits.append(np.exp(-0.5 * (np.arange(n, dtype=np.float64) - center) ** 2 / (0.12 * n) ** 2))
    # Two asymmetric peaks
    for f1, f2 in [(0.25, 0.7), (0.3, 0.75), (0.2, 0.6)]:
        inits.append(np.exp(-0.5 * ((np.arange(n, dtype=np.float64) - f1 * n) ** 2 +
                                    (np.arange(n, dtype=np.float64) - f2 * n) ** 2) / (0.1 * n) ** 2))
    # Random
    inits.append(rng.dirichlet(np.full(n, 1.0)))
    inits.append(rng.dirichlet(np.full(n, 0.5)))
    inits.append(rng.dirichlet(np.full(n, 2.0)))

    best = None
    best_score = np.inf
    for init in inits:
        init = np.clip(init, 0.0, 1000.0)
        s = init.sum()
        if s < 0.01:
            continue
        init = init * (target_sum / s)
        sc = fast_score(init)
        if sc < best_score:
            best_score = sc
            best = init.copy()

    # Verify with actual evaluator
    actual = evaluate_sequence(best.tolist())
    if np.isfinite(actual) and actual < best_score:
        best_score = actual
    print(f"Phase 1 done. Best initial score: {best_score:.6f}, n={n}")

    # ---- Phase 2: Subgradient optimization ----
    eta = 0.005
    best_ever = best.copy()
    best_ever_score = best_score
    no_improve = 0
    patience = 5000
    iter_count = 0
    use_lp = True
    lp_timer = 0

    while time.monotonic() < deadline:
        iter_count += 1
        lp_timer += 1

        # Compute convolution and score
        c = fft_self_conv(best)
        c_max = c.max()
        c_mean = c.mean()
        current_score = 2.0 * n * c_max / (target_sum * target_sum)

        if current_score < best_ever_score:
            best_ever_score = current_score
            best_ever = best.copy()
            no_improve = 0
            if iter_count % 10000 == 0:
                print(f"Iter {iter_count}: score={best_ever_score:.6f}")

        # Find hot lags
        spread = c_max - c_mean
        if spread < 1e-12:
            threshold = c_max
        else:
            threshold = c_max - 0.08 * spread
        hot_mask = c >= threshold
        hot_idx = np.where(hot_mask)[0]
        if len(hot_idx) > 15:
            hot_vals = c[hot_idx]
            top_pos = np.argsort(hot_vals)[-15:]
            hot_idx = hot_idx[top_pos]

        # Compute pressure and update
        pressure = compute_pressure(best, c, hot_idx)
        max_a = best.max()
        if max_a < 1e-12:
            break
        step = eta * max_a
        new_best = best - step * pressure
        new_best = np.clip(new_best, 0.0, 1000.0)
        new_sum = new_best.sum()
        if new_sum < 0.01:
            eta *= 0.5
            no_improve += 1
            continue
        new_best = new_best * (target_sum / new_sum)
        new_best = np.clip(new_best, 0.0, 1000.0)

        # Evaluate new candidate
        new_c = fft_self_conv(new_best)
        new_c_max = new_c.max()
        new_score = 2.0 * n * new_c_max / (target_sum * target_sum)

        if new_score < current_score * 1.001:
            best = new_best
            no_improve = 0
            eta = min(eta * 1.01, 0.05)
        else:
            no_improve += 1
            eta = max(eta * 0.99, 0.00005)

        # Periodic exploration when stuck
        if no_improve > patience:
            strategy = rng.integers(0, 6)
            if strategy == 0:
                best = best[::-1].copy()
            elif strategy == 1:
                best = best * np.exp(rng.normal(0, 0.03, n))
                best = np.clip(best, 0, 1000)
                best = best * target_sum / best.sum()
            elif strategy == 2:
                ramp = np.linspace(-1, 1, n) * best.max() * 0.02
                best = best + ramp
                best = np.clip(best, 0, 1000)
                best = best * target_sum / best.sum()
            elif strategy == 3:
                new_rand = rng.dirichlet(np.full(n, 1.0)) * target_sum
                t = 0.2 + 0.3 * rng.random()
                best = (1 - t) * best + t * new_rand
                best = np.clip(best, 0, 1000)
                best = best * target_sum / best.sum()
            elif strategy == 4:
                # Block perturbation
                w = int(rng.integers(max(1, n // 100), max(2, n // 10)))
                s0 = int(rng.integers(0, max(1, n - w)))
                best[s0:s0 + w] *= np.exp(rng.normal(0, 0.1, w))
                best = np.clip(best, 0, 1000)
                best = best * target_sum / best.sum()
            else:
                # Sine perturbation
                freq = rng.uniform(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                pert = 1 + 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                best = best * pert
                best = np.clip(best, 0, 1000)
                best = best * target_sum / best.sum()
            no_improve = 0
            patience = int(patience * 1.15)
            eta = 0.005

        # Anneal patience periodically
        if iter_count % 50000 == 0:
            patience = max(3000, int(patience * 0.85))
            eta = max(eta * 0.9, 0.00005)

        # Periodic LP direction (expensive, so infrequent)
        if use_lp and lp_timer > 20000 and iter_count % 50000 == 0:
            lp_start = time.monotonic()
            c = fft_self_conv(best)
            M = c.max() + 0.01 * M
            lp_result = lp_direction(best, M, n)
            lp_elapsed = time.monotonic() - lp_start
            print(f"LP solve took {lp_elapsed:.1f}s")
            if lp_result is not None and len(lp_result) == n:
                lp_result = np.clip(lp_result, 0, 1000)
                s_lp = lp_result.sum()
                if s_lp > 0.01:
                    lp_result = lp_result * (target_sum / s_lp)
                    # Mix with current
                    for t in [0.1, 0.2, 0.3, 0.5]:
                        mixed = (1 - t) * best + t * lp_result
                        mixed = np.clip(mixed, 0, 1000)
                        mixed = mixed * target_sum / mixed.sum()
                        sc = fast_score(mixed)
                        if sc < best_ever_score:
                            best = mixed
                            best_ever = mixed.copy()
                            best_ever_score = sc
                            print(f"LP improvement: score={best_ever_score:.6f}")
                            break
            if lp_elapsed > 60:
                use_lp = False
            lp_timer = 0

    # ---- Phase 3: Final verification and return ----
    best = best_ever
    final_score = evaluate_sequence(best.tolist())
    if np.isfinite(final_score):
        best_ever_score = min(best_ever_score, final_score)
    print(f"Final score: {final_score:.6f}")
    print(f"Total iterations: {iter_count}")

    return [float(x) for x in best]
