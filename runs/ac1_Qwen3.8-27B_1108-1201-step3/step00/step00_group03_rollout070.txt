Gradient-based optimization with annealed soft-max for minimizing the peak

Uses FFT convolution, softmax parameterization for non-negativity,
Adam optimizer, and multi-start over sequence lengths and initializations.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1

def soft_max_and_weights(x, tau):
    """Compute soft-max value and corresponding softmax weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp

    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Create a random initial sequence centered around 1."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k, amp in [(1, 0.25), (3, 0.15), (5, 0.1), (8, 0.08), (13, 0.05)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: (1) gradient-based optimization over multiple
    initializations and sequence lengths with annealed soft-max, (2)
    simulated annealing polish. Returns best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 20.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Define search configurations: (length, num_inits, init_types)
    configs = [
        (256, 4, ['chirp', 'random', 'multi_tone', 'provided']),
        (384, 3, ['chirp', 'random', 'multi_tone']),
        (512, 3, ['chirp', 'random', 'multi_tone']),
        (768, 2, ['chirp', 'random']),
        (1024, 2, ['chirp', 'random']),
    ]

    # Phase 1: Gradient-based optimization
    print("\n=== Phase 1: Gradient optimization ===")
    for n, n_inits, init_types in configs:
        if time.monotonic() > deadline - 60:
            break
        time_left = deadline - time.monotonic()
        time_per = time_left / (n_inits + 0.5)  # leave some buffer

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            init_type = init_types[min(init_i, len(init_types) - 1)]

            # Generate initial sequence
            if init_type == 'provided' and start_seq is not None:
                if len(start_seq) == n:
                    a0 = start_seq.copy()
                else:
                    # Resample to target length
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, start_seq)
            elif init_type == 'chirp':
                alpha = 0.4 + 0.2 * rng.random()
                a0 = make_chirp_init(n, alpha, rng)
            elif init_type == 'multi_tone':
                a0 = make_multi_tone_init(n, rng)
            else:
                a0 = make_random_init(n, rng, spread=0.2 + 0.3 * rng.random())

            a0 = np.clip(a0, 0.01, 1000.0)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)

            t0 = time.monotonic()
            theta_best, grad_score = optimize_gradient(n, theta_init, time_per, rng)
            elapsed = time.monotonic() - t0

            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score_fast(a_best)
            print(f"  n={n}, init='{init_type}', grad_score={grad_score:.6f}, actual={actual_score:.6f}, time={elapsed:.1f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        print(f"  Best after n={n}: {best_score:.6f}")

    # Phase 2: Polish with local search on best candidate
    print(f"\n=== Phase 2: Local search polish ===")
    if best_seq is not None:
        polish_time = min(120.0, max(10.0, deadline - time.monotonic()))
        print(f"  Starting polish, score={best_score:.6f}, time_available={polish_time:.1f}s")
        polished, polish_score = polish_local_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")

    # Phase 3: Try a few more different-length sequences if time permits
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 3: Additional exploration ===")
        remaining = deadline - time.monotonic()
        extra_lengths = [400, 600, 800, 1200]
        for n in extra_lengths:
            if time.monotonic() > deadline - 20:
                break
            a0 = make_chirp_init(n, 0.5, rng)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)
            t_per = min(40.0, remaining / 2.0)
            theta_best, grad_score = optimize_gradient(n, theta_init, t_per, rng)
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score_fast(a_best)
            print(f"  extra n={n}, score={actual_score:.6f}")
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final: verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
