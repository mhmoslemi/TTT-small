import numpy as np
import time

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
- FFT-based self-convolution for O(n log n) per iteration
- Gradient descent with active-set (top-k lags) for smooth optimization
- Simulated annealing to escape local minima
- Multiple structured initializations and periodic restarts
- Periodic verification via evaluate_sequence
"""


def fast_self_convolve(a):
    """Compute full self-convolution of a using FFT. O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_objective(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    b = fast_self_convolve(a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_gradient(a, n_top=1):
    """
    Compute gradient of J w.r.t. a using active set of top n_top lags.
    Returns (gradient, max_b, objective_value).
    
    grad[i] = (4n/(S^2*|S|)) * sum_{k in S} a[k-i] - 4n*b*/S^3
    where a[j] = 0 for j not in [0, n-1].
    """
    n = len(a)
    b = fast_self_convolve(a)
    max_b = float(np.max(b))
    S = float(np.sum(a))
    
    if S < 1e-10:
        return np.zeros(n), max_b, np.inf
    
    J = 2.0 * n * max_b / (S * S)
    
    # Find active set (top n_top lags)
    if n_top >= len(b):
        active = np.arange(len(b))
    else:
        part = np.argpartition(b, -n_top)[-n_top:]
        active = part[np.argsort(b[part])[::-1]]
    
    # Compute conv_term[i] = (1/|S|) * sum_{k in active} a[k-i]
    conv_term = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo <= hi:
            idx = int(k) - np.arange(lo, hi + 1)
            conv_term[lo:hi + 1] += a[idx]
    conv_term /= len(active)
    
    # Gradient: dJ/da[i] = (4n/S^2) * conv_term[i] - 4n*b*/S^3
    grad = (4.0 * n / (S * S)) * conv_term - (4.0 * n * max_b / (S * S * S)) * a
    
    return grad, max_b, J


def generate_initialization(n, rng, init_type):

        # Uniform
        a = np.ones(n)
    elif init_type == 1:

        a = np.exp(-np.arange(n) * 0.001)
    elif init_type == 2:
        # Random positive
        a = np.abs(rng.standard_normal(n)) + 0.01
    elif init_type == 3:
        # Lopsided (linear decrease)
        a = np.linspace(1.0, 0.05, n)
    elif init_type == 4:
        # Symmetric U-shape
        mid = n // 2
        a = np.ones(n)
        a[:mid] = np.linspace(0.1, 1.0, mid)
        a[mid:] = np.linspace(1.0, 0.1, n - mid)
    elif init_type == 5:
        # Two-hump
        a = np.ones(n) * 0.1
        w = n // 5
        a[:w] = 1.0
        a[n-w:] = 1.0
    elif init_type == 6:
        # Cosine envelope
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.01)
    elif init_type == 7:
        # Gaussian centered
        x = np.arange(n) - n / 2
        a = np.exp(-x**2 / (2 * (n/4)**2)) + 0.01
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def perturb_sequence(a, rng, perturbation_size=0.05):
    """Apply a random structural perturbation to a sequence."""
    n = len(a)
    a_new = a.copy()
    pert_type = rng.integers(0, 6)
    
    if pert_type == 0:
        # Add Gaussian noise
        a_new = a + perturbation_size * rng.standard_normal(n)
    elif pert_type == 1:
        # Scale a random block
        width = max(1, int(rng.integers(n // 20, n // 3)))
        start = int(rng.integers(0, n - width))
        a_new[start:start + width] *= rng.uniform(0.3, 3.0)
    elif pert_type == 2:
        # Move mass from one block to another
        w1 = max(1, int(n // 8))
        w2 = max(1, int(n // 8))
        s1 = int(rng.integers(0, n - w1))
        s2 = int(rng.integers(0, n - w2))
        amount = 0.1 * np.sum(a_new[s1:s1+w1])
        a_new[s1:s1+w1] -= amount / w1
        a_new[s2:s2+w2] += amount / w2
    elif pert_type == 3:
        # Mirror a portion
        width = max(1, int(rng.integers(n // 5, n // 2)))
        start = int(rng.integers(0, n - width))
        a_new[start:start+width] = a_new[start:start+width][::-1]
    elif pert_type == 4:
        # Swap two blocks
        w = max(1, int(n // 10))
        s1 = int(rng.integers(0, n - w))
        s2 = int(rng.integers(0, n - w))
        a_new[s1:s1+w], a_new[s2:s2+w] = a_new[s2:s2+w].copy(), a_new[s1:s1+w].copy()
    else:
        # Multiplicative noise
        a_new = a * np.exp(perturbation_size * 0.5 * rng.standard_normal(n))
    
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new / s
    return a_new

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 5.0

        if p.ndim == 1 and p.size > 0 and np.sum(np.maximum(p, 0)) > 0.01:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        n_default = len(parent)
    else:
        n_default = 2048
        parent = np.ones(n_default)
    
    # --- Try multiple initializations and lengths ---
    candidate_pool = []
    
    # Use parent's length and a few others
