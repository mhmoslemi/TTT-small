def mirror_descent_step(a, k_star, lr):
    """One step of mirror descent to reduce c[k_star]."""
    n = len(a)
    # grad[j] = 2 * a[k_star - j] (0 if out of range)
    grad = np.zeros(n)
    # For j in 0..n-1, grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k_star - j]
    
    # Multiplicative update
    a_new = a * np.exp(-lr * grad)
    total = a_new.sum()
    if total > 0:
        a_new /= total
    return a_new
