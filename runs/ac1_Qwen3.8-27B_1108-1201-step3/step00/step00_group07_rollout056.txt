    """Project v onto {x : x >= 0, sum(x) = s}. O(n log n) via sorting."""

    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb).real[:n]


def _simplex_proj(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_normalized(a):
    """Score assuming sum(a)=1: 2n * max(a*a)."""
    n = len(a)
    b = _fft_conv(a, a)
    return 2.0 * n * np.max(b)


def _smooth_grad(a, tau):
    """Gradient of (1/tau) log sum exp(tau * (a*a)_k) w.r.t. a."""
    n = len(a)
    b = _fft_conv(a, a)
    b_max = b.max()
    ez = np.exp(tau * (b - b_max))
    u = ez / ez.sum()
    # g_i = 2 * sum_j u[i+j] * a_j = 2 * (flip(u) * a)[2n-2-i]
    uf = u[::-1]
    conv = _fft_conv(uf, a)
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return b, g


def _gd_run(a0, budget, rng, tau0=2.0, tau1=100.0):
    """Projected gradient descent for one restart."""
    n = len(a0)
    a = np.clip(a0, 0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best = a.copy()
    best_sc = _score_normalized(a)
    tau = tau0
    eta = 1.0 / (n * 4)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < budget:
        b, g = _smooth_grad(a, tau)
        sc = 2.0 * n * b.max()
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Gradient step
        a = a - eta * g
        a = _simplex_proj(a, 1.0)

        # Schedule
        tau = min(tau * 1.003, tau1)
        # Adaptive eta
        gnorm = np.linalg.norm(g)
        if gnorm > 0:
            eta = min(eta * 1.01, 1.0 / (n * gnorm))

        it += 1
        if no_improve > 1000:
            # Perturb
            a += 0.05 * rng.standard_normal(n)
            a = _simplex_proj(a, 1.0)
            no_improve = 0
            eta = 1.0 / (n * 4)

    return best, best_sc
