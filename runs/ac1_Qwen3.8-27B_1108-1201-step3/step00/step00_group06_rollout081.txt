Escapes the uniform-sequence local minimum via random kicks, then refines with annealed softmax.

import numpy as np
import time
import math


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def _flatten_sequence(a, max_time, alpha_start=1.0, alpha_end=500.0, eta=0.3, verbose=False):
    """Flatten auto-convolution via iterative multiplicative gradient updates."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return a, float('inf')
    
    s_target = math.sqrt(2.0 * n)
    a = a * (s_target / s)
    
    n_fft = _next_pow2(3 * n + 2)
    t0 = time.monotonic()
    best_score = float('inf')
    best_a = a.copy()
    patience = 0
    patience_limit = 500
    
    it = 0
    while time.monotonic() - t0 < max_time:
        # Compute linear auto-convolution via FFT
        fa = np.fft.rfft(a, n_fft)
        c = np.fft.irfft(fa * fa, n_fft)
        c = c[:2 * n - 1].copy()
        c = np.maximum(c, 0.0)
        
        max_c = float(c.max())
        s = float(a.sum())
        score = 2.0 * n * max_c / (s * s)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > patience_limit:
            break
        
        # Annealing schedule
        frac = min(1.0, (time.monotonic() - t0) / max_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Softmax weights (centered to prevent overflow)
        c_shifted = c - max_c
        # Clip for numerical safety
        c_shifted = np.maximum(c_shifted, -20.0 / max(alpha, 0.1))
        w = np.exp(alpha * c_shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            break
        w = w / w_sum * (2 * n - 1)

        # = conv(a_rev, w)[i + n - 1]

        it += 1
    
    if verbose:
        print(f"    flatten: {it} iters, best={best_score:.6f}")
    
    return best_a, best_score


def _local_search(a, max_time, rng, verbose=False):

    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()
    n_fft = _next_pow2(3 * n + 2)
    t0 = time.monotonic()
    best_score = float('inf')

    c = np.fft.irfft(fa * fa, n_fft)[:2 * n - 1]
    max_c = float(c.max())
    s = float(a.sum())
    best_score = 2.0 * n * max_c / (s * s)
    
    it = 0
    while time.monotonic() - t0 < max_time:
        # Random block perturbation
        width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        # Multiplicative perturbation with varying strength
        sigma = rng.uniform(0.05, 0.5)
        candidate[start:stop] *= np.exp(rng.normal(0.0, sigma, stop - start))
        
        # Occasionally also perturb a second block
        if rng.random() < 0.4:
            width2 = int(rng.integers(max(1, n // 400), max(2, n // 20)))
            start2 = int(rng.integers(0, max(1, n - width2)))
            stop2 = min(n, start2 + width2)
            candidate[start2:stop2] *= np.exp(rng.normal(0.0, sigma, stop2 - start2))
        
        # Occasionally reverse a segment
        if rng.random() < 0.15:
            seg_len = int(rng.integers(max(2, n // 50), max(3, n // 5)))
            seg_start = int(rng.integers(0, max(1, n - seg_len)))
            candidate[seg_start:seg_start + seg_len] = candidate[seg_start:seg_start + seg_len][::-1]
        
        candidate = np.maximum(candidate, 0.0)
        s_cand = float(candidate.sum())
        if s_cand < 1e-10:
            continue
        
        # Evaluate
        fc = np.fft.rfft(candidate, n_fft)
        cc = np.fft.irfft(fc * fc, n_fft)[:2 * n - 1]
        cc = np.maximum(cc, 0.0)
        score = 2.0 * n * float(cc.max()) / (s_cand * s_cand)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
        
        it += 1
    
    if verbose:
        print(f"    local_search: {it} iters, best={best_score:.6f}")
    
    return best_a, best_score

"""
FFT-accelerated gradient-based flattening of auto-convolution with random-kick exploration.
Escapes the uniform-sequence local minimum via large multiplicative perturbations,
then converges using annealed-softmax multiplicative updates.
Final refinement via block-perturbation local search.
"""

import numpy as np
import time
import math


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def _convolution_fft(a, n_fft):
    """Compute linear auto-convolution of a using FFT."""
    fa = np.fft.rfft(a, n_fft)
    c = np.fft.irfft(fa * fa, n_fft)
    n = a.size
    return c[:2 * n - 1].copy()


def _score_from_conv(c, n):
    """Compute objective from convolution array."""
    max_c = float(c.max())
    s = float(c.sum()) ** 0.5  # sum(a)^2 = sum(conv(a,a)) ... wait no
    # Actually sum(c) = sum(a)^2
    # So S = sqrt(sum(c))
    s = math.sqrt(max(1e-30, float(c.sum())))
    return 2.0 * n * max_c / (s * s)


def _flatten_sequence(a, max_time, alpha_start=1.0, alpha_end=500.0, eta=0.3, patience_limit=400):
    """Flatten auto-convolution via iterative multiplicative gradient updates with annealed softmax."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n < 2:
        return a, float('inf')
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return a, float('inf')

    s_target = math.sqrt(2.0 * n)
    a = a * (s_target / s)

    n_fft = _next_pow2(3 * n + 2)
    t0 = time.monotonic()
    best_score = float('inf')
    best_a = a.copy()
    patience = 0

    it = 0
    while time.monotonic() - t0 < max_time:
        # Compute linear auto-convolution via FFT
        c = _convolution_fft(a, n_fft)
        c = np.maximum(c, 0.0)

        max_c = float(c.max())
        s = float(a.sum())
        score = 2.0 * n * max_c / (s * s)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > patience_limit:
            break

        # Annealing schedule (time-fraction based)
        frac = min(1.0, (time.monotonic() - t0) / max_time)
        alpha = alpha_start * (max(1.0, alpha_end / alpha_start)) ** frac

        # Softmax weights centered to prevent overflow
        c_shifted = c - max_c
        c_shifted = np.maximum(c_shifted, -30.0 / max(alpha, 0.01))
        w = np.exp(alpha * c_shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            break
        w = w / w_sum * (2 * n - 1)  # normalize so mean(w) = 1

        # Compute gradient: grad[i] = sum_j a[j] * w[i+j]
        # = conv(a_rev, w)[i + n - 1]
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n_fft)
        fw = np.fft.rfft(w, n_fft)
        conv_res = np.fft.irfft(fa_rev * fw, n_fft)
        grad = conv_res[n - 1: 2 * n - 1]

        # Remove DC for redistributive update
        grad = grad - grad.mean()

        # Multiplicative update
        g_scale = float(np.abs(grad).max())
        if g_scale < 1e-15:
            break
        a = a * np.exp(-eta * grad / g_scale)
        a = np.maximum(a, 0.0)

        # Renormalize
        s_new = float(a.sum())
        if s_new < 1e-10:
            break
        a = a * (s_target / s_new)

        # Clip to valid range
        a = np.minimum(a, 1000.0)

        it += 1

    return best_a, best_score


def _local_search(a, max_time, rng):
    """Block-perturbation local search for final refinement."""
    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()
    n_fft = _next_pow2(3 * n + 2)
    t0 = time.monotonic()

    # Initial score
    c = _convolution_fft(a, n_fft)
    c = np.maximum(c, 0.0)
    s = float(a.sum())
    best_score = 2.0 * n * float(c.max()) / (s * s)
    best_a = a.copy()

    # Adaptive sigma: start moderate, increase if stuck
    sigma = 0.2
    no_improve = 0
    imrove_limit = 300

    while time.monotonic() - t0 < max_time:
        if no_improve > imrove_limit:
            sigma = min(1.0, sigma * 1.5)
            imrove_limit = int(imrove_limit * 1.3)
            no_improve = 0

        candidate = a.copy()
        n_blocks = 1 if rng.random() < 0.7 else 2

        for _ in range(n_blocks):
            width = int(rng.integers(max(1, n // 300), max(2, n // 8)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, sigma, stop - start))

        # Occasional structural changes
        r = rng.random()
        if r < 0.1:
            # Reverse a segment
            seg_len = int(rng.integers(max(2, n // 40), max(3, n // 4)))
            seg_start = int(rng.integers(0, max(1, n - seg_len)))
            candidate[seg_start:seg_start + seg_len] = candidate[seg_start:seg_start + seg_len][::-1]
        elif r < 0.15:
            # Set a block to a constant
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width)))
            val = float(rng.uniform(0.0, 3.0))
            candidate[start:start + width] = val

        candidate = np.maximum(candidate, 0.0)
        s_cand = float(candidate.sum())
        if s_cand < 1e-10:
            continue

        c_cand = _convolution_fft(candidate, n_fft)
        c_cand = np.maximum(c_cand, 0.0)
        score = 2.0 * n * float(c_cand.max()) / (s_cand * s_cand)

        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def _make_initialization(n, init_type, rng):
    """Generate an initial sequence of length n."""
    x = np.linspace(0.0, 1.0, n)
    if init_type == 'randn':
        return np.abs(rng.normal(1.0, 1.5, n)) + 0.05
    elif init_type == 'randu':
        return rng.uniform(0.0, 2.0, n) + 0.05
    elif init_type == 'bowl':
        return (x - 0.5) ** 2 * 4.0 + 0.1
    elif init_type == 'hill':
        return 1.5 - 1.0 * (x - 0.5) ** 2
    elif init_type == 'cos1':
        return 1.0 + 0.7 * np.cos(2.0 * np.pi * rng.integers(2, max(3, n // 100)) * x)
    elif init_type == 'cos2':
        return 1.0 + 0.7 * np.cos(2.0 * np.pi * rng.integers(5, max(10, n // 80)) * x)
    elif init_type == 'sin':
        k = int(rng.integers(1, max(2, n // 300)))
        return 0.5 + 0.5 * np.abs(np.sin(2.0 * np.pi * k * x)) + 0.05
    elif init_type == 'expdec':
        return np.exp(-4.0 * (x - 0.5) ** 2) + 0.1
    elif init_type == 'power':
        return x ** 3 + 0.1
    elif init_type == 'power2':
        return (1.0 - x) ** 3 + 0.1
    elif init_type == 'gauss':
        return np.exp(-20.0 * (x - rng.uniform(0.3, 0.7)) ** 2) + 0.1
    elif init_type == 'two_bump':
        c1 = float(rng.uniform(0.2, 0.4))
        c2 = float(rng.uniform(0.6, 0.8))
        return np.exp(-50.0 * (x - c1) ** 2) + np.exp(-50.0 * (x - c2) ** 2) + 0.05
    else:
        return np.ones(n)


def _apply_kick(a, rng):
    """Apply a random perturbation to escape local minima."""
    n = a.size
    r = rng.random()
    if r < 0.4:
        # Multiplicative Gaussian kick
        sigma = float(rng.uniform(0.8, 2.5))
        return a * np.exp(rng.normal(0.0, sigma, n))
    elif r < 0.6:
        # Additive kick (clipped)
        sigma = float(rng.uniform(0.5, 2.0))
        return a + np.abs(rng.normal(0.0, sigma, n))
    elif r < 0.8:
        # Block replacement kick
        a_kicked = a.copy()
        n_blocks = int(rng.integers(2, max(3, n // 200)))
        for _ in range(n_blocks):
            width = int(rng.integers(max(1, n // 50), max(2, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            a_kicked[start:start + width] = rng.uniform(0.0, 3.0, width)
        return a_kicked
    else:
        # Mixed: interpolate with a random sequence
        t = float(rng.uniform(0.3, 0.8))
        b = np.abs(rng.normal(1.0, 1.0, n)) + 0.1
        return (1.0 - t) * a + t * b
