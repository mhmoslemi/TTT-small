"""
Minimize  n * 2 * max_k (a*a)_k / (sum a)^2  for a non-negative step function.

Method: projected gradient descent with a temperature-annealed soft-max
replacing max_k (a*a)_k (turns the non-smooth max into a smooth objective),
heavy-ball momentum, and exact Euclidean projection onto the simplex
{a >= 0, sum a = 1}.  The objective is scale-invariant, so we fix sum a = 1.

Several starting points (uniform, two-level periodic, cosine-windowed,
random, and the supplied height_sequence_1 when available) are tried at
several sequence lengths, each time-boxed.  The best finite score observed
is returned.  A valid uniform fallback is always available.
"""

import time
import numpy as np


def _simplex_project(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target} in Euclidean distance."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    # fix tiny drift
    s = out.sum()
    if s > 0:
        out = out * (target / s)
    return out


def _softmax_grad(a, alpha):
    """Return (s, grad_s, maxc, S) for the softened objective pieces."""
    # full linear convolution (symmetric)
    c = np.convolve(a, a, mode="full")
    cm = float(c.max())
    # numerically stable log-sum-exp
    e = np.exp(alpha * (c - cm))
    ssum = float(e.sum())
    p = e / ssum
    s = cm + np.log(ssum) / alpha
    # grad_s[i] = 2 * sum_j a[j] p[i+j]
    # = 2 * full_convolve(a, p_rev)[2n-2-i]
    n = a.size
    p_rev = p[::-1]
    conv = np.convolve(a, p_rev, mode="full")
    grad_s = 2.0 * conv[n - 1:2 * n - 1][::-1]
    S = float(a.sum())
    return s, grad_s, cm, S


def _one_optimize(n, a0, iters, alpha_start, alpha_end, lr0, lrdecay, beta,
                  seed, deadline, score_cb):
    """Run annealed projected gradient on a copy of a0; call score_cb(score)."""
    rng = np.random.default_rng(seed)
    a = _simplex_project(np.asarray(a0, dtype=np.float64).copy(), 1.0)
    if a.size != n or not np.isfinite(a).all() or a.sum() < 0.01:
        a = np.full(n, 1.0 / n)

    vel = np.zeros_like(a)
    last_eval = 0.0

    for it in range(iters):
        if time.monotonic() > deadline:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, iters - 1))
        lr = lr0 / (1.0 + lrdecay * it)

        s, grad_s, cm, S = _softmax_grad(a, alpha)
        # objective f = 2 n s / S^2
        # grad f = (2 n / S^2) grad_s - (4 n s / S^3)
        grad = (2.0 * n / (S * S)) * grad_s - (4.0 * n * s / (S ** 3))

        # occasional cheap exact score to track progress
        if (it - last_eval) * 1.0 > 0 and (it % 40 == 0 or it == 0):
            sc = _safe_score(a)
            if np.isfinite(sc):
                score_cb(sc)
            last_eval = it

        vel = beta * vel - lr * grad
        a = _simplex_project(a + vel, 1.0)

    sc = _safe_score(a)
    if np.isfinite(sc):
        score_cb(sc)
    return a.copy(), sc

def _make_starts(n, seed):

    starts = []

    # uniform
    starts.append(np.full(n, 1.0 / n))

    # two-level periodic 1,0,1,0,... normalized

    if two.sum() > 0:
        starts.append(two / two.sum())

    # two-level with 3-period
    three = (np.arange(n) % 3 == 0).astype(np.float64) * 0.5 + 0.1
    starts.append(three / three.sum())

    # cosine window
    x = np.linspace(0.0, np.pi, n)
    cosw = np.abs(np.cos(x)) + 0.05
    starts.append(cosw / cosw.sum())

    # random
    starts.append(rng.random(n) + 0.05)
    r2 = starts[-1] / starts[-1].sum()
    starts.append(r2)

    # symmetric random (palindromic)
    half = rng.random((n + 1) // 2) + 0.05
    pal = np.concatenate([half, half[-2::-1]]) if n % 2 == 0 else np.concatenate([half, half[-1::-1]])
    pal = pal[:n]
    starts.append(pal / pal.sum())

    # supplied sequence if present
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        h = h[h >= 0]
        if h.size == n and h.sum() > 0:
            starts.append(h / h.sum())
        elif h.size >= n and h.sum() > 0:
            starts.append(h[:n] / h[:n].sum())
        elif 0 < h.size < n and h.sum() > 0:
            pad = np.random.default_rng(seed).uniform(0.05, 1.0, n - h.size)
            mixed = np.concatenate([h, pad])
            starts.append(mixed / mixed.sum())
    except Exception:
        pass

    return starts

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    global _best_score, _best_seq
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = float(budget_s)
    # reserve time for final exact evaluation of the best
    deadline = start_time + max(5.0, total_budget - 20.0)

    _best_score = np.inf
    _best_seq = None

    def store(score, seq):
        nonlocal _best_score, _best_seq
        if np.isfinite(score) and score < _best_score:
            _best_score = float(score)
            _best_seq = np.array(seq, dtype=np.float64).copy()
            print(f"[{time.monotonic()-start_time:7.1f}s] new best = {_best_score:.6f} (len={len(_best_seq)})")

    # candidate lengths: keep moderate so convolve stays fast; larger n -> better bound
    for n in [2535, 4096, 8192, 2048]:
        if time.monotonic() > deadline:
            break
        # per-length time slice based on how many lengths remain
        slice_s = max(30.0, (deadline - time.monotonic()) * 0.28)
        len_deadline = time.monotonic() + slice_s

        # per-start annealing parameters
        alpha_start = 8.0
        alpha_end = 4000.0
        lr0 = 0.05
        lrdecay = 2.0e-4
        beta = 0.9
        iters = 400  # effectively limited by time

        count_starts = len(_make_starts(n, seed))
        # try a few distinct seeds for diversity on the best-starting structures
        for trial, sse in enumerate([seed, seed + 101, seed + 2025]):
            if time.monotonic() > len_deadline:
                break
            starts = _make_starts(n, sse)
            for si, st in enumerate(starts):
                if time.monotonic() > len_deadline:
                    break
                iters_here = max(50, int(iters * (len_deadline - time.monotonic()) + 1))
                a, sc = _one_optimize(
                    n, st, iters_here, alpha_start, alpha_end, lr0, lrdecay, beta,
                    sse + 7 * (si + 1) + 13 * (trial + 1), len_deadline, store)
                store(sc, a)

    # ---- final polish: short random block-perturbation local search on best ----
    if _best_seq is not None and time.monotonic() < deadline:
        base = _best_seq.copy()
        base = base / base.sum()
        for _ in range(60):
            if time.monotonic() > deadline:
                break
            cnd = base.copy()
            w = max(2, int(base.size // 60))
            s = int(rng.integers(0, max(1, base.size - w + 1)))
            cnd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.15)))
            cnd = _simplex_project(cnd, 1.0)
            sc = _safe_score(cnd)
            if np.isfinite(sc) and sc < _best_score:
                _best_score = sc
                base = cnd
                store(sc, base)

    # ---- final exact check / fallback ----
    if _best_seq is None or not np.all(np.isfinite(_best_seq)):
        _best_seq = np.ones(256)
        _best_score = _safe_score(_best_seq)

    out = [float(max(0.0, min(1000.0, x))) for x in _best_seq]
    final = _safe_score(out)
    if np.isfinite(final) and final < _best_score:
        _best_score = final
    print(f"final score = {_best_score:.6f}, length = {len(out)}, time = {time.monotonic()-start_time:.1f}s")
    return out
