def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    return a

"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a)**2)
by mirror-descent / Bregman gradient updates on log-weights.

The method:
- works with sequences normalized to sum 1,
- uses FFT linear convolution for the objective,
- uses a softmax active-set weight vector over convolution lags,
- updates log(a) with projected multiplicative steps,
- keeps the best sequence found and uses random logarithmic
  perturbations to escape stagnation.
"""

import numpy as np
import time


def next_pow2(m):
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


def central_sequence(n):
    a = np.empty(max(1, int(n)), dtype=np.float64)
    if a.size == 0:
        return a
    a[0] = 1.0
    for k in range(a.size - 1):
        a[k + 1] = a[k] * float(2 * k + 1) / float(2 * k + 2)
    s = a.sum()
    if s > 0.0 and np.isfinite(s):
        a /= s
    else:
        a[:] = 1.0
        a /= a.size
    return a


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def interpolate_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(max(1, n))
    if a.size == 0:
        return central_sequence(n)
    if a.size == n:
        return normalize_positive(a)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, n)
    b = np.interp(x1, x0, a)
    return normalize_positive(b)


def theta_to_a(theta):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0:
        return np.ones(1, dtype=np.float64)
    t = theta - float(np.max(theta))
    t = np.maximum(t, -745.0)
    a = np.exp(t)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(theta.size, dtype=np.float64) / float(theta.size)
    return a / s


def convolve_fft(a, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c, F


def cross_corr_shift(F_a, w, L, n):
    W = np.fft.rfft(w, L)
    v = np.fft.irfft(np.conj(F_a) * W, L)[:n]
    np.maximum(v, 0.0, out=v)
    return v


def score_and_weights(c, n, lam):
    c = np.asarray(c, dtype=np.float64)
    M = float(np.max(c))
    exact = float(2.0 * n * M)
    if M <= 0.0 or not np.isfinite(M):
        p = np.zeros_like(c)
        if c.size:
            p[0] = 1.0
        return exact, exact, p

    lam = max(1e-6, float(lam))
    lp = c - M
    lp *= lam
    np.clip(lp, -745.0, 0.0, out=lp)
    p = np.exp(lp, out=lp)
    exp_sum = float(p.sum())
    if exp_sum <= 0.0 or not np.isfinite(exp_sum):
        p.fill(0.0)
        p[int(np.argmax(c))] = 1.0
        exp_sum = 1.0
    mean_exp = exp_sum / float(c.size)
    if mean_exp <= 0.0 or not np.isfinite(mean_exp):
        search = exact
    else:
        search = float(2.0 * n * (M + np.log(mean_exp) / lam))
    p /= exp_sum
    return exact, search, p


def internal_score(a):
    a = normalize_positive(a)
    n = a.size
    L = next_pow2(3 * n)
    c, _ = convolve_fft(a, L)
    return float(2.0 * n * np.max(c))


def evaluate_theta(theta, n, L, lam):
    a = theta_to_a(theta)
    c, F = convolve_fft(a, L)
    exact, search, p = score_and_weights(c, n, lam)
    return a, c, F, exact, search, p


def make_initial_candidates(n, rng):
    cands = []
    base = central_sequence(n)
    cands.append(base)

    if n > 1:
        x = np.arange(n, dtype=np.float64)
        xs = x / float(n - 1)

        for sig in (0.10, 0.20, 0.35, 0.50):
            a = base * np.exp(rng.normal(0.0, sig, n))
            cands.append(normalize_positive(a))

        for t in (-0.6, -0.3, 0.3, 0.6):
            a = base * np.exp(t * xs)
            cands.append(normalize_positive(a))

        for w in (1.5, 3.0, 6.0, 12.0):
            a = np.exp(-w * np.square(xs - 0.5))
            cands.append(normalize_positive(a))

        cands.append(normalize_positive(base + base[::-1]))
        cands.append(normalize_positive(base[::-1]))

    try:
        h = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if h.size > 0:
            if h.size != n:
                h = interpolate_sequence(h, n)
            cands.append(normalize_positive(h))
    except Exception:
        pass

    return cands


def optimize_sequence(n, a0, deadline, rng, tag="opt"):
    n = int(max(2, n))
    a0 = normalize_positive(a0 if a0 is not None else central_sequence(n))
    start = time.monotonic()

    L = next_pow2(3 * n)
    theta = np.log(np.maximum(a0, 1e-300))
    theta -= float(theta.mean())
    theta = np.clip(theta, -50.0, 50.0)

    a, c, F, exact, search, _ = evaluate_theta(theta, n, L, 50.0)

    best_a = a.copy()
    best_theta = theta.copy()
    best_exact = float(exact)
    best_search = float(search)

    cur_theta = theta.copy()
    cur_a = a.copy()
    cur_c = c.copy()
    cur_F = F.copy()
    cur_exact = float(exact)
    cur_search = float(search)

    vel = np.zeros(n, dtype=np.float64)
    lr = 0.08
    min_lr = 1e-6
    max_lr = 0.25
    mu = 0.90
    stagnation = 0
    it = 0

    total_budget = max(1.0, float(deadline) - start)

    if tag:
        print(f"[{tag}] start score={best_exact:.10f} n={n}")

    while time.monotonic() < float(deadline):
        progress = min(1.0, (time.monotonic() - start) / total_budget)
        lam = 50.0 + 1450.0 * progress

        if not np.isfinite(cur_exact):
            cur_theta = best_theta.copy()
            cur_a, cur_c, cur_F, cur_exact, cur_search, _ = evaluate_theta(cur_theta, n, L, lam)

        cur_exact, cur_search, p = score_and_weights(cur_c, n, lam)

        v = cross_corr_shift(cur_F, p, L, n)
        g = v - float(np.mean(v))
        scale = float(np.max(np.abs(g)))

        if (not np.isfinite(scale)) or scale < 1e-14:
            th = cur_theta + rng.normal(0.0, 0.05, n)
            th -= float(th.mean())
            th = np.clip(th, -50.0, 50.0)
            a_new, c_new, F_new, exact_new, search_new, _ = evaluate_theta(th, n, L, lam)

            accept = False
            if np.isfinite(exact_new):
                accept = (
                    exact_new < best_exact - 1e-12 or
                    (exact_new <= cur_exact + 1e-10 and exact_new <= best_exact + 1e-6 and
                     search_new < cur_search - 1e-12)
                )

            if accept:
                cur_theta = th
                cur_a = a_new
                cur_c = c_new
                cur_F = F_new
                cur_exact = float(exact_new)
                cur_search = float(search_new)
                if exact_new < best_exact - 1e-12:
                    best_a = a_new.copy()
                    best_theta = th.copy()
                    best_exact = float(exact_new)
                    best_search = float(search_new)
                    if tag and (it == 1 or it % 5000 == 0):
                        print(f"[{tag}] it={it} score={best_exact:.10f} lam={lam:.1f} lr={lr:.4f}")
                stagnation = 0
            else:
                lr = max(min_lr, lr * 0.5)
                stagnation += 1
            it += 1
            continue

        direction = g / scale
        vel = mu * vel + (1.0 - mu) * direction
        vmax = float(np.max(np.abs(vel)))
        if (not np.isfinite(vmax)) or vmax == 0.0:
            vel.fill(0.0)
            lr = 0.05
            stagnation += 1
            it += 1
            continue
        if vmax > 1.0:
            vel /= vmax

        th = cur_theta - lr * vel
        th -= float(th.mean())
        range_th = float(np.max(th) - np.min(th))
        if not np.isfinite(range_th) or range_th > 100.0:
            th = th - float((np.min(th) + np.max(th)) / 2.0)
        th = np.clip(th, -50.0, 50.0)

        a_new, c_new, F_new, exact_new, search_new, _ = evaluate_theta(th, n, L, lam)

        accept = False
        if np.isfinite(exact_new):
            if exact_new < best_exact - 1e-12:
                accept = True
            elif (
                exact_new <= cur_exact + 1e-10 and
                exact_new <= best_exact + 1e-6 and
                search_new < cur_search - 1e-12
            ):
                accept = True

        if accept:
            cur_theta = th
            cur_a = a_new
            cur_c = c_new
            cur_F = F_new
            cur_exact = float(exact_new)
            cur_search = float(search_new)

            if exact_new < best_exact - 1e-12:
                best_a = a_new.copy()
                best_theta = th.copy()
                best_exact = float(exact_new)
                best_search = float(search_new)
                stagnation = 0
                if lr < max_lr:
                    lr = min(max_lr, lr * 1.04)
            else:
                stagnation = max(0, stagnation - 20)

            if tag and (it == 1 or it % 5000 == 0):
                print(f"[{tag}] it={it} score={best_exact:.10f} lam={lam:.1f} lr={lr:.4f}")
        else:
            lr = max(min_lr, lr * 0.7)
            stagnation += 1

            if stagnation > 900:
                improved = False

                for sigma in (0.02, 0.05, 0.10, 0.20, 0.40, 0.80):
                    th = best_theta + rng.normal(0.0, sigma, n)
                    th -= float(th.mean())
                    th = np.clip(th, -50.0, 50.0)
                    ar, cr, Fr, er, sr, _ = evaluate_theta(th, n, L, lam)
                    if np.isfinite(er) and (
                        er < best_exact - 1e-12 or
                        (er <= best_exact + 1e-8 and sr < cur_search - 1e-12)
                    ):
                        cur_theta = th
                        cur_a = ar
                        cur_c = cr
                        cur_F = Fr
                        cur_exact = float(er)
                        cur_search = float(sr)
                        if er < best_exact - 1e-12:
                            best_a = ar.copy()
                            best_theta = th.copy()
                            best_exact = float(er)
                            best_search = float(sr)
                            if tag:
                                print(f"[{tag}] restart improved to {best_exact:.10f}")
                        vel.fill(0.0)
                        lr = 0.06
                        stagnation = 0
                        improved = True
                        break

                if not improved:
                    for _ in range(6):
                        th = best_theta.copy()
                        w = int(rng.integers(max(1, n // 30), max(2, n // 4)))
                        st = int(rng.integers(0, n - w))
                        th[st:st + w] += rng.normal(0.0, 0.25, w)
                        th -= float(th.mean())
                        th = np.clip(th, -50.0, 50.0)
                        ar, cr, Fr, er, sr, _ = evaluate_theta(th, n, L, lam)
                        if np.isfinite(er) and (
                            er < best_exact - 1e-12 or
                            (er <= best_exact + 1e-8 and sr < cur_search - 1e-12)
                        ):
                            cur_theta = th
                            cur_a = ar
                            cur_c = cr
                            cur_F = Fr
                            cur_exact = float(er)
                            cur_search = float(sr)
                            if er < best_exact - 1e-12:
                                best_a = ar.copy()
                                best_theta = th.copy()
                                best_exact = float(er)
                                best_search = float(sr)
                                if tag:
                                    print(f"[{tag}] block restart improved to {best_exact:.10f}")
                            vel.fill(0.0)
                            lr = 0.06
                            stagnation = 0
                            improved = True
                            break

                if not improved:
                    stagnation = 450

        it += 1

    return best_a, best_exact


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    budget = max(1.0, float(budget_s))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    if budget >= 1200.0:
        n_large = 16384
        n_small = 2048
        small_time = max(80.0, 0.08 * budget)
    elif budget >= 450.0:
        n_large = 8192
        n_small = 1024
        small_time = max(30.0, 0.10 * budget)
    elif budget >= 120.0:
        n_large = 4096
        n_small = 512
        small_time = max(10.0, 0.12 * budget)
    else:
        n_large = 1024
        n_small = None
        small_time = 0.0

    reserve = 30.0 if budget > 120.0 else 5.0
    research_deadline = start + max(1.0, budget - reserve)

    global_best_a = central_sequence(n_large)
    global_best_score = internal_score(global_best_a)

    if n_small is not None and small_time > 5.0:
        small_deadline = min(start + small_time, research_deadline - 10.0)
        if small_deadline > start + 2.0:
            cands = make_initial_candidates(n_small, rng)
            best_small = None
            best_small_sc = np.inf
            for a in cands:
                sc = internal_score(a)
                if np.isfinite(sc) and sc < best_small_sc:
                    best_small_sc = sc
                    best_small = a
            if best_small is None:
                best_small = central_sequence(n_small)

            a_small, sc_small = optimize_sequence(n_small, best_small, small_deadline, rng, tag="small")
            if sc_small < global_best_score:
                global_best_a = a_small
                global_best_score = sc_small
            small_best = a_small
        else:
            small_best = central_sequence(n_large)
    else:
        small_best = central_sequence(n_large)

    large_cands = []
    large_cands.append(central_sequence(n_large))
    large_cands.append(interpolate_sequence(small_best, n_large))

    if n_large > 1:
        noisy = central_sequence(n_large) * np.exp(rng.normal(0.0, 0.12, n_large))
        large_cands.append(normalize_positive(noisy))

    try:
        h = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if h.size > 0:
            large_cands.append(interpolate_sequence(h, n_large))
    except Exception:
        pass

    primary = None
    primary_sc = np.inf
    secondary = None
    secondary_sc = np.inf

    for a in large_cands:
        sc = internal_score(a)
        if not np.isfinite(sc):
            continue
        if sc < primary_sc:
            secondary_sc = primary_sc
            secondary = primary
            primary_sc = sc
            primary = a
        elif sc < secondary_sc:
            secondary_sc = sc
            secondary = a

    if primary is None:
        primary = central_sequence(n_large)

    large_deadline = research_deadline
    if time.monotonic() < large_deadline:
        a_large, sc_large = optimize_sequence(n_large, primary, large_deadline, rng, tag="large")
        if sc_large < global_best_score:
            global_best_a = a_large
            global_best_score = sc_large

    # A final small deterministic cleanup on residual time can help if very little
    # was used by the main run.
    residual_deadline = start + budget - 5.0
    if time.monotonic() < residual_deadline:
        # Polish the best known sequence a little more at its own length.
        n_polish = len(global_best_a)
        mini_deadline = residual_deadline
        if mini_deadline > time.monotonic() + 2.0:
            a_pol, sc_pol = optimize_sequence(n_polish, global_best_a, mini_deadline, rng, tag="polish")
            if sc_pol < global_best_score:
                global_best_a = a_pol
                global_best_score = sc_pol

    best = normalize_positive(global_best_a)
    # Avoid returning absurd singleton or empty sequence.
    if best.size < 2:
        best = central_sequence(max(64, int(next_pow2(2))))

    print(f"final internal score={float(2.0 * len(best) * np.max(np.convolve(best, best)) / max(1e-300, best.sum()**2)):.10f} n={len(best)}")
    return [float(x) for x in best]
