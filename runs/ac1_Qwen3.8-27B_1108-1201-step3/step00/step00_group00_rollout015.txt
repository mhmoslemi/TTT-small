def fast_eval(a):
    """Fast evaluation of the objective using FFT-based convolution."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 2 * n - 1
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)
    M = b.max()
    return 2.0 * n * M / (S * S)


def make_perturbation(a, rng, scale):
    """Apply multiplicative perturbation to a."""
    noise = rng.normal(0, 1, size=a.shape)
    candidate = a * np.exp(scale * noise)
    return np.clip(candidate, 0, 1000)


def make_block_perturbation(a, rng, scale):
    """Apply block multiplicative perturbation."""
    n = len(a)
    width = max(2, int(rng.integers(n // 100, n // 5)))
    start = int(rng.integers(0, n - width))
    candidate = a.copy()
    candidate[start:start+width] *= np.exp(scale * rng.normal(0, 1))
    return np.clip(candidate, 0, 1000)

    """Reduce elements that contribute to the peak at k_star."""
    n = len(a)
    candidate = a.copy()
    # Find elements contributing to the peak
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    # Get the contributing pairs
    contrib = np.zeros(n)
    for i in range(lo, hi):
        contrib[i] = a[i] * a[k_star - i]
    # Reduce the top contributors

    top_indices = np.argsort(contrib[lo:hi])[::-1][:top_k] + lo
    candidate[top_indices] *= (1 - strength)
    # To maintain sum, increase some other elements
    other = np.setdiff1d(np.arange(n), top_indices)
    inc_idx = rng.choice(other, size=min(len(top_indices), len(other)), replace=False)
    inc_amount = a[top_indices].sum() * strength * rng.uniform(0.5, 1.0)
    candidate[inc_idx] += inc_amount / len(inc_idx)
    return np.clip(candidate, 0, 1000)

def smooth_objective_and_grad(a, T):
    """Compute smoothed objective and gradient using logsumexp."""
    n = len(a)
    N = 2*n - 1
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa*fa, N)
    S = a.sum()
    
    # Compute softmax of b/T
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    p = exp_b / exp_b.sum()
    
    # Smoothed max ≈ b_max + T*log(exp_b.sum()) = b_max + T*logsumexp((b-b_max)/T)
    smooth_max = b_max + T * np.log(exp_b.sum())
    
    f_val = 2.0 * n * smooth_max / (S*S)
    
    # Gradient: dL/da[i] = 2 * sum_k a[k-i] * p[k]
    # This is cross-correlation
    # c[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    # = conv(a, p_rev)[2n-2-i] where p_rev = p[::-1]
    p_rev = p[::-1]

    return f_val, grad

def fast_eval(a):
    """Fast objective evaluation using FFT."""
    n = a.shape[0]
    S = a.sum()
    if S < 1e-6:
        return np.inf
    fa = np.fft.rfft(a, 2*n-1)
    b = np.fft.irfft(fa * fa, 2*n-1)
    M = b.max()
    return 2.0 * n * M / (S * S)

def smooth_grad(a, T):
    """Compute smoothed objective and gradient using logsumexp."""
    n = a.shape[0]
    N = 2 * n - 1
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)
    S = a.sum()
    if S < 1e-6:
        return np.inf, np.zeros(n)
