LP-based optimization for low-peak autocorrelation sequences.
Uses linear programming to find directions that maximize sum while
keeping cross-correlation bounded, combined with local search.
"""

from scipy.sparse import csr_matrix

def build_conv_matrix(f):
    """Build sparse matrix A such that A @ b = convolve(f, b)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            if f[k - j] != 0:
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)

2*n*max(conv(a,a))/(sum(a))^2.

def build_conv_matrix(f):
    """Build sparse matrix A such that A @ b = convolve(f, b)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            v = f[k - j]
            if v > 0:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, time_limit=10):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        pass
    return None


def local_search_step(best, best_score, rng, n_trials=30, sigma=None):
    """Block perturbation local search."""
    n = len(best)
    if sigma is None:
        sigma = 0.04
    for _ in range(n_trials):
        candidate = best.copy()
        width = int(rng.integers(2, max(3, n // 25)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate[start:stop] *= np.exp(rng.normal(0.0, sigma, size=stop - start))
        candidate = np.clip(candidate, 0.0, None)
        if candidate.sum() < 0.01:
            continue
        score = evaluate_sequence(candidate.tolist())
        if score < best_score:
            best = candidate
            best_score = score
    return best, best_score


def sa_accept_step(best, best_score, rng, temperature):
    """One simulated annealing step."""
    n = len(best)
    candidate = best.copy()
    width = int(rng.integers(3, max(4, n // 15)))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
    candidate = np.clip(candidate, 0.0, None)
    if candidate.sum() < 0.01:
        return best, best_score
    score = evaluate_sequence(candidate.tolist())
    if score < best_score:
        return candidate, score
    if temperature > 1e-6 and np.isfinite(score) and np.isfinite(best_score):
        delta = score - best_score
        if delta < 10.0 * temperature:
            if rng.random() < np.exp(-delta / temperature):
                return candidate, score
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: LP direction + local search + SA for low autocorrelation peak."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)

    # --- Initialization ---
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        init = None

    if init is not None and init.ndim == 1 and init.size > 0 and np.isfinite(init).all():
        init = np.clip(init, 0.0, 1000.0)
    else:
        init = np.ones(256)

    n = len(init)
    best = init.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: score={best_score:.6f}, n={n}")

    # Try diverse initializations
    t_arr = np.arange(n, dtype=np.float64)
    c = n / 2.0
    init_list = [
        rng.random(n) + 0.01,
        np.exp(-2.0 * ((t_arr - 0.6 * n) / (0.35 * n)) ** 2),
        np.exp(-2.0 * ((t_arr - 0.7 * n) / (0.35 * n)) ** 2),
        np.exp(-1.5 * ((t_arr - 0.4 * n) / (0.25 * n)) ** 2) +
        np.exp(-1.5 * ((t_arr - 0.8 * n) / (0.25 * n)) ** 2) + 0.01,
        0.3 + 0.7 * np.cos(2 * np.pi * 2 * t_arr / n) ** 2,
        0.3 + 0.7 * np.cos(2 * np.pi * 5 * t_arr / n) ** 2,
        (t_arr / n) ** 2 + 0.01,
        (1.0 - t_arr / n) ** 2 + 0.01,
        np.abs(np.sin(np.pi * t_arr / n)) * (0.5 + 0.5 * np.cos(2 * np.pi * 3 * t_arr / n)) + 0.01,
        np.exp(-((t_arr - 0.3 * n) / (0.15 * n)) ** 2) * 2.0 +
        np.exp(-((t_arr - 0.65 * n) / (0.2 * n)) ** 2) + 0.01,
    ]

    for ic in init_list:
        ic = np.clip(ic, 0.0, 1000.0)
        if ic.sum() < 0.01:
            continue
        s = evaluate_sequence(ic.tolist())
        if s < best_score:
            best = ic.copy()
            best_score = s
            print(f"  Better init: {best_score:.6f}")

    print(f"After inits: score={best_score:.6f}")

    # --- Main optimization loop ---
    lp_count = 0
    no_improve = 0
    temperature = 0.05
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        prev_score = best_score

        # === LP direction finding ===
        f_norm = best / best.sum()
        conv_f = np.convolve(f_norm, f_norm)
        M = conv_f.max()

        # Solve LP
        time_left_lp = min(8.0, max(1.0, (deadline - time.monotonic()) * 0.05))
        lp_result = solve_lp_direction(f_norm, M, time_limit=time_left_lp)
        lp_count += 1

        if lp_result is not None and np.isfinite(lp_result).all():
            lp_sum = lp_result.sum()
            if lp_sum > 1.0 + 1e-7:
                g = lp_result / lp_sum
                # Try multiple blend factors (small to large)
                for t_blend in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 1.0]:
                    candidate = (1.0 - t_blend) * f_norm + t_blend * g
                    candidate = np.clip(candidate, 0.0, None)
                    candidate = candidate * best.sum()
                    if candidate.sum() < 0.01:
                        continue
                    score = evaluate_sequence(candidate.tolist())
                    if score < best_score:
                        best = candidate.copy()
                        best_score = score
                        no_improve = 0

        # Also try LP with slightly relaxed M for diversity
        if time.monotonic() < deadline - 5 and lp_result is not None:
            M_relaxed = M * 1.05
            lp_result2 = solve_lp_direction(f_norm, M_relaxed, time_limit=time_left_lp)
            lp_count += 1
            if lp_result2 is not None and np.isfinite(lp_result2).all():
                lp_sum2 = lp_result2.sum()
                if lp_sum2 > best.sum() * 1.001:
                    g2 = lp_result2 / lp_result2.sum()
                    for t_blend in [0.05, 0.1, 0.2, 0.4]:
                        candidate = (1.0 - t_blend) * f_norm + t_blend * g2
                        candidate = np.clip(candidate, 0.0, None)
                        candidate = candidate * best.sum()
                        if candidate.sum() < 0.01:
                            continue
                        score = evaluate_sequence(candidate.tolist())
                        if score < best_score:
                            best = candidate.copy()
                            best_score = score
                            no_improve = 0

        # === Local search ===
        if time.monotonic() < deadline - 3:
            n_ls = 15 if n > 1000 else 25
            best, best_score = local_search_step(best, best_score, rng, n_trials=n_ls,
                                                  sigma=0.03)

        # === Simulated annealing ===
        if no_improve > 15 and temperature > 1e-5:
            best, best_score = sa_accept_step(best, best_score, rng, temperature)
            temperature *= 0.93
            if best_score > prev_score:
                # Revert if SA made things too bad
                pass  # Keep it, SA is meant to explore

        if best_score < prev_score - 1e-10:
            no_improve = 0
            temperature = min(temperature * 1.05, 0.1)
        else:
            no_improve += 1
            temperature *= 0.995

        # Log
        if iteration % 10 == 0:
            remaining = deadline - time.monotonic()
            print(f"Iter {iteration}: score={best_score:.6f}, "
                  f"lp={lp_count}, no_improve={no_improve}, "
                  f"temp={temperature:.4f}, rem={remaining:.0f}s")

    # --- Final intensive local search with decreasing sigma ---
    remaining = deadline - time.monotonic()
    if remaining > 2:
        # Phase 1: moderate sigma
        n_p1 = max(20, int(remaining * 0.4 * 20))
        best, best_score = local_search_step(best, best_score, rng,
                                              n_trials=n_p1, sigma=0.03)
        print(f"After fine LS1: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        if remaining > 1:
            # Phase 2: small sigma
            n_p2 = max(20, int(remaining * 0.5 * 30))
            best, best_score = local_search_step(best, best_score, rng,
                                                  n_trials=n_p2, sigma=0.01)
            print(f"After fine LS2: {best_score:.6f}")

    print(f"FINAL: score={best_score:.6f}, n={len(best)}, lp_solves={lp_count}")
    return [float(x) for x in best]
