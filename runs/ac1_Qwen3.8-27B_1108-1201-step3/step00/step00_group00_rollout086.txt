"""
Hybrid nonnegative autocorrelation peak minimizer.

The objective is scale-invariant, so the search normalizes the sequence to
sum = sqrt(2n), making the evaluated score equal to the maximum linear
convolution (up to floating point).  The main iteration uses a smooth
log-sum-exp surrogate for the maximum convolution and a multiplicative
Adam-style update in the log domain.  Random anti-palindromic / quasi-periodic
initial points break the symmetric uniform fixed point.  Additional tracks use
different temperatures and noisy restarts, followed by a fast FFT-scored block
local search.
"""

import numpy as np
import time
import math


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _rescale_with_floor(a, target_sum, floor):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, floor)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 0.0:
        a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
        a = np.maximum(a, floor)
        s = float(a.sum())
        if not math.isfinite(s) or s <= 0.0:
            return np.full(a.shape, float(target_sum) / float(a.size), dtype=np.float64)
    return a * (float(target_sum) / s)


def _safe_positive(a, n, target_sum):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size != n:
        return None
    floor = max(1e-12, 1e-10 * float(target_sum) / float(n))
    return _rescale_with_floor(a, target_sum, floor)


def _score_key(x):
    return x[0]


def _rfft_square_conv(a, N, out_len):
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fa * fa, n=N)[:out_len]
    c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 0.0)
    return c, fa


def _corr_with_p(a, p, fa, N, n):
    L = 2 * n - 1
    pr = p[::-1].copy()
    fp = np.fft.rfft(pr, n=N)
    conv = np.fft.irfft(fa * fp, n=N)[: (3 * n - 2)]
    start = L - 1
    stop = L - n - 1
    q = conv[start:stop:-1]
    return np.asarray(q, dtype=np.float64).copy()


def _make_p(c, tau):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c
    M = float(np.max(c))
    if not math.isfinite(M):
        return np.full(c.shape, 1.0 / float(c.size), dtype=np.float64)
    if tau <= 1e-12:
        tau = 1e-12
    s = c - M
    w1 = np.exp(s / tau)
    s1 = float(np.sum(w1))
    if not math.isfinite(s1) or s1 <= 0.0:
        p = np.zeros_like(c)
        p[c == M] = 1.0
        p /= float(p.sum())
        return p
    p1 = w1 / s1
    tau2 = max(4.0 * tau, 1e-10)
    w2 = np.exp(s / tau2)
    s2 = float(np.sum(w2))
    if math.isfinite(s2) and s2 > 0.0:
        p2 = w2 / s2
    else:
        p2 = p1
    p = 0.85 * p1 + 0.15 * p2
    ps = float(np.sum(p))
    if math.isfinite(ps) and ps > 0.0:
        p = p / ps
    else:
        p = p1
    return p


def _fft_score_with_N(a, n, N):
    a = np.asarray(a, dtype=np.float64)
    c, _ = _rfft_square_conv(a, N, 2 * n - 1)
    if c.size == 0:
        return np.inf
    S = float(a.sum())
    if not math.isfinite(S) or S <= 0.0:
        return np.inf
    return float(2.0 * n * float(np.max(c)) / (S * S))


def _fft_score(a, n):
    N = _next_pow2(max(1, 3 * n - 2))
    return _fft_score_with_N(a, n, N)


def _make_initial_candidates(parent, n, rng, S0):
    cands = []
    idx = np.arange(n, dtype=np.float64)

    cand = _safe_positive(parent, n, S0)
    if cand is not None:
        cands.append(cand)

    if parent is not None and np.asarray(parent).size == n:
        for sigma in (0.25, 0.5, 0.9):
            cand = _safe_positive(parent * np.exp(rng.normal(0.0, sigma, size=n)), n, S0)
            if cand is not None:
                cands.append(cand)

    cand = _safe_positive(rng.exponential(1.0, size=n), n, S0)
    if cand is not None:
        cands.append(cand)

    cand = _safe_positive(np.exp(rng.normal(0.0, 1.0, size=n)), n, S0)
    if cand is not None:
        cands.append(cand)

    cand = _safe_positive(rng.gamma(2.0, 1.0, size=n), n, S0)
    if cand is not None:
        cands.append(cand)

    # Anti-palindromic random structures strongly suppress the central peak.
    half = n // 2
    if half > 0:
        r = rng.normal(0.0, 1.0, size=half)
        s = np.zeros(n, dtype=np.float64)
        s[:half] = r
        s[n - half:] = -r[::-1]

        st = np.tanh(s)
        for A in (0.45, 0.75, 0.92):
            cand = _safe_positive(1.0 + A * st, n, S0)
            if cand is not None:
                cands.append(cand)

        sg = np.sign(s)
        for A in (0.55, 0.85, 0.95):
            cand = _safe_positive(1.0 + A * sg, n, S0)
            if cand is not None:
                cands.append(cand)

    # Quasi-periodic / low-discrepancy perturbations.
    for alpha in (0.6180339887, 0.3819660113, 0.2000000000, 0.8000000000):
        frac = np.mod(idx * alpha, 1.0)
        ss = 2.0 * frac - 1.0
        for A in (0.45, 0.75, 0.92):
            cand = _safe_positive(1.0 + A * ss, n, S0)
            if cand is not None:
                cands.append(cand)

    # Fourier modes.
    if n > 1:
        for k in (1, 2, 3, 5, 8, 13):
            phase = rng.uniform(0.0, 2.0 * np.pi)
            theta = 2.0 * np.pi * k * idx / float(n) + phase
            for A in (0.40, 0.70, 0.90):
                cand = _safe_positive(1.0 + A * np.sin(theta), n, S0)
                if cand is not None:
                    cands.append(cand)
                cand = _safe_positive(1.0 - A * np.cos(theta), n, S0)
                if cand is not None:
                    cands.append(cand)

    # Edge-enhanced shapes.
    t = 2.0 * idx / float(n - 1) - 1.0 if n > 1 else np.zeros(n)
    for A in (0.30, 0.70, 1.05):
        cand = _safe_positive(1.0 + A * t * t, n, S0)
        if cand is not None:
            cands.append(cand)

    # Smooth random low-pass positive sequences from a random spectrum.
    Nrand = _next_pow2(n)
    for _ in range(3):
        m = Nrand // 2
        spec = np.zeros(m + 1, dtype=np.complex128)
        spec[0] = 5.0
        if m > 1:
            spec[1:m] = rng.normal(0.0, 1.0, size=m - 1) + 1j * rng.normal(0.0, 1.0, size=m - 1)
        spec[m] = float(abs(spec[m]))
        a = np.fft.irfft(spec, n=Nrand)[:n]
        cand = _safe_positive(np.abs(a) + 0.1, n, S0)
        if cand is not None:
            cands.append(cand)

    if not cands:
        cands.append(_safe_positive(np.ones(n, dtype=np.float64), n, S0))
    return cands


def _local_lse(
    a0,
    n,
    S0,
    N,
    stop_deadline,
    rng,
    label,
    floor,
    tau_init=0.45,
    tau_min=0.012,
    lr=0.09,
    max_step=0.9,
    b1=0.90,
    b2=0.99,
    eps=1e-8,
    restart_secs=40.0,
    restart_noise=0.35,
    step_noise=0.0,
    tau_decay=0.999,
):
    current = _rescale_with_floor(a0, S0, floor)
    start = time.monotonic()
    if stop_deadline <= start:
        return current, np.inf

    best_a = current.copy()
    best_score = np.inf
    last_improve = start
    last_print = start
    printed_score = np.inf

    mom1 = np.zeros(n, dtype=np.float64)
    mom2 = np.zeros(n, dtype=np.float64)
    b1p = 1.0
    b2p = 1.0
    tau = float(tau_init)

    out_len_c = 2 * n - 1

    while time.monotonic() < stop_deadline:
        c, fa = _rfft_square_conv(current, N, out_len_c)
        if c.size == 0:
            break

        score = float(np.max(c))
        if math.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = current.copy()
            last_improve = time.monotonic()
            if best_score < printed_score - 0.002 or time.monotonic() - last_print > 20.0:
                print(f"{label} fft_score={best_score:.6f} elapsed={time.monotonic() - start:.1f}s")
                printed_score = best_score
                last_print = time.monotonic()

        if not math.isfinite(score):
            current = best_a.copy()
            mom1.fill(0.0)
            mom2.fill(0.0)
            b1p = 1.0
            b2p = 1.0
            last_improve = time.monotonic()
            continue

        tau = max(float(tau_min), tau * float(tau_decay))
        p = _make_p(c, tau)
        q = _corr_with_p(current, p, fa, N, n)
        if q.size != n:
            q = np.zeros(n, dtype=np.float64)

        mc = float(np.dot(p, c))
        v = q - mc / S0
        wmean = float(np.dot(current, v))
        if math.isfinite(wmean):
            v = v - wmean / S0

        b1p *= b1
        b2p *= b2
        mom1 = b1 * mom1 + (1.0 - b1) * v
        mom2 = b2 * mom2 + (1.0 - b2) * (v * v)
        mhat = mom1 / (1.0 - b1p)
        vhat = mom2 / (1.0 - b2p)
        step = lr * mhat / (np.sqrt(vhat) + eps)

        if step_noise > 0.0:
            step = step + step_noise * rng.normal(0.0, 1.0, size=n)

        step = np.clip(step, -max_step, max_step)
        sp = float(np.dot(current, step))
        if math.isfinite(sp):
            step = step - sp / S0
        step = np.clip(step, 2.0 * max_step, -2.0 * max_step)

        current = current * np.exp(-step)
        current = _rescale_with_floor(current, S0, floor)

        if time.monotonic() - last_improve > restart_secs:
            r = rng.random()
            if r < 0.75:
                current = best_a * np.exp(rng.normal(0.0, restart_noise, size=n))
            else:
                rand = rng.exponential(1.0, size=n)
                current = 0.6 * best_a + 0.4 * rand
            current = _rescale_with_floor(current, S0, floor)
            mom1.fill(0.0)
            mom2.fill(0.0)
            b1p = 1.0
            b2p = 1.0
            tau = min(0.9, max(tau_init * 0.5, tau * 1.5))
            last_improve = time.monotonic()

    return best_a, best_score


def _local_block_search(a0, n, S0, N, stop_deadline, rng, ref_best_score, floor):
    current = _rescale_with_floor(a0, S0, floor)
    c, _ = _rfft_square_conv(current, N, 2 * n - 1)
    cur_score = float(np.max(c)) if c.size else np.inf

    best_score = ref_best_score
    best_a = current.copy()
    if math.isfinite(cur_score) and cur_score < best_score:
        best_score = cur_score

    start = time.monotonic()
    last_print = start
    printed_score = np.inf
    no_improve = 0

    low = max(2, n // 3000)
    high = max(low + 1, n // 8)
    shift_bound = max(1, n // 8)

    while time.monotonic() < stop_deadline:
        width = int(rng.integers(low, high))
        startpos = int(rng.integers(0, max(1, n - width)))
        cand = current.copy()

        if rng.random() < 0.06:
            shift = int(rng.integers(-shift_bound, shift_bound + 1))
            cand = np.roll(cand, shift)
        else:
            cand[startpos:startpos + width] *= np.exp(rng.normal(0.0, 0.12, size=width))
            if rng.random() < 0.15:
                cand *= np.exp(rng.normal(0.0, 0.04, size=n))

        cand = _rescale_with_floor(cand, S0, floor)
        c, _ = _rfft_square_conv(cand, N, 2 * n - 1)
        sc = float(np.max(c)) if c.size else np.inf

        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = cand.copy()
            current = cand
            no_improve = 0
            if best_score < printed_score - 0.002 or time.monotonic() - last_print > 30.0:
                print(f"block search fft_score={best_score:.6f} elapsed={time.monotonic() - start:.1f}s")
                printed_score = best_score
                last_print = time.monotonic()
        else:
            no_improve += 1
            if no_improve > 5000:
                current = best_a.copy()
                no_improve = 0

    return best_a, best_score


def _sanitize_list(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        arr = np.ones(64, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 0.0:
        arr = np.ones_like(arr, dtype=np.float64)
        s = float(arr.size)
    if s < 0.01:
        arr = arr * (0.02 / s)
        arr = np.clip(arr, 0.0, 1000.0)
    return [float(x) for x in arr]
