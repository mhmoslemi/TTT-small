    n = len(a)
    # p has length 2n-1
    # b = reverse of a, zero-padded to 2n-1
    b = np.zeros(2*n-1)

    g = 2 * conv[n-1 : 2*n-1]

import time
from scipy.signal import fftconvolve

def fft_convolve_self(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def compute_grad(a, p):
    """Compute gradient g = 2 * correlate(p, a, mode='valid') using FFT."""
    # correlate(p, a, mode='valid') = fftconvolve(p, a[::-1], mode='valid')
    result = fftconvolve(p, a[::-1], 'valid')
    return 2.0 * result
