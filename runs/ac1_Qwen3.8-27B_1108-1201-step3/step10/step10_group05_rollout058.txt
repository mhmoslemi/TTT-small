def fast_convolve(a):

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # conv_ap = ifft(A * P) → gives (a*p)[k] for k=0..N-1, we need k=0..n-1
    conv_ap = np.fft.irfft(A * P, N)[:n]
    # corr_ap[j] = sum_m a[m]*p[j+m]
    # This is the "anti-convolution": ifft(conj(A) * P) ... let me verify
    # ifft(conj(A)*P)[j] = (1/N) sum_k conj(A[k])*P[k]*exp(2πikj/N)
    # A[k] = sum_m a[m] exp(-2πimk/N)
    # conj(A[k]) = sum_m a[m] exp(2πimk/N)
    # conj(A[k])*P[k] = (sum_m a[m] exp(2πimk/N)) * (sum_l p[l] exp(-2πil k/N))
    # = sum_{m,l} a[m]*p[l] * exp(2πik(m-l)/N)
    # ifft of that at j: (1/N) sum_k (sum_{m,l} a[m]*p[l]*exp(2πik(m-l)/N)) * exp(2πikj/N)
    # = sum_{m,l} a[m]*p[l] * (1/N) sum_k exp(2πik(m-l+j)/N)
    # = sum_{m,l} a[m]*p[l] * δ_{m-l+j, 0} (mod N)
    # = sum_{m,l} a[m]*p[l] where l = m+j (mod N)
    # = sum_m a[m]*p[m+j] (for m+j < N, which is our case)
    # ✓ So ifft(conj(A)*P)[j] = sum_m a[m]*p[m+j]
    corr_ap = np.fft.irfft(np.conj(A) * P, N)[:n]
    grad = conv_ap + corr_ap
    return obj, grad

def lbgfs_smoothed_optimize(a0, time_limit, beta_start=50.0, beta_end=500.0, label=""):
    """
    Optimize using L-BFGS on the log-sum-exp smoothed objective.
    Uses softmax parameterization to handle the simplex constraint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    t0 = time.monotonic()
    
    # Smoothing schedule: anneal beta from beta_start to beta_end
    total_iters_estimate = max(int(time_limit / 0.5), 10)  # rough estimate
    beta_schedule = np.geomspace(beta_start, beta_end, total_iters_estimate)
    
    it = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Get current beta from schedule
        idx = min(it - 1, len(beta_schedule) - 1)
        beta = beta_schedule[idx]
        
        # Evaluate current
        conv = fast_convolve(a)
        cur_obj = float(conv.max())
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute smoothed objective and gradient
        sm_obj, grad = smoothed_objective_and_grad_fftp(a, beta)
        
        # Add regularization towards uniform (very small) to avoid boundary issues
        # grad_reg = grad - 0.01 * (a - 1/n) ... skip for now
        
        # Line search + gradient step on the simplex
        best_step_obj = sm_obj
        best_step_a = a.copy()
        step_sizes = [1.0/beta * 10, 1.0/beta * 5, 1.0/beta * 2, 1.0/beta, 1.0/beta * 0.5, 1.0/beta * 0.1, 0.01, 0.005, 0.001]
        
        for step in step_sizes:
            a_new = a - step * grad
            a_new = simplex_project(a_new)
            new_sm_obj, _ = smoothed_objective_and_grad_fftp(a_new, beta)
            if new_sm_obj < best_step_obj - 1e-10:
                best_step_obj = new_sm_obj
                best_step_a = a_new
        
        a = best_step_a
        
        # Nesterov momentum (applied every few steps)
        # (simplified: just use the step directly)
        
        # Check for convergence
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LBF {label} it={it} ({elapsed:.0f}s): obj={cur_obj:.8f}, sm_obj={sm_obj:.8f}, beta={beta:.1f}, best={best_obj:.8f}")
        
        if it > 200 and cur_obj > best_obj + 1e-10:
            stuck_count += 1
            if stuck_count > 5:
                # Reset to best
                a = best_a.copy()
                stuck_count = 0
        else:
            stuck_count = 0
    
    # Final evaluation
    conv = fast_convolve(a)
    final_obj = float(conv.max())
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    return best_a, best_obj

def accelerated_fictitious_play(a0, time_limit, solver=None, label=""):
    """
    Fictitious play with LP best response, adaptive extrapolation, and line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    b_prev = a.copy()
    momentum = 0.0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Solve LP best response
        b_new, M_lp = solve_lp_best_response(a, solver)
        if b_new is None:
            b_new = a.copy()
        
        # Extrapolation (Nesterov-like)
        diff = b_new - b_prev
        if momentum > 0.01:
            b_extrap = b_new + momentum * diff
            b_extrap = np.maximum(b_extrap, 0)
            se = b_extrap.sum()
            if se > 1e-15:
                b_extrap = b_extrap / se
            else:
                b_extrap = b_new
        else:
            b_extrap = b_new
        
        b_prev = b_new.copy()
        # Update momentum
        momentum = 0.9 * momentum + 0.1 * min(1.0, 50.0 / (iters + 10))
        
        # Line search: find best alpha in [0, 1] for a_new = (1-alpha)*a + alpha*b_extrap
        best_alpha = 0.0
        best_obj = float(fast_convolve(a).max())
        
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1 - alpha) * a + alpha * b_extrap
            a_cand = np.maximum(a_cand, 0)
            sc = a_cand.sum()
            if sc < 1e-15:
                continue
            a_cand = a_cand / sc
            obj = float(fast_convolve(a_cand).max())
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_alpha = alpha
        
        if best_alpha > 0:
            a_new = (1 - best_alpha) * a + best_alpha * b_extrap
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            score = 2.0 * n * float(fast_convolve(a_new).max())
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
            a = a_new
        
        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AFAFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_lp={M_lp:.10f}")
    
    return best_a, best_score, iters

def peak_aware_coord_descent(a0, time_limit, label=""):
    """
    Coordinate descent that prioritizes coordinates contributing most to the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Order coordinates by their contribution to the peak
        peak_idx = conv.argmax()
        priority = np.zeros(n)
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        for i in range(i_start, i_end + 1):
            j = peak_idx - i
            priority[i] = a[i] * a[j]
        
        # Also consider near-peak positions
        for k in range(len(conv)):
            if conv[k] > 0.95 * M and k != peak_idx:
                i_start2 = max(0, k - n + 1)
                i_end2 = min(n - 1, k)
                for i in range(i_start2, i_end2 + 1):
                    j = k - i
                    priority[i] += 0.5 * a[i] * a[j]
        
        order = np.argsort(-priority)
        
        for i in order:
            ai = a[i]
            if ai < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            
            best_delta = 0.0
            best_new_max = M
            
            # Generate candidates
            candidates = [0.0, -ai * 0.99, -ai * 0.5, -ai * 0.1]
            
            # Where affected lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.1 < d < ai * 0.1:
                        candidates.append(d)
            
            # Crossings of top affected lines
            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.1 < d < ai * 0.1:
                            candidates.append(d)
            
            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
        
        if not improved and sweeps > 50:
            break
    
    return best_a, best_score, sweeps

def fine_polish(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)
            
            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            
            best_delta = 0.0
            best_new_max = M
            
            # Fine fractional moves
            for factor in [0.9999, 0.9995, 0.999, 0.998, 0.997, 0.995, 0.99, 0.985, 0.98,
                          1.0001, 1.0005, 1.001, 1.002, 1.003, 1.005, 1.01, 1.02]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved and sweeps > 20:
            break
    
    return best_a, best_score

def make_monotone_init(n, alpha=1.0):
    """Create a monotonically decreasing initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = (1.0 - t) ** alpha
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def make_symmetric_bowl(n, alpha=2.0):
    """Create a symmetric bowl-shaped initialization (large in middle)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + alpha * t * t
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def make_anti_bowl(n, alpha=2.0):
    """Create an anti-bowl (large at edges, small in middle)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 - 0.5 * alpha * t * t
    a = np.maximum(a, 1e-6)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_perturb(a0, rng, strength=0.05):
    """Perturb by adding structured (smooth) noise."""
    n = len(a0)
    a = a0.copy()
    # Generate smooth random perturbation
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.linspace(0, 2 * np.pi * freq, n)
    smooth_noise = np.sin(t + phase) * 0.5 + np.sin(2 * t + 2 * phase) * 0.3 + rng.standard_normal(n) * 0.3
    smooth_noise = smooth_noise / (np.abs(smooth_noise).max() + 1e-15)
    a_new = a * (1.0 + strength * smooth_noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

        shifted = np.roll(best_seq / best_seq.sum(), shift)
        inits.append(shifted)
    
    for ii, init in enumerate(inits):
        rem = deadline - time.monotonic() - 60
        if rem < 20:
            break
        t7 = min(rem * 0.25, 40.0)
        a7, s7, _ = peak_aware_coord_descent(init, t7, label=f"p7_i{ii}")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, f"p7_i{ii}")
    
    # === Phase 8: Final intensive polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Peak-aware CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t8a = min(rem * 0.4, 80.0)
            a8a, s8a, _ = peak_aware_coord_descent(best_seq, t8a, label="p8a")
            best_seq, best_score = check_and_update(a8a, best_seq, best_score, "p8a")
        
        # Fine polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t8b = min(rem * 0.5, 60.0)
            a8b, s8b = fine_polish(best_seq, t8b, label="p8b")
            best_seq, best_score = check_and_update(a8b, best_seq, best_score, "p8b")
        
        # One more peak-aware CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a8c, s8c, _ = peak_aware_coord_descent(best_seq, rem * 0.6, label="p8c")
            best_seq, best_score = check_and_update(a8c, best_seq, best_score, "p8c")
        
        # Final fine
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a8d, s8d = fine_polish(best_seq, rem * 0.7, label="p8d")
            best_seq, best_score = check_and_update(a8d, best_seq, best_score, "p8d")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
