import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT (full convolution)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

    Solve: maximize sum(b) s.t. convolve(a_fixed, b)[k] <= M for k in active_peaks,
    b >= 0. If active_peaks is None, use all peaks.
    Returns (b_normalized, None) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_peaks is not None and len(active_peaks) < size:
        # Only use active constraints for speed
        T_sub = np.zeros((len(active_peaks), n))
        for ii, k in enumerate(active_peaks):
            T_sub[ii] = np.convolve(a_fixed, np.eye(n)[k % n]) if False else _conv_row(a_fixed, k)
    else:
        T_sub = build_conv_matrix(a_fixed)
    
    constraints = [T_sub @ b_var <= M * np.ones(len(T_sub)), cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M * cp.Constant(1)), constraints)
    
    # Actually, we want to just find a feasible point (sum(b)=1, conv <= M)
    # Then the best response is the one that minimizes max conv given sum=1
    # Let me reformulate: minimize M2 s.t. conv(a,b)[k] <= M2, sum(b)=1, b>=0
    
    M2_var = cp.Variable()
    constraints = [T_sub @ b_var <= M2_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M2_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(M2_var.value)
    except Exception:
        pass
    
    # Try alternate solvers
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    return b_val, float(M2_var.value)
        except Exception:
            continue
    return None, None


def _conv_row(a_fixed, k):
    """Get the k-th row of the convolve matrix (i.e., convolve(a_fixed, e_k))."""
    n = len(a_fixed)
    size = 2 * n - 1
    row = np.zeros(n)
    # convolve(a_fixed, e_j)[k] = a_fixed[k-j] if 0<=k-j<n
    # We want row such that row @ b = convolve(a_fixed, b)[k]
    # convolve(a_fixed, b)[k] = sum_j a_fixed[k-j] * b[j] for valid k-j
    # So row[j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            row[j] = a_fixed[idx]
    return row


def fista_lp_alternate(a0, time_limit, solver, n_active_frac=0.999):
    """
    FISTA-style LP alternating:
    - At each step, solve LP to find best response b given current y
    - Update using FISTA momentum
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    b_curr = fast_convolve(a)
    best_score = 2.0 * n * b_curr.max()
    t0 = time.monotonic()
    k_momentum = 1.0
    stall_iter = 0
    rng = np.random.default_rng(12345)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution at y
        b_y = fast_convolve(y)
        max_b_y = b_y.max()
        
        # Find active peaks (top fraction)
        threshold = max_b_y * n_active_frac
        active = np.where(b_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b_y))])
        # Limit active set size for LP speed
        if len(active) > 200:
            # Keep the largest 200
            vals = b_y[active]
            top_idx = active[np.argsort(vals)[::-1][:200]]
        else:
            top_idx = active
        
        # Solve LP for best response
        b_new, M_val = solve_lp_response(y, M_val=None, solver=solver, active_peaks=top_idx)
        
        if b_new is None:
            # Fallback: use subgradient step
            g = _compute_subgrad(y, top_idx)
            eta = 0.03 / (1 + 0.1 * it)
            log_a = np.log(np.maximum(y, 1e-300))
            log_a -= eta * (g - g.min())
            log_a -= log_a.max()
            b_new = np.exp(log_a)
            b_new = b_new / b_new.sum()
        
        # FISTA update
        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_momentum * k_momentum))
        theta = (1.0 + k_momentum) / (1.0 + k_new)
        
        y_new = (b_new + (k_momentum / k_new) * (y - a)) 
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_new.copy()
            sy = 1.0
        else:
            y_new = y_new / sy
        
        a = b_new.copy()
        y = y_new
        k_momentum = k_new
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_iter = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_iter = 0
        else:
            stall_iter += 1
        
        if stall_iter >= 80:
            # Perturb to escape local minimum
            noise = rng.standard_normal(n) * 0.01
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            a = y.copy()
            k_momentum = 1.0
            stall_iter = 0
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} |active|={len(top_idx)}")
    
    return best_a, best_score


def _compute_subgrad(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        row = _conv_row(a, k)
        # subgradient of conv(a,a)[k] w.r.t. a is: a[k-a_j] pattern
        # Actually for the max, we average subgradients over active set
        # d/dx_i of sum_j x_j * x[k-j] = x[k-i] + (if i != k-i) x[k-i] ... 
        # For full convolve: d/dx_i of C[k] = x[k-i] + x[i+k-(k-i)] ... 
        # Let me just use the row: C[k] = row @ a, so dC[k]/da = row (for one-sided)
        # But C[k] = sum_j a[j]*a[k-j], so dC[k]/da[i] = a[k-i] + a[i+k-i]... 
        # Actually C[k] = sum_j a[j] * a[k-j]. dC[k]/da[i] = a[k-i] (from j=i term) 
        #   + a[i] * 0 ... no wait.
        # C[k] = sum_j a[j] * a[k-j]. Only terms where j = i or k-j = i matter.
        # j = i: term is a[i] * a[k-i], derivative w.r.t. a[i] is a[k-i]
        # k-j = i, i.e., j = k-i: term is a[k-i] * a[i], derivative w.r.t. a[i] is a[k-i]
        # If i = k-i (i.e., 2i = k), these are the same term: a[i]^2, derivative is 2*a[i]
        # So dC[k]/da[i] = a[k-i] if k-i in [0,n-1], plus a[k-i] if i != k-i and k-i in [0,n-1]
        # Simplified: dC[k]/da[i] = a[k-i] + (a[k-i] if i != k-i else a[i]) ... 
        # Actually: if 2i == k, d/d a[i] of a[i]^2 = 2*a[i]. And k-i = i, so a[k-i] = a[i].
        # If 2i != k, d/d a[i] of (a[i]*a[k-i] + a[k-i]*a[i]) = 2*a[k-i].
        # Wait no. C[k] = sum_j a[j]*a[k-j]. The terms involving a[i] are:
        #   j=i: a[i]*a[k-i]
        #   j=k-i: a[k-i]*a[i]  (if k-i != i)
        # So if k-i != i: C[k] contains 2*a[i]*a[k-i], derivative = 2*a[k-i]
        # If k-i == i (i.e., k=2i): C[k] contains a[i]^2, derivative = 2*a[i] = 2*a[k-i]
        # So in all cases: dC[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
        # Wait, that's the derivative treating a as the variable and computing C[k] as a function.
        # But C[k] = sum_j a[j]*a[k-j] is a function of all a's.
        # ∂C[k]/∂a[i]: the terms where a[i] appears are j=i and j=k-i.
        # ∂/∂a[i] [a[i]*a[k-i]] = a[k-i] (treating a[k-i] as constant)
        # ∂/∂a[i] [a[k-i]*a[i]] = a[k-i] (treating a[k-i] as constant)
        # But if i == k-i, it's just one term: ∂/∂a[i] [a[i]^2] = 2*a[i] = 2*a[k-i]
        # If i != k-i: two terms, each gives a[k-i], total 2*a[k-i]
        # So always: ∂C[k]/∂a[i] = 2*a[k-i] if 0 <= k-i < n, else 0
        
        if 0 <= k - 0 < 2*n:  # always true for our range
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    g[i] += 2.0 * a[ki]
    
    g /= max(1, len(active_indices) * 2.0)  # average and normalize
    return g


def mirror_descent_fast(a0, time_limit, eta0=0.05, active_frac=0.999):
    """Fast mirror descent with Nesterov."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    prev_a = a.copy()
    prev_g = None
    rng = np.random.default_rng(999)
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    g[i] += 2.0 * a[ki]
        g /= max(1, len(active) * 2.0)
        
        # BB step size
        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if s_dot_s > 1e-15 and abs(s_dot_y) > 1e-15:
                eta_bb = s_dot_s / abs(s_dot_y)
                eta = 0.7 * eta0 + 0.3 * eta_bb
            else:
                eta = eta0 * (0.999 ** (it % 1000))
        else:
            eta = eta0 / (1 + 0.01 * it)
        
        prev_a = a.copy()
        prev_g = g.copy()
        
        # Mirror descent update in log space
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys
        
        a = a_new
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 500:
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter=5000):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    z = np.log(np.maximum(a_init, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    # Check convexity: for the logsumexp of conv, we need to be careful
    # The objective is: T * log(sum(exp(conv(a,a)/T)))
    # which is a smoothed version of max(conv(a,a))
    # Gradient w.r.t. a: (1/T) * sum_k exp(conv[k]/T) * d(conv[k])/da / sum_k exp(conv[k]/T)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        def func(z_vec):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_vec = _softmax(z_vec)
            b = fast_convolve(a_vec)
            b_shifted = b - b.max()
            logsumexp = b.max() + np.log(np.exp(b_shifted).sum())
            return (2.0 * n / T) * logsumexp
        
        def jac(z_vec):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_vec)
            a_vec = _softmax(z_vec)
            b = fast_convolve(a_vec)
            b_shifted = b - b.max()
            exp_b = np.exp(b_shifted)
            w = exp_b / exp_b.sum()
            
            # Gradient of logsumexp(conv(a,a)/T) w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                wk = 2.0 * w[k]
                for i in range(n):
                    ki = k - i
                    if 0 <= ki < n:
                        grad_a[i] += wk * a_vec[ki]
            
            # Convert to z-space (softmax parameterization)
            S = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - S)
            return (2.0 * n / T) * grad_z
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
        
        elapsed = time.monotonic() - t0
        a_check = _softmax(z)
        sc = 2.0 * n * fast_convolve(a_check).max()
        print(f"    L-BFGS T={T}: score={sc:.8f} ({elapsed:.0f}s)")
    
    return _softmax(z)


def _softmax(z):
    z = z - z.max()
    exp_z = np.exp(np.maximum(z, -700))
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def peak_directed_cd(a0, time_limit, n_steps=50000):
    """
    Peak-directed coordinate descent:
    For the dominant peak, identify contributing coordinates and try to
    reduce their values while redistributing mass to "quiet" positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find all near-peak positions
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for pk in peaks[:5]:  # Try top 5 peaks
            if improved:
                break
            # Find contributing coordinates for this peak
            # C[pk] = sum_j a[j]*a[pk-j], contributing pairs are (j, pk-j)
            contribs = np.zeros(n)
            for j in range(n):
                kj = pk - j
                if 0 <= kj < n:
                    contribs[j] = a[j] * a[kj]
            
            # Sort by contribution descending
            order = np.argsort(contribs)[::-1]
            top_contrib = order[:min(10, n)]
            
            # Find "quiet" positions (low contribution to all peaks)
            quiet_score = np.zeros(n)
            for pk2 in peaks[:5]:
                for j in range(n):
                    kj = pk2 - j
                    if 0 <= kj < n:
                        quiet_score[j] += a[j] * a[kj]
            
            # Quiet positions are those with low score and non-zero a
            quiet_threshold = np.percentile(quiet_score, 50)
            quiet_positions = np.where(quiet_score < quiet_threshold)[0]
            quiet_positions = quiet_positions[a[quiet_positions] > 1e-10]
            
            if len(quiet_positions) == 0:
                continue
            
            # Try moving mass from top contributors to quiet positions
            for src in top_contrib[:5]:
                if a[src] < 1e-10:
                    continue
                for dst in quiet_positions[:5]:
                    for frac in [0.001, 0.003, 0.005, 0.01]:
                        delta = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation
                a_trial = a.copy()
                idx = rng.integers(0, n, size=min(20, n))
                for i in idx:
                    a_trial[i] *= (1 + rng.standard_normal() * 0.01)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                else:
                    if stall > 1000:
                        break
        else:
            stall = 0
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def spectral_shaping(a0, time_limit, n_iters=200):
    """
    Spectral shaping: modify the power spectrum to reduce time-domain peaks.
    Works by identifying frequency components that contribute to peaks and
    attenuating them, then projecting back to non-negative.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Compute FFT of current signal
        fft_size = 1 << (2 * n - 1).bit_length() if (2*n-1) > 1 else 2
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        
        # The time-domain peak locations tell us which frequencies to attenuate
        # Peak of IFFT(power) corresponds to DC + all frequencies aligned constructively
        # We want to spread the power more evenly
        # Strategy: apply a "notch" at frequencies where the phase alignment causes peaks
        
        # Simpler approach: use the peak locations to compute a "target" spectrum
        # that has lower peak in time domain
        # Use iterative spectral balancing:
        # 1. Compute current time-domain signal (which is the convolution)
        # 2. Find peak
        # 3. In frequency domain, the peak at time t corresponds to all frequencies
        #    having the same phase at t. We can reduce the "coherence" by adding
        #    random phase or by spectral flattening.
        
        # Practical approach: random spectral perturbation + projection
        noise_freq = rng.standard_normal(len(power)) * 0.01 + 1j * rng.standard_normal(len(power)) * 0.01
        A_new = A * (1 + noise_freq / np.maximum(np.abs(A), 1e-10))
        
        # Or: spectral flattening (reduce dynamic range of power spectrum)
        p_min = np.percentile(power[power > 1e-10], 5) if np.any(power > 1e-10) else 1e-10
        p_max = np.percentile(power[power > 1e-10], 95) if np.any(power > 1e-10) else 1.0
        if p_max > p_min:
            # Compress dynamic range
            power_new = p_min + (p_max - p_min) * (power - p_min) / (p_max - p_min) * 1.0
            # This is identity... let's do actual compression
            power_compressed = p_min + (p_max - p_min) * np.sqrt((power - p_min) / (p_max - p_min))
            # Blend
            alpha = 0.05
            power_new = (1 - alpha) * power + alpha * power_compressed
            # Keep phase from original
            phase = np.angle(A)
            A_new = np.sqrt(power_new) * np.exp(1j * phase)
        
        # Inverse FFT and project to non-negative
        a_trial = irfft(A_new, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-10:
            continue
        a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_score


def structured_init(n, rng, style=0):
    """Create structured initial sequence with various shapes."""
    x = np.linspace(0, 1, n)
    if style == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 1:
        # Gaussian centered
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif style == 2:
        # Sine^2
        a = np.sin(np.pi * x) ** 2
    elif style == 3:
        # Super-Gaussian
        a = np.exp(-8 * (x - 0.5) ** 4)
    elif style == 4:
        # Double bump
        a = 0.5 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase A: Mirror descent (fast, broad)
    Phase B: FISTA-LP alternating (main workhorse)
    Phase C: LogSumExp L-BFGS (smooth refinement)
    Phase D: Peak-directed coordinate descent (targeted)
    Phase E: Spectral shaping + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n = len(parent)
            p_score = evaluate_local(parent)
            print(f"Parent: n={n}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
        n = 1200
        best_seq = structured_init(n, rng, style=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n}, score={best_score:.8f}")
    
    solver = get_lp_solver()
    print(f"LP solver: {solver}, n={len(best_seq)}, initial score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase A: Mirror Descent (fast broad optimization) ===
    rem = deadline - time.monotonic()
    t_a = min(rem * 0.20, 300.0)
    if t_a > 20:
        print(f"\n=== Phase A: Mirror Descent ({t_a:.0f}s) ===")
        a_md, _ = mirror_descent_fast(best_seq, t_a, eta0=0.04, active_frac=0.999)
        try_update(a_md, "mirror_descent")
    
    # === Phase B: FISTA-LP Alternating (main workhorse) ===
    rem = deadline - time.monotonic()
    t_b = min(rem * 0.35, 600.0)
    if t_b > 30:
        print(f"\n=== Phase B: FISTA-LP ({t_b:.0f}s) ===")
        a_lp, _ = fista_lp_alternate(best_seq, t_b, solver=solver, n_active_frac=0.999)
        try_update(a_lp, "fista_lp")
    
    # === Phase C: LogSumExp L-BFGS ===
    rem = deadline - time.monotonic()
    t_c = min(rem * 0.25, 400.0)
    if t_c > 30:
        print(f"\n=== Phase C: L-BFGS ({t_c:.0f}s) ===")
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lb = logsumexp_lbgfs(best_seq, t_c, T_stages=T_stages, maxiter=3000)
        try_update(a_lb, "lbgfs")
    
    # === Phase D: Peak-directed Coordinate Descent ===
    rem = deadline - time.monotonic()
    t_d = min(rem * 0.12, 180.0)
    if t_d > 15:
        print(f"\n=== Phase D: Peak CD ({t_d:.0f}s) ===")
        a_cd, _ = peak_directed_cd(best_seq, t_d, n_steps=80000)
        try_update(a_cd, "peak_cd")
    
    # === Phase E: Spectral shaping + second LP pass ===
    rem = deadline - time.monotonic()
    t_e = min(rem * 0.08, 120.0)
    if t_e > 10:
        print(f"\n=== Phase E: Spectral + LP polish ({t_e:.0f}s) ===")
        t_spec = t_e * 0.4
        t_lp2 = t_e * 0.6
        a_spec = spectral_shaping(best_seq, t_spec)
        try_update(a_spec, "spectral")
        if t_lp2 > 10:
            a_lp2, _ = fista_lp_alternate(best_seq, t_lp2, solver=solver, n_active_frac=0.9999)
            try_update(a_lp2, "lp_polish")
    
    # === Phase F: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t_f = min(rem, 60.0)
    if t_f > 5:
        print(f"\n=== Phase F: Final CD ({t_f:.0f}s) ===")
        a_fin, _ = peak_directed_cd(best_seq, t_f, n_steps=30000)
        try_update(a_fin, "final_cd")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
