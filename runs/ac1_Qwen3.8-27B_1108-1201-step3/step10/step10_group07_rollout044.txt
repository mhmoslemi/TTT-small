import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def find_lp_solver():

        try:

def build_conv_matrix(a):
    n = len(a)
    # T is (2n-1) x n, T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # So conv(a,b)[k] = sum_j a[k-j]*b[j] = (T @ b)[k]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver, prev_b=None):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    If prev_b is provided, use it as warm start.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if prev_b is not None and solver == cp.PDLP:
            prob.solve(solver=solver, warm_start=True,
                      pdlp_params={"duo_solution": prev_b})
        else:
            prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def subgradient_descent(a0, time_limit, lr_init=0.015, top_k=30, verbose=True):
    """
    Subgradient descent with multi-peak weighting, momentum, and simplex projection.
    Uses Nesterov acceleration for faster convergence.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)  # Momentum
    a_y = a.copy()  # Nesterov lookahead
    beta = 0.9  # Momentum coefficient
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 150:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-12:
                break
        
        # Compute subgradient at Nesterov point
        conv_y = fast_convolve(a_y)
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv_y.copy()
        else:
            peaks = np.argpartition(conv_y, -k_act)[-k_act:]
            pv = conv_y[peaks]
        
        # Softmax weighting
        temp = 0.005 * max(conv_y.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a_y[k - i_end:k - i_start + 1][::-1]
        
        # Update with momentum
        v = beta * v + lr * sg
        a_new = a - v
        a_new = simplex_project(a_new)
        
        # Nesterov update
        t_k = (1 + np.sqrt(1 + 4 * beta * beta * it)) / 2
        a_y = a_new + ((t_k - 1) / (t_k + 1)) * (a_new - a)
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def support_prune_and_reopt(a0, n_target, time剩余, solver, verbose=True):
    """
    Prune the support of a to n_target non-zero elements,
    then re-optimize using LP on the reduced problem.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()

    # Find top n_target coordinates by value
    if n_target < n:

        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 15))
        t = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # Sparse comb-like
        a = np.zeros(n)
        n_peaks = int(n / rng.integers(8, 20))
        positions = rng.choice(n, size=n_peaks, replace=False).astype(int)
        a[positions] = 1.0
        if a.sum() < 1:
            a = np.ones(n) / n
    elif init_type == 6:
        # Random smooth
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n, dtype=np.float64)
        a = np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * freq * t / n + phase), 0)
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
