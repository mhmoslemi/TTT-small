Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Minimax power iteration (Blahut-Arimoto style) with FFT-accelerated H(p)@v
2. Gaussian and structured initializations
3. Smart 1D line-search coordinate descent for polishing
4. Multi-length exploration
5. Random restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def hmv(p, v):
    """Compute H(p) @ v where H[i,j] = p[i+j]. Efficient via FFT."""
    n = len(v)
    p_rev = p[::-1]
    # convolve p_rev (len 2n-1) with v (len n) -> length 3n-2
    full = convolve_ab(p_rev, v)
    return full[n-1:2*n-1][::-1]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def gaussian_init(n, sigma_frac):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    a = a / a.sum()
    return a

def super_gaussian_init(n, sigma_frac, power):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-np.abs((x - center) / sigma) ** (2 * power))
    a = a / a.sum()
    return a

        a = np.ones(n) / n
    else:
        a = a / s

    p = conv_a.copy()
    p = p / p.sum()
    
    # Initialize v
    v = a.copy()
    v = v / np.linalg.norm(v)
    
    best_a = a.copy()
    best_score = 2.0 * n * conv_a.max()
    t0 = time.monotonic()
    
    damping = damping_init
    it = 0
    prev_lambda = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Inner: power iteration on H(p)
        for _ in range(inner_iters):
            Hv = hmv(p, v)
            Hv = np.maximum(Hv, 0)  # PF: should be non-negative
            norm = np.linalg.norm(Hv)
            if norm < 1e-15:
                break
            v_new = Hv / norm
            v = v_new
        
        # Compute lambda_max estimate
        Hv = hmv(p, v)
        lambda_est = float(np.dot(v, Hv))
        
        # Update p from v
        conv_v = fast_convolve(v)
        p_new = conv_v / conv_v.sum()
        
        # Track best a
        a_cand = v / v.sum()
        score = 2.0 * n * fast_convolve(a_cand).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_cand.copy()
        
        # Update p with damping
        p = (1 - damping) * p + damping * p_new
        p = np.maximum(p, 0)
        p = p / p.sum()
        
        # Adaptive damping
        if prev_lambda is not None:
            change = abs(lambda_est - prev_lambda) / max(lambda_est, 1e-15)
            if change < 0.001:
                damping = min(0.95, damping * 1.02)
            else:
                damping = max(0.3, damping * 0.98)
        prev_lambda = lambda_est
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MinMax {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, lambda={lambda_est:.8f}")
        
        # Convergence check
        if it > 100 and abs(lambda_est - prev_lambda) < 1e-10 * max(lambda_est, 1):
            break
    
    return best_a, best_score
