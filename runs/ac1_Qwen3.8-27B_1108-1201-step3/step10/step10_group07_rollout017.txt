  - Set to zero: x* = C_k/T

1. Compute C, L, Q vectors (length 2n-1)
2. Generate candidate x values: x=0, and for each k, the critical point
3. Filter: keep only x >= 0
4. Evaluate f(x) at each candidate
5. Pick the best

            #   For k=2i: Q_k=1, L_k=0, C_k = conv[k] - old_x^2
            #   For 0<=k-i<n and k-i!=i: L_k = 2*a[k-i], Q_k=0, C_k = conv[k] - 2*old_x*a[k-i]
            #   Otherwise: L_k=0, Q_k=0, C_k = conv[k]

            if 0 <= 2*i < total_len:
                C[2*i] = conv[2*i] - old_x * old_x
                Q[2*i] = 1.0
                # L[2*i] = 0 (already)
            
            # All other k where k-i is valid
            k_valid_start = i  # k-i >= 0 => k >= i
            k_valid_end = i + n - 1  # k-i <= n-1 => k <= i+n-1
            k_valid_end = min(k_valid_end, total_len - 1)

                indices = np.arange(k_valid_start, k_valid_end + 1)
                j_indices = indices - i  # = k - i
                # Exclude j = i (which is k = 2i)
                mask = j_indices != i
                if mask.any():
                    valid_indices = indices[mask]
                    valid_j = j_indices[mask]
                    C[valid_indices] = conv[valid_indices] - 2.0 * old_x * a[valid_j]
                    L[valid_indices] = 2.0 * a[valid_j]

            # f(x) = (C_k + L_k*x + Q_k*x^2) / (T+x)^2

def spectral_perturb(a, rng, strength=0.1):
    """Generate a perturbation that tries to flatten the spectrum."""
    n = len(a)
    A = rfft(a)
    # Compute target: flatten |A|
    mag = np.abs(A)
    mean_mag = mag.mean()
    # Create a "flattened" target
    target_mag = np.ones_like(mag) * mean_mag
    # Blend: new_mag = (1-strength)*mag + strength*target_mag
    new_mag = (1 - strength) * mag + strength * target_mag
    # Keep phase
    phase = A / (mag + 1e-30)
    new_A = new_mag * phase
    # Inverse FFT

    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s > 1e-15:

    else:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating with warm starts
2. Exact coordinate descent via critical-point enumeration
3. Spectral flattenization for structured perturbations

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv) for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def simplex_project(v):
    """Project vector onto simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached

def build_conv_matrix(a):

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)

        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1

    return b_val, M_val


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Exact coordinate descent: for each coordinate, find near-optimal value
    by evaluating all critical points of the objective.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    total_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        sweeps += 1
        sweep_improved = False
        S = a.sum()

        for i in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break
            old_x = a[i]
            T = S - old_x  # sum of others

            if T < 1e-15:
                continue

            # Compute C_k, L_k, Q_k for c[k](x) = C_k + L_k*x + Q_k*x^2
            C = conv.copy()
            L = np.zeros(total_len)
            Q = np.zeros(total_len)

            # k = 2i: quadratic term
            ki = 2 * i
            if 0 <= ki < total_len:
                C[ki] = conv[ki] - old_x * old_x
                Q[ki] = 1.0

            # Linear terms for valid k-i
            k_lo = max(0, i)
            k_hi = min(i + n - 1, total_len - 1)
            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                j_idx = idx - i
                mask = j_idx != i  # exclude k=2i
                if mask.any():
                    vi = idx[mask]
                    vj = j_idx[mask]
                    C[vi] = conv[vi] - 2.0 * old_x * a[vj]
                    L[vi] = 2.0 * a[vj]

            # Generate candidate x values
            # Critical points: x = (2*C_k - L_k*T) / (2*Q_k*T - L_k)
            denom = 2.0 * Q * T - L
            numer = 2.0 * C - L * T
            valid_mask = np.abs(denom) > 1e-20
            candidates = np.array([0.0, old_x])
            if valid_mask.any():
                x_crit = numer[valid_mask] / denom[valid_mask]
                x_crit = x_crit[x_crit > 0]
                if len(x_crit) > 0:
                    candidates = np.concatenate([candidates, x_crit])

            # Add heuristic candidates
            candidates = np.concatenate([candidates, [old_x * 0.5, old_x * 1.5]])

            # Clip
            max_x = 10.0 * S / n
            candidates = candidates[(candidates >= 0) & (candidates <= max_x)]

            if len(candidates) <= 2:
                # Check if setting to 0 helps
                if old_x > 0:
                    a[i] = 0.0
                    new_conv_i = conv.copy()
                    # Update conv: remove contribution of a[i]=old_x
                    k_lo2 = max(0, i)
                    k_hi2 = min(i + n - 1, total_len - 1)
                    if k_lo2 <= k_hi2:
                        j2 = np.arange(k_lo2, k_hi2 + 1) - i
                        mask2 = j2 != i
                        if mask2.any():
                            new_conv_i[j2[mask2]] -= 2.0 * old_x * a[j2[mask2]]
                    if 0 <= ki < total_len:
                        new_conv_i[ki] = C[ki]  # x=0, so just C
                    new_M = float(np.maximum(new_conv_i, 0).max())
                    new_S = S
                    new_score = 2.0 * n * new_M / (new_S * new_S)
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_a = a.copy()
                        conv = np.maximum(new_conv_i, 0)
                        M = new_M
                        S = new_S
                        sweep_improved = True
                    else:
                        a[i] = old_x
                continue

            # Vectorized evaluation
            X = candidates[None, :]
            vals = C[:, None] + L[:, None] * X + Q[:, None] * (X ** 2)
            max_vals = vals.max(axis=0)
            denom_vals = (T + candidates) ** 2
            f_vals = max_vals / denom_vals

            best_idx = int(np.argmin(f_vals))
            best_x = candidates[best_idx]
            best_f = f_vals[best_idx]
            current_f = M / (T + old_x) ** 2

            if best_f < current_f - 1e-14 and abs(best_x - old_x) > 1e-12:
                a[i] = best_x
                conv = fast_convolve(a)
                M = float(conv.max())
                S = a.sum()
                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                sweep_improved = True

        if not sweep_improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_perturb(a, strength=0.15):
    """Generate a perturbation that flattens the power spectrum."""
    n = len(a)
    A = rfft(a)
    mag = np.abs(A)
    mean_mag = mag.mean()
    if mean_mag < 1e-15:
        return a.copy()
    new_mag = (1.0 - strength) * mag + strength * mean_mag
    phase = A / (mag + 1e-30)
    new_A = new_mag * phase
    new_a = irfft(new_A, n)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s > 1e-15:
        new_a = new_a / s
    else:
        new_a = np.ones(n) / n
    return new_a


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Damped update
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.7
        else:
            alpha = max(0.2, 8.0 / (it + 5.0))

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1.0 + 1e-7):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            # Spectral perturbation to escape
            a_pert = spectral_perturb(a, strength=0.1)
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile improvements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    total_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, total_len - 1)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < total_len - 1:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.999, 0.998, 0.997, 0.9995, 1.0005, 1.001, 1.002, 1.003]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    else:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * np.arange(n) * rng.integers(3, 12) / n)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization combining LP alternating, exact coordinate descent,
    spectral perturbations, and fine polishing.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.40

    t_alloc = min(t1_end - time.monotonic(), 350.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Try from a diverse init
    init = diverse_init(n, rng, init_type=2)
    if time.monotonic() < t1_end - 60:
        t_alloc2 = min(t1_end - time.monotonic(), 100.0)
        a_lp2, s_lp2 = lp_alternating(init, t_alloc2, solver, verbose=False)
        try_update(a_lp2, "lp_diverse")

    # === Phase 2: Exact Coordinate Descent ===
    print(f"\n=== Phase 2: Exact CD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    t_alloc = min(t2_end - time.monotonic(), 400.0)
    print(f"  ExactCD budget={t_alloc:.0f}s")
    a_ecd, s_ecd = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
    try_update(a_ecd, "exact_cd")

    # Also from LP result
    if time.monotonic() < t2_end - 60:
        t_alloc2 = min(t2_end - time.monotonic(), 200.0)
        a_ecd2, s_ecd2 = exact_coord_descent(a_lp, t_alloc2, verbose=True, label="lp")
        try_update(a_ecd2, "ecd_lp")

    # === Phase 3: Spectral + CD hybrid ===
    print(f"\n=== Phase 3: Spectral Hybrid (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    for strength in [0.05, 0.1, 0.2, 0.3]:
        if time.monotonic() > t3_end:
            break
        a_spec = spectral_perturb(best_seq, strength=strength)
        # Quick CD on the spectral result
        t_alloc = min(t3_end - time.monotonic() - 10, 60.0)
        if t_alloc > 20:
            a_h, s_h = exact_coord_descent(a_spec, t_alloc, verbose=False, label=f"sp{strength}")
            try_update(a_h, f"spec_{strength}")

    # === Phase 4: Fine Polish ===
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 30

    t_fp = min(t4_end - time.monotonic(), 200.0)
    if t_fp > 30:
        print(f"  FinePolish budget={t_fp:.0f}s")
        a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
        try_update(a_fp, "fine_polish")

    # Final exact CD pass
    if time.monotonic() < t4_end - 20:
        t_final = min(t4_end - time.monotonic() - 5, 150.0)
        if t_final > 30:
            a_fin, s_fin = exact_coord_descent(best_seq, t_final, verbose=False, label="final")
            try_update(a_fin, "final_ecd")

    # Final fine polish
    if time.monotonic() < deadline - 10:
        t_final2 = min(deadline - time.monotonic() - 5, 60.0)
        if t_final2 > 10:
            a_fin2, s_fin2 = fine_polish(best_seq, t_final2, verbose=False)
            try_update(a_fin2, "final_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
