def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

    diag_k = 2 * diag_idx
    valid_diag = diag_k < size
    grad[valid_diag] -= w[diag_k[valid_diag]] * a[valid_diag]
    
    return lse, grad


def mirror_descent_v2(a0, time_limit, eta0=0.05, active_frac=0.9995):
    """Improved mirror descent with Nesterov and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        max_b = b.max()
        
        # Active set
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.mean()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a = a_new
        
        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            rng = np.random.default_rng(hash((it,)) % (2**31))
            noise = rng.standard_normal(n) * 0.003
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta * 0.8, 0.001)
        
        # Adaptive eta
        if it % 100 == 0 and no_improve < 50:
            eta = min(eta * 1.001, 0.2)
        elif no_improve > 100:
            eta = max(eta * 0.99, 0.001)
        
        if (it + 1) % 5000 == 0:
            print(f"    MD2 iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_optimize_v2(a0, time_limit, T_start=10.0, T_end=0.01, n_stages=20):
    """LogSumExp optimization with adaptive temperature cooling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    
    t0 = time.monotonic()
    
    # Adaptive temperature stages
    T_stages = np.geomspace(T_start, T_end, n_stages)
    
    for stage_idx, T in enumerate(T_stages):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Allocate time: more time for lower T
        frac = (stage_idx + 1) / n_stages
        t_alloc = min(remaining * 0.9, remaining)
        
        def func(z_val):
            a_val = softmax_safe(z_val)
            obj, _ = logsumexp_offset_and_grad(a_val, T)
            return obj
        
        def jac(z_val):
            a_val = softmax_safe(z_val)
            _, grad_a = logsumexp_offset_and_grad(a_val, T)
            # Convert a-gradient to z-gradient
            S = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S)
            return grad_z
        
        maxiter = max(50, int(t_alloc * 50))  # rough estimate
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxiter * 200,
                                   'disp': False})
            z = res.x
        except Exception:
            pass
        
        a = softmax_safe(z)
        score = evaluate_local(a)
        print(f"    LSE stage {stage_idx+1}/{n_stages} T={T:.4f} ({time.monotonic()-t0:.0f}s): score={score:.8f}")
    
    return a


def peak_shaving_cd(a0, time_limit, n_iters=100000):
    """Peak-shaving coordinate descent with exact line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        max_k = int(np.argmax(b))
        
        # Find the peak position
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        
        # Contributions to the peak
        contribs = a[idx] * a[max_k - idx]
        
        # Try reducing top contributors
        top_m = min(8, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-15:
                continue
            
            # Try several reduction factors
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                
                # Redistribute the difference proportionally to non-peak-contributing elements
                # Elements that don't contribute to the peak
                mask = np.ones(n, dtype=bool)
                # Mark elements that contribute to the top peaks
                top_peaks = np.argsort(b)[::-1][:5]
                peak_mask = np.zeros(n, dtype=bool)
                for pk in top_peaks:
                    pjs = max(0, pk - n + 1)
                   pje = min(n - 1, pk)
                    peak_mask[pjs:pje+1] = True
                
                mask[pos] = False
                non_peak = mask & ~peak_mask
                non_peak_sum = a_trial[non_peak].sum()
                
                if non_peak_sum > 1e-15:
                    a_trial[non_peak] += diff * a_trial[non_peak] / non_peak_sum
                else:
                    a_trial[mask] += diff * a_trial[mask] / a_trial[mask].sum()
                
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Try adding to low-contribution positions
                b_current = fast_convolve(a)
                # Find positions with low contribution to all peaks
                contrib_to_peaks = np.zeros(n)
                top_peaks = np.argsort(b_current)[::-1][:10]
                for pk in top_peaks:
                    pjs = max(0, pk - n + 1)
                    pje = min(n - 1, pk)
                    idx_p = np.arange(pjs, pje + 1)
                    contrib_to_peaks[idx_p] += a[idx_p] * a[pk - idx_p]
                
                # Try increasing low-contribution positions
                low_positions = np.argsort(contrib_to_peaks)[:20]
                for pos in low_positions:
                    a_trial = a.copy()
                    a_trial[pos] *= 1.01
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if not improved and stall > 500:
                    break
        else:
            stall = 0
        
        if (it + 1) % 20000 == 0:
            print(f"    CD {time.monotonic()-t0:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def nesterov_lp_v2(a0, time_limit, solver=None, max_iters=100000):
    """Nesterov-accelerated LP alternating."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 50:
                b_new = y.copy()
            else:
                continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 10:
                noise = rng.standard_normal(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
        
        if (it + 1) % 500 == 0:
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
