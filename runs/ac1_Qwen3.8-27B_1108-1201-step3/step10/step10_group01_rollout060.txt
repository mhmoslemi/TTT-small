        T[j:j + n, j] = a
    return csr_matrix(T)


def lp_best_response_minM(a_fixed, use_sparse=True, method='highs'):
    """
    Solve: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b, M_value).
    """
    n = len(a_fixed)
    size = 2 * n - 1

    if use_sparse:
        T = build_conv_matrix_sparse(a_fixed)
        # A_ub: [T, -1] of shape (size, n+1)
        neg_ones = csr_matrix((-np.ones((size, 1))))
        A_ub = hstack([T, neg_ones], format='csc')
        b_ub = np.zeros(size)
    else:
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j + n, j] = a_fixed
        A_ub = np.hstack([T, -np.ones((size, 1))])
        b_ub = np.zeros(size)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method)
        if res.success:
            b_val = res.x[:n]
            M_val = res.x[n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                b_val = np.ones(n) / n
                M_val = fast_convolve_cross(a_fixed, b_val).max()
            else:
                b_val = b_val / s
            return b_val, float(M_val)
    except Exception:
        pass

    # Fallback
    b_val = np.ones(n) / n
    M_val = fast_convolve_cross(a_fixed, b_val).max()
    return b_val, float(M_val)


def nesterov_lp_optimize(a0, time_limit, max_iters=100000, print_interval=2000):
    """Nesterov-accelerated LP alternation (min-M formulation)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_avg = a.copy()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = lp_best_response_minM(y, use_sparse=(n <= 5000))
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        x_avg = (it * x_avg + b_new) / (it + 1) if it > 0 else b_new.copy()

        score_y = 2.0 * n * fast_convolve_auto(y).max()
        score_x = 2.0 * n * fast_convolve_auto(x_avg).max()

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if score_x < best_score:
            best_score = score_x
            best_a = x_avg.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                noise = rng.standard_normal(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            if print_interval and (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, max_iters=500000, print_interval=5000):
    """Fast Nesterov subgradient method for max-conv minimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_avg = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve_auto(y)
        max_b = b.max()
        # Active set: positions within 1% of max
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]

        # Subgradient: average of 2*y[k-i] over active k
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g *= 2.0 / len(active)

        alpha = 2.0 / (it + 2)
        lr = alpha / max_b if max_b > 1e-15 else alpha

        y = y - lr * g
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x_avg.copy()
            sy = 1.0
        y = y / sy

        x_avg = (it * x_avg + y) / (it + 1) if it > 0 else y.copy()

        score_y = 2.0 * n * fast_convolve_auto(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_x = 2.0 * n * fast_convolve_auto(x_avg).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x_avg.copy()

        if print_interval and (it + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and multi-stage temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_loc(zv):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            av = np.exp(zv - zv.max())
            av = av / av.sum()
            bv = fast_convolve_auto(av)
            max_bv = bv.max()
            exp_bv = np.exp(bv - max_bv)
            lse = max_bv + np.log(exp_bv.sum())
            return (2.0 * n / T_stage) * lse

        def jac_loc(zv):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zv)
            av = np.exp(zv - zv.max())
            av = av / av.sum()
            bv = fast_convolve_auto(av)
            max_bv = bv.max()
            exp_bv = np.exp(bv - max_bv)
            w = exp_bv / exp_bv.sum()
            grad_a = np.zeros(n)
            for k in range(len(bv)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * av[k - idx]
            grad_a *= 2.0 / T_stage
            S = np.dot(grad_a, av)
            return av * (grad_a - S)

        iters = max(10, int(maxiter_per_stage / max(1, len(T_stages))))
        try:
            res = minimize(fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                                    'maxfun': iters * 200, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_coord_descent(a0, time_limit, n_steps=100000, top_m=6):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve_auto(a)
        top_peaks = np.argsort(b)[::-1][:top_m]

        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:3]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve_auto(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def harmonic_mean_iterate(a0, time_limit, max_iters=200000, step=0.1):
    """Fast harmonic-mean heuristic iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        eps = 1e-12
        b = 1.0 / (a + eps)
        b = b / b.sum()
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        score = 2.0 * n * fast_convolve_auto(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
