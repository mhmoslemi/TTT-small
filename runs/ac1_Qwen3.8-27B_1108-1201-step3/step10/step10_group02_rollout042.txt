def simplex_project(v, target=1.0):
    v=np.asarray(v,float)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

Strategy:
- Normalize to sum 1, then minimize 2*n*max(conv(a,a)).
- Use FFT-based gradients for LogSumExp-smoothed peak objectives.
- Multi-stage optimization:
    * structured / power-law / polynomial-root initializers,
    * mirror descent on active-peak / LogSumExp subgradients,
    * a game-theoretic mirror variant,
    * L-BFGS in log-coordinates with annealed smoothing,
    * gradient-guided coordinate polish,
    * occasional larger-length interpolation and refinement.
- All local computations are scale-invariant and use sum(a)=1.
"""

import time
import numpy as np
from scipy.optimize import minimize

_LS_STATE = None


def fast_conv2(x, y):
    """Linear convolution by FFT, non-negative sanitized."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    size = len(x) + len(y) - 1
    if size <= 0:
        return np.zeros(0, dtype=np.float64)
    fsize = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)
    z = np.fft.irfft(X * Y, fsize)[:size]
    z = np.asarray(z, dtype=np.float64)
    z[~np.isfinite(z)] = 0.0
    # Inputs used here are non-negative; tiny negative FFT noise is clipped.
    return np.maximum(z, 0.0)


def fast_conv_auto(a):
    """Auto-convolution a * a, non-negative sanitized."""
    a = np.asarray(a, dtype=np.float64)
    size = 2 * len(a) - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    z = np.fft.irfft(A * A, fsize)[:size]
    z = np.asarray(z, dtype=np.float64)
    z[~np.isfinite(z)] = 0.0
    return np.maximum(z, 0.0)


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = len(a)
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def local_score_normalized(a):
    """Assumes sum(a)=1. Score = 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    d = fast_conv_auto(a)
    m = float(d.max())
    return float(2.0 * n * m)


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    a = a / s
    return local_score_normalized(a)


def simplex_project(v, target=1.0):
    """Project v onto {x >= 0, sum(x)=target}."""
    v = np.asarray(v, dtype=np.float64)
    v = np.nan_to_num(v, nan=0.0, posinf=1e6, neginf=-1e6)
    n = len(v)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx - css < target + 1e-14
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho_pos = np.nonzero(cond)[0]
    rho = int(rho_pos[-1])
    theta = (css[rho] - target) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    sw = float(w.sum())
    if sw <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    if abs(sw - target) > 1e-12 * max(1.0, abs(target)):
        w = w * (target / sw)
    return w


def softmax_weights(d, sigma):
    """
    p_k proportional to exp((d_k - max(d)) / (sigma*max(d))).
    If sigma is tiny, gives uniform weight over (near) maximizers.
    """
    d = np.asarray(d, dtype=np.float64)
    maxd = float(d.max())
    L = len(d)
    if L == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.isfinite(maxd) or maxd <= 0.0:
        return np.ones(L, dtype=np.float64) / L
    if sigma <= 1e-12:
        p = (d >= maxd - 1e-10 * maxd).astype(np.float64)
        s = float(p.sum())
        if s <= 0.0:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(d))] = 1.0
            s = 1.0
        return p / s
    z = (d - maxd) / (sigma * maxd)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(d))] = 1.0
        return p
    return e / s


def Bw_vec(a, w):
    """
    Return gradient factor 2*B_w a where
    (B_w a)_i = sum_j w_{i+j} a_j.
    w has length 2n-1.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    a_rev = a[::-1]
    c = fast_conv2(a_rev, w)
    return 2.0 * c[n - 1:2 * n - 1]


def shift_sum(p, u):
    """
    g_i = sum_j u_j p_{i+j}, where p has length 2n-1 and u length n.
    """
    p = np.asarray(p, dtype=np.float64)
    u = np.asarray(u, dtype=np.float64)
    n = len(u)
    ur = u[::-1]
    c = fast_conv2(p, ur)
    return c[n - 1:2 * n - 1]


def candidate_sqrtpoly(n):
    """First n coefficients of (1-z)^(-1/2), normalized."""
    q = np.empty(n, dtype=np.float64)
    q[0] = 1.0
    for k in range(1, n):
        q[k] = q[k - 1] * ((2.0 * k - 1.0) / (2.0 * k))
    return normalize_seq(q)


def candidate_left_power(n, alpha=-0.5):
    i = np.arange(n, dtype=np.float64) + 1e-6
    v = np.power(i, alpha)
    return normalize_seq(v)


def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if new_n == old_n:
        return f.copy()
    fn = normalize_seq(f)
    if old_n <= 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / new_n
    xo = np.linspace(0.0, 1.0, old_n)
    xn = np.linspace(0.0, 1.0, new_n)
    g = np.interp(xn, xo, fn)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    return g / s


def make_base_candidates(n, parent, rng=None):
    """Generate a diverse set of initial sequences of length n."""
    c = []
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    eps = 1e-6

    # Uniform.
    c.append(np.ones(n, dtype=np.float64) / n)

    # One-sided power-law shapes.
    for alpha in (-0.5, -0.45, -0.4, -0.3):
        c.append(normalize_seq(np.power(i + eps, alpha)))
        c.append(normalize_seq(np.power(n - 1.0 - i + eps, alpha)))

    # Centered / U-shaped powers.
    y = np.minimum(x, 1.0 - x) + 1e-4
    for alpha in (-0.5, -0.25, 0.0, 0.25):
        c.append(normalize_seq(np.power(y, alpha)))

    # Polynomial square-root-like shape.
    q = candidate_sqrtpoly(n)
    c.append(q)
    c.append(q[::-1])

    # Smooth classical shapes.
    c.append(normalize_seq(np.cos(np.pi * x / 2.0) ** 2))
    c.append(normalize_seq(np.sin(np.pi * x) ** 1.5))
    c.append(normalize_seq(np.exp(-3.0 * x * (1.0 - x))))
    c.append(normalize_seq(np.exp(-4.0 * (x - 0.35) ** 2)))
    c.append(normalize_seq(np.exp(-4.0 * (x - 0.65) ** 2)))

    # One-sided parameter grid.
    for alpha in (-0.5, -0.4, -0.3):
        for beta in (0.0, -0.2, -0.4):
            v = np.power(i + eps, alpha) * np.power(n - 1.0 - i + eps, beta)
            c.append(normalize_seq(v))

    # Random smooth / restricted random candidates.
    if rng is not None:
        for _ in range(5):
            v = np.ones(n, dtype=np.float64)
            for _ in range(5):
                f = float(rng.integers(1, 8))
                ph = float(rng.uniform(0.0, 2.0 * np.pi))
                amp = float(rng.uniform(0.1, 0.8))
                v += amp * np.cos(2.0 * np.pi * f * x + ph)
            c.append(normalize_seq(v))

        # Binary / three-level random candidates.
        c.append(normalize_seq((rng.random(n) > 0.5).astype(np.float64)))
        c.append(normalize_seq(rng.integers(0, 3, size=n).astype(np.float64) / 2.0))

    # Parent interpolated and blended with a one-sided power law.
    if parent is not None:
        pi = interpolate_seq(parent, n)
        c.append(pi)
        lp = candidate_left_power(n, -0.5)
        for mix in (0.75, 0.5, 0.25):
            loga = mix * np.log(pi + 1e-300) + (1.0 - mix) * np.log(lp + 1e-300)
            tv = np.exp(loga - float(np.max(loga)))
            c.append(normalize_seq(tv))

    return c


def select_top_scored(scored, k):
    """Return k smallest-score entries from list of (score, array)."""
    selected = []
    rem = list(scored)
    for _ in range(k):
        if not rem:
            break
        mi = 0
        ms = rem[0][0]
        for ii in range(1, len(rem)):
            if rem[ii][0] < ms:
                ms = rem[ii][0]
                mi = ii
        selected.append(rem.pop(mi))
    return selected


def mirror_refine(a0, time_limit, sigma=0.05, eta0=0.08, max_eta=5.0,
                  nesterov=True, restart_iters=700, noise_base=0.01, label="mirror"):
    """
    Mirror descent / multiplicative weights with active-peak LogSumExp weights.
    Optimizes a normalized non-negative sequence.
    """
    a = normalize_seq(a0)
    n = len(a)
    t0 = time.monotonic()
    d0 = fast_conv_auto(a)
    best_score = float(2.0 * n * d0.max())
    best_a = a.copy()

    y = a.copy()
    cur_score = best_score
    eta = float(eta0)
    no_improve = 0
    it = 0
    last_print = t0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        dy = fast_conv_auto(y)
        score_y = float(2.0 * n * dy.max())
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()

        p = softmax_weights(dy, sigma)
        g = Bw_vec(y, p)
        gc = g - float(np.mean(g))

        loga = np.log(np.maximum(a, 1e-300))
        logan = loga - eta * gc
        logan -= float(np.max(logan))
        a_new = np.exp(logan)
        sn = float(a_new.sum())
        if sn <= 0.0 or not np.isfinite(sn):
            a_new = a.copy()
            sn = 1.0
        else:
            a_new = a_new / sn

        if nesterov:
            alpha = 2.0 / (float(it) + 2.0)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0.0)
            ys = float(y.sum())
            if ys <= 0.0 or not np.isfinite(ys):
                y = a_new.copy()
                ys = 1.0
            else:
                y = y / ys

        new_score = float(2.0 * n * fast_conv_auto(a_new).max())
        accepted = False

        # Allow slight non-monotonicity to escape shallow local minima.
        if new_score < cur_score * 1.02 + 1e-14:
            a = a_new
            cur_score = new_score
            eta = min(max_eta, eta * 1.08)
            accepted = True
        else:
            e = eta
            for _ in range(7):
                e *= 0.5
                if e < max(1e-7, eta0 * 0.01):
                    break
                logan = loga - e * gc
                logan -= float(np.max(logan))
                a_new = np.exp(logan)
                sn = float(a_new.sum())
                if sn <= 0.0 or not np.isfinite(sn):
                    break
                a_new = a_new / sn
                new_score = float(2.0 * n * fast_conv_auto(a_new).max())
                if new_score < cur_score * 1.02 + 1e-14:
                    a = a_new
                    cur_score = new_score
                    eta = e
                    accepted = True
                    break
            if not accepted:
                eta = max(max(1e-7, eta0 * 0.1), eta * 0.6)

        if accepted:
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if it % 2000 == 0 and time.monotonic() - last_print > 15.0:
            print(f"  {label} it={it} t={time.monotonic()-t0:.0f}s best={best_score:.8f} eta={eta:.5f}")
            last_print = time.monotonic()

        if no_improve >= restart_iters:
            noise_scale = noise_base * (1.0 + 0.25 * min(1.0, float(no_improve) / float(restart_iters)))
            a = a + rng.standard_normal(n) * noise_scale
            a = simplex_project(a)
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0)

        it += 1

    return normalize_seq(best_a), best_score


def game_mirror_refine(a0, time_limit, sigma_d=0.05, sigma_c=0.05, eta0=0.02,
                       update_p_every=5, restart_iters=900, label="game"):
    """
    Heuristic game-theoretic mirror refinement.
    p = weights over diagonals (peaks of a*a).
    c = M_p a (column payoffs), u = weights over active columns.
    gradient = shift_sum(p,u), then multiplicative update.
    """
    a = normalize_seq(a0)
    n = len(a)
    t0 = time.monotonic()
    d0 = fast_conv_auto(a)
    best_score = float(2.0 * n * d0.max())
    best_a = a.copy()

    y = a.copy()
    cur_score = best_score
    eta = float(eta0)
    no_improve = 0
    it = 0
    last_print = t0
    rng = np.random.default_rng(987654)

    while time.monotonic() - t0 < time_limit:
        if it % max(1, int(update_p_every)) == 0:
            dy = fast_conv_auto(y)
            score_y = float(2.0 * n * dy.max())
            if score_y < best_score - 1e-14:
                best_score = score_y
                best_a = y.copy()
            p = softmax_weights(dy, sigma_d)
        else:
            dy = fast_conv_auto(y)

        c = Bw_vec(y, p) * 0.5
        u = softmax_weights(c, sigma_c)
        g = shift_sum(p, u)
        g = g - float(np.mean(g))
        gnorm = float(np.max(np.abs(g)))

        if gnorm <= 1e-14:
            a = a + rng.standard_normal(n) * 0.01
            a = simplex_project(a)
            y = a.copy()
            no_improve += 50
            it += 1
            continue

        g = g / gnorm
        loga = np.log(np.maximum(a, 1e-300))
        logan = loga - eta * g
        logan -= float(np.max(logan))
        a_new = np.exp(logan)
        sn = float(a_new.sum())
        if sn <= 0.0 or not np.isfinite(sn):
            a_new = a.copy()
            sn = 1.0
        else:
            a_new = a_new / sn

        # Nesterov-like y update.
        alpha = 2.0 / (float(it) + 2.0)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0.0)
        ys = float(y.sum())
        if ys <= 0.0 or not np.isfinite(ys):
            y = a_new.copy()
            ys = 1.0
        else:
            y = y / ys

        new_score = float(2.0 * n * fast_conv_auto(a_new).max())
        accepted = False

        if new_score < cur_score * 1.03 + 1e-14:
            a = a_new
            cur_score = new_score
            eta = min(3.0, eta * 1.08)
            accepted = True
        else:
            e = eta
            for _ in range(6):
                e *= 0.5
                if e < max(1e-8, eta0 * 0.02):
                    break
                logan = loga - e * g
                logan -= float(np.max(logan))
                a_new = np.exp(logan)
                sn = float(a_new.sum())
                if sn <= 0.0 or not np.isfinite(sn):
                    break
                a_new = a_new / sn
                new_score = float(2.0 * n * fast_conv_auto(a_new).max())
                if new_score < cur_score * 1.03 + 1e-14:
                    a = a_new
                    cur_score = new_score
                    eta = e
                    accepted = True
                    break
            if not accepted:
                eta = max(max(1e-8, eta0 * 0.1), eta * 0.6)

        if accepted:
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if it % 2000 == 0 and time.monotonic() - last_print > 15.0:
            print(f"  {label} it={it} t={time.monotonic()-t0:.0f}s best={best_score:.8f} eta={eta:.5f}")
            last_print = time.monotonic()

        if no_improve >= restart_iters:
            noise_scale = 0.012 + 0.01 * min(1.0, float(no_improve) / float(restart_iters))
            a = a + rng.standard_normal(n) * noise_scale
            a = simplex_project(a)
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0)

        it += 1

    return normalize_seq(best_a), best_score


def _ls_compute(x, need_grad):
    """Obj/grad for L-BFGS on softmax(z), objective T*logsumexp(conv(a,a)/T)."""
    st = _LS_STATE
    if st is None:
        return 0.0, None
    if time.monotonic() > st["deadline"]:
        st["abort"] = True
        return 1e300, np.zeros_like(x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return 1e300, np.zeros_like(x, dtype=np.float64)

    z = x.astype(np.float64, copy=False)
    zm = z - float(np.mean(z))
    zm = np.clip(zm, -700.0, 700.0)
    a = np.exp(zm)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.ones(len(x), dtype=np.float64) / len(x)
    else:
        a = a / s

    d = fast_conv_auto(a)
    maxd = float(d.max())
    if maxd <= 0.0 or not np.isfinite(maxd):
        return 0.0, (np.zeros_like(x, dtype=np.float64) if need_grad else None)

    T = float(st["T"])
    if T <= 1e-15:
        w = np.zeros(len(d), dtype=np.float64)
        w[int(np.argmax(d))] = 1.0
        obj = maxd
    else:
        e = np.exp((d - maxd) / T)
        se = float(e.sum())
        if se <= 0.0 or not np.isfinite(se):
            w = np.zeros(len(d), dtype=np.float64)
            w[int(np.argmax(d))] = 1.0
            obj = maxd
        else:
            w = e / se
            obj = maxd + T * float(np.log(se))

    if not need_grad:
        return float(obj), None

    g = Bw_vec(a, w)
    S = float(np.dot(g, a))
    grad = a * (g - S)
    return float(obj), grad


def _ls_fun(x):
    val, _ = _ls_compute(x, False)
    return val


def _ls_jac(x):
    _, grad = _ls_compute(x, True)
    if grad is None:
        return np.zeros_like(x, dtype=np.float64)
    return grad


def lbgfs_refine(a0, time_limit, sigma_stages=None, maxiter_per_stage=None, label="lbgfs"):
    """
    L-BFGS in log-coordinates with annealed LogSumExp temperature.
    Uses a global state to avoid closures.
    """
    global _LS_STATE
    a = normalize_seq(a0)
    n = len(a)
    if maxiter_per_stage is None:
        maxiter_per_stage = max(50, min(250, 250000 // max(1, n)))
    if sigma_stages is None:
        sigma_stages = [0.30, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]

    deadline = time.monotonic() + max(1.0, float(time_limit))
    d0 = fast_conv_auto(a)
    best_score = float(2.0 * n * d0.max())
    best_a = a.copy()

    z = np.log(np.maximum(a, 1e-300)).astype(np.float64)
    z -= float(np.mean(z))

    last_print = time.monotonic()
    for sigma in sigma_stages:
        if time.monotonic() >= deadline:
            break
        dcur = fast_conv_auto(a)
        maxd = float(dcur.max())
        T = max(1e-15, float(sigma) * maxd)
        _LS_STATE = {"deadline": deadline, "T": T, "abort": False}
        z0 = z.copy()

        try:
            res = minimize(
                fun=_ls_fun,
                x0=z0,
                jac=_ls_jac,
                method="L-BFGS-B",
                options={
                    "maxiter": int(maxiter_per_stage),
                    "ftol": 1e-13,
                    "gtol": 1e-10,
                    "maxfun": int(maxiter_per_stage * 40),
                    "disp": False
                }
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x.astype(np.float64, copy=False) - float(np.mean(res.x))
            else:
                z = z0
        except Exception:
            z = z0

        zm = np.clip(z - float(np.mean(z)), -700.0, 700.0)
        at = np.exp(zm)
        s = float(at.sum())
        if s <= 0.0 or not np.isfinite(s):
            at = a.copy()
        else:
            at = at / s

        sc = local_score_normalized(at)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = at.copy()

        # Continue from the smoothed point; it is useful for lower temperatures.
        a = at
        z = np.log(np.maximum(a, 1e-300))
        z -= float(np.mean(z))

        if time.monotonic() - last_print > 10.0:
            print(f"  {label} sigma={sigma:.5f} t={time.monotonic()-deadline+time_limit:.0f}s best={best_score:.8f}")
            last_print = time.monotonic()

    return normalize_seq(best_a), best_score


def coordinate_polish(a0, time_limit, sigma=0.02, label="coord"):
    """
    Gradient-guided two-coordinate line search: move mass from high gradient
    coordinate to low gradient coordinate, accepting only exact improvements.
    """
    a = normalize_seq(a0)
    n = len(a)
    if n < 2:
        return a, local_score_normalized(a)

    t0 = time.monotonic()
    d0 = fast_conv_auto(a)
    best_score = float(2.0 * n * d0.max())
    best_a = a.copy()

    rng = np.random.default_rng(314159)
    no_improve = 0
    it = 0
    last_print = t0
    deltas = (0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05)

    while time.monotonic() - t0 < time_limit:
        d = fast_conv_auto(a)
        p = softmax_weights(d, sigma)
        g = Bw_vec(a, p)
        h = g - float(np.mean(g))

        order = np.argsort(h)
        k = max(3, n // 200)
        donors = order[-k:]
        receivers = order[:k]

        improved = False
        moves = []
        if len(donors) and len(receivers):
            moves.append((int(donors[0]), int(receivers[0])))
            moves.append((int(donors[-1]), int(receivers[-1])))
            for _ in range(3):
                ii = int(donors[int(rng.integers(len(donors)))])
                jj = int(receivers[int(rng.integers(len(receivers)))])
                moves.append((ii, jj))

        for ii, jj in moves:
            if ii == jj or a[ii] < 1e-12:
                continue
            orig_i = float(a[ii])
            orig_j = float(a[jj])
            for delta in deltas:
                amt = orig_i * delta
                if amt <= 0.0 or orig_i - amt <= -1e-15:
                    continue
                a[ii] = orig_i - amt
                a[jj] = orig_j + amt
                s = float(a.sum())
                if abs(s - 1.0) > 1e-10:
                    a = a / s
                    orig_i = float(a[ii])
                    orig_j = float(a[jj])

                sc = local_score_normalized(a)
                if np.isfinite(sc) and sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                a[ii] = orig_i
                a[jj] = orig_j
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                a = a + rng.standard_normal(n) * 0.005
                a = simplex_project(a)
                no_improve = 0

        if it % 4000 == 0 and time.monotonic() - last_print > 15.0:
            print(f"  {label} it={it} t={time.monotonic()-t0:.0f}s best={best_score:.8f}")
            last_print = time.monotonic()

        it += 1

    return normalize_seq(best_a), best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search driver. Returns a list of non-negative floats.
    """
    if budget_s is None:
        budget_s = 1800.0
    budget_s = float(max(1.0, budget_s))
    reserve = 25.0
    deadline = time.monotonic() + max(1.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")
    parent = None

    # Load any provided parent sequence.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and float(p.sum()) > 1e-10:
            parent = p / float(p.sum())
            bs = local_score_normalized(parent)
            if np.isfinite(bs):
                best_seq = parent.copy()
                best_score = bs
                print(f"Loaded parent sequence: n={len(parent)}, score={bs:.8f}")
    except Exception as ee:
        print("No usable parent sequence:", ee)

    if best_seq is None:
        n0 = 1200
        best_seq = candidate_left_power(n0, -0.5)
        best_score = local_score_normalized(best_seq)
        print(f"Default initializer: n={n0}, score={best_score:.8f}")

    work_seq = best_seq.copy()

    # Phase 1: quick mirror refinement of the parent/default.
    rem = deadline - time.monotonic()
    if rem > 30.0:
        t = min(rem * 0.07, 90.0)
        a_work, sc_work = mirror_refine(work_seq, t, sigma=0.06, eta0=0.08, label="parent_mirror")
        if np.isfinite(sc_work):
            if sc_work < best_score - 1e-15:
                best_score = sc_work
                best_seq = a_work.copy()
                print(f"Update parent_mirror: {sc_work:.8f}")
            if sc_work <= best_score + 1e-6:
                work_seq = a_work.copy()

    # Phase 2: LogSumExp L-BFGS on the current best.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        t = min(rem * 0.20, 320.0)
        a_work, sc_work = lbgfs_refine(work_seq, t, label="lbgfs_initial")
        if np.isfinite(sc_work):
            if sc_work < best_score - 1e-15:
                best_score = sc_work
                best_seq = a_work.copy()
                print(f"Update lbgfs_initial: {sc_work:.8f}")
            if sc_work <= best_score + 1e-6:
                work_seq = a_work.copy()

    # Phase 3: candidate generation and screening across several lengths.
    rem = deadline - time.monotonic()
    if rem > 120.0:
        base_n = len(work_seq)
        candidates_n = {base_n}
        for n in (base_n - 400, base_n, base_n + 400, base_n + 800,
                  1000, 1200, 1600, 2000, 2400, 3000, 3600):
            if 400 <= int(n) <= 4500:
                candidates_n.add(int(n))
        n_list = sorted(candidates_n)
        if len(n_list) > 5:
            # Keep a controlled mix: near current plus standard sizes.
            n_list = [n for n in n_list if n in (base_n, base_n + 400, base_n + 800, 1200, 2000, 3000)]
            n_list = sorted(set(n_list))
            if len(n_list) > 5:
                n_list = n_list[:5]

        parent_for_candidates = parent if parent is not None else best_seq
        pool = []
        for n in n_list:
            cands = make_base_candidates(n, parent_for_candidates, rng)
            scored = []
            for a in cands:
                a = normalize_seq(a)
                sc = local_score_normalized(a)
                if np.isfinite(sc):
                    scored.append((sc, a))
            if not scored:
                continue
            top = select_top_scored(scored, 3)
            pool.extend(top)

        if pool:
            pool_top = select_top_scored(pool, 8)
            screen_budget = min(rem * 0.30, 400.0)
            per_screen = max(10.0, screen_budget / max(1, len(pool_top)))
            for idx in range(len(pool_top)):
                if time.monotonic() >= deadline - 30.0:
                    break
                sc0, a0 = pool_top[idx]
                a2, sc2 = mirror_refine(a0, min(per_screen, 120.0),
                                        sigma=0.05, eta0=0.08, label=f"screen_{idx}")
                if np.isfinite(sc2):
                    if sc2 < best_score - 1e-15:
                        best_score = sc2
                        best_seq = a2.copy()
                        work_seq = a2.copy()
                        print(f"Update screen_{idx}: n={len(a2)}, score={sc2:.8f}")
                    elif sc2 <= best_score + 1e-6:
                        work_seq = a2.copy()

    # Phase 4: focused refinement of the global best.
    rem = deadline - time.monotonic()
    if rem > 40.0:
        t = min(rem * 0.08, 110.0)
        a_work, sc_work = mirror_refine(best_seq, t, sigma=0.10, eta0=0.06, label="mirror_coarse")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update mirror_coarse: {sc_score:.8f}")

    rem = deadline - time.monotonic()
    if rem > 40.0:
        t = min(rem * 0.12, 160.0)
        a_work, sc_work = mirror_refine(best_seq, t, sigma=0.03, eta0=0.04, label="mirror_fine")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update mirror_fine: {sc_work:.8f}")

    rem = deadline - time.monotonic()
    if rem > 40.0:
        t = min(rem * 0.09, 130.0)
        a_work, sc_work = game_mirror_refine(best_seq, t, sigma_d=0.04, sigma_c=0.04,
                                             eta0=0.02, update_p_every=8, label="game")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update game: {sc_work:.8f}")

    rem = deadline - time.monotonic()
    if rem > 60.0:
        t = min(rem * 0.12, 180.0)
        a_work, sc_work = lbgfs_refine(best_seq, t, label="lbgfs_late")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update lbgfs_late: {sc_work:.8f}")

    rem = deadline - time.monotonic()
    if rem > 30.0:
        t = min(rem * 0.07, 90.0)
        a_work, sc_work = coordinate_polish(best_seq, t, sigma=0.015, label="coord_late")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update coord_late: {sc_work:.8f}")

    # Phase 5: try a larger length if useful time remains.
    rem = deadline - time.monotonic()
    if rem > 250.0 and len(best_seq) < 3500:
        new_n = min(4200, max(len(best_seq) + 800, 2000))
        cands = [
            interpolate_seq(best_seq, new_n),
            candidate_left_power(new_n, -0.5),
            candidate_sqrtpoly(new_n)
        ]
        init_seq = None
        init_score = float("inf")
        for a in cands:
            a = normalize_seq(a)
            sc = local_score_normalized(a)
            if np.isfinite(sc) and sc < init_score:
                init_score = sc
                init_seq = a.copy()

        if init_seq is not None:
            if init_score < best_score - 1e-15:
                best_score = init_score
                best_seq = init_seq.copy()
                print(f"Update large_init: n={new_n}, score={best_score:.8f}")

            t = min(rem * 0.15, 160.0)
            a_work, sc_work = mirror_refine(init_seq, t, sigma=0.05, eta0=0.06, label="large_mirror")
            if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
                best_score = sc_work
                best_seq = a_work.copy()
                print(f"Update large_mirror: {sc_work:.8f}")

            rem = deadline - time.monotonic()
            if rem > 120.0 and len(best_seq) == new_n:
                t = min(rem * 0.10, 130.0)
                a_work, sc_work = lbgfs_refine(best_seq, t, label="large_lbgfs")
                if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
                    best_score = sc_work
                    best_seq = a_work.copy()
                    print(f"Update large_lbgfs: {sc_work:.8f}")

    # Phase 6: final polish.
    rem = deadline - time.monotonic()
    if rem > 25.0:
        t = min(rem * 0.60, 130.0)
        a_work, sc_work = coordinate_polish(best_seq, t, sigma=0.01, label="final_coord")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update final_coord: {sc_work:.8f}")

    rem = deadline - time.monotonic()
    if rem > 15.0:
        t = min(rem * 0.80, 60.0)
        a_work, sc_work = mirror_refine(best_seq, t, sigma=0.008, eta0=0.02, label="final_mirror")
        if np.isfinite(sc_work) and sc_work < best_score - 1e-15:
            best_score = sc_work
            best_seq = a_work.copy()
            print(f"Update final_mirror: {sc_work:.8f}")

    # Final sanity and exact local verification.
    if best_seq is None:
        best_seq = candidate_left_power(1200, -0.5)

    final = np.asarray(best_seq, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)
    final = normalize_seq(final)
    final_score = local_score_normalized(final)
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL: n={len(final)}, score={best_score:.10f}")

    return [float(x) for x in final]
```
