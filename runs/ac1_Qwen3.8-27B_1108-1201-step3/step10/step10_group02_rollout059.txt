"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using dual-informed accelerated LP alternating
with scipy HiGHS solver, FFT-based subgradients, LogSumExp L-BFGS smoothing, and
targeted coordinate descent. Key innovations: using LP dual solutions for weighted
subgradients (much more informative than uniform averaging over active peaks),
adaptive step-size scheduling, and multi-phase time allocation that prioritizes
the LP alternating method which has the best convergence properties for this problem.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.signal import fftconvolve


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_C_matrix(a):
    """Build C matrix (size x n) such that (C @ b)[k] = sum_i a[i]*b[k-i]."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_primal_lp(a):
    """Solve min M s.t. Cb <= M, sum(b)=1, b>=0. Returns (b, M, success)."""
    n = len(a)
    size = 2 * n - 1
    C = build_C_matrix(a)
    # Variables: x = [b_0, ..., b_{n-1}, M], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    # Inequality: C @ b - M <= 0, shape (size, n+1)
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds: b >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None, False
    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, False
    b_val = b_val / s
    M_val = float(res.x[n])
    return b_val, M_val, True


def solve_dual_lp(a):
    """Solve max lambda s.t. C^T y >= lambda, sum(y)=1, y>=0. Returns (y, lambda, success)."""
    n = len(a)
    size = 2 * n - 1
    C = build_C_matrix(a)
    C_T = C.T  # shape (n, size)
    # Variables: x = [y_0, ..., y_{size-1}, lambda], total size+1
    c_obj = np.zeros(size + 1)
    c_obj[size] = -1.0  # maximize lambda = minimize -lambda
    # Inequality: C^T y - lambda >= 0  =>  -C^T y + lambda <= 0
    A_ub = np.zeros((n, size + 1))
    A_ub[:, :size] = -C_T
    A_ub[:, size] = 1.0
    b_ub = np.zeros(n)
    # Equality: sum(y) = 1
    A_eq = np.zeros((1, size + 1))
    A_eq[0, :size] = 1.0
    b_eq = np.array([1.0])
    # Bounds: y >= 0, lambda free
    bounds = [(0, None)] * size + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None, False
    y_val = res.x[:size]
    y_val = np.maximum(y_val, 0)
    s = y_val.sum()
    if s < 1e-15:
        return None, None, False
    y_val = y_val / s
    lam_val = float(res.x[size])
    return y_val, lam_val, True


def compute_subgradient_from_dual(a, y_dual):
    """Compute weighted subgradient using dual distribution over lags."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(n)
    active = np.where(y_dual > 1e-10)[0]
    for k in active:
        wk = y_dual[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            val = 0.0
            kj = k - j
            jk = j - k
            if 0 <= kj < n:
                val += a[kj]
            if 0 <= jk < n:
                val += a[jk]
            if k == 2 * j and 0 <= j < n:
                val -= a[j]
            g[j] += wk * val
    return g


def compute_subgradient_uniform(a, active_frac=0.999):
    """Compute subgradient using uniform weighting over peaks near max."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * active_frac
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            kj = k - j
            jk = j - k
            val = 0.0
            if 0 <= kj < n:
                val += a[kj]
            if 0 <= jk < n:
                val += a[jk]
            if k == 2 * j:
                val -= a[j]
            g[j] += val
    g /= len(active)
    return g


def mirror_descent_step(a, g, eta):
    """Mirror descent update on simplex."""
    log_a = np.log(np.maximum(a, 1e-300))
    g_shifted = g - g.min()
    log_a_new = log_a - eta * g_shifted
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def mirror_descent_phase(a0, time_limit, eta0=0.05, active_frac=0.999):
    """Standard mirror descent with BB step size and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kj = k - j
                jk = j - k
                val = 0.0
                if 0 <= kj < n:
                    val += a[kj]
                if 0 <= jk < n:
                    val += a[jk]
                if k == 2 * j:
                    val -= a[j]
                g[j] += val
        g /= max(1, len(active))

        # BB step size
        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.5 * eta + 0.5 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-5, min(eta, 1.0))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        a_new = mirror_descent_step(a, g, eta)

        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys

        a = a_new
        score = evaluate_local(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            print(f"    MD iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score
