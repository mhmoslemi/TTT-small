import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv)/(sum)^2 assuming a is normalized to sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def score_of(a):
    """Compute full score."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    return score_normalized(a_norm)


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, **_solver_kwargs(solver))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def nesterov_soft_minimax(a0, time_limit, top_k=40, T_init=0.005, T_min=0.0005,
                          lr_init=0.015, momentum_init=0.85, verbose=False):
    """
    Nesterov-accelerated subgradient descent with soft-minimax peak weighting.
    Work on the simplex (sum=1). Objective: 2n * max_k conv_k.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    momentum = momentum_init
    lr = lr_init
    T = T_init

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = a + momentum * v

        # Compute convolution at y for gradient
        conv_y = fast_convolve(y)
        M_y = conv_y.max()

        # Soft peak weights
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv_y.copy()
        else:
            peaks = np.argpartition(conv_y, -k_act)[-k_act:]
            pv = conv_y[peaks]

        temp = T * max(M_y, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient: sum_k w_k * d(conv_k)/d(a_i)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # d(conv_k)/d(a_i) = a_{k-i} + a_i (for valid i in [i_start, i_end])
            # where a_{k-i} means y[k-i] (evaluated at y)
            a_part = y[k - i_end:k - i_start + 1][::-1]
            sg[i_start:i_end + 1] += w[idx] * (a_part + y[i_start:i_end + 1])

        # Update
        v = momentum * v - lr * sg
        a_new = simplex_project(a + v)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive learning rate and temperature
        if no_improve > 100:
            lr *= 0.9
            T *= 0.85
            momentum = min(0.95, momentum * 1.005)
            no_improve = 0
            if lr < 1e-12 or T < T_min * 0.1:
                break
        elif no_improve > 50:
            lr *= 0.95
            T = max(T_min, T * 0.95)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSGD it={it} ({elapsed:.0f}s): best={best_score:.9f} lr={lr:.6f} T={T:.6f}")

    return best_a, best_score


def precise_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent with ternary search for optimal value.
    Also tries pairwise mass transfers.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    def local_score(a_trial):
        """Quick score evaluation using incremental update."""
        conv_trial = fast_convolve(a_trial)
        return 2.0 * n * conv_trial.max()

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            # Affected range in convolution
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = conv[:k_start].max()
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
                unaff_max = max(unaff_max, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1].copy()

            def trial_M(new_val):
                """Compute new M if a[i] = new_val (others fixed)."""
                delta = new_val - a[i]
                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                return max(unaff_max, float(new_affected.max()))

            # Ternary search for optimal a[i] in [0, a[i]*2]
            lo = 0.0
            hi = max(a[i] * 2.0, 1e-10)
            if a[i] > 0:
                lo = max(0.0, a[i] * 0.1)

            best_val = a[i]
            best_M_trial = trial_M(a[i])

            for _ in range(12):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                M1 = trial_M(m1)
                M2 = trial_M(m2)
                if M1 <= M2:
                    hi = m2
                else:
                    lo = m1
                # Also check endpoints
                for cand in [lo, hi, (lo + hi) / 2]:
                    Mc = trial_M(cand)
                    if Mc < best_M_trial - 1e-15:
                        best_M_trial = Mc
                        best_val = cand

            if best_M_trial < M - 1e-13 and best_val != a[i]:
                # Apply update
                delta = best_val - a[i]
                a[i] = max(best_val, 0.0)
                changes = 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * (a[i] - delta) + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                # Renormalize
                s_new = a.sum()
                if s_new > 0:
                    # Score is scale-invariant, so normalization doesn't matter
                    # But keep sum=1 for consistency
                    a = a / s_new
                    conv = fast_convolve(a)
                    M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Try a few pairwise transfers
        if sweeps % 5 == 0 and time.monotonic() - t0 < time_limit:
            for _ in range(5):
                i1 = np.random.randint(n)
                i2 = np.random.randint(n)
                if i1 == i2 or a[i1] < 1e-15:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = -a[i1] * frac
                    a_trial = a.copy()
                    a_trial[i1] = max(a[i1] + delta, 0)
                    a_trial[i2] = a[i2] - delta
                    if a_trial[i2] < 0:
                        a_trial[i2] = 0
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    sc = local_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        conv = fast_convolve(a)
                        M = float(conv.max())
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.9f} M={M:.12f}")

    return best_a, best_score

def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n, dtype=np.float64)

    elif init_type == 1:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 2:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(4, 12))
        t = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # Smooth random via low-pass
        freqs = rng.integers(1, 8, size=n)
        phases = rng.uniform(0, 2 * np.pi, size=n)
        t = np.arange(n, dtype=np.float64) / n
        a = 0.5
        for f, p in zip(freqs[:6], phases[:6]):
            a = a + 0.1 * np.cos(2 * np.pi * f * t + p)
        a = np.maximum(a, 0.01)
    else:
        # Bump with varying width
        t = np.arange(n, dtype=np.float64)
        center = n / 2.0
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        print(f"Default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")
    print(f"Initial best: {best_score:.10f}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(candidate)})")

    # === Phase 1: LP Alternating with line search (55% time) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.10f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.55

    # Run from parent
    t_alloc = min(t1_end - time.monotonic(), 500.0)
    print(f"  LP v2 from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating_v2(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Try diverse initializations
    inits = [
        diverse_init(n, rng, 0),
        diverse_init(n, rng, 2),
        diverse_init(n, rng, 4),
        diverse_init(n, rng, 6),
    ]

    for idx, init in enumerate(inits):
        if time.monotonic() > t1_end - 30:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(remaining * 0.35, 150.0)
        print(f"  LP v2 init {idx + 1}/{len(inits)}, budget={t_alloc:.0f}s, init_score={score_normalized(init):.6f}")
        a_tmp, s_tmp = lp_alternating_v2(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_init{idx + 1}")

    # Second LP round on best
    if time.monotonic() < t1_end - 80:
        t_alloc = min(t1_end - time.monotonic(), 250.0)
        print(f"  LP v2 round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating_v2(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_round2")

    # Symmetrization check
    a_sym, improved = symmetrize_and_check(best_seq)
    if improved:
        try_update(a_sym, "symmetrized")

    # === Phase 2: Nesterov Soft-Minimax (20% time) ===
    print(f"\n=== Phase 2: Nesterov Subgradient (best={best_score:.10f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t2_end - time.monotonic(), 200.0)
    if t_alloc > 30:
        print(f"  Nesterov budget={t_alloc:.0f}s")
        a_ns, s_ns = nesterov_soft_minimax(best_seq, t_alloc, top_k=50,
                                           T_init=0.008, T_min=0.0003,
                                           lr_init=0.012, momentum_init=0.8,
                                           verbose=True)
        try_update(a_ns, "nesterov")

        # Also from LP result
        if time.monotonic() < t2_end - 40:
            a_ns2, s_ns2 = nesterov_soft_minimax(a_lp, min(t2_end - time.monotonic(), 100.0),
                                                  top_k=60, T_init=0.005, T_min=0.0003,
                                                  lr_init=0.008, momentum_init=0.85,
                                                  verbose=False)
            try_update(a_ns2, "nesterov_lp")

    # === Phase 3: Precise Coordinate Descent (20% time) ===
    print(f"\n=== Phase 3: Precise CD (best={best_score:.10f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t3_end - time.monotonic(), 250.0)
    if t_alloc > 30:
        print(f"  PCD budget={t_alloc:.0f}s")
        a_cd, s_cd = precise_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "pcd_main")

        # From LP result
        if time.monotonic() < t3_end - 60:
            t_alloc2 = min(t3_end - time.monotonic(), 120.0)
            a_cd2, s_cd2 = precise_coord_descent(a_lp, t_alloc2, verbose=True, label="lp")
            try_update(a_cd2, "pcd_lp")

        # From Nesterov result
        if time.monotonic() < t3_end - 30:
            t_alloc3 = min(t3_end - time.monotonic(), 60.0)
            a_cd3, s_cd3 = precise_coord_descent(a_ns, t_alloc3, verbose=False, label="ns")
            try_update(a_cd3, "pcd_ns")

    # === Phase 4: Final Polish (5% time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.10f}) ===")
    t4_end = deadline - 10

    # Very fine CD
    if time.monotonic() < t4_end - 20:
        t_fp = min(t4_end - time.monotonic() - 10, 80.0)
        if t_fp > 15:
            print(f"  Final PCD budget={t_fp:.0f}s")
            a_fin, s_fin = precise_coord_descent(best_seq, t_fp, verbose=True, label="final")
            try_update(a_fin, "final_pcd")

    # One more LP pass if time
    if time.monotonic() < t4_end - 15:
        t_final_lp = min(t4_end - time.monotonic() - 10, 40.0)
        if t_final_lp > 10:
            a_flp, s_flp = lp_alternating_v2(best_seq, t_final_lp, solver, verbose=False)
            try_update(a_flp, "final_lp")

    # Final symmetrization check
    a_sym_final, imp = symmetrize_and_check(best_seq)
    if imp:
        try_update(a_sym_final, "final_sym")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
