import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size, s=fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def hankel_matvec(w_seq, a):
    """Compute Q_w @ a where Q_w[i,j] = w_seq[i+j]."""
    n = len(a)
    len_w = len(w_seq)
    # (Q_w @ a)[i] = sum_j w_seq[i+j] * a[j]
    # This is a cross-correlation
    result = np.zeros(n)
    for i in range(n):
        max_j = min(n - 1, len_w - 1 - i)
        if max_j >= 0:
            result[i] = np.dot(w_seq[i:i + max_j + 1], a[:max_j + 1])
    return result

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-15:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex (v >= 0, sum(v) = 1)."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        avg_lp_time = np.mean(lp_times[-50:]) if len(lp_times) >= 10 else 1.0

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1

                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.04 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise

                y = y / y.sum()

            if (it + 1) % 500 == 0:

                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp={avg_lp_time*1000:.0f}ms")

def adam_logsumexp_refine(a0, time_limit, T_stages, lr_schedule='cosine', beta1=0.9, beta2=0.999, lr_init=0.5):
    """Adam optimizer with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_a = a0.copy()
    best_score = evaluate_local(a0)

    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        # Determine number of steps for this stage
        steps_per_stage = max(50, int(min(remaining / max(1, len(T_stages) - stage_idx), 2000)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            obj, grad = logsumexp_obj_grad(z, n, stage_T)
            step_count += 1

            # Learning rate schedule
            progress = step / max(1, steps_per_stage)
            if lr_schedule == 'cosine':
                lr = 0.1 * lr_init * (0.5 * (1 + np.cos(np.pi * progress))) + 0.1 * lr_init
            else:
                lr = lr_init / (1 + step * 0.01)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** step_count)
            v_hat = v / (1 - beta2 ** step_count)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if (stage_idx + 1) % 3 == 0:
            print(f"    Adam stage {stage_idx+1}/{len(T_stages)} T={stage_T}: score={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)
    best_a = a0.copy()
    best_score = evaluate_local(a0)

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_i):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_i, n, stage_T)
            return obj

        def jac(z_i):
            if time.monotonic() > deadline:
                return np.zeros_like(z_i)
            _, g = logsumexp_obj_grad(z_i, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            a_new = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a


def spectral_peak_shave(a0, time_limit, n_max_iters=50000):
    """
    Spectral peak shaving: uses FFT to compute autocorrelation,
    applies peak reduction in frequency domain with phase preservation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    target = 1.0 / (2 * n - 1)
    no_improve = 0
    alpha = 0.5

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        A = fft(a_pad)
        power = np.abs(A) ** 2
        b = np.real(ifft(power))[:2 * n - 1]

        # Only reduce peaks above target
        error = np.maximum(b - target, 0)
        if error.max() < 1e-10:
            break

        error_fft = np.real(fft(error, fft_size))
        new_power = power - alpha * error_fft
        new_power = np.maximum(new_power, 1e-20)

        phase = np.angle(A)
        new_A = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.real(ifft(new_A))[:n]

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                alpha *= 0.9
                if alpha < 0.01:
                    break

        if it % 5000 == 0 and it > 0:
            print(f"    Spectral {it}: score={best_score:.8f} alpha={alpha:.4f} ({time.monotonic()-t0:.0f}s)")

    return best_a


def multiplicative_subgradient(a0, time_limit, n_steps=200000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence from diverse templates."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.25) ** 2) + 0.5 * np.exp(-2 * (x - 0.75) ** 2)
    elif choice == 6:
        a = 1.0 + 0.3 * np.cos(np.pi * x)
    elif choice == 7:
        # U-shaped (mild)
        a = 0.3 + 0.7 * (2 * x - 1) ** 2
    elif choice == 8:
        # Double cosine
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        # Gaussian bump
        a = np.exp(-8 * (x - 0.5) ** 2) + 0.2

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
