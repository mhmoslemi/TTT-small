import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def lp_alternating(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    succession = 0  # for adaptive damping
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break
        
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Adaptive damping: start aggressive, get conservative
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.5
        elif it <= 100:
            alpha = 0.3
        else:
            alpha = max(0.1, 5.0 / it)
        
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def fine_coord_descent(a0, time_limit, verbose=True, n_steps=16):
    """
    Very fine coordinate descent. For each coordinate, try n_steps
    candidate values and pick the best one.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    # Pre-generate step factors
    # Mix of multiplicative and absolute steps
    step_factors = np.linspace(-0.4, 0.4, n_steps)
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-16:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            base = max(a[i], 1e-10)
            
            for factor in step_factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

def lp_best_response(a, solver):

    n = len(a)

    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:

def softmax_gd(a0, time_limit, verbose=True, lr_init=0.1, n_peaks=20):
    """
    Gradient descent with softmax parameterization.
    a = exp(v) / sum(exp(v)) ensures positivity and normalization.
    Escapes local minima that simplex projection can get stuck in.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    # Initialize v from a (inverse softmax)
    v = np.log(a + 1e-15) - np.log(a.sum() + 1e-15)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        # Compute softmax
        v_max = v.max()
        e_v = np.exp(v - v_max)
        a = e_v / e_v.sum()

        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-8:
                break

        # Find top peaks
        K = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -K)[-K:]
        pv = conv[peaks]

        # Softmax weighting over peaks
        temp = 0.002 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Gradient of sum_k w_k * conv(a,a)[k] w.r.t. a[j]
        # = sum_k w_k * 2 * a[k-j]
        grad_a = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad_a[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Convert gradient w.r.t. a to gradient w.r.t. v
        # a_j = exp(v_j) / sum(exp(v)), so da_j/dv_m = a_j*(delta_jm - a_m)
        # grad_v = a * (grad_a - a * sum(a * grad_a))... approximately
        # More precisely: grad_v_j = a_j * (grad_a_j - sum_m a_m * grad_a_m)
        # But this is for the softmax with the constraint. Let's use:
        # d(score)/dv_j = sum_i da_i/dv_j * grad_a_i
        # da_i/dv_j = a_i * (delta_ij - sum_m a_m * delta_im) = a_i * (delta_ij - a_j)
        # Wait, that's not right either. Let me be more careful.
        # a_i = exp(v_i) / Z where Z = sum exp(v_m)
        # da_i/dv_j = a_i * delta_ij - a_i * exp(v_j)/Z = a_i * (delta_ij - a_j)
        # So d(score)/dv_j = sum_i a_i * (delta_ij - a_j) * grad_a_i
        # = a_j * grad_a_j - a_j * sum_i a_i * grad_a_i
        # = a_j * (grad_a_j - sum_i a_i * grad_a_i)

        a_dot_grad = np.dot(a, grad_a)
        grad_v = a * (grad_a - a_dot_grad)

        # Update v
        v = v - lr * grad_v

        # Prevent v from drifting too far
        v = v - v.mean()

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftmaxGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

def peak_directed_cd(a0, time_limit, verbose=True):
    """Coordinate descent focusing on coordinates contributing to top peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top K peaks
        K = 15
        peak_indices = np.argpartition(conv, -K)[-K:]

        # Find contributing coordinates
        target_coords = set()
        for k_star in peak_indices:
            k_star = int(k_star)
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            for i in range(i_start, i_end + 1):
                if a[i] > 1e-8 and a[k_star - i] > 1e-8:
                    target_coords.add(i)
                    target_coords.add(k_star - i)

        # Optimize target coordinates
        for i in sorted(target_coords):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M
            base = max(a[i], 1e-10)

            factors = [-0.3, -0.15, -0.05, -0.01, 0.01, 0.05, 0.15, 0.3,
                       -0.005, 0.005, -0.5, 0.5, -0.1, 0.1]
            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and rounds % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / (center + 1), 0)
    elif init_type == 1:
        x = (t - center) / (center + 1)
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / (center + 1)
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        c1 = n * rng.uniform(0.25, 0.35)
        c2 = n * rng.uniform(0.65, 0.75)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        n_freq = int(rng.integers(3, 12))
        freqs = rng.uniform(1, 8, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = rng.uniform(0.1, 0.8, size=n_freq)
        a = np.ones(n) * 0.5
        for f, p, amp in zip(freqs, phases, amps):
            a = a + amp * np.cos(2 * np.pi * f * t / n + p)
        a = np.maximum(a, 0)
    elif init_type == 5:
        a = rng.random(n)
        k = int(rng.integers(3, 9))
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    elif init_type == 6:
        # Periodic with random period
        period = int(rng.integers(3, 15))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period) + 0.1 * rng.random(n)
        a = np.maximum(a, 0)
    else:
        a = rng.random(n)
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_update(candidate, label):
    """Try to update the global best with a candidate sequence."""
    global _best_seq, _best_score
    candidate = np.maximum(candidate, 0.0)
    s = candidate.sum()
    if s < 0.01:
        return
    try:
        sc = float(evaluate_sequence(candidate.tolist()))
    except Exception:
        sc = float('inf')
    if np.isfinite(sc) and sc < _best_score:
        _best_score = sc
        _best_seq = candidate.copy()
        print(f"  NEW BEST [{label}]: {_best_score:.8f} (n={len(candidate)})")


def propose_candidate():
    """Main search function: multi-phase optimization to minimize the score."""
    global _best_seq, _best_score

    deadline = time.monotonic() + 1760.0
    rng = np.random.default_rng(12345)

    # Load parent sequence
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    n = len(parent)
    init_seq = parent / parent.sum()
    _best_seq = init_seq.copy()
    _best_score = float(evaluate_sequence(_best_seq.tolist()))
    print(f"Parent: n={n}, score={_best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    # === Phase 1: LP Alternating (45% time) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.45

    # Main LP from parent - long run
    t_alloc = min(t1_end - time.monotonic(), 400.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(_best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # LP from diverse initializations
    n_inits = 8
    for idx in range(n_inits):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(remaining * 0.25, 80.0)
        init = diverse_init(n, rng, idx % 7)
        print(f"  LP init {idx+1}/{n_inits}, budget={t_alloc:.0f}s")
        a_tmp, _ = lp_alternating(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_init{idx+1}")

    # Second round on current best
    if time.monotonic() < t1_end - 80:
        t_alloc = min(t1_end - time.monotonic(), 200.0)
        print(f"  LP round 2, budget={t_alloc:.0f}s")
        a_lp2, _ = lp_alternating(_best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_r2")

    # === Phase 2: Softmax Gradient Descent (15% time) ===
    print(f"\n=== Phase 2: Softmax GD (best={_best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    if time.monotonic() < t2_end - 20:
        t_alloc = min(t2_end - time.monotonic(), 150.0)
        print(f"  SoftmaxGD, budget={t_alloc:.0f}s")
        a_sgd, _ = softmax_gd(_best_seq, t_alloc, lr_init=0.05, n_peaks=25, verbose=True)
        try_update(a_sgd, "softmax_gd")

        # Second round with different params
        if time.monotonic() < t2_end - 20:
            t_alloc2 = min(t2_end - time.monotonic(), 60.0)
            a_sgd2, _ = softmax_gd(_best_seq, t_alloc2, lr_init=0.02, n_peaks=40, verbose=False)
            try_update(a_sgd2, "softmax_gd2")

    # === Phase 3: Fine Coordinate Descent (25% time) ===
    print(f"\n=== Phase 3: Fine CD (best={_best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.25

    if time.monotonic() < t3_end - 30:
        t_alloc = min(t3_end - time.monotonic(), 200.0)
        print(f"  FineCD, budget={t_alloc:.0f}s")
        a_cd, _ = fine_coord_descent(_best_seq, t_alloc, verbose=True)
        try_update(a_cd, "fine_cd")

    # Peak-directed CD
    if time.monotonic() < t3_end - 40:
        t_alloc = min(t3_end - time.monotonic(), 100.0)
        if t_alloc > 20:
            print(f"  PeakCD, budget={t_alloc:.0f}s")
            a_pc, _ = peak_directed_cd(_best_seq, t_alloc, verbose=True)
            try_update(a_pc, "peak_cd")

    # === Phase 4: Random Restarts + Short Optimization (10% time) ===
    print(f"\n=== Phase 4: Restarts (best={_best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.10

    n_restarts = 6
    for r in range(n_restarts):
        if time.monotonic() > t4_end:
            break
        remaining_t4 = t4_end - time.monotonic()
        if remaining_t4 < 30:
            break

        # Different perturbation strategies
        if r % 3 == 0:
            # Gaussian noise
            noise_scale = rng.uniform(0.03, 0.08)
            a_pert = _best_seq + rng.standard_normal(n) * noise_scale
        elif r % 3 == 1:
            # Sparse perturbation (modify a few random coordinates)
            a_pert = _best_seq.copy()
            n_mod = int(rng.integers(10, 50))
            idxs = rng.choice(n, size=n_mod, replace=False)
            a_pert[idxs] = a_pert[idxs] * rng.uniform(0.3, 2.0, size=n_mod)
        else:
            # Blend with a diverse init
            init = diverse_init(n, rng, r % 7)
            blend = rng.uniform(0.1, 0.4)
            a_pert = (1 - blend) * _best_seq + blend * init

        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s > 0:
            a_pert = a_pert / s

        t_alloc = min(remaining_t4 * 0.35, 60.0)
        a_r, _ = lp_alternating(a_pert, min(t_alloc, 40.0), solver, verbose=False)
        try_update(a_r, f"restart_{r}_lp")

        if time.monotonic() < t4_end - 15:
            t_cd = min(t4_end - time.monotonic(), 25.0)
            a_r2, _ = fine_coord_descent(a_r, t_cd, verbose=False)
            try_update(a_r2, f"restart_{r}_cd")

    # === Phase 5: Final Polish (5% time) ===
    print(f"\n=== Phase 5: Final Polish (best={_best_score:.8f}) ===")

    if time.monotonic() < deadline - 20:
        t_alloc = min(deadline - time.monotonic() - 20, 60.0)
        print(f"  Final FineCD, budget={t_alloc:.0f}s")
        a_fin, _ = fine_coord_descent(_best_seq, t_alloc, verbose=True)
        try_update(a_fin, "final_cd")

    if time.monotonic() < deadline - 10:
        t_alloc = min(deadline - time.monotonic() - 10, 30.0)
        a_fin2, _ = peak_directed_cd(_best_seq, t_alloc, verbose=False)
        try_update(a_fin2, "final_peak")

    if time.monotonic() < deadline - 5:
        t_alloc = min(deadline - time.monotonic() - 5, 20.0)
        # One more LP pass
        a_fin3, _ = lp_alternating(_best_seq, t_alloc, solver, verbose=False)
        try_update(a_fin3, "final_lp")

    # Final verification
    final_score = float(evaluate_sequence(_best_seq.tolist()))
    if np.isfinite(final_score):
        _best_score = final_score

    print(f"\nFINAL SCORE: {_best_score:.8f} (n={len(_best_seq)})")
    return [float(x) for x in _best_seq]
