"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated LP alternating
with HiGHS solver, LogSumExp mirror descent refinement, and coordinate descent polish.
Key: fast LP solving with dense matrix construction via numpy, Nesterov acceleration
for O(1/t) convergence, and multi-phase optimization within the 1800s budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    m = len(b)
    size = n + m - 1
    if size <= 1:
        return a * b if m == n else np.array([a[0]*b[0]])
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix_dense(y):
    """Build dense (2n-1) x n matrix T such that T @ b = convolve(y, b)."""
    n = len(y)
    size = 2 * n - 1
    k = np.arange(size)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    T = np.where(valid, y[np.clip(idx, 0, n-1)], 0.0)
    return T

def solve_best_response(y):
    """Solve min M s.t. conv(y,b)_k <= M, sum(b)=1, b>=0."""
    n = len(y)
    T = build_conv_matrix_dense(y)
    size = T.shape[0]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: [T, -1] @ [b, M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'disp': False, 'dual_feasibility_tolerance': 1e-9,
                              'primal_feasibility_tolerance': 1e-9})
    
    if result.success:
        b = result.x[:n]
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-15:
            b = b / s
        M = float(result.x[-1])
        return b, M
    return None, None

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def nesterov_lp(a0, time_limit, verbose=True):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    prev_y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov extrapolation
        if it >= 2:
            beta = (it - 1) / (it + 2)
            y_new = y + beta * (y - prev_y)
            y_new = np.maximum(y_new, 0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
            prev_y = y
            y = y_new
        else:
            prev_y = y.copy()
        
        # Solve LP for best response to y
        b_new, M_val = solve_best_response(y)
        if b_new is None:
            continue
        
        a = b_new
        score = 2.0 * n * fast_convolve(a).max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 5000 == 0:

            print(f"  LP iter {it} ({elapsed:.0f}s): score={best_score:.8f}, M={M_val:.8f}")
        
        # Convergence check
        if it > 100 and abs(score - best_score) < 1e-12 and it % 10000 == 0:
            # Check if we're converged
            pass
    
    return best_a, best_score

def logsumexp_mirror_descent(a0, time_limit, T_start=5.0, T_end=200.0, 
                             n_stages=8, iters_per_stage=200):
    """LogSumExp smoothed mirror descent with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    T_schedule = np.geomspace(T_start, T_end, n_stages)
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        stage_time = time_limit / n_stages
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - t0 > stage_time * (T_schedule.index(T) + 1) + t0:
                break
            
            # Compute b = conv(a, a)
            b = fast_convolve(a)
            
            # Compute softmax weights
            T_b = T * b
            T_b_shifted = T_b - T_b.max()
            w = np.exp(T_b_shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
