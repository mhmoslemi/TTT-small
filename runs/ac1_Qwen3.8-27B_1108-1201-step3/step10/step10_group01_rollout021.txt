OK, one more important consideration: the "pairwise swap" variant of Frank-Wolfe.

In standard FW:
- v = e_{argmin s_i}
- a ← (1-t)a + tv

In pairwise swap:
- Find (i,j) that maximizes s_j - s_i (the steepest direction on the simplex)
- Move mass from j to i: a_j -= δ, a_i += δ
- Choose δ by line search

The pairwise swap is strictly better than standard FW because it uses a 2D face of the simplex instead of a 1D edge.

For our problem, the subgradient s_i = (2/|A|) * sum_{k in A} a_{k-i} (valid indices only).

The "largest" s_i is at positions that are "correlated" with the active peaks. The "smallest" s_i is at positions that are "anti-correlated."

So pairwise swap would move mass from "high-correlation" positions to "low-correlation" positions. This makes intuitive sense: we want to reduce the contribution to the peaks.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, active_set):
    """Compute subgradient of max_k conv(a,a)_k at positions in active_set."""
    n = len(a)
    s = np.zeros(n)
    for k in active_set:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                s[i] += 2.0 * a[j]
    s /= len(active_set)
    return s

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Adaptive parameters
    active_tol = 0.01  # positions within 1% of max are "active"
    n_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: positions within tolerance of max
        threshold = max_b * (1 - active_tol)
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        sg = compute_subgradient_direct(a, active)
        
        # Find steepest direction on simplex: max sg - min sg
        i_min = np.argmin(sg)
        i_max = np.argmax(sg)
        
        if sg[i_max] - sg[i_min] < tol:
            break  # Convergence
        
        # Line search: find optimal delta
        # We move delta from i_max to i_min
        # x(δ) = a - δ*e_{i_max} + δ*e_{i_min}
        # Minimize f(x(δ)) = max_k conv(x(δ), x(δ))_k
        
        # For efficiency, use a few tries
        best_delta = 0.0
        best_f = 2.0 * n * max_b
        
        # Try a range of deltas
        max_delta = a[i_max]  # Can't remove more than what's there
        for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            δ = frac * max_delta
            if δ < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i_max] -= δ
            a_trial[i_min] += δ
            f_trial = 2.0 * n * fast_convolve(a_trial).max()
            if f_trial < best_f:
                best_f = f_trial
                best_delta = δ
        
        if best_delta < 1e-15:

        a[i_max] -= best_delta

Actually, I can be smarter about the line search. Since we're only changing two coordinates, the change in the convolution is local:

conv(a + δ(e_i - e_j), a + δ(e_i - e_j))[k] = conv(a,a)[k] + δ * (conv(a, e_i - e_j) + conv(e_i - e_j, a))[k] + δ^2 * conv(e_i - e_j, e_i - e_j)[k]

The linear term is: conv(a, e_i)[k] - conv(a, e_j)[k] + conv(e_i, a)[k] - conv(e_j, a)[k]
= a[i] shifted + a[k-i] - a[j] shifted - a[k-j]
= 2a[k-i] (shifted by i) - 2a[k-j] (shifted by j) (approximately)

The quadratic term is: e_i*e_i - e_i*e_j - e_j*e_i + e_j*e_j, which is 1 at k=2i, 1 at k=2j, -1 at k=i+j (twice, but it's the same position).

So the change in conv at position k is:
Δ_k = 2δ(a[k-i] - a[k-j]) * [valid] + δ^2 (δ_ki+ki - δ_ki+kj - δ_kj+ki + δ_kj+kj)

This is O(1) per position k (just a few index lookups). So we can compute the entire new convolution in O(n) without an FFT!

This makes the line search much cheaper: O(n) per evaluation instead of O(n log n).

def conv_diff_two_coords(a, b, i, j, delta):
    """Compute new convolution after moving delta from j to i, given old conv b.
    Returns the new convolution vector."""
    n = len(a)
    # a_new = a + delta*(e_i - e_j)
    # conv(a_new, a_new) = conv(a,a) + delta*(conv(a, e_i) + conv(e_i, a) - conv(a, e_j) - conv(e_j, a)) + delta^2 * conv(e_i-e_j, e_i-e_j)
    # conv(a, e_i)[k] = a[k-i] (if valid) = 0 otherwise
    # conv(e_i, a)[k] = a[k-i] (if valid) = 0 otherwise
    # So the linear term is 2*(a[k-i] - a[k-j]) for valid k
    
    linear = np.zeros(2*n-1)
    # conv(a, e_i) is a shifted version of a: position k gets a[k-i]
    for k in range(i, i+n):
        linear[k] += a[k-i]
    for k in range(2*i, 2*i+1):  # conv(e_i, a): position k gets a[k-i]... 
        pass
    # Actually conv(e_i, a)[k] = a[k-i] if 0 <= k-i < n
    # This is the same as conv(a, e_i) by symmetry!
    # So linear term = 2 * (shifted_a_i - shifted_a_j)
    
    # Let me just compute it directly
    linear = np.zeros(2*n-1)
    for k in range(max(0, i), min(2*n-1, i+n)):
        j_idx = k - i
        if 0 <= j_idx < n:
            linear[k] += 2.0 * a[j_idx]
    for k in range(max(0, j), min(2*n-1, j+n)):
        j_idx = k - j
        if 0 <= j_idx < n:
            linear[k] -= 2.0 * a[j_idx]
    
    # Quadratic term: conv(e_i - e_j, e_i - e_j)
    # = conv(e_i, e_i) - conv(e_i, e_j) - conv(e_j, e_i) + conv(e_j, e_j)
    # conv(e_i, e_i)[k] = 1 if k=2i
    # conv(e_i, e_j)[k] = 1 if k=i+j
    # So quadratic = delta^2 * (1_{k=2i} + 1_{k=2j} - 2*1_{k=i+j})
    
    quad = np.zeros(2*n-1)
    if 2*i < 2*n-1:
        quad[2*i] = 1.0
    if 2*j < 2*n-1:
        quad[2*j] = 1.0
    if i+j < 2*n-1:
        quad[i+j] -= 2.0
    
    new_conv = b + delta * linear + delta*delta * quad
    return np.maximum(new_conv, 0)  # Should be non-negative but numerical errors
