            stagnation = 0
        else:
            stagnation += 1
        anneal_temp *= 0.997
        if stagnation > 60:
            noise_scale = 0.03 / np.sqrt(n)
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score * 1.003:
                    a = a_pert
                    conv = fast_convolve(a)
                    best_a = a.copy()
                    best_score = 2.0 * n * conv.max()
                stagnation = 0
                anneal_temp = max(anneal_temp, 0.015)
        if verbose and it % 100 == 0:
            print(f"    LP[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} K={K} alpha={alpha:.3f}")
    return best_a, best_score


def coord_descent_polish(a0, time_limit, verbose=True, label="", fine=False):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    if fine:
        factors = [0.997, 0.994, 0.998, 0.999, 1.001, 1.002, 1.006, 1.003]
    else:
        factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 2.0, 0.3]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if a[i] < 1e-16:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            best_delta = 0.0
            best_new_M = M
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    if not new_affected.flags['owndata']:
                        new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(old_val + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    if not changes.flags['owndata']:
                        changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > (30 if not fine else 15):
                break
        else:
            no_improve = 0
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def diverse_init(n, rng, init_type):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        t = np.arange(n)
        period = int(rng.integers(3, 15))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Chirp-like
        t = np.arange(n)
        rate = rng.uniform(2, 10)
        a = 0.5 + 0.4 * np.cos(np.pi * rate * t * t / n)
        a = np.maximum(a, 0)
    elif init_type == 7:
        # Sparse peaks
        a = np.zeros(n)
        num_peaks = int(rng.integers(5, 20))
        positions = rng.integers(0, n, num_peaks)
        heights = rng.uniform(0.1, 1.0, num_peaks)
        for pos, h in zip(positions, heights):
            width = int(rng.integers(2, 10))
            for d in range(-width, width + 1):
                idx = (pos + d) % n
                a[idx] += h * np.exp(-0.5 * (d / (width * 0.5)) ** 2)
    else:
        t = np.arange(n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t * 7 / n) + 0.2 * np.cos(2 * np.pi * t * 13 / n)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def upscale(a, target_n):
    old_n = len(a)
    if old_n == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n, endpoint=False)
    t_new = np.linspace(0, 1, target_n, endpoint=False)
    a_new = np.interp(t_new, t_old, a)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return a_new / s


def propose_candidate():
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = find_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(candidate, 0.0).astype(np.float64)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n_base = len(parent)
        try_update(parent, "parent")
    else:
        n_base = 1024
        parent = np.ones(n_base) / n_base
        try_update(parent, "uniform")

    # === Phase 1: LP alternating on base length (55% of time) ===
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.50
    print(f"\n=== Phase 1: LP Alternating on n={n_base} (best={best_score:.8f}) ===")

    t_alloc = min(t1_end - time.monotonic(), 500.0)
    if t_alloc > 30:
        print(f"  LP alternating from parent, budget={t_alloc:.0f}s")
        a_lp, s_lp = lp_alternating_refined(best_seq, t_alloc, solver,
                                            K_start=50, K_end=400, verbose=True, label="parent")
        try_update(a_lp, "lp_parent")

    # LP from diverse inits at same n
    inits_to_try = [
        (diverse_init(n_base, rng, 0), "tri"),
        (diverse_init(n_base, rng, 4), "periodic"),
        (diverse_init(n_base, rng, 6), "chirp"),
        (diverse_init(n_base, rng, 7), "sparse"),
    ]
    for idx, (init, name) in enumerate(inits_to_try):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(remaining * 0.5, 100.0)
        print(f"  LP init '{name}', budget={t_alloc:.0f}s")
        a_tmp, s_tmp = lp_alternating_refined(init, t_alloc, solver,
                                              K_start=30, K_end=200, verbose=False, label=name)
        try_update(a_tmp, f"lp_{name}")

    # Second round on best
    if time.monotonic() < t1_end - 60:
        t_alloc = min(t1_end - time.monotonic(), 250.0)
        print(f"  LP round 2 on best, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating_refined(best_seq, t_alloc, solver,
                                              K_start=80, K_end=500, verbose=True, label="round2")
        try_update(a_lp2, "lp_round2")

    # === Phase 2: Try n=2048 (25% of time) ===
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.25
    print(f"\n=== Phase 2: Try n=2048 (best={best_score:.8f}) ===")

    n2 = 2048
    # Upscale parent to 2048
    a_2048 = upscale(best_seq, n2)
    try_update(a_2048, "upscaled_2048")

    if time.monotonic() < t2_end - 30:
        t_alloc = min(t2_end - time.monotonic(), 300.0)
        print(f"  LP alternating n=2048, budget={t_alloc:.0f}s")
        a_2k, s_2k = lp_alternating_refined(a_2048, t_alloc, solver,
                                            K_start=50, K_end=300, verbose=True, label="n2048")
        try_update(a_2k, "lp_2048")

    # Also try structured init at 2048
    if time.monotonic() < t2_end - 40:
        a_chirp_2k = diverse_init(n2, rng, 6)
        t_alloc = min(t2_end - time.monotonic(), 120.0)
        print(f"  LP init chirp n=2048, budget={t_alloc:.0f}s")
        a_c2k, s_c2k = lp_alternating_refined(a_chirp_2k, t_alloc, solver,
                                              K_start=30, K_end=200, verbose=False, label="chirp2k")
        try_update(a_c2k, "chirp_2048")

    # === Phase 3: Coordinate descent polishing (20% of time) ===
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")

    t_alloc = min(t3_end - time.monotonic(), 200.0)
    if t_alloc > 30:
        print(f"  CD main, budget={t_alloc:.0f}s")
        a_cd, s_cd = coord_descent_polish(best_seq, t_alloc, verbose=True, label="main", fine=False)
        try_update(a_cd, "cd_main")

    if time.monotonic() < t3_end - 30:
        t_alloc2 = min(t3_end - time.monotonic(), 100.0)
        print(f"  CD fine, budget={t_alloc2:.0f}s")
        a_cd2, s_cd2 = coord_descent_polish(best_seq, t_alloc2, verbose=True, label="fine", fine=True)
        try_update(a_cd2, "cd_fine")

    # === Phase 4: Final LP + CD combo (remaining time) ===
    t4_end = deadline - 30
    print(f"\n=== Phase 4: Final refinement (best={best_score:.8f}) ===")

    if time.monotonic() < t4_end - 60:
        t_alloc = min(t4_end - time.monotonic() - 60, 150.0)
        print(f"  Final LP, budget={t_alloc:.0f}s")
        a_fin, s_fin = lp_alternating_refined(best_seq, t_alloc, solver,
                                              K_start=100, K_end=600, verbose=True, label="final_lp")
        try_update(a_fin, "final_lp")

    if time.monotonic() < t4_end - 20:
        t_alloc = min(t4_end - time.monotonic() - 20, 80.0)
        print(f"  Final fine CD, budget={t_alloc:.0f}s")
        a_fin2, s_fin2 = coord_descent_polish(best_seq, t_alloc, verbose=False, label="final_fine", fine=True)
        try_update(a_fin2, "final_fcd")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        result = [float(x) for x in best_seq]
    else:
        result = [1.0 / 1024] * 1024
        print(f"\nFALLBACK: uniform n=1024")

    return result
