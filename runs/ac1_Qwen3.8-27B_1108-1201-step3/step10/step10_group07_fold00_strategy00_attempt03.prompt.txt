def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

            no_improve += 1

def lp_alternating(a0, time_limit, solver, verbose=True):
    """
    LP-based best-response alternating minimization.
    a -> b*(a) -> a*(b*) -> ...
    Uses full iterate initially, then damped updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: best response b* to current a
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Evaluate b_new
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Decide update strategy
        if it <= 20:
            # Full iterate
            a = b_new.copy()
        elif it <= 50:
            # Aggressive damping
            alpha = 0.7
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            # Conservative damping
            alpha = max(0.3, 10.0 / (it + 10.0))
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        # Evaluate updated a
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check stagnation
        if score_a > best_score * (1 + 1e-8):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        # If stuck, try a random perturbation
        if stagnation > 30:
            noise_scale = 0.01 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.001:
                a = a_pert
            stagnation = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        # Triangle (peaked in middle)
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        # U-shape (high at edges)
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        # Cosine window
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Two humps
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Periodic
        period = int(rng.integers(3, 10))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # W-shape
        t = np.arange(n)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    else:
        # Smooth random
        freqs = rng.integers(0, 20, size=n)
        phases = rng.uniform(0, 2 * np.pi, size=n)
        a = 0.5 + 0.4 * np.cos(freqs * np.pi * 2 * np.arange(n) / n + phases)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def peak_shaving(a0, time_limit, verbose=True):
    """
    Focus specifically on reducing the top peaks.
    For each peak, identify contributing coordinate pairs and
    redistribute mass to reduce the peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top peaks
        K = 5
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.95:
                continue

            # Find the most contributing coordinate pairs
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Contributions: a[i] * a[k_star - i]
            pairs = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                pairs.append((i, j, a[i] * a[j]))

            pairs.sort(key=lambda x: x[2], reverse=True)

            # Try reducing the top contributing pair
            for (i, j, contrib) in pairs[:3]:
                if contrib < 1e-12:
                    continue
                # Try reducing the larger of a[i], a[j]
                if a[i] >= a[j]:
                    target = i
                    other = j
                else:
                    target = j
                    other = i

                # Try reducing target by various amounts
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = -a[target] * frac
                    if delta == 0:
                        continue

                    # Move mass to 'other' coordinate (keeps sum same approximately)
                    # Actually, let's just try reducing and see
                    new_a = a.copy()
                    new_a[target] = max(new_a[target] + delta, 0)

                    # Update convolution incrementally
                    k_s = target
                    k_e = min(target + n - 1, 2 * n - 2)
                    new_conv = conv.copy()
                    new_conv[k_s:k_e + 1] += 2.0 * delta * a[k_s - target:k_e - target + 1]
                    if 2 * target >= k_s and 2 * target <= k_e:
                        idx = 2 * target - k_s
                        new_conv[idx] = conv[2 * target] + 2.0 * delta * a[target] + delta * delta

                    new_M = float(new_conv.max())
                    if new_M < M - 1e-12:
                        a = new_a
