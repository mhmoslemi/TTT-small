def fast_convolve(a):

def evaluate_local(a):

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute gradient of sum_k lambda_k * conv(a,a)_k with respect to a.
    grad_i = 2 * sum_{m=0}^{n-1} lambda_{i+m} * a_m
    """
    n = len(a)
    grad = 2.0 * fast_cross_correlate(a, lambdas)
    return grad

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using IRLS (Iterative Reweighted

The problem is convex, so IRLS with power annealing converges to the

    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_correlate(a, b):
    """result[i] = sum_{m=0}^{n-1} a[m] * b[i+m]"""
    n = len(a)

    size = n + len(b) - 1

    A = rfft(a_flip, fft_size)
    B = rfft(b, fft_size)

    n = len(a)

    """grad_i = 2 * sum_{m=0}^{n-1} lambdas[i+m] * a[m]"""
    return 2.0 * fast_cross_correlate(a, lambdas)

            w = w / w.sum()

            grad = compute_irls_grad(a, w)

                f_trial = np.dot(w, b_trial)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using IRLS (Iterative Reweighted Least
Squares) with FFT-accelerated gradient, momentum, and power annealing.
The problem is convex (max of quadratic forms over a simplex), so IRLS with
increasing power converges to the global optimum. Includes multiple init
strategies, different sequence lengths, and coordinate descent polish.

import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_autocorr(a):
    """Same as fast_convolve for auto-convolution."""
    return fast_convolve(a)


def fast_cross_corr(a, b):
    """Cross-correlation: result[i] = sum_{m=0}^{n-1} a[m] * b[i+m]
    where len(a) = n, len(b) = 2n-1, len(result) = n."""
    n = len(a)
    a_flip = a[::-1]
    size = n + len(b) - 1
    fft_size = next_fast_len(size)
    A = rfft(a_flip, fft_size)
    B = rfft(b, fft_size)
    conv = irfft(A * B, fft_size)[:size]
    return conv[n - 1:n - 1 + n]


def evaluate_local(a):
    """Evaluate sequence using local computation (normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_irls(a, w):
    """Gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    """
    return 2.0 * fast_cross_corr(a, w)


def irls_power_schedule(p_start, p_end, n_steps):
    """Geometric power schedule from p_start to p_end."""
    return np.geomspace(p_start, p_end, n_steps)


def irls_optimize(a0, time_limit, p_start=0.5, p_end=200.0,
                  n_p_steps=35, steps_per_p=300, momentum=0.85):
    """
    IRLS optimization with power annealing and momentum.
    Minimizes max_k conv(a,a)_k over the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    else:
        a = a / s

    best_a = a.copy()
    b0 = fast_convolve(a)
    best_score = 2.0 * n * b0.max()
    t0 = time.monotonic()

    p_schedule = irls_power_schedule(p_start, p_end, n_p_steps)
    v = np.zeros(n)

    for pi, p in enumerate(p_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute IRLS weights
        b = fast_convolve(a)
        b_safe = np.maximum(b, 1e-30)

        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = w / w_sum

        grad = compute_grad_irls(a, w)

        # Clamp eta

        eta = max(eta, 1e-8)

def coordinate_descent_polish(a0, time_limit, n_steps=100000):
    """
    Fine-grained coordinate descent targeting peak-contributing elements.
    Tries to reduce the max convolution by adjusting individual coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find peak-contributing positions
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing top contributors
        top_m = min(4, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.995]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()

                b_trial = fast_convolve(a_trial)
                max_trial = b_trial.max()
                score = 2.0 * n * max_trial
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Try increasing a low contributor at the peak
                low_positions = idx[np.argsort(contribs)[:3]]
                found = False
                for pos in low_positions:
                    a_trial = a.copy()
                    a_trial[pos] *= 1.001
                    a_trial = a_trial / a_trial.sum()
                    b_trial = fast_convolve(a_trial)
                    score = 2.0 * n * b_trial.max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        found = True
                        break
                if not found and stall > 1000:
                    break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, eta0=0.05, active_frac=0.999,
                            nesterov=True):
    """
    Mirror descent on the simplex with Nesterov acceleration.
    Uses exponential weighting (entropic regularization).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_g = np.arange(js, je + 1)
            g[idx_g] += a[k - idx_g]
        g /= max(1, len(active))

        # Mirror descent update (exponential weights)
        g_shifted = g - g.min()
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2.0)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new

        # Track best
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, 0.01)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def logsumexp_smooth_max(a, T):
