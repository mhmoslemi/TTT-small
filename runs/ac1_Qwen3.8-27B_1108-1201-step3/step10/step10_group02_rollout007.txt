Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization on softmax

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective (with sum(a) normalization handled externally)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def power_mean_grad(z, p):
    """
    Compute objective value and gradient of 2*n*power_mean_p(conv(softmax(z), softmax(z))).
    """
    a = softmax(z)
    n = len(a)
    b = fast_convolve(a)
    
    # Power mean
    if p == 1:
        M_p = b.max()  # degenerate case, shouldn't happen
    else:
        b_p = b ** p
        sum_bp = b_p.sum()
        M_p = sum_bp ** (1.0 / p)
    
    # db/d b_k = b_k^(p-1) / M_p
    if p == 1:
        db = np.zeros_like(b)
        db[np.argmax(b)] = 1.0
    else:
        db = b ** (p - 1) / M_p
    
    # grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * db[i+j]
    # Use FFT: c[i] = conv(a, db_rev)[2n-2-i]
    db_rev = db[::-1]  # length 2n-1
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    DBR = rfft(db_rev, fft_size)
    conv_full = irfft(A * DBR, fft_size)[:conv_len]
    c = conv_full[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * c
    
    # Softmax chain rule
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    obj = 2.0 * n * M_p
    return obj, grad_z


def adam_optimize(z0, p_schedule, time_limit, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimization with a schedule of power-mean exponents.
    """
    n = len(z0)
    z = z0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_z = z.copy()
    best_a = softmax(z)
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()
    
    no_improve = 0
    for stage, p in enumerate(p_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()
        
        while time.monotonic() - t0 < time_limit:
            t += 1
            obj, grad = power_mean_grad(z, p)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Redirect gradient (to keep z centered, for numerical stability)
            z = z - z.mean()
            
            a = softmax(z)
            score = evaluate_local(a)
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 5000:
                break
        
        # Between stages, update Adam state
        # (keeping it is fine, the gradient scale changes with p)
        if stage % 2 == 1:
            print(f"    Stage {stage}: p={p}, best={best_score:.8f}, t={t}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_steps=200000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find peaks within 0.1% of max
        threshold = max_b * 0.999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:5]:  # limit to 5 peaks
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 2000:
                break
        else:
            stall = 0
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score

    choices = [0, 1, 2, 3, 4, 5, 6]

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization on softmax
parameterization with power-mean smoothing, multiple asymmetric initializations,
and coordinate descent polish. Key insight: non-symmetric sequences can achieve
lower peak autocorrelation than symmetric ones, since conv(a,a)[n-1] = sum a[i]*a[n-1-i]
can be made small by anti-correlating a with its reverse.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective for a non-negative sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def power_mean_obj_grad(z, p):
    """
    Compute objective and gradient of 2*n*power_mean_p(conv(softmax(z), softmax(z))).
    Uses normalized power mean for numerical stability.
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    if b_max < 1e-300:
        return 0.0, np.zeros_like(z)
    
    # Normalized power mean: M_p = b_max * (sum (b/b_max)^p)^(1/p)
    b_norm = b / b_max
    b_p = b_norm ** p
    sum_bp = b_p.sum()
    if sum_bp < 1e-300:
        return 0.0, np.zeros_like(z)
    M_p_norm = sum_bp ** (1.0 / p)
    M_p = b_max * M_p_norm
    
    # d M_p / d b_k = b_norm[k]^(p-1) / M_p_norm
    db = b_norm ** (p - 1) / M_p_norm
    
    # grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * db[i+j]
    # Computed via FFT: c[i] = conv(a, db_rev)[2n-2-i]
    db_rev = db[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    DBR = rfft(db_rev, fft_size)
    conv_full = irfft(A * DBR, fft_size)[:conv_len]
    c = conv_full[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * c
    
    # Softmax chain rule: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    obj = 2.0 * n * M_p
    return obj, grad_z


def adam_optimize(z0, p_schedule, time_limit, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8,
                  max_no_improve=8000, clip_grad=5.0):
    """
    Adam optimization with power-mean schedule and gradient clipping.
    Returns (best_a, best_score).
    """
    n = len(z0)
    z = z0.copy()
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = softmax_fn(z)
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()
    
    for stage, p in enumerate(p_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        no_improve = 0
        stage_best = best_score
        
        while time.monotonic() - t0 < time_limit:
            t += 1
            obj, grad = power_mean_obj_grad(z, p)
            
            # Gradient clipping for stability at high p
            g_norm = np.linalg.norm(grad)
            if g_norm > clip_grad:
                grad = grad * (clip_grad / g_norm)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.mean()
            
            a = softmax_fn(z)
            score = evaluate_local(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > max_no_improve:
                break
        
        if (stage + 1) % 3 == 0 or stage == len(p_schedule) - 1:
            elapsed = time.monotonic() - t0
            print(f"    Adam stage {stage+1}/{len(p_schedule)} (p={p}, {elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coord_desc_refine(a0, time_limit, n_steps=100000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        if time.monotonic() - t0 > time_limit:

        b = fast_convolve(a)
        max_b = b.max()
        
        # Find peaks within 0.05% of max
        threshold = max_b * 0.99995
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:8]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(12, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 3000:
                break
            # Periodic multi-coordinate nudge when stuck

                peak_k = int(np.argmax(b))

                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def make_asymmetric_init(n, rng):
    """Create an asymmetric initial sequence."""
    choice = int(rng.integers(0, 8))
    x = np.linspace(0, 1, n)
    
    if choice == 0:

    elif choice == 1:

        a = np.exp(-5.0 * (x - center) ** 2 / width)
    elif choice == 2:
        a = np.exp(2.5 * x)
    elif choice == 3:
        mid = rng.uniform(0.4, 0.6)
        sharp = rng.uniform(5, 15)
        a = 1.0 / (1.0 + np.exp(-sharp * (x - mid)))
    elif choice == 4:
        a = (1.0 - x + 0.01) ** rng.uniform(1.5, 3.0)
    elif choice == 5:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.7, 0.9)
        a = np.exp(-4.0 * (x - c1) ** 2 / 0.08) + 0.4 * np.exp(-4.0 * (x - c2) ** 2 / 0.04)
    elif choice == 6:
        a = np.cos(np.pi * x / 2) ** rng.uniform(1.0, 2.0)
    else:
        # Smooth random
        a = np.abs(rng.standard_normal(n)).cumsum()
        a = a[::-1]  # reverse to make it decreasing
        a = np.maximum(a, 0.001)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_gamma_init(n, rng):
    """Random gamma initialization (mixture of shapes)."""
    shape = rng.uniform(1.0, 4.0)
    a = rng.standard_gamma(shape=shape, size=n)
    # Add a small linear trend to break symmetry
    trend = 1.0 + rng.uniform(-0.5, 0.5) * np.linspace(0, 1, n)
    a = a * np.maximum(trend, 0.1)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
