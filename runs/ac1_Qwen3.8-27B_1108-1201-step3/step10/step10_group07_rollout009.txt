import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)

        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:

    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

    best_a = a.copy()

    it = 0
    stagnation = 0
    damping = 1.0

        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adaptive damping: decrease as we progress
        if it <= 10:
            damping = 1.0
        elif it <= 30:
            damping = 0.7
        elif it <= 60:
            damping = 0.4
        elif it <= 100:
            damping = 0.2
        else:
            damping = 0.1

        a_new = (1 - damping) * a + damping * b_new
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        score_a = 2.0 * n * fast_convolve(a_new).max()
        if score_a < best_score:

def fast_convolve(a):

    A = rfft(a, fft_size)

def convolve_ab(a, b):
    n_a = len(a)

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        # Sort coordinates by peak contribution (highest first)
        order = np.argsort(peak_scores)[::-1]

        # Try reducing high-contribution coordinates and adding to low ones
        for idx in range(min(20, n)):
            i = int(order[idx])
            if a[i] < 1e-12:
                continue

            # Find donor coordinates (low peak score)
            donor_candidates = []
            for d in range(n):
                if d == i or a[d] > a[i]:
                    continue
                if peak_scores[d] < peak_scores[i] * 0.3:
                    donor_candidates.append(d)

            if not donor_candidates:
                continue

            # Try transferring mass from i to a donor
            for frac in [0.05, 0.1, 0.15, 0.25]:
                delta = -a[i] * frac
                if delta > 0:
                    continue
                new_a = a.copy()
                new_a[i] = max(new_a[i] + delta, 0.0)

                # Update conv
                k_s = i
                k_e = min(i + n - 1, 2 * n - 2)
                new_conv = conv.copy()
                new_conv[k_s:k_e + 1] += 2.0 * (new_a[i] - a[i]) * a[k_s - i:k_e - i + 1]
                if 2 * i >= k_s and 2 * i <= k_e:
                    idx2 = 2 * i - k_s
                    d2 = new_a[i] - a[i]
                    new_conv[idx2] = conv[2 * i] + 2.0 * d2 * a[i] + d2 * d2

                new_M = float(new_conv.max())
                if new_M < M - 1e-12:
                    a = new_a
                    conv = new_conv
                    M = new_M
                    improved = True
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    break

        if not improved and rounds > 100:
            break

        if verbose and rounds % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDef round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.999, 0.998, 0.996, 0.997, 1.002, 1.003, 1.004, 1.001]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_aff = conv_aff + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_descent(a0, time_limit, lr_init=0.01, top_k=30, verbose=False):
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-12:
                break

        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a = simplex_project(a - lr * sg)
        conv = fast_convolve(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def propose_candidate():
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    global_best_seq = None
    global_best_score = float('inf')

    def try_update(candidate, label):
        nonlocal global_best_seq, global_best_score
        if candidate is None:
            return
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    if parent is not None:
        n_parent = len(parent)
        try_update(parent, "parent")
    else:
        n_parent = 1024
        try_update(np.ones(n_parent) / n_parent, "uniform")

    total_n = n_parent

    # === Phase 1: Shape exploration ===
    print(f"\n=== Phase 1: Shape Exploration (best={global_best_score:.8f}) ===")
    t1_end = time.monotonic() + 120.0
    best_shape = None
    best_shape_score = global_best_score

    for wt in range(8):
        if time.monotonic() > t1_end:
            break
        for n_try in [512, total_n, 2048]:
            a = smooth_window(n_try, wt, rng)
            sc = score_of(a)
            if sc < best_shape_score:
                best_shape_score = sc
                best_shape = a.copy()
            try_update(a, f"shape_wt{wt}_n{n_try}")

    if best_shape is None:
        best_shape = smooth_window(total_n, 0)

    print(f"  Best shape: score={best_shape_score:.8f} (n={len(best_shape)})")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating (best={global_best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    # LP from parent
    if parent is not None:
        t_alloc = min(t2_end - time.monotonic(), 300.0)
        if t_alloc > 30:
            print(f"  LP from parent, budget={t_alloc:.0f}s")
            a_lp, s_lp = lp_alternating(parent, t_alloc, solver, verbose=True)
            try_update(a_lp, "lp_parent")

    # LP from best shape
    t_alloc = min(t2_end - time.monotonic(), 250.0)
    if t_alloc > 30:
        print(f"  LP from shape, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating(best_shape, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_shape")

    # Bisection LP
    t_alloc = min(t2_end - time.monotonic(), 150.0)
    if t_alloc > 30:
        print(f"  Bisection LP, budget={t_alloc:.0f}s")
        a_bis, s_bis = bisection_lp(global_best_seq, t_alloc, solver, verbose=True)
        try_update(a_bis, "bisLP")

    # LP from diverse inits
    diverse_inits = [
        smooth_window(total_n, 0),
        smooth_window(total_n, 1),
        smooth_window(total_n, 3),
    ]
    for idx, di in enumerate(diverse_inits):
        if time.monotonic() > t2_end:
            break
        t_alloc = min(t2_end - time.monotonic(), 100.0)
        if t_alloc < 20:
            break
        a_di, s_di = lp_alternating(di, t_alloc, solver, verbose=False)
        try_update(a_di, f"lp_div{idx}")

    # === Phase 3: Subgradient + Coordinate Descent ===
    print(f"\n=== Phase 3: SGD + CD (best={global_best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.25

    t_alloc = min(t3_end - time.monotonic(), 120.0)
    if t_alloc > 20:
        print(f"  SGD, budget={t_alloc:.0f}s")
        a_sgd, s_sgd = subgradient_descent(global_best_seq, t_alloc, lr_init=0.008, top_k=40, verbose=True)
        try_update(a_sgd, "sgd")

    t_alloc = min(t3_end - time.monotonic(), 200.0)
    if t_alloc > 20:
        print(f"  CD, budget={t_alloc:.0f}s")
        a_cd, s_cd = coord_descent_line_search(global_best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

    # === Phase 4: Peak Deflation ===
    print(f"\n=== Phase 4: Peak Deflation (best={global_best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.2

    t_alloc = min(t4_end - time.monotonic(), 150.0)
    if t_alloc > 20:
        print(f"  PeakDef, budget={t_alloc:.0f}s")
        a_pd, s_pd = peak_deflation(global_best_seq, t_alloc, verbose=True)
        try_update(a_pd, "peak_def")

    t_alloc = min(t4_end - time.monotonic(), 100.0)
    if t_alloc > 20:
        a_cd2, s_cd2 = coord_descent_line_search(global_best_seq, t_alloc, verbose=False, label="pd")
        try_update(a_cd2, "cd_pd")

    # === Phase 5: Fine Polish ===
    print(f"\n=== Phase 5: Fine Polish (best={global_best_score:.8f}) ===")
    t5_end = deadline - 30

    t_alloc = min(t5_end - time.monotonic(), 120.0)
    if t_alloc > 15:
        print(f"  FinePolish, budget={t_alloc:.0f}s")
        a_fp, s_fp = fine_polish(global_best_seq, t_alloc, verbose=True)
        try_update(a_fp, "fine")

    # Final CD
    t_alloc = min(t5_end - time.monotonic(), 60.0)
    if t_alloc > 10:
        a_fin, s_fin = coord_descent_line_search(global_best_seq, t_alloc, verbose=False, label="final")
        try_update(a_fin, "final_cd")

    # Final fine polish
    t_alloc = min(t5_end - time.monotonic(), 40.0)
    if t_alloc > 10:
        a_fp2, s_fp2 = fine_polish(global_best_seq, t_alloc, verbose=False)
        try_update(a_fp2, "fine2")

    # Final verification
    if global_best_seq is not None:
        fs = float(evaluate_sequence(global_best_seq.tolist()))
        if np.isfinite(fs):
            global_best_score = fs

    print(f"\nFINAL SCORE: {global_best_score:.8f} (n={len(global_best_seq)})")
    return [float(x) for x in global_best_seq]
