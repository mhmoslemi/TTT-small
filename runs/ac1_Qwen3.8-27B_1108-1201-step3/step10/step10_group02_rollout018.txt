def interp_to_n(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

def build_conv_rows(a, peak_indices=None):
    """Build constraint rows for conv(a, b) <= M at given peak positions."""
    n = len(a)
    if peak_indices is None:
        peak_indices = range(2 * n - 1)
    peak_indices = list(peak_indices)
    rows = []

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    return np.array(rows) if rows else np.zeros((0, n))


def solve_lp_response(a_fixed, solver=None, top_k=None):

    M_var = cp.Variable()
    
    if top_k is not None and top_k < 2 * n - 1:
        conv = fast_autoconv(a_fixed)
        peak_idx = np.argsort(conv)[::-1][:top_k]
        C = build_conv_rows(a_fixed, peak_idx)
    else:
        C = build_conv_rows(a_fixed)
    
    constraints = [C @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for sv in [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:

        b = fast_autoconv(y)
        max_b = b.max()

        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

            sv = a - prev_a
            yv = g - prev_g

            sdy = np.dot(sv, yv)
            ydy = np.dot(yv, yv)
            if sds > 1e-15 and ydy > 1e-15 and abs(sdy) > 1e-15:
                eta_new = sds / abs(sdy)
                eta = 0.4 * eta + 0.6 * eta_new
            elif sds > 1e-15 and ydy > 1e-15:
                eta = sds / ydy
            eta = max(1e-5, min(eta, 1.0))
        prev_a = a.copy()
        prev_g = g.copy()

        log_a = np.log(np.maximum(a, 1e-300))

            sn = 1.0

            no_improve = 0

def logsumexp_smooth(z, T):
    """Compute smoothed objective and gradient in log-parameterization."""
    a = np.exp(z - z.max())
    s = a.sum()
    if s < 1e-300:
        return 1e15, np.zeros_like(z)
    a = a / s
    n = len(a)
    conv = fast_autoconv(a)
    max_c = conv.max()
    exp_c = np.exp(conv - max_c)
    w = exp_c / exp_c.sum()
    logsumexp_val = max_c + np.log(exp_c.sum())
    obj = (2.0 * n / T) * logsumexp_val

    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages=None):
    """Multi-stage L-BFGS on the log-sum-exp smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    for T in T_stages:
        el = time.monotonic() - t0
        if el > time_limit:
            break
        remaining = time_limit - el
        if remaining < 5:
            break
        maxit = max(50, int(remaining * 15))

        def func(zz):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_smooth(zz, T)
            return obj

        def jac(zz):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zz)
            _, grad = logsumexp_smooth(zz, T)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxit, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxit * 200, 'disp': False})
            z = res.x
        except Exception:
            pass

    a = np.exp(z - z.max())
    a = a / a.sum()
    return a

def perturbation_restart(a0, rng, scale=0.005):
    """Apply a structured perturbation and renormalize."""
    n = len(a0)
    # Mix of random and structured perturbations

    if choice == 0:
        noise = rng.standard_normal(n) * scale
    elif choice == 1:
        # Low-frequency perturbation

        noise = np.zeros(n)
        x = np.linspace(0, 1, n)
        for i, fi in enumerate(f):
            noise += fi * np.cos(2 * np.pi * (i + 1) * x)
        noise *= scale / max(1.0, np.max(np.abs(noise)))
    else:
        # Block perturbation

        noise = np.zeros(n)
        for b in range(n // block_size):
            noise[b * block_size:(b + 1) * block_size] = rng.standard_normal() * scale
    a_new = a0 + noise
    return normalize_nonneg(a_new)

        print(f"Default init: n=1200, score={compute_score(parent):.8f}")

    best_seq = parent.copy()
    best_score = compute_score(parent)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = normalize_nonneg(a)
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Multi-n exploration (400s) ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.25, 400.0)
    if t_phase1 > 30:
        print(f"  Budget: {t_phase1:.0f}s")
        parent_n = len(parent)
        n_candidates = sorted(set([
            parent_n,
            max(500, int(parent_n * 0.7)),
            max(500, int(parent_n * 0.85)),
            max(500, int(parent_n * 1.15)),
            max(500, int(parent_n * 1.3)),
            max(500, int(parent_n * 1.5)),
            max(500, int(parent_n * 2.0)),
        ]))
        n_candidates = [n for n in n_candidates if 400 <= n <= 5000]
        t_per = t_phase1 / len(n_candidates)
        print(f"  n values: {n_candidates}, time each: {t_per:.0f}s")

        for n_try in n_candidates:
            el = time.monotonic()
            if (deadline - el) < 15:
                break
            a_init = interp_to_n(parent, n_try)
            # Add small perturbation to avoid exact duplicate
            a_init = a_init + rng.standard_normal(n_try) * 0.001
            a_init = normalize_nonneg(a_init)
            t_alloc = min(t_per, deadline - el - 5)
            if t_alloc < 10:
                break
            a_md, sc = mirror_descent(a_init, t_alloc, eta0=0.07, active_frac=0.999, bb=True)
            try_update(a_md, f"md_n{n_try}")

    # === Phase 2: Main LP optimization (800s) ===
    print(f"\n=== Phase 2: LP Nesterov (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.45, 800.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s, n={len(best_seq)}")
        n_cur = len(best_seq)
        # Use reduced LP for speed
        top_k = max(20, n_cur // 40)
        a_lp, sc = lp_nesterov(best_seq, t_phase2, solver=solver, top_k=top_k)
        try_update(a_lp, "lp_nesterov")

    # === Phase 3: Mirror descent refinement (300s) ===
    print(f"\n=== Phase 3: MD Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.20, 300.0)
    if t_phase3 > 20:
        print(f"  Budget: {t_phase3:.0f}s")
        a_md2, sc = mirror_descent(best_seq, t_phase3, eta0=0.02, active_frac=0.9999, bb=True)
        try_update(a_md2, "md_refine")

    # === Phase 4: L-BFGS on smoothed objective (250s) ===
    print(f"\n=== Phase 4: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.18, 250.0)
    if t_phase4 > 15:
        print(f"  Budget: {t_phase4:.0f}s")
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a_lb = lbgfs_refine(best_seq, t_phase4, T_stages=T_stages)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Coordinate descent polish (200s) ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.15, 200.0)
    if t_phase5 > 10:
        print(f"  Budget: {t_phase5:.0f}s")
        a_cd, sc = peak_coordinate_descent(best_seq, t_phase5, n_steps=60000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Perturbation + re-optimize cycles (150s) ===
    print(f"\n=== Phase 6: Perturbation Cycles (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase6 = min(rem * 0.10, 150.0)
    if t_phase6 > 20:
        print(f"  Budget: {t_phase6:.0f}s")
        n_cycles = max(3, int(t_phase6 / 25))
        t_per_cycle = t_phase6 / n_cycles
        for cyc in range(n_cycles):
            el = time.monotonic()
            if (deadline - el) < 15:
                break
            a_pert = perturbation_restart(best_seq, rng, scale=0.003 + 0.002 * cyc)
            t_alloc = min(t_per_cycle, deadline - el - 5)
            if t_alloc < 8:
                break
            # Quick MD + CD
            a_q, _ = mirror_descent(a_pert, t_alloc * 0.6, eta0=0.03, active_frac=0.999)
            try_update(a_q, f"pert_md_{cyc}")
            if (deadline - el) > 10:
                a_q2, _ = peak_coordinate_descent(a_q, min(t_alloc * 0.4, deadline - el - 3), n_steps=20000)
                try_update(a_q2, f"pert_cd_{cyc}")

    # === Phase 7: Final fine L-BFGS (100s) ===
    print(f"\n=== Phase 7: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase7 = min(rem * 0.07, 100.0)
    if t_phase7 > 10:
        T_final = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a_fin = lbgfs_refine(best_seq, t_phase7, T_stages=T_final)
        try_update(a_fin, "final_lbgfs")

    # Final coordinate descent on best
    rem = deadline - time.monotonic()
    if rem > 15:
        a_polish, _ = peak_coordinate_descent(best_seq, rem - 5, n_steps=30000)
        try_update(a_polish, "final_polish")

    # Final verification
    final_score = compute_score(best_seq)
    print(f"\nFINAL: score={final_score:.8f} (n={len(best_seq)})")

    # Clamp for safety
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
