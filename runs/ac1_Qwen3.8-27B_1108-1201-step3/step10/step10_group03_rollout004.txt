3. Iterative peak equalization
4. Multi-n exploration and multi-start
5. LP best-response for final polish
Key novelty: FFT-accelerated gradients for the smoothed objective,
enabling thousands of L-BFGS iterations in the time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_logz(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def build_lse_funcs(n, T):
    """Build objective and gradient functions for LSE-smoothed problem.
    Parameters: z (log-softmax parameterization)
    Returns: objective value and gradient w.r.t. z
    """
    fft_size_corr = 1 << (3 * n - 2 - 1).bit_length()
    
    def obj_and_grad(z):
        a = softmax_from_logz(z)
        b = fast_convolve(a)
        b_shift = b / T
        b_max = b_shift.max()
        exp_ws = np.exp(b_shift - b_max)
        w = exp_ws / exp_ws.sum()
        lse = b_max + np.log(exp_ws.sum())
        obj = 2.0 * n * T * lse
        
        # Gradient w.r.t. a: 4n/T * (w *_ a)
        # (w *_ a)[j] = sum_k w[k] * a[k-j]
        # = ifft(FFT(w) * conj(FFT(a))) with proper padding
        W = rfft(w, fft_size_corr)
        A = rfft(a, fft_size_corr)
        cross = irfft(W * np.conj(A), fft_size_corr)
        grad_a = (4.0 * n / T) * cross[:n]
        
        # Chain rule for log-softmax: grad_z = a * (grad_a - sum(grad_a * a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        return obj, grad_z
    
    return obj_and_grad


def lbgfs_lse(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with FFT gradients."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-100)
    z = np.log(a_clipped)
    # Don't subtract mean - L-BFGS handles offset
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for T in T_stages:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break
        max_iter = max(50, int(remaining * 15))
        m = min(20, n // 10)  # L-BFGS memory
        
        obj_and_grad = build_lse_funcs(n, T)
        
        def obj(z):
            return obj_and_grad(z)[0]
        
        def grad(z):
            return obj_and_grad(z)[1]
        
        def obj_and_jac(z):
            return obj_and_grad(z)
        
        try:
            res = minimize(
                obj_and_jac, z, jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': max_iter * n * 2, 'm': m, 'disp': False}
            )
            z = res.x
            a_new = softmax_from_logz(z)
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                if (time.monotonic() - t0) % 30 < 0.5:
                    print(f"    LSE L-BFGS T={T}: score={sc:.10f} ({time.monotonic()-t0:.0f}s)")
        except Exception as e:
            pass
    
    return best_a, best_score


def adam_subgradient(a0, time_limit, n_restarts=3):
    """Adam-style subgradient method with simplex projection."""
    n = len(a0)
    best_a = normalize(a0).copy()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    rng = np.random.default_rng(12345)
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        if restart == 0:
            a = best_a.copy()
        else:
            a = best_a + rng.standard_normal(n) * 0.01
            a = normalize(a)
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        lr = 0.01
        t_step = 0
        
        for it in range(50000):
            if time.monotonic() - t0 > time_limit:
                break
            
            t_step += 1
            b = fast_convolve(a)
            M = b.max()
            
            # Find active set (positions near max)
            threshold = M * (1 - 0.01)
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(b))])
            
            # Subgradient: average of 2*a[k-j] over active k
            # (w *_ a)[j] where w = uniform on active set
            w = np.zeros(2 * n - 1)
            w[active] = 1.0 / len(active)
            
            # Compute (w *_ a) via FFT
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            # Actually we need size >= (2n-1) + n - 1
            fft_size2 = 1 << (3 * n - 2 - 1).bit_length()
            W = rfft(w, fft_size2)
            A = rfft(a, fft_size2)
            cross = irfft(W * np.conj(A), fft_size2)
            g = 2.0 * cross[:n]
            
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g * g
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            
            a_trial = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # Project onto simplex: a >= 0, sum(a) = 1
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial / s
            
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                # Decay lr slowly
                lr = max(0.001, lr * 0.9999)
            
            if (it + 1) % 5000 == 0:
                lr *= 0.5
    
    return best_a, best_score


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) < cssv) + 1
    theta = (cssv[rho - 1] - 1) / rho
    w = np.maximum(v - theta, 0)
    return w


def peak_equalization(a0, time_limit):
    """Iteratively equalize peaks of the convolution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    fft_size2 = 1 << (3 * n - 2 - 1).bit_length()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Target: reduce all peaks above M*(1-0.005)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Build weight vector
        w = np.zeros(2 * n - 1)
        # Weight proportional to excess
        w[active] = b[active] - threshold + 1e-10
        w = w / w.sum()
        
        # Compute direction: (w *_ a)
        W = rfft(w, fft_size2)
        A = rfft(a, fft_size2)
        cross = irfft(W * np.conj(A), fft_size2)
        direction = cross[:n]
        
        # Step: reduce a in the direction of the subgradient
        # Adaptive step size
        lr = 0.1 / (1 + it * 0.001)
        for step_scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_trial = a - lr * step_scale * direction
            a_trial = project_simplex(a_trial)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        # Triangular (optimal for L1 convolution)
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Near-uniform with slight modulation
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 7:
        # Flat top
        a = np.exp(-10 * (x - 0.5) ** 4)
    elif choice == 8:
        # Waterfall
        a = 1.0 / (0.5 + x)
    else:
        # Bimodal
        a = np.exp(-5 * (x - 0.25) ** 2) + np.exp(-5 * (x - 0.75) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
