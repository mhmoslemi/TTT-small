def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated Adam gradient descent
with aggressive LogSumExp temperature cooling, peak-directed coordinate descent,
Nesterov LP alternation, multi-length exploration, and simulated annealing restarts.
Key innovations: O(n log n) gradient via FFT cross-correlation, adaptive Adam optimizer,
peak-aware weighting, and multi-scale exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp(a_fixed, solver=None):
    """Solve LP: min M s.t. (a_fixed * b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver is None:
        solver = get_lp_solver()
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            return b_val / s, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, time_limit, solver=None, max_iters=50000):
    """Nesterov LP alternation for global convergence."""
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp(y, solver)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize(a)

        sc_a = eval_score(a)
        sc_y = eval_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 20:
                noise = rng.standard_normal(n) * 0.015
                y = normalize(y + noise)
                stall = 0

    return best_a, best_score


def compute_grad_lse(a, T):
    """Compute gradient of (2n/T)*LSE_T(b) where b=conv(a,a). Returns (grad, b, max_b)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    w = exp_shifted / Z
    # c_i = sum_k w[k] * a[k-i] for valid k-i
    # This is correlate(a, w, 'full')[:n]
    c = np.correlate(a, w, mode='full')[:n]
    grad = n * c
    # Project onto simplex tangent: subtract dot product with a
    proj = np.dot(grad, a)
    grad = grad - proj * a
    return grad, b, max_b


def lse_value(a, n, T):
    """Compute (2n/T) * LSE_T(b)."""
    b = fast_convolve(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    return (2.0 * n / T) * (max_b + np.log(np.exp(shifted).sum()) / T)


def adam_lse_optimize(a0, time_limit, T_stages=None, beta1=0.9, beta2=0.999,
                      lr_init=0.01, stall_patience=800, print_every=2000):
    """Adam optimizer on LogSumExp smoothed objective with temperature cooling."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.01, 0.005, 0.003]

    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    rng = np.random.default_rng(31337)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        stall = 0
        prev_best_local = best_score
        lr = lr_init

        while time.monotonic() - t0 < time_limit:
            step_count += 1

            grad, b, max_b = compute_grad_lse(a, T)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** step_count)
            v_hat = v / (1.0 - beta2 ** step_count)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = normalize(a_new)

            score = eval_score(a_new)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                # Occasionally accept non-improving moves (simulated annealing)
                if stall > 100 and rng.random() < 0.1:
                    a = a_new
                    stall = 50
                else:
                    a = a  # keep old

            # Adaptive learning rate
            if stall > 200:
                lr *= 0.8
                stall = 0
            if step_count % 500 == 0:
                lr = min(lr_init, lr * 1.01)

            if stall >= stall_patience:
                break

            if print_every and step_count % print_every == 0:
                elapsed = time.monotonic() - t0
                print(f"      Adam T={T:.3f} step={step_count} ({elapsed:.0f}s): score={best_score:.8f}")

        # Reset momentum between stages
        m = np.zeros(n)
        v = np.zeros(n)
        step_count = 0

        if time.monotonic() - t0 > time_limit:
            break
        # Perturb when cooling
        noise = rng.standard_normal(n) * 0.003
        a = normalize(a + noise)
        sc = eval_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def peak_directed_optimize(a0, time_limit, top_k=10, max_rounds=200000):
    """Peak-directed coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:top_k]

        improved = False
        for pk in top_peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            order = np.argsort(contribs)[::-1]

            for oi in order[:5]:
                pos = idx[oi]
                if a[pos] < 1e-14:
                    continue
                # Try reducing this position
                for factor in [0.995, 0.99, 0.98, 0.96]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = normalize(a_trial)
                    sc = eval_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing a position that's underutilized
            # Find position with lowest conv contribution to top peak
            pk = top_peaks[0]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            # Try increasing a random low position
            low_positions = np.where(a < np.median(a) * 0.5)[0]
            if len(low_positions) > 0:
                pos = rng.choice(low_positions)
                a_trial = a.copy()
                a_trial[pos] *= 1.05
                a_trial = normalize(a_trial)
                sc = eval_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            if stall > 500:
                # Perturb and reset
                noise = rng.standard_normal(n) * 0.005
                a = normalize(a + noise)
                stall = 0
                if eval_score(a) < best_score:
                    best_a = a.copy()
                    best_score = eval_score(a)

            if stall > 3000:
                break

        if rnd % 50000 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakDir {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def multi_start_gradient(a0, time_limit, n_starts=5, T=0.05):
    """Multiple gradient restarts with perturbation."""
    n = len(a0)
    best_a = normalize(a0.copy())
    best_score = eval_score(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    time_per_start = time_limit / n_starts

    for si in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        t_rem = time_limit - (time.monotonic() - t0)
        t_this = min(t_rem, time_per_start)
        if t_this < 10:
            break

        if si == 0:
            a_start = best_a.copy()
        elif si == 1:
            x = np.linspace(0, 2 * np.pi, n)
            pert = 0.02 * np.cos(3 * x) + 0.01 * np.sin(7 * x)
            a_start = normalize(best_a + pert)
        elif si == 2:
            a_start = normalize(best_a + rng.standard_normal(n) * 0.01)
        elif si == 3:
            # Shift the peaks
            a_shift = np.roll(best_a, n // 10)
            a_start = normalize(a_shift)
        else:
            # Inverse-like perturbation
            a_inv = 1.0 / (best_a + 0.01)
            a_start = normalize(a_inv + 0.5 * best_a)

        m = np.zeros(n)
        v = np.zeros(n)
        step_count = 0
        lr = 0.005
        stall = 0

        while time.monotonic() - t0 < time_limit - 5:
            step_count += 1
            grad, b, max_b = compute_grad_lse(a_start, T)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1.0 - 0.9 ** step_count)
            v_hat = v / (1.0 - 0.999 ** step_count)
            a_new = a_start - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = normalize(a_new)
            sc = eval_score(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                a_start = a_new
                stall = 0
            else:
                stall += 1
                if rng.random() < 0.05:
                    a_start = a_new
                    stall = 50
            if stall > 300:
                lr *= 0.8
                stall = 0
            if stall > 1000:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.2
    elif choice == 7:
        a = (1 - x) ** 0.5 + 0.1
    elif choice == 8:
        # Multi-bump
        a = np.zeros(n)
        for c in [0.15, 0.5, 0.85]:
            a += np.exp(-20 * (x - c) ** 2)
    else:
        # Smooth noise
        a = np.ones(n) + 0.2 * rng.standard_normal(n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov LP, FFT-accelerated Adam gradient
    descent with LogSumExp smoothing, peak-directed coordinate descent,
    multi-start exploration, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP ===
    print(f"\n=== Phase 1: Nesterov LP, n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 350.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t1, solver=solver, max_iters=100000)
        try_update(a_lp, "nlp_p1")

    # === Phase 2: Adam LogSumExp (main workhorse) ===
    print(f"\n=== Phase 2: Adam LSE gradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 30:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
        print(f"  T_stages: {len(T_stages)} stages, budget={t2:.0f}s")
        a_adam, _ = adam_lse_optimize(best_seq, t2, T_stages=T_stages,
                                       lr_init=0.008, stall_patience=1500)
        try_update(a_adam, "adam_p2")

    # === Phase 3: Peak-directed CD ===
    print(f"\n=== Phase 3: Peak-directed CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 15:
        a_peak, _ = peak_directed_optimize(best_seq, t3, top_k=12, max_rounds=300000)
        try_update(a_peak, "peak_p3")

    # === Phase 4: Second Adam pass with lower T ===
    print(f"\n=== Phase 4: Adam LSE low-T, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.18, 300.0)
    if t4 > 20:
        T_low = [0.03, 0.02, 0.01, 0.006, 0.004, 0.002, 0.001]
        print(f"  Low-T stages: {len(T_low)}, budget={t4:.0f}s")
        a_adam2, _ = adam_lse_optimize(best_seq, t4, T_stages=T_low,
                                        lr_init=0.003, stall_patience=2000)
        try_update(a_adam2, "adam_p4")

    # === Phase 5: Multi-start gradient ===
    print(f"\n=== Phase 5: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 200.0)
    if t5 > 20:
        a_ms, _ = multi_start_gradient(best_seq, t5, n_starts=5, T=0.03)
        try_update(a_ms, "multistart")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        alt_lengths = []
        for candidate_n in [800, 1000, 1500, 1800, 2000]:
            if candidate_n != n_cur:
                alt_lengths.append(candidate_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alt = min(rem * 0.5, 120.0)
            if t_alt < 20:
                break
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_seq(best_seq, alt_n)
            T_mid = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
            a_alt, _ = adam_lse_optimize(init_alt, t_alt, T_stages=T_mid,
                                          lr_init=0.005, stall_patience=800)
            try_update(a_alt, f"alt_n{alt_n}")

    # === Phase 7: Final peak-directed polish ===
    print(f"\n=== Phase 7: Final peak polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.4, 60.0)
        if t7 > 10:
            a_fin, _ = peak_directed_optimize(best_seq, t7, top_k=15, max_rounds=200000)
            try_update(a_fin, "peak_fin")

    # === Phase 8: Very low T Adam ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.5, 50.0)
        if t8 > 10:
            T_ult = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
            a_ult, _ = adam_lse_optimize(best_seq, t8, T_stages=T_ult,
                                          lr_init=0.002, stall_patience=3000)
            try_update(a_ult, "ult_lowT")

    # === Phase 9: Final LP if time ===
    rem = deadline - time.monotonic()
    if rem > 40:
        print(f"\n=== Phase 9: Final LP, score={best_score:.8f} ===")
        t9 = min(rem * 0.7, 80.0)
        a_final_lp, _ = nesterov_lp(best_seq, t9, solver=solver, max_iters=50000)
        try_update(a_final_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
