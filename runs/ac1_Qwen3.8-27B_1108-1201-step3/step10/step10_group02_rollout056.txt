from scipy.fft import rfft, irfft

# ... helper functions ...

def solve_lp_active(a, active_k, solver):
    """Solve LP with only active constraints."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a, active_k)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status in ('optimal', 'optimal_inaccurate'):
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(M_var.value)
    return None, None

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def solve_lp_fast(a, active_k, solver):
    """Solve LP: min M s.t. (a*b)_k <= M for k in active_k, sum(b)=1, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix only for active_k
    T = np.zeros((len(active_k), n))
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a[k - idx]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    return None

    if s < 1e-15:

    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    prev_best = best_score

        b_curr = fast_convolve(y)
        max_b = b_curr.max()
        
        # Adaptive active set
        threshold = max_b * active_frac
        active_k = np.where(b_curr >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(b_curr))])
        # Cap the active set size for LP speed
        if len(active_k) > 50:

            active_k = top_idx

        if b_new is None:
            # Fallback: use subgradient

            for k in active_k:

                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += y[k - idx]
            g /= max(1, len(active_k))
            # Gradient step

            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            b_new = a_trial
        
        # Nesterov update
        if nesterov:
            alpha = 2.0 / (it + 2.0)
            y = (1.0 - alpha) * y + alpha * b_new
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = b_new.copy()
            else:
                y = y / sy
        
        # Weighted average for a
        a = ((it + 1) * a + (it + 2) * b_new) / (2 * (it + 1))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = b_new.copy()
        else:
            a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            
            if stall >= 15:
                # Diversity kick
                kick_scale = 0.01 / np.sqrt(stall + 1)
                kick = rng.standard_normal(n) * kick_scale
                y = y + kick
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
                active_frac = max(0.99, active_frac - 0.001)

def logsumexp_refine(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
    
    n_stages = len(T_stages)
    iters_per_stage = max(30, 3000 // n_stages)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        def func_z(zv):
            av = np.exp(zv - zv.max())
            av = av / av.sum()
            bv = fast_convolve(av)
            max_bv = bv.max()
            exp_bv = np.exp(T * (bv - max_bv))
            logsumexp = max_bv + np.log(exp_bv.sum()) / T
            return 2.0 * n * logsumexp
        
        def jac_z(zv):
            av = np.exp(zv - zv.max())
            sv = av.sum()
            if sv < 1e-300:
                return np.zeros_like(zv)
            av = av / sv
            bv = fast_convolve(av)
            max_bv = bv.max()
            exp_bv = np.exp(T * (bv - max_bv))
            w = exp_bv / exp_bv.sum()
            
            # Gradient of logsumexp w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(bv)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * av[k - idx]
            
            # Convert to z-gradient (softmax)
            S = np.dot(grad_a, av)
            grad_z = av * (grad_a - S)
            return 2.0 * n * grad_z
        
        try:
            res = minimize(fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    a_out = np.exp(z_shifted)
    a_out = a_out / a_out.sum()
    return a_out

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    return None


def lp_alternate_nesterov(a0, time_limit, solver, active_frac=0.999,
                          cap_active=80, verbose=True):
    """LP alternating with Nesterov acceleration and adaptive active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    stall = 0
    prev_best = best_score

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_curr = fast_convolve(y)
        max_b = b_curr.max()

        threshold = max_b * active_frac
        active_k = np.where(b_curr >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(b_curr))])
        if len(active_k) > cap_active:
            top_idx = np.argsort(b_curr)[::-1][:cap_active]
            active_k = top_idx

        b_new = solve_lp_fast(y, active_k, solver)
        if b_new is None:
            g = np.zeros(n)
            for k in active_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += y[k - idx]
            g /= max(1, len(active_k))
            a_trial = a - 0.01 * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            b_new = a_trial / ss

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_new.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + (it + 2) * b_new) / (2.0 * (it + 1))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = b_new.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 20:
                kick_scale = 0.008 / np.sqrt(min(stall, 50) + 1.0)
                kick = rng.standard_normal(n) * kick_scale
                y = y + kick
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0

        if verbose and (it + 1) % 500 == 0:
            print(f"    LP-Nest it={it+1} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.8f} |act|={len(active_k)}")

    return best_a, best_score


def compute_lse_grad_a(a, T):
    """Compute gradient of (1/T)*log(sum_k exp(T*b_k)) w.r.t. a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    P = A * A
    b_vals = irfft(P, fft_size)[:size]
    scaled = T * (b_vals - b_vals.max())
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    # FFT of weights (odd-length, circular), reshaped to 2n
    W_full = ifft(fft(w, 2 * fft_size)).real[:n]
    # grad_a_i = (2/T) * sum_k w_k * a_{k-i}
    return 2.0 * np.convolve(W_full, a, mode='full')[:n] / T


def logsumexp_refine(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp objective with new efficient gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]

    n_stages = len(T_stages)
    iters_per_stage = max(40, min(500, 5000 // n_stages))

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break

        def func_lse(zv):
            av = np.exp(zv - zv.max())
            sv = av.sum()
            if sv < 1e-300:
                return 1e15
            av = av / sv
            bv = fast_convolve(av)
            max_bv = bv.max()
            lse = max_bv + np.log(np.exp(T * (bv - max_bv)).sum()) / T
            return 2.0 * n * lse

        def jac_lse(zv):
            av = np.exp(zv - zv.max())
            sv = av.sum()
            if sv < 1e-300:
                return np.zeros_like(zv)
            av = av / sv
            grad_a = compute_lse_grad_a(av, T)
            S = np.dot(grad_a, av)
            return 2.0 * n * av * (grad_a - S)

        try:
            res = minimize(fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    z_sh = z - z.max()
    a_out = np.exp(z_sh)
    a_out = a_out / a_out.sum()
    return a_out


def coord_desc_refine(a0, time_limit, n_steps=100000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for peak_k in peaks[:4]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9999, 0.9997, 0.9994, 0.999, 0.998, 0.996]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 800:
                break
        else:
            stall = 0

        if (step + 1) % 30000 == 0:
            print(f"    CoordDesc {time.monotonic()-t0:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def direct_subgradient(a0, time_limit, eta0=0.05, active_frac=0.999):
    """Direct subgradient method with adaptive step size for the max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    prev_a = a.copy()
    prev_g = None

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: g_i = (2/|S|) * sum_{k in S} a_{k-i}
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= max(1, len(active))

        # BB step size
        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-18 and y_dot_y > 1e-18:
                if abs(s_dot_y) > 1e-18:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 0.5))

        prev_a = a.copy()
        prev_g = g.copy()

        # Mirror descent update (exponential weighting)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-300:
            continue
        a = a_new / s_new

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if (it + 1) % 10000 == 0:
            print(f"    SubGrad it={it+1} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def make_diverse_init(n, rng, idx):
    """Create diverse initializations (emphasizing asymmetry)."""
    x = np.linspace(0, 1, n)
    m = idx % 10
    if m == 0:
        a = np.exp(-3.0 * x * (1 - x))
    elif m == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif m == 2:
        a = np.sin(np.pi * x) ** 1.5
    elif m == 3:
        a = np.exp(-2.5 * (x - 0.35)) ** 2
    elif m == 4:
        a = np.exp(-2.0 * (x - 0.6)) ** 2 * (1.0 + 0.4 * np.cos(4.0 * np.pi * x))
    elif m == 5:
        a = np.exp(-4.0 * x * (1 - x)) * (1.0 + 0.3 * np.sin(2.0 * np.pi * x))
    elif m == 6:
        a = np.maximum(np.cos(np.pi * (x - 0.4)), 0.0) ** 2
    elif m == 7:
        a = np.exp(-3.0 * (x - 0.5)) ** 2 + 0.3 * np.exp(-8.0 * (x - 0.2)) ** 2
    elif m == 8:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.4), 0.0) ** 1.5
    else:
        a = np.exp(-2.0 * (x - 0.5)) ** 2 * np.maximum(1.0 + 0.5 * np.cos(6.0 * np.pi * (x - 0.3)), 0.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
