Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP-based bidirectional

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (assumes sum(a)=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix_fast(a):
    """Build (2n-1) x n Toeplitz matrix T where T[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    size = 2 * n - 1
    # Use vectorized construction
    k_idx = np.arange(size)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    safe_diff = np.clip(diff, 0, n - 1)
    T = np.where(mask, np.take(a, safe_diff), 0.0)
    return T


def solve_response_lp_primal(a, verbose=False):
    """
    Given a (sum=1, non-neg), find b (sum=1, non-neg) minimizing max_k conv(a,b)_k.
    Uses scipy linprog with HiGHS solver.
    Returns (b, M) or (None, None) on failure.

    size = 2 * n - 1

    # Build T: T[k,j] = a[k-j] if 0<=k-j<n else 0
    T = build_conv_matrix_fast(a)

    # LP: min M s.t. T@b - M <= 0 (all k), sum(b) = 1, b >= 0
    # Variables: x = [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0

    # A_ub @ x <= b_ub: T @ b - 1 * M <= 0
    # A_ub is (2n-1) x (n+1)
    A_ub = np.empty((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(None, None)]

        # Update Polyak-Ruppert averages
        s_a = alpha_a * (it + 2)
        avg_a = (it * avg_a + s_a * a_new) / (it + s_a)
        avg_a = normalize(avg_a)

        s_b = alpha_b * (it + 2)
        avg_b = (it * avg_b + s_b * b_new) / (it + s_b)
        avg_b = normalize(avg_b)

        # Evaluate candidates
        for cand in [a_new, b_new, avg_a, avg_b, y_a, y_b]:
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        # Stall detection and noise injection
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                noise_scale = 0.02 / math.sqrt(stall_count)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall_count = 0

            if (it + 1) % 5000 == 0:
                print(f"      LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=50000):
    """Subgradient method to polish the solution."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Find active set (peaks within 0.1% of max)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient: grad of (a*a)[k] w.r.t. a is 2*a reversed/shifted
        g = np.zeros(n)
        for k in active:
            # (a*a)[k] = sum_j a[j]*a[k-j], grad w.r.t. a[j] = a[k-j] (+ a[j-k] if j>k)
            # More carefully: d/d(a[j]) sum_i a[i]*a[k-i] = a[k-j] + a[j-k] (if both in range)
            # But since we sum over all valid i, the gradient is:
            # For each term a[i]*a[k-i], d/d(a[j]) gives a[k-j] (from i=j term) + a[j-k] (from k-i=j term, i.e., i=k-j)
            # So grad[j] = a[k-j] + a[j-k] (each if in [0,n-1])
            # But if k-j == j (i.e., 2j=k), it's 2*a[j]
            js_valid = np.arange(max(0, k - n + 1), min(n, k + 1))
            for j in js_valid:
                kj = k - j
                if 0 <= kj < n:
                    g[j] += a[kj]
                    if kj != j and 0 <= (j - k) < n:
                        pass  # already counted
            # Simplified: grad_a[j] = a[k-j] if 0<=k-j<n, else 0
            # Actually for auto-convolution: (a*a)[k] = sum_j a[j]a[k-j]
            # d/d(a[j]) = a[k-j] (treating k-j as index into the other copy)
            # But by symmetry it's also a[j-k] if that's different...
            # Let's just use: grad[j] += a[(k-j) % n] ... no, let's be explicit
        # Actually, the correct gradient:
        # (a*a)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]
        # d/d(a[j]) for a specific j: a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, i.e., j>k)
        # But wait, in the sum, a[j] appears as the first factor when index=j, and as the
        # second factor when k-i=j, i.e., i=k-j. So:
        # d/d(a[j]) = a[k-j] + a[j-k] (first if 0<=k-j<n, second if 0<=j-k<n, i.e., j>k)
        # For j <= k: only first term: a[k-j] (if k-j < n, which is always true since k<=2n-2 and j>=0)
        # For j > k: only second term: a[j-k]
        # So: grad[j] = a[|k-j|] if |k-j| < n, which is always true for valid k,j
        # Wait, that's not right either. Let me just do it directly.

        g = np.zeros(n)
        for k in active:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    g[j] += a[kj]
                jk = j - k
                if 0 <= jk < n and jk != kj:
                    g[j] += a[jk]
        g /= len(active)

        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial /= st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.08)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-7, lr)

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """
    L-BFGS on LogSumExp-smoothed objective using log-parametrization.
    More efficient gradient computation using FFT.
    """
    n = len(a0)
    a = normalize(a0.copy())
    a_clipped = np.maximum(a, 1e-20)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    fft_size = 1 << (2 * n - 1).bit_length()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iter = max(20, int(remaining * 15))

        def obj_and_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()

            # Compute convolution
            A = rfft(a_cur, fft_size)
            b = irfft(A * A, fft_size)[:2 * n - 1]

            # LogSumExp
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            lse = b_max + np.log(exp_ws.sum())
            obj_val = 2.0 * n * lse

            # Gradient w.r.t. a
            w = exp_ws / exp_ws.sum()
            # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for the max part)
            # Use FFT: grad_a = 2 * (w convolved with a) ... actually
            # (a*a)[k] = sum_j a[j]*a[k-j], d/d(a[j]) for the LSE objective:
            # grad_a[j] = 2 * sum_k w[k] * (partial (a*a)[k] / partial a[j])
            # = 2 * sum_k w[k] * a[k-j] (for valid k-j)
            # This is 2 * (a * w)[j] where * is cross-correlation
            # = 2 * ifft(fft(a) * conj(fft(w))) ... let me just do it directly with FFT

            # Actually: sum_k w[k] * a[k-j] = (w * a)[j] in the cross-correlation sense
            # = convolution of w and rev(a) ... let me just use FFT
            W = rfft(w, fft_size)
            grad_a_fft = irfft(W * conj(A) * 2.0, fft_size)[:n]
            # Hmm, this gives sum_k w[k] * a[k-j] only if we're careful about the convolution definition
            # Let me just do it the naive way for correctness (it's O(n^2) but n isn't too large)
            # Actually for n up to 3000, O(n^2) is 9M ops, fast enough
            # But let's use FFT properly:
            # sum_k w[k] * a[k-j] for k in [0, 2n-2]:
            # This is the (2n-1)-length cross-correlation of w and a, evaluated at lag j
            # = ifft(fft(w) * conj(fft(a))) but with proper padding
            # Actually: corr[j] = sum_k w[k] * a[k-j]
            # This is the same as: let v[k] = w[k], u[k] = a[k]
            # corr = ifft(FFT(v) * conj(FFT(u))) ... no
            # Convolution: (v*u)[k] = sum_j v[j]*u[k-j]
            # Cross-correlation: (v⊛u)[k] = sum_j v[j]*u[j-k] = sum_j v[j+k]*u[j]
            # We want: sum_k w[k]*a[k-j] = sum_k w[k+j]*a[k] (substituting k->k+j)
            # = (w_rev ⊛ a)[j] ... this is getting confusing
            # Let me just compute it using the direct formula with numpy
            grad_a = np.zeros(n)
            # For each j, grad_a[j] = 2 * sum_k w[k] * a[k-j] (where 0<=k-j<n)
            # = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j] (clipping k to [0,2n-2])
            # This is a cross-correlation, computable via FFT:
            # Define w_padded of length fft_size, a_padded of length fft_size
            # Then ifft(fft(w_padded) * conj(fft(a_padded))) gives the cross-correlation
            W_pad = rfft(w, fft_size)
            A_pad = rfft(a_cur, fft_size)
            corr = irfft(W_pad * np.conj(A_pad), fft_size)
            # corr[j] = sum_k w[k] * a[k-j] (circular), but with proper padding it's linear
            # Actually for linear cross-correlation:
            # We need to be careful. Let's just use the direct approach for small n
            # or accept slight inaccuracy from circular effects

            # Simpler: use the fact that grad_a = 2 * (a_cross_correlate w)
            # where (a⊛w)[j] = sum_k a[k] w[k+j] ...
            # I'll just use the naive O(n*2n) computation which is fine for n<=5000
            # Actually n could be up to 5000, so n*2n = 50M, which is ~50ms. Too slow per iteration.
            # Let me use FFT properly.
            # 
            # The quantity we want is: for each j in [0,n-1]:
            #   S[j] = sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w[k] * a[k-j]
            # This is the linear cross-correlation of the sequences w (length 2n-1) and a (length n)
            # evaluated at lags j = 0, ..., n-1.
            #
            # Linear cross-correlation: (w ⊛ a)[j] = sum_k w[k] * a[k-j]
            # = ifft(FFT(w) * conj(FFT(a))) with zero-padding to length >= (2n-1) + n - 1 = 3n-2
            # But we already have fft_size >= 2n-1, so we might have circular artifacts.
            # For safety, let's use a larger FFT.
            #
            # Actually, since a has length n and w has length 2n-1, the linear cross-correlation
            # has length n + (2n-1) - 1 = 3n-2. We need fft_size >= 3n-2.
            # Our current fft_size is the next power of 2 >= 2n-1, which may be < 3n-2.
            # Let's just compute it with a separate (larger) FFT.
            
            fft_size_corr = 1 << (3 * n - 2).bit_length()
            W_corr = rfft(w, fft_size_corr)
            A_corr = rfft(a_cur, fft_size_corr)
            corr_full = irfft(W_corr * np.conj(A_corr), fft_size_corr)[:3*n-2]
            # corr_full[j] = sum_k w[k] * a[k-j] for j = 0, ..., n-1
            # Wait, I need to check the indexing. The cross-correlation (w ⊛ a)[tau] = sum_k w[k+tau] * a[k]
            # = sum_k w[k] * a[k-tau]. So tau = -j in my notation.
            # Let me just use: S[j] = sum_k w[k] * a[k-j]
            # In terms of cross-correlation: if C[tau] = sum_k w[k+tau]*a[k], then S[j] = C[-j] = C[(2n-1)-j] (mod)
            # This is getting too complicated. Let me just use the direct method for the gradient.
            # For n=1200, the direct computation is sum over n positions, each requiring O(n) work = O(n^2) = 1.4M ops.
            # At 30M ops/sec in numpy, that's ~50ms. For 1000 iterations, that's 50s. Acceptable.
            #
            # Better: use the relationship that S = ifft(fft(w) * conj(fft(a)))
            # with proper zero-padding. The issue is that a is only length n, but in the convolution
            # (a*a)[k], a is treated as zero outside [0,n-1].
            #
            # Let me define: a_full of length 2n-1: a_full[j] = a[j] for 0<=j<n, 0 otherwise.
            # Then S[j] = sum_k w[k] * a_full[k-j] for k-j in [0, n-1]
            # = sum_{m=max(-j,0)}^{min(2n-2-j, n-1)} w[m+j] * a_full[m]
            # This is (w_shifted ⊛ a_full) where w_shifted[j] = w[j]
            #
            # I'll just use the simpler approach: compute S via FFT with a_full of length 2n-1
            a_full = np.zeros(2 * n - 1)
            a_full[:n] = a_cur
            fft_sz = 1 << (2 * (2 * n - 1) - 1).bit_length()
            Wf = rfft(w, fft_sz)
            Af = rfft(a_full, fft_sz)
            # Cross-correlation: sum_k w[k] * a_full[k-j]
            # = ifft(FFT(w) * conj(FFT(a_full))) evaluated appropriately
            # Actually: ifft(FFT(w) * conj(FFT(a_full)))[m] = sum_k w[k] * a_full[k-m] (circular)
            # With zero-padding to fft_sz >= len(w) + len(a_full) - 1 = (2n-1) + (2n-1) - 1 = 4n-3
            # Our fft_sz = next_pow2(4n-3) which is >= 4n-3, so no circular artifacts.
            corr = irfft(Wf * np.conj(Af), fft_sz)
            # corr[j] = sum_k w[k] * a_full[k-j] for j = 0, ..., 2n-2
            # We want S[j] = sum_k w[k] * a_full[k-j] for j = 0, ..., n-1
            # But a_full[k-j] is a[k-j] when 0<=k-j<n, and 0 otherwise
            # So S[j] = corr[j] for j = 0, ..., n-1
            S = corr[:n]
            grad_a = 2.0 * S

            # Chain rule for log-parametrization: a = softmax(z)
            # a[j] = exp(z[j]) / sum(exp(z))
            # d(a[j])/d(z[ell]) = a[j] * (delta[j,ell] - a[ell])
            # grad_z[ell] = sum_j grad_a[j] * a[j] * (delta[j,ell] - a[ell])
            #             = grad_a[ell] * a[ell] - a[ell] * sum_j grad_a[j] * a[j]
            S_dot = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S_dot)

            return obj_val, grad_z

        # Use L-BFGS-B
        from scipy.optimize import minimize

        def obj_only(z):
            v, _ = obj_and_grad(z)
            return v

        def grad_only(z):
            _, g = obj_and_grad(z)
            return g

        try:
            res = minimize(
                obj_only, z, jac=grad_only, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-10)
    elif choice == 8:
        # Super-Gaussian
        a = np.exp(-8 * x * (1 - x))
    elif choice == 9:
        # Parabolic
        a = 4 * x * (1 - x) + 0.05
    elif choice == 10:
        # Sinc-like (windowed)
        a = np.sinc(4 * (x - 0.5))
        a = np.maximum(a, 0)
        if a.sum() < 0.01:
            a = np.exp(-4 * x * (1 - x))
    else:
        # Bimodal
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2) + 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_with_early_stop(a_fixed, max_time=0.5):
    """Solve the response LP with a time limit using HiGHS options."""
    n = len(a_fixed)
    size = 2 * n - 1

    T = build_conv_matrix_fast(a_fixed)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.empty((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
            return b_val, float(res.x[n])
    except Exception:
        pass
    return None, None


def fast_alternation_single_n(a_init, time_budget):
    """Run fast LP alternation for a single n value."""
    n = len(a_init)
    a = normalize(a_init.copy())
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    avg_a = a.copy()
    avg_b = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_budget:
        it += 1

        # Solve for b given y_a
        b_new, M1 = lp_with_early_stop(y_a)
        if b_new is None:
            b_new = y_b.copy()

        # Solve for a given y_b
        a_new, M2 = lp_with_early_stop(y_b)
        if a_new is None:
            a_new = y_a.copy()

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_b = (1 - alpha) * y_b + alpha * b_new
        y_b = normalize(y_b)
        y_a = (1 - alpha) * y_a + alpha * a_new
        y_a = normalize(y_a)

        # Polyak-Ruppert
        s = alpha * (it + 1)
        avg_a = (it * avg_a + s * a_new) / (it + s)
        avg_a = normalize(avg_a)
        avg_b = (it * avg_b + s * b_new) / (it + s)
        avg_b = normalize(avg_b)

        # Check improvement
        for cand in [a_new, avg_a, y_a, b_new, avg_b, y_b]:
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = normalize(cand.copy())

        # Stall handling
        if it % 300 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 4:
                scale = 0.03 / math.sqrt(stall)
                y_a += rng.standard_normal(n) * scale
                y_a = normalize(y_a)
                y_b += rng.standard_normal(n) * scale
                y_b = normalize(y_b)
                stall = 0

            if it % 3000 == 0:
                print(f"        n={n}, iter={it}, t={time.monotonic()-t0:.0f}s, score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Fast LP alternation across multiple n values
    2. Subgradient polishing
    3. LogSumExp L-BFGS refinement
    4. Final verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    parent_n = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = local_eval(parent)
            print(f"Parent: n={parent_n}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent_n = 1200
        best_seq = structured_init(parent_n, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={parent_n}, score={best_score:.10f}")

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: score={sc:.10f} (n={len(a)})")

    # === Phase 1: Fast LP alternation for primary n ===
    n_primary = parent_n
    print(f"\n=== Phase 1: Fast LP (n={n_primary}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = fast_alternation_single_n(best_seq, t1)
        try_update(a_lp, "fast_lp_primary")

    # === Phase 2: LP alternation for other n values ===
    print(f"\n=== Phase 2: Multi-n LP Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_candidates = []
        for mult in [0.5, 0.6, 0.7, 0.8, 1.1, 1.3, 1.5, 2.0, 2.5, 3.0]:
            cand_n = int(n_primary * mult)
            if 200 <= cand_n <= 8000 and cand_n != n_primary:
                n_candidates.append(cand_n)
        # Add some absolute values
        for abs_n in [500, 1000, 2000, 4000, 6000]:
            if abs_n not in n_candidates and 200 <= abs_n <= 8000:
                n_candidates.append(abs_n)

        t2_total = min(rem * 0.30, 400.0)
        if t2_total > 30 and len(n_candidates) > 0:
            t_per_n = t2_total / len(n_candidates)
            print(f"  Trying {len(n_candidates)} n values, {t_per_n:.0f}s each")
            for cand_n in sorted(n_candidates):
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                t_sub = min(t_per_n, rem * 0.9)
                if t_sub < 15:
                    break
                print(f"  n={cand_n}, budget={t_sub:.0f}s")
                init_cand = interpolate_sequence(best_seq, cand_n)
                a_cand, _ = fast_alternation_single_n(init_cand, t_sub)
                try_update(a_cand, f"lp_n{cand_n}")

    # === Phase 3: Subgradient polishing ===
    print(f"\n=== Phase 3: Subgradient Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 15:
        a_sg, sc_sg = subgradient_polish(best_seq, t3)
        try_update(a_sg, "subgrad")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 200.0)
    if t4 > 20:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lse, sc_lse = logsumexp_lbgfs(best_seq, t4, T_stages=T_stages)
        try_update(a_lse, "logsumexp")

    # === Phase 5: Second round of fast LP on best ===
    print(f"\n=== Phase 5: Fast LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 200.0)
    if t5 > 20:
        a_lp2, sc_lp2 = fast_alternation_single_n(best_seq, t5)
        try_update(a_lp2, "fast_lp_2")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        # Quick subgradient
        t_sg2 = min(rem * 0.3, 40.0)
        if t_sg2 > 10:
            a_fin, _ = subgradient_polish(best_seq, t_sg2)
            try_update(a_fin, "final_sg")

        # Very low T LSE
        rem = deadline - time.monotonic()
        if rem > 15:
            t_fin = min(rem * 0.5, 50.0)
            a_fin2, _ = logsumexp_lbgfs(best_seq, t_fin,
                                         T_stages=[0.1, 0.05, 0.02, 0.01, 0.005])
            try_update(a_fin2, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
