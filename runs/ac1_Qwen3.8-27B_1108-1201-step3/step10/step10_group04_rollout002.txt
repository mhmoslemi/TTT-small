def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the objective score."""
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Use the efficient algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_reduced_matrix(a, peak_indices):
    """Build the reduced convolution matrix for top-K peaks."""
    n = len(a)
    K = len(peak_indices)

        i_max = min(n - 1, k)
        T[alpha, i_min:i_max+1] = a[k - i_max:k - i_min + 1][::-1]
    return T

def subgradient_step(a, n_active=5):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    sorted_idx = np.argsort(b)[::-1]
    active = sorted_idx[:n_active]
    g = np.zeros(n)
    for k in active:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        g[i_min:i_max+1] += 2.0 * a[k - i_max:k - i_min + 1][::-1]
    g /= n_active
    return g, b.max()

def peak_shave(a0, time_budget, verbose=True):
    """Direct peak shaving: move mass from peak contributors to non-contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find contributors to the peak
        i_min = max(0, peak_k - n + 1)
        i_max = min(n - 1, peak_k)
        idx = np.arange(i_min, i_max + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Move mass from highest contributor to a random non-contributor
        src = idx[np.argmax(contrib)]
        
        # Find positions that don't contribute to the top 5 peaks
        top5 = np.argsort(b)[::-1][:5]
        in_peaks = set()
        for k in top5:

            in_peaks.update(range(im, ix + 1))
        
        candidates = [j for j in range(n) if j not in in_peaks]
        if not candidates:
            candidates = [j for j in range(n) if j != src]
        
        dst = rng.choice(candidates)
        
        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt

            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:

            a_trial = a_trial / ss
            sc = score_of(a_trial)

                best_score = sc

                    if ss > 1e-15:

def structured_init(n, seed=42):
    """Generate a structured initial sequence."""
    rng = np.random.default_rng(seed)

    # Raised cosine with slight perturbation
    a = np.cos(np.pi * (x - 0.5)) ** 2
    # Add small perturbation
    a += 0.01 * rng.random(n)
    a = np.maximum(a, 0)
    a = a / a.sum()

def interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result
