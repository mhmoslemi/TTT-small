"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak reduction,
LogSumExp L-BFGS with multi-stage annealing, and multi-n exploration.
The key innovation is an LP that directly finds the optimal first-order
direction to reduce the maximum convolution peak, combined with aggressive
local polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (assumes a >= 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_peak_reduction_lp(a, K, solver):
    """
    Solve LP to find direction d that reduces the top-K peaks most.
    
    Variables: d (n), t (scalar)
    Maximize: t
    Subject to:
        sum(d) = 0
        d[j] >= -a[j] for all j
        For each of top-K peaks k_i:
            2 * sum_j a[k_i - j] * d[j] <= (M - b[k_i]) - t
    """
    n = len(a)
    size = 2 * n - 1
    b = fast_convolve(a)
    M = b.max()
    
    # Get top-K peak indices
    sorted_idx = np.argsort(b)[::-1]
    active_k = sorted_idx[:K]
    
    # Build G matrix: G[i, j] = 2*a[k_i - j] if valid
    K_eff = len(active_k)
    G = np.zeros((K_eff, n))
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        G[i, idx] = 2.0 * a[k - idx]
    
    # Slack: M - b[k_i] for each active peak
    slack = M - b[active_k]
    
    # LP: maximize t
    # s.t. sum(d) = 0
    #      d >= -a
    #      G @ d + t <= slack  (i.e., G@d <= slack - t)
    #      t >= 0 (optional, but ensures we only take improving steps)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -a,
        G @ d_var + t_var <= slack,
        t_var >= 0
    ]
    
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            d_val = np.asarray(d_var.value, dtype=np.float64)
            t_val = float(t_var.value)
            d_val = np.maximum(d_val, -a)  # clip for numerical safety
            return d_val, max(0.0, t_val)
    except Exception:
        pass
    
    # Fallback solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = np.asarray(d_var.value, dtype=np.float64)
                t_val = float(t_var.value)
                d_val = np.maximum(d_val, -a)
                return d_val, max(0.0, t_val)
        except Exception:
            continue
    
    return None, 0.0


def lp_peak_reduction(a0, solver, time_limit, K_init=20, K_max=100):
    """
    Main optimization loop using LP-based peak reduction.
    Adaptively increases K (number of active peaks) as progress slows.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    
    K = K_init
    stall = 0
    prev_best = best_score
    lr = 1.0  # step size for LP direction
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        d, t_opt = solve_peak_reduction_lp(a, K, solver)
        
        if d is None:
            # LP failed, try subgradient step
            b = fast_convolve(a)
            M = b.max()
            top_k = np.argsort(b)[::-1][:K]
            g = np.zeros(n)
            for k in top_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += 2.0 * a[k - idx]
            g /= len(top_k)
            g = g - g.mean()  # project to sum=0
            a_trial = a + 0.01 * g  # note: g is "negative direction" so add
            # Wait, we want to REDUCE peaks. g points toward increasing peaks.
            # So we should go in direction -g.
            a_trial = a - 0.01 * g
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = local_eval(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                stall = 0
            else:
                stall += 1
            it += 1
            continue
        
        if t_opt < 1e-10:
            # Can't improve with current K, increase K
            K = min(K * 2, K_max)
            if K > K_max:
                # Add noise and reset
                a = a + rng.standard_normal(n) * 0.005
                a = normalize(a)
                K = K_init
            stall += 1
            it += 1
            continue
        
        # Take a step in direction d
        # d already satisfies sum(d)=0 and d >= -a
        # Try full step first, then damped
        step_factors = [1.0, 0.7, 0.5, 0.3, 0.1]
        improved = False
        for alpha in step_factors:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = local_eval(a_trial)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # If stalling, try adding noise
            if stall % 10 == 0:
                a = a + rng.standard_normal(n) * (0.002 / math.sqrt(stall // 10 + 1))
                a = normalize(a)
        
        it += 1
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}, K={K}, stall={stall}")
    
    return best_a, best_score

C[2] = sum_k w_padded[k]*a_padded[(k-2) mod 8] = w0*a[6] + w1*a[7] + w2*a0 + w3*a1 + w4*a2 = w2*a0 + w3*a1 + w4*a2 ✓ (matches R[2])

grad_a = 2 * ifft(conj(fft(w_padded)) * fft(a_padded)).real[:n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining
LP-based peak reduction, FFT-accelerated LogSumExp L-BFGS with multi-stage
annealing, adaptive subgradient steps, and systematic multi-n exploration.
Key innovations: (1) LP that directly optimizes the first-order peak reduction
direction with adaptive active-set sizing, (2) FFT-based gradient computation
for the LogSumExp objective reducing complexity to O(n log n) per iteration,
(3) aggressive multi-n sampling with interpolation-based initialization.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (assumes a >= 0, sum normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_peak_lp(a, K, solver):
    """
    Solve LP to find direction d reducing top-K peaks.
    Maximize t s.t. sum(d)=0, d>=-a, G@d + t <= slack.
    Returns (d, t_opt) or (None, 0).
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    sorted_idx = np.argsort(b)[::-1]
    active_k = sorted_idx[:K]
    K_eff = len(active_k)
    
    G = np.zeros((K_eff, n))
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        G[i, idx] = 2.0 * a[k - idx]
    
    slack = M - b[active_k]
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -a,
        G @ d_var + t_var <= slack,
        t_var >= 0
    ]
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    
    solvers_to_try = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = np.asarray(d_var.value, dtype=np.float64)
                t_val = max(0.0, float(t_var.value))
                d_val = np.maximum(d_val, -a)
                return d_val, t_val
        except Exception:
            continue
    return None, 0.0


def lp_peak_reduction(a0, solver, time_limit, K_init=15):
    """LP-based peak reduction with adaptive K and damping."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    K = K_init
    stall = 0
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit:
        d, t_opt = solve_peak_lp(a, K, solver)
        
        if d is not None and t_opt > 1e-12:
            improved = False
            for alpha in [1.0, 0.8, 0.5, 0.3, 0.1]:
                a_trial = a + alpha * d
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = local_eval(a_trial)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                stall += 1
        else:
            # LP couldn't help; use subgradient step
            b = fast_convolve(a)
            M = b.max()
            top_k = np.argsort(b)[::-1][:min(K, 50)]
            g = np.zeros(n)
            for k in top_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += 2.0 * a[k - idx]
            g /= len(top_k)

            a_trial = a - 0.005 * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()

                sc = local_eval(a_trial)

                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    stall = 0
            else:
                stall += 1
        
        it += 1
        # Adaptive K and noise
        if stall > 20:

                a = a + rng.standard_normal(n) * 0.003
                a = normalize(a)
                K = K_init
            stall = 0
        elif it % 50 == 0 and stall > 5:
            a = a + rng.standard_normal(n) * 0.001
            a = normalize(a)
        
        if (it + 1) % 100 == 0:
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}, K={K}")
    
    return best_a, best_score


def lse_objective_and_grad(z, T, n, fft_size):
    """Compute LogSumExp objective and gradient w.r.t. z (log-param)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    lse_val = b_max + np.log(exp_ws.sum())
    obj_val = (2.0 * n) * lse_val
    
    # Gradient via FFT: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    L = 1 << ((2 * n - 1 + n - 1 + 1).bit_length())
    W = rfft(w, L)
    A = rfft(a, L)
    grad_a = irfft(np.conj(W) * A, L)[:n]
    grad_a *= 2.0
    # Chain rule for softmax parametrization
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with FFT-based gradients."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    fft_size = 1 << (2 * n - 1 + n - 1 + 1).bit_length()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 25))
        
        def obj_fn(z_cur):
            z_cur = np.asarray(z_cur, dtype=np.float64)
            return lse_objective_and_grad(z_cur, T, n, fft_size)[0]
        
        def grad_fn(z_cur):
            z_cur = np.asarray(z_cur, dtype=np.float64)
            return lse_objective_and_grad(z_cur, T, n, fft_size)[1]
        
        try:
            res = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = local_eval(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def fast_subgradient(a0, time_limit, n_steps=50000):
    """Fast subgradient method with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 200 == 0:
            b = fast_convolve(a)
            M = b.max()
            threshold = M * 0.999
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(b))])
            g = np.zeros(n)
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += 2.0 * a[k - idx]
            g /= len(active)
            g = g - g.mean()
            a_trial = a - lr * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial / st
            sc = local_eval(a_trial)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
                lr = min(lr * 1.02, 0.08)
            else:
                no_improve += 1
                if no_improve >= 15:
                    lr *= 0.9
                    no_improve = 0
            lr = max(1e-6, lr)
        else:
            # Cheaper step: only update every few steps
            pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 8:
        a = (np.sinc((x - 0.5) * n * 0.3) ** 2)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) * (1.0 + 0.2 * rng.standard_normal(n))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
