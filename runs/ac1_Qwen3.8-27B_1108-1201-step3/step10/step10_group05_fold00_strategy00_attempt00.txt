def simplex_project(v):
    """Project a vector to the simplex (all elements non-negative and sum to 1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (css - 1)
    if cond.any():
        rho = cond[cond].sum()
        theta = (css[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    else:
        return np.ones(n)/n * v.sum()
