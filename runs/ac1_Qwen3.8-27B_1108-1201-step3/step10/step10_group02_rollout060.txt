The proximal operator for the simplex is: `proj_simplex(v) = argmin_{x in simplex} ||x - v||^2`. This can be computed in O(n log n) time.

For the smoothed version `f_smooth(a) = (1/T) log(sum_k exp(T * conv(a,a)[k]))`, the gradient is:
`grad f_smooth = (1/T) * (1/sum_k exp(T*b[k])) * T * d(sum_k exp(T*b[k]))/da`
= `(1/sum_k exp(T*b[k])) * T * sum_k exp(T*b[k]) * (d b[k]/d a)`
= `sum_k w_k * d b[k]/d a` where `w_k = exp(T*b[k]) / sum_j exp(T*b[j])`

This is the same as the subgradient but with a softmax weighting instead of the max. As T -> inf, it approaches the true subgradient.

**FISTA with smoothed objective:**
```
y_0 = a_0, t_0 = 1
for k = 0, 1, ...:
    g = grad f_smooth(y_k)
    a_{k+1} = proj_simplex(y_k - (1/L) * g)  where L is Lipschitz constant
    t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
    y_{k+1} = a_{k+1} + (t_k - 1)/t_{k+1} * (a_{k+1} - a_k)
```

The Lipschitz constant L of the smoothed objective can be bounded. For `f_smooth = (1/T) log(sum exp(T*conv(a,a)))`:
`||grad f_smooth|| <= max_k ||grad conv(a,a)[k]|| * (some weight)`

Actually, the Lipschitz constant of the gradient of the log-sum-exp is bounded by the Lipschitz constant of the max, which for `conv(a,a)` is... it depends on the values of `a`.

When `b` is fixed, `max_k conv(a, b)[k]` is the max of linear functions in `a`, so its subgradient at `a` is:
`g = (1/|A|) sum_{k in A} d(conv(a,b)[k])/d(a) = (1/|A|) sum_{k in A} b[shifted]`

Specifically, `conv(a,b)[k] = sum_j a[j]*b[k-j]`, so `d/d a[i] conv(a,b)[k] = b[k-i]` (if `0 <= k-i < n`).

So `g[i] = (1/|A|) sum_{k in A, 0 <= k-i < n} b[k-i] = (1/|A|) sum_{k in A} b[k-i] * 1[0 <= k-i < n]`.

This is just a "smeared" version of `b` at the active peak positions. It can be computed efficiently.

Then the projected gradient step is:
`a_new = proj_simplex(a - step_size * g)`

where `proj_simplex` is the Euclidean projection onto the simplex `{x: x >= 0, sum(x) = 1}`.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Alternating with Nesterov 
acceleration (HiGHS solver), LogSumExp-FISTA, and targeted coordinate descent.
Improvements: faster LP via precomputed sparse structure, multi-phase schedule,

import numpy as np

from scipy.fft import rfft, irfft

# ===== Core utilities =====

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate the objective function."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def precompute_conv_indices(n):
    """Precompute (row, col) indices for the convolution matrix T."""
    size = 2 * n - 1
    rows = []
    cols = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_response(a_fixed, rows, cols, n):
    """Solve LP: min t s.t. T@b <= t, sum(b)=1, b>=0.
    Returns optimal b."""
    size = 2 * n - 1
    nnz_T = len(cols) - size
    
    # Build A_ub = [T, -1]
    vals = np.empty(len(cols))
    vals[:nnz_T] = a_fixed[cols[:nnz_T]]
    vals[nnz_T:] = -1.0
    
    A_ub = sp.csr_matrix((vals, (rows, cols)), shape=(size, n + 1))
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(size),
                  A_eq=A_eq, b_eq=np.ones(1),
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.status == 0:
        b_opt = res.x[:n]
        b_opt = np.maximum(b_opt, 0)
        s = b_opt.sum()
        if s > 1e-15:
            b_opt = b_opt / s
            return b_opt
    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Alternating with Nesterov
acceleration (HiGHS solver), projected subgradient with backtracking, and
targeted coordinate descent. Key improvements: precomputed sparse LP structure,
HiGHS for speed, multi-phase schedule with adaptive restarts, and a dual-side
Nesterov scheme for better convergence.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def precompute_conv_indices(n):
    """Precompute (row, col) indices for the LP constraint matrix [T, -1]."""
    size = 2 * n - 1
    rows_list = []
    cols_list = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        rows_list.extend([k] * (je - js + 1))
        cols_list.extend(range(js, je + 1))
    # Add the -1 column entries
    rows_list.extend(range(size))
    cols_list.extend([n] * size)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_response(a_fixed, rows, cols, n):
    """Solve LP: min t s.t. conv(a_fixed, b)[k] <= t for all k, sum(b)=1, b>=0."""
    size = 2 * n - 1
    nnz_T = len(cols) - size
    vals = np.empty(len(cols), dtype=np.float64)
    vals[:nnz_T] = a_fixed[cols[:nnz_T]]
    vals[nnz_T:] = -1.0
    A_ub = sp.csr_matrix((vals, (rows, cols)), shape=(size, n + 1))
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(size),
                      A_eq=A_eq, b_eq=np.ones(1),
                      bounds=bounds, method='highs',
                      options={'presolve': True,
                               'primal_feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            b_opt = np.maximum(res.x[:n], 0)
            s = b_opt.sum()
            if s > 1e-15:
                return b_opt / s
    except Exception:
        pass
    return None


def compute_subgradient_conv(a, active_indices, other=None):
    """Compute subgradient of max_{k in active} conv(a, other)[k] w.r.t. a.
    If other is None, other = a (self-convolution)."""
    n = len(a)
    other = other if other is not None else a
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += other[k - idx]
    g /= max(1, len(active_indices))
    return g


def nesterov_lp_alternating(a0, time_limit, rows, cols, n, seed=42,
                             max_stall=250, verbose=True):
    """LP-Alternating with Nesterov acceleration."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    lp_fail_count = 0

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_response(y, rows, cols, n)
        if b_new is None:
            lp_fail_count += 1
            if lp_fail_count > 50:
                break
            b_new = y.copy()
        else:
            lp_fail_count = 0

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        else:
            a = y.copy()

        # Evaluate both
        score_a = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-13:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        elif score_y < best_score - 1e-13:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart when stalled
        if stall_count >= max_stall:
            noise_scale = min(0.04, 0.008 * np.sqrt(stall_count / 50.0))
            noise = rng.standard_normal(n) * noise_scale
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = a + noise * 0.3
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
            stall_count = 0
            if verbose:
                print(f"    [restart] iter={it+1} noise={noise_scale:.4f} score={best_score:.8f}")

        if verbose and (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt {it+1} ({elapsed:.0f}s): best={best_score:.8f} stall={stall_count}")

    return best_a, best_score


def mirror_descent(a0, time_limit, seed=0, eta0=0.06, active_frac=0.99,
                   nesterov=True, verbose=True):
    """Mirror descent with Nesterov acceleration for the max-conv objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    stall_count = 0
    rng = np.random.default_rng(seed)
    prev_a = a.copy()
    prev_g = None

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = compute_subgradient_conv(y, active)

        # BB step size
        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-18 and y_dot_y > 1e-18:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * min(eta_bb, 2.0)
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.999
        prev_a = a.copy()
        prev_g = g.copy()

        # Mirror descent update (exponential weighting)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a_new.copy()

        a = a_new

        score = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
