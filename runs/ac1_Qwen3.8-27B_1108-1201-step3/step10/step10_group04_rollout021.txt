    Alternates between flattening |FFT(a)| and projecting to non-negative simplex.
    """

    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

        A = np.fft.rfft(a, N)

        target = np.quantile(mag_sq[mag_sq > 0], target_quantile) if np.any(mag_sq > 0) else 1.0

        ratio = target / (mag_sq + 1e-30)

        mag_sq_new = np.maximum(mag_sq_new, 0)

        if s_new < 1e-15:
            # Failed, reset
            break
        a_new = a_new / s_new
        
        a = a_new
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-start initialization with length exploration
2. Frequency-domain flattening (Gerchberg-Saxton style) - primary new method
3. Multiplicative gradient descent (replicator dynamics) - new

5. LP-based minimax best-response (Nesterov accelerated)
6. Peak equalization and final polish

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def freq_domain_flatten(a0, time_limit, alpha_start=0.15, alpha_end=0.5,
                        target_frac=0.6, n_restarts=3):
    """
    Gerchberg-Saxton style frequency-domain flattening.
    Alternates between flattening |FFT(a)| and projecting to non-negative simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart > 0:
            runtime = time.monotonic() - t0
            total_est = runtime * (n_restarts - restart) / restart
            if total_est > time_limit:
                break
            # Perturb for restart
            rng = np.random.default_rng(restart * 7919 + 13)
            a = a + rng.random(n) * 0.05
            a = np.maximum(a, 0)
            a = a / a.sum()

        n_iters = 2000
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            # Adaptive alpha: start gentle, become more aggressive
            progress = it / n_iters
            alpha = alpha_start + (alpha_end - alpha_start) * progress

            # Forward to frequency domain
            A = np.fft.fft(a, N)
            mag_sq = np.abs(A) ** 2

            # Target level: quantile of the spectrum
            nonzero = mag_sq[mag_sq > 1e-30]
            if len(nonzero) > 0:
                target = np.quantile(nonzero, target_frac)
            else:
                target = 1.0
            if target < 1e-30:
                target = 1.0

            # Flatten: push peaks down, troughs up
            ratio = target / (mag_sq + 1e-30)
            # Soft power-law adjustment
            mag_sq_new = mag_sq * (ratio ** alpha)
            mag_sq_new = np.maximum(mag_sq_new, 0)

            # Reconstruct with new magnitude, same phase
            mag_new = np.sqrt(mag_sq_new)
            phase = np.angle(A)
            A_new = mag_new * np.exp(1j * phase)

            # Inverse transform
            a_new = np.real(np.fft.ifft(A_new, N))[:n]

            # Project to non-negative simplex
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

            a = a_new

            # Check improvement every 50 iterations
            if it % 50 == 0:
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = a.copy()

    # Final check
    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def freq_domain_softcap(a0, time_limit, cap_quantile=0.85, n_iterations=1000):
    """
    Frequency-domain flattening with soft capping.
    More aggressive peak reduction by directly capping the top of the spectrum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a, N)
        mag = np.abs(A)

        # Cap the top (1 - cap_quantile) fraction
        cap_level = np.quantile(mag[mag > 0], cap_quantile) if np.any(mag > 0) else 1.0

        # Soft cap: use smooth min
        beta = 1.0 + 0.5 * it / n_iterations  # increasing sharpness
        mag_capped = np.minimum(mag, cap_level + (mag - cap_level) * np.exp(-beta * np.maximum(0, mag - cap_level) / (cap_level + 1e-10)))

        # Boost the bottom slightly to maintain energy
        mean_mag = mag.mean()
        if mean_mag > 1e-10:
            scale = mean_mag / (mag_capped.mean() + 1e-10)
            mag_capped = mag_capped * np.sqrt(scale)  # sqrt to be gentle

        A_new = mag_capped * np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        if it % 50 == 0:
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()

    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def multiplicative_gd(a0, time_limit, n_active_frac=0.08):
    """
    Multiplicative (replicator) gradient descent on the probability simplex.
    Naturally maintains non-negativity via exponential parameterization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1 - n_active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient (vectorized per active index)
        g = np.zeros(n)
        for k in active:
            j1_start = max(0, k - n + 1)
            j1_end = min(n - 1, k)
            j1 = np.arange(j1_start, j1_end + 1)
            g[j1] += a[k - j1]
            j2_start = max(0, k)
            j2_end = min(n - 1, k + n - 1)
            j2 = np.arange(j2_start, j2_end + 1)
            g[j2] += a[j2 - k]
        g /= len(active)

        # Multiplicative update via log-space
        z = np.log(a + 1e-30) - lr * g
        a_new = np.exp(z - z.max())
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 5.0)
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve += 1
            lr = max(lr * 0.95, 0.05)

        if no_improve >= 50:
            rng = np.random.default_rng(step * 31 + 7)
            noise = rng.random(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            lr = 0.5

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MGD iter={step} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.3f}")

    return best_a, best_score


def lse_lbgfgs(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=10):
    """L-BFGS with LogSumExp approximation, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jac(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        max_iter = max(30, int(200 * remaining / (time_limit)))

        try:
            res = minimize(
                fun=lambda z_in: lse_obj(z_in, stage_T),
                x0=z,
                jac=lambda z_in: lse_jac(z_in, stage_T),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': max_iter * 100,
                         'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass

        if si % 3 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def nesterov_lp(a0, time_limit, solver, n_max_iters=100000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0 and it > 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                rng = np.random.default_rng(it * 13 + 37)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def peak_equalization(a0, time_limit, n_steps=20000):

            je = min(n - 1, k)

                a_trial[src] -= amt
                a_trial[dst] += amt

                ss = a_trial.sum()
                if ss < 1e-15:

                a_trial = a_trial / ss

                if score < best_score - 1e-14:

    print(f"\n=== Phase 1: Initialization (best={best_score:.6f}) ===")
    t1_deadline = time.monotonic() + 60
    n_init = len(best_seq)

    # Try a few different initializations at the same length
    inits = [best_seq.copy()]
    for i in range(3):
        inits.append(structured_init(n_init, rng))
    inits.append(random_init(n_init, rng))

    for idx, init_a in enumerate(inits):
        if time.monotonic() > t1_deadline:
            break
        sc = evaluate_safe(init_a)
        if sc < best_score:
            best_score = sc
            best_seq = init_a.copy()
            print(f"  Better init {idx}: {sc:.6f}")

    # Quick freq-domain pass on best
    if time.monotonic() < t1_deadline - 10:
        t_rem = t1_deadline - time.monotonic() - 5
        a_freq, _ = freq_domain_flatten(best_seq, min(t_rem, 30), alpha_start=0.1, alpha_end=0.3, n_restarts=1)
        try_update(a_freq, "freq_init")

    # ============================================================
    # Phase 2: Frequency-domain flattening (400s)
    # ============================================================
    print(f"\n=== Phase 2: Frequency-Domain Flattening (best={best_score:.6f}) ===")
    t2_deadline = time.monotonic() + 400

    # Main GS iterations
    t_rem = t2_deadline - time.monotonic() - 20
    if t_rem > 20:
        a_gs, _ = freq_domain_flatten(best_seq, t_rem * 0.5,
                                       alpha_start=0.1, alpha_end=0.4,
                                       target_frac=0.6, n_restarts=2)
        try_update(a_gs, "gs_main")

    # Soft-cap variant
    t_rem = t2_deadline - time.monotonic() - 10
    if t_rem > 15:
        a_cap, _ = freq_domain_softcap(best_seq, t_rem * 0.5,
                                        cap_quantile=0.85, n_iterations=800)
        try_update(a_cap, "softcap")

    # Another GS with different parameters
    t_rem = t2_deadline - time.monotonic()
    if t_rem > 10:
        a_gs2, _ = freq_domain_flatten(best_seq, t_rem,
                                        alpha_start=0.2, alpha_end=0.6,
                                        target_frac=0.5, n_restarts=1)
        try_update(a_gs2, "gs_aggr")

    # ============================================================
    # Phase 3: Multiplicative Gradient Descent (300s)
    # ============================================================
    print(f"\n=== Phase 3: Multiplicative GD (best={best_score:.6f}) ===")
    t3_deadline = time.monotonic() + 300

    t_rem = t3_deadline - time.monotonic()
    if t_rem > 20:
        a_mgd, _ = multiplicative_gd(best_seq, t_rem * 0.6, n_active_frac=0.05)
        try_update(a_mgd, "mgd_main")

    # Second pass with different active fraction
    t_rem = t3_deadline - time.monotonic()
    if t_rem > 15:
        a_mgd2, _ = multiplicative_gd(best_seq, t_rem, n_active_frac=0.15)
        try_update(a_mgd2, "mgd_wide")

    # ============================================================
    # Phase 4: LogSumExp L-BFGS (300s)
    # ============================================================
    print(f"\n=== Phase 4: LSE L-BFGS (best={best_score:.6f}) ===")
    t4_deadline = time.monotonic() + 300

    t_rem = t4_deadline - time.monotonic()
    if t_rem > 20:
        a_lse = lse_lbgfgs(best_seq, t_rem * 0.7, T_start=30.0, T_end=0.1, n_stages=8)
        try_update(a_lse, "lse_main")

    t_rem = t4_deadline - time.monotonic()
    if t_rem > 10:
        a_lse2 = lse_lbgfgs(best_seq, t_rem, T_start=5.0, T_end=0.05, n_stages=6)
        try_update(a_lse2, "lse_fine")

    # ============================================================
    # Phase 5: LP Minimax (400s)
    # ============================================================
    print(f"\n=== Phase 5: LP Minimax (best={best_score:.6f}) ===")
    t5_deadline = time.monotonic() + 400

    t_rem = t5_deadline - time.monotonic()
    if t_rem > 30:
        a_lp, _ = nesterov_lp(best_seq, t_rem * 0.6, solver=solver, n_max_iters=50000)
        try_update(a_lp, "lp_main")

    t_rem = t5_deadline - time.monotonic()
    if t_rem > 20:
        a_lp2, _ = nesterov_lp(best_seq, t_rem, solver=solver, n_max_iters=30000)
        try_update(a_lp2, "lp_fine")

    # ============================================================
    # Phase 6: Peak equalization + length tuning (200s)
    # ============================================================
    print(f"\n=== Phase 6: Peak Eq + Length (best={best_score:.6f}) ===")
    t6_deadline = time.monotonic() + 200

    # Peak equalization
    t_rem = t6_deadline - time.monotonic()
    if t_rem > 15:
        a_pe, _ = peak_equalization(best_seq, min(t_rem * 0.4, 60), n_steps=15000)
        try_update(a_pe, "peak_eq")

    # Try nearby lengths
    n_cur = len(best_seq)
    length_tries = []
    for delta in [-20, -10, -5, 5, 10, 20, 50]:
        nl = n_cur + delta
        if 200 <= nl <= 3000 and nl != n_cur:
            length_tries.append(nl)

    for nl in length_tries[:4]:
        t_rem = t6_deadline - time.monotonic()
        if t_rem < 15:
            break
        init_new = interpolate_sequence(best_seq, nl)
        sc_init = evaluate_safe(init_new)
        cur = init_new
        if sc_init < best_score:
            best_score = sc_init
            best_seq = cur.copy()
            print(f"  Length {nl} init: {sc_init:.6f}")

        # Quick optimization at new length
        t_alloc = min(t_rem * 0.3, 30)
        if t_alloc > 5:
            a_new, _ = freq_domain_flatten(cur, t_alloc * 0.5,
                                            alpha_start=0.1, alpha_end=0.3,
                                            target_frac=0.6, n_restarts=1)
            try_update(a_new, f"freq_n{nl}")

        t_alloc = t6_deadline - time.monotonic() - 5
        if t_alloc > 10:
            a_new2 = lse_lbgfgs(best_seq if len(best_seq) == nl else cur,
                                  min(t_alloc, 20), T_start=10.0, T_end=0.1,
                                  n_stages=5)
            try_update(a_new2, f"lse_n{nl}")

    # ============================================================
    # Phase 7: Final polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        # Final LP
        t_lp = min(t_rem * 0.3, 60)
        if t_lp > 5:
            a_fin, _ = nesterov_lp(best_seq, t_lp, solver=solver, n_max_iters=20000)
            try_update(a_fin, "final_lp")

        # Final LSE
        t_rem = deadline - time.monotonic()
        if t_rem > 8:
            t_lse = min(t_rem * 0.3, 30)
            if t_lse > 5:
                a_fin2 = lse_lbgfgs(best_seq, t_lse, T_start=3.0, T_end=0.05, n_stages=5)
                try_update(a_fin2, "final_lse")

        # Final peak eq
        t_rem = deadline - time.monotonic()
        if t_rem > 5:
            t_pe = min(t_rem * 0.3, 20)
            if t_pe > 3:
                a_fin3, _ = peak_equalization(best_seq, t_pe, n_steps=10000)
                try_update(a_fin3, "final_pe")

        # One more quick freq
        t_rem = deadline - time.monotonic()
        if t_rem > 5:
            a_fin4, _ = freq_domain_flatten(best_seq, min(t_rem, 15),
                                            alpha_start=0.05, alpha_end=0.15,
                                            target_frac=0.7, n_restarts=1)
            try_update(a_fin4, "final_freq")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
