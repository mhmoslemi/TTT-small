    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')

    return float(2.0 * n * b.max() / (s * s))

    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Flatten the spectrum of a to reduce auto-convolution peaks."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a_work = normalize(a)
    best_a = a_work.copy()
    best_score = local_eval(a_work)

        # Compute FFT
        A = fft(a_work, L)
        
        # Flatten magnitude
        mag = np.abs(A)
        mean_mag2 = np.mean(mag ** 2)
        # Soft flattening: |A_new| = sqrt(alpha * |A|^2 + (1-alpha) * mean(|A|^2))

        new_mag = np.sqrt(new_mag2)

        if sc < best_score:

            best_a = a_candidate.copy()
            a_work = a_candidate

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_flatten(a, time_budget=120.0, alpha_start=0.95, alpha_end=0.5):
    a_work = normalize(a)
    n = len(a_work)
    L = 1 << (2 * n - 1).bit_length()
    best_a = a_work.copy()
    best_score = local_eval(a_work)
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_budget:
        it += 1
        frac = min(1.0, it / 5000)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        A = fft(a_work, L)
        mag = np.abs(A)
        mag2 = mag * mag
        mean_mag2 = np.mean(mag2)
        new_mag2 = alpha * mag2 + (1.0 - alpha) * mean_mag2
        new_mag = np.sqrt(np.maximum(new_mag2, 0))
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_cand = np.real(ifft(A_new, L))[:n]
        a_cand = np.maximum(a_cand, 0)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand = a_cand / s
        sc = local_eval(a_cand)
        if sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            a_work = a_cand
    return best_a, best_score


def smoothed_gradient_descent(a0, time_budget=300.0, T0=5.0, T_end=0.01):
    """Minimize LogSumExp-smoothed max of conv(a,a) with gradient descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()

    T_stages = [T0, T0*0.6, T0*0.4, T0*0.25, T0*0.15, T0*0.08, T0*0.04,
                T0*0.02, T0*0.01, T0*0.005, T0*0.002, T_end]

    for T in T_stages:
        if time.monotonic() - t0 > time_budget:
            break
        stage_time = min(25.0, time_budget - (time.monotonic() - t0))
        lr = min(0.02, 1.0 / (n * T * 0.1))

        while time.monotonic() - t0 < time_budget and (time.monotonic() - t0 < t0 + stage_time or True):
            if time.monotonic() - t0 > time_budget:
                break
            b = fast_convolve(a)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()

            # Gradient: grad_a[j] = 2 * sum_k w[k] * a[k-j]
            # Use FFT cross-correlation
            L_grad = 1 << (3 * n).bit_length()
            W = rfft(w, L_grad)
            A_ff = rfft(a, L_grad)
            cross = irfft(W * A_ff.conj(), L_grad)[:n]
            grad_a = 2.0 * cross

            a_trial = a - lr * grad_a
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                lr *= 0.5
                continue
            a_trial = a_trial / s
            sc = local_eval(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            if sc < local_eval(a) + 1e-10:
                a = a_trial
                lr = min(lr * 1.005, 0.05)
            else:
                lr *= 0.7
                if lr < 1e-8:
                    break
    return best_a, best_score


def subgradient_polish(a0, time_budget=120.0):
    """Exact subgradient method with polyak step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    lr = 0.01

    while time.monotonic() - t0 < time_budget:
        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_trial = a_trial / s
        sc = local_eval(a_trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.002, 0.03)
        else:
            lr *= 0.95
            if lr < 1e-7:
                break
    return best_a, best_score


def generate_initials(n, rng):
    """Generate diverse initial sequences of length n."""
    initials = []
    x = np.linspace(0, 1, n)

    # Linear ramp (increasing)
    a = np.arange(1, n + 1, dtype=np.float64)
    initials.append(normalize(a))

    # Linear ramp (decreasing)
    a = np.arange(n, 0, -1, dtype=np.float64)
    initials.append(normalize(a))

    # Quadratic ramp
    a = np.arange(1, n + 1, dtype=np.float64) ** 2
    initials.append(normalize(a))

    # Sine window
    a = np.sin(np.pi * x) ** 2 + 0.1
    initials.append(normalize(a))

    # Cosine window (Highpass-like)
    a = np.cos(np.pi * x / 2) ** 3 + 0.05
    initials.append(normalize(a))

    # Gaussian centered
    a = np.exp(-3.0 * (x - 0.5) ** 2 / (0.15 ** 2))
    initials.append(normalize(a))

    # Exponential ramp
    a = np.exp(2.0 * x)
    initials.append(normalize(a))

    # Triangle (peak in middle)
    a = 1.0 - 2.0 * np.abs(x - 0.5)
    a = np.maximum(a, 0.01)
    initials.append(normalize(a))

    # Plateau with smooth edges
    a = np.ones(n)
    edge = max(3, n // 10)
    for i in range(edge):
        a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
        a[n-1-i] = a[i]
    initials.append(normalize(a))

    # Random (Dirichlet)
    a = rng.dirichlet(np.ones(n))
    initials.append(normalize(a))

    return initials


def multi_start_optimize(n, time_budget, rng):
    """Optimize multiple initializations and return the best."""
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()
    per_method_time = time_budget / 10.0

    initials = generate_initials(n, rng)

    for i_init, ainit in enumerate(initials):
        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 30:
            break

        alloc = min(remaining * 0.15, per_method_time)
        if alloc < 10:
            alloc = min(remaining * 0.3, 30.0)

        print(f"    Init {i_init}: score={local_eval(ainit):.8f}, time_alloc={alloc:.0f}s")

        # Spectral flattening
        a1, s1 = spectral_flatten(ainit, time_budget=min(alloc * 0.4, 40.0))
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
            print(f"      After spectral: {s1:.8f}")

        # Smoothed gradient descent
        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining > 30:
            t_grad = min(remaining * 0.3, 120.0)
            a2, s2 = smoothed_gradient_descent(a1, time_budget=t_grad)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
                print(f"      After smooth grad: {s2:.8f}")

        # Subgradient polish
        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining > 20:
            t_sg = min(remaining * 0.2, 60.0)
            a3, s3 = subgradient_polish(a2 if s2 <= s1 else a1, time_budget=t_sg)
            if s3 < best_score:
                best_score = s3
                best_a = a3.copy()
                print(f"      After subgrad: {s3:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_best_response_one_step(a_fixed):
    """One LP best-response step using scipy's linear programming."""
    from scipy.optimize import linprog
    n = len(a_fixed)
    size = 2 * n - 1

    # Minimize M subject to T @ b <= M, sum(b) = 1, b >= 0
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: T @ b - M <= 0 for each row
    # T is (2n-1) x n, so inequality matrix is (2n-1) x (n+1)
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(size)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    return None
