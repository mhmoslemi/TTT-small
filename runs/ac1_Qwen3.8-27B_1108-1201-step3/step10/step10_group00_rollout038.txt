- If k-j == j (i.e., k=2j): the term a[j]*a[j] = a[j]^2, d/d(a[j]) = 2*a[j].

- If k = 2j (and 0 <= j < n): derivative = 2*a[j]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_reduced(a_fixed, active_indices, n):
    """Build reduced (|active|) x n matrix for active constraints only."""
    T = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a_fixed[k - idx]
    return T


def get_lp_solver():

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_ob_grad(z, n, T_temp, top_k=10):
    """Compute LSE objective and gradient w.r.t. z with top-k hybrid."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Hybrid: use top-k + LSE for the rest
    if top_k > 0 and len(b) > top_k:
        sorted_idx = np.argsort(b)[::-1]
        top_indices = sorted_idx[:top_k]
        rest_indices = sorted_idx[top_k:]
        
        top_vals = b[top_indices]
        rest_vals = b[rest_indices] if len(rest_indices) > 0 else np.array([max_b - 1])
        
        # LSE on rest, plus max of top
        rest_max = rest_vals.max()
        rest_exp = np.exp((rest_vals - rest_max) / T_temp)
        rest_lse = rest_max + T_temp * np.log(rest_exp.sum())
        
        # Combine: max(top) and rest_lse via LSE
        combined_vals = np.concatenate([top_vals, np.array([rest_lse])])
        c_max = combined_vals.max()
        c_exp = np.exp((combined_vals - c_max) / T_temp)
        lse = c_max + T_temp * np.log(c_exp.sum())
    else:
        exp_terms = np.exp((b - max_b) / T_temp)
        lse = max_b + T_temp * np.log(exp_terms.sum())
    
    obj = 2.0 * n * lse
    
    # Gradient: compute subgradient at top-k positions (equal weight)
    # Plus LSE gradient for smoothness
    grad_a = np.zeros(n)
    
    # Top-k equal weight subgradient
    if top_k > 0 and len(b) >= top_k:
        sorted_idx = np.argsort(b)[::-1]
        top_indices = sorted_idx[:top_k]
        for k in top_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a[k - idx]
        grad_a /= top_k
    
    # Add LSE gradient component (weighted by soft weights)
    exp_terms = np.exp((b - max_b) / (T_temp * 2))
    w = exp_terms / exp_terms.sum()
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

        # Pressure: how much each coordinate contributes to top peaks
        pressure = np.zeros(n)
        for k in top_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += a[k - idx]
        pressure /= top_k
        
        # Find high-pressure and low-pressure coordinates
        p_threshold_high = np.percentile(pressure, 90)
        p_threshold_low = np.percentile(pressure, 10)
        
        high_coords = np.where(pressure > p_threshold_high)[0]
        low_coords = np.where(pressure < p_threshold_low)[0]
        
        if len(high_coords) == 0 or len(low_coords) == 0:
            break
        
        # Move a small amount from high to low
        h_idx = high_coords[rng.integers(len(high_coords))]
        l_idx = low_coords[rng.integers(len(low_coords))]
        
        move = delta * min(a[h_idx], 0.01)
        if move < 1e-10:
            continue
        
        a_new = a.copy()
        a_new[h_idx] -= move
        a_new[l_idx] += move
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller move
            pass
    
    return best_a, best_score


def structured_init(n, rng, method=0):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if method == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif method == 1:
        a = np.sin(np.pi * x) ** 2
    elif method == 2:
        a = np.exp(-4 * x * (1 - x))
    elif method == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif method == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif method == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif method == 6:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif method == 7:
        # Bi-modal
        a = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternating, adaptive LSE gradient descent,
    top-k mirror descent, peak-shaving CD, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, method=2)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP on parent ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=200000, active_frac=0.7)
        local_update(a_lp, "nlp_1")

    # === Phase 2: Adaptive LSE gradient descent ===
    print(f"\n=== Phase 2: LSE Grad Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        # Adaptive: start with higher T, cool based on improvement
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  {len(T_stages)} temperature stages, budget={t_alloc:.0f}s")
        a_lse, _ = lse_gradient_descent(best_seq, t_alloc, T_stages, top_k=15, lr=1.0)
        local_update(a_lse, "lse_grad")

    # === Phase 3: Top-k Mirror Descent ===
    print(f"\n=== Phase 3: Top-k Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_md, _ = topk_mirror_descent(best_seq, t_alloc, top_k=12)
        local_update(a_md, "topk_mirror")

    # === Phase 4: Second Nesterov LP round ===
    print(f"\n=== Phase 4: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=150000, active_frac=0.6)
        local_update(a_lp2, "nlp_2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration (score={best_score:.8f}) ===")
    target_lengths = [1000, 1100, 1400, 1600]
    # Remove current length
    target_lengths = [nl for nl in target_lengths if nl != len(best_seq)]
    
    for trial_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_sub = min(rem * 0.20, 160.0)
        if t_sub < 40:
            break
        
        print(f"  Trying n={trial_n} (budget={t_sub:.0f}s)")
        init_n = interpolate_sequence(best_seq, trial_n)
        
        # Quick LP
        t_lp = t_sub * 0.6
        a_n, _ = nesterov_lp_refine(init_n, t_lp, solver, n_max_iters=100000, active_frac=0.7)
        local_update(a_n, f"lp_n{trial_n}")
        
        # Quick LSE if time remains
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == trial_n:
            T_short = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_lse_n = lse_gradient_descent(best_seq, min(rem, 80.0), T_short, top_k=10, lr=0.8)
            local_update(a_lse_n, f"lse_n{trial_n}")

    # === Phase 6: Peak shaving CD ===
    print(f"\n=== Phase 6: Peak Shaving CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_ps, _ = peak_shaving_cd(best_seq, t_alloc, top_k=8)
        local_update(a_ps, "peak_shave")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 100.0)
    if t_alloc > 20:
        a_fin, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=100000, active_frac=0.8)
        local_update(a_fin, "final_lp")

    # === Phase 8: Final LSE polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        T_final = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin2 = lse_gradient_descent(best_seq, min(rem * 0.7, 60.0), T_final, top_k=10, lr=0.5)
        local_update(a_fin2, "final_lse")

    # === Phase 9: Ultimate mirror descent polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin3, _ = topk_mirror_descent(best_seq, min(rem * 0.6, 30.0), top_k=8)
        local_update(a_fin3, "final_mirror")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
