import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Returns (b, M, dual_weights)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver]
    for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    
    for s in solvers_to_try:
        try:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Bidirectional LP alternation with adaptive Nesterov averaging
2. Sparse convolution matrix construction for fast LP solves
3. FFT-based LogSumExp L-BFGS with aggressive temperature schedule
4. Multi-length exploration with smooth interpolation
5. Active-set guided subgradient for final polishing
Key improvements: sparse LP formulation, better convergence schedule,
targeted temperature annealing, and focused time allocation on the best n.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (sum-normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = lil_matrix((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T.tocsr()


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver]
    for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)

    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def bidirectional_lp(a0, time_limit, solver, verbose=True):
    """Bidirectional LP alternation with adaptive step-size Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()

    iteration = 0

        if b_new is not None:

            b = normalize((1.0 - rate) * b + rate * b_new)

        if a_new is not None:

def logsumexp_objective_and_grad(z, T_temp, n_iters_cache=None):
    """Compute LogSumExp objective and gradient in log-space parameterization.
    z is the log-parameterization: a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    b = fast_convolve(a)
    b_shift = b / T_temp
    b_max = b_shift.max()
    logsumexp_val = b_max + np.log(np.exp(b_shift - b_max).sum())
    obj = 2.0 * n * T_temp * logsumexp_val

    # Gradient: sum_k w[k] * 2 * (a[k-m] + a[m-k]) for each m
    w = np.exp(b_shift - b_max)
    w = w / w.sum()
    # Cross-correlation: grad_a[m] = 2 * sum_k w[k] * a[k-m] (with a[i]=0 outside [0,n))
    # This is conv(w_padded, a) evaluated appropriately
    # sum_k w[k] * a[k-m] for m in [0,n): this is the cross-correlation
    # Equivalent to: (w * a_flipped) reversed, but let's just do it via FFT
    fft_size = 1 << (n + len(w) - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr[m] = sum_k w[k] * a[k-m] = sum_k w[k] * a[k-m]
    # This is the correlation of w and a: corr[m] = sum_k w[k+m] * a[k]... 
    # Actually: sum_k w[k] * a[k-m]. Let's define c[m] = sum_k w[k] * a[k-m].
    # This is the cross-correlation (w * a)(m) in the convolution sense with one flipped.
    # c[m] = (w conv a_reversed)[m + n - 1]... let me just use the direct approach for clarity
    # Actually, c[m] = sum_k w[k] * a[k-m] where a[i]=0 for i outside [0,n-1]
    # = sum_{k} w[k] * a[k-m] = sum_{j=k-m} w[j+m] * a[j] = sum_j a[j] * w[j+m]
    # This is the cross-correlation of a and w at lag m.
    # cross_corr[m] = sum_j a[j] * w[j+m]
    # In terms of convolution: flip w, then convolve with a
    w_flipped = w[::-1]
    full_conv = np.convolve(a, w_flipped, mode='full')
    # cross_corr[m] = full_conv[m + len(w) - 1 - (n-1)] ... this is getting confusing
    # Let me just use the direct FFT approach:
    # c[m] = sum_k w[k] * a[k-m]
    # = sum_k w[k] * a[k-m] (a[i]=0 outside [0,n-1])
    # Pad a to length n, w to length 2n-1
    # c = ifft(fft(w) * conj(fft(a))) ... no that's not right either
    # Let me just use: c[m] = sum_k w[k] * a[(k-m) % 1] ... no
    # Direct computation is O(n * len(w)) = O(n * 2n) = O(n^2) which for n=1200 is 2.9M - fast enough
    c = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        # a[k-m] is non-zero when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            c[m_start:m_end + 1] += w[k] * a[k - (m_start:m_end + 1)]
    grad_a = 2.0 * c
    # Chain rule for softmax: grad_z[m] = a[m] * (grad_a[m] - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, verbose=True):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iter = max(10, int(remaining * 15))

        def func(z_vec):
            o, _ = logsumexp_objective_and_grad(z_vec, T, n)
            return o

        def func_grad(z_vec):
            o, g = logsumexp_objective_and_grad(z_vec, T, n)
            return o, g

        try:
            res = minimize(
                func_grad, z, jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        if verbose:
            print(f"    LSE T={T:.4f}: score={best_score:.10f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=50000):
    """Subgradient method with adaptive step size for final polishing."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Use soft active set: peaks within 0.1% of max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.08)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def multiplicative_update(a0, time_limit, n_steps=100000):
    """Exponential multiplicative weight updates for peak reduction."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    T = 0.1
    eta = 0.5

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Soft weights (peaked distribution over near-max entries)
        b_shift = b / T
        b_max = b_shift.max()
        w = np.exp(b_shift - b_max)
        w /= w.sum()

        # Gradient direction: sum_k w[k] * 2 * a[k-m]
        g = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[k] * a[k - idx]

        # Multiplicative update
        a_trial = a * np.exp(-eta * g)
        a_trial = a_trial / a_trial.sum()

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            # Slowly decrease T
            T = max(0.001, T * 0.999)
        else:
            eta = max(0.01, eta * 0.95)

        if step % 10000 == 0 and step > 0:
            print(f"    MultUpd step={step} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Bidirectional LP alternation (main workhorse)
    Phase 2: LogSumExp L-BFGS polishing
    Phase 3: Subgradient + multiplicative update refinement
    Phase 4: Multi-n exploration
    Phase 5: Final intensive polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s, deadline in {budget_s - 30:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Bidirectional LP alternation (main) ===
    print(f"\n=== Phase 1: Bidirectional LP (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = bidirectional_lp(best_seq, t1, solver)
        try_update(a_lp, "bidi_lp_main")

    # Second LP run from slightly perturbed start
    rem = deadline - time.monotonic()
    t1b = min(rem * 0.10, 120.0)
    if t1b > 20:
        perturbed = normalize(best_seq + rng.standard_normal(len(best_seq)) * 0.01)
        a_lp2, sc_lp2 = bidirectional_lp(perturbed, t1b, solver)
        try_update(a_lp2, "bidi_lp_perturb")

    # === Phase 2: LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 20:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        print(f"  T_stages: {T_stages}, budget={t2:.0f}s")
        a_lse, sc_lse = logsumexp_lbgfs(best_seq, t2, T_stages=T_stages)
        try_update(a_lse, "logsumexp")

    # === Phase 3: Subgradient + Multiplicative update ===
    print(f"\n=== Phase 3: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 100.0)
    if t3 > 10:
        a_sg, sc_sg = subgradient_polish(best_seq, t3, n_steps=30000)
        try_update(a_sg, "subgrad")

    rem = deadline - time.monotonic()
    t3b = min(rem * 0.05, 50.0)
    if t3b > 10:
        a_mu, sc_mu = multiplicative_update(best_seq, t3b, n_steps=50000)
        try_update(a_mu, "mult_upd")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = n_best
        for mult in [0.8, 0.9, 1.1, 1.2, 1.3, 1.5]:
            cand_n = int(base_n * mult)
            if 400 <= cand_n <= 5000 and cand_n not in n_candidates and cand_n != base_n:
                n_candidates.append(cand_n)
        # Also try some round numbers
        for rn in [800, 1000, 1500, 2000, 2500, 3000]:
            if 400 <= rn <= 5000 and rn not in n_candidates and rn != base_n:
                n_candidates.append(rn)

        n_candidates.sort(key=lambda x: abs(x - base_n))
        n_tried = 0
        for cand_n in n_candidates[:8]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.15, 60.0)
            if t_sub < 15:
                break
            n_tried += 1
            print(f"  Trying n={cand_n} (#{n_tried}), budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP for this n
            a_cand, _ = bidirectional_lp(init_cand, t_sub * 0.6, solver, verbose=False)
            try_update(a_cand, f"lp_n{cand_n}")
            # Quick LSE
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cand2, _ = logsumexp_lbgfs(a_cand, min(rem * 0.3, 25.0),
                                              T_stages=[5.0, 2.0, 1.0, 0.5, 0.1, 0.05],
                                              verbose=False)
                try_update(a_cand2, f"lse_n{cand_n}")

    # === Phase 5: Final intensive polish on best ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.10f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # LP polish
        t_fin1 = min(rem * 0.25, 80.0)
        if t_fin1 > 15:
            a_fin_lp, _ = bidirectional_lp(best_seq, t_fin1, solver, verbose=False)
            try_update(a_fin_lp, "final_lp")

        # Very low T LSE
        rem = deadline - time.monotonic()
        t_fin2 = min(rem * 0.30, 80.0)
        if t_fin2 > 10:
            a_fin_lse, _ = logsumexp_lbgfs(best_seq, t_fin2,
                                           T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                                           verbose=True)
            try_update(a_fin_lse, "final_lse")

        # Subgradient + multiplicative
        rem = deadline - time.monotonic()
        t_fin3 = min(rem * 0.20, 40.0)
        if t_fin3 > 5:
            a_fin_sg, _ = subgradient_polish(best_seq, t_fin3, n_steps=20000)
            try_update(a_fin_sg, "final_sg")

        rem = deadline - time.monotonic()
        t_fin4 = min(rem * 0.15, 30.0)
        if t_fin4 > 5:
            a_fin_mu, _ = multiplicative_update(best_seq, t_fin4, n_steps=30000)
            try_update(a_fin_mu, "final_mu")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
