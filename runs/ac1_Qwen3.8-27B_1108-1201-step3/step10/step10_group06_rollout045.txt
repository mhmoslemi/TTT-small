def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

    if ns < 0.01:
        return 1e10
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def lp_fictitious_play_v2(a0, time_limit, solver=None, verbose=False):
    """Improved LP FP with diminishing step sizes and time averaging."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    # Time average (running sum / count)
    a_sum = a.copy()
    avg_count = 1

    # Exponential moving average
    a_ema = a.copy()
    ema_alpha = 0.05

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Diminishing step size: t_k = alpha / (k + beta)^gamma
        beta = 50.0
        gamma = 0.7
        alpha = 2.0
        t_k = alpha / (iters + beta) ** gamma
        t_k = min(t_k, 0.5)  # Cap at 0.5

        # Try multiple step sizes and pick the best
        candidates = []
        for trial_t in [t_k, t_k * 0.5, t_k * 2.0, 0.05, 0.1, 0.2, 0.3, 0.5]:
            trial_t = max(0.01, min(trial_t, 1.0))
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            candidates.append((sc, cand))

        # Pick best candidate
        candidates.sort(key=lambda x: x[0])
        best_sc, best_cand = candidates[0]

        if best_sc < best_score - 1e-12:
            best_score = best_sc
            best_a = best_cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = best_cand

        # Update averages
        a_sum += a
        avg_count += 1
        a_ema = (1.0 - ema_alpha) * a_ema + ema_alpha * a

        # Check averages for improvement
        a_avg = a_sum / avg_count
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
            no_improve = 0

        a_ema = np.maximum(a_ema, 0)
        a_ema = a_ema / a_ema.sum()
        sc_ema = compute_score(a_ema)
        if sc_ema < best_score:
            best_score = sc_ema
            best_a = a_ema.copy()
            no_improve = 0

        if verbose and iters % 10 == 0:
            print(f"    FP-v2 n={n} it={iters}: best={best_score:.8f} t={t_k:.4f}")

        if no_improve > 50:
            # Reset EMA with more aggressive mixing
            a_ema = 0.5 * a_ema + 0.5 * b
            a_ema = np.maximum(a_ema, 0)
            a_ema = a_ema / a_ema.sum()
            no_improve = 0

    return best_a, best_score, iters

def basin_hopping_structured(a0, time_limit, rng, verbose=False):
    """Basin hopping with structured perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        perturb_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 90.0)

        pert_type = perturb_count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Peak-shift: reduce values at high-convolution positions
            conv_p = fast_convolve(a)
            top_k = min(15, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.03 * rng.random())
        elif pert_type == 1:
            # Gaussian noise
            sigma = 0.01 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            # Block swap
            blen = max(4, rng.integers(n // 10, n // 4))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        elif pert_type == 3:
            # Shift and mix
            shift = rng.integers(1, n // 5)
            shifted = np.roll(a, shift)
            mix = 0.3 + 0.4 * rng.random()
            perturbed = (1.0 - mix) * a + mix * shifted
        else:
            # Taper perturbation: modify edges
            taper = rng.uniform(0.5, 1.5)
            for i in range(min(20, n // 5)):
                perturbed[i] *= taper
                perturbed[n - 1 - i] *= 1.0 / taper

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh_{perturb_count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {perturb_count}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.2, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
