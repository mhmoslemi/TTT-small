"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response (main workhorse, 55% time)
2. Direct SQP (Sequential Quadratic Programming) refinement (20% time)
3. LogSumExp-smoothed Adam optimization (15% time)
4. Active-set equalization and diverse restarts (10% time)
Key improvements: correct Nesterov weights, constraint-reduced LP for speed,
SQP for direct non-convex optimization, and active-set exploitation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast aperiodic convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_constraints(a_fixed, n_constraints=None):
    """Build the convolution matrix T(a) for LP. Optionally reduce constraints."""
    n = len(a_fixed)
    size = 2 * n - 1
    if n_constraints is not None and n_constraints < size:
        # Only keep the most important constraints
        b_self = fast_convolve(a_fixed)
        idx = np.argsort(b_self)[::-1][:n_constraints]
        idx = np.sort(idx)
        T = np.zeros((n_constraints, n))
        for ii, k in enumerate(idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            T[ii, js:je+1] = a_fixed[k - je:k - js + 1][::-1]
        return T
    else:
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j + n, j] = a_fixed
        return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None

    global _lp_solver
    if _lp_solver is not None:

    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, n_constraints=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_constraints(a_fixed, n_constraints)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver == cp.SCS:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-7)
        elif solver == cp.ECOS:
            prob.solve(solver=cp.ECOS, verbose=False)
        else:
            prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver=None, n_constraints=None, max_iters=100000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_b = fast_convolve(a)
    best_score = 2.0 * n * best_b.max()
    t0 = time.monotonic()
    t_k = 1.0
    a_prev = a.copy()
    stall_count = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response to y
        b_new, M_lp = solve_lp_best_response(y, solver, n_constraints)
        if b_new is None:
            # Fallback: use subgradient step
            b_conv = fast_convolve(y)
            k_star = int(np.argmax(b_conv))
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            grad = np.zeros(n)
            grad[idx] = y[k_star - idx]
            grad[idx] += y[idx - js + (k_star - je):(k_star - js + 1)] if False else 0
            # Simpler: just use the peak location
            a_new = y.copy()
            a_new[k_star] -= 0.01
            a_new[(k_star + n//2) % n] += 0.01
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            a = (t_k / t_new) * a + (1.0 / t_new) * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a + (t_k / t_new) * (a - a_prev)
            y = np.maximum(y, 0)
            y = y / y.sum()
            a_prev = a
            t_k = t_new
            continue

        # Nesterov update with correct weights
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        a_candidate = (t_k / t_new) * a + (1.0 / t_new) * b_new
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = a_candidate.sum()
        if s_cand < 1e-15:
            a_candidate = y.copy()
            s_cand = 1.0
        a_candidate = a_candidate / s_cand

        # Update y (extrapolate)
        y_candidate = a_candidate + (t_k / t_new) * (a_candidate - a_prev)
        y_candidate = np.maximum(y_candidate, 0)
        s_y = y_candidate.sum()
        if s_y < 1e-15:
            y_candidate = a_candidate.copy()
        else:
            y_candidate = y_candidate / s_y

        a = a_candidate
        y = y_candidate
        a_prev = a
        t_k = t_new

        # Track best
        b_a = fast_convolve(a)
        score_a = 2.0 * n * b_a.max()
        b_y = fast_convolve(y)
        score_y = 2.0 * n * b_y.max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0 and it > 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Random restart perturbation
                rng = np.random.default_rng(it * 17 + 41)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, t={t_k:.1f}")

def sqp_refine(a0, time_limit, max_iters=5000):
    """SQP: linearize quadratic constraints, solve QP at each step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: top peaks within 10% of max
        threshold = max_b * 0.9
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        active = active[:20]  # Limit to 20 active constraints

        # Build QP: min t s.t. linearized constraints, sum(da)=0, a+da>=0
        # Linearization: (conv(a+da, a+da))_k ≈ b_k + 2*sum_j a_{k-j} da_j (for active k)
        # Actually: d/d da_j [(conv(a+da,a+da))_k] = a_{k-j} + a_{j-k} (symmetric part)
        # Since conv is symmetric: = 2 * a_{k-j} (when j and k-j both valid, counting both)
        # More carefully: (conv(a+da,a+da))_k = sum_j (a_j+da_j)(a_{k-j}+da_{k-j})
        # ≈ sum_j a_j*a_{k-j} + sum_j [a_j*da_{k-j} + da_j*a_{k-j}]
        # = b_k + 2*sum_j a_j*da_{k-j} (by symmetry, the two sums are equal)
        # = b_k + 2*sum_j da_j * a_{k-j}

        # QP variables: da (length n), t (scalar)
        # Objective: minimize t
        # Constraints:
        #   For each active k: b_k + 2*sum_j da_j * a_{k-j} <= t
        #   sum(da) = 0
        #   a_j + da_j >= 0  => da_j >= -a_j

        if len(active) == 1:
            # Single constraint: QP reduces to minimizing a quadratic
            k = active[0]
            # minimize t s.t. b_k + 2*(a_shifted @ da) <= t, sum(da)=0, da >= -a
            # => minimize b_k + 2*(a_shifted @ da) s.t. sum(da)=0, da >= -a
            # where a_shifted[j] = a[k-j] (if valid)
            a_shifted = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            a_shifted[js:je+1] = a[k - je:k - js + 1][::-1]

            da_var = cp.Variable(n)
            constraints = [cp.sum(da_var) == 0, da_var >= -a]
            prob = cp.Problem(cp.Minimize(2.0 * a_shifted @ da_var), constraints)
            try:
                prob.solve(solver=cp.SCS, verbose=False, max_iters=3000, eps=1e-7)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break
            da = np.asarray(da_var.value, dtype=np.float64)
            a_new = a + da
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new = a_new / sn
        else:
            # Multiple active constraints: use min t formulation
            da_var = cp.Variable(n)
            t_var = cp.Variable()
            constraints = [cp.sum(da_var) == 0, da_var >= -a]
            for k in active:
                a_shifted = np.zeros(n)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                a_shifted[js:je+1] = a[k - je:k - js + 1][::-1]
                constraints.append(b[k] + 2.0 * (a_shifted @ da_var) <= t_var)
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            try:
                prob.solve(solver=cp.SCS, verbose=False, max_iters=3000, eps=1e-7)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break
            da = np.asarray(da_var.value, dtype=np.float64)
            a_new = a + da
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new = a_new / sn

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            da = np.asarray(da_var.value, dtype=np.float64) * 0.5
            a_new = a + da
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                score_new = 2.0 * n * fast_convolve(a_new).max()
                if score_new < best_score - 1e-12:
                    best_score = score_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    break  # Converged

        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    SQP iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lse_adam(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15, lr_base=0.5):
    """LogSumExp-smoothed optimization with Adam optimizer on logits."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = softmax_from_z(z).copy()
    best_score = 2.0 * n * fast_convolve(best_a).max()

    def compute_grad_lse(z_in, T_val):
        """Grad of LSE objective w.r.t. logits."""
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = np.clip(Tb.max(), 0, 700)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # Grad w.r.t. a: sum_k w_k * d(conv(a,a))_k / da
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        # Chain rule through softmax: grad_z = a * (grad_a - dot(grad_a, a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z

    def compute_lse_value(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = np.clip(Tb.max(), 0, 700)
        exp_Tb = np.exp(Tb - max_Tb)
        return (2.0 * n / T_val) * (max_Tb + np.log(exp_Tb.sum()))

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_iters = max(10, int(remaining * 30))
        lr = lr_base * min(1.0, 50.0 / T_val)

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad = compute_grad_lse(z, T_val)
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-12:
                break

            # Adam update
            t_adam = it + 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Periodically check score
            if (it + 1) % 50 == 0:
                a_curr = softmax_from_z(z)
                sc = 2.0 * n * fast_convolve(a_curr).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_curr.copy()

        a_curr = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_curr).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_curr.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE-Adam stage={stage} T={T_val:.2f} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def subgradient_descent(a0, time_limit, n_iters=50000):
    """Direct subgradient descent on the non-convex max of quadratic forms."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: peaks within 5% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) > 10:
            active = np.array(b.argsort()[::-1][:10])

        # Compute subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(active)

        # Polyak step or fixed decreasing step
        lr = 0.1 / np.sqrt(it + 1)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2)
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.4 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    else:
        x = np.linspace(-1, 1, n)
        a = (1 - x**2) * (1 + 0.15 * np.cos(6 * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def optimize_at_length(a_init, time_budget, solver):
    """Run multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"      NEW BEST [{label}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    # Phase 1: Nesterov LP (main workhorse) - 55% of budget
    t_lp = time_budget * 0.55
    if t_lp > 10:
        a_lp, _ = nesterov_lp_optimize(best_a, t_lp, solver=solver,
                                        n_constraints=min(500, 2*n-1),
                                        max_iters=200000)
        try_update(a_lp, "nesterov_lp")

    # Phase 2: SQP refinement - 25% of budget
    rem = time_budget - (time.monotonic() - t0)
    t_sqp = min(rem * 0.5, time_budget * 0.25)
    if t_sqp > 5:
        a_sqp, _ = sqp_refine(best_a, t_sqp, max_iters=3000)
        try_update(a_sqp, "sqp")

    # Phase 3: LSE Adam - 15% of budget
    rem = time_budget - (time.monotonic() - t0)
    t_lse = min(rem * 0.6, time_budget * 0.15)
    if t_lse > 5:
        a_lse = lse_adam(best_a, t_lse, T_start=100.0, T_end=0.1, n_stages=10, lr_base=0.8)
        try_update(a_lse, "lse_adam")

    # Phase 4: Subgradient + final SQP - remaining
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        t_sg = rem * 0.5
        if t_sg > 3:
            a_sg, _ = subgradient_descent(best_a, t_sg, n_iters=20000)
            try_update(a_sg, "subgrad")
        rem2 = time_budget - (time.monotonic() - t0)
        if rem2 > 3:
            a_sqp2, _ = sqp_refine(best_a, rem2, max_iters=1000)
            try_update(a_sqp2, "sqp_final")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    global time_limit_global
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    time_limit_global = start_time + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s, internal limit: {time_limit_global - start_time:.0f}s")

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Optimize at base length (most time)
    print(f"\n=== Phase 1: Optimize at n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t1 = min(rem * 0.50, 800.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t1, solver)
        try_update_global(a_out, f"opt_n{n_base}")

    # Phase 2: Try longer lengths
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer_lengths = []
    for mult in [1.2, 1.5]:
        ln = int(n_base * mult)
        if ln > n_base and ln <= 2500:
            longer_lengths.append(ln)
    longer_lengths = sorted(set(longer_lengths))[:2]

    for ln in longer_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.30, 350.0)
        if t_alloc < 60:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"  Interpolation improved! score={sc_init:.6f}")

        a_out, _ = optimize_at_length(init_long, t_alloc, solver)
        try_update_global(a_out, f"opt_long_n{ln}")

    # Phase 3: Try a shorter length (sometimes helps)
    print(f"\n=== Phase 3: Shorter length (best={best_score:.6f}) ===")
    sn = max(400, int(n_base * 0.6))
    rem = time_limit_global - time.monotonic()
    if rem > 100:
        t_alloc = min(rem * 0.15, 150.0)
        if t_alloc > 30:
            print(f"  Trying n={sn}, budget={t_alloc:.0f}s")
            init_short = interpolate_sequence(best_seq, sn)
            a_out, _ = optimize_at_length(init_short, t_alloc, solver)
            try_update_global(a_out, f"opt_short_n{sn}")

    # Phase 4: Final intensive polish on the best
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 30:
        # More LP iterations
        t_lp = min(rem * 0.35, 120.0)
        if t_lp > 10:
            print(f"  Final LP: {t_lp:.0f}s")
            a_fin, _ = nesterov_lp_optimize(best_seq, t_lp, solver=solver,
                                             n_constraints=min(400, 2*len(best_seq)-1),
                                             max_iters=50000)
            try_update_global(a_fin, "final_lp")

        # SQP
        rem = time_limit_global - time.monotonic()
        if rem > 15:
            t_sqp = min(rem * 0.3, 60.0)
            if t_sqp > 5:
                print(f"  Final SQP: {t_sqp:.0f}s")
                a_sqp, _ = sqp_refine(best_seq, t_sqp, max_iters=2000)
                try_update_global(a_sqp, "final_sqp")

        # LSE Adam fine-tune
        rem = time_limit_global - time.monotonic()
        if rem > 15:
            t_lse = min(rem * 0.3, 50.0)
            if t_lse > 5:
                print(f"  Final LSE-Adam: {t_lse:.0f}s")
                a_lse = lse_adam(best_seq, t_lse, T_start=10.0, T_end=0.05,
                                  n_stages=8, lr_base=0.3)
                try_update_global(a_lse, "final_lse")

        # One more SQP
        rem = time_limit_global - time.monotonic()
        if rem > 10:
            t_sqp2 = min(rem * 0.4, 30.0)
            if t_sqp2 > 5:
                a_sqp2, _ = sqp_refine(best_seq, t_sqp2, max_iters=1000)
                try_update_global(a_sqp2, "final_sqp2")

        # Final LP squeeze
        rem = time_limit_global - time.monotonic()
        if rem > 10:
            t_lp2 = min(rem * 0.4, 30.0)
            if t_lp2 > 5:
                a_lp2, _ = nesterov_lp_optimize(best_seq, t_lp2, solver=solver,
                                                 n_constraints=min(300, 2*len(best_seq)-1),
                                                 max_iters=20000)
                try_update_global(a_lp2, "final_lp2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL VERIFIED SCORE: {best_score:.6f} (n={len(best_seq)})")
        else:
            print(f"\nWARNING: Final score is not finite! Using last known: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = evaluate_safe(best_seq)

    print(f"\n=== RESULT: score={best_score:.6f}, n={len(best_seq)} ===")
    return [float(x) for x in best_seq]
