def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        if delta < 0:
            a = a_trial
            n_accepted += 1
        else:
            accept_prob = math.exp(-delta / max(T, 1e-10))
            if rng.random() < accept_prob:
                a = a_trial
                n_accepted += 1
            else:
                n_rejected += 1

        if it % 200 == 0:
            # Re-optimize with quick CD
            a_opt, s_opt, _ = coord_descent_golden(a, 5.0, label=f"sa{label}_it{it}")
            if s_opt < best_score - 1e-14:
                best_score = s_opt
                best_a = a_opt.copy()
            a = a_opt

            elapsed = time.monotonic() - t0
            print(f"    SA-CD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.4f}")

    return best_a, best_score

def nesterov_smoothed_md(a0, time_limit, beta_init=20.0, lr_init=0.008, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y_proj)
        max_c = float(conv_y.max())
        if max_c < 1e-30:
            break
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[int(conv_y.argmax())] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient computation
        grad = np.zeros(n)
        active_k = np.where(weights > 1e-8)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            lr *= 0.85
            beta *= 1.5
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 50 == 0:
            beta *= 1.02

        gamma = (1.0 + math.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def generate_init_tapered(n):
    """Raised cosine taper initialization."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def generate_init_spectral(n, freq=1):
    """Spectral initialization with a single frequency modulation."""
    x = np.arange(n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def generate_init_bimodal(n):
    """Bimodal (two peaks) initialization."""
    x = np.arange(n)
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * 0.15
    a = np.exp(-0.5 * ((x - center1) / width) ** 2) + np.exp(-0.5 * ((x - center2) / width) ** 2)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def fine_polish(a0, time_limit, label=""):
    """Very fine polishing with narrow search ranges."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1].copy()
            coeffs = 2.0 * a[(k_start - i):(k_end - i + 1)]
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Very narrow search
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 1.001, 1.002, 1.005, 1.01]:
                delta = ai * (factor - 1.0)
                new_max = compute_1d_val(affected, coeffs, quad_idx, has_quad, unaff_max, delta)
                if new_max < M - 1e-16:
                    a[i] = max(ai + delta, 0.0)
                    conv[k_start:k_end + 1] += delta * coeffs
                    if has_quad:
                        conv[2 * i] += delta * delta
                    M = new_max
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
                    break

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization for minimum autocorrelation peak:
    1. Diverse initializations with quick optimization
    2. LP fictitious play (main workhorse)
    3. Golden-section coordinate descent with priority ordering
    4. Simulated annealing + CD
    5. Nesterov smoothed mirror descent
    6. Multi-length exploration
    7. Final intensive polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    
    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback init: n={n_init}, score={best_score:.6f}")

    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Diverse Inits (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    
    inits = {
        "parent": best_seq.copy(),
        "uniform": np.ones(n_base) / n_base,
        "tapered": generate_init_tapered(n_base),
        "spectral_f1": generate_init_spectral(n_base, 1),
        "spectral_f3": generate_init_spectral(n_base, 3),
        "bimodal": generate_init_bimodal(n_base),
    }

    # Quick CD on each (30s each)
    for name, init in inits.items():
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        t_budget = min(30.0, rem * 0.15)
        if t_budget < 5:
            continue
        a_opt, s_opt, _ = coord_descent_golden(init, t_budget, label=f"init_{name}")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"init_{name}")

    print(f"  After Phase 1: best={best_score:.6f}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    solver = None
    try:
        solver = get_lp_solver()
        print(f"  LP solver: {solver}")
    except Exception:
        print("  No LP solver available, skipping Phase 2")

    if solver is not None:
        rem = deadline - time.monotonic() - 500
        t2 = min(max(rem * 0.3, 100), 400.0)
        print(f"  Budget: {t2:.0f}s, n={len(best_seq)}")
        a2, s2, it2 = lp_fictitious_play(best_seq, t2, solver=solver, label="p2")
        best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_lpfp")

    # === Phase 3: Golden-section CD ===
    print(f"\n=== Phase 3: Golden CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    t3 = min(max(rem * 0.3, 60), 350.0)
    print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
    a3, s3, sw3 = coord_descent_golden(best_seq, t3, verbose=True, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_gcd")

    # === Phase 4: Simulated Annealing + CD ===
    print(f"\n=== Phase 4: SA-CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    t4 = min(max(rem * 0.25, 40), 250.0)
    print(f"  Budget: {t4:.0f}s")
    a4, s4 = simulated_annealing_cd(best_seq, t4, rng, label="p4")
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_sa")

    # === Phase 5: Nesterov Smoothed MD ===
    print(f"\n=== Phase 5: Nesterov MD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 250
    t5 = min(max(rem * 0.2, 30), 150.0)
    print(f"  Budget: {t5:.0f}s")
    a5, s5 = nesterov_smoothed_md(best_seq, t5, beta_init=25.0, lr_init=0.006, label="p5")
    best_seq, best_score = check_and_update(a5, best_seq, best_score, "p5_nmd")

    # === Phase 6: CD + Perturbation loop ===
    print(f"\n=== Phase 6: CD + Perturb (best={best_score:.6f}) ===")
    for rnd in range(5):
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t6 = min(rem * 0.2, 80.0)
        a6, s6, _ = coord_descent_golden(best_seq, t6, label=f"p6r{rnd}")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_cd_r{rnd}")

        if rnd < 4:
            # Perturb
            if rnd % 2 == 0:
                # Tapered perturbation
                perturbed = best_seq * (1.0 + 0.02 * rng.standard_normal(len(best_seq)))
                perturbed = np.maximum(perturbed, 0)
                sp = perturbed.sum()
                if sp > 1e-15:
                    perturbed = perturbed / sp
            else:
                # Restructure: zero out random 10% and redistribute
                perturbed = best_seq.copy()
                n_zero = max(5, len(best_seq) // 10)
                zero_idx = rng.choice(len(best_seq), size=n_zero, replace=False)
                removed_mass = perturbed[zero_idx].sum()
                perturbed[zero_idx] = 0
                nonzero = perturbed > 0
                if nonzero.sum() > 0:
                    perturbed[nonzero] += removed_mass / nonzero.sum()
            
            # Quick optimize perturbed
            t6b = min(rem * 0.1, 30.0)
            if t6b > 5:
                a6b, s6b, _ = coord_descent_golden(perturbed, t6b, label=f"p6p_r{rnd}")
                best_seq, best_score = check_and_update(a6b, best_seq, best_score, f"p6_pert_r{rnd}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for alt in [768, 1000, 1280, 1536, 2048]:
        if alt != n_best:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:3]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        t7 = min(rem * 0.3, 60.0)
        init_alt = interpolate_seq(best_seq, n_alt)
        a7, s7, _ = coord_descent_golden(init_alt, t7, label=f"p7_n{n_alt}")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, f"p7_n{n_alt}")

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    # CD pass
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t8 = min(rem * 0.4, 100.0)
        a8, s8, _ = coord_descent_golden(best_seq, t8, label="p8a")
        best_seq, best_score = check_and_update(a8, best_seq, best_score, "p8_gcd")

    # Fine polish
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t8b = min(rem * 0.5, 60.0)
        a8b, s8b = fine_polish(best_seq, t8b, label="p8b")
        best_seq, best_score = check_and_update(a8b, best_seq, best_score, "p8_fine")

    # One more CD
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a8c, s8c, _ = coord_descent_golden(best_seq, rem * 0.7, label="p8c")
        best_seq, best_score = check_and_update(a8c, best_seq, best_score, "p8c")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
