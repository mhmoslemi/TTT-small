def fast_conv(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_w_a(w, a):
    """Compute g[j] = sum_k w[k+j]*a[k] for j=0..n-1. w has length 2n-1, a has length n."""
    return correlate(w, a, mode='valid', method='fft')


def logsumexp_gradient(a, beta):
    """
    Compute gradient of logsumexp(beta*conv(a,a)) w.r.t. z where a=softmax(z).
    Returns (grad_z, c_max) where grad_z has length n.
    grad_z = 2*beta*a*(cross_corr(w,a) - S), w=softmax(beta*c), S=dot(w,c)
    """
    n = len(a)
    c = fast_conv(a)
    beta_c = beta * c
    shifted = beta_c - beta_c.max()
    exp_bc = np.exp(shifted)
    w = exp_bc / exp_bc.sum()
    S = np.dot(w, c)
    cc = cross_corr_w_a(w, a)
    grad_z = 2.0 * beta * a * (cc - S)
    return grad_z, c.max()


def score_of_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_conv(a).max()


def mirror_descent_lse(a0, time_limit, beta_start=0.5, beta_end=500.0, verbose=True):

def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = convolve(a_fixed, b). Shape (2n-1, n)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        cols = a_fixed[max(0, j - (n - 1)):min(n, j + 1)]
        # T[k,j] = a_fixed[k-j] if 0<=k-j<n
        # For column j: rows k where 0<=k-j<n, i.e., j<=k<=j+n-1
        lo = max(0, j)
        hi = min(size - 1, j + n - 1)
        T[lo:hi + 1, j] = a_fixed[lo - j:hi + 1 - j]
    return T


_lp_solver_name = None


def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def solve_lp_br(a_fixed, solver=None):
    """
    Solve LP best-response: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        prob.status = 'error'

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_fixed_point(a0, time_limit, solver=None, verbose=True):

    if s < 1e-15:

    else:
        a = a / s

    best_score = score_of_a(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP best response
        b_br, M_br = solve_lp_br(a, solver)
        if b_br is None:
            break

        it += 1

        # Try convex combination with decreasing t
        base_t = max(0.1, 2.0 / (it + 1.0))
        improved = False

        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            t = base_t * frac
            a_try = (1 - t) * a + t * b_br
            a_try = np.maximum(a_try, 0)
            ss = a_try.sum()
            if ss < 1e-15:
                continue
            a_try = a_try / ss
            sc = score_of_a(a_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

            # Just use the LP solution as a check

            if sc_br < best_score - 1e-12:
                best_score = sc_br
                best_a = b_br.copy()
                a = b_br.copy()
            else:

                a = 0.95 * a + 0.05 * b_br
                a = np.maximum(a, 0)
                a = a / a.sum()

def subgradient_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent using subgradient of the exact max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)

    best_a = a.copy()
    best_score = score_of_a(a)
    t0 = time.monotonic()
    eta = 1.0
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient: average over active set
        c = fast_conv(a)
        c_max = c.max()
        # Active set: entries within 1% of max
        threshold = c_max * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient: average of db[k]/da for k in active
        # db[k]/da[j] = 2*a[k-j] if 0<=k-j<n else 0
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += 2.0 * a[k - js:k - js + (je - js + 1)]
        g /= len(active)

        if not np.all(np.isfinite(g)):
            break

        # Mirror descent step
        for bt in range(6):
            z_new = z - eta * g
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            sc = score_of_a(a_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                z = z_new
                stall = 0
                eta = min(eta * 1.03, 10.0)
                break
            eta *= 0.5
        else:
            stall += 1
            if stall > 100:
                rng = np.random.default_rng(it + 99)
                z += rng.normal(0, 0.05, n)
                a = np.exp(z - z.max())
                a = a / a.sum()
                z = np.log(a)
                stall = 0
                eta = 1.0

        it += 1
        if verbose and it % 20000 == 0:
            print(f"      SubMD: t={time.monotonic()-t0:.0f}s, score={best_score:.6f}, |active|={len(active)}")

    return best_a, best_score


def init_interpolated(parent, n):
    """Interpolate parent sequence to length n (normalized to sum=1)."""
    old_n = len(parent)
    if old_n == n:
        a = parent.copy()
    else:
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, parent)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_perturbed_uniform(n, seed):
    """Uniform with small random perturbation to break symmetry."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_chirp(n, seed):
    """Chirp-like initialization: amplitude envelope with phase variation."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Modulated chirp amplitude
    freq = rng.uniform(2, 8)
    a = np.exp(-2 * x * (1 - x)) * (1 + 0.3 * np.cos(2 * np.pi * freq * x))
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        return np.ones(n) / n
    return a / a.sum()


def init_ramp_asymmetric(n, seed):
    """Asymmetric ramp: breaks the symmetry that makes b[n-1] large."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Asymmetric: peak off-center
    peak = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.1, 0.4)
    a = np.exp(-((x - peak) / width) ** 2)
    # Add asymmetry
    asym = rng.uniform(0.5, 1.5)
    a = a ** (1.0 + asym * (x - 0.5))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def init_two_bump(n, seed):
    """Two-bump initialization: energy at 1/3 and 2/3, low at ends."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    p1 = rng.uniform(0.2, 0.4)
    p2 = 1.0 - p1
    w = rng.uniform(0.05, 0.15)
    a = np.exp(-((x - p1) / w) ** 2) + np.exp(-((x - p2) / w) ** 2)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def evaluate_safe(a):
    """Safely evaluate sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_single_length(a_init, n, time_budget, solver, label=""):
    """
    Run multi-phase optimization at a fixed length.
    Returns (best_a, best_score).
    """
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_of_a(a)
    t_start = time.monotonic()

    def timed_remaining():
        return max(0.0, time_budget - (time.monotonic() - t_start))

    def try_update(a_new, src):
        nonlocal best_a, best_score
        if a_new is None:
            return
        a_new = np.maximum(np.asarray(a_new, dtype=np.float64), 0)
        ss = a_new.sum()
        if ss < 0.01:
            return
        a_new = a_new / ss
        sc = score_of_a(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            print(f"      NEW BEST [{src}] n={n}: {sc:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 1: Mirror descent with LogSumExp (fast, main workhorse)
    rem = timed_remaining()
    t1 = min(rem * 0.40, 300.0)
    if t1 > 10:
        print(f"    Phase 1 (MD-LSE): budget={t1:.0f}s")
        a_out, _ = mirror_descent_lse(best_a, t1, beta_start=0.5, beta_end=300.0, verbose=True)
        try_update(a_out, "md_lse")

    # Phase 2: LP fixed-point
    rem = timed_remaining()
    t2 = min(rem * 0.35, 300.0)
    if t2 > 15:
        print(f"    Phase 2 (LP-FP): budget={t2:.0f}s")
        a_out, _ = lp_fixed_point(best_a, t2, solver=solver, verbose=True)
        try_update(a_out, "lp_fp")

    # Phase 3: Another MD-LSE round (with higher beta_end)
    rem = timed_remaining()
    t3 = min(rem * 0.20, 150.0)
    if t3 > 10:
        print(f"    Phase 3 (MD-LSE-2): budget={t3:.0f}s")
        a_out, _ = mirror_descent_lse(best_a, t3, beta_start=50.0, beta_end=1000.0, verbose=False)
        try_update(a_out, "md_lse2")

    # Phase 4: Subgradient mirror descent on exact max
    rem = timed_remaining()
    t4 = min(rem * 0.10, 60.0)
    if t4 > 5:
        print(f"    Phase 4 (SubMD): budget={t4:.0f}s")
        a_out, _ = subgradient_mirror_descent(best_a, t4, verbose=False)
        try_update(a_out, "submd")

    # Phase 5: Final LP polish
    rem = timed_remaining()
    t5 = rem
    if t5 > 10:
        print(f"    Phase 5 (LP-polish): budget={t5:.0f}s")
        a_out, _ = lp_fixed_point(best_a, t5, solver=solver, verbose=False)
        try_update(a_out, "lp_polish")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Explores multiple initializations and lengths,
    using mirror descent + LP best-response optimization.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    global_deadline = t_global_start + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_global_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={ps:.6f}")
            best_seq = parent.copy()
            best_global_score = ps
    except Exception:
        pass

    def global_try_update(a_new, label):
        nonlocal best_seq, best_global_score
        if a_new is None:
            return
        a_new = np.maximum(np.asarray(a_new, dtype=np.float64), 0)
        ss = a_new.sum()
        if ss < 0.01:
            return
        a_new = a_new / ss
        sc = evaluate_safe(a_new)
        if np.isfinite(sc) and sc < best_global_score:
            best_global_score = sc
            best_seq = a_new.copy()
            print(f"  >>> GLOBAL BEST [{label}] n={len(a_new)}: {sc:.6f}")

    # Determine base length
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 1200
        best_seq = init_perturbed_uniform(n_base, seed)
        best_global_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_global_score:.6f}")

    # === Strategy: Try multiple initializations at base length ===
    # Limit total initializations to manage time
    n_inits = 4
    init_budget_per = min(200.0, (budget_s - 60.0) * 0.5 / n_inits)

    inits = []
    # Init 1: parent (or perturbed uniform)
    if parent is not None:
        inits.append(("parent", init_interpolated(parent, n_base)))
    else:
        inits.append(("uniform_pert", init_perturbed_uniform(n_base, seed)))

    # Init 2: asymmetric
    inits.append(("asym", init_ramp_asymmetric(n_base, seed + 1)))

    # Init 3: chirp
    inits.append(("chirp", init_chirp(n_base, seed + 2)))

    # Init 4: two bump
    inits.append(("twobump", init_two_bump(n_base, seed + 3)))

    print(f"\n=== Base length n={n_base}, {len(inits)} initializations ===")
    print(f"    Init budget each: ~{init_budget_per:.0f}s")

    for idx, (name, a_init) in enumerate(inits):
        rem_global = global_deadline - time.monotonic()
        if rem_global < 60:
            break

        # Use the current best as the init if it's better than this one
        sc_init = score_of_a(a_init)
        if best_seq is not None and score_of_a(best_seq) < sc_init:
            a_init = best_seq.copy()
            name = f"{name}->best"

        t_alloc = min(init_budget_per, rem_global * 0.3)
        if t_alloc < 30:
            t_alloc = min(rem_global * 0.25, 120.0)
        if t_alloc < 15:
            break

        print(f"\n  [{idx+1}/{len(inits)}] Init '{name}', budget={t_alloc:.0f}s")
        a_out, sc_out = optimize_single_length(a_init, n_base, t_alloc, solver, label=name)
        global_try_update(a_out, f"base_{name}")

    # === Strategy: Try longer lengths ===
    print(f"\n=== Longer lengths (best={best_global_score:.6f}, n={len(best_seq)}) ===")
    longer_n = [int(n_base * 1.5), int(n_base * 2.0), int(n_base * 2.5)]
    longer_n = sorted(set([x for x in longer_n if x > n_base and x <= 5000]))[:3]

    for ln in longer_n:
        rem_global = global_deadline - time.monotonic()
        if rem_global < 120:
            break
        t_alloc = min(rem_global * 0.20, 250.0)
        if t_alloc < 60:
            break

        print(f"\n  Longer n={ln}, budget={t_alloc:.0f}s")
        a_long = init_interpolated(best_seq, ln)
        # Add small perturbation
        rng2 = np.random.default_rng(ln)
        a_long = a_long + rng2.normal(0, 0.005, ln)
        a_long = np.maximum(a_long, 0)
        a_long = a_long / a_long.sum()

        a_out, sc_out = optimize_single_length(a_long, ln, t_alloc, solver, label=f"long{ln}")
        global_try_update(a_out, f"long_n{ln}")

    # === Strategy: Try shorter lengths ===
    print(f"\n=== Shorter lengths (best={best_global_score:.6f}) ===")
    shorter_n = [int(n_base * 0.6), int(n_base * 0.4)]
    shorter_n = sorted(set([x for x in shorter_n if x >= 100]))[:2]

    for sn in shorter_n:
        rem_global = global_deadline - time.monotonic()
        if rem_global < 90:
            break
        t_alloc = min(rem_global * 0.15, 150.0)
        if t_alloc < 30:
            break

        print(f"\n  Shorter n={sn}, budget={t_alloc:.0f}s")
        a_short = init_interpolated(best_seq, sn)
        a_out, sc_out = optimize_single_length(a_short, sn, t_alloc, solver, label=f"short{sn}")
        global_try_update(a_out, f"short_n{sn}")

    # === Final polish on the global best ===
    print(f"\n=== Final Polish (best={best_global_score:.6f}, n={len(best_seq)}) ===")
    rem_global = global_deadline - time.monotonic()
    if rem_global > 20:
        # One more round of MD-LSE with very high beta
        t_polish = min(rem_global * 0.4, 120.0)
        if t_polish > 10:
            print(f"  MD-LSE polish: {t_polish:.0f}s")
            a_out, _ = mirror_descent_lse(best_seq, t_polish, beta_start=100.0, beta_end=2000.0, verbose=False)
            global_try_update(a_out, "polish_md")

        # LP polish
        rem_global = global_deadline - time.monotonic()
        t_lp = min(rem_global * 0.4, 100.0)
        if t_lp > 10:
            print(f"  LP polish: {t_lp:.0f}s")
            a_out, _ = lp_fixed_point(best_seq, t_lp, solver=solver, verbose=False)
            global_try_update(a_out, "polish_lp")

        # Subgradient policy
        rem_global = global_deadline - time.monotonic()
        t_sub = min(rem_global * 0.3, 60.0)
        if t_sub > 5:
            print(f"  SubMD polish: {t_sub:.0f}s")
            a_out, _ = subgradient_mirror_descent(best_seq, t_sub, verbose=False)
            global_try_update(a_out, "polish_sub")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_global_score = final_score

    print(f"\n{'='*60}")
    print(f"FINAL: score={best_global_score:.6f}, n={len(best_seq)}")
    print(f"Total time: {time.monotonic()-t_global_start:.0f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
