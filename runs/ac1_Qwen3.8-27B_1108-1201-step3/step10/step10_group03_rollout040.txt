"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a refined multi-phase approach:
(1) Fine-grained LogSumExp L-BFGS with logarithmic temperature annealing,
(2) Active-set subgradient with adaptive step sizes,
(3) Systematic n-search with cubic interpolation,
(4) Peak-shaving via targeted coordinate updates,
(5) Multiple perturbation-restart cycles to escape local optima.
Key improvements over prior: much finer T schedule (40+ levels), FFT-based
gradient computation for speed, and a dedicated peak-shaving phase that
directly reduces the top convolution coefficients.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_lse_grad(a, T):
    """Compute gradient of 2n * LSE(conv(a,a)/T) w.r.t. a on the simplex."""
    n = len(a)
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    
    # grad[m] = (4n/T) * sum_k w[k] * a[k-m] (with bounds)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        m = np.arange(js, je + 1)
        grad[m] += wk * a[k - m]
    return (4.0 * n / T) * grad


def lse_objective_and_grad_z(z, T, n):
    """Objective and gradient in log-parameterization a = softmax(z)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    obj = (2.0 * n / T) * (b_max + np.log(exp_ws.sum()))
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        m = np.arange(js, je + 1)
        grad_a[m] += wk * a[k - m]
    grad_a *= (4.0 * n / T)
    
    # Chain rule: grad_z = a * (grad_a - a^T grad_a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_optimize(a0, time_limit, T_list=None):
    """Multi-stage LSE optimization with L-BFGS."""
    n = len(a0)
    a = normalize(a0)
    a_clip = np.maximum(a, 1e-300)
    z = np.log(a_clip) - np.log(a_clip.sum())
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_list is None:
        # Logarithmic schedule from 50 down to 0.001, 40 levels
        T_list = np.geomspace(50.0, 0.001, 40)
    
    for Ti in T_list:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break
        remaining = time_limit - elapsed
        max_iter = max(30, int(remaining * 8))
        
        def obj(z):
            return lse_objective_and_grad_z(z, Ti, n)[0]
        
        def jac(z):
            return lse_objective_and_grad_z(z, Ti, n)[1]
        
        try:
            res = minimize(
                obj, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    return best_a, best_score


def active_subgradient(a0, time_limit, n_steps=50000):
    """Active-set subgradient: focus on top peaks only."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        # Use top 3 peaks (or all within 0.1% of max)
        threshold = M * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            m = np.arange(js, je + 1)
            g[m] += 2.0 * a[k - m]
        g /= len(active)
        
        # Projected gradient step on simplex
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            break
        a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.005, 0.05)
        else:
            lr *= 0.99
            lr = max(lr, 1e-8)
    
    return best_a, best_score


def peak_shaving(a0, time_limit, n_iter=20000):
    """Directly reduce the peak coefficients by adjusting a."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.005
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        # Find positions where b_k > 0.999 * M
        active = np.where(b > M * 0.999)[0]
        
        # For each active peak k, compute the "pressure" on each a_j
        pressure = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            m = np.arange(js, je + 1)
            pressure[m] += a[k - m]
        
        # Normalize pressure
        p_sum = pressure.sum()
        if p_sum < 1e-15:
            break
        pressure /= p_sum
        
        # Move a away from high-pressure regions
        a_trial = a - lr * (pressure - 1.0 / n)
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            break
        a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.002, 0.02)
        else:
            lr *= 0.995
            lr = max(lr, 1e-9)
    
    return best_a, best_score


def cubic_interpolate(a, new_n):
    """Cubic interpolation of sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    
    # Use cubic spline via np.interp fallback (scipy not always available for this)
    # Simple approach: use np.interp (linear) then smooth
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    # Apply a light smoothing to mitigate linear interpolation artifacts
    if new_n > 10:
        kernel = np.array([1, 2, 3, 2, 1], dtype=np.float64)
        kernel /= kernel.sum()
        a_smooth = np.convolve(a_interp, kernel, mode='same')
        a_smooth = np.maximum(a_smooth, 0)
        ss = a_smooth.sum()
        if ss > 1e-15:
            a_interp = a_smooth / ss
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def perturbation_restart(a0, n_perturbs=5, rng=None):
    """Try several random perturbations and return the best."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for _ in range(n_perturbs):
        # Gaussian perturbation in log-space
        z = np.log(np.maximum(a, 1e-300))
        z_pert = z + rng.standard_normal(n) * 0.1
        a_pert = np.exp(z_pert - z_pert.max())
        a_pert = a_pert / a_pert.sum()
        sc = 2.0 * n * fast_convolve(a_pert).max()
        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
    
    # Also try "smoothing" perturbation
    for sigma in [0.3, 0.5, 1.0]:
        kernel_size = int(2 * sigma * n / 10) + 1
        if kernel_size % 2 == 0:
            kernel_size += 1
        if kernel_size > 1:
            x = np.arange(kernel_size) - kernel_size // 2
            kernel = np.exp(-x**2 / (2 * sigma**2))
            kernel /= kernel.sum()
            a_smooth = np.convolve(a, kernel, mode='same')
            a_smooth = np.maximum(a_smooth, 0)
            ss = a_smooth.sum()
            if ss > 1e-15:
                a_smooth = a_smooth / ss
                sc = 2.0 * n * fast_convolve(a_smooth).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_smooth.copy()
    
    return best_a, best_score


def power_sum_optimize(a0, time_limit, p_values=None):
    """Optimize using power-sum smooth approximation of max."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if p_values is None:
        p_values = [50, 20, 10, 5, 3]
    
    a_clip = np.maximum(a, 1e-300)
    z = np.log(a_clip) - np.log(a_clip.sum())
    
    for p in p_values:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break
        remaining = time_limit - elapsed
        max_iter = max(20, int(remaining * 5))
        
        def obj_p(z):
            z_s = z - z.max()
            a_cur = np.exp(z_s) / np.exp(z_s).sum()
            b = fast_convolve(a_cur)
            bp = b ** p
            return 2.0 * n * (bp.sum()) ** (1.0 / p)
        
        def grad_p(z):
            z_s = z - z.max()
            a_cur = np.exp(z_s) / np.exp(z_s).sum()
            b = fast_convolve(a_cur)
            bp = b ** p
            Sp = bp.sum()
            factor = 2.0 * n / (p * Sp ** (1.0 - 1.0/p))
            
            # grad w.r.t. a: factor * sum_k p * b_k^(p-1) * S_k(a)
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                coeff = factor * (b[k] ** (p - 1))
                if abs(coeff) < 1e-30:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                m = np.arange(js, je + 1)
                grad_a[m] += coeff * a_cur[k - m]
            
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj_p, z, jac=grad_p, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    return best_a, best_score


def structured_init(n, rng):

    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Load parent, try quick n-search to find best dimension
    2. Fine-grained LSE L-BFGS (40 temperature levels, log-spaced)
    3. Power-sum optimization (complementary smooth approximation)
    4. Active-set subgradient
    5. Peak shaving
    6. Perturbation restarts + re-optimization
    7. Final ultra-low-T polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 1: Quick n-search ===
    print(f"\n=== Phase 1: Quick n-Search (current best: {best_score:.10f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        n_candidates = []
        base_n = n_best
        for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0, 4.0]:
            cand_n = int(base_n * mult)
            if 100 <= cand_n <= 8000 and cand_n not in n_candidates:
                n_candidates.append(cand_n)
        # Also try some absolute values
        for abs_n in [400, 600, 800, 1000, 1500, 2000, 3000, 4000, 6000]:
            if abs_n not in n_candidates:
                n_candidates.append(abs_n)
        
        n_candidates.sort()
        quick_budget = min(rem * 0.15, 250.0)
        per_n_budget = quick_budget / max(1, len(n_candidates))
        
        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(per_n_budget * 1.5, rem * 0.3)
            if t_sub < 10:
                break
            
            init_cand = cubic_interpolate(best_seq, cand_n)
            # Quick LSE with few T levels
            T_quick = [10.0, 3.0, 1.0, 0.3, 0.1, 0.03]
            a_cand, sc_cand = lse_optimize(init_cand, t_sub, T_list=T_quick)
            try_update(a_cand, f"quick_n{cand_n}")
    
    # === Phase 2: Fine-grained LSE on best n ===
    print(f"\n=== Phase 2: Fine LSE (score={best_score:.10f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 600.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s, T levels: 40 (log-spaced 50 -> 0.001)")
        T_fine = np.geomspace(50.0, 0.001, 40)
        a_lse, sc_lse = lse_optimize(best_seq, t2, T_list=T_fine)
        try_update(a_lse, "lse_fine")
    
    # === Phase 3: Power-sum optimization ===
    print(f"\n=== Phase 3: Power-Sum (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 20:
        a_ps, sc_ps = power_sum_optimize(best_seq, t3, p_values=[30, 15, 10, 7, 5, 3])
        try_update(a_ps, "power_sum")
    
    # === Phase 4: Active-set subgradient ===
    print(f"\n=== Phase 4: Active Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 10:
        a_sg, sc_sg = active_subgradient(best_seq, t4, n_steps=80000)
        try_update(a_sg, "subgrad")
    
    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 120.0)
    if t5 > 10:
        a_peak, sc_peak = peak_shaving(best_seq, t5, n_iter=30000)
        try_update(a_peak, "peak_shave")
    
    # === Phase 6: Perturbation + re-optimize ===
    print(f"\n=== Phase 6: Perturbation Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_restarts = min(5, int(rem / 60))
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert, _ = perturbation_restart(best_seq, n_perturbs=8, rng=rng)
            # Quick re-optimize
            T_reopt = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
            t_sub = min(rem * 0.3, 80.0)
            a_re, _ = lse_optimize(a_pert, t_sub, T_list=T_reopt)
            try_update(a_re, f"perturb_{r}")
    
    # === Phase 7: Final ultra-low-T polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very low T LSE
        t_polish = min(rem * 0.4, 200.0)
        T_ultra = np.geomspace(1.0, 0.0005, 25)
        a_final, _ = lse_optimize(best_seq, t_polish, T_list=T_ultra)
        try_update(a_final, "ultra_low_T")
        
        # Peak shaving again
        rem = deadline - time.monotonic()
        if rem > 20:
            t_ps = min(rem * 0.3, 100.0)
            a_ps2, _ = peak_shaving(best_seq, t_ps, n_iter=50000)
            try_update(a_ps2, "peak_shave_2")
        
        # Final subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            t_sg = min(rem * 0.3, 60.0)
            a_sg2, _ = active_subgradient(best_seq, t_sg, n_steps=50000)
            try_update(a_sg2, "subgrad_2")
    
    # === Phase 8: One more n-search if time allows ===
    rem = deadline - time.monotonic()
    if rem > 100:
        print(f"\n=== Phase 8: Final n-Search (score={best_score:.10f}) ===")
        n_best_now = len(best_seq)
        for mult in [0.9, 1.1, 1.3, 1.5]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            cand_n = int(n_best_now * mult)
            if cand_n == n_best_now or cand_n < 100 or cand_n > 10000:
                continue
            t_sub = min(rem * 0.3, 80.0)
            init_cand = cubic_interpolate(best_seq, cand_n)
            T_mid = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
            a_cand, _ = lse_optimize(init_cand, t_sub, T_list=T_mid)
            try_update(a_cand, f"final_n{cand_n}")
    
    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
