    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    s = a.sum()
    if s < 1e-15:

def active_set_qcp(p0, solver, active_k=30, time_limit=60):
    """Solve min t s.t. (p*p)[k] <= t for k in top-|A| peaks, sum(p)=1, p>=0."""
    n = len(p0)
    p = normalize(p0)
    best_p = p.copy()
    best_score = local_eval(p)
    t0 = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        E = fast_convolve(p)
        M = E.max()
        # Active set: top k positions
        active = np.argsort(E)[::-1][:active_k]
        
        # Build QCP
        p_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(p_var) == 1]

            js = range(max(0, k - n + 1), min(n, k + 1))
            quad = 0
            for j in js:
                quad = quad + p_var[j] * p_var[k - j]
            constraints.append(quad <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                p_new = np.asarray(p_var.value, dtype=np.float64)
                p_new = np.maximum(p_new, 0)
                s = p_new.sum()
                if s > 1e-15:
                    p_new = p_new / s
                    sc = local_eval(p_new)
                    if sc < best_score:
                        best_score = sc
                        best_p = p_new.copy()
                        p = p_new
        except Exception:
            break
    
    return best_p, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def fft_cross_corr_full(w, p):
    """Compute (w ⋆ p)[j] = sum_k w[k]*p[k-j] for j=0..len(p)-1 using FFT.
