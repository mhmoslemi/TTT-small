def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    return irfft(A * A, fft_size)[:size]

    return irfft(A * B, fft_size)[:size]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

        try:
            prob.solve(solver=cp.ECOS)

            return a.copy(), float('inf')

    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with adaptive multi-factor grids (primary engine)
2. LP-based best-response alternating for basin jumping
3. Sparsification (zeroing coordinates) attempts
4. Multi-scale refinement and random restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [T @ b_var <= M_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    b_val[b_val < 1e-12] = 0.0
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def coord_descent(a0, time_limit, verbose=True, label="", phase=1):
    """
    Coordinate descent with exact 1D optimization.
    Works on unnormalized vectors. Score = 2*n*max(conv)/sum^2.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (s * s)
    best_a = a.copy()
    best_s = s
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Factor grids - progressively refined
    if phase == 1:
        coarse_grid = [0.0, 0.1, 0.25, 0.5, 0.75, 1.33, 1.75, 2.5, 4.0]
        base_scale = 1.0 / n
    else:
        coarse_grid = [0.0, 0.25, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.7, 2.5]
        base_scale = 1.0 / n

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            a_i = a[i]

            # Skip very small values early (save time)
            if a_i < 1e-12 * s and sweeps > 20 and phase == 1:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            # Unaffected max
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = 0.0
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:2 * n - 1].max())
                unaff_max = max(unaff_max, ur)

            # Precompute
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]
            idx_2i = 2 * i - k_start

            best_delta = 0.0
            best_new_score = best_score

            # Try multiplier factors
            for factor in coarse_grid:
                if factor == 1.0:
                    continue
                new_a_i = a_i * factor
                delta = new_a_i - a_i
                if abs(delta) < 1e-15:
                    continue

                new_s = s + delta
                if new_s < 1e-10:
                    continue

                # New affected conv values
                new_aff = conv_aff + 2.0 * delta * a_shifted
                # Fix quadratic at k=2i
                new_aff = new_aff.copy()
                new_aff[idx_2i] = conv[2 * i] + 2.0 * delta * a_i + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                new_score = 2.0 * n * new_M / (new_s * new_s)

                if new_score < best_new_score - 1e-13:
                    best_new_score = new_score
                    best_delta = delta

            # Try absolute values near zero (for coordinates that might benefit from being set to a small positive value)
            if a_i < base_scale * 2 and sweeps > 5:
                for abs_val in [base_scale * 0.5, base_scale, base_scale * 2, base_scale * 4]:
                    delta = abs_val - a_i
                    if abs(delta) < 1e-15:
                        continue
                    new_s = s + delta
                    if new_s < 1e-10:
                        continue
                    new_aff = conv_aff + 2.0 * delta * a_shifted
                    new_aff = new_aff.copy()
                    new_aff[idx_2i] = conv[2 * i] + 2.0 * delta * a_i + delta * delta
                    new_M = max(unaff_max, float(new_aff.max()))
                    new_score = 2.0 * n * new_M / (new_s * new_s)
                    if new_score < best_new_score - 1e-13:
                        best_new_score = new_score
                        best_delta = delta

            # Apply best change
            if best_delta != 0.0:
                old_a_i = a_i
                a[i] = a[i] + best_delta
                s += best_delta

                # Update conv incrementally
                changes = 2.0 * best_delta * a_shifted
                conv[k_start:k_end + 1] += changes
                conv[2 * i] += best_delta * best_delta

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                    best_s = s
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] p{phase} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f} nnz={int((a > 1e-10).sum())}")

    # Normalize and verify
    s_final = best_a.sum()
    if s_final < 1e-10:
        best_a = a.copy()
        s_final = s
    best_a = best_a / s_final
    final_score = 2.0 * n * fast_convolve(best_a).max()
    return best_a, final_score


def lp_alternate(a0, time_limit, solver, verbose=True):
    """LP best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        sc_b = 2.0 * n * fast_convolve(b_new).max()
        if sc_b < best_score - 1e-12:
            best_score = sc_b
            best_a = b_new.copy()

        # Damped update
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.6
        else:
            alpha = max(0.2, 5.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = b_new.copy()
        else:
            a = a / ss

        sc_a = 2.0 * n * fast_convolve(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()

        if sc_a > best_score * 1.005:
            stall += 1
        else:
            stall = max(0, stall - 1)

        if stall > 15:
            # Random perturbation to escape
            noise = 0.02 / np.sqrt(n) * np.random.standard_normal(n)
            a_p = np.maximum(a + noise, 0)
            a_p = a_p / a_p.sum()
            sc_p = 2.0 * n * fast_convolve(a_p).max()
            if sc_p < best_score * 1.01:
                a = a_p
            stall = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def sparsify(a0, time_limit, verbose=True):
    """Try zeroing random subsets of coordinates to find sparser solutions."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Pick a random subset to zero out (small subset)
        n_zero = int(n * rng.uniform(0.02, 0.15))
        zeros = rng.choice(n, size=n_zero, replace=False)

        a_try = a.copy()
        a_try[zeros] = 0.0
        ss = a_try.sum()
        if ss < 1e-10:
            continue
        a_try = a_try / ss

        sc = 2.0 * n * fast_convolve(a_try).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_try.copy()
            if verbose:
                print(f"    Sparsify: new best={best_score:.8f}, zeroed {n_zero} coords")

    return best_a, best_score


def diverse_init(n, rng, idx):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if idx == 0:
        a = np.maximum(1.0 - np.abs(t - center) / max(center, 1), 0)
    elif idx == 1:
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif idx == 2:
        x = (t - center) / center
        pw = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** pw
    elif idx == 3:
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif idx == 4:
        period = int(rng.integers(3, 15))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif idx == 5:
        a = 1.0 - 0.4 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif idx == 6:
        # Sawtooth
        period = int(rng.integers(5, 30))
        a = np.abs((t % period) / period - 0.5) + 0.1
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t * rng.uniform(1, 8) / n)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a
