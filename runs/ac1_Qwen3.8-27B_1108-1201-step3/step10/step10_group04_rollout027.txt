```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):

        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

    try:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:

                    break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_logsumexp(a0, time_limit, T_start=500.0, T_end=0.01, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # More iterations at lower temperatures
        if si < n_stages // 2:
            iters_per_stage = 50
        else:
            iters_per_stage = 100

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 3
    elif choice == 7:
        # Hann window
        x = np.linspace(0, 1, n)
        a = 0.5 - 0.5 * np.cos(2 * np.pi * x)
    elif choice == 8:
        # Flat-top window approximation
        x = np.linspace(0, 1, n)
        a = 1 - 1.93 * np.abs(np.cos(np.pi * x)) + 1.29 * np.abs(np.cos(2 * np.pi * x)) - 0.39 * np.abs(np.cos(3 * np.pi * x))
        a = np.maximum(a, 0)
    else:

        x = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    
    s = a.sum()
    if s < 1e-15:

    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def combine_sequences(a1, a2, t):
    """Combine two sequences of the same length."""
    a1 = np.asarray(a1, dtype=np.float64)
    a2 = np.asarray(a2, dtype=np.float64)
    if len(a1) != len(a2):
        a2 = interpolate_sequence(a2, len(a1))
    combo = (1 - t) * a1 + t * a2
    combo = np.maximum(combo, 0)
    s = combo.sum()
    if s < 1e-15:
        return a1 / a1.sum()
    return combo / s
```

```python
def lse_jacobian_fast(z_in, T_val, a, n):
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = min(Tb.max(), 700.0)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # Compute grad_a = 2 * sum_k w_k * (a[k-i] + a[i-k])
    # = 2 * (conv(w, a))[i] + 2 * (conv(w, a_rev))[i]
    # But w has length 2n-1, a has length n
    
    # Actually, let's use the direct formula:
    # grad_a[i] = 2 * sum_k w_k * a[k-i] (for k-i in [0, n-1])
    #            + 2 * sum_k w_k * a[i-k] (for i-k in [0, n-1])
    # But a[k-i] + a[i-k] = 2*a[k-i] when k-i and i-k are both in range,
    # which happens when |k-i| < n.
    
    # Simpler: just use the loop (it's O(n * 2n) = O(n^2) which is fine for n=1200)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    return a * (grad_a - S)
```

This is O(n * 2n) = O(n^2) in the worst case, but in practice, only a few w_k are significant (the ones corresponding to the top peaks). So it's much faster in practice.
