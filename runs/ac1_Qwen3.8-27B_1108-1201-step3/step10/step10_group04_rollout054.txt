        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # Gradient of sum_k w_k * (a*a)_k with respect to a
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Adapt iterations based on remaining time
        frac_remaining = remaining / time_limit
        iters = max(10, int(80 * frac_remaining))
        
        def func(z_in):
            return lse_objective(z_in, stage_T)
        
        def jac(z_in):
            return lse_jacobian(z_in, stage_T)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 100,
                         'disp': False}

            a_candidate = softmax_from_z(z_candidate)
            score = 2.0 * n * fast_convolve(a_candidate).max()

                best_score = score
                best_a = a_candidate.copy()

            pass

            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.2f}) ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, max_pairs=None):
    """Coordinate descent: optimize 2 coordinates at a time."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if max_pairs is None:
        max_pairs = n * 3  # Try each coordinate in 3 random pairings
    
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    for iteration in range(max_pairs * 50):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick two random coordinates
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] < 1e-15:
            continue
        
        # Optimize the split between i and j
        base = a[i] + a[j]
        
        def objective(frac):
            nonlocal a
            a_trial = a.copy()
            a_trial[i] = base * frac
            a_trial[j] = base * (1 - frac)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            return sc
        
        # Use golden section search on [0, 1]
        try:
            res = minimize_scalar(objective, bounds=(0.0, 1.0), method='bounded',
                                 options={'maxiter': 50})
            if res.success and res.fun < best_score - 1e-12:
                a[i] = base * res.x
                a[j] = base * (1 - res.x)
                best_score = res.fun
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 500:
            # Try a different pairing strategy
            no_improve = 0
    
    return best_a, best_score

def subgradient_descent(a0, time_limit, n_active=5):
    """Subgradient method with Polyak step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = np.maximum(a, 0)
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b
        
        # Polyak step size
        gap = current_score - lb
        if gap < 1e-12:
            break
        alpha = gap / (max_b * max_b + 1e-15) * 0.5  # Scale down
        
        # Compute subgradient using top n_active peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:n_active]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Projected subgradient step
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def nesterov_lp(a0, time_limit, solver=None):
    """Nesterov-accelerated LP best-response."""
    import cvxpy as cp
    
    def get_solver():
        for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
        return cp.ECOS
    
    if solver is None:
        solver = get_solver()
    
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Precompute the convolution matrix structure
    # T is (2n-1) x n where T[k, j] = a[j] if k-n+1 <= j <= k, else 0
    # We'll build it on the fly since it depends on a
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build convolution matrix for current y
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            js = max(0, j)  # k - j >= 0 => k >= j
            je = min(2 * n - 2, j + n - 1)  # k - j <= n-1 => k <= j+n-1
            T[js:je+1, j] = y[j]
        
        # Solve LP: min M s.t. T @ b <= M, sum(b) = 1, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            continue
        b_val = b_val / sb
        
        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = (1 - alpha) * a + alpha * b_val
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar, minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z_in, T_val, n_val, t_start, time_limit):

        if w[k] < 1e-15:
            continue
        js = max(0, k - n_val + 1)
        je = min(n_val - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_optimize(a0, time_limit, T_start=30.0, T_end=0.01, n_stages=50, seed_offset=0):
    """LogSumExp L-BFGS with many-stage geometric annealing."""
    n = len(a0)
    z = z_from_a(a0.astype(np.float64)).copy()
    t_start = time.monotonic()
    best_a = a0.astype(np.float64).copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 2:
            break

        frac_remaining = remaining / time_limit
        iters = max(15, int(100 * frac_remaining))

        def func(z_in):
            return lse_objective(z_in, stage_T, n, t_start, time_limit)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T, n, t_start, time_limit)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_candidate = res.x
            a_candidate = softmax_from_z(z_candidate)
            score = 2.0 * n * fast_convolve(a_candidate).max()
            if score < best_score:
                best_score = score
                best_a = a_candidate.copy()
            z = z_candidate
        except Exception:
            pass

        if (si + 1) % 10 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.3f}) ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=12345):
    """Coordinate descent: optimize 2 coordinates at a time via 1D search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = n * 2

    for iteration in range(n * 1000):
        if time.monotonic() - t_start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] + a[j] < 1e-15:
            continue

        base = a[i] + a[j]

        def obj_1d(frac):
            a_trial = a.copy()
            a_trial[i] = base * frac
            a_trial[j] = base * (1.0 - frac)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            return sc

        try:
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40})
            if res.success and res.fun < best_score - 1e-14:
                a[i] = base * res.x
                a[j] = base * (1.0 - res.x)
                best_score = res.fun
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

    return best_a, best_score


def targeted_peak_shave(a0, time_limit, seed=777):
    """Targeted peak shaving: identify top peaks, find best coordinate moves to reduce them."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = np.maximum(a, 0)
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)

    for outer in range(200000):
        if time.monotonic() - t_start > time_limit:
            break

        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_k = sorted_idx[:min(5, len(b))]
        current_max = b.max()

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = int(idx[np.argmax(contrib)])

            # Find destination: position that contributes least to any top peak
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for jj in range(js2, je2 + 1):
                    peak_areas.add(jj)

            non_peak = [jj for jj in range(n) if jj not in peak_areas and jj != src]
            if non_peak:
                dst = int(rng.choice(non_peak))
            else:
                dst = int(rng.integers(n))
                if dst == src:
                    dst = (dst + 1) % n

            for frac in [0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.01, 0.08))
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

    return best_a, best_score


def nesterov_lp(a0, time_limit):
    """Nesterov-accelerated LP best-response with adaptive step."""
    solver = None
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = solver_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = time.monotonic()

    for it in range(100000):
        if time.monotonic() - t_start > time_limit:
            break

        # Build convolution matrix T (2n-1) x n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            js = j
            je = min(2 * n - 2, j + n - 1)
            T[js:je + 1, j] = y[j]

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            continue
        b_val = b_val / sb

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = (1.0 - alpha) * a + alpha * b_val
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def subgradient_method(a0, time_limit, n_active=5):
    """Subgradient method with Polyak step size and random restarts."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lb = 2.0 * n / (2.0 * n - 1.0)
    t_start = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    for step in range(2000000):
        if time.monotonic() - t_start > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b

        gap = current_score - lb
        if gap < 1e-13:
            break

        # Polyak step with damping
        alpha = 0.3 * gap / (max_b * max_b + 1e-15)
        alpha = min(alpha, 0.5)

        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:n_active]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 1000:
            # Random perturbation
            noise = rng.random(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_score


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.4) ** 2) + 0.3 * np.exp(-5 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - np.abs(2 * x - 1))
        a = a ** 2 + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
