def fast_convolve(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Score for a sequence normalized to sum=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def solve_best_response_lp(a, verbose=False):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]
    subject to b >= 0, sum(b) = 1.
    Uses scipy.optimize.linprog with HiGHS.
    """
    from scipy.optimize import linprog
    n = len(a)
    size = 2 * n - 1
    
    # Build sparse constraint matrix A_ub: size x (n+1)
    # Row k: [a[k], a[k-1], ..., 0, ..., 0, -1] (shifted based on k)
    # (T @ b)_k = sum_j a[k-j] * b_j for valid j
    rows = []
    cols = []
    vals = []

        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
        # M coefficient
        rows.append(k)
        cols.append(n)  # M is variable n
        vals.append(-1.0)
    
    A_ub = csc_matrix((vals, (rows, cols)), shape=(size, n+1))
    
    # Objective: minimize M (variable index n)
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(size),
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    b_val = np.maximum(res.x[:n], 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = max(0.0, res.x[n])
    return b_val, float(M_val)

Let me now think about the **Nesterov acceleration** more carefully.

The standard Nesterov acceleration for the outer loop:
- Maintain y (Nesterov point) and a (current point)
- Each iteration:
  1. Solve LP: b = best_response(y)
  2. alpha = 2/(t+2)
  3. y_new = (1-alpha)*y + alpha*b
  4. a_new = ((t+1)*a + alpha*(t+2)*b) / ((t+1) + alpha*(t+2))
  5. a = normalize(a_new), y = normalize(y_new)

This is the FISTA-style update applied to the block coordinate descent.

The convergence rate of Nesterov-accelerated block coordinate descent is O(1/t^2) for the objective value, which is optimal for first-order methods.

However, in practice, the convergence might be slower because:
1. The LP gives the exact best response, but the outer convergence is limited by the "geometry" of the problem
2. The objective is non-smooth (max of quadratics), so the Nesterov rate might not apply directly

To address this, I can:
- Use **restarts** when the improvement stalls
- Use **adaptive step sizes**
- Combine with **smoothing** (LogSumExp) for the final polish

Peak shaving idea:
1. Compute b = conv(a, a), find argmax k*
2. The peak b[k*] = sum_j a[j] * a[k*-j]
3. To reduce the peak, we want to reduce a[j] * a[k*-j] for the j's that contribute most

5. So we reduce a at positions that contribute to the peak and increase a at positions that don't

A simple approach:

- Reduce a[j] and a[k*-j] by a small amount epsilon
- Distribute the removed mass to positions far from any peak

def peak_shave(a, n_steps=1000):
    """Custom peak shaving heuristic."""
    a = normalize(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    n = len(a)
    
    for step in range(n_steps):
        b = fast_convolve(a)
        k_star = b.argmax()
        
        # Find contributing pairs
        # b[k_star] = sum_j a[j] * a[k_star - j]
        js = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        contributions = a[js] * a[k_star - js]
        
        # Find the position with the largest contribution
        j_max = js[np.argmax(contributions)]
        
        # Find a position with minimum "peak contribution" (safest to increase)
        # A position i is safe if reducing a at i doesn't increase any peak
        # Simple heuristic: pick the position with minimum a[i] * max_k a[k-i]
        peak_contact = np.array([a[i] * fast_convolve_selective(b, i, n) for i in range(n)])
