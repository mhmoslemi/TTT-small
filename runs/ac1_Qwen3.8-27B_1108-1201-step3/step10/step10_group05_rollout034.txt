def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

            prob.solve(solver=solver_name)

def spectral_flatten(a, num_iters=100, damp=0.5):
    """Fienup-style spectral flattening with non-negativity constraint."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(num_iters):
        F = rfft(a, fft_size)
        power = np.abs(F) ** 2

        # Target: geometric mean of power spectrum
        log_power = np.log(np.maximum(power, 1e-30))
        target_log = np.mean(log_power)
        target = np.exp(target_log) * np.ones_like(power)

        # Interpolate between current and target
        new_power = power * (1 - damp) + target * damp
        new_power = np.maximum(new_power, 0)

        # Reconstruct with same phase
        magnitude_ratio = np.sqrt(new_power / np.maximum(power, 1e-30))
        new_F = F * magnitude_ratio

        a_new = irfft(new_F, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

    return best_a


def spectral_perturb(a, rng, strength=0.3, num_iters=30):
    """Perturb the sequence by adding noise to its power spectrum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(num_iters):
        F = rfft(a, fft_size)
        power = np.abs(F) ** 2
        avg_p = power.mean()
        noise = rng.standard_normal(len(power)) * strength * 0.1 * avg_p
        new_power = np.maximum(power + noise, 0)
        ratio = np.sqrt(new_power / np.maximum(power, 1e-30))
        new_F = F * ratio
        a = irfft(new_F, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

    return a

                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            # Also try slightly larger steps
            for factor in [0.95, 0.97, 0.99, 1.01, 1.03, 1.05]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Adaptive damping
        alpha = min(1.0, 8.0 / (iters + 10))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Try LP solution directly
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def nesterov_smoothed_md(a0, time_limit, beta_init=25.0, lr_init=0.008, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            if lr < 1e-10:
                break

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def peak_targeted_perturb(a0, rng, strength=0.04):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.15):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 5):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def nesterov_subgradient(a0, time_limit, label=""):
    """Subgradient with Nesterov acceleration on the exact objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M - 0.02 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        a_new = a - step * grad
        a_new = simplex_project(a_new)

        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Spectral flattening for novel initialization/improvement
    2. Enhanced coordinate descent with ternary search
    3. LP fictitious play
    4. Nesterov smoothed mirror descent
    5. Multi-length exploration with spectral perturbation
    6. Intensive final polishing
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(777)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: Spectral flattening ===
    print(f"\n=== Phase 1: Spectral Flattening (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(80.0, rem * 0.1)
    print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
    a1 = spectral_flatten(best_seq, num_iters=200, damp=0.4)
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_spec")

    # Immediate CD after spectral flatten
    t1b = min(120.0, (deadline - time.monotonic()) * 0.15)

    best_seq, best_score = check_and_update(a1b, best_seq, best_score, "p1_cd")

    # === Phase 2: Enhanced coordinate descent on best ===
    print(f"\n=== Phase 2: Enhanced CD (best={best_score:.6f}) ===")
    t2 = min(200.0, (deadline - time.monotonic()) * 0.2)
    print(f"  Budget: {t2:.0f}s, n={len(best_seq)}")
    a2, s2, sw2 = enhanced_coord_descent(best_seq, t2, verbose=True, label="p2")
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_ecd")

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.6f}) ===")
    t3 = min(250.0, (deadline - time.monotonic()) * 0.2)
    print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
    a3, s3, it3 = lp_fictitious_play(best_seq, t3, solver=solver, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_lpfp")

    # CD after LP
    t3b = min(100.0, (deadline - time.monotonic()) * 0.08)
    a3b, s3b, _ = enhanced_coord_descent(a3, t3b, label="p3b_cd")
    best_seq, best_score = check_and_update(a3b, best_seq, best_score, "p3b_cd")

    # === Phase 4: Nesterov Smoothed MD ===
    print(f"\n=== Phase 4: Nesterov MD (best={best_score:.6f}) ===")
    t4 = min(150.0, (deadline - time.monotonic()) * 0.15)
    print(f"  Budget: {t4:.0f}s")
    a4, s4 = nesterov_smoothed_md(best_seq, t4, beta_init=30.0, lr_init=0.006, label="p4")
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_nmd")

    # CD after MD
    t4b = min(80.0, (deadline - time.monotonic()) * 0.08)
    a4b, s4b, _ = enhanced_coord_descent(a4, t4b, label="p4b_cd")
    best_seq, best_score = check_and_update(a4b, best_seq, best_score, "p4b_cd")

    # === Phase 5: Multi-start with perturbations ===
    print(f"\n=== Phase 5: Perturbation Loop (best={best_score:.6f}) ===")
    for rnd in range(5):
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        t5 = min(rem * 0.15, 80.0)

        # Different perturbation strategies
        if rnd == 0:
            perturbed = peak_targeted_perturb(best_seq, rng, strength=0.03)

            perturbed = spectral_perturb(best_seq, rng, strength=0.5, num_iters=20)

        else:
            perturbed = peak_targeted_perturb(best_seq, rng, strength=0.06)

    for alt in [896, 1024, 1152, 1280, 1408, 1536]:
        if alt != n_best:
            alt_lengths.add(alt)
    alt_lengths = list(alt_lengths)[:3]

    for n_alt in alt_lengths:

        if rem < 30:

        t6 = min(rem * 0.25, 60.0)

    print(f"\n=== Phase 7: Subgradient Nesterov (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 30:
        t7 = min(rem * 0.5, 120.0)
        a7, s7 = nesterov_subgradient(best_seq, t7, label="p7")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_subgrad")

    # === Phase 8: Final intensive polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t8 = min(rem * 0.5, 100.0)
        a8, s8, _ = enhanced_coord_descent(best_seq, t8, label="p8")
        best_seq, best_score = check_and_update(a8, best_seq, best_score, "p8_ecd")

    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t8b = min(rem * 0.7, 80.0)
        a8b, s8b = fine_coord_descent(best_seq, t8b, label="p8b")
        best_seq, best_score = check_and_update(a8b, best_seq, best_score, "p8b_fine")

    rem = deadline - time.monotonic() - 10

        best_seq, best_score = check_and_update(a8c, best_seq, best_score, "p8c_fine")

    # One more enhanced CD pass if time remains
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a8d, s8d, _ = enhanced_coord_descent(best_seq, rem * 0.8, label="p8d")
        best_seq, best_score = check_and_update(a8d, best_seq, best_score, "p8d")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
