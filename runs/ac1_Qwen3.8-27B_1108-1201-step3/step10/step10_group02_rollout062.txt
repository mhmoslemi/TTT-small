"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive weighted mirror descent with backtracking, LP alternating,
LogSumExp L-BFGS, and targeted coordinate descent.
Key innovations: exponential peak weighting for smoother subgradients,
guaranteed-improvement backtracking, adaptive active set tightening,
and multi-n exploration for global improvement.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Project to probability simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, active_indices):
    """Compute uniform subgradient of max(conv(a,a)) at active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def compute_weighted_subgradient(a, b, T):
    """Compute exponentially-weighted subgradient using soft peak selection."""
    n = len(a)
    max_b = b.max()
    if T < 1e-10:
        active = np.where(b >= max_b * (1 - 1e-9))[0]
        return compute_subgradient(a, active)
    w = np.exp((b - max_b) / T)
    total_w = w.sum()
    if total_w < 1e-300:
        active = np.where(b == max_b)[0]
        return compute_subgradient(a, active)
    w = w / total_w
    # Use FFT-based cross-correlation for efficiency
    size = len(b)
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # cross-correlation: g[i] = sum_j a[j] * w[i+j]
    # = conv(a, w_reversed)[size-1-i]
    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    conv = irfft(A * W_rev, fft_size)[:size + n - 1]
    g = conv[size - 1 : size - 1 - n : -1]
    g = g[::-1]
    return g


def md_step_with_backtrack(a, g, eta0, max_bt=8):
    """Mirror descent step with backtracking line search. Returns (best_a, best_score, eta_used)."""
    n = len(a)
    b_curr = fast_convolve(a)
    score_curr = b_curr.max()
    eta = eta0
    best_a = a.copy()
    best_score = score_curr
    log_a = np.log(np.maximum(a, 1e-300))
    g_shifted = g - g.min()

    for _ in range(max_bt):
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-300:
            break
        a_new = a_new / s_new
        score_new = fast_convolve(a_new).max()
        if score_new < best_score - 1e-15:
            best_a = a_new
            best_score = score_new
            break
        eta *= 0.5
        if eta < 1e-8:
            break
    return best_a, best_score, eta

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

    """Find a working LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k] s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    return None, None

def lse_obj_grad(z, T, n):
    """LogSumExp objective and gradient in softmax parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    logsumexp = max_b + T * np.log(exp_b.sum())
    obj = 2.0 * n * logsumexp

    # Gradient wrt a: grad_a[i] = (2n/T) * sum_k w[k] * a[k-i]
    # Use FFT-based cross-correlation
    size = len(b)
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    A = rfft(a, fft_size)
    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    conv = irfft(A * W_rev, fft_size)[:size + n - 1]
    grad_a = conv[size - 1 : size - 1 - n : -1]
    grad_a = grad_a[::-1]
    grad_a *= 2.0 / T

    # Convert to z-gradient (softmax): grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on LogSumExp smoothed objective with decreasing temperature."""
    n = len(a0)
    a0 = normalize(a0.copy())
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = lse_obj_grad(z_inner, T, n)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = lse_obj_grad(z_inner, T, n)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': maxiter_per_stage * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    return a

            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9998, 0.9997, 0.9995, 0.999, 0.998, 0.997, 0.995]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Simultaneous small reduction of top contributors
                a_trial = a.copy()
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                for pos in top_positions:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.9995
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    if stall > 800:
                        break
        else:
            stall = 0

        if (step + 1) % 15000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: score={best_score:.8f} stall={stall}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.35) ** 2 / 0.15)
    else:
        a = 0.7 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    return normalize(a)
