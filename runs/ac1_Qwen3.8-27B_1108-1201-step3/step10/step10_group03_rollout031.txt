import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (assumes normalized, non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Returns (b, M, lambda_dual) where lambda_dual is the dual variable distribution."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    # constraints: T @ b <= M * ones, sum(b) = 1
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver] + [s2 for s2 in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s2 != solver]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                bs = b_val.sum()
                if bs > 1e-15:
                    b_val = b_val / bs
                    return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_score(a):
    """Compute objective for normalized non-negative a."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def get_solver():
    for s in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def solve_br(a_fixed, solver):
    """Best response: given a, find b>=0, sum(b)=1 min max(conv(a,b))."""
    n = len(a_fixed)

    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(Mv), cons)
    for s in [solver] + [s2 for s2 in [cp.ECOS, cp.SCS, cp.PDLP] if s2 != solver]:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = np.asarray(bv.value, dtype=np.float64)
                val = np.maximum(val, 0)
                ss = val.sum()
                if ss > 1e-15:
                    return val / ss
        except Exception:
            continue
    return None

The objective is: 2n * max_m (a*a)[m] / (sum a)^2

For a probability distribution a on {0, ..., n-1}, (a*a)[m] is the probability that two independent draws from a differ by m. So we're asking: what distribution on {0, ..., n-1} minimizes the maximum "difference probability" times 2n?

The "difference distribution" is the distribution of X - Y where X, Y are i.i.d. with distribution a. We want to minimize the maximum point probability of this difference distribution.

This is a well-studied problem in additive combinatorics and probability theory. The key results:
- For a uniform distribution on a finite abelian group, the difference distribution is also uniform, so the max is 1/|G| = 1/n (for a group of size n). But we're on Z, not a group, so boundary effects matter.
- For a "flat" distribution (one that's as uniform as possible), the difference distribution is nearly flat, with max ≈ 1/n.
- The optimal (for non-negative distributions on Z) is not known in closed form for general n.

The connection to the problem: we're on {0, 1, ..., n-1} (not a group), so the difference X-Y ranges from -(n-1) to n-1 (2n-1 values). The "ideal" flat distribution would have each difference probability = 1/(2n-1), giving objective = 2n/(2n-1) ≈ 1. The best we can do is approximately 2n/(2n-1) + correction, where the correction depends on the boundary behavior.

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]
