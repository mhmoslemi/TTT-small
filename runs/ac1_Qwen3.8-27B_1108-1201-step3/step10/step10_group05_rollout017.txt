"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def logsumexp_smooth(a, top_k=20, beta=50.0):
    """Compute smooth approximation to max(conv(a,a)) using log-sum-exp of top-k values."""
    conv = fast_convolve(a)
    M = conv.max()
    # Use log-sum-exp for smoothness
    scaled = beta * (conv - M)
    scaled = np.clip(scaled, -50, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    smooth_max = M - np.log(Z) / beta  # This is actually M + log(mean(exp(scaled))) / beta... 
    # Actually: logsumexp(conv) = M + log(sum(exp(conv - M))) = M + log(sum(exp(beta*(conv-M)))) / beta
    # = M + log(Z) / beta
    smooth_max = M + np.log(Z) / beta
    # Gradient of smooth_max w.r.t. a[i]
    # d/d a[i] of logsumexp(conv) = (1/Z) * sum_k Z_k * d conv[k]/d a[i]
    # d conv[k]/d a[i] = a[k-i] (for valid k-i) + a[i] (for k=i, the quadratic term)
    # Actually: conv[k] = sum_j a[j]*a[k-j], so d conv[k]/d a[i] = a[k-i] + (a[i] if 2i=k else 0)
    # More precisely: d conv[k]/d a[i] = sum_j [delta(j=i)*a[k-j] + delta(k-j=i)*a[j]]
    # = a[k-i] (if 0<=k-i<n) + a[i] (if k=2i, but only once since j=i appears once)
    # Wait: conv[k] = sum_j a[j]*a[k-j]. d/d a[i] = a[k-i] (from j=i term) + a[i] (from k-j=i term, i.e., j=k-i)
    # If i = k-i (i.e., k=2i), these are the same term, so d/d a[i] = 2*a[i]... no wait.
    # d/d a[i] of a[j]*a[k-j] is: (da[j]/da[i])*a[k-j] + a[j]*(da[k-j]/da[i])
    # = delta(j=i)*a[k-j] + a[j]*delta(k-j=i)
    # = delta(j=i)*a[k-j] + a[j]*delta(j=k-i)
    # Summing over j: a[k-i] (if valid) + a[i] (if k-i is valid, which it is when we summed)
    # So d conv[k]/d a[i] = a[k-i] + a[i] if k != 2i, and 2*a[i] if k = 2i.
    # Wait no: if k = 2i, then k-i = i, so a[k-i] = a[i], and we get a[i] + a[i] = 2*a[i].
    # If k != 2i: we get a[k-i] + a[i].
    # So: d conv[k]/d a[i] = a[k-i] + a[i] for all valid k (where 0<=k-i<n).
    # (When k=2i, this gives 2*a[i] which is correct.)
    
    grad = np.zeros(len(a))
    active_k = np.where(weights / Z > 1e-10)[0]
    for k in active_k:
        w = weights[k] / Z
        i_start = max(0, k - len(a) + 1)
        i_end = min(len(a) - 1, k)
        j = np.arange(i_start, i_end + 1)
        # d conv[k]/d a[i] = a[k-i] + a[i]
        grad[j] += w * (a[k - j] + a[j])
    return smooth_max, grad

def smooth_gradient_descent(a0, time_limit, beta=30.0, lr_init=0.05, top_k=50, label=""):
    """Accelerated gradient descent on smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    gamma = 1.0
    y = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = lr / np.sqrt(1 + 0.01 * it)

        # Nesterov lookahead
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Compute smooth gradient
        smooth_val, grad = logsumexp_smooth(y, top_k=top_k, beta=beta)

        # Gradient step + projection
        a_new = a - step * grad
        a_new = simplex_project(a_new)

        # Evaluate exact objective
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if no_improve > 50:
            theta = no_improve / 50.0
            lr = lr_init / (1 + 0.1 * theta)
            beta = min(beta * (1 + 0.05 * (no_improve - 50)), 200.0)
            no_improve = 0

    return best_a, best_score

def coord_descent_exact(a0, time_limit, label=""):
    """Exact coordinate descent minimizing the true max of convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1].copy()
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Generate candidates more aggressively
            candidates = [0.0, -ai]

            # Find where each affected line hits unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 2 < d < ai * 2:
                        candidates.append(d)

            # Crossings of top affected lines
            top_k = min(8, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 2 < d < ai * 2:
                            candidates.append(d)

            # Fractional moves
            for frac in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.3 * ai)

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

    return best_a, best_score

def fourier_init(n, rng, mode="random"):
    """Generate initialization based on Fourier domain considerations."""
    if mode == "uniform":
        return np.ones(n) / n
    elif mode == "sine_mod":
        # Modulate uniform with a sine wave
        x = np.arange(n)
        freq = rng.uniform(0.5, 0.45 * n)  # between 0.5 and n/2 cycles
        amp = rng.uniform(0.1, 0.6)
        a = 1.0 + amp * np.sin(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a = a / a.sum()
        return a
    elif mode == "two_freq":
        x = np.arange(n)
        f1 = rng.uniform(0.3, 0.3)
        f2 = rng.uniform(0.2, 0.3)
        a1 = rng.uniform(0.2, 0.5)
        a2 = rng.uniform(0.2, 0.5)
        a = 1.0 + a1 * np.cos(2 * np.pi * f1 * x / n) + a2 * np.cos(2 * np.pi * f2 * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
        a = a / a.sum()
        return a
    elif mode == "envelope_sine":
        # Envelope (bell) times oscillation
        x = np.arange(n)
        center = n / 2
        sigma = rng.uniform(n * 0.2, n * 0.5)
        envelope = np.exp(-(x - center)**2 / (2 * sigma**2))
        freq = rng.randint(2, n // 4)
        oscillation = 1.0 + rng.uniform(0.3, 0.7) * np.sin(2 * np.pi * freq * x / n)
        a = envelope * np.maximum(oscillation, 0.1)
        a = a / a.sum()
        return a
    elif mode == "power_law":
        x = np.arange(n)
        # Power law decay from center
        center = n / 2 - 0.5
        a = 1.0 / (1.0 + np.abs(x - center) ** rng.uniform(0.5, 2.0))
        a = a / a.sum()
        return a
    else:
        # Random with smoothing
        a = rng.random(n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
        return a

def regret_matching_fp(a0, time_limit, label=""):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

        a = np.ones(n) / n
    else:
        a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    # In our game: Player 1 picks a distribution a over coordinates
    # Player 2 picks a lag k
    # Payoff for (a, k) is conv_a[k] = sum_i a[i]*a[k-i]
    # We want to minimize max_k conv_a[k]
    
    # In fictitious play, Player 1 plays a best response to Player 2's empirical distribution
    # and Player 2 plays a best response to Player 1's empirical distribution.
    
    # Player 2's best response to a: pick k that maximizes conv_a[k]
    # Player 1's best response to q (distribution over k): minimize sum_k q[k]*conv_a[k] = a^T (sum_k q[k] B_k) a
    # This is a quadratic form, which we can optimize with coordinate descent.

        emp_k[k_best] += alpha

            # W is symmetric: W[i,j] = emp_k[i+j] = W[j,i] ✓
            
            # Coordinate descent on a^T W a

                ai = a_br[i]
                if ai < 1e-15:
                    continue

                # = sum_{j'} a[j']*W[i,j'] + sum_{i'} a[i']*W[i',i]
                # = 2 * sum_{j'} a[j']*W[i,j'] (by symmetry)

                j_valid = np.arange(n)
                k_vals = i + j_valid
                k_valid = k_vals <= 2*n-2
                w_row = np.where(k_valid, emp_k[np.minimum(k_vals, 2*n-2)], 0.0)
                # Actually: W[i,j] = emp_k[i+j] for i+j in [0, 2n-2]

def perturb_and_optimize(a0, rng, time_limit, label=""):
    """Perturb the sequence and then optimize."""
    n = len(a0)
    
    # Generate perturbation
    mode = rng.integers(0, 4)
    if mode == 0:
        # Gaussian noise
        noise = rng.standard_normal(n) * 0.1 * a0.max()
        a_new = a0 + noise
    elif mode == 1:
        # Multiplicative noise
        a_new = a0 * (1.0 + rng.standard_normal(n) * 0.15)
    elif mode == 2:
        # Zero out some elements and redistribute
        mask = rng.random(n) > 0.15
        a_new = a0 * mask
        # Redistribute to zeros
        n_zeros = (~mask).sum()
        if n_zeros > 0:
            a_new[~mask] = a0[mask].mean() * rng.random(n_zeros) * 0.5
    else:
        # Coordinate swap perturbation
        a_new = a0.copy()
        n_swaps = rng.integers(5, n // 10)
        for _ in range(n_swaps):
            i, j = rng.integers(0, n, size=2)
            a_new[i], a_new[j] = a_new[j], a_new[i]
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy(), 2.0 * n * fast_convolve(a0).max()
    a_new = a_new / s
    
    # Optimize
    a_opt, score = coord_descent_exact(a_new, time_limit, label=label)
    return a_opt, score

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
