```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. QP-based fictitious play with FFT-accelerated Hankel mat-vec (main innovation)
2. LogSumExp L-BFGS with aggressive multi-stage annealing
3. Adaptive length exploration with interpolation
4. Direct coordinate descent / subgradient polishing
Key: The QP best response min a^T Q(q) a s.t. sum(a)=1, a>=0 
is solved via FISTA with O(n log n) per iteration using the Hankel structure.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def hankel_matvec(q, a):
    """Compute Q*a where Q[i,j] = q[i+j], using FFT-based convolution."""
    n = len(a)
    b = a[::-1]
    total_len = len(q) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    Q = rfft(q, fft_size)
    B = rfft(b, fft_size)
    conv = irfft(Q * B, fft_size)
    return conv[n-1:n-1+n]

def project_simplex(v):
    """Project vector onto probability simplex (non-negative, sum=1)."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_qp_fista(q, a_init, n_fista_iters=2000, tol=1e-10):
    """
    Solve min a^T Q(q) a s.t. sum(a)=1, a>=0 using FISTA.
    Q(i,j) = q(i+j). Matrix-vector product via FFT.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    t = 1.0
    
    # Estimate Lipschitz constant
    # L = 2 * ||Q||_2, estimated via power iteration
    v = a.copy()
    for _ in range(10):
        v = hankel_matvec(q, v)
        nv = np.linalg.norm(v)
        if nv < 1e-15:
            break
        v = v / nv
    q_norm = abs(v @ hankel_matvec(q, v))
    L = max(2.0 * q_norm, 1e-10)
    step = 1.0 / L
    
    for k in range(n_fista_iters):
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        if k > 0:
            momentum = (t - 1.0) / t_new
            y = a + momentum * (a - a_prev)
        else:
            y = a
            t = t_new
            continue
        
        # Gradient step
        grad = 2.0 * hankel_matvec(q, y)
        a_new = project_simplex(y - step * grad)
        
        # Check convergence
        diff = np.abs(a_new - a).max()
        
        a_prev = a
        a = a_new
        t = t_new
        
        if diff < tol:
            break
    
    return a

def qp_fictitious_play(a_init, time_limit, n_inner_iters=1000):
    """
    QP-based fictitious play:
    - Maintain distribution q over lags
    - Player 1's best response: solve QP min a^T Q(q) a
    - Player 2's best response: k* = argmax (a*a)[k]
    - Update q using running average
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    # Initialize q uniformly
    q = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Running average of q for convergence
    q_avg = q.copy()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Phase 1: Solve QP for a given q_avg
        a_qp = solve_qp_fista(q_avg, a, n_fista_iters=n_inner_iters)
        
        # Evaluate
        b = fast_convolve(a_qp)
        score = 2.0 * n * b.max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_qp.copy()
        
        # Phase 2: Update q with best response
        k_star = int(np.argmax(b))
        e_k = np.zeros(2 * n - 1)
        e_k[k_star] = 1.0
        
        # Fictitious play update (empirical average)
        alpha = 1.0 / (it + 1)
        q_avg = (1.0 - alpha) * q_avg + alpha * e_k
        
        # Also maintain a "recent" q for exploration
        if it < 100:
            q = e_k.copy()  # Start with pure best response
        else:
            q = (1.0 - 0.1) * q + 0.1 * e_k
        
        a = a_qp
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP-FP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror Descent Game (Multiplicative Weights) with FFT-accelerated Hankel mat-vec (main novel approach)
2. QP-based fictitious play with FISTA solver exploiting Hankel structure
3. LogSumExp L-BFGS with aggressive multi-stage annealing
4. Peak equalization polish
5. Adaptive multi-length exploration with interpolation
Key innovation: The zero-sum game structure (Player 1: sequence a, Player 2: lag k)
solved via multiplicative weights with O(n log n) per iteration using the Hankel
matrix structure of the expected gradient.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def hankel_matvec(q, a):
    """Compute Q*a where Q[i,j] = q[i+j], using FFT-based convolution. O(n log n)."""
    n = len(a)
    b = a[::-1]
    total_len = len(q) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    Qf = rfft(q, fft_size)
    Bf = rfft(b, fft_size)
    conv = irfft(Qf * Bf, fft_size)
    return conv[n - 1:n - 1 + n]


def project_simplex(v):
    """Project vector onto probability simplex (non-negative, sum=1)."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert sequence to log-space (for softmax parameterization)."""
    return np.log(np.maximum(a, 1e-300))


def solve_qp_fista(q, a_init, n_iters=1000, tol=1e-12):
    """
    Solve min a^T Q(q) a s.t. sum(a)=1, a>=0 using FISTA.
    Exploits Hankel structure: Q[i,j] = q[i+j], mat-vec in O(n log n).
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Add small regularization to ensure PSD-ness
    q_reg = q.copy()
    reg = 0.001
    q_reg[0] += reg  # This adds reg*I to Q since Q[i,i] = q[2i]... 
    # Actually, to add reg*I we'd need to modify the mat-vec. Let's just add to the gradient.

    a_prev = a.copy()
    t = 1.0

    # Estimate step size via power iteration
    v = np.ones(n) / np.sqrt(n)
    for _ in range(20):
        v_new = 2.0 * hankel_matvec(q, v)
        nv = np.linalg.norm(v_new)
        if nv < 1e-15:
            break
        v = v_new / nv
    eigenval = abs(v @ (2.0 * hankel_matvec(q, v)))
    L = max(eigenval, 1e-8)
    step = 1.0 / L

    for k in range(n_iters):
        if k > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            momentum = (t - 1.0) / t_new
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a
            else:
                y = y / sy
        else:
            t_new = 1.0
            y = a

        grad = 2.0 * hankel_matvec(q, y)
        a_new = project_simplex(y - step * grad)

        diff = np.abs(a_new - a).max()
        a_prev = a
        a = a_new
        t = t_new

        if diff < tol:
            break

    return a


def mirror_descent_game(a_init, time_limit, eta=0.05, use_adaptive_eta=True):
    """
    Mirror descent / Multiplicative Weights for the zero-sum game:
    - Player 1: choose a in simplex (sequence)
    - Player 2: choose k in {0,...,2n-2} (lag)
    - Payoff: (a*a)[k]
    
    Uses O(n log n) per iteration via FFT-accelerated Hankel mat-vec.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Player 2's strategy (distribution over lags)
    q = np.ones(2 * n - 1) / (2 * n - 1)

    # Track running averages
    a_avg = a.copy()
    q_avg = q.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iter_count = 0

    for t in range(5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Adaptive learning rate
        if use_adaptive_eta:
            eta_t = eta / (1.0 + 0.0001 * t)
        else:
            eta_t = eta

        # Compute convolution
        b = fast_convolve(a)
        score = 2.0 * n * b.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Update Player 2 (q): multiplicative weights
        # q[k] proportional to q[k] * exp(-eta * b[k])
        log_q = np.log(np.maximum(q, 1e-300)) - eta_t * b
        log_q -= log_q.max()
        q_new = np.exp(log_q)
        q_sum = q_new.sum()
        if q_sum < 1e-300:
            q = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            q = q_new / q_sum

        # Update Player 1 (a): multiplicative weights with expected gradient
        # grad[i] = 2 * sum_k q[k] * a[k-i] = 2 * hankel_matvec(q, a)[i]
        grad = 2.0 * hankel_matvec(q, a)
        log_a = np.log(np.maximum(a, 1e-300)) - eta_t * grad
        a = softmax_safe(log_a)

        # Update running averages
        alpha = 1.0 / (t + 1)
        a_avg += alpha * (a - a_avg)
        q_avg += alpha * (q - q_avg)

        iter_count = t + 1

        # Periodic evaluation of running average
        if (t + 1) % 2000 == 0:
            b_avg = fast_convolve(a_avg)
            score_avg = 2.0 * n * b_avg.max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()
            if (t + 1) % 20000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MD-Game iter={t+1} ({elapsed:.0f}s): cur={score:.6f}, avg={score_avg:.6f}, best={best_score:.6f}")

    # Final: also try solving QP with q_avg for a potentially better answer
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 1.0:
        a_qp = solve_qp_fista(q_avg, a_avg, n_iters=500)
        b_qp = fast_convolve(a_qp)
        score_qp = 2.0 * n * b_qp.max()
        if score_qp < best_score:
            best_score = score_qp
            best_a = a_qp.copy()

    return best_a, best_score


def qp_fictitious_play(a_init, time_limit, n_inner=500):

    t0 = time.monotonic()

            break

        b = fast_convolve(a_qp)

            best_score = score
            best_a = a_qp.copy()

        # Update q_avg (fictitious play: empirical average of best responses)
        k_star = int(np.argmax(b))
        e_k = np.zeros(2 * n - 1)
        e_k[k_star] = 1.0

        if (it + 1) % 500 == 0:

def lse_objective_and_grad(z, T_val, n):
    """Compute LogSumExp-smoothed objective and gradient in log-space parameterization."""
    a = softmax_safe(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_vals = np.exp((b - max_b) / T_val)
    log_sum = max_b / T_val + np.log(exp_vals.sum())
    obj = T_val * log_sum

    # Gradient with respect to z
    w = exp_vals / exp_vals.sum()
    grad_a = 2.0 * hankel_matvec(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z * (2.0 * n)

            return obj

        def jac(z_in):
            _, g = lse_objective_and_grad(z_in, stage_T, n)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iters, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': max_iters * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if (si + 1) % 5 == 0:
            a_check = softmax_safe(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE-LBFGS stage {si+1}/{n_stages} ({elapsed:.0f}s): T={stage_T:.3f}, score={sc:.6f}")

    return softmax_safe(z)


def peak_equalize(a_init, time_limit, n_steps=50000):
    """Directly reduce the peak autocorrelation value via targeted adjustments."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]

        src_local = int(np.argmax(contrib))
        src = idx[src_local]

        # Find a good destination (low contribution to peaks)
        b_sorted = np.argsort(b)[::-1]
        top_peaks = b_sorted[:min(10, len(b_sorted))]
        peak_zones = set()
        for pk in top_peaks:
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            peak_zones.update(range(js2, je2 + 1))

        non_peak = [j for j in range(n) if j not in peak_zones]
        if non_peak:
            # Pick from non-peak zones with low current value
            candidates = sorted(non_peak, key=lambda j: a[j])[:20]
        else:
            candidates = [j for j in range(n) if j != src]

        dst = candidates[rng.integers(len(candidates))]

        # Try several transfer fractions
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            # Random perturbation to escape
            if no_improve > 50:
                k = rng.integers(max(2, n // 20), n // 5)
                idx_p = rng.choice(n, k, replace=False)
                a_trial = a.copy()
                a_trial[idx_p] += rng.normal(0, 0.02, k)
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score + 0.01:  # Allow small temporary increase
                        a = a_trial
                no_improve = 0

        if no_improve > 2000:
            break

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq step={step+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2 * (1 + 0.15 * np.cos(16 * np.pi * x))
    else:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(2, 10) * np.linspace(0, 1, n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def subgradient_polish(a_init, time_limit):
    """Polish using active-set subgradient with backtracking."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Backtracking line search
        improved = False
        alpha = lr
        for bt in range(8):
            z_new = z - alpha * g
            a_new = softmax_safe(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 1.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.9, 0.01)
            if no_improve >= 50:
                rng = np.random.default_rng(step * 7 + 1)
                noise = rng.random(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.1

        if no_improve > 500:
            break

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple optimization strategies at the parent
    length and nearby lengths, returning the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_limit_global = t_global_start + max(10.0, float(budget_s) - 40.0)

    def time_remaining():
        return time_limit_global - time.monotonic()

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    n_base = len(parent) if parent is not None else 1200

    if parent is not None:
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Parent: n={n_base}, score={best_score:.6f}")
    else:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s) ***")

    n = n_base

    # === Phase 1: Mirror Descent Game (main workhorse) ===
    print(f"\n=== Phase 1: Mirror Descent Game (n={n}, budget from remaining) ===")
    rem = time_remaining()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_md, sc_md = mirror_descent_game(best_seq, t1, eta=0.05)
        try_update(a_md, "MD-game")
        # Also try with different eta
        rem = time_remaining()
        t1b = min(rem * 0.15, 200.0)
        if t1b > 20:
            a_md2, sc_md2 = mirror_descent_game(best_seq, t1b, eta=0.1)
            try_update(a_md2, "MD-game-eta0.1")

    # === Phase 2: QP Fictitious Play ===
    print(f"\n=== Phase 2: QP Fictitious Play (n={n}) ===")
    rem = time_remaining()
    t2 = min(rem * 0.25, 350.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_qp, sc_qp = qp_fictitious_play(best_seq, t2, n_inner=300)
        try_update(a_qp, "QP-FP")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (n={n}) ===")
    rem = time_remaining()
    t3 = min(rem * 0.20, 250.0)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a_lse = lse_lbgfs(best_seq, t3, T_start=200.0, T_end=0.005, n_stages=15)
        try_update(a_lse, "LSE-LBFGS")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient Polish (n={n}) ===")
    rem = time_remaining()
    t4 = min(rem * 0.15, 150.0)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s")
        a_sg, sc_sg = subgradient_polish(best_seq, t4)
        try_update(a_sg, "SubGrad")

    # === Phase 5: Peak equalization ===
    print(f"\n=== Phase 5: Peak Equalization (n={n}) ===")
    rem = time_remaining()
    t5 = min(rem * 0.15, 120.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        a_pe, sc_pe = peak_equalize(best_seq, t5)
        try_update(a_pe, "PeakEq")

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = time_remaining()
    alt_lengths = []
    for mult in [1.2, 1.5, 0.8]:
        alt_n = int(n_base * mult)
        if alt_n != n_base and 400 <= alt_n <= 3000:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:2]:
        rem = time_remaining()
        if rem < 60:
            break
        t_alt = min(rem * 0.2, 120.0)
        if t_alt < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_init = evaluate_safe(init_alt)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_alt.copy()
            print(f"    Interpolated init already better: {sc_init:.6f}")

        # Quick MD game at new length
        a_alt, _ = mirror_descent_game(init_alt, t_alt * 0.6, eta=0.05)
        try_update(a_alt, f"MD-n{alt_n}")

        rem = time_remaining()
        if rem > 20:
            a_alt2 = lse_lbgfs(best_seq, min(rem * 0.4, 60.0), T_start=50.0, T_end=0.01, n_stages=8)
            try_update(a_alt2, f"LSE-n{alt_n}")

    # === Phase 7: Final polish on best ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_remaining()
    if rem > 20:
        a_fin, _ = peak_equalize(best_seq, min(rem * 0.5, 60.0))
        try_update(a_fin, "Final-PeakEq")

    rem = time_remaining()
    if rem > 15:
        a_fin2, _ = subgradient_polish(best_seq, min(rem * 0.5, 40.0))
        try_update(a_fin2, "Final-SubGrad")

    rem = time_remaining()
    if rem > 10:
        a_fin3, _ = mirror_descent_game(best_seq, min(rem * 0.6, 30.0), eta=0.03)
        try_update(a_fin3, "Final-MD")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed_total = time.monotonic() - t_global_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, total_time={elapsed_total:.0f}s)")
    return [float(x) for x in best_seq]
```
