"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated gradient methods,
adaptive temperature annealing, momentum, peak-shaving, and simulated annealing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_fft(a, w):
    """Compute v[i] = sum_{d=0}^{n-1} a[d] * w[d+i] for i = 0, ..., n-1.
    w has length 2n-1."""
    n = len(a)
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    return conv[n-1:2*n-1]

def lse_objective_and_grad(a, T):
    """Compute LogSumExp smoothed objective and its gradient.
    Objective: (2n/T) * log(sum_k exp(T * b[k])) where b = convolve(a, a)
    Gradient: 4n * v where v[i] = sum_k w[k] * a[k-i] * I(0<=k-i<n)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_Tb = np.exp(T * (b - max_b))
    exp_sum = exp_Tb.sum()
    if exp_sum < 1e-300:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    else:
        w = exp_Tb / exp_sum
    obj = (2.0 * n / T) * (max_b + np.log(exp_sum))
    
    # Gradient
    v = cross_corr_fft(a, w)
    grad = 4.0 * n * v
    
    return obj, grad, b, w

def project_simplex(a):
    """Project a onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    # Find the largest threshold t such that sum(max(a - t, 0)) = 1
    sorted_a = np.sort(a)[::-1]
    cumsum = np.cumsum(sorted_a)
    # Find rho: largest index such that sorted_a[rho] > (cumsum[rho] - 1) / (rho + 1)
    cond = sorted_a > (cumsum - 1.0) / np.arange(1, n + 1)
    rho = np.argmax(~cond)  # first index where cond is False
    if rho == 0:
        return np.ones(n) / n
    t = (cumsum[rho - 1] - 1.0) / rho
    return np.maximum(a - t, 0.0)

def momentum_lse_descent(a0, time_limit, T_initial=20.0, T_final=0.05,
                         momentum0=0.9, print_interval=200):
    """Momentum-based gradient descent on LogSumExp smoothed objective.
    Adaptive temperature cooling.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    v_mom = np.zeros(n)
    t0 = time.monotonic()
    step = 0
    
    # Total number of steps we'll do
    # Estimate: each step takes about O(n log n) time
    # For n=1200, that's maybe 0.1ms per step, so we can do ~10000 steps in 1s
    # In 600s, that's ~6 million steps (optimistic)
    # Let's be conservative
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential temperature cooling
        T = T_initial * (T_final / T_initial) ** progress
        T = max(T, T_final)
        
        obj, grad, b, w = lse_objective_and_grad(a, T)
        
        # Adaptive learning rate: Barzilai-Borwein style
        # Use the norm of the gradient to scale the step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Learning rate: start moderate, decrease as T decreases
        lr = 0.1 / (1.0 + 0.1 * T)
        
        # Momentum update
        v_mom = momentum0 * v_mom - lr * grad
        a_new = a + v_mom
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        step += 1
        
        if step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"      MOLSE {elapsed:.0f}s step={step} T={T:.4f} score={best_score:.8f} grad_norm={grad_norm:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated multiplicative
gradient descent (mirror descent), adaptive temperature annealing, LP-based
refinement, and peak-directed coordinate descent.

Key innovation: O(n log n) gradient computation via FFT cross-correlation,
enabling 10-100x more optimization iterations than the previous O(n^2) approach.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(a, w):
    """Compute v[i] = sum_{d=0}^{n-1} a[d] * w[d+i] for i = 0,...,n-1.
    w has length 2n-1."""
    n = len(a)
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(A * W, fft_size)
    return conv_full[n - 1: 2 * n - 1]


def compute_lse_gradient(a, T):
    """Compute gradient of (2n/T)*logsumexp(T*b) where b=convolve(a,a).
    Returns (objective_value, gradient, b, w)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    # Avoid overflow
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        obj = 2.0 * n * max_b
    else:
        w = exp_shifted / exp_sum
        obj = (2.0 * n / T) * (T * max_b + np.log(exp_sum))
    
    v = cross_corr_fft(a, w)
    grad = 4.0 * n * v
    return obj, grad, b, w


def score_fn(a):
    """Compute the actual score 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find rho: largest k such that sorted_v[k-1] > (cumsum[k-1] - 1) / k
    # (1-indexed in the formula)
    rho_candidates = (sorted_v * np.arange(1, n + 1) > (cumsum - 1.0))
    rho = int(np.sum(rho_candidates))
    if rho == 0:
        return np.ones(n) / n
    theta = (cumsum[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def mirror_descent(a0, time_limit, T_start=30.0, T_end=0.03, print_every=500):
    """Mirror descent (multiplicative weight) on LogSumExp-smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Exponential temperature schedule
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient
        obj, grad, b, w = compute_lse_gradient(a, T)
        
        # Adaptive learning rate
        # The gradient magnitude is roughly 4n * max(a) * something
        # For normalized a, max(a) ≈ 1/support_size
        # We want lr * |grad| to be O(1) for a good step
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-15:
            break
        lr = 0.3 / (grad_max + 1e-10)
        
        # Multiplicative update (mirror descent with entropy regularization)
        log_update = -lr * grad
        log_update = np.clip(log_update, -20.0, 20.0)  # Prevent overflow
        a_new = a * np.exp(log_update)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        new_score = score_fn(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # If the step didn't help, try a smaller step
            lr2 = lr * 0.5
            log_update2 = np.clip(-lr2 * grad, -20.0, 20.0)
            a_new2 = a * np.exp(log_update2)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-300:
                a_new2 = a_new2 / s_new2
            if score_fn(a_new2) < best_score:
                best_score = score_fn(a_new2)
                best_a = a_new2.copy()
                a = a_new2
            # If still no improvement, just accept the step (for exploration)
            # but keep best_a
            a = a_new  # Move forward anyway to escape local optima
        
        step += 1
        if step % print_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      MirrorDesc {elapsed:5.0f}s step={step:6d} T={T:.5f} best={best_score:.8f} cur={score_fn(a):.8f}")
    
    return best_a, best_score


def momentum_descent(a0, time_limit, T_start=20.0, T_end=0.05):
    """Additive gradient descent with Nesterov momentum on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v_mom = np.zeros(n)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    step = 0
    momentum = 0.85
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        obj, grad, b, w = compute_lse_gradient(a, T)
        
        # Nesterov momentum
        v_mom = momentum * v_mom - 0.05 / (1.0 + T / 10.0) * grad
        a_trial = a + v_mom
        
        # Project to simplex
        a_trial = project_simplex(a_trial)
        
        new_score = score_fn(a_trial)
        if new_score < best_score:
            best_score = new_score
            best_a = a_trial.copy()
        a = a_trial
        step += 1
        
        if step % 500 == 0 and elapsed > 5:
            print(f"      MomDesc {elapsed:5.0f}s step={step:6d} T={T:.5f} best={best_score:.8f}")
    
    return best_a, best_score


def lp_refine(a0, time_limit):
    """LP-based refinement using CVXPY."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    # Find a working solver
    solver = None
    for sol_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sol_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = sol_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    print(f"      LP solver: {solver}")
    
    y = a.copy()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP: max sum(b) s.t. conv(a, b) <= M*1, b >= 0
        M = fast_convolve(a).max()
        
        b_var = cp.Variable(n, nonneg=True)
        size = 2 * n - 1
        T_mat = np.zeros((size, n))
        for j in range(n):
            T_mat[j:j + n, j] = a
        
        constraints = [T_mat @ b_var <= M, cp.sum(b_var) == 1.0]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False, max_iter=500)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            break
        b_val = b_val / sb
        
        # Nesterov update
        alpha = 2.0 / (iteration + 2)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Primal average
        a = ((iteration + 1) * a + alpha * (iteration + 2) * b_val) / (iteration + 1 + alpha * (iteration + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        sc = score_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = score_fn(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        iteration += 1
        if iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP {elapsed:5.0f}s iter={iteration:4d} M={M:.8f} best={best_score:.8f}")
    
    return best_a, best_score
