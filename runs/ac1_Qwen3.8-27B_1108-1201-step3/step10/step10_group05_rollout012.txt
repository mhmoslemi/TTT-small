import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.zeros(n)

def build_conv_matrix(a):

    n = len(a)
    T = np.zeros((2 * n - 1, n))

def spectral_init(n, rng, num_components=8):
    """Generate a sequence with approximately flat power spectrum, projected to non-negative."""
    # Random phases
    phases = rng.uniform(-np.pi, np.pi, n)
    # Target flat spectrum
    target_mag = np.ones(n)
    target_mag[0] = 0  # zero mean
    # Inverse FFT
    spectrum = target_mag * np.exp(1j * phases)
    a_raw = np.real(irfft(spectrum, n)) * n  # Scale
    # Project to non-negative simplex
    a_raw = np.maximum(a_raw, 0)
    s = a_raw.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_raw / s

    # Polyak averaging
    avg_a = np.zeros(n)
    
    # History for extrapolation
    a_prev = a.copy()
    
    while time.monotonic() - t0 < time_limit:

        elapsed = time.monotonic() - t0

            peak_idx = np.where(conv_a >= 0.95 * M)[0]
            grad = np.zeros(n)
            for k in peak_idx:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                j = np.arange(i_start, i_end + 1)
                grad[j] += a[k - j]
            grad /= len(peak_idx)

            # Extrapolation (Nesterov-like)

            else:
                a_new = b
            
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:

            else:
                a_new = a.copy()
            
            a_prev = a.copy()
            a = a_new
            
            # Polyak average
            avg_a = ((iters - 1) / iters) * avg_a + (1.0 / iters) * a
        
        # Evaluate

        if score < best_score - 1e-14:

**Key insight**: The LP fictitious play converges to the value of the zero-sum game. The optimal sequence is a Nash equilibrium. At the Nash equilibrium, the LP best response should satisfy max(conv(a*, a*)) = max(conv(a*, b*)) for all b (i.e., a* is also a best response to itself).

So the algorithm should:
1. Iterate: solve LP to get best response b to current a
2. Update a towards b
3. Stop when a ≈ b (convergence)
4. At convergence, a is approximately optimal

The rate of convergence of fictitious play is O(1/sqrt(T)), so we need many iterations. With a budget of ~1000 seconds and each LP taking ~1-5 seconds, we can get ~200-1000 iterations.

To speed up convergence:
- Use **extrapolation** (Nesterov acceleration): a_{t+1} = a_t + alpha_t * (b_t - a_t) + momentum * (a_t - a_{t-1})
- Use **adaptive step size**: alpha_t = 2/(t+2) (Polyak) or alpha_t = 1/sqrt(t)

Let me also think about a completely different approach: **direct optimization with L-BFGS**.

The objective f(a) = max_k conv(a,a)[k] is convex in a (since it's the max of linear functions... wait, no, conv(a,a) is quadratic, not linear). So f(a) is not convex.

- Actually, conv(a,a)[k] = sum_i a[i]*a[k-i] is a quadratic form. It's not convex in general (the Hessian is not PSD).

So direct optimization is harder. The LP fictitious play is the right approach.

**Idea: Use the dual of the LP.**

The dual is:
maximize sum(y) + z  (where y >= 0 are dual vars for T@b <= M, z is dual var for sum(b)=1)
subject to T^T @ y + z * 1 <= 0  (n constraints)
         M * sum(y) + z = 1  (dual of... hmm, this is getting complicated)

Actually, let me write it more carefully. The primal is:
min_{b, M} M
s.t. T b - M * 1 <= 0  (dual var y >= 0)
     sum(b) = 1  (dual var z, free)
     b >= 0

The Lagrangian is:
L = M + y^T (T b - M * 1) + z (sum(b) - 1) - mu^T b
  = M(1 - sum(y)) + b^T (T^T y + z * 1 - mu) - z

For the dual function to be bounded below, we need:
1 - sum(y) = 0  =>  sum(y) = 1
T^T y + z * 1 - mu = 0  =>  mu = T^T y + z * 1 >= 0  (since mu >= 0)

So the dual is:
maximize -z  (since L at optimum = -z when sum(y)=1)
subject to sum(y) = 1
           T^T y + z * 1 >= 0  =>  z <= -min_j (T^T y)_j
           y >= 0

maximize -z subject to z <= -min_j (T^T y)_j
= maximize min_j (T^T y)_j  (since we want to maximize -z, and z can be as large as -min_j (T^T y)_j)
= maximize min_j (T^T y)_j
subject to sum(y) = 1, y >= 0

Where (T^T y)_j = sum_k T[k,j] * y[k] = sum_k a[k-j] * y[k] (for valid k-j).

So the dual is:
maximize min_j (sum_k a[k-j] * y[k])
subject to sum(y) = 1, y >= 0

This is also a zero-sum game! Player 1 chooses y (a probability distribution over the 2n-1 convolution positions), and Player 2 chooses j (a coordinate of b). The payoff is (T^T y)_j = the "weighted sum" of a shifted by j.

This is symmetric to the primal (by the minimax theorem), which makes sense.

The dual gives us a way to certify the optimality: if we find y such that min_j (T^T y)_j >= V, then V is a lower bound on the primal value.

    n = len(a)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)

        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.zeros(n)

def fine_coord_descent(a0, time_limit, label=""):
    """Fine coordinate descent with small multiplicative adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end+1:].max()))

            affected = conv[k_start:k_end+1]
            num_affected = k_end - k_start + 1
            coeffs = a[:num_affected] + ai
            quad_offset = 2 * i - k_start
            has_quad = (0 <= quad_offset < num_affected)

            best_delta = 0.0
            best_new_max = M

            # Very fine multiplicative steps
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97,
                          1.001, 1.002, 1.005, 1.01, 1.02, 1.03]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_aff = affected + delta * coeffs
                if has_quad:
                    new_aff = new_aff.copy()
                    new_aff[quad_offset] += delta * delta
                new_max = max(unaff_max, float(np.max(new_aff)))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end+1] += best_delta * coeffs
                if has_quad:
                    conv[2*i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score
