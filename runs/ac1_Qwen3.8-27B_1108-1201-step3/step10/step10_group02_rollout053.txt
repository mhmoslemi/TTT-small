def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ b = convolve(f, b)."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for i in range(n):
        A[i:i+n, i] = f
    return A

def solve_best_response_lp(f, solver=None):
    """Solve: min M s.t. (f*b)_k <= M, b >= 0, sum(b) = 1"""
    n = len(f)

    M = cp.Variable()

    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solvers = [cp.SCS, cp.CLARABEL, cp.ECOS]
    else:
        solvers = [solver]
    
    for s in solvers:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                bs = b_val.sum()
                if bs < 1e-15:

                b_val = b_val / bs
                return b_val, float(M.value)
        except Exception:
