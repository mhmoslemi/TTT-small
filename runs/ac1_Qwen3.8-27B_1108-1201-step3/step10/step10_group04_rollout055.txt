def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def solve_lp_best_response(a_fixed, solver=None, warm_start=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def nesterov_lp_clean(a0, time_limit, solver=None, n_max_iters=200000,
                      verbose=True):
    """
    Clean Nesterov-accelerated LP best-response iteration.
    Uses proper Nesterov momentum: y_t = x_t + beta_t*(x_t - x_{t-1})
    with beta_t = (t-1)/(t+1) for the standard schedule.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    # Initialize both players
    x = a.copy()  # current a
    x_prev = a.copy()  # previous a
    y = a.copy()  # "extrapolated" point for LP

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()

    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if it == 1:
            y = x.copy()
        else:
            beta = (it - 1.0) / (it + 1.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Solve LP: find best response to y
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Update x with averaging step
        alpha = 2.0 / (it + 2.0)
        x_prev = x.copy()
        x = (1.0 - alpha) * x + alpha * b_new
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            x = b_new.copy()
        else:
            x = x / sx

        # Track best
        score_x = 2.0 * n * fast_convolve(x).max()
        score_b = 2.0 * n * fast_convolve(b_new).max()

        if score_x < best_score - 1e-12:
            best_score = score_x
            best_a = x.copy()
            stall_count = 0
        if score_b < best_score - 1e-12:
            best_score = score_b
            best_a = b_new.copy()
            stall_count = 0

        # Stall detection and restart
        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1

            if stall_count >= 15:
                # Perturb to escape
                noise = rng.random(n) * 0.05
                x = x + noise
                x = np.maximum(x, 0)
                x = x / x.sum()
                x_prev = x.copy()
                stall_count = 0
                if verbose:
                    print(f"    NesterovLP restart at iter={it}, score={best_score:.6f}")

        if verbose and (it % 5000 == 0 or it == 1):
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def lbgfs_logsumexp(a0, time_limit, T_start=500.0, T_end=0.01, n_stages=15):
    """L-BFGS on LogSumExp with cosine annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 50.0)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 50.0)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        # Vectorized gradient computation
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    # Cosine annealing schedule
    temps = []
    for i in range(n_stages):
        frac = i / (n_stages - 1)
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * frac))
        temps.append(T)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        max_iter = max(30, int(200 * (T_start / stage_T) ** 0.3))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': max_iter * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if si % 3 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def mirror_descent_subgrad(a0, time_limit, n_active_frac=0.1):
    """Mirror descent on the simplex with subgradient and backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0
    rng = np.random.default_rng(42)

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: average of shifted a over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Remove mean (project gradient onto tangent space of simplex)
        g = g - g.mean()

        # Backtracking line search
        alpha = lr
        improved = False
        for bt in range(8):
            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.1, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.85, 0.005)
            if no_improve >= 50:
                noise = rng.random(n) * 0.02
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.1

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_steps=100000):
    """Coordinate descent: try adjusting pairs of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(31415)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick two random coordinates
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # Try moving mass from i to j (and vice versa)
        best_local = None
        best_local_score = best_score

        for direction in [1, -1]:
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac * direction
                if a[i] + amt < -1e-15 or a[j] - amt < -1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] += amt
                a_trial[j] -= amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score - 1e-14:
                    best_local_score = score
                    best_local = a_trial.copy()

        if best_local is not None:
            a = best_local
            best_score = best_local_score
            best_a = a.copy()

    return best_a, best_score


def peak_deflation(a0, time_limit, n_steps=50000):

    For the peak at position k*, the contribution to (a*a)_{k*} from element j is a_j * a_{k*-j}.
    We reduce the largest contributors.

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

        k_star = int(np.argmax(b))

        # Find contributions to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]

        # Find the top contributor
        top_local = np.argmax(contrib)
        src_global = idx[top_local]

        # Positions that don't contribute to the peak: those outside [js, je]

            dst_global = idx[dst_local]
        else:
            dst_global = rng.integers(n)

                dst_global = (dst_global + 1) % n

        # Try to move a small amount

            amt = a[src_global] * frac
            a_trial = a.copy()
            a_trial[src_global] -= amt
            a_trial[dst_global] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Clean Nesterov-accelerated LP best-response with adaptive step and warm-starting
2. LogSumExp L-BFGS with cosine annealing
3. Mirror descent with subgradient and backtracking line search
4. Peak deflation: directly targets the maximum autocorrelation peak
5. Coordinate descent local search
6. Multi-length exploration with interpolation and perturbed restarts
Key improvements over baseline: proper Nesterov momentum schedule, adaptive alpha,
peak-aware local search, and systematic restart strategy to escape local minima.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_clean(a0, time_limit, solver=None, n_max_iters=200000, verbose=False):
    """
    Clean Nesterov-accelerated LP best-response.
    Standard Nesterov: y_t = x_t + beta_t*(x_t - x_{t-1}), beta_t = (t-1)/(t+1)
    Then x_{t+1} = (1-alpha_t)*x_t + alpha_t*BR(y_t), alpha_t = 2/(t+2)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    x = a.copy()
    x_prev = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(1, n_max_iters + 1):

            y = x.copy()

            y = np.maximum(y, 0)

def lbgfs_logsumexp(a0, time_limit, T_start=300.0, T_end=0.05, n_stages=12):
    """L-BFGS on LogSumExp with cosine annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 40.0)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 40.0)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    # Cosine annealing
    temps = []
    for i in range(n_stages):
        frac = i / (n_stages - 1)
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * frac))
        temps.append(T)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        max_iter = max(20, int(150 * (1 + T_start / stage_T) ** 0.2))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': max_iter * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if si % 3 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def mirror_descent_subgrad(a0, time_limit, n_active_frac=0.08):
    """Mirror descent with subgradient and backtracking line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.15
    no_improve = 0
    rng = np.random.default_rng(789)

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g = g - g.mean()

        alpha = lr
        improved = False
        for bt in range(8):
            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.08, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.9, 0.01)
            if no_improve >= 40:
                noise = rng.random(n) * 0.015
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.15

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score
