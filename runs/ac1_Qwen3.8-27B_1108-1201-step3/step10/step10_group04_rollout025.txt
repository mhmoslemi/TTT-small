import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """FFT-based cross-convolution of a and b."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n matrix T such that T @ b = cross_conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(size, j + n)
        T[lo:hi, j] = a_fixed[max(0, n - 1 - (j - lo)):n - hi + j + 1]
    # Simpler construction
    T = np.zeros((size, n))
    for j in range(n):
        # (a*b)_k = sum_j a[k-j]*b[j], so T[k,j] = a[k-j] if valid
        ks = range(max(0, j), min(size, j + n))
        for k in ks:
            idx = k - j
            if 0 <= idx < n:
                T[k, j] = a_fixed[idx]
    return T


def build_conv_matrix_fast(a_fixed):
    """Faster construction of convolution matrix using broadcasting."""
    n = len(a_fixed)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = np.arange(size)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    T = np.where((idx >= 0) & (idx < n), a_fixed[np.clip(idx, 0, n-1)], 0.0)
    return T


def softmax_from_z(z):
    """Compute softmax of z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert a to log domain."""
    return np.log(np.maximum(a, 1e-300))

        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b * (1.0 - active_frac)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def mirror_descent_bt(a0, time_limit, n_active_frac=0.05):
    """Mirror descent with backtracking line search in log domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b = compute_subgradient(a, n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        alpha = lr
        improved = False
        for bt in range(12):
            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()

            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.85, 0.005)
            if no_improve >= 50:
                rng = np.random.default_rng(step * 31 + 7)
                noise = rng.standard_normal(n) * 0.05
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a = a_pert / sp
                    z = z_from_a(a)
                no_improve = 0
                lr = 0.5

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            if elapsed > 60:
                break

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver=None, n_max_iters=500000):
    """
    Nesterov-accelerated LP best-response optimization.
    Main workhorse for convergence.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    lp_fail_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            lp_fail_count += 1
            if lp_fail_count > 20:
                break
            continue
        lp_fail_count = 0

        # Nesterov step with optimal schedule
        alpha = 2.0 / ((it + 1.0) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Also track the "x" sequence (slightly different for Nesterov)
        if it == 0:
            a = b_new.copy()
        else:
            a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate both a and y
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Periodic stall detection and perturbation
        if it % 200 == 0 and it > 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                rng = np.random.default_rng(it * 13 + 37)
                # Gentle perturbation
                noise = rng.standard_normal(n) * 0.02
                y_pert = y + noise
                y_pert = np.maximum(y_pert, 0)
                sp = y_pert.sum()
                if sp > 1e-15:
                    y = y_pert / sp
                stall_count = 0

        # Periodic Nesterov restart (resets momentum for fresh convergence)
        if it % 5000 == 0 and it > 0:
            # Keep best as new starting point
            y = best_a.copy()
            a = best_a.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=15):
    """L-BFGS with LogSumExp approximation, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = np.clip(Tb.max(), 0, 700)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = np.clip(Tb.max(), 0, 700)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, 200 // n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 1000,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_flat_top_init(n, rng):
    """
    Generate a non-negative sequence with approximately flat power spectrum.
    Uses the idea of flat-top window design in the frequency domain.
    """
    method = rng.integers(0, 6)

    if method == 0:
        # Cosine-modulated with positive coefficients
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x) + 0.05 * np.cos(6 * np.pi * x)
    elif method == 1:
        # Raised cosine variant
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
        a = a ** 0.5  # Soften
        a = 1.0 - 0.3 * np.cos(2 * np.pi * x) + a * 0.3
    elif method == 2:
        # Inverse-FFT of flat magnitude spectrum
        freq = np.linspace(0, 1, n, endpoint=False)
        mag = 1.0 + 0.1 * np.cos(2 * np.pi * freq * 3) + 0.05 * np.sin(2 * np.pi * freq * 7)
        mag = np.maximum(mag, 0.1)
        phase = rng.standard_normal(n) * 0.3
        spectrum = mag * np.exp(1j * phase)
        a = np.real(np.fft.ifft(spectrum))
        # Make real and positive
        a = np.abs(a)
    elif method == 3:
        # Product of two smooth windows
        x = np.linspace(0, 1, n)
        w1 = 0.5 * (1 + np.cos(2 * np.pi * x))
        w2 = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.25))
        a = w1 + w2 + 0.5
    elif method == 4:
        # Gaussian convolution with itself (gives wider Gaussian)
        x = np.linspace(-2, 2, n)
        sigma1 = rng.uniform(0.5, 0.8)
        sigma2 = rng.uniform(0.3, 0.6)
        g1 = np.exp(-x**2 / (2 * sigma1**2))
        g2 = np.exp(-x**2 / (2 * sigma2**2))
        a = g1 * g2  # This is exp(-x^2/2 * (1/s1^2 + 1/s2^2))
        # Blend with uniform
        a = 0.5 * a / a.max() + 0.5
    else:
        # Multi-scale cosine
        x = np.linspace(0, 1, n)
        a = 1.0
        for k in range(1, 6):
            amp = 0.2 / k
            a += amp * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = a * (1 + 0.1 * np.cos(20 * np.pi * x))
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2) * (1 + 0.2 * np.cos(8 * x))
    elif choice == 8:
        # Flat-top window (sine)
        x = np.linspace(0, 1, n)
        a = (0.2156 - 0.4160 * np.cos(2 * np.pi * x) + 0.2781 * np.cos(4 * np.pi * x)
             - 0.08359 * np.cos(6 * np.pi * x) + 0.006947 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        # Blackman-Harris window
        x = np.linspace(0, 1, n)
        a = (0.35875 - 0.48829 * np.cos(2 * np.pi * x) + 0.14128 * np.cos(4 * np.pi * x)
             - 0.01168 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def local_search_peaks(a0, time_limit, n_steps=50000):
    """Local search targeting peak reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find the top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Try moving mass from peak contributors to non-contributing positions
        improved = False
        src_ranking = np.argsort(contrib)[::-1][:10]
        
        for si in src_ranking[:5]:
            src = idx[si]
            if a[src] < 1e-10:
                continue
            
            # Find good destinations (low contribution to peak)
            contrib_all = np.zeros(n)
            for j in range(n):
                k2 = peak_k
                if 0 <= j < n and 0 <= k2 - j < n:
                    contrib_all[j] = a[j] * a[k2 - j]
            
            # Sort by contribution (lowest first)
            dest_ranking = np.argsort(contrib_all)[:20]
            
            for di in dest_ranking[:5]:
                dst = di
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.08)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

    return best_a, best_score


def optimize_comprehensive(a_init, time_budget, solver, label=""):
    """
    Multi-phase optimization at a fixed length.
    Phase A: Mirror descent (quick)
    Phase B: LP Nesterov (main)
    Phase C: LogSumExp L-BFGS (refinement)
    Phase D: Local peak search (polish)
    """
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(len(a_init)) / len(a_init)
    best_a = best_a / best_a.sum()
    best_score = evaluate_safe(best_a)
    t0 = time.monotonic()
    n = len(best_a)

    def try_update(a, src_label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    NEW BEST [{src_label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase A: Quick mirror descent
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.12, 60.0)
    if t_a > 10:
        a_md, _ = mirror_descent_bt(best_a, t_a, n_active_frac=0.05)
        try_update(a_md, f"md_{label}")

    # Phase B: LP Nesterov (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.50, 600.0)
    if t_b > 20:
        a_lp, _ = nesterov_lp_optimize(best_a, t_b, solver=solver, n_max_iters=300000)
        try_update(a_lp, f"lp_{label}")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.20, 150.0)
    if t_c > 15:
        a_lb = lbgfs_logsumexp_optimize(best_a, t_c, T_start=50.0, T_end=0.05, n_stages=15)
        try_update(a_lb, f"lse_{label}")

    # Phase D: Local peak search
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.10, 60.0)
    if t_d > 10:
        a_ls, _ = local_search_peaks(best_a, t_d, n_steps=30000)
        try_update(a_ls, f"ls_{label}")

    # Phase E: Final LP polish
    rem = time_budget - (time.monotonic() - t0)
    if rem > 20:
        t_e = min(rem * 0.08, 40.0)
        if t_e > 5:
            a_lp2, _ = nesterov_lp_optimize(best_a, t_e, solver=solver, n_max_iters=50000)
            try_update(a_lp2, f"lp2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function that finds the best sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """

    end_time = start_time + max(1.0, float(budget_s) - 60.0)  # Safety margin
    
    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None

            parent = p / p.sum()
            p_score = evaluate_safe(parent)

    except Exception:

    if parent is None:

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):

            return
        a = np.maximum(a, 0).astype(np.float64)

        sc = evaluate_safe(a)

    n_base = len(best_seq)
    remaining = end_time - time.monotonic()
    print(f"\n=== Starting optimization (best={best_score:.6f}, n={n_base}, remaining={remaining:.0f}s) ===")

        rem = end_time - time.monotonic()

            sa = init_a.sum()

            else:
                init_a = best_seq.copy()
        
        sc_init = evaluate_safe(init_a)
        print(f"  Init {ii+1}/{n_inits}: score={sc_init:.6f}")
        
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_a.copy()
        
        # Quick optimize this init
        a_opt, _ = optimize_comprehensive(init_a, t_init * 0.7, solver, label=f"init{ii}")
        try_update(a_opt, f"init{ii}_opt")

    # === Phase 2: Main optimization on best sequence (40% of time) ===
    print(f"\n--- Phase 2: Main optimization (best={best_score:.6f}) ---")
    rem = end_time - time.monotonic()
    t_phase2 = min(rem * 0.40, 500.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s")
        a_opt2, _ = optimize_comprehensive(best_seq, t_phase2, solver, label="main")
        try_update(a_opt2, "main_opt")

    # === Phase 3: Multi-length exploration (25% of time) ===
    print(f"\n--- Phase 3: Multi-length exploration (best={best_score:.6f}) ---")
    rem = end_time - time.monotonic()
    
    # Candidate lengths around the base
    candidate_lengths = []
    for mult in [0.75, 0.85, 0.95, 1.1, 1.2, 1.35, 1.5, 1.75, 2.0]:
        ln = int(n_base * mult)
        if 200 <= ln <= 5000 and ln != n_base:
            candidate_lengths.append(ln)
    
    # Sort by proximity to base length (prefer nearby lengths)
    candidate_lengths.sort(key=lambda x: abs(x - n_base))
    candidate_lengths = candidate_lengths[:5]  # Try at most 5
    
    for ln in candidate_lengths:
        rem = end_time - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.18, 180.0)
        if t_alloc < 30:
            break
        
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        
        # Initialize by interpolating
        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"    Interpolation improved! score={sc_init:.6f}")
        
        # Also try a fresh spectral init at this length
        alt_init = spectral_flat_top_init(ln, rng)
        sc_alt = evaluate_safe(alt_init)
        if sc_alt < sc_init:
            init_long = alt_init
            if sc_alt < best_score:
                best_score = sc_alt
                best_seq = init_long.copy()
        
        a_opt, _ = optimize_comprehensive(init_long, t_alloc, solver, label=f"n{ln}")
        try_update(a_opt, f"opt_n{ln}")

    # === Phase 4: Final intensive optimization (remaining time) ===
    print(f"\n--- Phase 4: Final optimization (best={best_score:.6f}, n={len(best_seq)}) ---")
    rem = end_time - time.monotonic()
    if rem > 30:
        # Extended LP run
        t_lp = min(rem * 0.40, 200.0)
        if t_lp > 15:
            print(f"  LP polish: {t_lp:.0f}s")
            a_fin, _ = nesterov_lp_optimize(best_seq, t_lp, solver=solver, n_max_iters=200000)
            try_update(a_fin, "final_lp")
        
        # LogSumExp refinement
        rem = end_time - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.25, 80.0)
            if t_lse > 10:
                print(f"  LSE refinement: {t_lse:.0f}s")
                a_fin2 = lbgfs_logsumexp_optimize(best_seq, t_lse, T_start=10.0, T_end=0.02, n_stages=12)
                try_update(a_fin2, "final_lse")
        
        # Local peak search
        rem = end_time - time.monotonic()
        if rem > 15:
            t_ls = min(rem * 0.20, 50.0)
            if t_ls > 5:
                print(f"  Local search: {t_ls:.0f}s")
                a_fin3, _ = local_search_peaks(best_seq, t_ls, n_steps=20000)
                try_update(a_fin3, "final_ls")
        
        # One more LP pass
        rem = end_time - time.monotonic()
        if rem > 10:
            t_lp2 = min(rem * 0.35, 40.0)
            if t_lp2 > 5:
                a_fin4, _ = nesterov_lp_optimize(best_seq, t_lp2, solver=solver, n_max_iters=50000)
                try_update(a_fin4, "final_lp2")

    # === Final: try a few more lengths if time permits ===
    rem = end_time - time.monotonic()
    if rem > 60 and best_seq is not None:
        print(f"\n--- Phase 5: Additional length exploration (remaining={rem:.0f}s) ---")
        n_best = len(best_seq)
        for offset in [-100, -50, 50, 100, 150]:
            ln = n_best + offset
            if ln < 200 or ln > 5000:
                continue
            rem = end_time - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.2, 60.0)
            if t_alloc < 15:
                break
            
            init = interpolate_sequence(best_seq, ln)
            a_opt, _ = optimize_comprehensive(init, t_alloc, solver, label=f"extra_n{ln}")
            try_update(a_opt, f"extra_n{ln}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, elapsed={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
