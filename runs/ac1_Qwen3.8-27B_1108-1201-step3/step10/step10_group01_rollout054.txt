    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def fast_autocorr(a):

    a = np.asarray(a, dtype=np.float64)

        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1)
    rho = np.where(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_val_and_weights(c, T):
    """Compute LogSumExp value and softmax weights for vector c with temperature T."""
    max_c = c.max()
    exp_tc = np.exp(T * (c - max_c))
    sum_exp = exp_tc.sum()
    val = max_c + np.log(sum_exp) / T
    w = exp_tc / sum_exp
    return val, w


def autocorr_gradient_fast(a, w):
    """Compute gradient of LogSumExp autocorrelation objective.
    
    grad[i] = (4n/T) * sum_m a[m] * w[i+m]
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return corr[:n]


def fista_logsumexp(a0, time_limit, T_stages=None, max_iter_per_stage=10000):
    """FISTA with LogSumExp smoothing on the probability simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)
    a_fista = a.copy()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        x = a_fista.copy()
        y = a_fista.copy()
        t_k = 1.0
        # Initial Lipschitz estimate
        c0 = fast_autocorr(y)
        L = T * n * np.sum(a_fista ** 2) * 1.5 + 1e-8
        alpha = 1.0 / L
        n_iter_this_stage = max(100, min(max_iter_per_stage, int(remaining * 50)))

        for it in range(n_iter_this_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Gradient at y
            c_y = fast_autocorr(y)
            _, w_y = lse_val_and_weights(c_y, T)
            grad_y = (4.0 * n / T) * autocorr_gradient_fast(y, w_y)

            # Backtracking
            x_new = project_simplex(y - alpha * grad_y)
            # Simple check: ensure function decreased
            L = L * 1.0  # Keep same L for efficiency (conservative)
            alpha = 1.0 / L

            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
            y = project_simplex(y)
            x = x_new
            t_k = t_new

            # Periodic evaluation
            if it % 500 == 0:
                sc = evaluate_local(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    a_fista = x.copy()

        # End of stage
        sc = evaluate_local(x)
        if sc < best_score:
            best_score = sc
            best_a = x.copy()
        a_fista = best_a.copy()

    # Final evaluation
    sc = evaluate_local(best_a)
    if sc < best_score:
        best_score = sc
    return best_a, best_score


def smooth_best_response(a_fixed, T_br=10.0, n_steps=3000):
    """Solve min_b (1/T_br)*log(sum exp(T_br * conv(a_fixed, b))) s.t. b in simplex.
    Uses FISTA with FFT-based gradients."""
    n = len(a_fixed)
    b = np.ones(n) / n
    t0 = time.monotonic()

    x = b.copy()
    y = b.copy()
    t_k = 1.0
    L = T_br * n * np.sum(a_fixed ** 2) * 2.0 + 1e-8
    alpha = 1.0 / L

    for it in range(n_steps):
        if time.monotonic() - t0 > n_steps * 0.002:
            break

        # Compute conv(a_fixed, y)
        ay = fast_convolve(a_fixed, y)
        _, w = lse_val_and_weights(ay, T_br)

        # Gradient: T^T @ w = cross_correlation(a_fixed, w)
        grad = autocorr_gradient_fast(a_fixed, w)

        # FISTA step
        x_new = project_simplex(y - alpha * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_simplex(y)
        x = x_new
        t_k = t_new

    return x


def nesterov_smooth_alternation(a0, time_limit, T_br=10.0, inner_steps=1500,
                                  stall_patience=15, print_interval=500):
    """Nesterov-accelerated alternating minimization with smooth best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    max_iters = 100000

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Inner: best response to y
        b_new = smooth_best_response(y, T_br=T_br, n_steps=inner_steps)

        # Nesterov dual averaging
        alpha_n = 2.0 / (it + 2)
        y = (1.0 - alpha_n) * y + alpha_n * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Primal iterated average
        a = ((it + 1) * a + (alpha_n * (it + 2)) * b_new) / (it + 1 + alpha_n * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Track best
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.03 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                ys = y.sum()
                if ys > 1e-15:
                    y = y / ys
                else:
                    y = a.copy()
                stall = 0

            if print_interval and (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NSA iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coord_descent_peaks(a0, time_limit, n_steps=50000, top_m=6):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
            # Small random perturbation
            if stall % 100 == 0:
                idx_r = rng.integers(0, n)
                a_trial = a.copy()
                a_trial[idx_r] *= 0.95
                a_trial = a_trial / a_trial.sum()
                sc = evaluate_local(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score
