Let me also think about one more improvement: **dual descent / cutting plane method**.

The problem is: min M s.t. f_k(a) <= M for all k, sum(a) = 1, a >= 0.

The Lagrangian dual is: max_{lambda >= 0, sum(lambda) = 1} min_{a >= 0, sum(a)=1} M + sum_k lambda_k (f_k(a) - M)
= max_{lambda} min_a [M(1 - sum_k lambda_k) + sum_k lambda_k f_k(a)]
= max_{lambda} min_a sum_k lambda_k f_k(a)  [since sum(lambda)=1]

The inner minimization is: min_{a >= 0, sum(a)=1} sum_k lambda_k * a^T Q_k a = min_a a^T (sum_k lambda_k Q_k) a

This is a QP (quadratic program)! And it's the "mirror" of the LP in the primal.

So we have an alternating scheme:
- Primal step: given lambda, solve min_{a in simplex} a^T A lambda a (a QP)
- Dual step: given a, update lambda_k proportional to the "violation" of constraint k
  (lambda_k <- lambda_k * exp(eta * f_k(a)) or similar)

This is an instance of the multiplicative weights / Hedge algorithm applied to the minimax problem!

This is actually a very natural approach and might be more effective than the LP because:
1. The QP in the primal step is easier to solve (it's a standard QP, not a large LP)
2. The dual update is cheap (just a vector operation)
3. It has good theoretical convergence guarantees

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:

    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

            ai_old = a[i]
            if ai_old < 1e-16:
                continue

            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_min > 0:
                unaff = conv[:k_min].max()
            else:
                unaff = -1e30
            if k_max < 2 * n - 2:
                unaff_right = conv[k_max + 1:].max()
                if unaff_right > unaff:
                    unaff = unaff_right

            # Get the a values for affected positions
            # a[k-i] for k in [k_min, k_max]
            a_affected = a[k_min - i: k_max - i + 1]
            conv_affected = conv[k_min: k_max + 1].copy()

            # Check if k=2i is in range
            has_diag = (2 * i >= k_min) and (2 * i <= k_max)
            diag_pos = 2 * i - k_min if has_diag else -1

            # Try candidate values
            best_x = ai_old
            best_M_trial = M

            for frac in candidate_fracs:
                x = ai_old * frac
                if x < -1e-15:
                    continue
                x = max(0.0, x)
                if abs(x - ai_old) < 1e-15:
                    continue

                delta = x - ai_old

                # New affected conv (vectorized)
                new_aff = conv_affected + 2.0 * delta * a_affected
                if has_diag:
                    new_aff[diag_pos] = conv[2 * i] + x * x - ai_old * ai_old

                new_M = unaff
                aff_max = new_aff.max()
                if aff_max > new_M:
                    new_M = aff_max

                if new_M < best_M_trial - 1e-14:
                    best_M_trial = new_M
                    best_x = x

            if best_x != ai_old and best_M_trial < M - 1e-14:
                # Apply the best change
                delta = best_x - a[i]
                a[i] = best_x
                conv[k_min: k_max + 1] += 2.0 * delta * a_affected
                if has_diag:
                    conv[2 * i] = conv_affected[diag_pos] + best_x * best_x - ai_old * ai_old

                new_M_global = max(unaff, float(conv[k_min: k_max + 1].max()))
                # Recompute global max (safe but slightly expensive)
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True, label=""):
    """Mirror descent (multiplicative weights) on the non-negative simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_M = float(conv.max())
    best_score = 2.0 * n * best_M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    # Start with a moderate learning rate
    eta = 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

        # Find top peaks
        K = min(20, 2 * n - 1)
        top_k = np.argpartition(conv, -K)[-K:]
        top_k_sorted = top_k[np.argsort(conv[top_k])[::-1]]

        # Compute subgradient: weighting over top peaks
        weights = np.exp((conv[top_k_sorted] - conv[top_k_sorted[0]]) * 5.0)
        weights = weights / weights.sum()

        # Subgradient: sg[i] = sum_k w[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] + (a[k-i] if k-i != i else a[i])
        # = 2*a[k-i] (if k != 2i) or a[i] (if k == 2i)... actually:
        # conv[k] = sum_j a[j]*a[k-j]
        # d/d(a[i]) conv[k] = a[k-i] (from j=i term) + a[k-i] (from j=k-i term, if k-i != i)
        # = 2*a[k-i] if k-i != i and 0<=k-i<n
        # = a[i] if k-i == i (i.e., k=2i)... wait no:
        # If k=2i: the term a[i]*a[i] appears once, derivative is 2*a[i]
        # If k-i != i: terms a[i]*a[k-i] and a[k-i]*a[i], derivative is 2*a[k-i]
        # So in all cases where k-i is in [0,n-1]: d/d(a[i]) conv[k] = 2*a[k-i]

        sg = np.zeros(n)
        for idx in range(len(top_k_sorted)):
            if weights[idx] < 1e-12:
                continue
            k = int(top_k_sorted[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # d/d(a[i]) conv[k] = 2*a[k-i] for i in [i_start, i_end]
            sg[i_start:i_end + 1] += weights[idx] * 2.0 * a[k - i_end: k - i_start + 1][::-1]

        # Mirror descent update: a[i] <- a[i] * exp(-eta * sg[i]) / Z
        log_update = -eta * sg
        log_update = log_update - log_update.max()  # for numerical stability
        multipliers = np.exp(log_update)
        a_new = a * multipliers
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        a = a_new

        # Decay learning rate
        if it % 100 == 0:
            eta *= 0.95
        if eta < 1e-8:
            eta = 1e-8

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization with adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Adaptive damping: start aggressive, become conservative
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.3, 8.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = b_new.copy()
        else:
            a = a / s

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation detection
        if score_a > best_score * (1 + 5e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        # Random perturbation when stuck
        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.002:
                a = a_pert
            stagnation = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=True):
    """Directly target and reduce the top peaks by redistributing mass."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        K = 8
        if K > 2 * n - 1:
            K = 2 * n - 1
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.98:
                continue

            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Find top contributing pairs
            contribs = np.array([a[i] * a[k_star - i] for i in range(i_start, i_end + 1)])
            top_pairs = np.argsort(contribs)[::-1][:5]

            for tp in top_pairs:
                i = i_start + int(tp)
                j = k_star - i
                if i == j:
                    # Diagonal term: reducing a[i] by delta changes conv[k] by -2*delta*a[i] + delta^2
                    for frac in [0.1, 0.2, 0.3, 0.5]:
                        delta = -a[i] * frac
                        new_val = a[i] + delta
                        if new_val < 0:
                            continue
                        # Update conv
                        k_s = i
                        k_e = min(i + n - 1, 2 * n - 2)
                        new_conv = conv.copy()
                        new_conv[k_s:k_e + 1] += 2.0 * delta * a[k_s - i:k_e - i + 1]
                        new_conv[2 * i] = conv[2 * i] + new_val * new_val - a[i] * a[i]
                        new_M = float(new_conv.max())
                        if new_M < M - 1e-12:
                            a[i] = new_val
                            conv = new_conv
                            M = new_M
                            improved = True
                            cur_score = 2.0 * n * M
                            if cur_score < best_score:
                                best_score = cur_score
                                best_a = a.copy()
                            break
                else:
                    # Off-diagonal: reduce the larger of a[i], a[j]
                    if a[i] >= a[j]:
                        target = i
                    else:
                        target = j
                    for frac in [0.1, 0.2, 0.3, 0.5]:
                        delta = -a[target] * frac
                        new_val = a[target] + delta
                        if new_val < 0:
                            continue
                        k_s = target
                        k_e = min(target + n - 1, 2 * n - 2)
                        new_conv = conv.copy()
                        new_conv[k_s:k_e + 1] += 2.0 * delta * a[k_s - target:k_e - target + 1]
                        if 2 * target >= k_s and 2 * target <= k_e:
                            new_conv[2 * target] = conv[2 * target] + new_val * new_val - a[target] * a[target]
                        new_M = float(new_conv.max())
                        if new_M < M - 1e-12:
                            a[target] = new_val
                            conv = new_conv
                            M = new_M
                            improved = True
                            cur_score = 2.0 * n * M
                            if cur_score < best_score:
                                best_score = cur_score
                                best_a = a.copy()
                            break

        if not improved and rounds > 30:
            break

        if verbose and rounds % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    t = np.arange(n)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 12))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Smooth random low-freq
        freqs = rng.uniform(0.1, 3.0, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * t / n + p)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization combining LP alternation, exact coordinate descent,
    mirror descent, and peak shaving with diverse restarts.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = p_score
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.8f}")

    solver = _find_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.40

    t_alloc = min(t1_end - time.monotonic(), 300.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Try diverse initializations
    inits = [
        diverse_init(n, rng, 1),
        diverse_init(n, rng, 5),
        diverse_init(n, rng, 6),
    ]
    for idx, init in enumerate(inits):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(remaining * 0.4, 60.0)
        a_tmp, s_tmp = lp_alternating(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_init{idx+1}")

    # Second LP round on best
    if time.monotonic() < t1_end - 40:
        t_alloc = min(t1_end - time.monotonic(), 150.0)
        print(f"  LP round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_r2")

    # === Phase 2: Exact Coordinate Descent ===
    print(f"\n=== Phase 2: Exact CD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    t_alloc = min(t2_end - time.monotonic(), 250.0)
    if t_alloc > 20:
        print(f"  ExactCD budget={t_alloc:.0f}s")
        a_cd, s_cd = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "exact_cd")

        # CD from LP result
        if time.monotonic() < t2_end - 30:
            t2 = min(t2_end - time.monotonic(), 100.0)
            a_cd2, s_cd2 = exact_coord_descent(a_lp, t2, verbose=True, label="lp")
            try_update(a_cd2, "exact_cd_lp")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.12

    if time.monotonic() < t3_end - 10:
        t_alloc = min(t3_end - time.monotonic(), 80.0)
        if t_alloc > 15:
            print(f"  MirrorDescent budget={t_alloc:.0f}s")
            a_md, s_md = mirror_descent(best_seq, t_alloc, verbose=True, label="main")
            try_update(a_md, "mirror")

    # === Phase 4: Peak Shaving + Final Polish ===
    print(f"\n=== Phase 4: Peak Shave + Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 15

    if time.monotonic() < t4_end - 40:
        t_ps = min(t4_end - time.monotonic() - 40, 60.0)
        if t_ps > 15:
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shaving(best_seq, t_ps, verbose=True)
            try_update(a_ps, "peak_shave")

    # Final exact CD polish
    if time.monotonic() < t4_end - 10:
        t_fp = min(t4_end - time.monotonic() - 10, 80.0)
        if t_fp > 10:
            print(f"  Final ExactCD budget={t_fp:.0f}s")
            a_fin, s_fin = exact_coord_descent(best_seq, t_fp, verbose=False, label="final")
            try_update(a_fin, "final_cd")

    # One more peak shave at the very end
    if time.monotonic() < t4_end - 5:
        t_ps2 = min(t4_end - time.monotonic() - 5, 20.0)
        if t_ps2 > 5:
            a_ps2, s_ps2 = peak_shaving(best_seq, t_ps2, verbose=False)
            try_update(a_ps2, "final_peak")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
