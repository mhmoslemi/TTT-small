def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(a, b):
    """Cross-correlation: result[i] = sum_k a[k] * b[k-i] (zero-padded)"""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * np.conj(B), fft_size)[:size]

        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.fft import rfft, irfft

import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly (assumes sum=1 for simplicity)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def build_conv_matrix(a):
    """Build convolution matrix T: T[k,j] = a[k-j] if valid."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_rows(a, row_indices):
    """Build subset of convolution matrix rows."""
    n = len(a)
    T = np.zeros((len(row_indices), n))
    for ii, k in enumerate(row_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        T[ii, j_start:j_end + 1] = a[k - j_end:k - j_start + 1]
    return T

    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, eps=1e-10, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, eps=1e-10, max_time=30)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else fast_convolve(b_val).max()
    return b_val, M_val


def lp_best_response_dual(a, solver):
    """Solve dual LP: max lambda s.t. cross_corr(y,a) >= lambda, sum(y)=1, y>=0.
    Then extract b from complementary slackness or just use the dual value."""
    n = len(a)
    # Dual: max lambda s.t. sum_k y[k]*a[k-j] >= lambda for all j, sum(y)=1, y>=0
    # Variables: y (length 2n-1), lambda
    # (sum_k y[k]*a[k-j]) = cross-correlation
    # Build matrix: C[j,k] = a[k-j] if 0<=k-j<=n-1 else 0
    C = np.zeros((n, 2 * n - 1))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        C[j, k_start:k_end + 1] = a[0:n - (k_end - j_start)]
    
    y_var = cp.Variable(2 * n - 1, nonneg=True)
    lam_var = cp.Variable()
    constraints = [
        C @ y_var >= lam_var,
        cp.sum(y_var) == 1
    ]
    prob = cp.Problem(cp.Maximize(lam_var), constraints)
    try:
        prob.solve(solver=solver, eps=1e-10, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, eps=1e-10, max_time=30)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    # We need b from the primal. Use complementary slackness or just solve primal with warm start.
    # For now, return the dual value and let caller use primal.
    dual_val = float(lam_var.value) if lam_var.value is not None else float('inf')
    # Just solve primal as fallback
    return lp_best_response_primal(a, solver)


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating with adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Solve LP for best response
        b_new, M_lp = lp_best_response_primal(a, solver)
        if not np.isfinite(M_lp):
            break

        # Evaluate b_new
        conv_b = fast_convolve(b_new)
        score_b = float(2.0 * n * conv_b.max())
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Adaptive damping
        if it <= 10:
            alpha = 1.0
        elif it <= 40:
            alpha = 0.8
        elif it <= 100:
            alpha = 0.5
        else:
            alpha = max(0.2, 15.0 / (it + 20.0))

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Evaluate
        conv_a = fast_convolve(a)
        score_a = float(2.0 * n * conv_a.max())
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation tracking
        if abs(score_a - prev_score) < 1e-12:
            stagnation += 1
        else:
            stagnation = 0
        prev_score = score_a

        # Perturb if stuck
        if stagnation > 20:
            # Gradient-informed perturbation
            peak = np.argmax(conv_a)
            grad = np.zeros(n)
            i_start = max(0, peak - n + 1)
            i_end = min(n - 1, peak)
            grad[i_start:i_end + 1] = 2.0 * a[peak - i_end:peak - i_start + 1][::-1]
            
            noise_scale = 0.005 / np.sqrt(n)
            a_pert = a - noise_scale * grad / max(np.max(np.abs(grad)), 1e-10)
            a_pert += np.random.standard_normal(n) * noise_scale * 0.5
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.10f} cur={score_a:.10f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=True, label=""):
    """Enhanced coordinate descent with very fine single-coordinate search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Sort coordinates by value (larger first) for efficiency
    coord_order = np.argsort(a)[::-1]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in coord_order:
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1].copy()
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Fine multiplicative search (more factors, smaller steps)
            ai = a[i]
            factors = [0.99, 0.995, 0.998, 0.999, 1.001, 1.002, 1.005, 1.01,
                       0.98, 0.97, 0.95, 0.9, 1.02, 1.05, 1.1, 0.5, 2.0, 0.0]

            for factor in factors:
                if factor == 0.0:
                    delta = -ai
                else:
                    delta = ai * (factor - 1.0)
                new_val = ai + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * ai + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    old_ai = a[i] - best_delta
                    changes[idx] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_coord_descent(a0, time_limit, verbose=True, label=""):
    """Pair coordinate descent: adjust pairs (i,j) simultaneously."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    no_improve = 0

    # Get top coordinates to form pairs
    top_coords = np.argsort(a)[::-1][:min(200, n)]

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Try pairs among top coordinates
        for idx_i in range(len(top_coords)):
            i = top_coords[idx_i]
            if a[i] < 1e-15:
                continue
            for idx_j in range(idx_i + 1, min(idx_i + 10, len(top_coords))):
                j = top_coords[idx_j]
                if a[j] < 1e-15:
                    continue
                if abs(i - j) > n // 2:
                    continue  # Skip pairs that are too far apart (less interaction)

                # Try shifting mass between i and j
                total = a[i] + a[j]
                if total < 1e-15:
                    continue

                # Parameterize: a[i] = total * (1-t), a[j] = total * t
                # Try several values of t
                current_t = a[j] / total
                t_candidates = [0.0, 1.0,
                                current_t * 0.8, current_t * 0.9, current_t * 0.95,
                                current_t * 1.05, current_t * 1.1, current_t * 1.2,
                                0.3, 0.5, 0.7]

                best_t = current_t
                best_new_M = M

                for t_new in t_candidates:
                    t_new = max(0.0, min(1.0, t_new))
                    if abs(t_new - current_t) < 1e-8:
                        continue

                    new_ai = total * (1.0 - t_new)
                    new_aj = total * t_new
                    delta_i = new_ai - a[i]
                    delta_j = new_aj - a[j]

                    # Compute new conv in the affected range
                    k_lo = min(i, j)
                    k_hi = max(i + n - 1, j + n - 1)
                    k_hi = min(k_hi, 2 * n - 2)

                    if k_lo > 0:
                        unaff_left = conv[:k_lo].max()
                    else:
                        unaff_left = -1e30
                    if k_hi < 2 * n - 2:
                        unaff_right = conv[k_hi + 1:].max()
                    else:
                        unaff_right = -1e30
                    unaff_max = max(unaff_left, unaff_right)

                    # Compute affected conv values
                    # Challenge: need conv values in [k_lo, k_hi]
                    # For positions where only i is affected: add 2*delta_i * a[k-i]
                    # For positions where only j is affected: add 2*delta_j * a[k-j]
                    # For positions where both are affected: add both
                    # Plus quadratic terms at k=2i and k=2j
                    # And cross term at k=i+j: delta_i*a[j] + delta_j*a[i] + ...
                    
                    # Simpler: just recompute conv for the affected range
                    # affected_len = k_hi - k_lo + 1
                    # For each k in [k_lo, k_hi]:
                    #   new_conv[k] = conv[k] + delta_i*(a[k-i] if valid) + a[k-i]*(delta_i if valid)
                    #                + delta_j*(a[k-j] if valid) + a[k-j]*(delta_j if valid)
                    #                + correction for k=2i, k=2j, k=i+j
                    
                    # This is getting complex. Let me just do a full FFT eval for the candidate.
                    # Since we're only trying ~10 candidates per pair, and pairs are ~200*10=2000,
                    # that's 20000 FFTs per round. Each FFT is O(n log n) ~ 10K ops. Total: 200M ops per round.
                    # That might be too slow. Let me limit the number of pairs tried per round.
                    
                    # Actually, let me just do incremental updates properly.
                    new_conv_affected = np.zeros(k_hi - k_lo + 1)
                    for offset in range(k_hi - k_lo + 1):
                        k = k_lo + offset
                        val = conv[k]
                        # Contribution from a_i term
                        ki = k - i
                        if 0 <= ki < n:
                            val += delta_i * a[ki] + (a[i] + delta_i) * 0  # cross term handled below
                            val += 0  # we need: new_a[i]*a[k-i] + a[k-i]*new_a[i] - a[i]*a[k-i] - a[k-i]*a[i]
                            # = (new_a[i] - a[i]) * a[k-i] + a[k-i] * (new_a[i] - a[i])
                            # = 2 * delta_i * a[k-i] (if k-i != i, i.e., k != 2i)
                            # If k == 2i: = delta_i * a[i] + a[i] * delta_i + delta_i^2 = 2*delta_i*ai + delta_i^2
                        kj = k - j
                        if 0 <= kj < n:
                            val += 2.0 * delta_j * a[kj]
                        # Quadratic corrections
                        if k == 2 * i:
                            val += delta_i * delta_i
                        if k == 2 * j:
                            val += delta_j * delta_j
                        # Cross terms: a[i]*a[j] appears at k=i+j in both a_i and a_j contributions
                        # When we change a[i] by delta_i, the term a[i]*a[j] changes by delta_i*a[j]
                        # When we change a[j] by delta_j, the term a[j]*a[i] changes by delta_j*a[i]
                        # These are already included in the 2*delta_i*a[k-i] when k-i=j (k=i+j)
                        # and 2*delta_j*a[k-j] when k-j=i (k=i+j)
                        # But we double-counted: at k=i+j, we add 2*delta_i*a[j] + 2*delta_j*a[i]
                        # The actual change is: delta_i*a[j] + delta_j*a[i] (from a_i's contribution)
                        #                           + delta_j*a[i] + delta_i*a[j] (from a_j's contribution)
                        # = 2*(delta_i*a[j] + delta_j*a[i])
                        # And our formula gives: 2*delta_
