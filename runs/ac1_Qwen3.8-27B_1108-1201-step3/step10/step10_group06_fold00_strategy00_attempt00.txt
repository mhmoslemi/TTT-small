def evaluate_sequence(seq):
    n_val = len(seq)
    if n_val < 2:
        return float('inf')
    try:
        seq_arr = np.array(seq, dtype=np.float64)
        seq_sum = seq_arr.sum()
        if seq_sum < 0.01:

        conv_result = fast_convolve(seq_arr)

        return 2.0 * n_val * max_conv_val / (seq_sum)**2
    except Exception as e:

        return float('inf')
