        alpha = 2.0 / (it + 2.0)
        a_avg = (1.0 - alpha) * a_avg + alpha * a_new
        a_avg = np.maximum(a_avg, 0)
        sa = a_avg.sum()
        if sa > 1e-15:
            a_avg = a_avg / sa
        
        b_avg = (1.0 - alpha) * b_avg + alpha * b
        b_avg = np.maximum(b_avg, 0)
        sb = b_avg.sum()
        if sb > 1e-15:
            b_avg = b_avg / sb
        
        # Update working point (blend of imitation and current)
        a = 0.5 * a_avg + 0.5 * a_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        else:
            a = a_avg.copy()
        
        # Track best
        for cand in [a, a_avg, b_avg, a_new, b]:
            sc = 2.0 * n * fast_conv(cand).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
        
        # Stall detection with perturbation
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 8:
                noise = rng.standard_normal(n) * 0.01
                a_avg = a_avg + noise
                a_avg = np.maximum(a_avg, 0)
                a_avg = a_avg / a_avg.sum()
                stall = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lse_lbgfs(a0, time_limit, T_stages=None):
    """LogSumExp-smoothed L-BFGS with multi-stage temperature cooling."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z = np.log(a_init + 1e-30)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01]
    
    n_stages = len(T_stages)
    
    for si, T_stage in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters_per = max(30, int(30000 // n_stages))
        
        def func(z_loc):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            al = np.exp(z_loc - z_loc.max())
            al = al / al.sum()
            bl = fast_conv(al)
            mb = bl.max()
            eb = np.exp(bl - mb)
            lse = mb + np.log(eb.sum())
            return (2.0 * n / T_stage) * lse
        
        def jac(z_loc):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_loc)
            al = np.exp(z_loc - z_loc.max())
            al = al / al.sum()
            bl = fast_conv(al)
            mb = bl.max()
            eb = np.exp(bl - mb)
            w = eb / eb.sum()
            # Gradient of max_b w.r.t. a: for each k, d/d a[j] of c[k] = w[k] * a[k-j]
            grad_a = np.zeros(n)
            # Vectorized: grad_a[j] = sum_k w[k] * a[k-j] for valid k
            # This is a correlation: grad_a = convolve(w, a) (cross correlation)
            # Actually: c[k] = sum_j a[j]*a[k-j], dc[k]/da[j] = a[k-j] + a[j] (if k-j valid and j valid)
            # For auto-conv: dc[k]/da[j] = a[k-j] if 0<=k-j<n, plus a[j] if j==k-j (i.e., k=2j)
            # But since we're using the weighted sum: sum_k w[k] * dc[k]/da[j]
            # = sum_k w[k] * a[k-j] (for valid k-j)
            # This is a correlation of w with a
            # Let's compute it directly
            for k in range(len(bl)):
                if w[k] < 1e-14:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    idx = np.arange(lo, hi + 1)
                    grad_a[idx] += w[k] * a[k - idx]
            # Account for the fact that c[k] = sum_j a[j]*a[k-j], so dc[k]/da[j] = a[k-j]
            # (each a[j] appears once in c[k] for each valid k)
            # Actually for auto-convolution, a[j] appears in c[k] for k = j (with a[0]) through k = j+n-1 (with a[n-1])
            # dc[k]/da[j] = a[k-j] (for the term a[j]*a[k-j])
            # But also if k-j = j, i.e., k=2j, then a[j]*a[j] = a[j]^2, derivative is 2*a[j]
            # The formula above handles this correctly since when k=2j, a[k-j] = a[j]
            # Wait no: c[k] = sum_{j:0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # dc[k]/da[j] = a[k-j] (if 0<=k-j<n)
            # But a[j] also appears as a[k-j] in the term where j' = k-j: a[k-j']*a[j'] = a[j]*a[k-j]
            # So dc[k]/da[j] = a[k-j] + a[j] (if both k-j and j are valid indices, and k-j != j)
            # If k-j == j (i.e., k=2j), then dc[k]/da[j] = 2*a[j]
            # Let me redo: c[k] = sum_j a[j] * a[k-j]
            # dc[k]/da[j] = a[k-j] (from the first factor) + a[k-j'] where j'=k-j... 
            # No. Let's be careful. c[k] = sum_{j'} a[j'] * a[k-j']
            # dc[k]/da[m] = sum_{j'} [delta_{j',m} * a[k-j'] + a[j'] * delta_{k-j',m}]
            # = a[k-m] (from first term, j'=m) + a[k-m] (from second term, k-j'=m => j'=k-m)
            # = 2*a[k-m] (if both m and k-m are valid)
            # Wait no: second term: a[j'] * delta_{k-j',m} => j' = k-m, so a[k-m]
            # But we need k-m to be a valid index: 0 <= k-m < n
            # And m must be valid: 0 <= m < n
            # So dc[k]/da[m] = a[k-m] + a[k-m] = 2*a[k-m] if 0<=k-m<n and 0<=m<n
            # But wait, that's only when m != k-m, i.e., k != 2m
            # If k = 2m, then the term a[m]*a[m] = a[m]^2, derivative is 2*a[m]
            # And the formula gives 2*a[k-m] = 2*a[m]. Correct!
            # So dc[k]/da[m] = 2*a[k-m] if 0<=k-m<n, else 0
            # Hmm wait, that's not right either. Let me be even more careful.
            # c[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[i]=0 if i not in [0,n-1])
            # dc[k]/da[m] = sum_j [delta(j,m)*a[k-j] + a[j]*delta(k-j,m)]
            # = a[k-m] + a[k-m]
            # = 2*a[k-m] (if 0<=k-m<n)
            # = a[k-m] (if k-m == m, i.e., the same term is counted once... no)
            # Actually no, the sum is over all j. When j=m: contribution is a[k-m] (from the first product rule)
            # When k-j=m, i.e., j=k-m: contribution is a[k-m] (from the second product rule)
            # These are the same when m=k-m, i.e., k=2m. In that case, we get a[m] + a[m] = 2*a[m].
            # When m != k-m (i.e., k != 2m), we get a[k-m] + a[k-m] = 2*a[k-m].
            # So in all cases: dc[k]/da[m] = 2*a[k-m] if 0<=k-m<n, else 0.
            # 
            # Wait, that can't be right. Let's check with a simple example.
            # a = [a0, a1], n=2
            # c[0] = a0*a0 = a0^2. dc[0]/da[0] = 2*a0, dc[0]/da[1] = 0.
            # Using formula: dc[0]/da[0] = 2*a[0-0] = 2*a0. Correct.
            # dc[0]/da[1] = 2*a[0-1] = 2*a[-1] = 0 (out of range). Correct.
            # c[1] = a0*a1 + a1*a0 = 2*a0*a1. dc[1]/da[0] = 2*a1, dc[1]/da[1] = 2*a0.
            # Using formula: dc[1]/da[0] = 2*a[1-0] = 2*a1. Correct.
            # dc[1]/da[1] = 2*a[1-1] = 2*a0. Correct.
            # c[2] = a1*a1 = a1^2. dc[2]/da[0] = 0, dc[2]/da[1] = 2*a1.
            # Using formula: dc[2]/da[0] = 2*a[2-0] = 2*a[2] = 0 (out of range). Correct.
            # dc[2]/da[1] = 2*a[2-1] = 2*a1. Correct.
            # 
            # Great! So the gradient is:
            # grad_a[m] = sum_k w[k] * 2*a[k-m] (for valid k-m)
            # = 2 * (w * a)[m] where (w*a) is the cross-correlation
            # cross_corr[m] = sum_k w[k] * a[k-m]
            # This is the same as: cross_corr = convolve(w_reversed, a) ... 
            # Actually cross_corr[m] = sum_k w[k] * a[k-m]
            # Let l = k-m: cross_corr[m] = sum_l w[l+m] * a[l]
            # This is the correlation of w with a, which equals convolve(flip(w), a)
            # Or equivalently: it's the auto-convolution-like operation
            
            # Let me just compute it with FFT
            # cross_corr[m] = sum_k w[k] * a[k-m]
            # = sum_k w[k] * a[k-m] where a is zero-padded
            # This is the cross-correlation <w, shift_m(a)>
            # = inverse FFT of conj(FFT(w)) * FFT(a) ... but we need to be careful with linear vs circular
            
            # For now, use the direct method (it's O(n^2) but n=1200 is manageable)
            # Actually, let's use the loop but vectorize over k
            # grad_a[m] = 2 * sum_{k: 0<=k-m<n} w[k] * a[k-m]
            # = 2 * sum_{l=0}^{n-1} w[l+m] * a[l] (where w is zero-padded)
            # This is: grad_a = 2 * (correlation of w and a)
            # correlation[m] = sum_l a[l] * w[l+m]
            
            # Use FFT for speed:
            # correlation = ifft(fft(a) * conj(fft(w))) ... need to be careful
            # Actually: correlation[m] = sum_l a[l] * w[l+m]
            # = (a * flip(w))[m] in the convolution sense? No...
            # Let's just use the direct computation for now (n=1200, 2n-1=2399 terms)
            # It's O(n * (2n-1)) = O(n^2) which for n=1200 is ~2.8M ops, fine.
            
            pass  # We already computed grad_a above, but with factor 1 instead of 2
            # Let me fix: the gradient of the max_b part is:
            # d(max_b)/da[m] ≈ sum_k w[k] * dc[k]/da[m] = sum_k w[k] * 2*a[k-m]
            # So we need to multiply grad_a by 2
            grad_a *= 2.0
            
            # Now the full objective is: (2n/T) * LSE(b)
            # d/dz of this: d/da * da/dz where a = softmax(z) = exp(z-zmax)/sum(exp(z-zmax))
            # da/dz_i = a[i] * (delta_{ij} - a[j])
            # d_obj/dz = (2n/T) * sum_m (grad_a[m] / a[m]) * a[m] * (delta - a) ... 
            # = (2n/T) * (grad_a - sum_m grad_a[m] * a[m]) ... 
            # Wait: d_obj/dz_i = (2n/T) * sum_m (d_b/dz_i) 
            # d_b/dz_i = sum_m (db/da_m) * (da_m/dz_i) = sum_m grad_a[m] * a[m] * (delta_{mi} - a[i])
            # = grad_a[i]*a[i] - a[i]*sum_m grad_a[m]*a[m]
            # = a[i] * (grad_a[i] - dot(grad_a, a))
            # So d_obj/dz_i = (2n/T) * a[i] * (grad_a[i] - dot(grad_a, a))
            # But we're using z as the variable and a = softmax(z), and the objective in terms of z is:
            # F(z) = (2n/T) * LSE(conv(softmax(z), softmax(z)))
            # dF/dz_i = (2n/T) * a[i] * (2*n_correl[i] - S) where S = sum_m grad_a_raw[m]*a[m]
            # Hmm, I'm getting confused. Let me just use the chain rule properly.
            
            # F(z) = (2n/T) * LSE(b(a(z)))
            # dF/dz_i = (2n/T) * sum_k (w[k]) * db_k/dz_i
            # db_k/dz_i = sum_m (db_k/da_m) * (da_m/dz_i)
            # da_m/dz_i = a_m * (delta_mi - a_i)
            # db_k/da_m = 2*a[k-m] (as derived above)
            # So db_k/dz_i = sum_m 2*a[k-m] * a_m * (delta_mi - a_i)
            # = 2*a[k-i]*a_i - 2*a_i * sum_m a[k-m]*a_m
            # = 2*a_i * (a[k-i] - c_k)  where c_k = conv(a,a)[k]
            # 
            # Therefore: dF/dz_i = (2n/T) * sum_k w[k] * 2*a_i * (a[k-i] - c_k)
            # = (4n/T) * a_i * (sum_k w[k]*a[k-i] - sum_k w[k]*c_k)
            # = (4n/T) * a_i * (cross_corr[i] - dot(w, c))
            # where cross_corr[i] = sum_k w[k]*a[k-i]
            
            # But wait, I already have grad_a computed as sum_k w[k] * 2*a[k-i] = 2*cross_corr
            # So cross_corr = grad_a / 2
            # And dot(w, c) = sum_k w[k] * c[k]
            
            S = np.dot(w, bl)  # sum_k w[k] * c[k]
            cross_corr = grad_a / 2.0  # sum_k w[k] * a[k-i]
            jac_val = (2.0 * n / T_stage) * a * (cross_corr - S)
            return jac_val
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_polish(a0, time_limit, n_steps=100000):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:5]
        
        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-14:
                    continue
                for factor in [0.98, 0.97, 0.96, 0.95]:
                    a_trial = a.copy()
                    reduction = a[pos] * (1.0 - factor)
                    a_trial[pos] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    osum = others.sum()
                    if osum < 1e-15:
                        break
                    a_trial[mask] = others + reduction * others / osum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_conv(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing top-2 contributors simultaneously
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top2 = idx[np.argsort(contribs)[::-1][:2]]
            a_trial = a.copy()
            total_reduce = 0
            for pos in top2:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.97
                    total_reduce += a[pos] * 0.03
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_conv(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break
        
        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def random_perturb_refine(a0, time_limit, n_trials=50):
    """Try random perturbations and keep improvements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    per_trial = max(0.5, time_limit / n_trials)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Generate perturbation
        scale = 0.01 * (trial / n_trials + 0.3)
        pert = rng.standard_normal(n) * scale
        # Also try structured perturbations
        if trial % 3 == 1:
            x = np.linspace(0, 2 * np.pi, n)
            freq = rng.integers(1, 10)
            pert = scale * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
        elif trial % 3 == 2:
            x = np.linspace(0, 1, n)
            pert = scale * (2 * x - 1) ** 2 * rng.uniform(0.5, 2.0)
        
        a_pert = a + pert
        a_pert = np.maximum(a_pert, 0)
        ps = a_pert.sum()
        if ps < 1e-15:
            continue
        a_pert = a_pert / ps
        
        # Quick refinement: a few L-BFGS steps
        z = np.log(a_pert + 1e-30)
        T_stage = 2.0
        
        def func(z_loc):
            al = np.exp(z_loc - z_loc.max())
            al = al / al.sum()
            bl = fast_conv(al)
            mb = bl.max()
            eb = np.exp(bl - mb)
            return (2.0 * n / T_stage) * (mb + np.log(eb.sum()))
        
        def jac(z_loc):
            al = np.exp(z_loc - z_loc.max())
            al = al / al.sum()
            bl = fast_conv(al)
            mb = bl.max()
            eb = np.exp(bl - mb)
            w = eb / eb.sum()
            gc = np.zeros(n)
            for k in range(len(bl)):
                if w[k] < 1e-14:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    idx = np.arange(lo, hi + 1)
                    gc[idx] += w[k] * a[k - idx] if False else al[k - idx]
            gc *= 2.0
            S = np.dot(gc, al)
            return al * (gc - S) * (2.0 * n / T_stage)
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
        except Exception:
            a_new = a_pert
        
        sc = 2.0 * n * fast_conv(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.3 * (2 * x - 1) ** 2  # U-shaped
    elif choice == 1:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x)
    elif choice == 2:
        a = 1.0 + 0.25 * (2 * x - 1) ** 4
    elif choice == 3:
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 4:
        a = 1.0 + 0.15 * np.sin(4 * np.pi * x)
    else:
        a = 1.0 + 0.2 * (2 * x - 1) ** 2 * np.cos(np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()

    """

    best_seq = None

            p_score = eval_loc(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception) as e:
        print(f"No parent: {e}")
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_loc(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a_cand, label):
        nonlocal best_seq, best_score
        if a_cand is None:
            return
        a_cand = np.maximum(np.asarray(a_cand, dtype=np.float64), 0)
        s = a_cand.sum()
        if s < 0.01:
            return
        a_cand = a_cand / s
        sc = eval_loc(a_cand)
        if np.isfinite(sc) and sc < best_score:
            improve = best_score - sc
            best_score = sc
            best_seq = a_cand.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (improve={improve:.6e}, n={len(a_cand)})")
    
    # === Phase 1: Fictitious Play (fast, no cvxpy) ===
    print(f"\n=== Phase 1: Fictitious Play (n={n_best}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_fp, _ = fictitious_play(best_seq, t1, n_inner=30, use_lp_every=100)
        try_update(a_fp, "fictitious_play")
    
    # === Phase 2: LogSumExp L-BFGS (main refinement) ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 20:
        T_stages = [25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.008]
        print(f"  {len(T_stages)} stages, budget={t2:.0f}s")
        a_lb = lse_lbgfs(best_seq, t2, T_stages=T_stages)
        try_update(a_lb, "lse_lbgfs")
    
    # === Phase 3: Peak-directed coordinate descent ===
    print(f"\n=== Phase 3: Peak CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 10:
        a_cd, _ = peak_polish(best_seq, t3, n_steps=200000)
        try_update(a_cd, "peak_cd")
    
    # === Phase 4: Random perturbation exploration ===
    print(f"\n=== Phase 4: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    if t4 > 10:
        a_rp, _ = random_perturb_refine(best_seq, t4, n_trials=40)
        try_update(a_rp, "perturb")
    
    # === Phase 5: Second LSE pass with lower temperatures ===
    print(f"\n=== Phase 5: LSE Pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 120.0)
    if t5 > 10:
        T_stages2 = [3.0, 1.5, 0.8, 0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.005]
        a_lb2 = lse_lbgfs(best_seq, t5, T_stages=T_stages2)
        try_update(a_lb2, "lse_pass2")
    
    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Length variation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        alt_n = int(n_cur * 1.15)
        t_alt = min(rem * 0.3, 120.0)
        if t_alt > 20:
            print(f"  Trying n={alt_n}")
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, _ = fictitious_play(init_alt, t_alt * 0.5, n_inner=25, use_lp_every=0)
            try_update(a_alt, f"fp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_alt2 = lse_lbgfs(best_seq, rem * 0.5,
                                    T_stages=[2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02])
                try_update(a_alt2, f"lse_n{alt_n}")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 5:
            a_fin = lse_lbgfs(best_seq, t_fin,
                              T_stages=[0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005])
            try_update(a_fin, "final_lse")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        a_cd2, _ = peak_polish(best_seq, min(rem * 0.5, 30.0), n_steps=50000)
        try_update(a_cd2, "final_cd")
    
    # Final verification
    if best_seq is not None:
        fs = eval_loc(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
