    """The given evaluation function"""
    n_val = len(a)

def compute_full_convolution(a):
    """
    Fast convolution using FFT as defined in the program.
    This is computed in full linear fashion without wrapping.
    """
    a = np.array(a)
    n = len(a)
    conv_result = np.zeros(2 * n - 1)
    conv_fft = np.fft.fft(np.concatenate([a, np.zeros(n-1)]))
    conv_result[:] = np.real(np.fft.ifft(conv_fft * conv_fft[:2*n-1]))[:2*n-1]
    return conv_result
