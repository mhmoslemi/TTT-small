Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines Fourier-parameterized L-BFGS for global search, focused mirror descent
with Nesterov acceleration, FFT-accelerated pairwise coordinate descent, and
fine-grained coordinate polish. Uses adaptive peak weighting and diverse
restarts to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_simplex(a):
    """Project onto the probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_focused_subgradient(a, active_frac=0.999, alpha=5.0):
    """
    Compute focused subgradient of max(conv(a,a)).
    Uses power weighting to focus on dominant peaks.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), b, max_b

    # Find active peaks
    threshold = max_b * active_frac
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    # Power-weighted focus
    weights = (b[active] / max_b) ** alpha
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / w_sum

    # Compute subgradient: g[i] = sum_k w_k * 2*a[k-i] (for valid k-i)
    g = np.zeros(n)
    for wk, k in zip(weights, active):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * wk * a[j]

    return g, b, max_b

def pairwise_coordinate_descent(a0, time_limit, n_outer=100000):
    """
    Targeted pairwise coordinate descent: moves mass from peak contributors
    to safe positions.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributing pairs to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        # Sort by contribution (descending)
        order = np.argsort(contribs)[::-1]
        top_contributors = idx[order[:8]]

        improved = False
        for pos in top_contributors:
            if a[pos] < 1e-10:
                continue

            # Find safe receiver positions (low contribution to peak)
            # A position p is "safe" if a[peak_k - p] is small (when valid)
            safe_scores = np.zeros(n)
            for p in range(n):
                j = peak_k - p
                if 0 <= j < n:
                    safe_scores[p] = a[j]
                else:
                    safe_scores[p] = 0.0
            # Also prefer positions where a is already small
            safe_scores += 0.5 * a
            # Sort by safety (ascending - lower is safer)
            safe_order = np.argsort(safe_scores)
            # Exclude the source position
            safe_candidates = [p for p in safe_order[:20] if p != pos][:5]

            for recv in safe_candidates:
                if a[recv] > 0.5 * a[pos]:
                    continue  # Don't move to a position with more mass

                    a_trial[pos] -= delta
                    a_trial[recv] += delta

                    a_trial = normalize_simplex(a_trial)
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:

                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

            if stall > 50:

                delta = a[perturb_pos] * 0.05
                if delta > 1e-12:

                    a_trial[perturb_pos] -= delta
                    # Distribute to 3 random safe positions

                    a_trial = normalize_simplex(a_trial)
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:

                        best_score = score
                        best_a = a.copy()

def fourier_param_log(z, n, k_max):
    """
    Evaluate the Fourier-parameterized sequence:
    a[i] = exp(sum_k z[k] * cos(2*pi*k*i/n)) normalized.
    z[0] is the DC term (shifted for numerical stability).
    """
    i = np.arange(n)
    log_a = np.zeros(n)
    for k in range(1, k_max + 1):
        log_a += z[k - 1] * np.cos(2.0 * np.pi * k * i / n)
    # Subtract max for numerical stability
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return a / s


def fourier_param_obj_and_grad(z, n, k_max, T=1.0):
    """
    Objective and gradient for Fourier parameterization using log-sum-exp smoothing.
    """
    a = fourier_param_log(z, n, k_max)
    b = fast_convolve(a)
    max_b = b.max()

    # LogSumExp smoothing
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    logsumexp = max_b + T * np.log(exp_b.sum())

    obj = 2.0 * n * logsumexp

    # Gradient with respect to a
    grad_a = np.zeros(n)
    size = len(b)
    for k in range(size):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += (2.0 / T) * w[k] * a[k - idx]

    # Chain rule: d(obj)/d(z_m) = sum_i (d obj/d a_i) * (d a_i / d z_m)
    # a_i = exp(log_a_i) / sum(exp(log_a_j))
    # d a_i / d z_m = a_i * (cos(2*pi*m*i/n) - sum_j a_j * cos(2*pi*m*j/n))
    grad_z = np.zeros(k_max)
    for m in range(1, k_max + 1):
        cos_m = np.cos(2.0 * np.pi * m * np.arange(n) / n)
        S = np.dot(a, cos_m)
        grad_z[m - 1] = np.dot(grad_a, a * (cos_m - S))

    return obj, grad_z


def fourier_param_optimize(a0, time_limit, k_max=15, T_stages=None):
    """
    Optimize the Fourier parameterization using L-BFGS with annealing.
    """
    n = len(a0)
    # Initialize z from the given sequence (least-squares fit in log domain)
    a_init = normalize_simplex(a0)
    log_a = np.log(np.maximum(a_init, 1e-300))
    log_a_centered = log_a - log_a.mean()
    i = np.arange(n)
    z = np.zeros(k_max)
    for m in range(1, k_max + 1):
        cos_m = np.cos(2.0 * np.pi * m * i / n)
        z[m - 1] = 2.0 * np.dot(log_a_centered, cos_m) / n

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = evaluate_local(a_init)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break

        def obj_func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            val, _ = fourier_param_obj_and_grad(z_inner, n, k_max, T)
            return val

        def grad_func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, g = fourier_param_obj_and_grad(z_inner, n, k_max, T)
            return g

        try:
            res = minimize(fun=obj_func, x0=z, jac=grad_func, method='L-BFGS-B',
                           options={'maxiter': 2000, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'disp': False})
            z = res.x
        except Exception:
            pass

        # Evaluate the resulting sequence
        a_new = fourier_param_log(z, n, k_max)
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

    return best_a, best_score


def spectral_flatten(a0, time_limit, n_iters=500):
    """
    Spectral flattening: iterate between time-domain (non-negativity projection)
    and frequency-domain (spectrum flattening).
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # Use FFT size for linear convolution
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT of zero-padded a
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Flatten the power spectrum: compress peaks
        p_mean = power.mean()
        if p_mean < 1e-300:
            break
        # Adaptive compression strength
        compression = 0.1 + 0.2 * (it / min(it + 1, 100))
        power_new = power / (1.0 + compression * (power / p_mean - 1.0) ** 2)

        # Recombine with original phase
        phase = A / np.maximum(np.abs(A), 1e-300)
        A_new = np.sqrt(np.maximum(power_new, 0)) * phase
        a_new = irfft(A_new, fft_size)[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0)
        a_new = normalize_simplex(a_new)

        score = evaluate_local(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralFlatten iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif choice == 1:
        # Gaussian-like
        a = np.exp(-5 * (x - 0.5) ** 2)
    elif choice == 2:
        # Horn/Poisson window variant
        a = np.exp(-4 * (x - 0.5))
    elif choice == 3:
        # Product of cosines (symmetric)
        a = (1 + np.cos(2 * np.pi * x)) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif choice == 4:
        # Flat top
        center = np.abs(x - 0.5) * 2
        a = np.where(center < 0.5, 1.0, 0.5 * (1 + np.cos(2 * np.pi * (center - 0.5))))
    else:
        # Random smooth
        a = np.abs(np.sin(3 * np.pi * x) + 0.5 * np.sin(7 * np.pi * x)) + 0.1
    a = np.maximum(a, 0)
    return normalize_simplex(a)


def quick_optimize(a0, n_iters=50, eta=0.03):
    """Quick mirror descent for initialization ranking."""
    n = len(a0)
    a = normalize_simplex(a0)
    for it in range(n_iters):
        g, b, max_b = compute_focused_subgradient(a, active_frac=0.999, alpha=5.0)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        g_norm = g_shifted.mean()
        if g_norm > 1e-15:
            g_shifted = g_shifted / g_norm * 0.5
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a = normalize_simplex(np.exp(log_a_new))
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_simplex(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Fourier-parameterized global search,
    focused mirror descent, spectral flattening, and targeted coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n = len(best_seq)
    print(f"Main optimization with n={n}")

    # === Phase 1: Diverse initialization (10% time) ===
    print(f"\n=== Phase 1: Diverse Initialization ===")
    rem = deadline - time.monotonic()
    t_p1 = rem * 0.08
    if t_p1 > 15:
        candidates = [best_seq.copy()]
        for i in range(8):
            c = structured_init(n, rng)
            c = quick_optimize(c, n_iters=30, eta=0.03)
            candidates.append(c)
        # Also try a spectral-flattened version of parent
        c_spec, _ = spectral_flatten(best_seq, t_alloc_seconds=10, n_iters=100)
        candidates.append(c_spec)

        # Rank all candidates
        scored = []
        for c in candidates:
            sc = evaluate_local(c)
            scored.append((sc, c))
        scored.sort(key=lambda x: x[0])

        try_update(scored[0][1], "init_rank1")
        try_update(scored[1][1], "init_rank2")
        try_update(scored[2][1], "init_rank3")

    # === Phase 2: Fourier-parameterized L-BFGS (12% time) ===
    print(f"\n=== Phase 2: Fourier L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p2 = rem * 0.12
    if t_p2 > 15:
        a_four, _ = fourier_param_optimize(best_seq, t_p2, k_max=15,
                                           T_stages=[15.0, 8.0, 4.0, 2.0, 1.0,
                                                      0.5, 0.2, 0.1, 0.05, 0.02, 0.01])
        try_update(a_four, "fourier_lbgfs")

    # === Phase 3: Spectral flattening (10% time) ===
    print(f"\n=== Phase 3: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p3 = rem * 0.10
    if t_p3 > 15:
        a_spec, _ = spectral_flatten(best_seq, t_p3, n_iters=2000)
        try_update(a_spec, "spectral_flatten")

    # === Phase 4: Main mirror descent (35% time) ===
    print(f"\n=== Phase 4: Focused Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p4 = rem * 0.35
    if t_p4 > 30:
        print(f"  Budget: {t_p4:.0f}s")
        # Run multiple passes with different parameters
        a_md, _ = mirror_descent_optimize(best_seq, t_p4 * 0.6,
                                          eta0=0.04, active_frac=0.999,
                                          nesterov=True, bb_adapt=True,
                                          alpha_start=15.0, alpha_end=3.0)
        try_update(a_md, "mirror_p4a")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_md2, _ = mirror_descent_optimize(best_seq, min(rem * 0.4, t_p4 * 0.4),
                                               eta0=0.02, active_frac=0.9999,
                                               nesterov=True, bb_adapt=True,
                                               alpha_start=5.0, alpha_end=1.5)
            try_update(a_md2, "mirror_p4b")

    # === Phase 5: Pairwise coordinate descent (25% time) ===
    print(f"\n=== Phase 5: Pairwise CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p5 = rem * 0.25
    if t_p5 > 20:
        a_pcd, _ = pairwise_coordinate_descent(best_seq, t_p5, n_outer=500000)
        try_update(a_pcd, "pairwise_cd")

    # === Phase 6: Fine coordinate descent (15% time) ===
    print(f"\n=== Phase 6: Fine CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p6 = rem * 0.15
    if t_p6 > 15:
        a_fcd, _ = fine_coordinate_descent(best_seq, t_p6, n_steps=500000)
        try_update(a_fcd, "fine_cd")

    # === Phase 7: Final refinement combo (remaining time) ===
    print(f"\n=== Phase 7: Final Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Quick mirror descent with very low eta
        t_md_final = min(rem * 0.4, 120)
        if t_md_final > 10:
            a_fin, _ = mirror_descent_optimize(best_seq, t_md_final,
                                               eta0=0.005, active_frac=0.99999,
                                               nesterov=True, bb_adapt=True,
                                               alpha_start=3.0, alpha_end=1.0)
            try_update(a_fin, "final_md")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = fine_coordinate_descent(best_seq, min(rem * 0.6, 100), n_steps=200000)
            try_update(a_fin2, "final_cd")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = pairwise_coordinate_descent(best_seq, min(rem * 0.5, 60), n_outer=100000)
            try_update(a_fin3, "final_pcd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
