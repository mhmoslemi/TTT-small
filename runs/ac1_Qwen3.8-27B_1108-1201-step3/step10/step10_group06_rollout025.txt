def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):

    n = len(a)
    conv = fast_convolve(a)

import numpy as np

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.CBC, cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                return name
        except Exception:

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS', 'CBC']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def coord_descent(a0, time_limit, random_order=False, verbose=False, label=""):
    """Fast coordinate descent with O(1) amortized max tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve_sweeps = 0
    rng = np.random.default_rng(42)
    
    order = list(range(n))
    
    while time.monotonic() - t0 < time_limit:
        if random_order:
            rng.shuffle(order)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                continue
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            # Affected values
            affected = conv[k_lo:k_hi + 1].copy()
            
            # Partner indices
            partners = np.arange(k_lo, k_hi + 1) - i
            valid = (partners >= 0) & (partners < n)
            a_partners = np.zeros(k_hi - k_lo + 1)
            a_partners[valid] = a[partners[valid]]
            
            # Quadratic index
            quad_idx = 2 * i - k_lo if (2 * i >= k_lo and 2 * i <= k_hi) else -1
            
            # Golden section search for optimal x
            def eval_at(x):
                delta = x - v
                if delta == 0:
                    return 2.0 * n * max(unaff_max, float(affected.max()))
                temp = affected + 2.0 * delta * a_partners
                if quad_idx >= 0:
                    temp[quad_idx] = affected[quad_idx] + delta * (2.0 * v + delta)
                return 2.0 * n * max(unaff_max, float(temp.max()))
            
            lo = 0.0
            hi = max(3.0 * v, 2.0 / n)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_at(c)
            fd = eval_at(d)
            for _ in range(22):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_at(d)
            best_x = (lo + hi) * 0.5
            
            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                changes = 2.0 * delta * a_partners
                if quad_idx >= 0:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max())
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        no_improve_sweeps = 0 if improved else no_improve_sweeps + 1
        if no_improve_sweeps >= 2:
            break
        if verbose and (len(order) % (n // 10) == 0):
            pass
    
    s_best = best_a.sum()
    if s_best > 0.01:
        best_a = best_a / s_best
        best_score = 2.0 * n * float(fast_convolve(best_a).max())
    
    return best_a, best_score

def subgradient_descent(a0, time_limit, verbose=False):
    """Subgradient with peak-focused weighting and adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        
        # Focus on top peaks
        k_peak = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -k_peak)[-k_peak:]
        pv = conv[peaks]
        temp = 0.005 * max_val
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            a_new = a
        else:
            a_new = a_new / cs
        
        sc = 2.0 * n * float(fast_convolve(a_new).max())
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.998
        if no_improve > 80:
            lr *= 0.6
            no_improve = 0
        if no_improve > 400:
            break
    
    return best_a, best_score

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def basin_hopping(a0, time_limit, rng, verbose=False):

            for k in peaks:

                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.03 - 0.04 * rng.random())
        elif pert_type == 1:

            sigma = 0.02 + 0.03 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            # Block swap
            blen = rng.integers(max(2, n // 10), n // 3)
            i1 = rng.integers(0, n - blen)

            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        elif pert_type == 3:

            shift = rng.integers(2, max(3, n // 5))
            shifted = np.roll(a, shift)
            w = 0.2 + 0.3 * rng.random()
            perturbed = (1.0 - w) * a + w * shifted

            center = rng.uniform(0.15, 0.85)

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=float)
    center = n * shift
    width = n * rng.uniform(0.2, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_two_hump(n, rng):
    t = np.arange(n, dtype=float)

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if rem < 40:
            break
        t_sub = min(rem * 0.3, 100.0)
        a_init = interpolate_seq(best_seq, alt_n)
        try:
            a_sub, s_sub = coord_descent(a_init, t_sub)
            sc = float(evaluate_sequence(a_sub.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_sub.copy()
                print(f"  NEW BEST n={alt_n}: {sc:.8f}")
        except Exception:
            continue
    
    # Phase 3: Deep coordinate descent
    print(f"\n=== Phase 3: Deep CD (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        t_cd = min(rem * 0.25, 150.0)
        print(f"  CD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_cd)
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST CD: {sc:.8f}")
        
        # Perturb and retry
        rem2 = deadline - time.monotonic() - 450
        if rem2 > 20 and rnd < 3:
            pert = best_seq + rng.standard_normal(n_work) * 0.015
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            t_cd2 = min(rem2 * 0.3, 40.0)
            a_cd2, s_cd2 = coord_descent(pert, t_cd2)
            sc2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST CD-pert: {sc2:.8f}")
    
    # Phase 4: Basin hopping
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 300
    if rem > 40:
        t_bh = min(rem * 0.7, 250.0)
        print(f"  BH budget={t_bh:.0f}s")
        try:
            a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
            sc = float(evaluate_sequence(a_bh.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_bh.copy()
                print(f"  NEW BEST BH: {sc:.8f}")
        except Exception as e:
            print(f"  BH error: {e}")
    
    # Phase 5: Subgradient + final polish
    print(f"\n=== Phase 5: Subgradient + Final (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_sg = min(rem * 0.4, 80.0)
        print(f"  SG budget={t_sg:.0f}s")
        try:
            a_sg, s_sg = subgradient_descent(best_seq, t_sg)
            sc = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_sg.copy()
                print(f"  NEW BEST SG: {sc:.8f}")
        except Exception:
            pass
    
    # Final coordinate descent polish
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_final = min(rem * 0.6, 50.0)
        print(f"  Final CD budget={t_final:.0f}s")
        a_fin, s_fin = coord_descent(best_seq, t_final)
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST final: {sc:.8f}")
    
    # Last resort: one more short polish
    rem = deadline - time.monotonic() - 10
    if rem > 8:
        a_last, s_last = coord_descent(best_seq, min(rem * 0.7, 20.0))
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
    
    if best_seq is None:
        best_seq = np.ones(512) / 512
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
