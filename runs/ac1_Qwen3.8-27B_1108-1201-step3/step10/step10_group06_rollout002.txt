def fast_convolve(a):
    """Compute full linear convolution using FFT."""

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

        return float('inf')

    return 2.0 * n * float(np.max(conv)) / (s * s)
