def simplex_project(v):

    u = np.sort(v)[::-1]

        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)

    else:
        return np.ones(n) / n

def gradient_optimization(a0, time_limit, label=""):
    """Gradient-based optimization using log-sum-exp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    
    beta = 40.0  # smoothing parameter
    lr = 0.005
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute smoothed objective gradient
        conv = fast_convolve(a)
        max_c = conv.max()
        scaled = beta * (conv - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k in range(len(weights)):
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                grad[i] += 2.0 * w * a[k - i]
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - lr * momentum
        a_new = simplex_project(np.maximum(a_new, 0))
        
        # Evaluate
        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            a = a_new  # still update to explore
            no_improve += 1
            
        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.7
            beta *= 1.2
            no_improve = 0
            if lr < 1e-8:
                break
        elif it % 20 == 0:
            beta *= 1.005
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradOpt {label} it={it} ({elapsed:.0f}s): score={score:.6f}, beta={beta:.1f}, lr={lr:.2e}")

    return best_a, best_score

        s2 = a_new2.sum()
        if s2 > 1e-15:
            a_new2 = a_new2 / s2
        
        # Pick better candidate
        sc1 = 2.0 * n * float(fast_convolve(a_new).max())
        sc2 = 2.0 * n * float(fast_convolve(a_new2).max())
        
        if sc1 < sc2:
            a = a_new
            score = sc1
        else:
            a = a_new2
            score = sc2
            
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 100:
            break
            
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpecShape {label} it={it} ({elapsed:.0f}s): score={score:.6f}")

    return best_a, best_score

def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(np.max(conv[:k_start]))
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(np.max(conv[k_end + 1:]))
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1].copy()
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Fine grid of multipliers
            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                          0.993, 0.99, 0.985, 0.98, 0.97, 0.95,
                          1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.003, 1.005,
                          1.007, 1.01, 1.015, 1.02, 1.03, 1.05]:
                delta = ai * (factor - 1.0)
                if delta < -ai - 1e-15:
                    continue
                delta = max(delta, -ai)
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(np.max(new_affected)))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

def random_perturb(a, rng, strength=0.05):
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Pressure: how much each coordinate contributes to peaks
    pressure = np.zeros(n)
    peak_threshold = M * 0.8
    peak_pos = np.where(conv >= peak_threshold)[0]
    for k in peak_pos:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            pressure[i] += a[i] * a[k - i]
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def propose_candidate():
    """
    Multi-phase optimization:
    1. Nesterov LP Fictitious Play (primary, most time)
    2. Gradient-based optimization with log-sum-exp
    3. Smart coordinate descent
    4. Spectral shaping
    5. Peak reduction
    6. Fine polishing
    7. Multi-length exploration
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}")

    # === Phase 1: Nesterov LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play + Nesterov (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    print(f"  Budget: {t1:.0f}s")
    a1, s1, it1 = lp_fictitious_play_nesterov(best_seq, t1, solver=solver, label="p1")

    # === Phase 2: Gradient optimization ===
    print(f"\n=== Phase 2: Gradient Optimization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

        t2 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = gradient_optimization(best_seq, t2, label="p2")
        best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_grad")

    # === Phase 3: Smart Coordinate Descent ===
    print(f"\n=== Phase 3: Smart CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.2, 300.0)
        print(f"  Budget: {t3:.0f}s")
        a3, s3 = smart_coord_descent(best_seq, t3, label="p3")
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_scd")

    # === Phase 4: Spectral Shaping ===
    print(f"\n=== Phase 4: Spectral Shaping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t4 = min(rem * 0.1, 150.0)
        print(f"  Budget: {t4:.0f}s")
        a4, s4 = spectral_shaping(best_seq, t4, label="p4")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_spec")

    # === Phase 5: Perturbation + Re-optimize ===
    print(f"\n=== Phase 5: Perturb + Re-optimize (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for rnd in range(3):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            perturbed = random_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
            t5 = min(rem * 0.15, 100.0)
            a5, s5 = smart_coord_descent(perturbed, t5, label=f"p5_r{rnd}")
            best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_r{rnd}")

    # === Phase 6: Another round of LP Fictitious Play ===
    print(f"\n=== Phase 6: LP FP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t6 = min(rem * 0.2, 200.0)
        print(f"  Budget: {t6:.0f}s")
        a6, s6, it6 = lp_fictitious_play_nesterov(best_seq, t6, solver=solver, label="p6")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6_lpfp2")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_current = len(best_seq)

            if alt != n_current:

    if rem > 30:
        t8 = min(rem * 0.4, 120.0)
        a8, s8 = fine_coord_descent(best_seq, t8, label="p8a")

    rem = deadline - time.monotonic()
    if rem > 20:
        t8b = min(rem * 0.4, 60.0)
        a8b, s8b = smart_coord_descent(best_seq, t8b, label="p8b")
        best_seq, best_score = check_and_update(a8b, best_seq, best_score, "p8b_scd")

    rem = deadline - time.monotonic()
    if rem > 10:
        a8c, s8c = fine_coord_descent(best_seq, rem * 0.6, label="p8c")
        best_seq, best_score = check_and_update(a8c, best_seq, best_score, "p8c_fine")

    # Final gradient pass
    rem = deadline - time.monotonic()
    if rem > 15:
        a9, s9 = gradient_optimization(best_seq, rem * 0.5, label="p9")
        best_seq, best_score = check_and_update(a9, best_seq, best_score, "p9_grad")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
