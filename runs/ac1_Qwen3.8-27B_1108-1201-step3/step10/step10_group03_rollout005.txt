"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel Fourier-band 
parameterization for low-dimensional optimization, combined with 
bidirectional LP alternation and LogSumExp-annealed L-BFGS.
Key innovation: projecting the sequence to a k-band Fourier series 
reduces the effective dimensionality from n to ~2k, enabling 
much more efficient global search.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_fixed_a(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver] + [alt for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if alt != solver]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                ss = b_val.sum()
                if ss > 1e-15:
                    return b_val / ss
        except Exception:
            continue
    return None


def lp_alternation(a0, time_limit, solver, max_iters=10000):
    """Bidirectional LP alternation with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    y_a = a.copy()
    y_b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_fixed_a(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = normalize((1.0 - alpha) * y_b + alpha * b_new)

        a_new = solve_lp_fixed_a(y_b, solver)  # by symmetry of the problem
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = normalize((1.0 - alpha) * y_a + alpha * a_new)

        # Evaluate averaged points
        for candidate in [y_a, y_b, 
                          normalize(0.5*(y_a + y_b))]:
            sc = 2.0 * n * fast_convolve(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                y_a = normalize(y_a + rng.standard_normal(n) * 0.01)
                y_b = normalize(y_b + rng.standard_normal(n) * 0.01)
                stall = 0

    return best_a, best_score


def fourier_band_optimize(a0, n_bands, time_limit):
    """
    Optimize by restricting the DFT to the first n_bands frequency components.
    This reduces the problem to ~2*(n_bands-1) parameters.
    """
    n = len(a0)
    a = normalize(a0)
    N = 1 << (2 * n - 1).bit_length()
    
    A_full = rfft(a, N)
    
    # Parameters: Re and Im of A[1:n_bands]
    n_params = 2 * min(n_bands - 1, N // 2)
    if n_params == 0:
        return a, local_eval(a)
    
    # Initial values
    x0 = np.zeros(n_params)
    for i in range(1, min(n_bands, N // 2 + 1)):
        idx = 2 * (i - 1)
        x0[idx] = A_full[i].real
        x0[idx + 1] = A_full[i].imag
    
    # DC component (fix to maintain normalization)
    A_dc = A_full[0].real  # sum of a = 1 (normalized)
    
    # Fixed high-frequency components (set to 0 for band-limited)
    
    def get_sequence(x):
        A = np.zeros(N // 2 + 1, dtype=complex)
        A[0] = A_dc
        for i in range(1, min(n_bands, N // 2 + 1)):
            idx = 2 * (i - 1)
            A[i] = x[idx] + 1j * x[idx + 1]
        a_seq = irfft(A, N)[:n]
        return np.maximum(a_seq, 0)
    
    def objective(x):
        a_seq = get_sequence(x)
        s = a_seq.sum()
        if s < 1e-10:
            return 1e15
        a_norm = a_seq / s
        b = fast_convolve(a_norm)
        return 2.0 * n * b.max()
    
    def gradient(x):
        a_seq = get_sequence(x)
        s = a_seq.sum()
        if s < 1e-10:
            return np.zeros(n_params)
        a_norm = a_seq / s
        b = fast_convolve(a_norm)
        M = b.max()
        # Subgradient of max: uniform over argmax
        active = b >= M * (1 - 1e-10)
        w = active / active.sum()
        # grad of b[k] w.r.t. a: 2*a[k-j] for valid j
        grad_a = np.zeros(n)
        for k in np.where(w > 0)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_norm[k - idx]
        # Chain rule: a_seq = max(irfft(A(x)), 0) / sum
        # This is complex; use numerical gradient
        eps = 1e-7
        grad = np.zeros(n_params)
        for i in range(n_params):
            xp = x.copy()
            xp[i] += eps
            xm = x.copy()
            xm[i] -= eps
            grad[i] = (objective(xp) - objective(xm)) / (2 * eps)
        return grad
    
    best_x = x0.copy()
    best_score = objective(x0)
    t0 = time.monotonic()
    
    # Nelder-Mead first (derivative-free, good for low dim)
    try:
        res = minimize(objective, x0, method='Nelder-Mead',
                      options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
        if res.fun < best_score:
            best_score = res.fun
            best_x = res.x
    except Exception:
        pass
    
    # L-BFGS with numerical gradient
    if time.monotonic() - t0 < time_limit - 5:
        try:
            res2 = minimize(objective, best_x, method='L-BFGS-B',
                           options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-10,
                                    'maxfun': 100000})
            if res2.fun < best_score:
                best_score = res2.fun
                best_x = res2.x
        except Exception:
            pass
    
    # Reconstruct
    a_best = get_sequence(best_x)
    s = a_best.sum()
    if s > 1e-10:
        a_best = a_best / s
