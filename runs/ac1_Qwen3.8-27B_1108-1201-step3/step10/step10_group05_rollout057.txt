def fista_smoothed(a_init, time_limit, beta_init=20.0, beta_growth=1.05, label=""):
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_nesterov = 1.0
    t0 = time.monotonic()
    beta = beta_init
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    it = 0
    no_improve = 0
    lr = 1.0 / np.sqrt(beta * n)  # Initial learning rate
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov update
        if it > 1:
            t_prev = t_nesterov
            t_nesterov = (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov**2)) / 2.0
            y = a + (t_prev - 1.0) / t_nesterov * (a - y)
        
        # Project y onto simplex
        y = simplex_project(np.maximum(y, 0.0))
        
        # Compute smoothed gradient at y
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -100, 0)
        weights = np.exp(scaled)
        weights /= weights.sum()
        
        # Gradient via FFT cross-correlation
        L = 1 << (2 * n).bit_length()
        W = rfft(weights, L)
        Y = rfft(y, L)
        grad = 2.0 * irfft(np.conj(W) * Y, L)[:n]
        
        # Mirror descent update (exponentiated gradient)
        # a_new[i] = a[i] * exp(-lr * grad[i]) / sum(a[j] * exp(-lr * grad[j]))
        log_update = -lr * grad
        log_update -= log_update.max()  # For numerical stability
        update = np.exp(log_update)
        a_new = a * update
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        
        # Evaluate
        conv_a = fast_convolve(a_new)
        score = 2.0 * n * conv_a.max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 50:
            beta *= beta_growth
            lr /= 1.5
            no_improve = 0
            if lr < 1e-12:
                break
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_a, best_score
