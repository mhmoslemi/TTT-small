def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        proj = np.maximum(v - theta, 0.0)
    else:
        proj = np.ones(n) / n
    return proj * target_sum

def gs_flat_spectrum(n, n_iter=5000, seed=42):
    Gerchberg-Saxton algorithm to find a non-negative sequence with near-flat power spectrum.
    The target is |FFT(a_padded)| = constant, which corresponds to a flat autocorrelation.
    Non-negativity constraint makes perfect flatness impossible, so we find the best compromise.
    rng = np.random.default_rng(seed)
    fft_size = 2 * n - 1
    N = 1 << (fft_size - 1).bit_length()
    
    # Initialize with random phases, unit magnitude
    A = np.zeros(N, dtype=np.complex128)
    half = N // 2
    for k in range(half + 1):
        A[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
    for k in range(1, half + 1):
        A[N - k] = np.conj(A[k])
    
    a = np.ones(n) / n
    
    for it in range(n_iter):
        # Time domain: IFT, clip to non-negative, keep support
        a_full = np.fft.ifft(A).real
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        
        # Frequency domain: FFT, enforce flat magnitude
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A_new = np.fft.fft(a_padded)
        
        mag = np.abs(A_new)
        avg_mag = mag.mean()
        if avg_mag < 1e-15:
            break
        # Replace magnitude with average, keep phase
        with np.errstate(divide='ignore', invalid='ignore'):
            phase = A_new / np.maximum(mag, 1e-30)
        A = avg_mag * phase
        # Restore conjugate symmetry
        for k in range(1, half + 1):
            A[N - k] = np.conj(A[k])
    
    conv = fast_convolve(a)
    score = 2.0 * n * conv.max()
    return a, score

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

        prob.solve(solver=solver)

        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:

                break

                continue

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

    return b_val, float(M_var.value)

def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)

        sweeps += 1
        improved = False
        for i in range(n):

                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)

                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:

    return best_a, best_score, sweeps
