conv(a,a)[k*] = sum_i a[i] * a[k*-i]

We can try to reduce this by:
- Reducing the products of large a[i] * a[k*-i]
- Specifically, find the pair (i, j) with i+j = k* that maximizes a[i]*a[j]
- Transfer mass from the larger of a[i], a[j] to a position that doesn't contribute to any peak

This is a greedy "peak shaving" approach. It's local but can be very effective when combined with other methods.

def fast_conv(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 2 * n
    # Use next power of 2 for efficiency
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2*n-1]

def score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def compute_subgrad_topk(a, K=10, T=0.1):
    """Compute weighted subgradient using top-K peaks."""
    n = len(a)
    b = fast_conv(a)
    # Find top K peaks
    sorted_idx = np.argsort(b)[::-1][:K]
    # Soft weights based on peak values
    peak_vals = b[sorted_idx]
    weights = np.exp(peak_vals / (T * peak_vals.max() + 1e-300))
    weights /= weights.sum()

        je = min(n - 1, k)

            g[j] += w * a[k - j]
    return g, b.max()

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)


def softmax_robust(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    if n > 3000:
        return None, None, None
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    # Try to get dual variables for the constraint T@b <= M
    dual_vals = None
    try:
        dual_list = prob.dual_values
        if dual_list is not None and len(dual_list) > 0:
            dual_vals = np.asarray(dual_list[0], dtype=np.float64)
            if dual_vals.shape[0] == 2 * n - 1:
                dual_vals = np.maximum(dual_vals, 0)
    except Exception:
        dual_vals = None
    return b_val, float(M_var.value), dual_vals

def nesterov_lp_improved(a0, time_limit, solver=None, n_max_iters=200000):
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    t_nest = 1.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, dual = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Standard Nesterov acceleration
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        gamma = (t_nest - 1.0) / t_next

        # Update y: y = (1 - gamma) * a + gamma * b_new
        y = (1.0 - gamma) * y + gamma * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update a: a = b_new (the iterate)
        a = b_new.copy()

        score_a = compute_score(a)
        score_y = compute_score(y)

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

        t_nest = t_next

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def topk_mirror_descent(a0, time_limit, K=20, T_param=0.03, denoise_sigma=0.0):
    n = len(a0)
    a = normalize_nonneg(a0)
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0
    rng = np.random.default_rng(77777)
    prev_best = best_score
    stall_counter = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        # Top-K peak weighting
        sorted_idx = np.argsort(b)[::-1][:K]
        peak_vals = b[sorted_idx]
        scale = T_param * max_b + 1e-300
        weights = np.exp((peak_vals - peak_vals[0]) / scale)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * 2.0 * a[k - idx]

        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        gap = best_score - 2.0 * n / (2.0 * n - 1)
        if gap < 1e-9:
            break

        alpha = lr
        improved = False
        for bt in range(10):
            z_new = z - alpha * g
            a_new = softmax_robust(z_new)
            if denoise_sigma > 0:
                a_new = gaussian_filter1d(a_np(a_new), denoise_sigma)
                a_new = normalize_nonneg(a_new)
            sc = compute_score(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
                a = a_new
                no_improve = 0
                stall_counter = 0
                lr = min(alpha * 1.05, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.9, 0.01)
            if no_improve >= 25:
                noise = rng.random(n) * 0.03
                a = a + noise
                a = normalize_nonneg(a)
                z = np.log(np.maximum(a, 1e-300))
                no_improve = 0
                lr = 0.5

        if (it_check(best_score, prev_best, stall_counter) < 0):
            stall_counter += 1
            if stall_counter >= 40:
                rng2 = np.random.default_rng(step * 31 + 7)
                a = a * (1.0 + rng2.normal(0, 0.05, n))
                a = normalize_nonneg(a)
                z = np.log(np.maximum(a, 1e-300))
                stall_counter = 0
                prev_best = best_score
                lr = 0.5
        else:
            prev_best = best_score

        if step % 100000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-MD step={step}: score={best_score:.6f} lr={lr:.3f}")

    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=200000):
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stuck = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[k_star - idx]

        max_pos = np.argmax(products)
        j1 = idx[max_pos]
        j2 = k_star - j1

        if a[j1] >= a[j2]:
            src = j1
        else:
            src = j2

        top_k = np.argsort(b)[::-1][:8]
        unsafe = set()
        for k2 in top_k:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            unsafe.update(range(js2, je2 + 1))

        safe_positions = [j for j in range(n) if j not in unsafe]
        if not safe_positions:
            safe_positions = [j for j in range(n) if j != src]

        improved = False
        for attempt in range(3):
            dst = rng.choice(safe_positions)
            for frac in [0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize_nonneg(a_trial)
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stuck = 0
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stuck += 1
            if stuck > 1000:
                break

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=8):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_robust(z_in)
        b = fast_conv(a)
        Tb = T_val * b
        m = Tb.max()
        if m > 700:
            m = 700
        e = np.exp(Tb - m)
        lse = m + np.log(e.sum())
        return (2.0 * n / T_val) * lse

    def lse_jac(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_robust(z_in)
        b = fast_conv(a)
        Tb = T_val * b
        m = Tb.max()
        if m > 700:
            m = 700
        e = np.exp(Tb - m)
        w = e / e.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in, T_v=stage_T):
            return lse_obj(z_in, T_v)

        def jac(z_in, T_v=stage_T):
            return lse_jac(z_in, T_v)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 30, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': 20000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_robust(z)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Correct Nesterov-accelerated LP best-response
2. Top-K peak-equalization mirror descent with optional spectral smoothing
3. Aggressive multi-length exploration (n=1200 to n=5000)
4. Peak-shaving local search
5. Log-sum-exp L-BFGS with temperature annealing
Key: dual-aware step sizing, FFT for large n, adaptive time allocation.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)


def softmax_robust(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    if n > 2500:
        return None, None
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

def nesterov_lp(a0, time_limit, solver=None):
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    t_nest = 1.0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        gamma = (t_nest - 1.0) / t_next
        y = (1.0 - gamma) * y + gamma * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = b_new.copy()
        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
        t_nest = t_next
        if (it + 1) % 500 == 0:
            print(f"    NLP it={it+1} t={time.monotonic()-t0:.0f}s: {best_score:.6f}")
    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=8):

def _run_lse_stage(z, T_val, n, time_limit, t0):
    def func(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_robust(z_in)
        b = fast_conv(a)
        Tb = T_val * b
        m = Tb.max()
        if m > 700:
            m = 700
        e = np.exp(Tb - m)
        return (2.0 * n / T_val) * (m + np.log(e.sum()))

    def jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_robust(z_in)
        b = fast_conv(a)
        Tb = T_val * b
        m = Tb.max()
        if m > 700:
            m = 700
        e = np.exp(Tb - m)
        w = e / e.sum()
        grad_a = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 30, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': 20000, 'disp': False})
        return res.x
    except Exception:
        return z


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
