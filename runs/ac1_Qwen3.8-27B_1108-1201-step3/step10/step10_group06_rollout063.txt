                break
            v = a[i]
            if v < 1e-14:
                continue
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            n_aff = k_hi - k_lo + 1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(3.0 * v, 5.0 / n)

            lo = 0.0
            hi = upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)

            def _eval(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv_c = nv.copy()
                    nv_c[quad_idx] = c_vals[quad_idx] + x * x
                    nv = nv_c
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            fc = _eval(c)
            fd = _eval(d)
            for _ in range(30):
                if hi - lo < 1e-14 * max(1.0, abs(v)):
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _eval(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _eval(d)

            best_x = (lo + hi) * 0.5
            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        alpha = 2.0 / (iters + 2.0)
        a = (1.0 - alpha) * a + alpha * b
        a = np.maximum(a, 0)
        cs = a.sum()
        if cs < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs2 = cand.sum()
            if cs2 < 1e-10:
                continue
            cand = cand / cs2
            sc = compute_score(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()
                no_improve = 0
            else:
                no_improve += 1

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.8f}, M={M_val:.6f}")

        if no_improve > 50:
            if iters > 20:
                break

    return best_a, best_score, iters

def basin_hopping(a0, time_limit, rng, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 80.0)
        if rem < 60:
            break

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            conv_p = fast_convolve(a)
            top_k = min(10, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            sigma = 0.03
            for k in peaks:
                kn = int(k)
                i_start = max(0, kn - n + 1)
                i_end = min(n - 1, kn)
                idx = np.arange(i_start, i_end + 1)
                for ii in idx:
                    perturbed[ii] *= (1.0 - sigma * rng.random())
        elif pert_type == 1:
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            blen = max(2, rng.integers(n // 10, n // 3))
            i1 = int(rng.integers(0, n - blen))
            i2 = int(rng.integers(0, n - blen))
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        elif pert_type == 3:
            shift = int(rng.integers(1, max(2, n // 5)))
            shifted = np.roll(a, shift)
            perturbed = 0.7 * a + 0.3 * shifted
        else:
            # Asymmetry push
            mid = n // 2
            if rng.random() < 0.5:
                perturbed[:mid] *= (1.0 + 0.1 * rng.random())
                perturbed[mid:] *= (1.0 - 0.05 * rng.random())
            else:
                perturbed[:mid] *= (1.0 - 0.05 * rng.random())
                perturbed[mid:] *= (1.0 + 0.1 * rng.random())

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, min(t_alloc, 60.0), label=f"bh{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_skewed(n, rng):
    """Initialize with a skewed (asymmetric) distribution."""
    beta1 = rng.uniform(1.5, 4.0)
    beta2 = rng.uniform(1.5, 4.0)
    a = (1.0 + np.arange(n) / n) ** beta1 * (1.0 - np.arange(n) / n + 1e-10) ** beta2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
