def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a[j]
    return T

    # For row k: columns max(0, k-n+1) to min(n-1, k) get value a[j]

    # This means T[k,j] = a[j] * (1 if j <= k < j+n else 0)

    T = np.zeros((size, n))

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def proximal_peak_equalization(a, time_limit, n_steps=50000):
    """Proximal gradient method targeting the active peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()

    else:
        a = a / s

    t0 = time.monotonic()
    
    eta = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: peaks within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: average of gradients of active quadratic forms
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Gradient step
        a_try = a - eta * g
        
        # Project onto {a >= 0, sum(a) = 1} using soft thresholding
        # Find tau such that sum(max(0, a_try - tau)) = 1
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try <= 0:
            a_try = a.copy()
        elif s_try != 1:
            # Simple: just normalize
            a_try = a_try / s_try
        
        score = 2.0 * n * fast_convolve(a_try).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-6:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
