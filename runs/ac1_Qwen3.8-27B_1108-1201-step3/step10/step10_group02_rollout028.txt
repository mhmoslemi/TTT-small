def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))

    """Mirror descent with Barzilai-Borwein step size and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(max_its):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = compute_subgradient(y, active)

        if bb_adapt and it >= 3 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-5, min(eta, 1.0))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new
        score = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 7 + 13)
            noise_scale = 0.003 * np.sqrt(no_improve / 500)
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return _lp_solver_cached
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov LP alternating for refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(42424)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            if stall_rounds >= 10:
                noise_scale = 0.01 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def paired_perturbation_search(a0, time_limit, n_trials=200000):
    """
    Novel perturbation-based local search using paired coordinate adjustments.
    For each peak in the convolution, finds the pair (i,j) with i+j=k that
    contributes most, and tries reducing both while compensating elsewhere.
    Also tries random global perturbations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False

        # Strategy 1: Paired reduction at peak-contributing positions
        peak_k = peaks[trial % len(peaks)]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if len(idx) >= 2:
            # Top contributing pairs
            order = np.argsort(contribs)[::-1]
            for oi in range(min(4, len(order))):
                i_local = idx[order[oi]]
                j_local = peak_k - i_local
                if i_local == j_local:
                    # Single coordinate
                    a_trial = a.copy()
                    for factor in [0.998, 0.995, 0.99, 0.985, 0.98]:
                        a_trial = a.copy()
                        a_trial[i_local] *= factor
                        diff = a[i_local] * (1 - factor)
                        if diff < 1e-16:
                            continue
                        mask = np.ones(n, dtype=bool)
                        mask[i_local] = False
                        others = a_trial[mask]
                        os = others.sum()
                        if os < 1e-15:
                            break
                        a_trial[mask] = others + diff * others / os
                        a_trial = a_trial / a_trial.sum()
                        score = evaluate_local(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                else:
                    # Paired: reduce both i and j
                    if j_local < 0 or j_local >= n:
                        continue
                    a_trial = a.copy()
                    for factor in [0.998, 0.995, 0.99, 0.985, 0.98, 0.97]:
                        a_trial = a.copy()
                        a_trial[i_local] *= factor
                        a_trial[j_local] *= factor
                        diff_i = a[i_local] * (1 - factor)
                        diff_j = a[j_local] * (1 - factor)
                        total_diff = diff_i + diff_j
                        if total_diff < 1e-16:
                            break
                        mask = np.ones(n, dtype=bool)
                        mask[i_local] = False
                        mask[j_local] = False
                        others = a_trial[mask]
                        os = others.sum()
                        if os < 1e-15:
                            break
                        a_trial[mask] = others + total_diff * others / os
                        a_trial = a_trial / a_trial.sum()
                        score = evaluate_local(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            # Strategy 2: Random global perturbation
            if trial % 3 == 0:
                a_trial = a.copy()
                perturbation_scale = 0.005 / (1 + stall * 0.01)
                mask = rng.random(n) < 0.3
                a_trial[mask] *= (1.0 + rng.standard_normal(mask.sum()) * perturbation_scale)
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts > 1e-10:
                    a_trial = a_trial / ts
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True

            # Strategy 3: Transfer mass from peak contributors to non-contributors
            if not improved and trial % 5 == 0:
                a_trial = a.copy()
                peak_k2 = int(np.argmax(b))
                js2 = max(0, peak_k2 - n + 1)
                je2 = min(n - 1, peak_k2)
                idx2 = np.arange(js2, je2 + 1)
                contribs2 = a[idx2] * a[peak_k2 - idx2]
                top_pos = idx2[np.argsort(contribs2)[::-1]][:2]
                # Transfer to a random low-contributing position
                non_peak = np.setdiff1d(np.arange(n), top_pos)
                if len(non_peak) > 0:
                    target = non_peak[rng.integers(0, len(non_peak))]
                    for src in top_pos:
                        if a[src] > 1e-12:
                            a_trial = a.copy()
                            transfer = a[src] * 0.005
                            a_trial[src] -= transfer
                            a_trial[target] += transfer
                            a_trial = np.maximum(a_trial, 0)
                            ts = a_trial.sum()
                            if ts > 1e-10:
                                a_trial = a_trial / ts
                                score = evaluate_local(a_trial)
                                if score < best_score - 1e-14:
                                    a = a_trial
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    break

        if not improved:
            stall += 1
            if stall > 20000:
                break
        else:
            stall = 0

        if (trial + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Perturb trial={trial+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func_lse(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_value_and_grad_z(z_inner, T)
            return obj

        def jac_lse(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_value_and_grad_z(z_inner, T)
            return grad

        try:
            res = minimize(fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-13, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_stable(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence with various patterns."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    else:
        # M-shaped: high at edges, low in middle
        a = 0.3 + 0.7 * (1 - np.sin(np.pi * x))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Adam-softmax, mirror descent,
    LP alternating, L-BFGS, and perturbation-based local search.
    Explores multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Adam with softmax parameterization and temperature annealing ===
    print(f"\n=== Phase 1: Adam-Softmax ({best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_1 = [50.0, 30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5,
                      2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        a_adam, _ = adam_softmax_optimize(best_seq, t_alloc, T_stages=T_stages_1, lr0=0.3)
        try_update(a_adam, "adam_softmax")

    # === Phase 2: Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 2: Mirror Descent ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 420.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc,
                                          eta0=0.05, active_frac=0.999,
                                          nesterov=True, bb_adapt=True)
        try_update(a_md, "mirror_descent")

    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: Nesterov LP ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 4: L-BFGS on log-sum-exp ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 220.0)
    if t_alloc > 20:
        T_stages_4 = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages_4, maxiter_per_stage=5000)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Paired Perturbation Search ===
    print(f"\n=== Phase 5: Paired Perturbation ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pert, _ = paired_perturbation_search(best_seq, t_alloc, n_trials=300000)
        try_update(a_pert, "perturbation")

    # === Phase 6: Second Mirror Descent pass (finer) ===
    print(f"\n=== Phase 6: Refine Mirror Descent ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 80.0)
    if t_alloc > 10:
        a_md2, _ = mirror_descent_optimize(best_seq, t_alloc,
                                           eta0=0.015, active_frac=0.9999,
                                           nesterov=True, bb_adapt=True)
        try_update(a_md2, "mirror_descent_fine")

    # === Phase 7: Try different n values ===
    print(f"\n=== Phase 7: Multi-n exploration ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_candidates = [int(n_best * 0.85), n_best, int(n_best * 1.15), int(n_best * 1.3)]
        n_candidates = sorted(set([max(400, nc) for nc in n_candidates]))
        t_per_n = rem * 0.7 / max(1, len(n_candidates))
        for nc in n_candidates:
            if time.monotonic() > deadline - 30:
                break
            if nc == n_best:
                continue
            print(f"  Trying n={nc} (budget={t_per_n:.0f}s)")
            a_interp = interpolate_sequence(best_seq, nc)
            a_nc, _ = mirror_descent_optimize(a_interp, t_per_n,
                                              eta0=0.04, active_frac=0.999,
                                              nesterov=True, bb_adapt=True)
            try_update(a_nc, f"multi_n_{nc}")

    # === Phase 8: Final Adam-Softmax polish ===
    print(f"\n=== Phase 8: Final Adam Polish ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.3, 100.0)
    if t_alloc > 10:
        T_stages_8 = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_final_adam = adam_softmax_optimize(best_seq, t_alloc, T_stages=T_stages_8, lr0=0.1)
        try_update(a_final_adam, "final_adam")

    # === Phase 9: Final perturbation polish ===
    print(f"\n=== Phase 9: Final Perturbation ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.5, 60.0)
    if t_alloc > 5:
        a_final_pert, _ = paired_perturbation_search(best_seq, t_alloc, n_trials=100000)
        try_update(a_final_pert, "final_pert")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
