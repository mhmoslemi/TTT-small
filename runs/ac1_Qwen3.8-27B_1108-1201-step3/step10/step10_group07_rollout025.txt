def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):

    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        cp.sum(b_var) == 1

    try:

def nesterov_lp_alternating(a0, time_limit, solver, verbose=True):

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    momentum = np.zeros(n)
    beta = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Nesterov extrapolation
        if it > 1:
            beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y = a + ((1.0 + beta) / (1.0 + beta_t)) * (a - a_prev)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            a_prev = a.copy()
            beta = beta_t
        else:
            y = a.copy()

        # Solve LP best response
        if it <= 10 or it % 5 == 0:
            b_new, M_lp = lp_best_response(y, solver)

        if not np.isfinite(M_lp):
            break

        # Evaluate
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Damped update with adaptive alpha

            alpha = 1.0
        elif it <= 20:
            alpha = 0.8
        elif it <= 50:
            alpha = 0.6
        else:
            alpha = max(0.3, 8.0 / (it + 5.0))

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0.0)
        as_val = a.sum()
        if as_val < 1e-15:
            a = b_new.copy()
        else:
            a = a / as_val

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a

        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Focus on coordinates that contribute to top peaks
        K = 8
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        # Build set of "important" coordinates
        important = set()
        for k_star in peak_indices:
            k_star = int(k_star)
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            for i in range(i_start, i_end + 1):
                important.add(i)

        # Process important coordinates first, then the rest
        coords = list(important)
        coords.sort()
        for i in range(n):
            if i not in important:
                coords.append(i)

        for i in coords:
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1].copy()

            best_delta = 0.0
            best_new_M = M
            base = max(a[i], 1e-8)

            # Try a range of multiplicative factors
            for factor in [-0.8, -0.6, -0.4, -0.25, -0.1, -0.04, 0.04, 0.1, 0.25, 0.4, 0.6, 0.8]:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def spectral_shaping(a0, time_limit, verbose=True):
    """
    Uses FFT to reshape the spectrum for flatter auto-convolution.
    Iteratively adjusts the phase/magnitude to flatten |A(w)|^2.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power.sum() / len(power)

        # Desired magnitude: flatten towards target
        # Use soft clipping: scale down peaks, scale up valleys
        ratio = target / (power + 1e-20)
        # Blend between current and target
        blend = 0.3
        desired_mag = np.abs(A) * (1.0 - blend + blend * np.sqrt(np.maximum(ratio, 0.1)))

        # Reconstruct with desired magnitude and current phase
        phase = np.angle(A)
        A_new = desired_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        # Damped update
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

        if it % 100 == 0:
            # Add small perturbation to escape local minima
            rng = np.random.default_rng(it)
            a += rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Very fine multiplicative coordinate descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.997, 0.994, 0.991, 0.9985, 1.0015, 1.009, 1.012, 1.015]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / (center + 1e-10), 0)
    elif init_type == 1:
        x = (t - center) / (center + 1e-10)
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / (center + 1e-10)
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        c1 = n * rng.uniform(0.15, 0.3)
        c2 = n * rng.uniform(0.7, 0.85)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # U-shaped with noise
        x = (t - center) / (center + 1e-10)
        a = 0.2 + 0.8 * x * x + 0.05 * rng.standard_normal(n)
    elif init_type == 5:
        # Periodic with period ~n/3
        period = max(3, n // 3)
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * t / period)
    elif init_type == 6:
        # Arcsine-like
        x = (t - center) / (center + 1e-10)
        a = 1.0 / np.sqrt(np.maximum(1.0 - x * x, 0.01))
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def subgradient_descent(a0, time_limit, lr_init=0.01, top_k=30, verbose=True):
    """Subgradient descent with multi-peak weighting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining LP alternating with Nesterov acceleration,
    peak-focused coordinate descent, spectral shaping, and fine polishing.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: Quick diverse initialization scan ===
    print(f"\n=== Phase 1: Init Scan (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.05
    inits = [diverse_init(n, rng, i) for i in range(7)]
    for idx, init in enumerate(inits):
        sc = 2.0 * n * fast_convolve(init).max()
        if sc < best_score:
            try_update(init, f"init_{idx}")
    print(f"  After init scan: best={best_score:.8f}")

    # === Phase 2: Nesterov LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: Nesterov LP Alt (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.45

    # Run from parent/best
    t_alloc = min(t2_end - time.monotonic(), 400.0)
    print(f"  NLP from best, budget={t_alloc:.0f}s")
    a_lp, s_lp = nesterov_lp_alternating(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "nlp_main")

    # Run from a few diverse inits
    for idx in [0, 1, 6]:
        if time.monotonic() > t2_end - 30:
            break
        remaining = t2_end - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(remaining * 0.4, 120.0)
        print(f"  NLP init {idx}, budget={t_alloc:.0f}s")
        a_tmp, s_tmp = nesterov_lp_alternating(inits[idx], t_alloc, solver, verbose=False)
        try_update(a_tmp, f"nlp_init{idx}")

    # Second round on best
    if time.monotonic() < t2_end - 60:
        t_alloc = min(t2_end - time.monotonic(), 200.0)
        print(f"  NLP round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = nesterov_lp_alternating(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "nlp_r2")

    # === Phase 3: Peak-Focused Coordinate Descent ===
    print(f"\n=== Phase 3: Peak CD (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t3_end - time.monotonic(), 250.0)
    if t_alloc > 30:
        print(f"  PeakCD budget={t_alloc:.0f}s")
        a_cd, s_cd = peak_focused_cd(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "peakcd")

        if time.monotonic() < t3_end - 40:
            t_alloc2 = min(t3_end - time.monotonic(), 100.0)
            a_cd2, s_cd2 = peak_focused_cd(a_lp, t_alloc2, verbose=False, label="lp")
            try_update(a_cd2, "peakcd_lp")

    # === Phase 4: Spectral Shaping + SGD ===
    print(f"\n=== Phase 4: Spectral + SGD (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    if time.monotonic() < t4_end - 30:
        t_ss = min(t4_end - time.monotonic() - 30, 80.0)
        if t_ss > 20:
            print(f"  Spectral budget={t_ss:.0f}s")
            a_ss, s_ss = spectral_shaping(best_seq, t_ss, verbose=True)
            try_update(a_ss, "spectral")

    if time.monotonic() < t4_end - 20:
        t_sgd = min(t4_end - time.monotonic() - 20, 80.0)
        if t_sgd > 20:
            print(f"  SGD budget={t_sgd:.0f}s")
            a_sgd, s_sgd = subgradient_descent(best_seq, t_sgd, lr_init=0.008, top_k=40, verbose=True)
            try_update(a_sgd, "sgd")

    # === Phase 5: Fine Polish + Final CD ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.8f}) ===")
    t5_end = deadline - 30

    if time.monotonic() < t5_end - 40:
        t_fp = min(t5_end - time.monotonic() - 40, 120.0)
        if t_fp > 20:
            print(f"  FinePolish budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine")

    # Another peak CD pass
    if time.monotonic() < t5_end - 20:
        t_pc = min(t5_end - time.monotonic() - 20, 80.0)
        if t_pc > 15:
            a_pc, s_pc = peak_focused_cd(best_seq, t_pc, verbose=False, label="final")
            try_update(a_pc, "final_cd")

    # Final NLP round
    if time.monotonic() < t5_end - 10:
        t_fn = min(t5_end - time.monotonic() - 10, 60.0)
        if t_fn > 15:
            a_fn, s_fn = nesterov_lp_alternating(best_seq, t_fn, solver, verbose=False)
            try_update(a_fn, "final_nlp")

    # Final fine polish
    if time.monotonic() < deadline - 5:
        t_ff = min(deadline - time.monotonic() - 5, 30.0)
        if t_ff > 10:
            a_ff, s_ff = fine_polish(best_seq, t_ff, verbose=False)
            try_update(a_ff, "final_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
