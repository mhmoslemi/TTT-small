from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def subgradient(a, active_frac=0.999):
    """Compute subgradient of max_k (a*a)[k]"""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * active_frac
    active = np.where(b >= threshold)[0]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active))
    return g, b

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):

def compute_subgrad(a, active_frac=0.999):
    """Compute subgradient of max_k conv(a,a)[k] at a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * active_frac
    active = np.where(b >= threshold)[0]
    if len(active) == 0:

    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active))
    return g, b, active

def spectral_flatten(a0, time_limit, alpha=0.7, n_iters=None):
    """
    Spectral flattening: iteratively modify the power spectrum to be flatter,
    then project back to non-negative simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()

    if n_iters is None:
        n_iters = 50000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute power spectrum
        A = rfft(a, fft_size)
        P = A * A  # This is real and non-negative (|A|^2 for real a)
        # Actually for real a, rfft gives complex, so P = |A|^2
        P = np.abs(A) ** 2

        # Flatten: P_new = P^alpha / sum(P^alpha) * sum(P)
        P_shifted = P - P.min() + 1e-15
        P_new = P_shifted ** alpha
        P_new = P_new / P_new.sum() * P.sum()

        # Recover amplitude
        new_amp = np.sqrt(np.maximum(P_new, 0))
        # Keep original phase
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)

        # Inverse FFT
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            continue
        a_new = a_new / a_new_sum

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()

        if it % 2000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={it} ({elapsed:.1f}s): best={best_score:.8f}")

    return best_a, best_score


def contribution_coord_descent(a0, time_limit, n_steps=50000, top_k=8):

    Contribution-aware coordinate descent:
    For the dominant peak, score each coordinate by its contribution to the peak
    divided by its impact on other peaks, then reduce the highest-scored coordinate.
    """

    s = a.sum()

    stall = 0

        if time.monotonic() - t0 > time_limit:
            break

        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find all significant peaks
        threshold = max_b * 0.998
        peaks = np.where(b >= threshold)[0]

        # For the main peak, compute contribution scores
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        idx = idx[idx >= 0] & (idx < n) if len(idx) > 0 else idx

        if len(idx) == 0:
            break

        # Contribution to main peak
        partner_idx = peak_k - idx
        valid = (partner_idx >= 0) & (partner_idx < n)
        idx_valid = idx[valid]
        partner_valid = partner_idx[valid]

        if len(idx_valid) == 0:
            break

        contrib = a[idx_valid] * a[partner_valid]

        # Impact on other peaks (approximate: max contribution to second peak)
        other_impact = np.zeros(len(idx_valid))
        if len(peaks) > 1:
            other_peaks = peaks[peaks != peak_k][:3]  # Look at up to 3 other peaks
            for op in other_peaks:
                op_js = max(0, op - n + 1)
                op_je = min(n - 1, op)
                # Which of our idx_valid indices contribute to this other peak?
                op_partner = op - idx_valid
                op_valid = (op_partner >= 0) & (op_partner < n)
                if np.any(op_valid):
                    other_impact[op_valid] = np.maximum(
                        other_impact[op_valid],
                        a[idx_valid[op_valid]] * a[op_partner[op_valid]]
                    )

        top_indices = idx_valid[np.argsort(score)[::-1][:top_k]]

        improved = False
        for pos in top_indices:
            if a[pos] < 1e-12:

            # Try several reduction factors with line search

            for factor in [0.999, 0.998, 0.995, 0.99, 0.985, 0.98]:

                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                trial_score = 2.0 * n * fast_convolve(a_trial).max()
                if trial_score < best_trial_score:
                    best_trial_score = trial_score
                    best_factor = factor
                    best_trial = a_trial.copy()

                a = best_trial

                break

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    choices.append(a)

    result = None
    best_sc = float('inf')
    for a in choices:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = evaluate_local(a)
        if sc < best_sc:
            best_sc = sc
            result = a.copy()

    if result is None:
        result = np.ones(n) / n
    return result


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining mirror descent, spectral flattening,
    contribution-aware coordinate descent, and multi-scale refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 1200
        best_seq = diverse_initialization(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}] (t={elapsed:.0f}s): {sc:.8f} (n={len(a)})")

    # === Phase 1: Mirror Descent from parent (40% of time) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 1: Mirror Descent (score={best_score:.8f}, n={len(best_seq)}, time={rem:.0f}s) ===")
    t_alloc = rem * 0.35
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_fast(best_seq, t_alloc, eta0=0.07,
                                      active_frac=0.999, nesterov=True,
                                      adapt_eta=True, restart_every=500)
        try_update(a_md, "mirror_descent")

    # === Phase 2: Spectral Flattening (15% of time) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 2: Spectral Flattening (score={best_score:.8f}, time={rem:.0f}s) ===")
    t_alloc = rem * 0.15
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        for alpha in [0.7, 0.5, 0.3]:
            rem2 = deadline - time.monotonic()
            if rem2 < 10:
                break
            a_sf, _ = spectral_flatten(best_seq, rem2 * 0.4, alpha=alpha)
            try_update(a_sf, f"spectral_a{alpha}")

    # === Phase 3: Contribution Coordinate Descent (20% of time) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 3: Coord Descent (score={best_score:.8f}, time={rem:.0f}s) ===")
    t_alloc = rem * 0.25
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = contribution_coord_descent(best_seq, t_alloc, n_steps=100000, top_k=10)
        try_update(a_cd, "coord_desc")

    # === Phase 4: Try different n values (15% of time) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-length search (score={best_score:.8f}, time={rem:.0f}s) ===")
    t_per_n = rem * 0.12
    if t_per_n > 15:
        n_best = len(best_seq)
        n_candidates = [int(n_best * f) for f in [0.7, 0.8, 0.9, 1.1, 1.25, 1.5]]
        n_candidates = sorted(set(max(200, min(5000, nn)) for nn in n_candidates))
        for nn in n_candidates:
            rem2 = deadline - time.monotonic()
            if rem2 < 10:
                break
            if nn == n_best:
                continue
            print(f"  Trying n={nn}...")
            a_interp = interpolate_sequence(best_seq, nn)
            # Quick mirror descent
            a_opt, _ = mirror_descent_fast(a_interp, min(t_per_n, rem2 - 5),
                                           eta0=0.05, active_frac=0.999,
                                           nesterov=True, adapt_eta=True,
                                           restart_every=400)
            try_update(a_opt, f"n={nn}")
            # Quick coord descent
            rem3 = deadline - time.monotonic()
            if rem3 > 15:
                a_cd2, _ = contribution_coord_descent(a_opt, min(t_per_n * 0.5, rem3 - 5),
                                                      n_steps=50000, top_k=8)
                try_update(a_cd2, f"n={nn}_cd")

    # === Phase 5: Gauss-Seidel peak reduction (10% of time) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 5: GS Peak Reduction (score={best_score:.8f}, time={rem:.0f}s) ===")
    t_alloc = rem * 0.12
    if t_alloc > 10:
        a_gs, _ = gauss_seidel_peak_reduction(best_seq, t_alloc, n_iters=20000)
        try_update(a_gs, "gauss_seidel")

    # === Phase 6: Final refinement mirror descent (10% of time) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Refinement (score={best_score:.8f}, time={rem:.0f}s) ===")
    t_alloc = rem * 0.15
    if t_alloc > 10:
        a_md2, _ = mirror_descent_fast(best_seq, t_alloc, eta0=0.01,
                                       active_frac=0.9999, nesterov=True,
                                       adapt_eta=True, restart_every=800)
        try_update(a_md2, "refine_md")

    # === Phase 7: Final coordinate descent polish ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}, time={rem:.0f}s) ===")
    t_alloc = rem * 0.2
    if t_alloc > 5:
        a_polish, _ = contribution_coord_descent(best_seq, t_alloc, n_steps=100000, top_k=12)
        try_update(a_polish, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.1f}s)")

    if best_seq is None:
        best_seq = np.ones(1000) / 1000

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
