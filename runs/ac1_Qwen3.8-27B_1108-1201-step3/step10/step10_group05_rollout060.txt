    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def coord_descent_enhanced(a0, time_limit, verbose=False, label=""):
    """
    Enhanced coordinate descent with CORRECT coefficients (2*a[j]).
    Uses binary search for optimal delta when the transition point is needed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    # Precompute coefficient base: coeff_full = 2*a (will be updated)
    # For coordinate i, affected positions are k=i..min(i+n-1, 2n-2)
    # Change at position k=i+j is: delta * 2*a[j] + delta^2 * I(j==i)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        # Randomize coordinate order every few sweeps to escape local optima
        if sweeps % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            n_affected = k_end - k_start + 1

            # Unaffected max
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            base = max(unaff_left, unaff_right)

            # Affected values
            affected = conv[k_start:k_end + 1].copy()
            # Coefficients: 2*a[0], 2*a[1], ..., 2*a[n_affected-1]
            # (since k-i ranges from 0 to n_affected-1)
            coeffs = 2.0 * a[:n_affected]
            # Quadratic term at position j=i (if i < n_affected)
            quad_pos = -1
            if i < n_affected:
                quad_pos = i

            current_max_aff = float(affected.max())
            if current_max_aff <= base + 1e-15:
                continue  # Already at base level, can't improve

            # Find optimal delta via binary search on [−ai, 0]
            # We want to minimize max(base, max_j(affected[j] + delta*coeffs[j] + delta^2*I(j==quad_pos)))
            # For delta < 0, the affected values generally decrease

            def obj_at_delta(d):
                """Compute max of affected at given delta."""
                vals = affected + d * coeffs
                if quad_pos >= 0:
                    vals = vals.copy()
                    vals[quad_pos] += d * d
                return float(vals.max())

            # Binary search for the delta where max_affected = base
            # or the best delta in [-ai, 0]
            lo = -ai
            hi = 0.0
            best_delta = 0.0
            best_obj = M

            # Evaluate at a few key points first
            candidates = [0.0, -ai, -0.5*ai, -0.1*ai, -0.01*ai]

            # Add transition candidates: where each affected line hits base
            for j in range(n_affected):
                if coeffs[j] > 1e-15:
                    d = (base - affected[j]) / coeffs[j]
                    if -ai <= d <= 0:
                        candidates.append(d)

            # Add pairwise intersection of top affected values
            top_k = min(8, n_affected)
            top_idx = np.argsort(affected)[-top_k:]
            for jj in range(len(top_idx)):
                for kk in range(jj + 1, len(top_idx)):
                    i1, i2 = top_idx[jj], top_idx[kk]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai <= d <= 0:
                            candidates.append(d)

            # Evaluate all candidates
            for d in candidates:
                d = max(d, -ai)
                if d > 0:
                    d = 0.0
                obj = max(base, obj_at_delta(d))
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_delta = d

            # If best_delta is not 0, refine with binary search
            if best_delta != 0.0:
                # Binary search for the optimal point
                # The objective max(base, max_affected(d)) is non-increasing in d (for d<0)
                # Actually it's: for d going from 0 to -ai, affected values decrease
                # So max_affected(d) is non-increasing for d in [-ai, 0] (going from 0 toward -ai)
                # We want the most negative d such that max_affected(d) >= base
                # i.e., find where max_affected(d) crosses base

                # Let's do a binary search: we want the largest |d| (most negative d)
                # such that max_affected(d) >= base
                # Actually we want the d that minimizes max(base, max_affected(d))
                # Since max_affected is non-increasing as d goes more negative,
                # the optimum is where max_affected = base (if reachable) or d = -ai

                lo2 = best_delta - 1e-10
                hi2 = min(best_delta + 1e-10, 0.0)
                if lo2 < -ai:
                    lo2 = -ai

                # Check if we can go more negative
                if obj_at_delta(-ai) > base:
                    # Even at d=-ai, affected max > base, so optimal is d=-ai
                    best_delta = -ai
                    best_obj = max(base, obj_at_delta(-ai))
                else:
                    # Find where max_affected = base
                    lo3 = best_delta  # at this point obj < M (improvement found)
                    hi3 = 0.0

                    # Actually, let's just do a fine binary search
                    # We know the optimum is in [lo, 0] and the function is non-increasing
                    # (more negative d → lower max_affected, until it hits base)
                    # The optimum is at the point where max_affected = base

                    # Binary search for d where max_affected(d) = base
                    d_lo = -ai
                    d_hi = 0.0
                    # max_affected(d_lo) <= base, max_affected(d_hi) >= base (usually)
                    if obj_at_delta(d_lo) >= base:
                        best_delta = d_lo
                    else:
                        for _ in range(40):
                            d_mid = (d_lo + d_hi) / 2
                            if obj_at_delta(d_mid) >= base:
                                d_lo = d_mid
                            else:
                                d_hi = d_mid
                        best_delta = d_lo
                        best_obj = base

            # Apply best delta
            if best_delta != 0.0 and best_obj < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                # Update convolution
                delta = best_delta
                conv[k_start:k_end + 1] += delta * coeffs
                if quad_pos >= 0:
                    conv[2 * i] += delta * delta
                M = max(base, float(conv[k_start:k_end+1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Periodic recompute to avoid drift
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    # Final recompute
    conv_final = fast_convolve(best_a)
    M_final = float(conv_final.max())
    score_final = 2.0 * n * M_final
    if score_final < best_score:
        best_score = score_final

    return best_a, best_score, sweeps

def fine_polish(a0, time_limit, label=""):
    """Very fine polishing with tight binary search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            n_affected = k_end - k_start + 1

            if k_start > 0:
                base = float(conv[:k_start].max())
            else:
                base = -1e30
            if k_end < 2 * n - 2:
                base = max(base, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = 2.0 * a[:n_affected]
            quad_pos = i if i < n_affected else -1

            current_max_aff = float(affected.max())
            if current_max_aff <= base + 1e-16:
                continue

            # Try very small perturbations around current value
            # Also try the binary-search optimal
            best_delta = 0.0
            best_new_M = M

            def eval_delta(d):
                vals = affected.copy()
                vals += d * coeffs
                if quad_pos >= 0:
                    vals[quad_pos] += d * d
                return max(base, float(vals.max()))

            # Binary search for optimal negative delta
            if current_max_aff > base:
                # Find where max_affected = base
                d_lo = -ai
                if eval_delta(d_lo) > base:
                    best_delta = -ai
                    best_new_M = eval_delta(-ai)
                else:
                    d_hi = 0.0
                    for _ in range(50):
                        d_mid = (d_lo + d_hi) / 2
                        if eval_delta(d_mid) >= base:
                            d_lo = d_mid
                        else:
                            d_hi = d_mid
                    best_delta = d_lo
                    best_new_M = base

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                delta = best_delta
                a[i] = max(ai + delta, 0.0)
                conv[k_start:k_end + 1] += delta * coeffs
                if quad_pos >= 0:
                    conv[2 * i] += delta * delta
                M = best_new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

        if sweeps % 10 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

    conv_final = fast_convolve(best_a)
    best_score = 2.0 * len(best_a) * float(conv_final.max())
    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        alpha = min(1.0, 8.0 / (iters + 3))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-20:
        return a.copy()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def spectral_perturb(a0, rng, strength=0.1):
    """Perturb in the frequency domain to create diverse starting points."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a = a / s

    fft_a = rfft(a)
    # Add random phase perturbation
    phase_noise = rng.standard_normal(len(fft_a)) * strength
    fft_new = fft_a * np.exp(1j * phase_noise)
    a_new = irfft(fft_new, 2 * n - 1)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_subgrad(a0, time_limit, label=""):
    """Subgradient with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M - 0.02 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        a_new = a - step * grad
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

        a6, s6, _ = coord_descent_enhanced(init_alt, t6, label=f"p6_n{n_alt}")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_n{n_alt}")

    # === Phase 7: Nesterov subgradient ===
    print(f"\n=== Phase 7: Nesterov Subgrad (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    t7 = min(max(rem * 0.3, 20), 80.0)
    print(f"  Budget: {t7:.0f}s")
    a7, s7 = nesterov_subgrad(best_seq, t7, label="p7")
    best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_subgrad")

    # === Phase 8: CD + LP + CD cycle ===
    print(f"\n=== Phase 8: CD+LP+CD cycle (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 40:
        t8a = rem * 0.3
        t8b = rem * 0.2
        t8c = rem * 0.3
        a8a, _, _ = coord_descent_enhanced(best_seq, t8a, label="p8a")
        best_seq, best_score = check_and_update(a8a, best_seq, best_score, "p8a")
        a8b, _, _ = lp_fictitious_play(best_seq, t8b, solver=solver, label="p8b")
        best_seq, best_score = check_and_update(a8b, best_seq, best_score, "p8b")
        a8c, _, _ = coord_descent_enhanced(best_seq, t8c, label="p8c")
        best_seq, best_score = check_and_update(a8c, best_seq, best_score, "p8c")

    # === Phase 9: Final intensive polishing ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t9 = rem * 0.7
        print(f"  Fine polish budget: {t9:.0f}s")
        a9, s9 = fine_polish(best_seq, t9, label="p9")
        best_seq, best_score = check_and_update(a9, best_seq, best_score, "p9_fine")

    # Final CD pass
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a10, _, _ = coord_descent_enhanced(best_seq, rem * 0.8, label="p10")
        best_seq, best_score = check_and_update(a10, best_seq, best_score, "p10_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
