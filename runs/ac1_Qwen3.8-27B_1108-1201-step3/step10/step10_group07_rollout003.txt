import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def mwu_optimization(a0, time_limit, verbose=True):
    """
    Multiplicative Weight Updates for min max_k (a*a)[k] s.t. a>=0, sum(a)=1.
    Key idea: at each step, find peaks and multiplicatively reduce
    coordinates that contribute to those peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    eta = 0.15  # initial step size
    K = 10  # number of peaks to consider
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top K peaks
        if K >= 2 * n - 1:
            peak_idx = np.arange(2 * n - 1)
            peak_val = conv.copy()
        else:
            peak_idx = np.argpartition(conv, -K)[-K:]
            peak_val = conv[peak_idx]

        # Adaptive temperature: soft weighting over peaks
        M_max = float(peak_val.max())
        M_min = float(peak_val.min())
        temp = max(0.01 * M_max, 1e-10)
        w = np.exp(-(peak_val - M_max) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            w = w / ws

        # Compute subgradient: d[j] = sum_k w[k] * a[k-j] (valid indices only)
        d = np.zeros(n)
        for idx in range(len(peak_idx)):
            if w[idx] < 1e-14:
                continue
            k = int(peak_idx[idx])
            # a[k-j] is valid when 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            d[j_start:j_end + 1] += w[idx] * a[k - j_end:k - j_start + 1][::-1]

        # Normalize subgradient
        d_max = float(d.max())
        if d_max < 1e-15:
            d_max = 1.0

        # Adaptive step size: reduce eta when M is small (converging)
        eta_eff = eta * max(M / (M_max + 1e-15), 0.3)

        # Multiplicative update
        a_new = a * np.exp(-eta_eff * d / d_max)
        a_new = np.maximum(a_new, 1e-20)  # prevent underflow

        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        conv = fast_convolve(a_new)
        new_M = float(conv.max())
        new_score = 2.0 * n * new_M

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                eta *= 0.85
                no_improve = 0
                if eta < 0.001:
                    # Try with smaller K
                    K = max(3, K // 2)
                    no_improve = 0
                    eta = 0.05

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU it={it} ({elapsed:.0f}s): best={best_score:.8f} M={M:.6f} eta={eta:.4f}")

    return best_a, best_score


def spectral_init(n, rng):
    """
    Generate a sequence with a relatively flat power spectrum.
    Uses iterative spectral flattening.
    """
    a = np.ones(n) / n
    for _ in range(50):
        A = rfft(a, 2 * n)
        # Flatten magnitude: target is 1/sqrt(|A|^2)
        mag = np.abs(A)
        mag[mag < 1e-10] = 1e-10
        target_mag = 1.0 / np.sqrt(mag)
        target_mag = target_mag / target_mag.max()
        # New phase-preserving spectrum
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    return a


def diverse_init(n, rng, init_type=0):

def adaptive_coord_descent(a0, time_limit, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_left = conv[:k_start].max() if k_start > 0 else -1e30
            unaff_right = conv[k_end + 1:].max() if k_end < 2 * n - 2 else -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            base = max(a[i], 1e-8)
            factors = [-0.5, -0.3, -0.1, -0.03, 0.03, 0.1, 0.3, 0.5]

            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lp_alternating(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 10:
            a = b_new.copy()
        elif it <= 30:
            alpha = 0.6
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            alpha = max(0.2, 8.0 / (it + 5.0))
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 5e-5):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
