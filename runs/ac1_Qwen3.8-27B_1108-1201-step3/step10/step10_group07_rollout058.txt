def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

        return a.copy(), float('inf')

    if s < 1e-15:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

            scale1 = max(a[i] * 0.3, 1e-5 / n)
            scale2 = max(a[i + 1] * 0.3, 1e-5 / n)

            for f1 in [-1.0, -0.5, -0.2, 0.0, 0.2, 0.5, 1.0]:
                d1 = f1 * scale1
                if d1 < -a[i]:
                    continue
                for f2 in [-1.0, -0.5, -0.2, 0.0, 0.2, 0.5, 1.0]:
                    d2 = f2 * scale2
                    if d2 < -a[i + 1]:
                        continue

                    new_aff = conv_aff + d1 * d1_coeff + d2 * d2_coeff
                    # Add quadratic and cross terms
                    new_aff = new_aff.copy()
                    if quad1_idx >= 0:
                        new_aff[quad1_idx] += d1 * d1
                    if quad2_idx >= 0:
                        new_aff[quad2_idx] += d2 * d2
                    # Cross term at k = 2i+1
                    cross_k = 2 * i + 1
                    if k_start <= cross_k <= k_end:
                        new_aff[cross_k - k_start] += 2.0 * d1 * d2

                    new_M = max(unaff_max, float(new_aff.max()))
                    if new_M < best_new_M - 1e-14:
                        best_new_M = new_M
                        best_d1 = d1
                        best_d2 = d2

            if (abs(best_d1) > 1e-16 or abs(best_d2) > 1e-16) and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_d1, 0.0)
                a[i + 1] = max(a[i + 1] + best_d2, 0.0)

                new_aff = conv_aff + best_d1 * d1_coeff + best_d2 * d2_coeff
                new_aff = new_aff.copy()
                if quad1_idx >= 0:
                    new_aff[quad1_idx] += best_d1 * best_d1
                if quad2_idx >= 0:
                    new_aff[quad2_idx] += best_d2 * best_d2
                cross_k = 2 * i + 1
                if k_start <= cross_k <= k_end:
                    new_aff[cross_k - k_start] += 2.0 * best_d1 * best_d2

                conv[k_start:k_end + 1] = new_aff
                M = best_new_M
                improved = True
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        if verbose and rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD[{label}] round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_descent(a0, time_limit, lr_init=0.01, top_k=30, verbose=True):
    """Subgradient descent with multi-peak softmax weighting and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    total_convs = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-12:
                break

        # Compute subgradient with softmax weighting over top peaks
        k_act = min(top_k, total_convs)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a = a - lr * sg
        a = simplex_project(a)
        conv = fast_convolve(a)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization with adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Evaluate b_new
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Adaptive damping
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.7
        else:
            alpha = max(0.3, 15.0 / (it + 10.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = b_new.copy()
        else:
            a = a / a_sum

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 1e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            rng = np.random.default_rng(it)
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps > 1e-15:
                a_pert = a_pert / ps
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.002:
                a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if init_type == 0:
        # Triangle
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        # U-shape
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 2:
        # Cosine window (raised cosine)
        x = (t - center) / center
        power = rng.uniform(0.5, 2.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1))), 1e-10) ** power
    elif init_type == 3:
        # Two humps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Flat-top with smooth edges
        edge = n * 0.2
        a = np.ones(n)
        a[:int(edge)] = np.linspace(0.3, 1.0, int(edge))
        a[-int(edge):] = np.linspace(1.0, 0.3, int(edge))
    elif init_type == 5:
        # Super-Gaussian
        x = (t - center) / center
        p = rng.uniform(3.0, 10.0)
        a = np.exp(-np.abs(x) ** p)
    elif init_type == 6:
        # Random smooth
        n_freq = rng.integers(3, 15)
        freqs = rng.uniform(1, 20, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = rng.uniform(0.1, 1.0, size=n_freq)
        a = 1.0 + 0.3 * sum(amps[i] * np.cos(2 * np.pi * freqs[i] * t / n + phases[i]) for i in range(n_freq))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def fine_polish(a0, time_limit, verbose=True):
    """Very fine multiplicative coordinate descent for last-mile improvements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    total_convs = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, total_convs - 1)

            if k_start > 0:
                unaff_max = max(conv[:k_start].max(), conv[k_end + 1:].max()) if k_end < total_convs - 1 else conv[:k_start].max()
            else:
                unaff_max = conv[k_end + 1:].max() if k_end < total_convs - 1 else -1e30

            a_vec = a[:k_end - k_start + 1]
            conv_aff = conv[k_start:k_end + 1]
            quad_idx = 2 * i - k_start if 2 * i >= k_start and 2 * i <= k_end else -1

            best_delta = 0.0
            best_new_M = M

            # Very fine multiplicative steps
            for factor in [0.997, 0.9985, 0.9995, 1.0005, 1.0015, 1.003]:
                delta = ai * (factor - 1.0)
                if abs(delta) < 1e-18:
                    continue

                new_aff = conv_aff + 2.0 * delta * a_vec
                if quad_idx >= 0:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] = conv_aff[quad_idx] + 2.0 * delta * ai + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                changes = 2.0 * best_delta * a_vec
                changes = changes.copy()
                if quad_idx >= 0:
                    changes[quad_idx] = 2.0 * best_delta * ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = best_new_M
                improved = True
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: LP alternating -> Exact CD -> Pair CD -> Subgradient -> Fine polish.
    Uses diverse initializations and the provided parent sequence.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(42)

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.10f}")

    print(f"Initial best: {best_score:.10f} (n={n})")

    # Find LP solver
    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return False
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(candidate)})")
            return True
        return False

    # === Try diverse initializations ===
    print("\n=== Trying diverse initializations ===")
    for idx in range(8):
        init = diverse_init(n, rng, idx)
        sc = score_of(init)
        if sc < best_score:
            print(f"  Init {idx}: {sc:.10f} (improvement!)")
            if try_update(init, f"init{idx}"):
                pass
        else:
            print(f"  Init {idx}: {sc:.8f}")

    # === Phase 1: LP Alternating (45% time) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.10f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.45

    # LP from current best
    t_alloc = min(t1_end - time.monotonic(), 350.0)
    print(f"  LP from best, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_best")

    # LP from best cosine init
    if time.monotonic() < t1_end - 60:
        init_cos = diverse_init(n, rng, 2)
        t_alloc = min(t1_end - time.monotonic(), 120.0)
        print(f"  LP from cosine init, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating(init_cos, t_alloc, solver, verbose=False)
        try_update(a_lp2, "lp_cosine")

    # Second LP round on best
    if time.monotonic() < t1_end - 40:
        t_alloc = min(t1_end - time.monotonic(), 150.0)
        print(f"  LP round 2, budget={t_alloc:.0f}s")
        a_lp3, s_lp3 = lp_alternating(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp3, "lp_r2")

    # === Phase 2: Exact Coordinate Descent (30% time) ===
    print(f"\n=== Phase 2: Exact CD (best={best_score:.10f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.30

    t_alloc = min(t2_end - time.monotonic(), 400.0)
    if t_alloc > 30:
        print(f"  ExactCD budget={t_alloc:.0f}s")
        a_cd, s_cd = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

        # Also from LP result
        if time.monotonic() < t2_end - 60:
            t_alloc2 = min(t2_end - time.monotonic(), 150.0)
            a_cd2, s_cd2 = exact_coord_descent(a_lp, t_alloc2, verbose=False, label="lp")
            try_update(a_cd2, "cd_lp")

    # === Phase 3: Pair CD + Subgradient (15% time) ===
    print(f"\n=== Phase 3: PairCD + SGD (best={best_score:.10f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    # Pair coordinate descent
    if time.monotonic() < t3_end - 30:
        t_pcd = min(t3_end - time.monotonic() - 30, 100.0)
        if t_pcd > 20:
            print(f"  PairCD budget={t_pcd:.0f}s")
            a_pcd, s_pcd = pair_coord_descent(best_seq, t_pcd, verbose=True, label="main")
            try_update(a_pcd, "pair_cd")

    # Subgradient
    if time.monotonic() < t3_end - 10:
        t_sgd = min(t3_end - time.monotonic() - 10, 80.0)
        if t_sgd > 15:
            print(f"  SGD budget={t_sgd:.0f}s")
            a_sgd, s_sgd = subgradient_descent(best_seq, t_sgd, lr_init=0.008, top_k=40, verbose=True)
            try_update(a_sgd, "sgd")

    # === Phase 4: Exact CD again + Fine Polish (10% time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.10f}) ===")
    t4_end = deadline - 30

    # Another exact CD pass
    if time.monotonic() < t4_end - 60:
        t_cd = min(t4_end - time.monotonic() - 60, 120.0)
        if t_cd > 20:
            print(f"  ExactCD final, budget={t_cd:.0f}s")
            a_cd3, s_cd3 = exact_coord_descent(best_seq, t_cd, verbose=True, label="final")
            try_update(a_cd3, "cd_final")

    # Fine polish
    if time.monotonic() < t4_end - 20:
        t_fp = min(t4_end - time.monotonic() - 20, 80.0)
        if t_fp > 15:
            print(f"  FinePolish budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine")

    # One more exact CD
    if time.monotonic() < t4_end - 10:
        t_final = min(t4_end - time.monotonic() - 10, 40.0)
        if t_final > 10:
            a_fin, s_fin = exact_coord_descent(best_seq, t_final, verbose=False, label="last")
            try_update(a_fin, "last_cd")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
