"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LP alternation with
dual-guided peak squeezing, Nesterov acceleration, and targeted coordinate
descent. Key innovations: LP dual variable extraction for active peak
identification, directed mass redistribution away from binding peaks, and
multi-timescale refinement with adaptive step sizing.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_with_dual(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), solve LP for best response b.
    Also extract dual variables (lambda) for active peak identification.
    Returns (b, M, lambda) or (None, None, None)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    lambda_var = cp.Variable(size, nonneg=True)

    # Primal:
    # min M
    # s.t. T @ b <= M (size constraints)
    #      sum(b) = 1
    #      b >= 0, M >= 0

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1, M_var >= 0]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    # Also set up dual to extract lambda
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None, None

        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None, None
        b_val = b_val / s
        M_val = float(M_var.value)
    except Exception:
        return None, None, None

    # Extract dual variables from the primal constraints
    # The dual of T@b <= M gives us lambda
    lambda_dual = None
    try:

        # In cvxpy, inequalities are of the form expr <= 0 in the internal form
        # T@b - M <= 0, so dual values correspond to these
        for con in constraints:

                dv = con._dual_value
                if dv is not None:

                    if dv.shape[0] == size or (hasattr(dv, '__len__') and len(dv) == size):
                        lambda_dual = np.maximum(dv, 0)
                        break
    except Exception:
        pass

        conv_ab = T @ b_val
        max_val = conv_ab.max()
        threshold = max_val * 0.999

        ls = lambda_dual.sum()
        if ls > 0:
            lambda_dual /= ls
        else:
            lambda_dual = np.ones(size) / size

    return b_val, M_val, lambda_dual


def nesterov_lp_with_dual(a0, time_limit, solver=None, max_iters=5000,
                           stall_patience=15, top_k_for_squeeze=5):
    """Nesterov LP alternation with dual-guided peak squeezing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()  # Nesterov momentum variable
    x_avg = a.copy()  # Iterated average
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, lam = solve_lp_with_dual(y, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal iterated average (heavy-ball style)
        a = ((it + 1) * x_avg + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
            x_avg = a.copy()

        # Dual-guided peak squeezing
        if (it + 1) % 10 == 0 and lam is not None:
            b_curr = fast_convolve(a)
            max_b_curr = b_curr.max()
            # Identify top contributing peaks using dual weights
            effective = lam * b_curr
            top_peaks = np.argsort(effective)[::-1][:top_k_for_squeeze]
            
            # For each top peak, find and reduce top contributors
            for pk in top_peaks[:3]:
                if a.sum() < 1e-15:
                    break
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                total_contrib = contribs.sum()
                if total_contrib < 1e-15:
                    continue
                
                # Find the top contributor
                top_order = np.argsort(contribs)[::-1][:min(3, len(idx))]
                for oi in top_order:
                    pos = idx[oi]
                    if a[pos] < 1e-12:
                        continue
                    # Compute "safety" of other positions (low contribution to all top peaks)
                    # Move mass from pos to a position with low contribution to all top peaks
                    pos_contribs = np.zeros(n)
                    for p2 in top_peaks[:3]:
                        js2 = max(0, p2 - n + 1)
                        je2 = min(n - 1, p2)
                        if js2 <= pos <= je2:
                            pos_contribs[pos] = 0  # it's at the peak
                    # Find safe positions: low contribution to all top peaks
                    for p2 in top_peaks[:3]:
                        js2 = max(0, p2 - n + 1)
                        je2 = min(n - 1, p2)
                        for jj in range(js2, je2 + 1):
                            pos_contribs[jj] += a[p2 - jj]
                    
                    # Safe positions: those with low total contribution AND away from pos
                    safe_mask = pos_contribs < 0.5 * pos_contribs.mean()
                    safe_positions = np.where(safe_mask)[0]
                    safe_positions = safe_positions[safe_positions != pos]
                    
                    if len(safe_positions) == 0:
                        safe_positions = np.arange(n)
                        safe_positions = safe_positions[safe_positions != pos]
                    
                    # Move a fraction of mass
                    frac = 0.05
                    amount = a[pos] * frac
                    a[pos] -= amount
                    # Distribute proportionally to existing values at safe positions
                    safe_vals = a[safe_positions].copy()
                    sv = safe_vals.sum()
                    if sv > 1e-15:
                        a[safe_positions] += amount * safe_vals / sv
                    else:
                        a[safe_positions] += amount / len(safe_positions)
                    
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    x_avg = a.copy()
                    y = a.copy()
                    break

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stall detection
        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.015 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP-dual iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_obj_and_grad(z, n, T_stage):
    """Compute LogSumExp-smoothed objective and gradient."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    lse = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T_stage) * lse

    # Gradient
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0 / T_stage
    S = np.dot(grad_a, a)
    grad = a * (grad_a - S)
    return obj, grad


def lbgfs_logsumexp(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing, multi-stage temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02]

    n_stages = len(T_stages)
    iters_per = max(20, maxiter_per_stage // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_loc(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad(z, n, T_stage)
            return obj

        def jac_loc(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, grad = logsumexp_obj_and_grad(z, n, T_stage)
            return grad

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_squeeze_cd(a0, time_limit, n_steps=100000, top_m=5):
    """Aggressive peak-directed coordinate descent with mass relocation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        for peak_k in top_peaks[:4]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:5]

            for oi in order:
                pos = idx[oi]
                partner = peak_k - pos
                if a[pos] < 1e-12:
                    continue
                
                # Try reducing pos and redistributing
                for factor in [0.995, 0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-16:
                        continue
                    
                    # Redistribute to minimize increase in max convolution
                    # Prefer positions that are far from current peak partners
                    # Compute "danger" for each position
                    danger = np.zeros(n)
                    for pk in top_peaks[:3]:
                        js2 = max(0, pk - n + 1)
                        je2 = min(n - 1, pk)
                        for jj in range(js2, je2 + 1):
                            danger[jj] += a[pk - jj]
                    
                    # Safe positions: low danger, not pos, not partner
                    safe = danger.copy()
                    safe[pos] = np.inf
                    if partner != pos:
                        safe[partner] = np.inf
                    
                    # Pick top safe positions by value (to maintain mass distribution)
                    safe_vals = a.copy()
                    safe_vals[safe == np.inf] = 0
                    # Scale by inverse danger
                    with np.errstate(divide='ignore', invalid='ignore'):
                        score_pos = safe_vals / (1e-10 + safe)
                    
                    top_safe = np.argsort(score_pos)[::-1][:min(20, n)]
                    top_safe = top_safe[top_safe != pos]
                    if len(top_safe) == 0:
                        top_safe = np.arange(n)
                        top_safe = top_safe[top_safe != pos]
                    
                    sv = a_trial[top_safe].sum()
                    if sv < 1e-15:
                        break
                    a_trial[top_safe] += diff * a_trial[top_safe] / sv
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Fallback: simultaneous small reduction
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.998
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSqueeze {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000, top_k=3):
    """Subgradient method targeting top-k active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: top-k peaks
        active = np.argsort(b)[::-1][:top_k]
        # Weight by proximity to max
        weights = b[active] / (max_b * top_k)

        # Weighted average gradient over active positions
        g = np.zeros(n)
        for wi, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[wi] * a[k - idx]
        g /= weights.sum()

        # Momentum step
        momentum = beta * momentum + (1 - beta) * g
        a_trial = a - lr * momentum
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            lr *= 0.9
            if lr < 1e-8:
                lr = 0.005
                momentum *= 0.5
            no_improve = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        # Two-peak structure
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    else:
        # Flat with small periodic modulation
        a = 1.0 + 0.1 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP alternation with dual guidance -> 
    LogSumExp L-BFGS -> Peak squeezing CD -> Subgradient -> 
    Multi-start diversity -> Final fine-tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Primary Nesterov LP with dual (longest) ===
    print(f"\n=== Phase 1: NLP-dual n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 40:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = nesterov_lp_with_dual(best_seq, t1, solver=solver,
                                             max_iters=30000, stall_patience=12)
        try_update(a_lp, "nlp_dual_p1")

    # === Phase 2: LogSumExp L-BFGS refinement ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.025, 0.015]
        print(f"  T_stages: {len(T_stages)}, budget={t2:.0f}s")
        a_lb = lbgfs_logsumexp(best_seq, t2, T_stages=T_stages, maxiter_per_stage=15000)
        try_update(a_lb, "lbgfs_p2")

    # === Phase 3: Peak squeezing coordinate descent ===
    print(f"\n=== Phase 3: Peak Squeeze CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 15:
        a_cd, sc_cd = peak_squeeze_cd(best_seq, t3, n_steps=150000, top_m=6)
        try_update(a_cd, "peak_squeeze")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 10:
        a_sg, sc_sg = subgradient_refine(best_seq, t4, n_steps=80000, top_k=3)
        try_update(a_sg, "subgrad")

    # === Phase 5: Second LP pass ===
    print(f"\n=== Phase 5: LP Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 180.0)
    if t5 > 30:
        a_lp2, sc_lp2 = nesterov_lp_with_dual(best_seq, t5, solver=solver,
                                               max_iters=20000, stall_patience=10)
        try_update(a_lp2, "nlp_p5")

    # === Phase 6: Multi-start with perturbations ===
    print(f"\n=== Phase 6: Multi-start, score={best_score:.8f} ===")

        n_cur = len(best_seq)
        x_space = np.linspace(0, 2 * np.pi, n_cur)
        perturbations = [
            0.02 * np.cos(3 * x_space) + 0.015 * np.sin(5 * x_space),

        ]
        t_each = min(rem / 4.0, 50.0)

            if time.monotonic() > deadline - 30:

            if t_each < 15:

            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            # Use peak squeeze for perturbed starts (faster)

            try_update(a_res, f"perturb_{pi}")

    # === Phase 7: Alternative length exploration ===
    print(f"\n=== Phase 7: Alt length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        alt_n = int(n_cur * 1.3)

            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)

            rem = deadline - time.monotonic()
            if rem > 20:
                a_alt2, sc_alt2 = peak_squeeze_cd(best_seq, rem * 0.3, n_steps=30000, top_m=5)
                try_update(a_alt2, f"squeeze_alt")

    # === Phase 8: Final fine-tuning polish ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 40.0)
        if t_fin > 8:
            a_fin = lbgfs_logsumexp(best_seq, t_fin,
                                     T_stages=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005],
                                     maxiter_per_stage=8000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, sc_cd2 = peak_squeeze_cd(best_seq, min(rem * 0.5, 30.0),
                                         n_steps=50000, top_m=6)
        try_update(a_cd2, "final_squeeze")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg2, sc_sg2 = subgradient_refine(best_seq, min(rem * 0.6, 20.0),
                                            n_steps=30000, top_k=4)
        try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
