                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * eta_bb
                else:
                    eta = 0.9 * eta
                eta = max(5e-6, min(eta, 0.5))
            else:
                eta *= 0.995

        prev_a = a.copy()
        prev_g = g.copy()

        # Mirror descent update (exponential weighting)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        g_max = g.max() - g.min()
        if g_max < 1e-15:
            g_shifted = g * eta
        else:
            g_shifted = g_shifted / g_max * eta
        log_a_new = log_a - g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        # Nesterov acceleration
        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart with noise if stuck
        if no_improve >= 300 + 100 * (it // 50000):
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.003 + 0.002 * np.sqrt(min(no_improve / 500.0, 5.0))
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            else:
                a = best_a.copy()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.6)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.10f} "
                  f"eta={eta:.6f} |S|={len(active)} pct={pct:.1f}")

    return best_a, best_score


def fista_lse_optimize(a0, time_limit):
    """
    FISTA (Accelerated Gradient) on log-sum-exp smoothed objective.
    Temperature adapts based on peak-to-average ratio.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # FISTA parameters
    y = a.copy()
    tk = 1.0
    L = 2.0 * n * n  # Lipschitz constant estimate (rough)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute LSE objective and gradient with adaptive temperature
        b = fast_convolve(y)
        max_b = b.max()
        avg_b = b.mean()
        peak_ratio = max_b / (avg_b + 1e-30)

        # Adaptive temperature: higher when peak dominates, lower when flat
        T = max(0.01, min(5.0, 0.5 * (peak_ratio - 1.0) + 0.1))

        # LSE computation
        b_scaled = b / T
        b_scaled -= b_scaled.max()
        exp_b = np.exp(b_scaled)
        sum_exp = exp_b.sum()
        if sum_exp < 1e-300:
            w = np.ones(len(b)) / len(b)
        else:
            w = exp_b / sum_exp

        lse_val = T * (np.log(sum_exp) + b_scaled.max())

        # Gradient: d/da[i] LSE = (1/T) * sum_k w[k] * 2*a[k-i] (for valid k-i)
        grad_a = np.zeros(n)
        # Only iterate over significant weights for efficiency
        significant = np.where(w > 1e-8)[0]
        for k in significant:
            ki = int(k)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                grad_a[idx] += (2.0 / T) * w[ki] * a[ki - idx]

        # FISTA update with backtracking
        eta = 1.0 / L
        x_new = y - eta * grad_a
        x_new = np.maximum(x_new, 0)
        x_new_sum = x_new.sum()
        if x_new_sum < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / x_new_sum

        # Evaluate
        score = 2.0 * n * fast_convolve(x_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = x_new.copy()

        # FISTA momentum update
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        beta = (tk - 1.0) / tk_new
        tk = tk_new
        y = x_new + beta * (x_new - a)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        a = x_new

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-LSE iter={it+1} ({elapsed:.0f}s): best={best_score:.10f} T={T:.4f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None):
    """Nesterov LP alternating for refinement with adaptive blending."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    lp_count = 0

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        lp_count += 1

        # Nesterov-style update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Weighted average for a
        w_a = (it + 1.0)
        w_b = alpha * (it + 2.0)
        a = (w_a * a + w_b * b_new) / (w_a + w_b)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 10:
                noise_scale = 0.01 / np.sqrt(stall)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} LPs={lp_count}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_lse(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_inner = _softmax_lse(z_inner)
            b = fast_convolve(a_inner)
            b_s = b / T
            b_s -= b_s.max()
            return float(T * (np.log(np.exp(b_s).sum()) + b_s.max()))

        def jac_lse(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            a_inner = _softmax_lse(z_inner)
            b = fast_convolve(a_inner)
            b_s = b / T
            b_s -= b_s.max()
            exp_b = np.exp(b_s)
            sum_exp = exp_b.sum()
            w = exp_b / sum_exp

            grad_a = np.zeros(n)
            significant = np.where(w > 1e-10)[0]
            for k in significant:
                ki = int(k)
                i_lo = max(0, ki - n + 1)
                i_hi = min(n - 1, ki)
                if i_lo <= i_hi:
                    idx = np.arange(i_lo, i_hi + 1)
                    grad_a[idx] += (2.0 / T) * w[ki] * a_inner[ki - idx]

            S = np.dot(grad_a, a_inner)
            grad_z = a_inner * (grad_a - S)
            return grad_z

        try:
            res = minimize(fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-15, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        if (stage_i + 1) % 3 == 0:
            elapsed = time.monotonic() - t0
            a_curr = _softmax_lse(z)
            sc = 2.0 * n * fast_convolve(a_curr).max()
            print(f"    LSE-LBFGS stage={stage_i+1}/{n_stages} ({elapsed:.0f}s): T={T} score={sc:.10f}")

    return _softmax_lse(z)


def _softmax_lse(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def coordinate_descent_refine(a0, time_limit, n_steps=100000):
    """Coordinate descent with pairwise adjustments targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, int(peak_k) - n + 1)
            je = min(n - 1, int(peak_k))
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[int(peak_k) - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.997, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            # Try pairwise: reduce one contributor, increase another
            if not improved and len(top_positions) >= 2:
                p1 = top_positions[0]
                p2 = top_positions[-1]
                if a[p1] > 1e-10 and a[p2] > 1e-15:
                    for frac in [0.01, 0.02, 0.05]:
                        a_trial = a.copy()
                        transfer = a[p1] * frac
                        a_trial[p1] -= transfer
                        a_trial[p2] += transfer
                        if a_trial[p1] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            stall += 1
            if stall > 500:
                # Try random pairs
                rng = np.random.default_rng(step * 17 + 3)
                for _ in range(10):
                    p1, p2 = rng.integers(0, n, 2)
                    if p1 == p2 or a[p1] < 1e-12:
                        continue
                    transfer = a[p1] * 0.05
                    a_trial = a.copy()
                    a_trial[p1] -= transfer
                    a_trial[p2] += transfer
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if not improved and stall > 2000:
                    break
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.10f} stall={stall}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.15)
    else:
        a = (1 - x) * (0.3 + 0.7 * np.sin(np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive mirror descent, FISTA-LSE, LP alternating, L-BFGS,
    and coordinate descent in a carefully timed sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 1: Adaptive Mirror Descent (score={best_score:.10f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc,
                                          eta0=0.05, nesterov=True, bb_adapt=True)
        try_update(a_md, "mirror_descent_p1")

    # === Phase 2: FISTA on LSE ===
    print(f"\n=== Phase 2: FISTA-LSE (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fista, _ = fista_lse_optimize(best_seq, t_alloc)
        try_update(a_fista, "fista_lse")

    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: Nesterov LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "nesterov_lp")

    # === Phase 4: L-BFGS on LSE ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 225.0)
    if t_alloc > 15:
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                               maxiter_per_stage=5000)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Second Mirror Descent (finer) ===
    print(f"\n=== Phase 5: Fine Mirror Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_md2, _ = mirror_descent_optimize(best_seq, t_alloc,
                                           eta0=0.015, nesterov=True, bb_adapt=True)
        try_update(a_md2, "mirror_descent_p2")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Final LP + Polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.5, 60.0)
    t_cd = rem - t_lp - 5
    if t_lp > 10:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver)
        try_update(a_lp2, "final_lp")
    if t_cd > 5:
        a_cd2, _ = coordinate_descent_refine(best_seq, t_cd, n_steps=50000)
        try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
