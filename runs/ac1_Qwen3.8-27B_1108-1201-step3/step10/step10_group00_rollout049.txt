Let me also think about a different approach: **iterative peak reduction**. 

Algorithm:
1. Start with some initial sequence
2. Compute conv(a,a)
3. Find the peak location k*
4. Reduce the "contribution" to that peak by adjusting a
5. Repeat

The contribution to `(conv(a,a))[k]` from a specific coordinate `a_j` is `a_j * a_{k-j}` (if k-j is valid). So to reduce the peak at k*, we want to reduce the products `a_j * a_{k*-j}`.

This is like a multiplicative weight update: for each j, the "cost" of increasing a_j is related to how much it contributes to the current peaks.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. log-space z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z, n, T, deadline):

    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lp_obj_grad(z, n, p, deadline):
    """Compute Lp-norm objective and gradient w.r.t. log-space z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    # Use shifted and scaled b for numerical stability

    # Normalize by max for stability

    lp_val = max_b * (np.sum(b_scaled ** p)) ** (1.0 / p)
    obj = 2.0 * n * lp_val

    # Gradient of (sum b_k^p)^(1/p) w.r.t. a
    # = (1/p) * (sum b_k^p)^(1/p - 1) * p * sum_k b_k^(p-1) * grad b_k / grad a
    # = (sum b_k^p)^(1/p - 1) * sum_k b_k^(p-1) * grad b_k / grad a

    grad_z = a * (grad_a - S)

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 25:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    elif choice == 7:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 8:
        a = np.sin(np.pi * x) ** 0.7
    else:
        a = 1 + 0.15 * np.sin(6 * np.pi * x)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def random_structured_init(n, rng):
    """Random structured initialization with sinusoidal mix."""
    a = np.ones(n) / n
    x = np.linspace(0, 2 * np.pi, n)
    n_comps = rng.integers(1, 5)
    for _ in range(n_comps):
        freq = rng.integers(1, 12)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.3)
        a += amp * np.cos(freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def full_pipeline(a0, n, time_budget, deadline, label_prefix=""):
    """Run the full optimization pipeline: adaptive LSE -> Lp -> subgradient -> fine LSE."""
    best = a0.copy()
    best_sc = evaluate_local(a0)

    def try_update(a, tag):
        nonlocal best, best_sc
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
            print(f"  {label_prefix}{tag}: {sc:.8f}")
        return a

    # Step 1: Adaptive LSE
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.4, time_budget * 0.4)
    if t1 > 10:
        a1, _ = adaptive_lse_refine(best, t1, deadline)
        try_update(a1, "adapt_lse")

    # Step 2: Lp bridge (p=4, 8, 16)
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, time_budget * 0.2)
    if t2 > 8:
        a2 = lbgfs_lp(best, t2, [4, 8, 16, 32], maxiter_per_stage=2000, deadline=deadline)
        try_update(a2, "lp_bridge")

    # Step 3: Multiplicative subgradient
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, time_budget * 0.15)
    if t3 > 5:
        a3, _ = multiplicative_subgradient(best, t3, n_steps=100000, deadline=deadline)
        try_update(a3, "mult_sg")

    # Step 4: Fine LSE polish
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, time_budget * 0.25)
    if t4 > 5:
        a4 = lbgfs_lse(best, t4, [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001],
                       maxiter_per_stage=4000, deadline=deadline)
        try_update(a4, "fine_lse")

    return best, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Pipeline: Adaptive LSE -> Lp-norm bridge -> MultiSubGrad -> Fine LSE.
    With multi-length exploration and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}, budget={float(budget_s)-20:.0f}s")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Main pipeline on parent ===
    print(f"\n=== Phase 1: Main Pipeline n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.35, 600.0)
    if t_main > 30:
        print(f"  Budget: {t_main:.0f}s")
        a_main, sc_main = full_pipeline(best_seq, n_best, t_main, deadline, "M1:")
        local_update(a_main, "main_pipeline")

    # === Phase 2: Second adaptive LSE (continue from best) ===
    print(f"\n=== Phase 2: Adaptive LSE Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        a_adapt2, _ = adaptive_lse_refine(best_seq, t2, deadline)
        local_update(a_adapt2, "adapt_lse_2")

    # === Phase 3: Lp bridge round 2 ===
    print(f"\n=== Phase 3: Lp Bridge Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 15:
        a_lp2 = lbgfs_lp(best_seq, t3, [16, 32, 64], maxiter_per_stage=2000, deadline=deadline)
        local_update(a_lp2, "lp_bridge_2")

    # === Phase 4: Try n=1500 ===
    print(f"\n=== Phase 4: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1500 = min(rem * 0.12, 200.0)
        print(f"  Budget: {t_1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = full_pipeline(init_1500, 1500, t_1500, deadline, "1500:")
        local_update(a_1500, "pipeline_n1500")

    # === Phase 5: Try n=2000 ===
    print(f"\n=== Phase 5: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_2000 = min(rem * 0.10, 150.0)
        print(f"  Budget: {t_2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, _ = full_pipeline(init_2000, 2000, t_2000, deadline, "2000:")
        local_update(a_2000, "pipeline_n2000")

    # === Phase 6: Perturbation escape + quick re-optimize ===
    print(f"\n=== Phase 6: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        strengths = [0.005, 0.01, 0.02, 0.03, 0.05]
        for trial in range(len(strengths)):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=strengths[trial])
            t_sub = min(rem_sub * 0.3, 40.0)
            a_opt, _ = adaptive_lse_refine(a_pert, t_sub, deadline)
            local_update(a_opt, f"pert_{strengths[trial]}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            a_init = random_structured_init(n_best, rng)
            t_sub = min(rem_sub * 0.25, 30.0)
            a_opt, _ = adaptive_lse_refine(a_init, t_sub, deadline)
            local_update(a_opt, f"restart_{trial}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n=== Phase 8: Final Ultra-Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        T_ufine = [1.0, 0.5, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
        t_ufine = min(rem * 0.35, 50.0)
        if t_ufine > 5:
            a_ufine = lbgfs_lse(best_seq, t_ufine, T_ufine, maxiter_per_stage=3000, deadline=deadline)
            local_update(a_ufine, "ultra_fine_lse")

    # === Phase 9: Final multiplicative subgradient ===
    rem = deadline - time.monotonic()
    if rem > 8:
        a_final_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.4, 15.0),
                                                    n_steps=80000, deadline=deadline)
        local_update(a_final_ms, "final_mult_sg")

    # === Phase 10: One more adaptive LSE ===
    rem = deadline - time.monotonic()
    if rem > 8:
        a_final_adapt, _ = adaptive_lse_refine(best_seq, min(rem * 0.5, 15.0), deadline)
        local_update(a_final_adapt, "final_adapt")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
