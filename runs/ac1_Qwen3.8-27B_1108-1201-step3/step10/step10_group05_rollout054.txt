So in all cases where the derivative is nonzero, it's `2*a_{k-i}`.

The subgradient of `f(a) = max_k (a*a)[k]` is:
`{sum_{k in S} lambda_k * g_k : lambda in simplex}`
where `g_k[i] = 2*a_{k-i}` (with the convention a_m = 0 if m out of range).

At the optimum, for all i with a_i > 0: `sum_{k in S} lambda_k * 2*a_{k-i} = constant`.

This gives us a system of equations to solve for the optimal lambda. If we can solve this, we get the exact gradient direction.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.GLOP, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

1. CORRECTED smart coordinate descent (fixes gradient coefficient bug: 2*a[k-i] not a[k-i]+a[i])
2. LP fictitious play with adaptive damping and dual certificates
3. Exact subgradient with active-set identification for polishing
4. Peak shaving and multi-length exploration
5. Alternating LP+CD cycles for escaping local optima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.GLOP, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def solve_dual_certificate(a, solver=None):
    """Find lower bound on M via dual: max c s.t. sum_k lambda_k * a[k-i] >= c for all i."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    lambda_var = cp.Variable(2 * n - 1, nonneg=True)
    c_var = cp.Variable()
    # Build matrix: D[i,k] = a[k-i] if 0<=k-i<n else 0
    D = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for k in range(n):
            D[i, i + k] = a[k]
    constraints = [
        D @ lambda_var >= c_var,
        cp.sum(lambda_var) == 1
    ]
    prob = cp.Problem(cp.Maximize(c_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return float(c_var.value)

def corrected_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with CORRECTED coefficient: 2*a[k-i] (not a[k-i]+a[i])."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 0.01:
        a = np.ones(n) / n
        S = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        S = a.sum()

        for i in range(n):
            ai = a[i]
            if ai < 1e-10:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            idx = np.arange(0, k_end - k_start + 1)
            coeffs = 2.0 * a[idx]  # CORRECT: 2*a[k-i]
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            cur_score = 2.0 * n * M / (S * S)

            # Generate candidates
            candidates = [0.0, -0.999 * ai, -0.5 * ai, -0.1 * ai,
                          0.05 * ai, 0.2 * ai, 0.5 * ai]

            # Where each affected line hits unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-12:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.2 < d < ai * 1.2:
                        candidates.append(d)

            # Top line crossings
            top_k = min(6, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-12:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.2 < d < ai * 1.2:
                            candidates.append(d)

            # Quadratic crossings
            if has_quad:
                for j in range(len(affected)):
                    if j == quad_idx:
                        continue
                    A2 = 1.0
                    B2 = coeffs[quad_idx] - coeffs[j]
                    C2 = affected[quad_idx] - affected[j]
                    disc = B2 * B2 - 4.0 * A2 * C2
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        for d in [(-B2 + sq) / 2.0, (-B2 - sq) / 2.0]:
                            if -ai * 1.2 < d < ai * 1.2:
                                candidates.append(d)

            best_delta = 0.0
            best_new_score = cur_score

            for delta in candidates:
                if delta < -ai - 1e-10:
                    continue
                delta = max(delta, -ai + 1e-12)
                if abs(delta) < 1e-16:
                    continue
                new_S = S + delta
                if new_S < 0.01:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                new_score = 2.0 * n * new_max / (new_S * new_S)

                if new_score < best_new_score - 1e-14:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                S = a.sum()
                # Update M
                new_aff_max = float(conv[k_start:k_end + 1].max())
                M = max(unaff_max, new_aff_max)
                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Periodic recompute
        if sweeps % 20 == 0:
            a = np.maximum(a, 0)
            S = a.sum()
            if S < 0.01:
                break
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M / (S * S)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CCD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f}")

    a = np.maximum(a, 0)
    S = a.sum()
    if S > 0.01:
        conv = fast_convolve(a)
        sc = 2.0 * n * float(conv.max()) / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, label=""):
    """Very fine polishing CD with correct coefficients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 0.01:
        return a0, 2.0 * n * float(fast_convolve(a0).max()) / (a0.sum()**2)

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        S = a.sum()

        for i in range(n):
            ai = a[i]
            if ai < 1e-10:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            idx = np.arange(0, k_end - k_start + 1)
            coeffs = 2.0 * a[idx]
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            cur_score = 2.0 * n * M / (S * S)

            best_delta = 0.0
            best_new_score = cur_score

            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                          1.001, 1.002, 1.003, 1.005, 1.007, 1.01, 1.015]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_S = S + delta
                if new_S < 0.01:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                new_score = 2.0 * n * new_max / (new_S * new_S)
                if new_score < best_new_score - 1e-15:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                S = a.sum()
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 10 == 0:
            a = np.maximum(a, 0)
            S = a.sum()
            if S < 0.01:
                break
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M / (S * S)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def subgradient_polish(a0, time_limit, label=""):
    """Subgradient method with correct gradient and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 0.01:
        return a0, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Compute subgradient at y (correct: 2*y[k-i])
        conv_y = fast_convolve(y)
        M = conv_y.max()
        # Use near-peak for stability
        threshold = M * 0.95
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                grad[i] += 2.0 * rho * y[k - i]

        # Gradient step
        a_new = a - step * grad
        a_new = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if no_improve > 200:
            break

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 0.01:
        return a0, float('inf'), 0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        alpha = min(1.0, 20.0 / (iters + 10))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * float(fast_convolve(b_new).max())
        if score_b < best_score - 1e-12:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")

    return best_a, best_score, iters

def peak_targeted_perturb(a0, rng, strength=0.05):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    threshold = M * 0.9
    peak_positions = np.where(conv >= threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            pressure[i] += a[i] * a[k - i]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 0.01:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 0.01:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization with corrected coordinate descent:
    1. LP fictitious play (main driver)
    2. Corrected CD polishing
    3. Subgradient + perturbation cycling
    4. Multi-length exploration
    5. Intensive final polish
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback init: n={n_init}, score={best_score:.6f}")

    # === Phase 1: LP Fictitious Play (main optimization) ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 600
    t1 = min(max(rem, 100), 500.0)
    print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
    a1, s1, it1 = lp_fictitious_play(best_seq, t1, solver=solver, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_lpfp")

    # === Phase 2: Corrected CD Polishing ===
    print(f"\n=== Phase 2: Corrected CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    t2 = min(max(rem, 60), 300.0)
    print(f"  Budget: {t2:.0f}s, n={len(best_seq)}")
    a2, s2, sw2 = corrected_coord_descent(best_seq, t2, verbose=True, label="p2")
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_ccd")

    # === Phase 3: Subgradient + Perturbation Cycle ===
    print(f"\n=== Phase 3: SubGrad + Perturb (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    t3 = min(max(rem, 60), 250.0)
    a3, s3 = subgradient_polish(best_seq, t3, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_subgrad")

    # === Phase 4: Alternating LP+CD with perturbation ===
    print(f"\n=== Phase 4: Alt LP+CD + Perturb (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for rnd in range(6):
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break

        # Perturb
        if rnd % 2 == 0:
            perturbed = peak_targeted_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
        else:
            # Random restructure
            a = best_seq.copy()
            frac = 0.05 + 0.05 * rnd
            zero_out = rng.random(n_best) < frac
            n_zeros = zero_out.sum()
            if n_zeros < n_best * 0.5:
                a[zero_out] = 0
                a = np.maximum(a * (1 + rng.standard_normal(n_best) * 0.1), 0)
                s = a.sum()
                if s > 0.01:
                    perturbed = a / s
                else:
                    perturbed = best_seq.copy()
            else:
                perturbed = best_seq.copy()

        # Quick LP on perturbed
        t4a = min(rem * 0.3, 80.0)
        a4, s4, _ = lp_fictitious_play(perturbed, t4a, solver=solver, label=f"p4_lp_r{rnd}")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, f"p4_lp_r{rnd}")

        # CD polish
        t4b = min(rem * 0.2, 60.0)
        a4b, s4b, _ = corrected_coord_descent(best_seq, t4b, label=f"p4_cd_r{rnd}")

    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = [n_best]
    for alt in [768, 1024, 1280, 1536, 2048]:
        if alt != n_best and 512 <= alt <= 2560:
            alt_lengths.append(alt)

    for n_alt in alt_lengths[:4]:
        rem = deadline - time.monotonic() - 100
        if rem < 40:
            break
        t5 = min(rem * 0.3, 80.0)
        if n_alt == n_best:
            init_alt = best_seq.copy()
        else:
            init_alt = interpolate_seq(best_seq, n_alt)
        a5, s5, _ = corrected_coord_descent(init_alt, t5, label=f"p5_n{n_alt}")
        best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_n{n_alt}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 30:
        t6a = min(rem * 0.4, 100.0)
        a6, s6, _ = corrected_coord_descent(best_seq, t6a, verbose=True, label="p6a")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6a_ccd")

    rem = deadline - time.monotonic() - 20
    if rem > 20:
        t6b = min(rem * 0.5, 60.0)
        a6b, s6b = fine_coord_descent(best_seq, t6b, label="p6b")
        best_seq, best_score = check_and_update(a6b, best_seq, best_score, "p6b_fine")

    rem = deadline - time.monotonic() - 10
    if rem > 10:
        a6c, s6c, _ = corrected_coord_descent(best_seq, rem * 0.6, label="p6c")
        best_seq, best_score = check_and_update(a6c, best_seq, best_score, "p6c_ccd")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
        print(f"Verified final score: {best_score:.6f}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
