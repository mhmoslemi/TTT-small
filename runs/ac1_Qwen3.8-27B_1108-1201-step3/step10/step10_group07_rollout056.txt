def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lp_best_response(a, solver):

        T @ b_var <= M_var * ones,

    b_val = np.asarray(b_var.value, dtype=np.float64)

def freq_shaping(a, sharpen=0.5, n_iter=20):
    """
    Frequency-domain auto-correlation shaping.
    Modify the DFT magnitude to flatten the auto-correlation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for it in range(n_iter):
        # Compute DFT
        A = fft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: IDFT(|A|^2) should be flat
        # |A|^2 is the power spectrum
        power = mag ** 2
        
        # We want power to be as flat as possible (minimize variance)
        # while keeping A[0] = 1 (DC constraint)
        # And IDFT(A) >= 0
        
        # Strategy: compress the dynamic range of the power spectrum
        # while keeping DC fixed
        power[0] = 1.0  # Keep DC
        # Compress: new_power = 1 - sharpen + sharpen * (power / max_power)
        max_p = power[1:].max() if n > 1 else 1.0
        if max_p > 0:
            power[1:] = 1.0 - sharpen + sharpen * (power[1:] / max_p)
        
        # Reconstruct with new magnitude, old phase
        new_mag = np.sqrt(power)
        new_A = new_mag * np.exp(1j * phase)
        new_A[0] = 1.0
        
        # IDFT
        new_a = np.real(ifft(new_A))
        
        # Threshold negatives
        new_a = np.maximum(new_a, 0.0)
        
        # Renormalize
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a / s_new
        
        # Check if improvement
        if score_of(new_a) < score_of(a):
            a = new_a
        else:
            # Try smaller sharpen
            break
    
    return a

def exact_coord_descent(a0, time_limit, verbose=True, label=""):

    Exact coordinate descent: for each coordinate, find the opt value
    by minimizing the quadratic form (given all other coords fixed).
    """

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / n * n  # keep as is, normalize later
    
    # Normalize to sum=1 for optimization
    a = a / a.sum()
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-18:
                continue
            
            # When we change a[i] by delta, conv changes as:
            # new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2 (if k=2i)
            # for k in [i, min(i+n-1, 2n-2)]
            # (with appropriate bounds)
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            # Affected convolution values
            # conv_affected[k] for k in [k_start, k_end]
            a_shifted = a[k_start - i:k_end + 1]  # a[k-i] for k in [k_start, k_end]
            conv_aff = conv[k_start:k_end + 1].copy()
            
            # The new conv at position k when a[i] changes by delta:
            # new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2 * I(k==2i)
            # Let c[k] = conv[k], b[k] = a[k-i], d = I(k==2i)
            # new_conv[k] = c[k] + 2*delta*b[k] + d*delta^2
            # We want to min max_k new_conv[k]
            # This is: min max_k (d[k]*delta^2 + 2*b[k]*delta + c[k])
            # The d term is only non-zero at k=2i (if in range)
            
            d = np.zeros(k_end - k_start + 1)
            if 2 * i >= k_start and 2 * i <= k_end:
                d[2 * i - k_start] = 1.0
            
            # Objective: min over delta >= -a[i] of max_k (d[k]*delta^2 + 2*b[k]*delta + c[k])
            # Also need new_conv[k] >= 0 for all k (but since a >= 0, conv >= 0 always)
            
            # This is a 1D minimax problem!
            # For each k: f_k(delta) = d[k]*delta^2 + 2*b[k]*delta + c[k]
            # We want min_delta max_k f_k(delta)
            
            # The optimal delta is where the "active" functions are equal.
            # For a 1D minimax of quadratics, we can use bisection on the value M.
            # For a given target M, check if exists delta s.t. f_k(delta) <= M for all k.
            # f_k(delta) <= M means:
            #   If d[k] > 0: delta^2 + 2*b[k]/d[k]*delta + (c[k]-M)/d[k] <= 0
            #     => delta in [root1, root2] (quadratic inequality)
            #   If d[k] == 0: 2*b[k]*delta + c[k] - M <= 0
            #     => If b[k] > 0: delta <= (M - c[k]) / (2*b[k])
            #     => If b[k] < 0: delta >= (M - c[k]) / (2*b[k])
            #     => If b[k] == 0: need c[k] <= M
            # Plus constraint delta >= -a[i]
            
            # Bisection on M:
            lo_M = max(unaff_max, float(conv_aff.min()))  # lower bound on achievable M
            hi_M = M  # current M
            
            if hi_M - lo_M < 1e-14:
                continue
            
            best_delta = 0.0
            best_M = M
            
            for _ in range(30):  # 30 bisection iterations ≈ 2^-30 precision
                mid_M = (lo_M + hi_M) / 2.0
                
                # Feasible set for delta given target mid_M:
                # Start with delta in [-a[i], +inf)
                d_lo = -a[i]
                d_hi = 1e10  # effectively infinite
                
                for k_idx in range(len(conv_aff)):
                    c_k = conv_aff[k_idx]
                    b_k = a_shifted[k_idx]
                    d_k = d[k_idx]
                    
                    if d_k > 0:
                        # delta^2 + 2*b_k*delta + (c_k - mid_M) <= 0
                        # roots: -b_k ± sqrt(b_k^2 - (c_k - mid_M))
                        disc = b_k * b_k - (c_k - mid_M)
                        if disc < 0:
                            # No real roots, quadratic always positive → infeasible
                            d_lo = 1e10  # mark infeasible
                            break
                        elif disc == 0:
                            root = -b_k
                            d_lo = max(d_lo, root)
                            d_hi = min(d_hi, root)
                        else:
                            sqrt_disc = np.sqrt(disc)
                            root1 = -b_k - sqrt_disc
                            root2 = -b_k + sqrt_disc
                            d_lo = max(d_lo, root1)
                            d_hi = min(d_hi, root2)
                    elif d_k == 0:
                        if b_k > 1e-18:
                            # 2*b_k*delta + c_k - mid_M <= 0
                            # delta <= (mid_M - c_k) / (2*b_k)
                            bound = (mid_M - c_k) / (2.0 * b_k)
                            d_hi = min(d_hi, bound)
                        elif b_k < -1e-18:
                            # delta >= (mid_M - c_k) / (2*b_k)
                            bound = (mid_M - c_k) / (2.0 * b_k)
                            d_lo = max(d_lo, bound)
                        else:
                            # c_k - mid_M <= 0 must hold
                            if c_k > mid_M:
                                d_lo = 1e10
                                break
                
                if d_lo <= d_hi:
                    # Feasible! M = mid_M is achievable
                    hi_M = mid_M
                    best_delta = (d_lo + d_hi) / 2.0  # midpoint of feasible range
                    best_M = mid_M
                else:
                    lo_M = mid_M
            
            if best_M < M - 1e-14:
                # Apply the update
                a[i] = max(a[i] + best_delta, 0.0)
                # Update conv
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * a[i] + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_descent(a0, time_limit, lr_init=0.02, top_k=20, verbose=True):
    """Subgradient descent with multi-peak weighting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.003 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 10:
            a = b_new.copy()
        elif it <= 30:
            alpha = 0.6
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            alpha = max(0.2, 8.0 / (it + 10.0))
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 1e-8):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            noise_scale = 0.005 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.997, 0.994, 0.998, 0.999, 1.001, 1.002, 1.006, 1.009]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain auto-correlation shaping (novel: modifies DFT magnitude to flatten correlation)
2. LP-based best-response alternating minimization (primary driver)
3. Exact coordinate descent via 1D minimax bisection (novel: precise per-coordinate optimization)
4. Subgradient descent with softmax peak weighting
5. Block Nelder-Mead for local refinement
6. Fine-grained polishing for last-mile improvements
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_time=60)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=60)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def freq_shaping(a, sharpen=0.5, n_iter=30):
    """
    Frequency-domain auto-correlation shaping.
    Modify the DFT magnitude to compress the power spectrum dynamic range,
    which flattens the auto-correlation in the time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        A = fft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        power = (mag ** 2).copy()
        power[0] = 1.0

        if n > 1:
            max_p = power[1:].max()
            if max_p > 1e-15:
                power[1:] = 1.0 - sharpen + sharpen * (power[1:] / max_p)

        new_mag = np.sqrt(np.maximum(power, 0))
        new_A = new_mag * np.exp(1j * phase)
        new_A[0] = 1.0

        new_a = np.real(ifft(new_A))
        new_a = np.maximum(new_a, 0.0)

        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a / s_new

        sc = 2.0 * n * fast_convolve(new_a).max()
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
        else:
            # Try with reduced sharpen
            break

    return best_a


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Exact coordinate descent using 1D minimax bisection per coordinate.
    For each coordinate, finds the optimal value by bisection on the
    target max convolution value.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Affected data
            a_shifted = a[k_start - i:k_end + 1]  # a[k-i] for k in range
            conv_aff = conv[k_start:k_end + 1]

            # Check if 2i is in affected range
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_k = 2 * i

            # Bisection on target M
            lo_M = max(unaff_max, float(conv_aff.min()))
            hi_M = M

            if hi_M - lo_M < 1e-15:
                continue

            best_delta = 0.0
            best_M = M

            for _ in range(35):
                mid_M = (lo_M + hi_M) / 2.0

                # Feasible delta range: start with [-a[i], +inf)
                d_lo = -a[i]
                d_hi = 1e10

                # Linear constraints (k != 2i)
                if has_quad:
                    mask_quad = (np.arange(k_start, k_end + 1) == quad_k)
                    mask_lin = ~mask_quad
                else:
                    mask_lin = np.ones(len(conv_aff), dtype=bool)

                if mask_lin.any():
                    b_lin = a_shifted[mask_lin]
                    c_lin = conv_aff[mask_lin]

                    # b > 0: upper bound
                    pos_mask = b_lin > 1e-18
                    if pos_mask.any():
                        ub = (mid_M - c_lin[pos_mask]) / (2.0 * b_lin[pos_mask])
                        d_hi = min(d_hi, float(ub.min()))

                    # b < 0: lower bound
                    neg_mask = b_lin < -1e-18
                    if neg_mask.any():
                        lb = (mid_M - c_lin[neg_mask]) / (2.0 * b_lin[neg_mask])
                        d_lo = max(d_lo, float(lb.max()))

                    # b = 0: need c <= mid_M
                    zero_mask = (b_lin > -1e-18) & (b_lin < 1e-18)
                    if zero_mask.any():
                        if c_lin[zero_mask].max() > mid_M:
                            d_lo = 1e10
                            break

                # Quadratic constraint (k = 2i)
                if has_quad:
                    b_q = a[i]  # a[2i - i] = a[i]
                    c_q = conv[quad_k]
                    # delta^2 + 2*b_q*delta + (c_q - mid_M) <= 0
                    disc = b_q * b_q - (c_q - mid_M)
                    if disc < 0:
                        d_lo = 1e10
                        break
                    elif disc < 1e-30:
                        root = -b_q
                        d_lo = max(d_lo, root)
                        d_hi = min(d_hi, root)
                    else:
                        sqrt_d = np.sqrt(disc)
                        d_lo = max(d_lo, -b_q - sqrt_d)
                        d_hi = min(d_hi, -b_q + sqrt_d)

                if d_lo <= d_hi:
                    hi_M = mid_M
                    best_delta = (d_lo + d_hi) / 2.0
                    best_M = mid_M
                else:
                    lo_M = mid_M

            if best_M < M - 1e-14 and best_delta != 0.0:
                # Apply update
                old_ai = a[i]
                a[i] = max(old_ai + best_delta, 0.0)
                # Update conv
                # The change in conv[k] = 2*delta*a[k-i] + delta^2*I(k=2i)
                # But note: a[k-i] should be the OLD value of a at position k-i
                # (which hasn't changed since we only modified a[i])
                delta = a[i] - old_ai
                changes = 2.0 * delta * a_shifted
                if has_quad:
                    idx = quad_k - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * old_ai + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_descent(a0, time_limit, lr_init=0.02, top_k=30, verbose=True):
    """Subgradient descent with multi-peak softmax weighting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.002 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score
