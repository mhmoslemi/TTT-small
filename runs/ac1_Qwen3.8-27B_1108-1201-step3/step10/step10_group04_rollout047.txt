def fast_convolve(a):
    """Fast self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_cross(a, b):
    """Fast cross-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_normalized(a):
    """Compute the score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    b = fast_convolve(a)

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

    s = b_val.sum()

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()  # Nesterov auxiliary variable
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP to get best response
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = (t_k - 1.0) / t_k
        beta = it / t_k
        
        # Standard Nesterov:
        # y = (1 - alpha) * a + alpha * b_new  (this is the "y" update)

        # y_new = a_new + momentum*(a_new - a_old)

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)

        if sa < 1e-15:
            a = b_new.copy()
        else:
            a = a / sa
        
        # Update y with momentum
        y = a + beta * (a - a_old)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Track best
        sc_a = score_normalized(a)
        sc_y = score_normalized(y)
        if sc_a < best_score:

            best_a = a.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov accelerated best-response (main workhorse, fixed formulation)
2. LogSumExp L-BFGS multi-stage annealing for fine polishing
3. Subgradient descent with momentum for final refinement
4. Multi-length exploration around parent sequence
Key improvements: correct Nesterov, PDLP solver, focused time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_func(z_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = lse_objective_grad(z_val, T_start, n)
        return obj

    def lse_jac(z_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_val)
        _, grad = lse_objective_grad(z_val, T_start, n)
        return grad

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 30

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def subgradient_polish(a0, time_limit, n_steps=500000, eta_start=0.5, eta_end=0.01):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    gamma = 0.7

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.95)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            z = gamma * z + (1.0 - gamma) * z  # keep z, barely contract
            # Actually just don't update
            pass

        # Simple: always update z but track best
        z = z - eta * grad_z
        a = softmax_from_z(z)
        sc_cur = score_normalized(a)
        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    else:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
