def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

Let me think about a different approach: **iterative reweighted least squares (IRLS)**.

The problem is: minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1.

This can be reformulated using the log-sum-exp approximation:
minimize log(sum_k exp(beta * (a*a)[k])) / beta

for large beta. This is a smooth approximation to the max problem.

The gradient of this objective w.r.t. a is:
nabla_a log(sum_k exp(beta * (a*a)[k])) = sum_k [exp(beta*(a*a)[k]) / sum_j exp(beta*(a*a)[j])] * 2 * (a * 1_{[k-n+1, k]})

where (a * 1_{[k-n+1, k]})[i] = a[k-i] if i is in [k-n+1, k], else 0.

This is essentially a weighted sum of shifted versions of a, where the weights are softmax over the convolution values.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)

    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lp_best_response(a, solver, n):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt, active_lags) where active_lags are the binding constraints.
    """
    T = build_conv_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    conv_constraints = T @ b_var <= M_var * ones
    constraints = [conv_constraints, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf'), np.array([n - 1])

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf'), np.array([n - 1])

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf'), np.array([n - 1])
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')

    # Identify active (binding) constraints
    conv_ab = convolve_ab(a, b_val)
    active_lags = np.where(conv_ab >= M_val * 0.999)[0]
    if len(active_lags) == 0:
        active_lags = np.array([n - 1])

    return b_val, M_val, active_lags


def subgradient_descent(a0, time_limit, lr_init=0.02, top_k=30, beta_temp=0.002, verbose=True):
    """
    Subgradient descent with softmax-weighted multi-peak subgradient.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-12:
                break

        # Compute multi-peak subgradient with softmax weighting
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        temp = beta_temp * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient computation
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # sg[i] += w[k] * 2 * a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] is a[k-i_end : k-i_start+1] reversed
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Gradient step + projection
        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def adaptive_coord_descent(a0, time_limit, verbose=True, fine=False):
    """
    Coordinate descent with adaptive step sizes and incremental convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    if fine:
        factors = [0.997, 0.994, 0.998, 1.002, 1.006, 1.003, 0.999, 1.001]
    else:
        factors = [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            base = a[i]
            for factor in factors:
                if fine:
                    delta = base * (factor - 1.0)
                else:
                    delta = base * factor
                new_val = base + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * base + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > (15 if fine else 30):
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True, label=""):
    """
    LP-based best-response alternating minimization with adaptive damping.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    alpha = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp, active_lags = lp_best_response(a, solver, n)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Adaptive damping
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.3, 8.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss
        else:
            a = best_a.copy()
            break

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation detection
        if score_a > best_score * (1 + 5e-6):

        else:
            stagnation = max(0, stagnation - 1)

def diverse_init(n, seed, init_type):
    """Generate diverse initializations with different structural priors."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center

    if init_type == 0:
        # Triangle
        a = np.maximum(1.0 - np.abs(x), 0)
    elif init_type == 1:
        # U-shape
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        # Cosine power
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Two humps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Linear ramp
        a = (t + 1) / (n + 1)
    elif init_type == 5:
        # Exponential decay
        tau = n * rng.uniform(0.3, 0.8)
        a = np.exp(-t / tau)
    elif init_type == 6:
        # Asymmetric Gaussian
        sigma_l = n * rng.uniform(0.1, 0.3)
        sigma_r = n * rng.uniform(0.1, 0.3)
        mu = n * rng.uniform(0.3, 0.7)
        a = np.where(t < mu,
                     np.exp(-0.5 * ((t - mu) / sigma_l) ** 2),
                     np.exp(-0.5 * ((t - mu) / sigma_r) ** 2))
    elif init_type == 7:
        # Sinusoidal modulation
        freq = rng.uniform(1, 5)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 8:
        # Wasserfall (1/i decay)
        a = 1.0 / (t + 1.0)
    else:
        # Smooth random
        a = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 3) * t / n)
        a = a + 0.2 * np.sin(2 * np.pi * rng.uniform(3, 7) * t / n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def peak_shaving(a0, time_limit, verbose=True):
    """Targeted peak reduction by redistributing mass from contributing pairs."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        K = 8
        if K > 2 * n - 1:
            K = 2 * n - 1
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.9:
                continue

            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Find top contributing pairs
            contributions = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                contributions.append((i, j, a[i] * a[j]))
            contributions.sort(key=lambda x: x[2], reverse=True)

            for (i, j, contrib) in contributions[:5]:
                if contrib < 1e-14:
                    continue

                # Try reducing the larger element
                if a[i] >= a[j]:
                    target = i
                else:
                    target = j

                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = -a[target] * frac
                    if delta == 0:
                        continue

                    new_a = a.copy()
                    new_a[target] = max(new_a[target] + delta, 0.0)

                    k_s = target
                    k_e = min(target + n - 1, 2 * n - 2)
                    a_shifted = a[k_s - target:k_e - target + 1]
                    new_affected = conv[k_s:k_e + 1] + 2.0 * delta * a_shifted
                    if 2 * target >= k_s and 2 * target <= k_e:
                        idx = 2 * target - k_s
                        new_affected = new_affected.copy()
                        new_affected[idx] = conv[2 * target] + 2.0 * delta * a[target] + delta * delta

                    if k_s > 0:
                        unaff_max = max(conv[:k_s].max(), conv[k_e + 1:].max()) if k_e < 2 * n - 2 else conv[:k_s].max()
                    else:
                        unaff_max = conv[k_e + 1:].max() if k_e < 2 * n - 2 else -1e30

                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < M - 1e-13:
                        a = new_a
                        conv[k_s:k_e + 1] = new_affected
                        M = new_M
                        improved = True
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        break

        if not improved and rounds > 30:
            break

        if verbose and rounds % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave r={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def ipm_refine(a0, time_limit, verbose=True):
    """
    Interior-point-like refinement using log-sum-exp barrier.
    Minimizes log(sum exp(beta * conv_k)) which approximates max(conv_k).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.001
    beta = 50.0  # Sharpness of log-sum-exp
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0:
            beta *= 2.0  # Increase sharpness over time
            lr *= 0.7

        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        # Log-sum-exp gradient
        shifted = conv - conv.max()
        log_sum = np.log(np.exp(shifted * beta).sum())
        w = np.exp(shifted * beta) / np.exp(log_sum)

        # Gradient of log-sum-exp w.r.t. a
        sg = np.zeros(n)
        significant = w > 1e-10
        for k in np.where(significant)[0]:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[k] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a = a - lr * sg
        a = simplex_project(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    IPM it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.0f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating from parent (main workhorse) ===
    print(f"\n=== Phase 1: LP Alt from parent (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    t_alloc = min(t1_end - time.monotonic(), 500.0)
    print(f"  LP alt parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True, label="parent")
    try_update(a_lp, "lp_parent")

    # LP from diverse inits
    inits_to_try = [
        diverse_init(n, 100, 4),  # Linear ramp
        diverse_init(n, 101, 6),  # Asymmetric Gaussian
        diverse_init(n, 102, 0),  # Triangle
        diverse_init(n, 103, 5),  # Exponential decay
        diverse_init(n, 104, 8),  # Wasserfall
    ]

    for idx, init in enumerate(inits_to_try):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(remaining * 0.3, 60.0)
        a_tmp, s_tmp = lp_alternating(init, t_alloc, solver, verbose=False, label=f"init{idx}")
        try_update(a_tmp, f"lp_init{idx}")

    # Second LP round on best
    if time.monotonic() < t1_end - 30:
        t_alloc = min(t1_end - time.monotonic(), 200.0)
        a_lp2, s_lp2 = lp_alternating(best_seq, t_alloc, solver, verbose=True, label="r2")
        try_update(a_lp2, "lp_r2")

    # === Phase 2: Subgradient Descent ===
    print(f"\n=== Phase 2: SGD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    if time.monotonic() < t2_end:
        t_alloc = min(t2_end - time.monotonic(), 150.0)
        a_sgd, s_sgd = subgradient_descent(best_seq, t_alloc, lr_init=0.01, top_k=40, verbose=True)
        try_update(a_sgd, "sgd")

        if time.monotonic() < t2_end - 30:
            a_sgd2, s_sgd2 = subgradient_descent(a_lp, min(t2_end - time.monotonic(), 80.0),
                                                  lr_init=0.005, top_k=50, verbose=False)
            try_update(a_sgd2, "sgd_lp")

    # === Phase 3: IPM Refinement ===
    print(f"\n=== Phase 3: IPM (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.15

    if time.monotonic() < t3_end:
        t_alloc = min(t3_end - time.monotonic(), 100.0)
        a_ipm, s_ipm = ipm_refine(best_seq, t_alloc, verbose=True)
        try_update(a_ipm, "ipm")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: CD (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.25

    if time.monotonic() < t4_end:
        t_alloc = min(t4_end - time.monotonic(), 200.0)
        a_cd, s_cd = adaptive_coord_descent(best_seq, t_alloc, verbose=True, fine=False)
        try_update(a_cd, "cd")

        if time.monotonic() < t4_end - 60:
            a_cd2, s_cd2 = adaptive_coord_descent(a_lp, min(t4_end - time.monotonic(), 100.0),
                                                  verbose=False, fine=False)
            try_update(a_cd2, "cd_lp")

    # === Phase 5: Peak Shaving + Fine Polish ===
    print(f"\n=== Phase 5: Polish (best={best_score:.8f}) ===")
    t5_end = deadline - 10

    if time.monotonic() < t5_end - 40:
        t_ps = min(t5_end - time.monotonic() - 40, 80.0)
        if t_ps > 15:
            a_ps, s_ps = peak_shaving(best_seq, t_ps, verbose=True)
            try_update(a_ps, "peak")

    if time.monotonic() < t5_end - 20:
        t_fp = min(t5_end - time.monotonic() - 20, 80.0)
        if t_fp > 10:
            a_fp, s_fp = adaptive_coord_descent(best_seq, t_fp, verbose=True, fine=True)
            try_update(a_fp, "fine_cd")

    # Final CD pass
    if time.monotonic() < t5_end - 5:
        t_final = min(t5_end - time.monotonic() - 5, 30.0)
        if t_final > 5:
            a_fin, s_fin = adaptive_coord_descent(best_seq, t_final, verbose=False, fine=True)
            try_update(a_fin, "final")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
