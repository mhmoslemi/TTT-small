def softmax(z):

    s=e.sum()
    if s<=0 or not finite: return np.ones(len(z))/len(z)

"""
Efficient peak-flattening optimizer for min 2*n*max(conv(a,a))/(sum(a))^2.

Main ideas:
- Normalize sequences to sum 1 and solve the scale-invariant problem.
- Use FFT-based linear auto-convolution and a duality-friendly Hankel operator
  h_i = sum_k w_k a_{i+k} for fast gradients/subgradients.
- Primary method: Adam in log-domain for smoothed LogSumExp objectives
  (effective temperature T = tau / (2n)), followed by L-BFGS refinement,
  a multiplicative max-min dual/primal refinement, and short active-peak polish.
- Multi-length multi-start search around the supplied parent construction.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    minimize = None
    HAVE_SCIPY = False


# -----------------------------
# Fast FFT-based building blocks
# -----------------------------

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def convolve_pair(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if x.size == 0 or y.size == 0:
        return np.array([], dtype=np.float64)
    size = x.size + y.size - 1
    nfft = _next_pow2(size)
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    return np.fft.irfft(X * Y, nfft)[:size]


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    nfft = _next_pow2(size)
    A = np.fft.rfft(a, nfft)
    return np.fft.irfft(A * A, nfft)[:size]


def cross_corr(f, w):
    """
    h_i = sum_j f_j w_{i+j}, i = 0..n-1, where len(f)=n and len(w)=2n-1
    (or more generally any length where i+j is valid for all i,j).
    """
    f = np.asarray(f, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = f.size
    L = w.size
    size = n + L - 1
    nfft = _next_pow2(size)
    F = np.fft.rfft(f[::-1], nfft)
    W = np.fft.rfft(w, nfft)
    c = np.fft.irfft(F * W, nfft)[:size]
    start = n - 1
    return c[start:start + n]


def normalize_nonnegative(a, target_sum=1.0):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) * (target_sum / a.size)
    return a * (target_sum / s)


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=100.0, neginf=-100.0)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def score_local(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    n = a.size
    c = fast_convolve(a)
    if c.size == 0:
        return float("inf")
    mc = float(np.max(c))
    if not np.isfinite(mc):
        return float("inf")
    return float(2.0 * n * mc / (s * s))


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    if f.size == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    f = f / s
    if f.size == 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    old_x = np.linspace(0.0, 1.0, f.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return out / ss


def structured_init(n, rng, mode=0):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-8
    mode = int(mode) % 14

    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = (1.0 - x) ** 1.0 + 0.03
    elif mode == 2:
        a = (1.0 - x) ** 2.0 + 0.02
    elif mode == 3:
        a = (1.0 - x) ** 0.7 + 0.04
    elif mode == 4:
        a = np.exp(-3.0 * x) + 0.05
    elif mode == 5:
        a = np.exp(-5.0 * x) + 0.05
    elif mode == 6:
        a = np.exp(-6.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-6.0 * (x - 0.8) ** 2) + 0.02
    elif mode == 7:
        a = np.exp(-8.0 * (x - 0.35) ** 2) + 0.4 * np.exp(-4.0 * (x - 0.85) ** 2) + 0.02
    elif mode == 8:
        a = 1.0 / np.sqrt(x + eps) + 0.5 / np.sqrt(1.0 - x + eps)
    elif mode == 9:
        a = 1.0 / np.sqrt(x + eps) + 0.05
    elif mode == 10:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x) + 0.1
    elif mode == 11:
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1
    elif mode == 12:
        a = 0.5 + 0.25 * np.sin(6.0 * np.pi * x) + 0.1
    else:
        a = 1.0 + 0.15 * rng.standard_normal(n)

    a = np.maximum(np.nan_to_num(a, nan=0.0, posinf=10.0, neginf=-10.0), 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def update_best(best_seq, best_score, candidate, label):
    if candidate is None:
        return best_seq, best_score
    c = np.asarray(candidate, dtype=np.float64)
    if c.size == 0:
        return best_seq, best_score
    c = np.maximum(c, 0.0)
    c = np.nan_to_num(c, nan=0.0, posinf=1000.0, neginf=0.0)
    s = float(c.sum())
    if s < 0.01:
        return best_seq, best_score
    c = c / s
    sc = score_local(c)
    if np.isfinite(sc) and sc < best_score:
        print(f"    NEW BEST [{label}]: {sc:.10f}  n={c.size}")
        return c, sc
    return best_seq, best_score

# LogSumExp globals for scipy.optimize (needed to keep no closures)

GL_N = 0
GL_T = 1.0


def _lse_weights(c, T):
    """Return normalized weights for LSE standard objective."""
    mc = float(np.max(c))
    if mc <= 0.0 or not np.isfinite(mc):
        n = c.size
        return np.ones(n, dtype=np.float64) / float(max(1, n))
    x = T * (c - mc)
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        mask = c >= (mc - 1e-12 * max(1.0, mc))
        if not np.any(mask):
            mask[int(np.argmax(c))] = True
        w = mask.astype(np.float64)
        sw = float(w.sum())
        if sw <= 0.0:
            w[int(np.argmax(c))] = 1.0
            sw = 1.0
        return w / sw
    return ex / s


def smooth_tf_lse(z):
    if GL_N <= 0:
        return 0.0
    n = int(GL_N)
    a = softmax(z)
    c = np.maximum(fast_convolve(a), 0.0)
    if c.size == 0:
        return 0.0
    mc = float(np.max(c))
    T = float(GL_T)
    if mc <= 0.0 or not np.isfinite(mc) or T <= 0.0:
        return 0.0
    x = T * (c - mc)
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        return float(mc)
    return float(mc + np.log(s) / T)


def smooth_tf_lse_grad(z):
    if GL_N <= 0:
        return np.array([], dtype=np.float64)
    n = int(GL_N)
    a = softmax(z)
    c = np.maximum(fast_convolve(a), 0.0)
    T = float(GL_T)
    w = _lse_weights(c, T)
    H = cross_corr(a, w)
    S = float(np.dot(a, H))
    return 2.0 * a * (H - S)

def adam_lse_optimize(
    a0,
    time_limit,
    tau_schedule=(20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02),
    lr=0.03,
    eval_interval=10,
    patience=60,
    print_interval=2500
):
    a = normalize_nonnegative(a0)
    n = a.size
    if n < 2 or time_limit <= 0.0:
        return a

    loga = np.log(np.maximum(a, 1e-300))
    loga -= float(np.mean(loga))

    best_a = a.copy()
    best_score = score_local(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    no_improve = 0
    rng = np.random.default_rng(123456789)

    tau_list = [float(x) for x in list(tau_schedule)]
    Ltau = len(tau_list)
    weights = np.arange(1, Ltau + 1, dtype=np.float64)
    fractions = weights / float(weights.sum())

    t0 = time.monotonic()
    tau_idx = 0

    eval_interval = max(3, int(eval_interval))
    print_interval = max(eval_interval, int(print_interval))

    while tau_idx < Ltau and (time.monotonic() - t0) < time_limit:
        tau = tau_list[tau_idx]
        stage_frac = float(np.sum(fractions[:tau_idx + 1]))
        stage_deadline = t0 + time_limit * stage_frac
        if stage_deadline < time.monotonic():
            stage_deadline = time.monotonic() + max(1.0, time_limit - (time.monotonic() - t0))

        step = lr
        it = 0

        while (time.monotonic() - t0) < time_limit and time.monotonic() < stage_deadline:
            it += 1
            a = softmax(loga)
            c = np.maximum(fast_convolve(a), 0.0)
            if c.size == 0:
                break
            mc = float(np.max(c))
            if not np.isfinite(mc) or mc <= 0.0:
                break

            T = tau / (2.0 * float(n))
            if T <= 0.0 or not np.isfinite(T):
                break

            w = _lse_weights(c, T)
            H = cross_corr(a, w)
            S = float(np.dot(a, H))
            grad = 2.0 * a * (H - S)

            scale = float(np.max(np.abs(grad)))
            if (not np.isfinite(scale)) or scale <= 1e-30:
                scale = 1.0
            g = grad / scale

            t += 1
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            a1 = (1.0 - 0.9) ** t
            a2 = (1.0 - 0.999) ** t
            mhat = m / max(1e-30, 1.0 - a1)
            vhat = v / max(1e-30, 1.0 - a2)

            loga = loga - step * mhat / (np.sqrt(vhat) + 1e-8)

            if (it % 250) == 0:
                loga -= float(np.mean(loga))

            if (it % eval_interval) == 0:
                a_trial = softmax(loga)
                sc = score_local(a_trial)
                if np.isfinite(sc):
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        no_improve = 0
                        step = min(step * 1.04, 0.18)
                    else:
                        no_improve += 1
                        if no_improve > patience:
                            base = np.log(np.maximum(best_a, 1e-300))
                            noise = rng.standard_normal(n) * 0.02
                            loga = base + noise
                            loga -= float(np.mean(loga))
                            m = np.zeros(n, dtype=np.float64)
                            v = np.zeros(n, dtype=np.float64)
                            t = 0
                            no_improve = 0
                            step = lr
                            break

                    if (it % print_interval) == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      adam_lse tau={tau:7.3f} t={elapsed:7.2f}s best={best_score:.10f}")

        tau_idx += 1

    return best_a


def lbgfs_lse_optimize(
    a0,
    time_limit,
    tau_schedule=(20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1),
    maxiter_base=700
):
    a = normalize_nonnegative(a0)
    n = a.size
    if n < 2 or time_limit <= 0.0 or (not HAVE_SCIPY) or (minimize is None):
        return a

    loga = np.log(np.maximum(a, 1e-300))
    loga -= float(np.mean(loga))
    best_a = a.copy()
    best_score = score_local(a)

    tau_list = [float(x) for x in list(tau_schedule)]
    L = len(tau_list)
    t0 = time.monotonic()

    for idx, tau in enumerate(tau_list):
        if (time.monotonic() - t0) > time_limit:
            break

        stage_time = time_limit / float(L)
        if stage_time < 5.0:
            break

        GL_N = n
        GL_T = tau / (2.0 * float(n))

        z = loga - float(np.mean(loga))

        # Conservative iteration/fun caps to respect wall-clock budgets.
        per_iter_cost = 0.04 + 0.00006 * float(n)
        maxiter = int(min(int(maxiter_base), max(60, stage_time / max(per_iter_cost, 0.025))))
        maxfun = int(max(200, 3 * maxiter))

        options = {
            "maxiter": int(maxiter),
            "maxfun": int(maxfun),
            "ftol": 1e-12,
            "gtol": 1e-10,
            "maxls": 20,
            "maxcor": 20,
            "disp": False
        }

        try:
            res = minimize(
                fun=smooth_tf_lse,
                x0=z,
                jac=smooth_tf_lse_grad,
                method="L-BFGS-B",
                options=options
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, "x"):
            z_new = np.asarray(res.x, dtype=np.float64)
            z_new = np.nan_to_num(z_new, nan=0.0, posinf=100.0, neginf=-100.0)
            a_new = softmax(z_new)
            sc_new = score_local(a_new)

            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                print(f"      lbgfs tau={tau:7.3f}: exact best {best_score:.10f}")

            # Continue from the new point unless it degraded too much.
            stage_start_score = score_local(softmax(loga))
            accept = np.isfinite(sc_new) and (sc_new <= max(stage_start_score, best_score) * 1.01 + 1e-10)
            if accept:
                loga = z_new - float(np.mean(z_new))
        else:
            pass

    return best_a


def dual_acm_optimize(
    a0,
    time_limit,
    inner_iters=16,
    eta=0.22,
    t_v=0.30,
    gamma=0.42,
    patience=45
):
    a = normalize_nonnegative(a0)
    n = a.size
    if n < 2 or time_limit <= 0.0:
        return a

    loga = np.log(np.maximum(a, 1e-300))
    loga -= float(np.mean(loga))

    best_a = a.copy()
    best_score = score_local(a)
    current_score = best_score
    eta_cur = eta
    eta_start = eta

    L = 2 * n - 1
    u = np.ones(L, dtype=np.float64) / float(L)
    no_improve = 0
    total_resets = 0
    rng = np.random.default_rng(987654321)

    t0 = time.monotonic()

    while (time.monotonic() - t0) < time_limit:
        # Initialize/update dual from current peaks when useful.
        if no_improve == 0 and (u < 1e-14).any():
            c = np.maximum(fast_convolve(a), 0.0)
            su = float(c.sum())
            if su > 1e-15:
                u = c / su
            else:
                u = np.ones(L, dtype=np.float64) / float(L)

        # --- Inner max-min dual refinement for fixed a ---
        for _ in range(int(inner_iters)):
            h = cross_corr(a, u)
            min_h = float(np.min(h))
            scale_h = float(np.std(h))
            if (not np.isfinite(scale_h)) or scale_h <= 1e-15:
                break

            ex = (h - min_h) / (max(t_v, 1e-8) * scale_h)
            ex = np.clip(ex, 0.0, 50.0)
            vv = np.exp(-ex)
            sv = float(vv.sum())
            if (not np.isfinite(sv)) or sv <= 1e-300:
                break
            vv = vv / sv

            sk = convolve_pair(vv, a)
            ski = float(sk.mean())
            ssk = float(np.std(sk))
            if (not np.isfinite(ssk)) or ssk <= 1e-15:
                break

            eu = gamma * (sk - ski) / ssk
            eu = np.clip(eu, -18.0, 18.0)
            u_new = u * np.exp(eu)
            su = float(u_new.sum())
            if (not np.isfinite(su)) or su <= 1e-300:
                break
            u_new = u_new / su

            # Small entropy/mixing for robustness.
            unif = 1.0 / float(L)
            u = 0.985 * u_new + 0.015 * unif
            su = float(u.sum())
            if su <= 1e-300:
                u = np.ones(L, dtype=np.float64) / float(L)
            else:
                u = u / su

        # --- Primal update from current dual h ---
        h = cross_corr(a, u)
        S = float(np.dot(a, h))
        Hc = h - S
        scale = float(np.sqrt(np.dot(a, Hc * Hc)))
        if (not np.isfinite(scale)) or scale <= 1e-15:
            break

        loga_new = loga - eta_cur * Hc / scale
        a_trial = softmax(loga_new)
        sc_trial = score_local(a_trial)
        if not np.isfinite(sc_trial):
            break

        if sc_trial < best_score:
            best_score = sc_trial
            best_a = a_trial.copy()

        if sc_trial <= current_score + 1e-11:
            a = a_trial
            loga = loga_new
            current_score = sc_trial
            no_improve = 0
            eta_cur = min(eta_cur * 1.025, 0.65)
        else:
            no_improve += 1
            eta_cur = max(eta_cur * 0.80, 0.015)
            if no_improve > int(patience):
                # Restart around current best with small noise and fresh dual.
                base = np.log(np.maximum(best_a, 1e-300))
                noise = rng.standard_normal(n) * 0.015
                loga = base + noise
                loga -= float(np.mean(loga))
                a = softmax(loga)
                current_score = score_local(a)
                c = np.maximum(fast_convolve(a), 0.0)
                su = float(c.sum())
                if su > 1e-15:
                    u = c / su
                else:
                    u = np.ones(L, dtype=np.float64) / float(L)
                eta_cur = eta_start
                no_improve = 0
                total_resets += 1
                if total_resets > 8:
                    break

    return best_a


def final_polish(
    a0,
    time_limit,
    fracs=(0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001),
    lr=0.06
):
    a = normalize_nonnegative(a0)
    n = a.size
    if n < 2 or time_limit <= 0.0:
        return a

    loga = np.log(np.maximum(a, 1e-300))
    loga -= float(np.mean(loga))
    best_a = a.copy()
    best_score = score_local(a)
    current_score = best_score

    t0 = time.monotonic()
    fracs = list(fracs)

    for frac in fracs:
        step = lr
        reject_streak = 0
        frac_start_best = best_score

        while (time.monotonic() - t0) < time_limit:
            c = np.maximum(fast_convolve(a), 0.0)
            if c.size == 0:
                break
            mc = float(np.max(c))
            if not np.isfinite(mc) or mc <= 0.0:
                break

            k = max(1, int(len(c) * float(frac)))
            if k >= len(c):
                idx = np.arange(len(c))
                vals = c
            else:
                part = np.argpartition(c, len(c) - k)
                idx = part[len(c) - k:]
                vals = c[idx]

            temp = max(mc * 1e-3, 1e-30)
            ve = (vals - mc) / temp
            ve = np.clip(ve, -60.0, 0.0)
            w = np.zeros(len(c), dtype=np.float64)
            w[idx] = np.exp(ve)
            sw = float(w.sum())
            if (not np.isfinite(sw)) or sw <= 1e-300:
                w[int(np.argmax(c))] = 1.0
                sw = 1.0
            w = w / sw

            H = cross_corr(a, w)
            S = float(np.dot(a, H))
            Hc = H - S
            scale = float(np.sqrt(np.dot(a, Hc * Hc)))
            if (not np.isfinite(scale)) or scale <= 1e-15:
                break

            loga_new = loga - step * Hc / scale
            a_trial = softmax(loga_new)
            sc_trial = score_local(a_trial)
            if not np.isfinite(sc_trial):
                break

            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()

            if sc_trial <= current_score + 1e-12:
                a = a_trial
                loga = loga_new
                current_score = sc_trial
                reject_streak = 0
                if sc_trial < frac_start_best - 1e-12:
                    step = min(step * 1.02, 0.25)
            else:
                step *= 0.72
                reject_streak += 1
                if step < 1e-5 or reject_streak > 900:
                    break

    return best_a


def optimize_sequence(a0, time_limit, name="stage", do_lbgfs=True, do_dual=True):
    a = normalize_nonnegative(a0)
    best = a.copy()
    best_score = score_local(a)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))

    def _try(fn, label, **kwargs):
        global best, best_score
        try:
            res = fn(**kwargs)
            if res is not None:
                best, best_score = update_best(best, best_score, res, label)
        except Exception:
            pass

    rem = deadline - time.monotonic()
    if rem > 15:
        t_adam = min(rem * 0.32, deadline - time.monotonic())
        _try(adam_lse_optimize, name + "_adam", a0=best, time_limit=t_adam)

    rem = deadline - time.monotonic()
    if do_lbgfs and rem > 25 and best.size <= 5000:
        t_lb = min(rem * 0.28, deadline - time.monotonic(), 500.0)
        if t_lb > 20:
            _try(lbgfs_lse_optimize, name + "_lbgfs", a0=best, time_limit=t_lb)

    rem = deadline - time.monotonic()
    if do_dual and rem > 30:
        t_dual = min(rem * 0.25, deadline - time.monotonic(), 260.0)
        if t_dual > 20:
            _try(dual_acm_optimize, name + "_dual", a0=best, time_limit=t_dual)

    rem = deadline - time.monotonic()
    if rem > 12:
        t_pol = min(rem * 0.85, deadline - time.monotonic(), 360.0)
        _try(final_polish, name + "_polish", a0=best, time_limit=t_pol)

    return best


def propose_candidate(seed=42, budget_s=1800.0):
    t_start = time.monotonic()
    deadline = t_start + max(5.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) + 12345)

    best_seq = None
    best_score = float("inf")

    # ---- Load parent construction if present ----
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            sp = float(p.sum())
            if sp > 1e-12:
                best_seq = p / sp
                best_score = score_local(best_seq)
                print(f"Parent loaded: n={best_seq.size}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(1200, rng, 1)
        best_score = score_local(best_seq)
        print(f"No parent; using structured init: score={best_score:.10f}")

    n_initial = int(best_seq.size)

    # Quick local comparison of several structured initializations at same length.
    if deadline - time.monotonic() > 20:
        for mode in range(1, 14):
            if time.monotonic() + 5 > deadline:
                break
            cand = structured_init(n_initial, rng, mode)
            sc = score_local(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"Structured mode {mode} improved to {best_score:.10f}")

    # ---- Multi-length plans ----
    length_candidates = [n_initial]
    for factor, minn in [(1.5, 1536), (2.0, 2048), (3.0, 3072), (4.0, 4096)]:
        nn = int(max(float(minn), float(n_initial) * float(factor)) / 256.0) * 256
        if nn > 4096:
            nn = 4096
        if nn < 128:
            nn = 128
        length_candidates.append(nn)

    seen = []
    for x in length_candidates:
        isnew = True
        for y in seen:
            if abs(int(x) - int(y)) <= 255:
                isnew = False
                break
        if isnew:
            seen.append(int(x))

    first = length_candidates[0]
    ordered = [first]
    for y in sorted(seen):
        if y != first:
            ordered.append(y)

    m = len(ordered)
    if m == 1:
        fracs=[1.0]
    elif m == 2:
        fracs=[0.60, 0.40]
    elif m == 3:
        fracs=[0.45, 0.30, 0.25]
    elif m == 4:
        fracs=[0.38, 0.27, 0.21, 0.14]
    else:
        fracs=[]
        total_w=0.0
        for i in range(m):
            w = 1.0 / (1 + 0.8 * i)
            fracs.append(w)
            total_w += w
        for i in range(m):
            fracs[i] = fracs[i] / total_w

    total_budget = max(0.0, deadline - time.monotonic())
    allocs = [total_budget * f for f in fracs]

    for idx, nlen in enumerate(ordered):
        if time.monotonic() + 25 > deadline:
            break
        rem = deadline - time.monotonic()
        if rem <= 0:
            break
        t_alloc = min(float(allocs[idx]), rem)
        if t_alloc < 25:
            break

        if idx == 0:
            init = best_seq.copy()
        else:
            init = interpolate_sequence(best_seq, nlen)

        do_lb = (nlen <= 4096) and (t_alloc > 60)
        do_d = t_alloc > 90
        tag = f"len_{nlen}"
        res = optimize_sequence(init, t_alloc, name=tag, do_lbgfs=do_lb, do_dual=do_d)
        best_seq, best_score = update_best(best_seq, best_score, res, tag)

    # ---- A few perturbed multi-start refinements at the best length ----
    rem = deadline - time.monotonic()
    if rem > 160:
        nbest = best_seq.size
        attempts = 3
        for pi in range(attempts):
            if time.monotonic() + 60 > deadline:
                break
            base = best_seq
            if pi == 0:
                pert = base + 0.010 * rng.standard_normal(nbest)
            elif pi == 1:
                x = np.linspace(0.0, 2.0 * np.pi, nbest)
                pert = base + 0.018 * np.cos(3.0 * x) + 0.012 * np.sin(5.0 * x)
            else:
                pert = structured_init(nbest, rng, 4 + pi)
            pert = normalize_nonnegative(pert)
            t = max(60.0, min(180.0, (deadline - time.monotonic()) / float(attempts - pi + 1)))
            res = optimize_sequence(pert, t, name=f"pert_{pi}", do_lbgfs=(t > 80), do_dual=(t > 130))
            best_seq, best_score = update_best(best_seq, best_score, res, f"pert_{pi}")

    # ---- Final global polish ----
    rem = deadline - time.monotonic()
    if rem > 20:
        res = final_polish(
            best_seq,
            min(150.0, rem * 0.9),
            fracs=(0.10, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008),
            lr=0.05
        )
        best_seq, best_score = update_best(best_seq, best_score, res, "final_polish")

    # Ensure a valid finite return even if something went wrong.
    if best_seq is None or best_seq.size == 0:
        best_seq = structured_init(1200, rng, 1)
        best_score = score_local(best_seq)

    out = []
    for x in best_seq:
        xf = float(x)
        if not np.isfinite(xf):
            xf = 0.0
        if xf < 0.0:
            xf = 0.0
        if xf > 1000.0:
            xf = 1000.0
        out.append(xf)

    # Optional external validation.
    try:
        ext = evaluate_sequence(out)
        if np.isfinite(ext):
            best_score = float(ext)
    except Exception:
        pass

    print(f"FINAL returned: local_score={best_score:.10f}, n={len(out)}")
    return out
