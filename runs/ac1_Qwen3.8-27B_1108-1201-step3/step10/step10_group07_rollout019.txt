import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lp_best_response(a, solver='ECOS'):
    """
    Solve LP: min M s.t. conv(a,b) <= M*1, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    # Build convolution matrix: T[k, j] = a[j] if 0 <= k-j < n, else 0
    # T is (2n-1) x n
    # T[k, j] = a[j] for j in [max(0, k-n+1), min(n-1, k)]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating_anderson(a0, time_limit, solver='ECOS', verbose=True, m_anderson=5):
    """
    LP best-response alternating with Anderson acceleration.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    
    # Anderson acceleration history
    history_x = []  # recent iterates
    history_g = []  # corresponding g(x) values
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # Solve LP: best response b* to current a
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            stagnation += 1
            if stagnation > 5:
                break
            continue
        
        # Evaluate b_new
        score_b = score_normalized(b_new)
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Anderson acceleration update
        if it <= m_anderson:
            # Just use the direct iterate
            a = b_new.copy()
        else:
            # Maintain history
            history_x.append(a.copy())
            history_g.append(b_new.copy())
            if len(history_x) > m_anderson:
                history_x.pop(0)
                history_g.pop(0)
            
            m = len(history_x)
            # Compute residuals r_k = x_k - g(x_k)
            residuals = np.array([history_x[i] - history_g[i] for i in range(m)])
            # Solve min ||sum alpha_i * r_i||^2 s.t. sum alpha_i = 1
            # This is: min ||G * alpha||^2 s.t. 1^T alpha = 1
            # where G[i] = r_i
            # Using the standard Anderson update:
            # Gamma = R^T R + beta * I, solve Gamma * alpha = R^T * 1
            R = residuals  # shape (m, n)
            beta = 0.1  # regularization
            Gamma = R @ R.T + beta * np.eye(m)
            rhs = R @ np.ones(m)
            try:
                alpha = np.linalg.solve(Gamma, rhs)
            except np.linalg.LinAlgError:
                alpha = np.ones(m) / m
            
            # Anderson extrapolated point
            a_new = sum(alpha[i] * history_g[i] for i in range(m))
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            # Check if extrapolation helped
            score_an = score_normalized(a_new)
            if score_an < score_b * 1.01:
                a = a_new
            else:
                a = b_new.copy()
        
        # Evaluate current a
        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Stagnation check
        if score_a > best_score * (1 + 1e-7):
            stagnation += 1
        else:
            stagnation = 0
        
        # Random restart if stuck
        if stagnation > 20:
            noise = np.random.standard_normal(n) * 0.02
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = score_normalized(a_pert)
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0
            history_x = []
            history_g = []
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-Anderson it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def spectral_gs(a0, time_limit, verbose=True):
    """
    Gerchberg-Saxton style spectral descent.
    Alternates between frequency domain (flatten power spectrum) 
    and time domain (enforce non-negativity).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    needed = 2 * n - 1
    fft_size = 1 << (needed - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    noise_level = 0.0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Frequency domain step: flatten power spectrum
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: flat spectrum (geometric mean of non-DC)
        mean_power = np.mean(power[1:])
        target_mag = np.sqrt(mean_power) * np.ones(len(A))
        target_mag[0] = A[0].real  # preserve DC
        
        # Mix with current magnitude (damping)
        alpha_mix = min(1.0, 0.3 + 0.1 * (it / 50.0))
        current_mag = np.abs(A)
        new_mag = (1 - alpha_mix) * current_mag + alpha_mix * target_mag
        new_mag[0] = A[0].real
        
        # New A with updated magnitude, keeping phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Time domain step: enforce non-negativity
        a_new = irfft(A_new, fft_size)[:n]
        
        # Add small noise to escape local minima
        if noise_level > 0:
            a_new += np.random.standard_normal(n) * noise_level
        
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        a = a_new
        score = score_normalized(a)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            # Decrease noise
            noise_level = max(0, noise_level * 0.95)
        else:
            no_improve += 1
            if no_improve > 100:
                # Increase noise to escape
                noise_level = min(0.05, noise_level + 0.005)
                no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  GS it={it} ({elapsed:.0f}s): best={best_score:.8f} noise={noise_level:.4f}")
    
    return best_a, best_score


def coord_descent_peak(a0, time_limit, verbose=True):
    """
    Peak-focused coordinate descent.
    For each coordinate, try adjustments that reduce the maximum convolution value.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_start_M = M
        
        # Process coordinates in order of their contribution to peaks
        # First pass: all coordinates
        for i in range(n):
            if a[i] < 1e-18:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected = slice(k_start, k_end + 1)
            
            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            a_shifted = a[k_start - i: k_end - i + 1]
            conv_affected = conv[k_start: k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            # Try various multiplicative factors
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv_affected + 2.0 * delta * a_shifted
                # Fix the quadratic term
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-13:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes_arr = changes.copy()
                    changes_arr[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                    conv[k_start: k_end + 1] += changes_arr
                else:
                    conv[k_start: k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start: k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD sweep={sweeps} ({elapsed:.0f}s): M={M:.10f} score={2*n*M:.8f}")
    
    return best_a, best_score

def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    if init_type == 0:
        # Triangle
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        # U-shape
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        # Cosine window
        x = (t - center) / center
        power = rng.uniform(0.5, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Two humps
        c1 = n * rng.uniform(0.25, 0.35)
        c2 = n * rng.uniform(0.65, 0.75)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Periodic (low frequency)
        period = int(rng.integers(4, 16))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period + rng.uniform(0, 2*np.pi))
    elif init_type == 5:
        # W-shape
        a = 1.0 - 0.4 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Exponential decay from center
        a = np.exp(-np.abs(t - center) / (n * 0.3))
    elif init_type == 7:
        # Random smooth
        n_freq = 5
        freqs = rng.integers(1, 8, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = rng.uniform(0.3, 1.0, size=n_freq)
        a = np.ones(n) * 0.3
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * t / n + p)
        a = np.maximum(a, 0)
    elif init_type == 8:
        # Chirp magnitude (structured)
        # |exp(i * c * t^2 / (2*pi))| = 1, not useful
        # Instead use a "Barker-like" pattern
        period = int(n ** 0.5)
        pattern = np.array([1, 1, 1, -1, 1, -1, -1][:period])
        a = np.tile(pattern, n // period + 1)[:n]
        a = np.abs(a) * (0.5 + 0.5 * np.cos(np.pi * t / n))  # window it
        
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def find_best_solver():
    """Find the best available LP solver."""
    cached = getattr(find_best_solver, '_cached', None)
    if cached is not None:
        return cached
    
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver test {solver_name}: {elapsed:.3f}s")
                find_best_solver._cached = solver_name
                return solver_name
        except Exception as e:
            print(f"  Solver test {solver_name} failed: {e}")
            continue
    
    find_best_solver._cached = cp.ECOS
    return cp.ECOS
