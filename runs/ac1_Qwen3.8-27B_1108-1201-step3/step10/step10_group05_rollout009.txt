import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Periodically recompute to avoid drift
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
            if 2.0 * n * M < best_score:
                best_score = 2.0 * n * M
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCd {label} sweep={sweeps} ({elapsed:.0f}s): M={M:.8f}, score={2*n*M:.6f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1].copy()
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Very fine fractional moves
            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                          0.993, 0.99, 0.985, 0.98, 0.97,
                          1.001, 1.002, 1.003, 1.005, 1.007, 1.01, 1.015, 1.02, 1.03]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 30 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def lp_fp_optimal_step(a0, time_limit, solver=None, label=""):
    """LP fictitious play with golden-section optimal step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Use golden section to find optimal step
        t_opt = optimal_step_golden(a, b_new, max_iter=40)
        
        # Also try a few fixed steps for robustness
        steps_to_try = [t_opt, min(1.0, t_opt * 1.1), min(1.0, t_opt * 0.9),
                       min(1.0, 5.0 / (iters + 10.0)), 0.5, 0.3, 0.1]

        best_step = 0.0
        best_cand = None
        best_cand_score = best_score

        for st in steps_to_try:
            if st < 0 or st > 1:
                continue
            cand = (1.0 - st) * a + st * b_new
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_cand_score - 1e-15:
                best_cand_score = sc
                best_cand = cand.copy()
                best_step = st

        if best_cand is not None:
            a = best_cand
            if best_cand_score < best_score - 1e-14:
                best_score = best_cand_score
                best_a = a.copy()
        else:
            # Fallback: small step
            alpha = min(1.0, 5.0 / (iters + 10.0))
            a = (1.0 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Also try the LP solution directly
        sc_b = 2.0 * n * fast_convolve(b_new).max()
        if sc_b < best_score - 1e-14:
            best_score = sc_b
            best_a = b_new.copy()

        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP-opt {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}, t={best_step:.4f}")

        # Convergence check
        if iters > 5 and abs(M_b - best_score / (2.0 * n)) < 1e-8:
            # Close to equilibrium
            pass

    return best_a, best_score, iters

def pair_swap_optimize(a0, time_limit, label=""):
    """Optimize by transferring mass between pairs of coordinates simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    # Find peak-contributing coordinates
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        conv = fast_convolve(a)
        M = float(conv.max())
        peak_k = int(np.argmax(conv))

        # Find top contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        contributions = {}
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if j < n:
                contributions[i] = a[i] * a[j]
        
        top_contribs = sorted(contributions.keys(), key=lambda x: contributions[x], reverse=True)[:10]

        # Try pair swaps among top contributors and their neighbors
        candidates_pairs = []
        for ci in top_contribs:
            for cj in range(max(0, ci - 3), min(n, ci + 4)):
                if cj != ci and cj < n:
                    candidates_pairs.append((ci, cj))

        for (i, j) in candidates_pairs:
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue
            s_ij = a[i] + a[j]
            if s_ij < 1e-15:
                continue

            # Binary search for best split
            # Try moving mass from i to j and vice versa
            best_frac = 0.5  # current fraction for i
            best_M = M

            # Try various splits
            for frac in np.linspace(0.0, 1.0, 21):
                new_ai = frac * s_ij
                new_aj = (1.0 - frac) * s_ij
                delta_i = new_ai - a[i]
                delta_j = new_aj - a[j]
                if abs(delta_i) < 1e-18 and abs(delta_j) < 1e-18:
                    continue

                # Compute new peak (approximate: only check affected terms)
                # Affected conv indices from i: [i, i+n-1]
                # Affected conv indices from j: [j, j+n-1]
                affected_min = min(i, j)
                affected_max = max(i + n - 1, j + n - 1)
                affected_min = max(0, affected_min)
                affected_max = min(2 * n - 2, affected_max)

                new_conv_seg = conv.copy()
                # Update from i
                for k in range(max(0, i), min(2*n-1, i+n)):
                    partner = k - i
                    if 0 <= partner < n:
                        new_conv_seg[k] += delta_i * a[partner]
                        if partner == j:
                            new_conv_seg[k] += delta_i * delta_j + delta_j * a[i]
                # Update from j (careful about double counting when partner==i)
                for k in range(max(0, j), min(2*n-1, j+n)):
                    partner = k - j
                    if 0 <= partner < n:
                        if partner != i:
                            new_conv_seg[k] += delta_j * a[partner]
                        # else already handled above

                new_M = float(new_conv_seg.max())
                if new_M < best_M - 1e-15:
                    best_M = new_M
                    best_frac = frac

            if best_frac != 0.5 and best_M < M - 1e-14:
                new_ai = best_frac * s_ij
                new_aj = (1.0 - best_frac) * s_ij
                a[i] = new_ai
                a[j] = new_aj
                conv = fast_convolve(a)
                M = float(conv.max())
                if 2.0 * n * M < best_score - 1e-14:
                    best_score = 2.0 * n * M
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score

def simulated_annealing(a0, time_limit, label="", n_restarts=5):
    """Simulated annealing with structured moves for escaping local minima."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    T_init = 0.001
    T_min = 1e-8
    cooling = 0.995

    T = T_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1
        cur_score = 2.0 * n * fast_convolve(a).max()

        # Choose a random move type
        move_type = rng.random()
        
        if move_type < 0.4:
            # Single coordinate change
            i = rng.integers(0, n)
            delta = rng.standard_normal() * a[i] * 0.5
            a_new = a.copy()
            a_new[i] = max(0.0, a[i] + delta)
        elif move_type < 0.7:
            # Block change: modify a small block
            block_size = rng.integers(2, 6)
            start = rng.integers(0, max(1, n - block_size))
            a_new = a.copy()
            block_sum = a[start:start+block_size].sum()
            if block_sum > 1e-15:
                noise = np.abs(rng.standard_normal(block_size)) + 0.1
                new_block = noise / noise.sum() * block_sum
                a_new[start:start+block_size] = new_block
        elif move_type < 0.85:
            # Pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                a_new = a.copy()
                s_ij = a[i] + a[j]
                frac = rng.random()
                a_new[i] = frac * s_ij
                a_new[j] = (1.0 - frac) * s_ij
            else:
                a_new = a.copy()
        else:
            # Peak-targeted: reduce a peak contributor
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)
            idxs = list(range(i_start, i_end + 1))
            i = idxs[rng.integers(0, len(idxs))]
            a_new = a.copy()
            a_new[i] = a[i] * 0.8
            # Redistribute to random other position
            j = rng.integers(0, n)
            a_new[j] += a[i] * 0.2

        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            T *= cooling
            continue
        a_new = a_new / sn

        new_score = 2.0 * n * fast_convolve(a_new).max()
        delta_score = new_score - cur_score

        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T, 1e-15)):
            a = a_new
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve = 1
        else:
            no_improve += 1

        T *= cooling
        if no_improve > 500:
            T *= 0.8
            no_improve = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA {label} it={it} ({elapsed:.0f}s): T={T:.2e}, best={best_score:.6f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_init(n, rng, mode=0):
    """Generate structured initializations based on spectral properties."""
    if mode == 0:
        # Cosine-based
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x / n) + 0.1 * np.cos(4 * np.pi * x / n)
        a = np.maximum(a, 0)
    elif mode == 1:
        # Sinc-like
        x = np.arange(n) - n / 2
        a = np.abs(np.sinc(x / (n * 0.3)))
        a = np.maximum(a, 0.01)
    elif mode == 2:
        # Bimodal
        x = np.arange(n)
        center = n / 2
        a = np.exp(-0.5 * ((x - center * 0.3) / (n * 0.1))**2) + np.exp(-0.5 * ((x - center * 1.7) / (n * 0.1))**2)
        a = np.maximum(a, 0)
    elif mode == 3:
        # Random with low-pass smoothing
        a = np.abs(rng.standard_normal(n))
        # Simple low-pass
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if j < n:
                pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * np.abs(noise))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

                break
            t8 = min(rem * 0.15, 50.0)
            init8 = spectral_init(n_best, rng, mode=mode)
            a8, s8, _ = smart_coord_descent(init8, t8, label=f"p8m{mode}")
            best_seq, best_score = check_and_update(a8, best_seq, best_score, f"p8m{mode}")

    # === Phase 9: Final intensive polishing ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t9 = min(rem * 0.5, 120.0)
            a9, s9 = fine_coord_descent(best_seq, t9, label="p9")
            best_seq, best_score = check_and_update(a9, best_seq, best_score, "p9_fine")

        rem = deadline - time.monotonic()
        if rem > 20:
            t9b = min(rem * 0.6, 60.0)
            a9b, s9b, _ = smart_coord_descent(best_seq, t9b, label="p9b")
            best_seq, best_score = check_and_update(a9b, best_seq, best_score, "p9b_scd")

        rem = deadline - time.monotonic()
        if rem > 15:
            a9c, s9c = fine_coord_descent(best_seq, rem * 0.7, label="p9c")
            best_seq, best_score = check_and_update(a9c, best_seq, best_score, "p9c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
