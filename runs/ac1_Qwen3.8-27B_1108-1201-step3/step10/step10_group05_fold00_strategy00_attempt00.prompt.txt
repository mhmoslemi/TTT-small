def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def nesterov_smoothed_md(a0, time_limit, beta_init=20.0, lr_init=0.01, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov momentum
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        # Compute smooth gradient via log-sum-exp
        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient: grad[i] = 2 * sum_k weights[k] * y_proj[k-i] (for valid k-i)
        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 80:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 30 == 0:
            beta *= 1.01

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.9
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.2):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 5):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.2))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def subgradient_nesterov(a0, time_limit, label=""):
    """Subgradient method with Nesterov acceleration on the exact (non-smooth) objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        # Average over near-peak indices for a more stable subgradient
        threshold = M - 0.01 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        # Gradient step + projection
        a_new = a - step * grad
        a_new = simplex_project(a_new)

        # Nesterov average (for convergence guarantee)
        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        # Evaluate
        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score
