def lse_lbgfs_softmax(a0, time_limit, T_stages=None):

    z = np.log(a0 + 1e-300)  # Initialize from a0
    t0 = time.monotonic()

        z_shift = z - z.max()
        e = np.exp(z_shift)
        return e / e.sum()
    
    for T in T_stages:
        def func(z):
            a = softmax_val(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            lse = max_b + np.log(exp_b.sum())
            return 2.0 * n * lse / T
        
        def grad(z):
            a = softmax_val(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()

            # Using FFT:

            # (standard softmax gradient)
            S = np.dot(a, grad_a)
            grad_z = a * (grad_a - S)

        z = res.x

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast LogSumExp L-BFGS with FFT-accelerated gradients and aggressive cooling
2. LP alternation for refinement
3. Peak-equalization Newton steps
4. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def lse_obj_and_grad_z(z, n, T, fft_size):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. z (log-param)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    lse_val = max_b + np.log(exp_b.sum())
    obj = 2.0 * n * lse_val / T
    
    # Gradient w.r.t. a: sum_k w_k * db_k/da
    w = exp_b / exp_b.sum()
    # db_k/da_i ≈ 2*a[k-i] (ignoring the factor-1 correction)
    # grad_a[i] = sum_k w[k] * 2*a[k-i] = 2 * correlate(w, a)[i]
    # Using FFT: correlate(w, a) = ifft(fft(w) * conj(fft(a))) ... 
    # Actually let me use the convolution trick:
    # grad_a[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * (w convolved with a_reversed)[i + n - 1]
    
    W = rfft(w, fft_size)
    Ar = rfft(a[::-1].copy(), fft_size)
    c = irfft(W * Ar, fft_size)
    grad_a = 2.0 * c[n-1:2*n-1]
    
    # Softmax chain rule: grad_z = a * (grad_a - dot(a, grad_a))
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S) * (2.0 * n / T)
    
    return obj, grad_z

def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=500):
    """LogSumExp L-BFGS with FFT gradients."""
    n = len(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30, 20, 15, 10, 7, 5, 3.5, 2.5, 2, 1.5, 1, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01]
    
    best_z = z.copy()
    best_score = float('inf')
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func_jac(z_local):
            obj, grad = lse_obj_and_grad_z(z_local, n, T, fft_size)
            return obj, grad
        
        try:
            res = minimize(func_jac, z, jac=True, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_stage, 'ftol': 1e-18, 'gtol': 1e-12})
            z = res.x
        except Exception:
            pass
        
        # Track best
        a_check = softmax_from_z(z)
        sc = evaluate_local(a_check)
        if sc < best_score:
            best_score = sc
            best_z = z.copy()
    
    return softmax_from_z(best_z)

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except:

    return cp.ECOS

def lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    import cvxpy as cp
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except:
        pass
    return None

    """Nesterov-accelerated LP alternation."""
    n = len(a0)
    a = np.maximum(a0, 0)

    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = lp_best_response(y, solver)
        if b_new is None:
            break
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Iterated average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        for candidate in [a, y]:
            sc = evaluate_local(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
    
    return best_a

def peak_coord_descent(a0, time_limit, n_steps=30000):
    """Coordinate descent targeting top peaks."""
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_peak = np.argmax(b)
        
        # Find top contributors to the max peak
        js = max(0, max_peak - n + 1)
        je = min(n - 1, max_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_peak - idx]
        top_pos = idx[np.argmax(contribs)]
        
        # Try reducing
        improved = False
        for factor in [0.99, 0.98, 0.97, 0.95]:
            a_trial = a.copy()
            a_trial[top_pos] *= factor
            diff = a[top_pos] * (1 - factor)
            if diff < 1e-15:
                break
            mask = np.ones(n, dtype=bool)
            mask[top_pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                break
            a_trial[mask] = others + diff * others / others_sum
            a_trial = a_trial / a_trial.sum()
            sc = evaluate_local(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        if not improved and step % 10000 == 9999:
            # Random restart of search direction
            pass
    
    return best_a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
