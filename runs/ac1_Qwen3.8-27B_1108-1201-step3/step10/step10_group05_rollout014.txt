    """Gerchberg-Saxton style spectral initialization."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    for it in range(n_iter):
        fft_size = 1 << (2 * n - 1).bit_length()
        F = rfft(a, fft_size)
        mag = np.abs(F)
        phase = np.angle(F)
        target = 1.0 / (2.0 * n)
        target_mag = np.sqrt(target)
        new_mag = mag * (target_mag / (mag + 1e-30)) ** alpha
        new_F = new_mag * np.exp(1j * phase)
        a_new = irfft(new_F, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if it % 50 == 0 and it > 0:
            score = 2.0 * n * fast_convolve(a_new).max()
            # print(f"    GS iter {it}: score={score:.6f}")
        a = a_new
    return a

def spectral_gs_refine(a, n_iter=100, alpha=0.3):
    """Refine an existing sequence using GS iterations."""
    n = len(a)
    a = a.copy()
    for it in range(n_iter):
        fft_size = 1 << (2 * n - 1).bit_length()
        F = rfft(a, fft_size)
        mag = np.abs(F)
        phase = np.angle(F)
        s = a.sum()
        target = s * s / (2.0 * n)
        target_mag = np.sqrt(target)
        new_mag = mag * (target_mag / (mag + 1e-30)) ** alpha
        new_F = new_mag * np.exp(1j * phase)
        a_new = irfft(new_F, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score_new = 2.0 * n * fast_convolve(a_new).max()
        score_old = 2.0 * n * fast_convolve(a).max()
        if score_new > score_old * 1.01:
            alpha *= 0.9
        a = a_new
        if alpha < 0.01:
            break
    return a

def freq_domain_subgradient(a, n_peaks=5, lr=0.01):
    """Frequency-domain subgradient step targeting autocorrelation peaks."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.95
    peak_positions = np.where(conv >= peak_threshold)[0]
    peak_positions = peak_positions[:n_peaks]
    
    fft_size = 1 << (2 * n - 1).bit_length()
    F = rfft(a, fft_size)
    mag = np.abs(F)
    phase = np.angle(F)
    
    for k in peak_positions:
        cos_vals = np.cos(2.0 * np.pi * np.arange(fft_size // 2 + 1) * k / fft_size)
        reduction = lr * cos_vals
        reduction = np.maximum(reduction, 0)
        mag = mag * (1.0 - reduction * 0.1)
    
    new_F = mag * np.exp(1j * phase)
    a_new = irfft(new_F, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with smart 1D candidate evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.5 * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 300:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps

def pairwise_coord_descent(a0, time_limit, label=""):
    """Coordinate descent with pairwise coordinated moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            if i + 1 >= n:
                break
            aj = a[i + 1]
            if aj < 1e-15:
                continue

            k_start = i
            k_end = min(i + n, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1].copy()
            len_affected = len(affected)

            best_score_pair = best_score
            best_ai = ai
            best_aj = aj

            for di_frac in [-0.9, -0.7, -0.5, -0.3, -0.1, 0.0, 0.1, 0.3, 0.5, 0.7, 0.9]:
                for dj_frac in [-0.9, -0.7, -0.5, -0.3, -0.1, 0.0, 0.1, 0.3, 0.5, 0.7, 0.9]:
                    di = di_frac * ai
                    dj = dj_frac * aj
                    new_ai = ai + di
                    new_aj = aj + dj
                    if new_ai < -1e-15 or new_aj < -1e-15:
                        continue
                    new_ai = max(new_ai, 0.0)
                    new_aj = max(new_aj, 0.0)

                    d_ai = new_ai - ai
                    d_aj = new_aj - aj
                    if abs(d_ai) < 1e-18 and abs(d_aj) < 1e-18:
                        continue

                    for k_idx in range(len_affected):
                        k = k_start + k_idx
                        contrib_i = 0.0
                        contrib_j = 0.0
                        quad_term = 0.0
                        if 0 <= k - i < n:
                            contrib_i = d_ai * a[k - i]
                            if k - i == i:
                                quad_term += d_ai * d_ai
                        if 0 <= k - (i + 1) < n:
                            contrib_j = d_aj * a[k - (i + 1)]
                            if k - (i + 1) == i + 1:
                                quad_term += d_aj * d_aj
                        if i + 1 <= k - i - 1 and 0 <= k - i - 1 < n and 0 <= k - i < n:
                            if k - (i + 1) == i and k - i == i + 1:
                                quad_term += 2.0 * d_ai * d_aj

                        affected[k_idx] += contrib_i + contrib_j + quad_term

                    new_M = max(unaff_max, float(affected.max()))
                    new_score = 2.0 * n * new_M
                    if new_score < best_score_pair - 1e-12:
                        best_score_pair = new_score
                        best_ai = new_ai
                        best_aj = new_aj
                    affected[k_start - k_start:k_end + 1 - k_start] -= d_ai * (a[(max(0, k_start - i)):min(n, k_end + 1 - i)] + ai) if False else 0
                    affected[:] = conv[k_start:k_end + 1]

            if best_ai != ai or best_aj != aj:
                a[i] = best_ai
                a[i + 1] = best_aj
                conv = fast_convolve(a)
                M = float(conv.max())
                if 2.0 * n * M < best_score - 1e-14:
                    best_score = 2.0 * n * M
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

            quad_idx = 2 * i - k_start if has_quad else -1

                if delta < -ai:

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    b = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        b = b_new

        alpha = min(1.0, 10.0 / (iters + 5))
        a_new = (1 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")

    return best_a, best_score, iters

def simulated_annealing_step(a, rng, temperature):
    """Perturb a sequence with temperature-controlled noise."""
    n = len(a)
    noise = rng.standard_normal(n) * temperature
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def spectral_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    F = rfft(a, fft_size)
    mag = np.abs(F)
    phase = np.angle(F)
    
    noise = rng.standard_normal(mag.shape) * strength
    new_mag = mag * (1.0 + noise)
    new_mag = np.maximum(new_mag, 0)
    
    new_F = new_mag * np.exp(1j * phase)
    a_new = irfft(new_F, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    return best_seq, best_score

def nesterov_smoothed_md(a0, time_limit, beta_init=30.0, lr_init=0.008, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 30 == 0:
            beta *= 1.01

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

    print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")
    a4, s4, _ = lp_fictitious_play(best_seq, t4, solver=solver, label="p4")
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_lpfp")

    # === Phase 5: Simulated annealing with spectral perturbation ===
    print(f"\n=== Phase 5: SA + Spectral Perturb (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    t5 = min(max(rem * 0.4, 30), 200.0)
    print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
    
    sa_start = time.monotonic()
    temp_start = 0.05
    temp_end = 0.005
    sa_iter = 0
    current = best_seq.copy()
    current_score = best_score
    
    while time.monotonic() - sa_start < t5:
        sa_iter += 1
        progress = (time.monotonic() - sa_start) / t5
        temp = temp_start * (temp_end / temp_start) ** progress
        
        if sa_iter % 3 == 0:
            perturbed = spectral_perturb(current, rng, strength=temp)
        else:
            perturbed = simulated_annealing_step(current, rng, temp)
        
        p_score = 2.0 * len(current) * fast_convolve(perturbed).max()
        
        if p_score < current_score or rng.random() < np.exp(-(p_score - current_score) / (temp * 0.1)):
            current = perturbed
            current_score = p_score
        
        if p_score < best_score - 1e-10:
            best_seq, best_score = check_and_update(perturbed, best_seq, best_score, f"p5_sa_{sa_iter}")
        
        if sa_iter % 50 == 0:
            print(f"    SA iter={sa_iter} temp={temp:.4f}: current={current_score:.6f}, best={best_score:.6f}")
    
    # Quick CD on SA result
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t5b = min(rem * 0.3, 60.0)
        a5b, s5b, _ = smart_coord_descent(current, t5b, label="p5_cd")
        best_seq, best_score = check_and_update(a5b, best_seq, best_score, "p5_cd")

    # === Phase 6: Nesterov smoothed MD ===
    print(f"\n=== Phase 6: Nesterov MD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 250
    t6 = min(max(rem * 0.4, 30), 150.0)
    print(f"  Budget: {t6:.0f}s")
    a6, s6 = nesterov_smoothed_md(best_seq, t6, beta_init=40.0, lr_init=0.005, label="p6")
    best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6_nmd")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best]
        for alt in [768, 1024, 1280, 1536, 2048]:
            if alt != n_best and 512 <= alt <= 2560:
                alt_lengths.append(alt)
        alt_lengths = alt_lengths[:4]

        for n_alt in alt_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t7 = min(rem * 0.4, 80.0)
            if n_alt == n_best:
                init_alt = best_seq.copy()
            else:
                init_alt = interpolate_seq(best_seq, n_alt)
            a7, s7, _ = smart_coord_descent(init_alt, t7, label=f"p7_n{n_alt}")
            best_seq, best_score = check_and_update(a7, best_seq, best_score, f"p7_n{n_alt}")

    # === Phase 8: Another LP FP round on best ===
    print(f"\n=== Phase 8: Final LP FP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 120
    if rem > 60:
        t8 = min(rem * 0.6, 120.0)
        a8, s8, _ = lp_fictitious_play(best_seq, t8, solver=solver, label="p8")
        best_seq, best_score = check_and_update(a8, best_seq, best_score, "p8_lpfp")

    # === Phase 9: Final intensive polishing ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t9 = min(rem * 0.5, 60.0)
            a9, s9 = fine_coord_descent(best_seq, t9, label="p9")
            best_seq, best_score = check_and_update(a9, best_seq, best_score, "p9_fine")

        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a9b, s9b, _ = smart_coord_descent(best_seq, rem * 0.6, label="p9b")
            best_seq, best_score = check_and_update(a9b, best_seq, best_score, "p9b_scd")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a9c, s9c = fine_coord_descent(best_seq, rem * 0.8, label="p9c")
            best_seq, best_score = check_and_update(a9c, best_seq, best_score, "p9c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
