import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_safe(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    size = 2 * n - 1
    g = np.zeros(n)
    # For each k with weight w_k, add w_k * 2 * a_{k-i} for valid i
    # Vectorized: g_i = 2 * sum_k w_k * a_{k-i} (where k-i in [0,n-1])
    # This is a cross-correlation of w with a
    for k in range(size):
        if abs(weights[k]) < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * weights[k] * a[k - idx]
    return g


def solve_dual_lp(a, max_iters=50):
    """Solve the LP to find optimal dual weights w that minimize ||gradient||.
    The dual of 'min max_k conv(a,b) s.t. b>=0, sum(b)=1' gives us w.
    But actually, for the subgradient question, we want the w that gives
    the steepest descent direction.
    
    Actually, the optimal subgradient for mirror descent is the one that
    minimizes the dual function. For the max function, any convex combination
    of active gradients is a valid subgradient. We want the one that gives
    the best (most negative) decrease.
    
    A simpler approach: w_k proportional to (a*a)_k^p for large p (approximates 
    the indicator of the max). Or solve the LP directly.
    
    For now, let's use: w_k = (a*a)_k^P / sum((a*a)_k^P) for large P.
    This is the "entropic" approximation to the indicator of the argmax.
    """
    n = len(a)
    b = fast_convolve(a)
    # Use power weighting (soft argmax)
    b_pos = np.maximum(b, 0)
    # Use a high power to concentrate on peaks
    b_max = b.max()
    if b_max < 1e-30:
        return np.ones(2*n-1) / (2*n-1)
    # Geometric weighting: w_k ~ exp(beta * b_k)
    beta = 50.0  # concentration parameter
    b_shifted = b - b_max
    w = np.exp(beta * b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.ones(2*n-1) / (2*n-1)
    return w / w_sum


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_for_dual_weights(a, solver):
    """Solve LP to find optimal dual weights for the best response problem.
    
    The LP is: maximize sum(b) subject to conv(a,b)_k <= 1 for all k, b >= 0.
    The dual gives us weights w_k such that the optimal b satisfies
    b_i is positive only where sum_k w_k * a_{k-i} = 1 (complementary slackness).
    
    Actually, let me think about this differently. We want to find the w that
    gives the best subgradient direction for minimizing max(conv(a,a)).
    
    The subgradient of max_k (a*a)_k is: g_i = sum_k w_k * 2*a_{k-i}
    where w is any probability on the argmax set.
    
    For mirror descent, we want to maximize the decrease, which means we want
    the subgradient that points most "outward" from the current point.
    
    An alternative: solve the LP that gives the best response b (minimizing
    max conv(a,b)), and the dual of that LP gives us w. Then use w to compute
    the subgradient for a.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    
    # Build constraint matrix implicitly using cvxpy's convolve-like operations
    # conv(a, b)_k = sum_j a_j * b_{k-j}
    # We need: for all k, sum_j a_j * b_{k-j} <= M
    # This is: A @ b <= M * 1 where A is the convolution matrix
    
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            # Extract dual variables
            # The dual of min M s.t. T@b <= M, sum(b)=1, b>=0
            # Dual: max sum(b_dual) s.t. ... 
            # Actually, let's just use the primal solution to identify active constraints
            b_val = np.asarray(b_var.value, dtype=np.float64)
            M_val = M_var.value
            # Active constraints: those where T@b is close to M
            conv_ab = T @ b_val
            active = conv_ab >= M_val * 0.999
            # Uniform weights on active constraints
            w = np.zeros(size)
            w[active] = 1.0 / active.sum()
            return w
    except Exception:
        pass
    return None


def mirror_descent_v2(a0, time_limit, eta0=0.06, beta_concentrate=50.0,
                      use_lp_dual=False, solver=None, nesterov=True):
    """
    Improved mirror descent with concentration-based dual weights and backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution at Nesterov point
        b = fast_convolve(y)
        max_b = b.max()
        
        # Compute dual weights using concentration
        b_shifted = b - max_b
        w = np.exp(beta_concentrate * b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum
        
        # Compute subgradient
        g = compute_subgradient_full(y, w)
        
        # Normalize subgradient for mirror descent
        # In mirror descent on simplex with entropy, the update is:
        # a_new_i = a_i * exp(-eta * g_i) / Z
        # We need to normalize g so that the step size is meaningful
        g_centered = g - np.dot(g, y)  # Remove the component along y (for normalization invariance)
        
        # Backtracking line search
        a_trial = a.copy()
        eta_trial = eta
        current_score = 2.0 * n * b.max()  # Score at y point
        actual_score = evaluate_local(a)
        
        for bt in range(20):
            log_a = np.log(np.maximum(a, 1e-300))
            log_a_new = log_a - eta_trial * (g_centered - g_centered.min())
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new = a_new / a_new.sum()
            
            score_new = evaluate_local(a_new)
            if score_new < actual_score - 1e-15:
                a_trial = a_new
                break
            eta_trial *= 0.6
            if eta_trial < 1e-8:
                a_trial = a_new
                break
        
        a = a_trial
        
        # Nesterov update
        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a.copy()
            else:
                y = y / ys
        
        # Update best
        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt eta
        if no_improve == 100:
            eta *= 0.7
        elif no_improve == 200:
            eta *= 0.7
        elif no_improve >= 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = eta0
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD2 iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def admm_optimize(a0, time_limit, rho0=10.0, n_inner=5):
    """
    ADMM for minimizing max(conv(a,a)) subject to a>=0, sum(a)=1.
    
    Variables: a (on simplex), b (convolution values), M (max value)
    Constraint: conv(a) = b, b_k <= M
    
    Simplified 2-block ADMM:
    Block 1: (a, M) - minimize M + (rho/2)||conv(a) - b + u||^2 + (rho/2)||b_clipped - b + u_b||^2
    Block 2: b - minimize (rho/2)||conv(a) - b + u||^2 + (rho/2)||b - b_clip||^2
             where b_clip = min(b, M) (projection onto {x: x_k <= M})
    
    Actually, let me use a simpler formulation:
    
    minimize M
    subject to a >= 0, sum(a) = 1, conv(a) <= M*1
    
    ADMM with splitting:
    x = a (simplex constraint)
    z = M (scalar)
    Coupling: conv(a) <= z * 1
    
    Augmented Lagrangian:
    L = z + (rho/2)||max(conv(a) - z, 0) + u||^2 ... 
    """
    # Let me use a different, cleaner formulation.
    # 
    # minimize f(a) + g(v) subject to h(a) = v
    # where f(a) = 0 if a on simplex else inf
    #       g(v) = max(v) (the max of the vector v)
    #       h(a) = conv(a)
    #
    # ADMM:
    # a-update: min (rho/2)||conv(a) - v + u||^2 s.t. a on simplex
    # v-update: min max(v) + (rho/2)||conv(a) - v + u||^2
    # u-update: u = u + conv(a) - v
    #
    # The v-update has a clean solution (water-filling).
    # The a-update is non-convex but we use projected gradient.
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    size = 2 * n - 1
    v = fast_convolve(a).copy()
    u = np.zeros(size)
    rho = rho0
    best_a = a.copy()
    best_score = 2.0 * n * v.max()
    t0 = time.monotonic()
    
    # Precompute convolution matrix structure for efficiency
    # We'll use FFT for convolutions
    
    for outer in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # V-update: minimize max(v_new) + (rho/2)||conv(a) + u - v_new||^2
        c = fast_convolve(a) + u
        # Solution: v_new_k = min(c_k, M*) where M* satisfies sum_{k:c_k>M*}(c_k - M*) = 1/rho
        # Use bisection
        c_sorted = np.sort(c)[::-1]
        # M* is such that sum of (c_k - M*) for c_k > M* equals 1/rho
        # Let m = number of elements > M*. Then M* = (sum of top m elements - 1/rho) / m
        # We need to find the right m.
        running_sum = np.cumsum(c_sorted)
        # For m elements: M_m = (running_sum[m-1] - 1/rho) / m
        # We need M_m < c_sorted[m-1] (the m-th largest) and M_m >= c_sorted[m] (the (m+1)-th)
        # Binary search on m
        m_lo, m_hi = 1, size
        m_star = 1
        while m_lo <= m_hi:
            m_mid = (m_lo + m_hi) // 2
            M_mid = (running_sum[m_mid - 1] - 1.0/rho) / m_mid
            if m_mid < size:
                if M_mid >= c_sorted[m_mid]:
                    m_star = m_mid
                    m_lo = m_mid + 1
                else:
                    m_hi = m_mid - 1
            else:
                m_star = m_mid
                break
        M_opt = (running_sum[m_star - 1] - 1.0/rho) / m_star
        M_opt = max(M_opt, 0)
        v_new = np.minimum(c, M_opt)
        
        # A-update: minimize (rho/2)||conv(a) - v_new + u||^2 s.t. a >= 0, sum(a) = 1
        # Use projected gradient with a few inner iterations
        target = v_new - u
        for inner in range(n_inner):
            conv_a = fast_convolve(a)
            diff = conv_a - target
            # Gradient: 2 * cross_corr(diff, a) (by chain rule)
            # cross_corr(diff, a)_i = sum_k diff_k * a_{k-i} (with appropriate bounds)
            # This is the same as the convolution-like operation
            # grad_i = 2 * sum_k diff_k * a_{k-i} for k-i in [0,n-1]
            # = 2 * (diff * a)_i in the cross-correlation sense
            # Using FFT: cross_corr(diff, a) = irfft(conj(rfft(diff)) * rfft(a)) ... 
            # Actually, it's the convolution of diff with reversed a, or equivalently
            # cross-correlation.
            # Let's use: grad = 2 * ifft(conj(fft(diff_padded)) * fft(a_padded))[:n]
            # But we need to be careful about the definition.
            # 
            # (diff * a)_i = sum_k diff_k * a_{k-i} where a_j = 0 for j outside [0,n-1]
            # This is the cross-correlation of diff and a.
            # In terms of convolution: cross_corr(diff, a) = convolve(diff_reversed, a_reversed) reversed
            # Or: it's the "anti-convolution"
            # 
            # Simplest: use the fact that if we define the linear cross-correlation:
            # (x ⋆ y)_i = sum_k x_k * y_{k-i}
            # Then in FFT domain: rfft(x ⋆ y) = conj(rfft(x)) * rfft(y) (with proper zero-padding)
            fft_size = 1 << (size - 1).bit_length()
            D = rfft(diff, fft_size)
            A = rfft(a, fft_size)
            grad_half = irfft(np.conj(D) * A, fft_size)[:n]
            grad = 2.0 * grad_half
            
            # Projected gradient step on simplex
            step_size = 1.0 / (rho * np.linalg.norm(grad) + 1e-10)
            a_trial = a - step_size * grad
            # Project onto simplex
            a_trial = project_simplex(a_trial)
            
            # Check improvement
            if fast_convolve(a_trial).max() < fast_convolve(a).max() + 1e-10:
                a = a_trial
            else:
                step_size *= 0.5
                a_trial = a - step_size * grad
                a = project_simplex(a_trial)
        
        # U-update
        u = u + fast_convolve(a) - v_new
        
        # Update best
        conv_a = fast_convolve(a)
        score = 2.0 * n * conv_a.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adapt rho
        if outer % 100 == 99:
            rho *= 1.1
        
        if (outer + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ADMM iter={outer+1} ({elapsed:.0f}s): score={best_score:.8f} rho={rho:.1f} M={M_opt:.6f}")
    
    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def admm_optimize_v2(a0, time_limit, rho0=5.0, n_inner=3):
    """
    ADMM v2: Cleaner implementation with better a-update.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    size = 2 * n - 1
    v = fast_convolve(a).copy()
    u = np.zeros(size)
    rho = rho0
    best_a = a.copy()
    best_score = 2.0 * n * v.max()
    t0 = time.monotonic()
    fft_size = 1 << (size - 1).bit_length()
    
    stall = 0
    prev_best = best_score
    
    for outer in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # V-update (water-filling)
        c = fast_convolve(a) + u
        c_sorted = np.sort(c)[::-1]
        running_sum = np.cumsum(c_sorted)
        # Find m* such that M_m* = (running_sum[m*-1] - 1/rho) / m*
        # satisfies M_m* < c_sorted[m*-1] and (m* == size or M_m* >= c_sorted[m*])
        m = size
        for m_try in range(1, size + 1):
            M_try = (running_sum[m_try - 1] - 1.0 / rho) / m_try
            if m_try == size or M_try >= c_sorted[m_try]:
                m = m_try
                break
        M_opt = (running_sum[m - 1] - 1.0 / rho) / m
        M_opt = max(M_opt, 0.0)
        v_new = np.minimum(c, M_opt)
        
        # A-update: projected gradient
        target = v_new - u
        conv_a = fast_convolve(a)
        for inner in range(n_inner):
            diff = conv_a - target
            D = rfft(diff, fft_size)
            A = rfft(a, fft_size)
            grad = 2.0 * irfft(np.conj(D) * A, fft_size)[:n]
            
            # Adaptive step size
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 1.0 / (rho * gnorm + 1e-10)
            
            a_trial = project_simplex(a - step * grad)
            new_conv_max = fast_convolve(a_trial).max()
            if new_conv_max < conv_a.max() + 1e-12:
                a = a_trial
                conv_a = fast_convolve(a)
            else:
                step *= 0.5
                a = project_simplex(a - step * grad)
                conv_a = fast_convolve(a)
        
        # U-update
        u = u + conv_a - v_new
        
        # Track best
        score = 2.0 * n * conv_a.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adapt rho every 50 iterations
        if (outer + 1) % 50 == 0:
            if stall == 0:
                rho *= 1.2
            else:
                rho *= 0.9
            rho = max(0.1, min(rho, 10000.0))
        
        if (outer + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ADMM2 iter={outer+1} ({elapsed:.0f}s): score={best_score:.8f} rho={rho:.1f}")
        
        # Early stopping if very slow improvement
        if stall > 5000:
            break
    
    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, n_steps=100000):
    """
    Improved coordinate descent with simultaneous moves and better peak targeting.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks
        threshold = max_b * 0.99999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        
        improved = False
        
        for peak_k in peaks[:3]:  # Focus on top 3 peaks
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            # Try reducing top contributors
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                # Try multiple reduction factors
                for factor in [0.9999, 0.9995, 0.999, 0.998, 0.997, 0.995]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    # Redistribute to others (proportionally)
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            # Also try: move mass FROM a peak contributor TO a low element
            if not improved and len(peaks) <= 5:
                # Pick a random low element to receive mass
                low_candidates = np.where(a < np.median(a) * 0.5)[0]
                if len(low_candidates) > 0:
                    target_pos = rng.choice(low_candidates)
                    for pos in top_positions[:3]:
                        if a[pos] < 1e-14:
                            continue
                        for transfer_frac in [0.001, 0.005, 0.01, 0.02]:
                            transfer = a[pos] * transfer_frac
                            a_trial = a.copy()
                            a_trial[pos] -= transfer
                            a_trial[target_pos] += transfer
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = evaluate_local(a_trial)
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Try random perturbation
            if stall % 10 == 0:
                perturbation_scale = min(0.05, 0.005 * np.sqrt(stall / 10))
                a_trial = a + rng.standard_normal(n) * perturbation_scale
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                elif stall > 2000:
                    break
        else:
            stall = 0
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD2 {elapsed:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def dual_guided_gradient_descent(a0, time_limit, max_iters=200000):
    """
    Gradient descent on the simplex with dual-guided subgradient.
    Uses the LP dual to get optimal weights.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Get active set
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Uniform weights on active set (simple but effective)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        # Subgradient
        g = compute_subgradient_full(a, w)
        # Center it (for simplex: the gradient of a function on the simplex
        # is only defined up to addition of a constant)
        g_centered = g - g.mean()
        
        # Projected gradient step
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break
        
        step_size = step / gnorm
        a_trial = a - step_size * g_centered
        a_trial = project_simplex(a_trial)
        
        score = evaluate_local(a_trial)
        if score < best_score - 1e-14:
            a = a_trial
            best_score = score
            best_a = a.copy()
            stall = 0
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.7
            stall += 1
        
        if stall > 200:
            # Perturb and restart
            rng = np.random.default_rng(it * 17 + 3)
            a = a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            step = 0.1
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DG iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} step={step:.4f}")
    
    return best_a, best_score


def multi_start_optimize(parent, budget_s, n_starts=4):
    """Try multiple starting points and return the best result."""
    rng = np.random.default_rng(999)
    best_seq = parent.copy()
    best_score = evaluate_local(parent)
    
    starts = [parent.copy()]
    
    # Add perturbed versions
    for i in range(n_starts - 1):
        scale = 0.01 * (i + 1)
        perturbed = parent + rng.standard_normal(len(parent)) * scale
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        starts.append(perturbed)
    
    # Add interpolated versions with structured sequences
    n = len(parent)
    x = np.linspace(0, 1, n)
    structured_list = [
        np.cos(np.pi * x / 2) ** 2,
        np.sin(np.pi * x) ** 1.5,
        np.exp(-3 * x * (1 - x)),
    ]
    for struct in structured_list[:2]:
        struct = np.maximum(struct, 0)
        struct = struct / struct.sum()
        for alpha in [0.1, 0.3]:
            interp = (1 - alpha) * parent + alpha * struct
            interp = np.maximum(interp, 0)
            interp = interp / interp.sum()
            starts.append(interp)
    
    # Print initial scores
    for i, st in enumerate(starts):
        sc = evaluate_local(st)
        print(f"    Start {i}: score={sc:.8f}")
    
    return best_seq, best_score
