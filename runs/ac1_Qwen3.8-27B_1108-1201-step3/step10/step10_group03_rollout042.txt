Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LogSumExp L-BFGS with
correct gradient, multi-restart strategy, subgradient with Polyak step,
direct LP refinement, and multi-length exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def logsumexp_score(z, T, n):
    """Compute LogSumExp-smoothed score from log-parameters."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    lse = b_max + np.log(np.exp(b_shift - b_max).sum())
    return 2.0 * n * lse


def logsumexp_grad(z, T, n):
    """Compute gradient of LogSumExp-smoothed score w.r.t. log-parameters z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()

    # Gradient of sum_k w[k]*b[k] w.r.t. a[j]:
    # db[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    # So grad_a[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    # = 2 * (w * a_flipped) in convolution terms
    #
    # Efficient computation: cross-correlation via FFT
    size = len(b)  # 2n-1
    N = 1 << (size + n - 1).bit_length()  # padding for full cross-correlation
    W = rfft(w, N)
    A = rfft(a, N)
    # Cross-correlation (w * a)[j] = sum_k w[k]*a[k-j]
    # = ifft(conj(fft(w)) * fft(a))[j]
    cross_corr = irfft(np.conj(W) * A, N)[:n]
    grad_a = 2.0 * cross_corr

    # Chain rule for softmax parameterization: a = softmax(z)
    # dz/dz_k = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return 2.0 * n * T * (b_max + np.log(exp_ws.sum())), grad_z


def logsumexp_grad_full(z, T, n):
    """Return (value, gradient) of LogSumExp objective."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    lse = b_max + np.log(exp_ws.sum())
    val = 2.0 * n * T * lse

    size = len(b)
    N = 1 << (size + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    cross_corr = irfft(np.conj(W) * A, N)[:n]
    grad_a = 2.0 * cross_corr
    S = np.dot(grad_a, a)
    grad_z = 2.0 * n * T * a * (grad_a - S)
    return val, grad_z

def subgradient_optimize(a0, time_limit, n_steps=None, tol=1e-14):
    """Subgradient descent with Polyak step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if n_steps is None:
        n_steps = 50000

    # Track running average for Polyak step
    avg_a = a.copy()
    lr_base = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        current_score = 2.0 * n * M

        # Find active set (peaks within 0.1% of max)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: grad_a[j] = 2 * sum_{k in active} a[k-j] / len(active)
        size = len(b)
        N = 1 << (size + n - 1).bit_length()
        w = np.zeros(size)
        w[active] = 1.0 / len(active)
        W = rfft(w, N)
        A = rfft(a, N)
        cross_corr = irfft(np.conj(W) * A, N)[:n]
        grad_a = 2.0 * cross_corr

        # Polyak step size
        gamma = (current_score - best_score) / (np.dot(grad_a, grad_a) + 1e-15)
        gamma = max(min(gamma, lr_base), 1e-10)

        # Update
        a_trial = a - gamma * grad_a
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s > 1e-15:
            a_trial = a_trial / s

        sc = compute_score(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial

        # Running average (also good solution)
        alpha = 2.0 / (step + 2)
        avg_a = (1 - alpha) * avg_a + alpha * a
        sc_avg = compute_score(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

    return best_a, best_score


def lp_best_response(a_fixed, n_iters=50):
    """Solve LP: find b>=0, sum(b)=1 minimizing max conv(a_fixed, b).
    Uses a simple iterative approach instead of full LP solver."""
    n = len(a_fixed)
    size = 2 * n - 1

    # Build T matrix: T @ b = conv(a_fixed, b)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed

    # Initialize b uniformly
    b = np.ones(n) / n
    best_b = b.copy()

    for it in range(n_iters):
        cb = T @ b
        M = cb.max()
        # Find binding constraints
        active = np.where(cb >= M * 0.95)[0]
        # Simple projected gradient on the LP
        # Minimize M s.t. T@b <= M, sum(b)=1, b>=0
        # Subgradient of M w.r.t. b: average of rows of T for active constraints
        grad = T[active, :].mean(axis=0)
        # Projected gradient step
        lr = M / (np.dot(grad, grad) + 1e-15)
        lr = min(lr, 0.1)
        b_trial = b - lr * grad
        b_trial[np.where(b_trial < 0)[0]] = 0
        s = b_trial.sum()
        if s > 1e-15:
            b_trial /= s
            sc = (T @ b_trial).max()
            if sc < M:
                b = b_trial
                best_b = b.copy()

    return best_b, (T @ best_b).max()

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def random_perturb(a, sigma, rng):
    """Add random perturbation and re-normalize."""
    n = len(a)
    a_trial = a + rng.standard_normal(n) * sigma
    a_trial = np.maximum(a_trial, 0)
    s = a_trial.sum()
    if s < 1e-15:
        return a.copy()
    return a_trial / s

            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick L-BFGS
            T_small = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]
            a_cand, _ = lbgfs_refine(init_cand, t_sub, T_stages=T_small, max_iter_per_stage=200)
            try_update(a_cand, f"n{cand_n}")

    # === Phase 6: Random restart + local optimization ===
    print(f"\n=== Phase 6: Random Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = min(10, int(rem / 30))
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            # Perturb best and optimize
            sigma = 0.02 * (1 + r * 0.5)
            init_r = random_perturb(best_seq, sigma, rng)
            t_sub = min(rem * 0.2, 50.0)
            T_small = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
            a_r, _ = lbgfs_refine(init_r, t_sub, T_stages=T_small, max_iter_per_stage=200)
            try_update(a_r, f"restart_r{r}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very low T L-BFGS
        t_fin = min(rem * 0.5, 120.0)
        if t_fin > 15:
            T_final = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin, _ = lbgfs_refine(best_seq, t_fin, T_stages=T_final, max_iter_per_stage=500)
            try_update(a_fin, "final_lbgfs")

        # Subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            a_sg, _ = subgradient_optimize(best_seq, min(rem * 0.4, 60.0), n_steps=20000)
            try_update(a_sg, "final_sg")

        # One more L-BFGS from the subgradient result
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = lbgfs_refine(a_sg, min(rem * 0.5, 40.0),
                                     T_stages=[0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                     max_iter_per_stage=300)
            try_update(a_fin2, "final_combo")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL LOCAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_score(best_seq)

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
