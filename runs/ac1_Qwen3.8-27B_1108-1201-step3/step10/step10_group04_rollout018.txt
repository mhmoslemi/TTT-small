"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp L-BFGS (main smooth optimizer)
2. Active-set subgradient with Polyak step size and bundle smoothing
3. Multi-restart local search with diverse perturbations
4. Length exploration via interpolation
5. Peak-specific fine-tuning (Wefel-style peak shaving)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def softmax_from_z(z):
    """Convert z-vector to probability simplex via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to z-vector (log)."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective_and_grad(z_in, T_val, n):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700.0
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient w.r.t. a
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Gradient w.r.t. z: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_objective(z_in, T_val, n):
    obj, _ = lse_objective_and_grad(z_in, T_val, n)
    return obj


def lse_grad(z_in, T_val, n):
    _, g = lse_objective_and_grad(z_in, T_val, n)
    return g


def optimize_lse(a_init, T_starts, n_stages_per_T, time_limit):
    """Multi-stage L-BFGS on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = z_from_a(a_init).copy()
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_obj = compute_objective(a_init)
    
    for T in T_starts:
        if time.monotonic() - t0 > time_limit:
            break
        max_iter = n_stages_per_T
        # Use scipy L-BFGS-B
        for restart in range(3):  # Multiple L-BFGS restarts per temperature
            if time.monotonic() - t0 > time_limit:
                break
            
            def func(z_in):
                obj, _ = lse_objective_and_grad(z_in, T, n)
                return obj
            
            def jac(z_in):
                _, g = lse_objective_and_grad(z_in, T, n)
                return g
            
            try:
                res = minimize(
                    fun=func, x0=z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-12, 'maxfun': max_iter * 200,
                             'disp': False}
                )

def subgradient_descent(a_init, time_limit, top_k=5, bundle_size=10):
    """Subgradient descent with Polyak step size and bundle smoothing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_obj = compute_objective(a)
    f_lower = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    
    t0 = time.monotonic()
    z = z_from_a(a)
    
    # Bundle of recent subgradients (in z-space)
    bundle = []
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at current point
        b = fast_convolve(a)
        max_b = b.max()
        # Find top-k peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:top_k]
        
        # Compute subgradient (average over active peaks)
        g_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += a[k - idx]
        g_a /= len(active)
        
        # Convert to z-gradientspace
        S = np.dot(g_a, a)
        g_z = a * (g_a - S)
        
        # Add to bundle
        bundle.append(g_z)
        if len(bundle) > bundle_size:
            bundle.pop(0)
        
        # Average bundle
        g_avg = np.mean(bundle, axis=0)
        
        # Polyak step size
        g_norm_sq = np.dot(g_avg, g_avg)
        if g_norm_sq < 1e-30:
            break
        gap = best_obj - f_lower
        alpha = min(gap / g_norm_sq, 1.0)
        # Clamp alpha
        alpha = max(alpha, 1e-4)
        
        # Update
        z_new = z - alpha * g_avg
        a_new = softmax_from_z(z_new)
        
        # Evaluate
        obj_new = compute_objective(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            # Reject step, reduce step size
            z_new = z - alpha * 0.5 * g_avg
            a_new = softmax_from_z(z_new)
            obj_new = compute_objective(a_new)
            if obj_new < best_obj:
                z = z_new
                a = a_new
            # If still no improvement, add noise
            if it % 200 == 199:
                rng = np.random.default_rng(it)
                noise = rng.normal(0, 0.01, n)
                a = a + a * noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                bundle = []  # Clear bundle
        
        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it} ({elapsed:.0f}s): obj={best_obj:.8f}, gap={gap:.6f}")
        
        if gap < 1e-8:
            break
    
    return best_a, best_obj


def wefel_peak_shaving(a_init, time_limit, top_k=3):
    """Wefel-style peak shaving: directly reduce the top peaks."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:top_k]
        
        improved = False
        for k in top_peaks:
            # Find the j that contributes most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Try reducing the top contributor
            top_contrib_shift = np.argsort(contrib)[::-1]
            
            for tc in top_contrib_shift[:3]:
                src = idx[tc]
                # Find a destination that doesn't contribute to any top peak
                peak_areas = set()
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_areas.update(range(js2, je2 + 1))
                
                # Try moving mass to positions outside peak areas
                candidates = [j for j in range(n) if j not in peak_areas]
                if not candidates:
                    candidates = [j for j in range(n) if j != src]
                
                # Try a few destinations
                dst_choices = rng.choice(candidates, size=min(5, len(candidates)), replace=False)
                
                for dst in dst_choices:
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        a_trial = a.copy()
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        obj = compute_objective(a_trial)
                        if obj < best_obj - 1e-12:
                            a = a_trial
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved and it % 1000 == 999:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                a_trial = a.copy()
                amt = a[i] * rng.uniform(0.01, 0.05)
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    obj = compute_objective(a_trial)
                    if obj < best_obj:
                        a = a_trial
                        best_obj = obj
                        best_a = a.copy()
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Compute full linear convolution a * b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def softmax_from_z(z):
    """Convert z-vector to probability simplex via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to z-vector (log)."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z_in, T_val, n):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700.0
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient w.r.t. a
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    # Gradient w.r.t. z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def _lse_func(z_in, T_val, n):
    """Adapter for scipy minimize (objective only)."""
    obj, _ = lse_obj_grad(z_in, T_val, n)
    return obj


def _lse_jac(z_in, T_val, n):
    """Adapter for scipy minimize (gradient only)."""
    _, g = lse_obj_grad(z_in, T_val, n)
    return g


def run_lse_lbfgs(a_init, temps, time_limit, label=""):
    """Run L-BFGS with log-sum-exp at multiple temperatures."""
    n = len(a_init)
    z = z_from_a(a_init).copy()
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = compute_objective(a_init)

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(20, int(remaining * 8))

        res = minimize(
            fun=_lse_func, x0=z, jac=_lse_jac, method='L-BFGS-B',
            args=(T, n),
            options={'maxiter': max_iter, 'ftol': 1e-16,
                     'gtol': 1e-12, 'maxfun': max_iter * 300,
                     'maxls': 50, 'disp': False}
        )
        z = res.x
        a_cand = softmax_from_z(z)
        sc = compute_objective(a_cand)
        if sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            print(f"    LSE {label} T={T:.2f}: {sc:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def subgradient_polyak(a_init, time_limit, top_k=5, bundle_size=8):
    """Subgradient descent with Polyak step size and bundle averaging."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_objective(a)
    f_lower = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    z = z_from_a(a)
    bundle = []
    stall_count = 0
    rng = np.random.default_rng(42)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        gap = best_score - f_lower
        if gap < 1e-10:
            break

        # Find top-k peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:top_k]

        # Compute subgradient (average over active peaks)
        g_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += a[k - idx]
        g_a /= len(active)

        # Convert to z-space gradient
        S_grad = np.dot(g_a, a)
        g_z = a * (g_a - S_grad)

        # Bundle averaging
        bundle.append(g_z)
        if len(bundle) > bundle_size:
            bundle.pop(0)
        g_avg = np.mean(bundle, axis=0)

        # Polyak step size
        g_norm_sq = np.dot(g_avg, g_avg)
        if g_norm_sq < 1e-30:
            break
        alpha = min(gap / g_norm_sq, 2.0)
        alpha = max(alpha, 1e-5)

        # Try the step
        z_new = z - alpha * g_avg
        a_new = softmax_from_z(z_new)
        obj_new = compute_objective(a_new)

        if obj_new < best_score - 1e-14:
            best_score = obj_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
            stall_count = 0
        else:
            # Backtrack
            z_new = z - alpha * 0.1 * g_avg
            a_new = softmax_from_z(z_new)
            obj_new = compute_objective(a_new)
            if obj_new < best_score:
                z = z_new
                a = a_new
                best_score = obj_new
                best_a = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count >= 50:
                    # Escaping perturbation
                    noise = rng.normal(0, 0.02, n)
                    a = a + a * noise
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    a = a / ss
                    z = z_from_a(a)
                    bundle = []
                    stall_count = 0

        if it % 10000 == 0 and it > 0:
            print(f"    SubGrad it={it} ({time.monotonic()-t0:.0f}s): score={best_score:.8f} gap={gap:.6f}")

    return best_a, best_score


def wefel_shave(a_init, time_limit, max_iter=100000):
    """Wefel-style iterative peak shaving."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    top_k = 3

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:top_k]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_contribs = np.argsort(contrib)[::-1]

            for tc in top_contribs[:5]:
                src = idx[tc]
                if a[src] < 1e-10:
                    continue

                # Find non-contributing positions
                peak_areas = set()
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        peak_areas.add(j2)

                candidates = [j for j in range(n) if j not in peak_areas and a[j] < a.max() * 0.5]
                if not candidates:
                    candidates = [j for j in range(n) if j != src]
                if not candidates:
                    continue

                dst = candidates[rng.integers(len(candidates))]

                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    obj = compute_objective(a_trial)
                    if obj < best_score - 1e-12:
                        a = a_trial
                        best_score = obj
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random exploration
            if it % 100 == 99:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    a_trial = a.copy()
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        obj = compute_objective(a_trial)
                        if obj < best_score:
                            a = a_trial
                            best_score = obj
                            best_a = a.copy()

            if it % 5000 == 4999:
                print(f"    Wefel it={it} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(7 * x) + 0.15 * np.cos(11 * x) + 0.1 * np.cos(13 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n * 3)
        a = a ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    """

    t_deadline = t_start + max(1.0, budget_s - 40.0)

    if best_a is None:

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s) ***")

    n = len(best_a)

    # === Phase 1: Multi-temperature L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: L-BFGS LogSumExp (n={n}, budget check) ===")
    rem = t_deadline - time.monotonic()
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        # Coarse to fine temperature schedule
        temps_coarse = [300.0, 100.0, 40.0, 15.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
        a_out, _ = run_lse_lbfgs(best_a, temps_coarse, t1, label="P1")
        try_update(a_out, "P1_lse")

    # === Phase 2: Subgradient with Polyak step ===
    print(f"\n=== Phase 2: Subgradient Polyak (best={best_score:.8f}) ===")
    rem = t_deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        a_out, _ = subgradient_polyak(best_a, t2, top_k=5, bundle_size=8)
        try_update(a_out, "P2_subgrad")
        # Second pass with different parameters
        rem = t_deadline - time.monotonic()
        if rem > 30:
            t2b = min(rem * 0.10, 150.0)
            if t2b > 15:
                a_out, _ = subgradient_polyak(best_a, t2b, top_k=10, bundle_size=5)
                try_update(a_out, "P2b_subgrad")

    # === Phase 3: Wefel peak shaving ===
    print(f"\n=== Phase 3: Wefel Peak Shaving (best={best_score:.8f}) ===")
    rem = t_deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 15:
        a_out, _ = wefel_shave(best_a, t3)
        try_update(a_out, "P3_wefel")

    # === Phase 4: Fine L-BFGS after Wefel ===
    print(f"\n=== Phase 4: Fine L-BFGS (best={best_score:.8f}) ===")
    rem = t_deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 15:
        temps_fine = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        a_out, _ = run_lse_lbfgs(best_a, temps_fine, t4, label="P4")
        try_update(a_out, "P4_lse")

    # === Phase 5: Multi-restart ===
    print(f"\n=== Phase 5: Multi-Restart (best={best_score:.8f}) ===")
    rem = t_deadline - time.monotonic()
    t5 = min(rem * 0.15, 300.0)

            if r == 0:
                noise = rng.normal(0, 0.015, n)
                a_pert = a_pert + a_pert * noise
            elif r == 1:
                k = max(10, n // 10)
                idx = rng.choice(n, k, replace=False)
                a_pert[idx] *= rng.uniform(0.6, 1.4, k)
            elif r == 2:
                i = rng.integers(0, max(1, n - 200))
                j = min(n, i + rng.integers(100, 250))
                a_pert[i:j] *= rng.uniform(0.5, 1.5)
            elif r == 3:
                # Move mass from peak area to random
                b = fast_convolve(best_a)
                pk = int(np.argmax(b))
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contrib = best_a[idx] * best_a[pk - idx]
                src = idx[np.argmax(contrib)]
                dst = rng.integers(n)
                while dst == src:
                    dst = rng.integers(n)
                a_pert[src] *= 0.7
                a_pert[dst] *= 2.0
            elif r == 4:
                # Frequency-domain perturbation
                A = rfft(a_pert, 2 * n)
                A_mag = np.abs(A)
                A_phase = np.angle(A)
                A_mag *= (1 + rng.normal(0, 0.05, len(A_mag)))

                seg_len = max(5, n // 50)
                for _ in range(3):
                    i = rng.integers(0, max(1, n - seg_len * 2))
                    a_pert[i:i + seg_len], a_pert[i + seg_len:i + 2 * seg_len] = \

            a_pert = np.maximum(a_pert, 0)
            ss = a_pert.sum()
            if ss < 1e-15:
                continue
            a_pert = a_pert / ss

            # Quick optimization: L-BFGS at moderate then low temp
            temps_r = [3.0, 0.5]

            try_update(a_cand, f"restart_{r}")

    rem = t_deadline - time.monotonic()

            t_avail = min((t_deadline - time.monotonic()) * 0.3, 120.0)
            if t_avail < 10:
                continue
            print(f"  n={ln} (budget={t_avail:.0f}s)")

            a_interp = interpolate_seq(best_a, ln)
            sc_interp = evaluate_safe(a_interp)
            if sc_interp < best_score:
                best_score = sc_interp
                best_a = a_interp.copy()

            # Optimize at this length

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}, n={len(best_a)}) ===")
    rem = t_deadline - time.monotonic()
    if rem > 20:

        fs = evaluate_safe(best_a)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_a)}, time={time.monotonic()-t_start:.0f}s")

    return [float(max(0.0, min(1000.0, x))) for x in best_a]
