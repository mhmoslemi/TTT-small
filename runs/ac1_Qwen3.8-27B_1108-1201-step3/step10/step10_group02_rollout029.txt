def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def compute_subgradient_fft(a, active_weights):
    """
    Compute subgradient of max(conv(a,a)) using active set weights.
    active_weights: array of length 2n-1, non-negative, sums to 1.
    Gradient: g_i = 2 * sum_k w_k * a_{k-i} (with boundary handling)
    This is 2 * (w circ a) at index i, computable via FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # Cross-correlation: (corr)[i] = sum_k w[k] * a[k-i]
    # = sum_k w[k] * a[mod(k-i, n)] for circular, but we need linear
    # Linear cross-correlation via FFT:
    W = rfft(active_weights, fft_size)
    A_conj = rfft(a, fft_size)
    # Linear cross-correlation of w and a:
    # (xcorr)[i] = sum_k w[k+i] * a[k] for i >= 0
    # (xcorr)[i] = sum_k w[k] * a[k-i] for i < 0
    # We want: g_i = sum_k w[k] * a[k-i] for 0 <= k-i < n
    # This is the linear cross-correlation at lag i
    # xcorr = ifft(conj(FFT(w)) * FFT(a)) gives circular xcorr
    # For linear: zero-pad
    W_pad = rfft(active_weights, fft_size)
    A_pad = rfft(a, fft_size)
    xcorr = irfft(W_pad * np.conj(A_pad), fft_size)[:size]
    # xcorr[i] = sum_k w[k] * a[k-i] (linear, with zero-padding for out of range)
    # Actually let me be more careful:
    # irfft(rfft(w) * conj(rfft(a))) gives circular cross-correlation
    # With zero-padding to size >= len(w)+len(a)-1, it gives linear cross-correlation
    # xcorr[i] = sum_j w[j+i] * a[j] = sum_k w[k] * a[k-i]
    # So xcorr[i] for i in [0, n-1] gives us what we want
    g = 2.0 * xcorr[:n]
    return g


def lse_obj_grad(a, T):
    """
    Compute LogSumExp objective and gradient for the normalized sequence.
    f(a) = log(sum_k exp(conv(a,a)[k] / T))
    Returns (objective_value, gradient_wrt_a)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    logsumexp = max_b / T + np.log(exp_b.sum())

    # Gradient via FFT cross-correlation
    g = compute_subgradient_fft(a, w)
    g /= T

    return logsumexp, g


def lse_lbfgs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):

    t0 = time.monotonic()

        if remaining < 5:
            break

        # Use a simple projected gradient method with backtracking
        step = 1.0 / (n * T)

        stagnant = 0

        for it in range(maxiter_per_stage):
            if time.monotonic() - t0 > time_limit - 2:

            # Projected gradient step on simplex
            a_trial = a - step * g
            # Project onto simplex: a_trial - min(a_trial, tau) where tau makes sum = 1
            a_sorted = np.sort(a_trial)[::-1]
            css = np.cumsum(a_sorted)
            rho = np.nonzero(a_sorted * np.arange(1, n + 1) > (css - 1))[0][-1]
            tau = (css[rho] - 1) / (rho + 1.0)
            a_trial = np.maximum(a_trial - tau, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial / ss

            score = evaluate_local(a_trial)
            if score < evaluate_local(a) - 1e-12:
                a = a_trial
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                stagnant = 0
                step = min(step * 1.1, 5.0 / n)
            else:
                step *= 0.7
                stagnant += 1

                    eta_bb = s_dot_s / abs(s_dot_y)
                else:
                    eta_bb = s_dot_s / y_dot_y
                eta = 0.7 * eta + 0.3 * eta_bb
            else:
                eta *= 0.998
            eta = max(1e-6, min(eta, 2.0))

        prev_a = a.copy()
        prev_g = g.copy()

        # Mirror descent update with backtracking
        best_trial = a.copy()
        best_trial_score = best_score
        current_eta = eta

        for bt in range(8):
            g_shifted = g - g.min()
            log_a = np.log(np.maximum(a, 1e-300))
            log_a_new = log_a - current_eta * g_shifted
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-300:
                break
            a_new = a_new / a_new_sum

            score = evaluate_local(a_new)
            if score < best_score - 1e-13:
                best_trial = a_new
                best_trial_score = score
                break
            elif score < evaluate_local(a) + 1e-10:
                best_trial = a_new
                best_trial_score = min(score, best_trial_score)
            current_eta *= 0.5

        a = best_trial
        eta = current_eta * 2.0 if current_eta < eta else eta  # restore full step for next

        score = best_trial_score
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve >= 300 and restart_count < max_restarts:
            noise_scale = 0.003 + 0.002 * restart_count
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            restart_count += 1
            eta = eta0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD v2 iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine_v2(a0, time_limit, solver=None, n_max_iters=20000):
    """Nesterov LP alternating with adaptive mixing."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    last_lp_time = 0
    lp_time_budget = 2.0  # seconds per LP solve

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        lp_start = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        last_lp_time = time.monotonic() - lp_start
        if b_new is None:
            continue

        # Nesterov mixing
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 15:
                noise_scale = 0.01 / np.sqrt(stall)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            # If LP is too slow, reduce frequency
            if last_lp_time > lp_time_budget:
                # Skip some iterations
                for skip in range(5):
                    if time.monotonic() - t0 > time_limit - 5:
                        break
                    b_new2, _ = solve_lp_best_response(y, solver)
                    if b_new2 is not None:
                        a = 0.5 * a + 0.5 * b_new2
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        sc = 2.0 * n * fast_convolve(a).max()
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()

    return best_a, best_score


def peak_equalization(a0, time_limit, n_iter=5000):
    """
    Water-filling approach: try to equalize the top convolution values.
    At each step, identify the top peaks and try to reduce them by
    adjusting the contributing a values.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            # Find top contributors to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[pk - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Try reducing the top contributor
            top_pos = idx[np.argmax(contribs)]
            if a[top_pos] < 1e-10:
                continue

            # Try several reduction factors
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97]:
                a_trial = a.copy()
                old_val = a_trial[top_pos]
                a_trial[top_pos] = old_val * factor
                diff = old_val - a_trial[top_pos]
                # Redistribute to the element with smallest contribution to this peak
                min_pos = idx[np.argmin(contribs)]
                a_trial[min_pos] += diff
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()

                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        # Also try: reduce all top contributors simultaneously
        if not improved:
            pk = peak_indices[0]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Reduce top 3 contributors, add to bottom 3
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            bot3 = idx[np.argsort(contribs)[:3]]
            a_trial = a.copy()
            total_red = 0
            for p in top3:
                if a_trial[p] > 1e-10:
                    red = a_trial[p] * 0.002
                    a_trial[p] -= red
                    total_red += red
            for p in bot3:
                a_trial[p] += total_red / len(bot3)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = evaluate_local(a_trial)
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence with various patterns."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    elif choice == 5:
        # Blackman-like window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        # Raised cosine with parameter
        alpha = 0.5
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5) / (2 * alpha)))
        a = np.maximum(a, 0)
    else:
        # Slight perturbation of uniform
        a = 1.0 + 0.1 * np.sin(2 * np.pi * 3 * x) + 0.05 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    a = a + 1e-8  # Ensure strictly positive
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_method(a0, time_limit, eta0=0.01, n_iter=100000):
    """
    Subgradient method with polyak step size for the non-smooth objective.
    Uses the fact that we know a lower bound (1) and upper bound (current best).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b  # with sum=1

        # Polyak step size
        gap = current_score - lower_bound
        if gap < 1e-3:
            gap = 1e-3
        step = gap / (2.0 * n * max_b)  # Approximate Lipschitz constant
        step = max(1e-6, min(step, 0.1))

        # Compute subgradient at the peak
        peak_k = int(np.argmax(b))
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[peak_k - idx]
        # Subtract mean to keep on simplex
        g -= g.mean()

        # Update
        a_trial = a - step * g
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / ss
        a = a_trial

        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.002
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_score


def fast_refine(a0, time_limit):
    """
    Quick refinement using adaptive mirror descent with multiple strategies.
    Designed to be fast (for trying many n values).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(888)

    # Phase 1: Mirror descent with adaptive eta
    eta = 0.03 / np.log(max(n, 2))
    no_improve = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit - 2:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: positions within 99.5% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Limit active set size for speed
        if len(active) > 50:
            # Keep the top 50
            active = active[np.argsort(b[active])[::-1][:50]]

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Update
        g_centered = g - g.mean()
        a_trial = a * np.exp(-eta * g_centered)
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-300:
            a_trial = a.copy()
        else:
            a_trial = a_trial / ss

        score = evaluate_local(a_trial)
        if score < best_score - 1e-13:
            a = a_trial
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                eta *= 0.95

        if no_improve >= 200:
            noise = rng.standard_normal(n) * 0.002
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            eta = 0.03 / np.log(max(n, 2))

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase, multi-length optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Strategy:
    1. Start from parent (n=1200) or structured init.
    2. Quick optimize at multiple n values to find the best length.
    3. Deep optimize at the best n using mirror descent + LP + peak equalization.
    4. Final polish with coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_parent = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 0: Quick multi-n exploration ===
    print(f"\n=== Phase 0: Multi-n Exploration ===")
    rem = deadline - time.monotonic()
    n_candidates = [n_parent]
    # Add nearby sizes
    for factor in [1.2, 1.5, 2.0, 2.5, 3.0, 4.0]:
        n_new = int(n_parent * factor)
        if n_new > 200 and n_new < 10000:
            n_candidates.append(n_new)
    # Also try some round numbers
    for n_try in [1000, 1500, 2000, 3000, 4000]:
        if n_try not in n_candidates and 200 < n_try < 10000:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))
    n_candidates = [n for n in n_candidates if 200 <= n <= 8000]

    print(f"  Testing n values: {n_candidates}")

    # Quick evaluation at each n
    n_score_map = {}
    quick_time_each = min(15.0, max(3.0, (rem * 0.15) / len(n_candidates)))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        a_try = interpolate_sequence(best_seq, n_try)
        # Quick optimize
        a_opt, sc = fast_refine(a_try, quick_time_each)
        n_score_map[n_try] = sc
        print(f"    n={n_try}: score={sc:.8f}")
        if sc < best_score:
            best_seq = a_opt
            best_score = sc

    # Pick the best n for deep optimization
    if n_score_map:
        best_n = min(n_score_map, key=n_score_map.get)
    else:
        best_n = n_parent
    print(f"  Best n for deep optimization: {best_n} (score={n_score_map.get(best_n, best_score):.8f})")

    # === Phase 1: Deep mirror descent at best n ===
    print(f"\n=== Phase 1: Deep Mirror Descent (n={best_n}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_best_n = interpolate_sequence(best_seq, best_n) if len(best_seq) != best_n else best_seq.copy()
        a_md, _ = mirror_descent_v2(a_best_n, t_alloc, eta0=0.04, active_frac=0.999)
        try_update(a_md, "mirror_descent_deep")

    # === Phase 2: LP alternating ===
    print(f"\n=== Phase 2: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30 and len(best_seq) <= 3000:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine_v2(best_seq, t_alloc, solver=solver, n_max_iters=20000)
        try_update(a_lp, "lp_alternating")
    elif len(best_seq) > 3000:
        print(f"  Skipping LP (n={len(best_seq)} too large)")

    # === Phase 3: LogSumExp gradient method ===
    print(f"\n=== Phase 3: LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, _ = lse_lbfgs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=3000)
        try_update(a_lb, "logsumexp")

    # === Phase 4: Peak equalization ===
    print(f"\n=== Phase 4: Peak Equalization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        a_pe, _ = peak_equalization(best_seq, t_alloc, n_iter=20000)
        try_update(a_pe, "peak_equalization")

    # === Phase 5: Subgradient method ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_method(best_seq, t_alloc, eta0=0.005, n_iter=50000)
        try_update(a_sg, "subgradient")

    # === Phase 6: Final mirror descent polish (very small eta) ===
    print(f"\n=== Phase 6: Final Polish MD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_fin, _ = mirror_descent_v2(best_seq, t_alloc, eta0=0.002, active_frac=0.9999)
        try_update(a_fin, "final_md")

    # === Phase 7: Try a few more n values with remaining time ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Additional n exploration (score={best_score:.8f}) ===")
        # Try n values we haven't deeply optimized
        additional_ns = [n for n in n_score_map if n != best_n]
        additional_ns.sort(key=lambda x: n_score_map[x])
        for n_try in additional_ns[:3]:
            if time.monotonic() > deadline - 20:
                break
            t_alloc2 = min(rem * 0.3, 60.0)
            if t_alloc2 < 10:
                break
            a_try = interpolate_sequence(best_seq, n_try) if len(best_seq) != n_try else best_seq.copy()
            a_opt2, sc2 = mirror_descent_v2(a_try, t_alloc2, eta0=0.02, active_frac=0.999)
            try_update(a_opt2, f"n={n_try}_refined")
            rem = deadline - time.monotonic()
            if rem < 15:
                break

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
