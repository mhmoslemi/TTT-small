"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA on LogSumExp-smoothed
objective with adaptive temperature cooling, combined with LP alternation
refinement and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_pair(a, b):
    """Linear convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1) / ind > 0
    k = np.max(ind[cond])
    theta = (cssv[k - 1] - 1) / k
    return np.maximum(v - theta, 0)


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_lse_gradient(a, T):
    """Compute gradient of (2n/T) * log(sum exp(c_k/T)) where c = a*a."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    p = np.exp((c - max_c) / T)
    p = p / p.sum()
    
    # s[i] = sum_j a[j] * p[i+j]
    p_flipped = p[::-1]  # length 2n-1
    conv = fast_convolve_pair(a, p_flipped)  # length 3n-2
    s = conv[2*n-2 : 3*n-2]  # length n
    
    # w[i] = sum_{j=0}^{n-1} p[i+j]
    p_ext = np.zeros(2*n)
    p_ext[:2*n-1] = p
    p_cum = np.concatenate([[0.0], np.cumsum(p_ext)])
    w = p_cum[n:] - p_cum[:n]
    
    grad = (2.0 * n / T) * (s + a * w)
    return grad


def fista_lse(a0, T_schedule, max_iters, time_limit, verbose=True):
    """FISTA with LogSumExp smoothing and adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t_curr = 1.0
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Phase through temperature schedule
    for T_idx, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        phase_iters = max(20, max_iters // len(T_schedule))
        
        for it in range(phase_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad = compute_lse_gradient(y, T)
            
            # Backtracking line search
            alpha = 0.1 / (n / max(1, T))  # initial guess
            alpha = min(alpha, 10.0 / n)
            alpha = max(alpha, 1e-6)
            
            for ls_iter in range(20):
                a_new = project_simplex(y - alpha * grad)
                # Simple decrease check (skip for speed, use fixed step)
                break
            
            a_new = project_simplex(y - alpha * grad)
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0
            y = a_new + ((t_curr - 1.0) / t_next) * (a_new - a)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 0:
                y = y / sy
            t_curr = t_next
            
            a = a_new
            
            # Evaluate periodically

                score = evaluate_local(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if verbose:
                        elapsed = time.monotonic() - t0
                        print(f"    FISTA T={T:.3f} iter={it+1} ({elapsed:.1f}s): score={best_score:.8f}")
    
    # Final evaluation
    score = evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

def lp_best_response(a_fixed, max_iters=100):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Uses scipy linprog for simplicity."""
    from scipy.optimize import linprog
    
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build T matrix: T @ b = conv(a, b)
    # T has shape (size, n)
    # T[k, j] = a[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    # LP: minimize M s.t. T @ b <= M, sum(b) = 1, b >= 0
    # Variables: [b_0, ..., b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality: T @ b - M <= 0
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': max_iters * 10})
        if res.status == 0:
            b_val = res.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            M_val = res.x[n]
            return b_val, M_val
    except Exception:
        pass
    return None, None


def alternation_optimize(a0, time_limit, n_rounds=50):
    """LP alternation: fix a, solve for b; fix b, solve for a; iterate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for round_i in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve for b given a
        b_new, M_b = lp_best_response(a)
        if b_new is not None:
            a = b_new
        
        # Solve for a given b (same thing since it's symmetric)
        a_new, M_a = lp_best_response(a)
        if a_new is not None:
            a = a_new
        
        # Evaluate
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def coordinate_descent_peaks(a0, time_limit, n_steps=20000):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    else:

            break
        
        c = fast_convolve(a)
        max_c = c.max()
        peak_k = int(np.argmax(c))

        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_pos = idx[np.argmax(contribs)]
        for factor in [0.99, 0.98, 0.97, 0.95]:

            a_trial[top_pos] *= factor
            diff = a[top_pos] * (1.0 - factor)

            # Redistribute

            mask[top_pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:

1) FISTA with LogSumExp smoothing for fast initial optimization
2) Nesterov-accelerated LP alternation (using scipy HiGHS) for precise refinement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_pair(a, b):
    """Linear convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using O(n) algorithm (approx)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / ind > 0
    k = np.max(ind[cond])
    theta = (cssv[int(k) - 1] - 1.0) / k
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_T_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b, M) or (None, None) on failure."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    T = build_T_matrix(a_fixed)
    
    # Variables: [b_0, ..., b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            b_val = np.asarray(res.x[:n], dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            M_val = float(res.x[n])
            return b_val, M_val
    except Exception:
        pass
    return None, None

def fista_lse_optimize(a0, time_limit, T_schedule=None, max_iters_total=100000,

            break

                break

            max_c = c.max()

            if p_sum < 1e-30:
                p = np.ones_like(c) / len(c)

                p = p / p_sum

            alpha = max(alpha, 1e-8)

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0

                y = y / sy
            else:

            a = a_new
            
            if (it + 1) % 200 == 0:

                    best_a = a.copy()
                    if print_interval and (it + 1) % print_interval == 0:

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        peak_k = int(np.argmax(c))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        for factor in [0.995, 0.99, 0.985, 0.98, 0.97]:
            top_pos = idx[np.argmax(contribs)]
            if a[top_pos] < 1e-12:
                break
            a_trial = a.copy()
            a_trial[top_pos] *= factor
            diff = a[top_pos] * (1.0 - factor)
            if diff < 1e-15:
                continue
            mask = np.ones(n, dtype=bool)
            mask[top_pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                break
            a_trial[mask] = others + diff * others / others_sum
            a_trial = a_trial / a_trial.sum()
            
            score = evaluate_local(a_trial)
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
        
        if not improved:
            # Try simultaneous reduction of top-2 contributors
            top2 = idx[np.argsort(contribs)[::-1][:2]]
            a_trial = a.copy()
            for pos in top2:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.998
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = evaluate_local(a_trial)
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
        
        if step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000, print_interval=5000):
    """Subgradient method targeting active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        threshold = max_c * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = evaluate_local(a_trial)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)
        
        if step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    else:
        a = (1 - x) ** 0.7 + 0.1
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
