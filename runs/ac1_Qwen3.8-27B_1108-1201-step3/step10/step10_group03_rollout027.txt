**New idea: Use the "saddle point" structure more directly**

The problem can be written as:
`min_a max_lambda sum_k lambda_k * conv(a,a)[k]`
subject to `sum(a) = 1, a >= 0, lambda >= 0, sum(lambda_k) = 1`

By min-max theorem (Sion's theorem), if the problem is convex-concave, we can swap min and max:
`= max_lambda min_a sum_k lambda_k * conv(a,a)[k]`
subject to the same constraints

The inner problem is: `min_a sum_k lambda_k * conv(a,a)[k]` subject to `sum(a)=1, a>=0`.

Now, `sum_k lambda_k * conv(a,a)[k] = a^T (sum_k lambda_k Q_k) a` where `Q_k` is the quadratic form matrix for the k-th convolution entry.

`sum_k lambda_k Q_k` is a symmetric matrix (let's call it `Q_lambda`). If it's PSD, then the inner problem is a convex QP, which we can solve efficiently.

Is `Q_lambda` PSD? We have `Q_lambda = sum_k lambda_k Q_k` where `Q_k[j,l] = 1` if `j+l=k`, else 0.

`Q_lambda[j,l] = sum_k lambda_k * 1{j+l=k} = lambda_{j+l}` (for valid `j+l`).

So `Q_lambda` is a Hankel matrix with entries `lambda_{j+l}`!

A Hankel matrix is PSD if and only if the sequence `lambda` is a "moment sequence" (i.e., there exists a positive measure whose moments are `lambda`). By the Hamburger moment problem, this is equivalent to all the Hankel matrices being PSD.

So `Q_lambda` is PSD iff the `2x2` minors are non-negative, i.e., `lambda_k * lambda_{k+2} >= lambda_{k+1}^2` for all `k`. This is the condition that `log(lambda_k)` is concave (if all `lambda_k > 0`).

So if we choose `lambda` such that `log(lambda_k)` is concave (e.g., `lambda_k = exp(-c*k^2)` or `lambda_k = 1/(1+c*k)`), then `Q_lambda` is PSD, and the inner problem is a convex QP!

This gives us a tractable approach:

2. Solve the inner QP: `min_a a^T Q_sigma a` subject to `sum(a)=1, a>=0`
3. Evaluate the outer objective at the solution
4. Optimize over `sigma`

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(n, 0, -1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)


def hankel_matvec(lam, a):
    """Compute Q_lambda @ a where Q_lambda[j,l] = lam[j+l]."""
    n = len(a)
    result = np.empty(n)
    for j in range(n):
        result[j] = np.dot(lam[j:j+n], a)
    return result

def solve_lp_fixed(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None


def bidirectional_lp(a0, time_limit, solver, max_iters=500000):
    """Alternating LP with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(42)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_fixed(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        a_new = solve_lp_fixed(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 5:
                noise_scale = 0.01 / math.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0

            if (it + 1) % 5000 == 0:
                print(f"    BiDirLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def qp_dual_refinement(a0, n_rounds=50, p_power=2.0):
    """Iterative dual ascent: alternate between forming lambda from conv(a,a)
    and solving the QP. Provides lower bounds via min-max duality."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_lb = 0.0
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > 120:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Form lambda from b using power weighting
        b_pos = np.maximum(b, 1e-15)
        lam = b_pos ** p_power
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.ones(2*n-1) / (2*n-1)
        else:
            lam = lam / lam_sum
        
        # Solve QP using FISTA
        a_new = solve_qp_fista(lam, n, max_iters=3000, lr=0.4)
        
        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        
        # Lower bound: the QP optimal value
        qp_val = a_new @ hankel_matvec(lam, a_new)
        lb = 2.0 * n * qp_val
        if lb > best_lb:
            best_lb = lb
        
        if (rnd + 1) % 10 == 0:
            print(f"    QP-dual round {rnd+1}: ub={best_score:.10f}, lb={best_lb:.10f}, gap={best_score-best_lb:.2e}")

    return best_a, best_score, best_lb


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)

            break
        
        def lse_obj(z_vec):
            z_max = z_vec.max()
            exp_z = np.exp(z_vec - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def lse_grad(z_vec):
            z_max = z_vec.max()
            exp_z = np.exp(z_vec - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        max_iter = max(30, int((time_limit - (time.monotonic() - t0)) * 15))
        
        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',

            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def local_eval(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
