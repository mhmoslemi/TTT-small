"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated LP alternating

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import minimize


def fast_convolve(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return np.array([np.dot(a, a)])
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective function locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, size)
        for k in range(start, end):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def solve_lp_best_response(a_fixed, solver_name='CLARABEL'):
    """Given fixed a (normalized), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver_name]
    for alt in ['CLARABEL', 'PDLP', 'ECOS', 'SCS']:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    b_val /= sv
                return b_val, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def nesterov_lp_alternate(a0, time_limit, solver_name='CLARABEL'):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_lp_best_response(y, solver_name)
        if b_new is None:
            b_new = y.copy()
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        t_iter = it + 1
        a = (t_iter * a + alpha * (it + 2) * b_new) / (t_iter + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        cur_best = min(score_a, score_y)
        
        if cur_best < best_score - 1e-11:
            best_score = cur_best
            best_a = a.copy() if score_a <= score_y else y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count >= 200:
            noise_scale = 0.01 / np.sqrt(stall_count / 200)

            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            a = a + noise * 0.5

            stall_count = 0
        
        if (it + 1) % 500 == 0:

def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_grad(a, T):
    """Compute gradient of (1/T)*log(sum_k exp(T*conv(a,a)[k]))."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(T * (b - max_b))
    w = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w[k] * a[k - idx]
    grad *= 2.0
    return grad, max_b + np.log(exp_b.sum()) / T


def lse_lbgfs(a0, time_limit, T_list=None):
    """LogSumExp L-BFGS refinement with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_list is None:
        T_list = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_in):
            a_in = softmax_safe(z_in)
            _, obj = logsumexp_grad(a_in, T)
            return obj
        
        def jac(z_in):
            a_in = softmax_safe(z_in)
            grad_a, _ = logsumexp_grad(a_in, T)
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        
        maxiter = max(20, int(remaining * 200))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax_safe(z)


def coord_descent_polish(a0, time_limit):
    """Adaptive coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:

    t0 = time.monotonic()
    stall = 0
    
    factors_large = [0.999, 0.998, 0.996, 0.993, 0.988, 0.98]
    factors_small = [0.9999, 0.9998, 0.9995, 0.999, 0.9985, 0.9975, 0.995, 0.99]
    
    step = 0
    max_steps = 500000
    use_small = False
    
    while step < max_steps:
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        threshold = max_b * 0.99999
        peaks = np.where(b >= threshold)[0]
        
        if len(peaks) > 20:
            peak_vals = [b[k] for k in peaks]
            top_peak_idx = np.argsort(peak_vals)[::-1][:5]
            peaks = peaks[top_peak_idx]
        
        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            factors = factors_small if use_small else factors_large
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in factors:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask].copy()
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved:
                for pos in top_positions:
                    if a[pos] < 1e-12:
                        continue
                    for factor in [1.001, 1.005, 1.01]:
                        a_trial = a.copy()
                        a_trial[pos] *= factor
                        diff = a_trial[pos] - a[pos]
                        if diff < 1e-18:
                            continue
                        mask = np.ones(n, dtype=bool)
                        mask[pos] = False
                        others = a_trial[mask].copy()
                        others_sum = others.sum()
                        if others_sum < 1e-15:
                            break
                        a_trial[mask] = others - diff * others / others_sum
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = evaluate_local(a_trial)
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            stall += 1
            if stall > 50 and not use_small:
                use_small = True
            if stall > 500:
                # Try simultaneous reduction of top contributors
                b = fast_convolve(a)
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:4]]
                
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.9999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            if stall > 10:
                use_small = False
            stall = 0
        
        step += 1
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                t_alt = min(rem * 0.4, 200.0)
                if t_alt > 20:
                    a_result, sc = nesterov_lp_alternate(a_interp, t_alt, solver_name='CLARABEL')
                    try_update(a_result, f"size_{ns}_lp")
                    rem2 = total_deadline - time.monotonic()
                    if rem2 > 15:
                        a_result2, _ = coord_descent_polish(a_result, min(rem2 * 0.3, 60.0))
                        try_update(a_result2, f"size_{ns}_polish")
    
    # === Phase 7: Final coordinate descent ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
        rem = total_deadline - time.monotonic()
        t7 = min(rem * 0.5, 100.0)
        if t7 > 10:
            a_result, sc = coord_descent_polish(best_seq, t7)
            try_update(a_result, "final_polish")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    
    if best_seq is not None:
        result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with the actual evaluate function
        try:
            actual = evaluate_sequence(result)
            print(f"Actual evaluation: {actual:.8f}")
        except Exception:
            pass
        return result
    return [1.0]
