from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse one-sided initializations (ramp, exponential)
2. LP-based best-response alternating minimization
3. Adaptive coordinate descent with incremental convolution
4. Fine-grained polishing
5. Multi-length exploration (n=512, 1024, 2048)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy()
    return b_val / s


def lp_alternating(a0, time_limit, solver, verbose=False):
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time_limit - (time.monotonic() - t0) < 2.0:
            break
        
        b_new = lp_best_response(a, solver)
        score_b = score_normalized(b_new)
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Damped update
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.7
        elif it <= 60:
            alpha = 0.5
        else:
            alpha = max(0.2, 6.0 / it)
        
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if score_a > best_score * (1 + 1e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)
        
        if stagnation > 40:
            # Perturb
            noise = np.random.standard_normal(n) * 0.02 / np.sqrt(n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
            if score_normalized(a_pert) < best_score * 1.01:
                a = a_pert
            stagnation = 0
        
        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

def coordinate_descent(a0, time_limit, verbose=False, fine=False):
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    
    if fine:
        factors = [0.995, 0.997, 0.998, 1.002, 1.003, 1.005, 0.99, 1.01]
        tol = 1e-15
    else:
        factors = [-0.4, -0.25, -0.15, -0.08, -0.03, 0.03, 0.08, 0.15, 0.25, 0.4]
        tol = 1e-14
    
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(n):
            if a[i] < 1e-17:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:

            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            # a[k - i] for k in [k_start, k_end]
            idx_start = k_start - i  # = 0
            idx_end = k_end - i  # = n-1 (kapped)

            conv_affected = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in factors:
                delta = a[i] * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv_affected + 2.0 * delta * a_shifted
                # Quadratic correction at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    qidx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[qidx] += delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - tol:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - tol:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:

                    changes[qidx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:

        else:
            no_improve_sweeps = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def diverse_init(n, rng, init_type):
    t = np.arange(n, dtype=np.float64)
    if init_type == 0:
        # Linear ramp (increasing)
        a = t + 1.0
    elif init_type == 1:
        # Exponential ramp
        a = np.exp(2.0 * t / n)
    elif init_type == 2:
        # Sqrt ramp
        a = np.sqrt(t + 1.0)
    elif init_type == 3:
        # Quadratic ramp
        a = (t + 1.0) ** 2
    elif init_type == 4:
        # Log ramp
        a = np.log(t + 1.0) + 0.1
    elif init_type == 5:
        # Triangle (symmetric)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0.01)
    elif init_type == 6:
        # U-shape
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.1 + 0.9 * x * x
    elif init_type == 7:
        # Random smooth
        freq = rng.integers(2, 6)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    elif init_type == 8:
        # Power law
        a = (t + 1.0) ** rng.uniform(0.3, 1.5)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def make_ramp_init(n, power=1.0, reverse=False):
    t = np.arange(n, dtype=np.float64)
    if reverse:
        t = n - 1 - t
    a = (t + 1.0) ** power
    a = a / a.sum()
    return a
