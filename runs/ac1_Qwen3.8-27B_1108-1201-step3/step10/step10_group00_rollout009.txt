"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp gradient (O(n log n) per eval) with L-BFGS
2. Ultra-fine temperature annealing (T down to 0.002)
3. Direct peak-targeted subgradient polishing
4. Multi-length exploration (n=1200, 1500, 1800)
5. Structured perturbation restarts with re-optimization
Key innovation: FFT cross-correlation for gradient makes each eval ~100x faster
than the loop-based approach, enabling far more L-BFGS iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_a(w, a):
    """Compute grad_a[j] = sum_m w[j+m]*a[m] via FFT cross-correlation. O(n log n)."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return conv[n - 1:2 * n - 1]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector (sum=1)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective_and_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z (log-space)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    T_b = T * (b - max_b)
    exp_Tb = np.exp(T_b)
    sum_exp = exp_Tb.sum()
    w = exp_Tb / sum_exp
    lse = max_b + (1.0 / T) * np.log(sum_exp)
    obj = 2.0 * n * lse

    # grad_a = 2 * compute_grad_a(w, a) [factor 2 from db/da of auto-conv]
    grad_a = 2.0 * compute_grad_a(w, a)

    # Chain rule: z -> a = softmax(z)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing. FFT-accelerated."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_z = z.copy()
    best_obj = lse_objective_and_grad(z, n, T_stages[0])[0]

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        def func(z_curr):
            if time.monotonic() > deadline:
                return 1e15
            return lse_objective_and_grad(z_curr, n, stage_T)[0]

        def jac(z_curr):
            if time.monotonic() > deadline:
                return np.zeros_like(z_curr)
            return lse_objective_and_grad(z_curr, n, stage_T)[1]

        iters = max(20, maxiter_per_stage // len(T_stages))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Track best
        true_score = evaluate_local(softmax_from_z(z))
        if true_score < best_obj or True:
            obj_now = lse_objective_and_grad(z, n, stage_T)[0]
            if obj_now < best_obj:
                best_obj = obj_now
                best_z = z.copy()

        if (stage_idx + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            a_check = softmax_from_z(z)
            sc = evaluate_local(a_check)
            print(f"    L-BFGS stage {stage_idx+1}/{len(T_stages)} T={stage_T} ({elapsed:.0f}s): score={sc:.8f}")

    return softmax_from_z(best_z)


def direct_subgradient_polish(a0, time_limit, n_steps=50000):
    """Direct subgradient on the true max (no smoothing). Multiplicative weights."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.02
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find peaks within 0.1% of max
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: average of shifted a at peak locations
        g = np.zeros(n)
        for k in active:
            g += 2.0 * compute_grad_a(np.ones_like(b) / len(active), a)
        # Actually compute_grad_a with uniform w gives the average of a[k-j]
        # Let me just do it directly for correctness
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Multiplicative update (mirror descent on simplex)
        eta = eta_base * scale
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi * 4, n)
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    T_coarse = [10.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03]

    # === Phase 1: Fine L-BFGS on parent (n=1200) ===
    print(f"\n=== Phase 1: Fine L-BFGS n=1200 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, {len(T_fine)} T-stages")
        a_fin = lbgfs_refine(best_seq, t_alloc, T_fine, maxiter_per_stage=15000)
        local_update(a_fin, "lbgfs_fine_1200")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: SubGrad polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_sg, _ = direct_subgradient_polish(best_seq, t_alloc, n_steps=50000)
        local_update(a_sg, "subgrad_1200")

    # === Phase 3: Second L-BFGS round (from updated best) ===
    print(f"\n=== Phase 3: L-BFGS Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 40:
        a_r2 = lbgfs_refine(best_seq, t_alloc, T_fine, maxiter_per_stage=12000)
        local_update(a_r2, "lbgfs_r2")

    # === Phase 4: Interpolate to n=1500 and optimize ===
    print(f"\n=== Phase 4: n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t_1500 = min(rem * 0.20, 300.0)
        print(f"  Budget: {t_1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500 = lbgfs_refine(init_1500, t_1500 * 0.7, T_medium, maxiter_per_stage=10000)
        local_update(a_1500, "lbgfs_1500")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == 1500:
            a_1500b = lbgfs_refine(best_seq, min(rem * 0.5, 100.0), T_coarse, maxiter_per_stage=8000)
            local_update(a_1500b, "lbgfs_1500b")

    # === Phase 5: Interpolate to n=1800 and optimize ===
    print(f"\n=== Phase 5: n=1800 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_1800 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_1800:.0f}s")
        init_1800 = interpolate_sequence(best_seq, 1800)
        a_1800 = lbgfs_refine(init_1800, t_1800 * 0.7, T_medium, maxiter_per_stage=8000)
        local_update(a_1800, "lbgfs_1800")

    # === Phase 6: Perturbation restarts ===
    print(f"\n=== Phase 6: Perturbation restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.015 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 80.0)
            a_opt = lbgfs_refine(a_pert, t_sub, T_coarse, maxiter_per_stage=6000)
            local_update(a_opt, f"pert_{trial}_n{n_cur}")

    # === Phase 7: Final fine polish on best ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.4, 120.0)
        T_final = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
        a_fin = lbgfs_refine(best_seq, t_fin, T_final, maxiter_per_stage=10000)
        local_update(a_fin, "final_lbgfs")

    # === Phase 8: Final subgradient ===
    rem = deadline - time.monotonic()
    if rem > 20:
        a_fin2, _ = direct_subgradient_polish(best_seq, min(rem * 0.4, 30.0), n_steps=30000)
        local_update(a_fin2, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)}, elapsed={elapsed:.0f}s)")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
