    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 1e-15:

    n = len(a)
    b = fast_convolve(a / s)
    return float(2.0 * n * b.max())


def get_solver():

    for name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                    sy = y.sum()
                    if sy > 1e-15:
                        y = y / sy
                    stall = 0
                    prev_best = best_score

        if it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    NL iter={it} ({elapsed:.0f}s): best={best_score:.8f} M_LP={M_b:.8f}")

    return best_a, best_score


def multiplicative_weights_optimize(a0, time_limit, lr_init=0.02, top_k=20):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    theta = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300))
    a = np.exp(theta - theta.max())
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t_start < time_limit:
        step += 1
        b = fast_convolve(a)
        max_b = b.max()

        # Average subgradient over top-k peaks
        top_idx = np.argsort(b)[::-1][:min(top_k, len(b))]
        g = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(top_idx)

        # Mirror descent update: theta -= lr * g
        theta = theta - lr * g
        a_new = np.exp(theta - theta.max())
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)

        if step % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    MW step={step} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def lse_obj(z, n, T):
    """LogSumExp smoothed objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    lse = max_b + np.log(exp_b.sum())
    return (2.0 * n / T) * lse


def lse_grad(z, n, T):
    """Gradient of LogSumExp smoothed objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0 / T
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_lse(a0, time_limit, T_stages=None, maxiter_total=10000):
    """L-BFGS with LogSumExp multi-stage temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300))
    t_start = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02]

    n_stages = len(T_stages)
    iters_per = max(30, maxiter_total // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t_start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 5:
            break

        def func_loc(z):
            return lse_obj(z, n, T_stage)

        def jac_loc(z):
            return lse_grad(z, n, T_stage)

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def coord_descent_polish(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_start = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:5]
        improved = False

        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.985, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Simultaneous reduction
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break

        if step % 10000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    CD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, seed):
    """Create diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.4 * np.exp(-4 * (x - 0.7) ** 2)
    else:
        a = (1 - x) ** 0.8 + 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Nesterov LP alternation (primary)
    2. Multiplicative weights (exploration)
    3. LogSumExp L-BFGS (refinement)
    4. Coordinate descent (polish)
    5. Multi-start diversity
    6. Alternative lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, seed)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP (primary workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP, n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = rem * 0.45
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc = nesterov_lp_optimize(best_seq, t1, solver, max_iters=200000, stall_patience=12)
        try_update(a_lp, "nlp1")

    # === Phase 2: Multiplicative weights ===
    print(f"\n=== Phase 2: Multiplicative Weights, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = rem * 0.12
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_mw, _ = multiplicative_weights_optimize(best_seq, t2, lr_init=0.015, top_k=30)
        try_update(a_mw, "mw")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = rem * 0.20
    if t3 > 20:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]
        print(f"  T stages: {len(T_stages)}, budget={t3:.0f}s")
        a_lb = lbgfs_lse(best_seq, t3, T_stages=T_stages, maxiter_total=8000)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Second LP pass on refined result ===
    print(f"\n=== Phase 4: LP Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = rem * 0.18
    if t4 > 30:
        print(f"  Budget: {t4:.0f}s")
        a_lp2, _ = nesterov_lp_optimize(best_seq, t4, solver, max_iters=80000, stall_patience=10)
        try_update(a_lp2, "nlp2")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coord Descent, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = rem * 0.12
    if t5 > 15:
        a_cd, _ = coord_descent_polish(best_seq, t5, n_steps=60000)
        try_update(a_cd, "cd")

    # === Phase 6: Multi-start diversity ===
    print(f"\n=== Phase 6: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 60.0)
        for pi in range(4):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            x = np.linspace(0, 2 * np.pi, n_cur)
            if pi == 0:
                pert = 0.04 * np.cos(2 * x) + 0.02 * np.sin(3 * x)
            elif pi == 1:
                pert = 0.03 * rng.standard_normal(n_cur)
            elif pi == 2:
                pert = 0.03 * np.sin(5 * x) + 0.01 * np.cos(7 * x)
            else:
                pert = 0.05 * np.cos(x) - 0.02 * np.cos(2 * x)
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            # Short LP + L-BFGS
            a_res, _ = nesterov_lp_optimize(a_pert, t_each * 0.5, solver,
                                             max_iters=15000, stall_patience=8)
            try_update(a_res, f"perturb_lp_{pi}")
            rem = deadline - time.monotonic()
            if rem > 10:
                a_res2 = lbgfs_lse(a_res, min(rem * 0.3, 15.0),
                                    T_stages=[3.0, 1.5, 1.0, 0.5, 0.3, 0.15, 0.08],
                                    maxiter_total=3000)
                try_update(a_res2, f"perturb_lb_{pi}")

    # === Phase 7: Alternative length ===
    print(f"\n=== Phase 7: Alt length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        for alt_factor in [1.25, 0.8]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            alt_n = int(n_cur * alt_factor)
            alt_n = max(200, min(3000, alt_n))
            t_alt = min(rem * 0.35, 100.0)
            if t_alt < 20:
                continue
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_optimize(init_alt, t_alt * 0.6, solver,
                                             max_iters=20000, stall_patience=8)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_alt2 = lbgfs_lse(a_alt, min(rem * 0.3, 20.0),
                                    T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                    maxiter_total=3000)
                try_update(a_alt2, f"lb_n{alt_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 30.0)
        a_fin = lbgfs_lse(best_seq, t_fin,
                           T_stages=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01],
                           maxiter_total=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_cd2, _ = coord_descent_polish(best_seq, min(rem * 0.6, 20.0), n_steps=30000)
        try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
