def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build matrix T such that (T @ b)[k] = conv(a_fixed, b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def project_simplex(v):
    """Project vector onto probability simplex (non-negative, sum=1)."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Shape the spectrum of a towards flatness.

def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max_k conv(a,a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 500 == 0 and it > 0:
            stall_count += 1
            if stall_count >= 8:
                # Random restart perturbation
                rng = np.random.default_rng(it * 13 + 37)
                noise = rng.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
                
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP-Spec iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def mirror_descent_spectral(a0, time_limit, n_active_frac=0.05, spectral_every=200):
    """Mirror descent with periodic spectral shaping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.2
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b = compute_subgradient(a, n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        # Backtracking line search
        alpha = lr
        improved = False
        for bt in range(8):
            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()

            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 2.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.9, 0.01)
            
            # Periodic spectral shaping
            if no_improve == spectral_every:
                a_spec = spectral_shaping(a, alpha=0.6)
                score_spec = 2.0 * n * fast_convolve(a_spec).max()
                if score_spec < best_score:
                    best_score = score_spec
                    best_a = a_spec.copy()
                    a = a_spec.copy()
                    z = z_from_a(a)
                    no_improve = 0
                    lr = 0.2
            elif no_improve >= 50:
                # Random perturbation
                rng = np.random.default_rng(step * 31 + 7)
                noise = rng.random(n) * 0.02
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.2

    return best_a, best_score


def peak_shaving_local_search(a0, time_limit, n_steps=50000):
    """Local search focused on equalizing top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_k = sorted_idx[:min(8, len(b))]
        top_vals = b[top_k]
        
        best_move_score = best_score
        best_move_a = None
        
        for k in top_k[:4]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Try moving to various destinations
            for _ in range(5):
                dst = rng.integers(n)
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_move_score:
                        best_move_score = score
                        best_move_a = a_trial.copy()

        if best_move_a is not None:
            a = best_move_a
            best_score = best_move_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Larger random perturbation
                idx = rng.choice(n, min(n//5, 100), replace=False)
                a_trial = a.copy()
                a_trial[idx] *= rng.uniform(0.3, 1.5, len(idx))
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15 and ss < 1e6:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                if stall > 500:
                    break

    return best_a, best_score


def lse_lbfgs(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=10):
    """L-BFGS with LogSumExp approximation."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = np.min(Tb.max(), 700)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = np.min(Tb.max(), 700)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    from scipy.optimize import minimize
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        maxiter = max(20, int(remaining * 3))
        try:
            res = minimize(
                fun=lambda z_in: lse_objective(z_in, stage_T),
                x0=z,
                jac=lambda z_in: lse_jacobian(z_in, stage_T),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
