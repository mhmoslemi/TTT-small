"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Saddle-point (Hedge vs QP best-response) with Hankel-structured fast matvec
2. Exponentiated gradient with adaptive softmax smoothing
3. Smart coordinate descent with exact per-coordinate 1D minimax
4. Fine-grained coordinate descent for final polishing
5. Multi-length exploration with interpolation

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def hankel_matvec(lam, a):
    """Compute (Q a)[i] = sum_j lam[i+j] * a[j] using FFT-based convolution."""
    n = len(a)
    a_flip = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    L = rfft(lam, fft_size)
    AF = rfft(a_flip, fft_size)
    conv_full = irfft(L * AF, fft_size)[:size]
    return conv_full[n - 1:2 * n - 1]

def compute_smooth_grad(a, beta):
    """Compute smoothed gradient via softmax over convolution."""
    n = len(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    scaled = beta * (conv - M)
    scaled = np.clip(scaled, -60, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    if Z < 1e-30:
        weights = np.zeros_like(conv)
        weights[int(np.argmax(conv))] = 1.0
        Z = 1.0
    weights = weights / Z
    g = 2.0 * hankel_matvec(weights, a)
    return g, M, weights

def qp_solve_pgd(lam, a_init, max_iters=3000, tol=1e-12):
    """Solve min a^T Q a s.t. a>=0, sum(a)=1 via projected gradient.
    Q[i,j] = lam[i+j]. Uses backtracking line search."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    lr = 2.0 / (np.max(np.abs(lam)) * 2 * n + 1e-10)
    lr = min(lr, 1.0)
    prev_obj = a @ hankel_matvec(lam, a)

    for it in range(max_iters):
        grad = 2.0 * hankel_matvec(lam, a)
        success = False
        for _ in range(50):
            a_new = simplex_project(a - lr * grad)
            obj_new = a_new @ hankel_matvec(lam, a_new)
            descent = lr * np.sum(grad * (a - a_new))
            if obj_new <= prev_obj - 1e-4 * descent:
                success = True
                break
            lr *= 0.5
        if not success:
            return a
        if prev_obj - obj_new < tol * max(1.0, abs(prev_obj)):
            a = a_new
            break
        prev_obj = obj_new
        a = a_new
    return a

def saddle_point_optimize(a0, time_limit, beta_init=80.0, eta_init=0.5, label=""):
    """Saddle-point: Hedge on lambda (max side) + QP best response (min side)."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    n_conv = 2 * n - 1
    lam = np.ones(n_conv) / n_conv

    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()

    avg_a = a.copy()
    avg_lam = lam.copy()
    it = 0
    beta = beta_init
    eta = eta_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Max side: update lambda via multiplicative weights based on current a's convolution
        conv_a = fast_convolve(a)
        shifted = beta * (conv_a - conv_a.max())
        shifted = np.clip(shifted, -60, 0)
        exp_w = np.exp(shifted)
        Z = exp_w.sum()
        if Z < 1e-30:
            exp_w = np.zeros(n_conv)
            exp_w[int(np.argmax(conv_a))] = 1.0
            Z = 1.0
        exp_w /= Z
        lam = (1.0 - eta) * lam + eta * exp_w
        lam = np.maximum(lam, 0)
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.ones(n_conv) / n_conv
        else:
            lam /= lam_sum

        # Min side: solve QP best response
        a_br = qp_solve_pgd(lam, a, max_iters=1500)

        # Nesterov averages
        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_br
        avg_lam = (1.0 - gamma_it) * avg_lam + gamma_it * lam

        # Evaluate candidates
        for cand in [a_br, avg_a]:
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * float(fast_convolve(cand_n).max())
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand_n.copy()
                no_improve = 0
            else:
                no_improve += 1

        a = a_br

        # Adaptive params
        if no_improve > 60:
            eta = min(eta * 1.05, 1.0)
            beta *= 1.02
            no_improve = 0
            if beta > 5000:
                break

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Saddle {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.0f}, eta={eta:.3f}")

    return best_a, best_score

def smart_coord_descent(a0, time_limit, label=""):

        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[k_start - i:k_end - i + 1] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai * 0.999]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.2 < d < ai * 0.5:
                        candidates.append(d)

            top_k = min(8, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.2 < d < ai * 0.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(0.1 * frac * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score

        else:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a7c, s7c = fine_coord_descent(best_seq, rem * 0.8, label="fp3")
        best_seq, best_score = check_and_update(a7c, best_seq, best_score, "phase7c")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
