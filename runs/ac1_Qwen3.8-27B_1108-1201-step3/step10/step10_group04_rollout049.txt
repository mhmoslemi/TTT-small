Let me focus on making the subgradient approach work well:

**Subgradient method with averaging:**
1. Start with `a_0` (some initialization)
2. At each step:
   a. Compute `c = conv(a_t, a_t)`
   b. Find active set: `A_t = {k : c_k >= (1-ε) * max(c)}`
   c. Compute subgradient: `g_t = (1/|A_t|) * sum_{k in A_t} ∇c_k`
      where `∇c_k = T_k @ a_t` (the cross-correlation of a with the "shifted" version)
   d. Compute step size: `alpha_t = 2 * (f(a_t) - f_lb) / ||g_t||^2` (Polyak step)
   e. Update: `a_{t+1} = project_simplex(a_t - alpha_t * g_t)`
   f. Update running average: `a_avg = (t * a_avg + a_{t+1}) / (t+1)`
3. Return the best of {a_t, a_avg}

The Polyak step size requires knowing f_lb. We have `f_lb = 2n/(2n-1)`.

Let me now think about the **deflation** approach, which is a novel idea:

**Deflation approach:**
1. Start with a good solution `a`.
2. Compute the convolution `c = conv(a, a)`.
3. Identify the top peaks: `k_1, k_2, ..., k_m` where `c_{k_i}` is large.
4. For each peak k_i, the contribution to the objective is `c_{k_i} = sum_j a[j] * a[k_i - j]`.
5. We want to reduce these peaks. One way: find a "deflation direction" d such that:
   - `d` reduces the peaks (i.e., `sum_j d[j] * a[k_i - j] + a[j] * d[k_i - j] < 0`)
   - `d` is "orthogonal" to the already-flat parts (doesn't increase other c_k)
   - `1^T d = 0` (preserves the sum constraint)
6. Move: `a_new = project_simplex(a - alpha * d)`

This is similar to the Lanczos algorithm used in signal processing for spectral flatness.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_from_z(z):
    z_shifted = z - z.max()

    s = exp_z.sum()

def compute_subgradient(a):
    """Compute subgradient of max_k c_k using active set near the max."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    if max_c < 1e-300:
        return np.zeros(n), max_c
    threshold = max_c - 0.05 * max_c
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_c


def subgradient_optimize(a0, time_limit, n_active_frac=0.05, use_averaging=True):
    """Subgradient method with Polyak step size and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    lb = lower_bound(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    avg_a = a.copy()
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        batch = 10 if n > 500 else 20
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            g, max_c = compute_subgradient(a)
            g_norm_sq = np.dot(g, g)
            if g_norm_sq < 1e-30:
                return best_a, best_score
            f_current = 2.0 * n * max_c
            gap = f_current - lb
            if gap < 1e-10:
                return best_a, best_score
            alpha = 2.0 * gap / g_norm_sq
            alpha = min(alpha, 0.5)
            a_new = project_simplex(a - alpha * g)
            if use_averaging:
                avg_a = (step * avg_a + a_new) / (step + 1)
            a = a_new
            step += 1

        f_a = 2.0 * n * fast_convolve(a).max()
        if f_a < best_score - 1e-12:
            best_score = f_a
            best_a = a.copy()

            f_avg = 2.0 * n * fast_convolve(avg_a).max()
            if f_avg < best_score - 1e-12:
                best_score = f_avg
                best_a = avg_a.copy()

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """Nesterov-accelerated fictitious play with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    beta = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        alpha = (beta - 1.0) / beta_new

        y = np.maximum(y + alpha * (y - a_prev), 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a_prev = a.copy()

        a = y + (alpha / beta_new) * (b_new * sy - y)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        beta = beta_new

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if it % 200 == 0 and it > 0:
            if best_score >= prev_best - 1e-8:
                stall += 1
            else:
                stall = 0
            prev_best = best_score

            if stall >= 8:
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall = 0

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def deflation_refine(a0, time_limit, n_peaks=3):
    """Spectral deflation: directly reduce top peaks by finding deflation directions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, None
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        current_score = 2.0 * n * max_c

        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:n_peaks]

        # For each top peak, compute the "pressure" vector
        # The gradient of c_k at a is: (grad c_k)[j] = a[k-j] (for valid indices)
        # We want to move in a direction that reduces all top peaks simultaneously
        pressure = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += a[k - idx]
        pressure /= n_peaks

        # Deflation direction: -pressure, projected to be orthogonal to 1
        # (to preserve sum constraint)
        d = -pressure + np.mean(pressure)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break

        # Try multiple step sizes
        improved = False
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.2]:
            a_trial = a + alpha * d / d_norm
            a_trial = project_simplex(a_trial)
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try random perturbation
            noise = rng.normal(0, 0.01, n)
            a_trial = project_simplex(a + noise)
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True

        if not improved and outer > 50:
            break

        if outer % 50 == 0 and outer > 0:
            elapsed = time.monotonic() - t0
            print(f"    Deflation outer={outer} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

            idx = rng.choice(n, k, replace=False)
            a_trial = a.copy()
            a_trial[idx] += rng.normal(0, T * 0.05, k)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
        elif move_type == 1:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                a_trial = a.copy()
                amt = a[i] * rng.uniform(0.01, 0.08)
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-10:
                    a_trial = a_trial / ss
                else:
                    a_trial = a.copy()
            else:
                a_trial = a.copy()
        else:
            i = rng.integers(0, n - 3)
            j = rng.integers(i + 2, n)
            a_trial = a.copy()
            a_trial[i:j + 1] = a_trial[i:j + 1][::-1]
            a_trial = a_trial / a_trial.sum()

        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        delta = score_trial - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(7 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1 - x))
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 6:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif choice == 7:
        # Oscillating window
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.2 * np.cos(10 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        # Flat with edge taper
        x = np.linspace(0, 1, n)
        taper = np.minimum(1, np.minimum(x, 1 - x) * n / 5)
        a = taper
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Run multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 1e-10:
            return
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  NEW BEST [{src}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    # Phase 1: Subgradient (fast, no LP)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.25, 120.0)
    if t1 > 10:
        a_out, _ = subgradient_optimize(best_a, t1, n_active_frac=0.05)
        try_update(a_out, f"subgrad_{label}")

    # Phase 2: Nesterov LP (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.45, 500.0)
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_a, t2, solver=solver, max_iters=50000)
        try_update(a_out, f"nesterov_{label}")

    # Phase 3: Deflation
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.15, 100.0)
    if t3 > 10:
        a_out, _ = deflation_refine(best_a, t3, n_peaks=3)
        try_update(a_out, f"deflate_{label}")

    # Phase 4: Another subgradient pass (to polish)
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.10, 60.0)
    if t4 > 5:
        a_out, _ = subgradient_optimize(best_a, t4, n_active_frac=0.03)
        try_update(a_out, f"subgrad2_{label}")

    # Phase 5: SA for exploration
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.05, 30.0)
    if t5 > 3:
        a_out, _ = simulated_annealing_short(best_a, t5, seed=int(time.monotonic() * 1000) % 100000)
        try_update(a_out, f"sa_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_limit = t_start + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get initial sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1000

    if parent is None:
        n_base = 1000
        best_seq = structured_init(n_base, rng)
        best_score = 2.0 * n_base * fast_convolve(best_seq).max()
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        nn = len(a)
        sc = 2.0 * nn * fast_convolve(a).max()
        # Also verify with evaluate_sequence
        sc_eval = evaluate_safe(a)
        if np.isfinite(sc_eval):
            sc = min(sc, sc_eval)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={nn}, t={elapsed:.0f}s)")

    # === Phase 1: Multi-length exploration ===
    print(f"\n=== Phase 1: Multi-length exploration (best={best_score:.6f}) ===")

    # Determine n values to try
    n_candidates = [n_base]
    for mult in [0.75, 0.85, 1.15, 1.3, 1.5]:
        ln = int(n_base * mult)
        if 200 <= ln <= 4000 and ln != n_base:
            n_candidates.append(ln)
    n_candidates = list(dict.fromkeys(n_candidates))  # deduplicate preserving order
    n_candidates.sort()

    # Allocate time: give more to n_base
    n_explore = min(len(n_candidates), 5)
    n_to_try = n_candidates[:n_explore]

    # Quick exploration: subgradient only (fast)
    for n_val in n_to_try:
        rem = time_limit - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.15, 60.0)
        if t_alloc < 10:
            break

        if n_val == n_base and best_seq is not None:
            init_a = best_seq.copy()
        else:
            init_a = interpolate_sequence(best_seq, n_val)

        print(f"  Quick explore n={n_val}, budget={t_alloc:.0f}s")
        a_out, _ = subgradient_optimize(init_a, t_alloc, n_active_frac=0.05)
        try_update_global(a_out, f"explore_n{n_val}")

    # Also try a different structured init at n_base
    rem = time_limit - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.1, 40.0)
        if t_alloc > 10:
            alt_init = structured_init(n_base, rng)
            print(f"  Alt init at n={n_base}, budget={t_alloc:.0f}s")
            a_out, _ = subgradient_optimize(alt_init, t_alloc, n_active_frac=0.05)
            try_update_global(a_out, f"altinit_n{n_base}")

    # === Phase 2: Full optimization at best n ===
    print(f"\n=== Phase 2: Full optimization (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit - time.monotonic()
    if rem > 100:
        t_alloc = min(rem * 0.55, 800.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t_alloc, solver, label=f"full_n{len(best_seq)}")
        try_update_global(a_out, f"full_n{len(best_seq)}")

    # === Phase 3: Try one more length if time allows ===
    print(f"\n=== Phase 3: Additional length (best={best_score:.6f}) ===")
    rem = time_limit - time.monotonic()
    if rem > 120:
        # Try the second-best n from candidates
        best_n_used = len(best_seq)
        alt_ns = [n for n in n_to_try if n != best_n_used]
        if alt_ns:
            n_alt = alt_ns[0]
            t_alloc = min(rem * 0.2, 200.0)
            if t_alloc > 60:
                print(f"  Trying n={n_alt}, budget={t_alloc:.0f}s")
                init_a = interpolate_sequence(best_seq, n_alt)
                a_out, _ = optimize_at_length(init_a, t_alloc, solver, label=f"alt_n{n_alt}")
                try_update_global(a_out, f"alt_n{n_alt}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit - time.monotonic()
    if rem > 30:
        # Deflation polish
        t1 = min(rem * 0.3, 60.0)
        if t1 > 5:
            a_out, _ = deflation_refine(best_seq, t1, n_peaks=5)
            try_update_global(a_out, "final_deflate")

        # Subgradient with small active set (fine-tuning)
        rem = time_limit - time.monotonic()
        t2 = min(rem * 0.3, 50.0)
        if t2 > 5:
            a_out, _ = subgradient_optimize(best_seq, t2, n_active_frac=0.02)
            try_update_global(a_out, "final_subgrad")

        # One more Nesterov LP pass
        rem = time_limit - time.monotonic()
        t3 = min(rem * 0.3, 60.0)
        if t3 > 10:
            a_out, _ = nesterov_lp_optimize(best_seq, t3, solver=solver, max_iters=10000)
            try_update_global(a_out, "final_lp")

        # SA micro-perturbation
        rem = time_limit - time.monotonic()
        t4 = min(rem * 0.1, 20.0)
        if t4 > 3:
            a_out, _ = simulated_annealing_short(best_seq, t4, seed=999)
            try_update_global(a_out, "final_sa")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            final_score = min(best_score, fs)
            print(f"\nFINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
            if fs < best_score:
                best_score = fs

    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in best_seq]
