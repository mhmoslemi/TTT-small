- `sum_j d_j * a_{k_i - j} = 0` for all `i` (make the derivative zero, i.e., don't change the active peaks to first order)
- `sum_j d_j = 0` (preserve normalization)

- `sum_j d_j * a_{k_i - j} = -ε` for all `i` (reduce each active peak by ε)
- `sum_j d_j = 0`

This is an underdetermined linear system (m+1 equations, n unknowns). The minimum-norm solution is `d = A^T (A A^T)^{-1} b` where `A` is the (m+1) × n matrix and `b` is the target vector.

Then we take a step: `a_new = normalize(a + α * d)` for a small step size α.

This is essentially a Gauss-Newton step for the equality-constrained problem.

```python
def active_set_newton(a, eps=1e-6, max_active=20):
    """
    Find active peaks and compute a Newton-type step to equalize them.
    """
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    
    # Find active peaks: those within eps of M
    active = np.where(c >= M - eps)[0]
    if len(active) > max_active:
        # Take the largest ones
        indices = np.argsort(-c[active])[:max_active]
        active = active[indices]
    
    m = len(active)
    if m == 0:
        return a, M
    
    # Build the gradient matrix A
    # A[i, j] = a[active[i] - j] if 0 <= active[i] - j < n, else 0
    # Plus a row of ones for the normalization constraint
    A = np.zeros((m + 1, n))
    for i, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[i, js:je+1] = a[k - je:k - js + 1][::-1]
    A[m, :] = 1.0  # normalization constraint
    
    # Solve minimum-norm solution to A * d = b
    # where b = [-eps, -eps, ..., -eps, 0] (reduce each active peak, maintain sum)
    b = np.zeros(m + 1)
    b[:m] = -0.001  # small reduction target
    b[m] = 0.0
    
    # d = A^T (A A^T)^{-1} b
    AA = A @ A.T
    try:
        sol = np.linalg.solve(AA, b)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(AA, b, rcond=None)[0]
    d = A.T @ sol
    
    # Ensure a + d >= 0
    alpha = 1.0
    if d.min() < 0:
        # Limit step to maintain non-negativity
        min_ratio = np.min(a[d < 0] / (-d[d < 0]))
        alpha = max(0, min_ratio) * 0.9  # safety factor
    
    a_new = a + alpha * d
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    
    return a_new, c.max()
```

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    cross = np.fft.ifft(np.conj(W) * A).real
    # cross[i] = sum_k w[k] * a[(i-k) mod fft_size]
    # For i in [0, 3n-3], and since w[0..2n-2] and a[0..n-1] are the only non-zero:
    # cross[i] = sum_{k: 0<=k<=2n-2, 0<=i-k<=n-1} w[k] * a[i-k]
    # = sum_{k: max(0,i-n+1)<=k<=min(i,2n-2)} w[k] * a[i-k]
    
    # We want grad_a[i] = sum_k w[k] * a[k-i] for i in [0, n-1]
    # Let j = k-i, so k = i+j, j in [0, n-1] and i+j in [0, 2n-2]
    # grad_a[i] = sum_{j=0}^{n-1} w[i+j] * a[j] (where w[i+j]=0 if i+j >= 2n-1)
    
    # This is the cross-correlation at lag i: sum_j a[j] * w[i+j]
    # In terms of the `cross` array above:
    # cross[i] = sum_k w[k] * a[i-k] = sum_{k: 0<=i-k<=n-1} w[k]*a[i-k]
    # Let m = i-k, so k = i-m: cross[i] = sum_{m: 0<=m<=n-1, 0<=i-m<=2n-2} w[i-m]*a[m]
    # = sum_{m=0}^{min(n-1,i)} w[i-m]*a[m] (since i < n < 2n-2 always)
    
    # But we want: grad_a[i] = sum_{j=0}^{n-1} w[i+j]*a[j]
    # = sum_{j=0}^{n-1-i+i} ... 
    
    # Hmm, let me just define it differently.
    # grad_a[i] = sum_k w[k] * a[k-i] where a[k-i] = 0 if k-i notin [0,n-1]
    # = sum_{k=i}^{i+n-1} w[k] * a[k-i] (since k-i must be in [0,n-1], so k in [i, i+n-1])
    # Also k must be in [0, 2n-2], so k in [i, min(i+n-1, 2n-2)]
    
    # This is: grad_a[i] = sum_{j=0}^{n-1} w[i+j] * a[j] (with w[k]=0 for k>=2n-1)
    
    # In terms of cross-correlation: this is (a cross-correlated with w) at lag -i
    # Or equivalently, it's the "reversed" cross-correlation.
    
    # Let me just compute it directly for clarity (n=384, so O(n^2) is fine):
    grad_a = np.zeros(n)
    for i in range(n):
        # grad_a[i] = sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
        max_j = min(n-1, 2n-2-i)
        if max_j >= 0:
            grad_a[i] = np.dot(w[i:i+max_j+1], a[:max_j+1])
    
    grad_a *= 2.0 / t
    
    # Chain rule for softmax: grad_z = (I - outer(a,a)) @ grad_a
    grad_z = grad_a - np.sum(grad_a * a) * a
    
    return obj, grad_z

`x = w_ext` has length `2n-1` (non-zero part), `y = a` has length `n`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization in softmax
parameterization with a temperature continuation method. The log-sum-exp
smooths the max function, allowing gradient-based optimization. Multiple
sequence lengths and initializations are tried to avoid local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lsq_obj_grad(z, t):
    """
    Compute t*logsumexp(conv(softmax(z))/t) and its gradient w.r.t. z.
    """
    a = softmax(z)
    n = len(a)
    c = fast_convolve(a)
    
    # Objective
    c_scaled = c / t
    c_max = c_scaled.max()
    lse = c_max + np.log(np.exp(c_scaled - c_max).sum())
    obj = t * lse
    
    # Softmax weights over convolution entries
    w = np.exp(c_scaled - c_max)
    w /= w.sum()
    
    # Gradient w.r.t. a: (2/t) * crosscorr(w, a)
    # Using FFT: ifft(conj(fft(w)) * fft(a))[:n]
    fft_size = 1 << (3 * n - 3).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(W) * A, fft_size)
    grad_a = cross[:n] * (2.0 / t)
    
    # Chain rule: softmax Jacobian
    # d(a_i)/d(z_j) = a_i * (delta_ij - a_j)
    # grad_z_j = sum_i grad_a_i * d(a_i)/d(z_j) = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    s = np.sum(grad_a * a)
    grad_z = a * (grad_a - s)
    
    return obj, grad_z


def lsq_obj_only(z, t):
    """Objective only (for verification)."""
    a = softmax(z)
    c = fast_convolve(a)
    c_scaled = c / t
    c_max = c_scaled.max()
    lse = c_max + np.log(np.exp(c_scaled - c_max).sum())
    return t * lse


def optimize_with_lnbfgs(a_init, t_schedule, max_iter_per_t=200, tol=1e-10):
    """
    Run L-BFGS with temperature continuation.
    a_init: initial point (non-negative, will be normalized)
    t_schedule: list of temperatures, from high to low
    """
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    
    n = len(a)
    z = np.log(a + 1e-300)  # log parameterization
    
    best_obj = np.inf
    best_a = a.copy()
    
    for t in t_schedule:
        result = minimize(
            lsq_obj_grad, z, args=(t,),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter_per_t,
                'ftol': tol,
                'gtol': 1e-8,
                'maxfun': 1000,
            }
        )
        z = result.x
        a_new = softmax(z)
        c_new = fast_convolve(a_new)
        max_c = c_new.max()
        
        if max_c < best_obj:
            best_obj = max_c
            best_a = a_new.copy()
        
        # Update z for next temperature (warm start)
        # Actually, z is already at the optimum for this t
        # Just continue to next t
    
    return best_a, best_obj

def _optimize_with_time(a_init, t_schedule, deadline, time_limit, holder):

    best_obj = np.inf
    best_a = a.copy()

            break
        if time.monotonic() > deadline - 50:

        result = minimize(

def _generate_inits(a_best, rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine taper
    c = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
    c = c ** 1.5
    inits.append(c / c.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.3*n))**2)
    inits.append(g / g.sum())
    
    # Edge-enhanced (U-shaped)
    u = (x - 0.5*(n-1))**2 / (0.5*n)**2
    ue = 0.5 + np.exp(u)
    inits.append(ue / ue.sum())
    
    # Sine
    s1 = np.sin(np.pi * (x + 0.5) / n)
    inits.append(s1 / s1.sum())
    
    # Perturbed best
    for scale in [0.01, 0.05, 0.1]:
        p = a_best + rng.standard_normal(n) * scale
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        d = rng.dirichlet(np.ones(n) * alpha)
        inits.append(d)
    
    return inits


def _coordinate_descent(a_init, time_limit=30):
    """Coordinate descent: try multiplicative perturbations on random coordinates."""
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    factors = [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5,
               0.75, 1.0/0.75, 0.9, 1.0/0.9, 0.6, 1.0/0.6]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    # Try the other direction too
                    break
        
        if not improved:
            # Try all coordinates with a few factors
            any_improved = False
            for i in range(n):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-15:
                    continue
