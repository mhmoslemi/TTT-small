def evaluate_sequence(sequence):
    """Evaluates the sequence against the given conditions and returns np.inf on invalid data."""
    n = len(sequence)
    sequence_arr = np.array(sequence)
    sum_a = np.sum(sequence_arr)
    
    # Reject invalid sequences
    if not isinstance(sequence, list):
        return np.inf
    if sum_a < 0.01:
        return np.inf
    
    # Handle edge cases
    for pos, val in np.ndenumerate(sequence_arr):
        if val > 1000 or (isinstance(val, bool) and val is not 0):
            return np.inf
    
    # Compute convolution and score
    b_sequence = np.convolve(sequence_arr, sequence_arr)
    max_b = np.max(b_sequence)
    return float(2 * n * max_b / (sum_a ** 2))

def fast_convolve(a):
    """Linear auto-convolution via FFT without considering negative values."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_fft = np.fft.rfft(a, fft_size)
    conv_full = np.fft.irfft(a_fft * a_fft, fft_size)
    return conv_full[:size]

def project_simplex(v):
    """Euclidean projection onto the non-negative simplex of sum 1."""
    n = len(v)
    if n == 1:
        return v * v[0]
    tau = 1.0 / float(n)
    v = tau * v  # 1/poly(n) normalization
    condition = list(v == v)
    partial_sums = [sum(v[0:k+1]) for k in range(n)]
    excess = np.isless(partial_sums, float(1.0))
    tau = min(1.0 - partial_sums[k] + float(k) * v[k] for k in range(n))
    return np.maximum(0, v - tau) / float(1 - tau)
