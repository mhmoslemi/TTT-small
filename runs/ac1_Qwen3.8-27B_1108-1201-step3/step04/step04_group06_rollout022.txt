Let me also think about a "dual" approach. The problem is:
Minimize `t` subject to `conv(a,a)[k] <= t` for all `k`, `sum(a) = 1`, `a >= 0`.

The Lagrangian is:
`L(a, t, lambda, mu, nu) = t + sum_k lambda_k * (conv(a,a)[k] - t) + mu * (1 - sum(a)) + sum_i nu_i * (-a[i])`

The stationarity condition for `a_i` is:
`sum_k lambda_k * (a[k-i] + a[i-k]) - mu - nu_i = 0` (with appropriate bounds)

This gives: `a[i]` should be proportional to `sum_k lambda_k * (a[k-i] + a[i-k])` (when `a[i] > 0`).

This is a fixed-point equation! We can iterate:
1. Start with `a` and `lambda = softmax(c / temp)`
2. Update `a[i] = max(0, sum_k lambda[k] * (a[k-i] + a[i-k]))` (with bounds)
3. Normalize `a`
4. Update `lambda = softmax(c / temp)`
5. Repeat

This is essentially the EG method, but with a more structured update.

Actually, I realize this is exactly what the EG method does! The subgradient of `max(c)` at the active set is `sum_k w[k] * grad(c[k])`, and `grad(c[k])[i] = a[k-i] + a[i-k]` (for valid indices). The EG update `a *= exp(-eta * sg)` is the multiplicative version of the projected gradient step.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved exponentiated gradient
method with adaptive temperature, running averages, coordinate descent polishing,
and multi-scale exploration. Key improvements over previous approach:
1. Adaptive temperature with periodic "heating" to escape local minima

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute the objective for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.003, temp_decay=0.9997,

    Exponentiated gradient with adaptive temperature and periodic heating.
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    avg_a = np.zeros(n, dtype=np.float64)
    count = 0
    
    eta = eta0
    temp = temp0
    base_temp = temp0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax weights over top-k lags

            peek = c[indices]

            indices = np.arange(len(c))
            peek = c.copy()
        
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(indices)):
            wi = w[idx]
            if wi < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * wi * a[k - je:k - js + 1][::-1]
        
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        # Multiplicative update
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:

        else:

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        avg_a += (a_new - avg_a) / count

            no_improve = 0
        else:

        c = c_new

                best_a = avg_a.copy()

        temp *= temp_decay
        
        # Periodic heating

            c = fast_convolve(a)

            eta = eta0 * 0.3

            count = 0

        if c_avg.max() < best_obj:

            best_a = avg_a.copy()

def coordinate_descent(a0, time_limit, n_inner=10, verbose=False):
    """
    Coordinate descent polishing. For each coordinate, try several multiplicative
    factors and keep the best.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    factors = [0.5, 0.7, 0.85, 0.92, 1.0/0.92, 1.0/0.85, 1.0/0.7, 1.5,
               0.6, 1.0/0.6, 0.8, 1.0/0.8, 0.95, 1.0/0.95]
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Random subset of coordinates
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-12:
                continue
            
            best_factor = 1.0
            best_local_obj = best_obj
            
            for factor in factors:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_local_obj:
                    best_local_obj = obj2
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] = old_val * best_factor
                a = a / a.sum()
                c = fast_convolve(a)
                obj = c.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = old_val
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a.copy(), best_obj


def spectral_flatten(a0, time_limit, verbose=False):
    """
    Spectral domain optimization: flatten the power spectrum |FFT(a)|^2
    to reduce the peak auto-correlation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    eta = 0.5
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute circular auto-correlation via FFT
        N = 1 << (2 * n - 1).bit_length()
        F = np.fft.rfft(a, N)
        P = np.abs(F) ** 2
        
        # Circular auto-correlation
        r = np.fft.irfft(P, N)[:2*n-1]
        
        # Find peaks
        r_max = r.max()
        if r_max < 1e-15:
            break
        
        # Update: in frequency domain, reduce power at frequencies
        # that contribute to the peak
        # r[k] = sum_j P[j] * exp(2*pi*i*j*k/N)
        # To reduce r at peak lag, we can reduce P at all frequencies
        # proportionally, but weighted by the peak location
        
        # Simpler: iterate in time domain
        # For each coordinate, compute its contribution to the peak
        peak_lags = np.where(r > 0.8 * r_max)[0]
        
        sg = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += a[k-je:k-js+1][::-1]
        
        if sg.max() < 1e-15:
            break
        
        # Gradient step
        a_new = a - eta * sg / sg.max()
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:

            eta *= 0.95
            if eta < 1e-8:
                break
        
        if it % 1000 == 0 and verbose:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}, eta={eta:.6f}")
    
    return best_a.copy(), best_obj


def generate_inits(rng, n, best_a=None):

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
