            best_obj = obj_exact
            best_a = a.copy()

        s *= s_factor
        if s < s_min:
            s = s_min

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    c = -np.ones(n)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': 30, 'feasibility_tolerance': 1e-9})
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def lp_refine(a0, time_limit, max_iters=200):
    """LP-based iterative refinement with line search."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    obj_current, conv = compute_obj(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = obj_current

    t0 = time.monotonic()
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        M_lp = M * 0.9995
        g, g_sum = solve_lp_direction(a, M_lp)

        if g_sum <= 1.0 + 1e-7 or np.linalg.norm(g - a) < 1e-14:
            no_improve += 1
            M_lp = M * 1.001
            if M_lp > M * 1.01:
                break
            continue
        else:
            d = g - a
            # Simple line search
            best_t = 0.0
            best_trial = obj_current
            for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                a_trial = np.maximum(a + t * d, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                obj_trial = compute_obj(a_trial)[0]
                if obj_trial < best_trial:
                    best_trial = obj_trial
                    best_t = t

            if best_t > 0 and best_trial < best_obj - 1e-12:
                a_new = np.maximum(a + best_t * d, 0.0)
                sn = a_new.sum()
                if sn > 1e-15:
                    a = a_new / sn
                _, conv = compute_obj(a)
                M = conv.max()
                obj_current = 2.0 * n * M
                if obj_current < best_obj:
                    best_obj = obj_current
                    best_a = a.copy()
                    no_improve = 0
            else:
                no_improve += 1

        iteration += 1

        if no_improve >= 30:
            # Perturb
            rng = np.random.default_rng(iteration * 17 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = project_and_normalize(best_a + noise)
            _, conv = compute_obj(a)
            M = conv.max()
            no_improve = 0

    return best_a, best_obj


def subgradient_step(a, conv, lr, momentum, v, top_k, temp):
    """One momentum subgradient step targeting top-k peaks of convolution."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]

    v = momentum * v - lr * sg
    a_new = a + v
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new, v


def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.95,
                       top_k=80, temp=0.00015):
    """Subgradient descent with momentum."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = compute_obj(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        a, v = subgradient_step(a, conv, lr, momentum, v, top_k, temp)
        _, conv = compute_obj(a)
        cur_obj = 2.0 * n * conv.max()
        step_count += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 30 == 0:
            lr *= 0.995
            temp *= 1.003

        if no_improve >= 300:
            rng = np.random.default_rng(step_count * 11 + 5)
            noise = rng.standard_normal(n) * 0.002
            a = project_and_normalize(best_a + noise)
            v = np.zeros(n)
            _, conv = compute_obj(a)
            no_improve = 0

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return project_and_normalize(new_seq)


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)

    # Antisymmetric (sine) - key insight: reduces middle convolution peak
    sine = 1.0 + 0.3 * np.sin(2.0 * np.pi * x / n)
    inits['antisym'] = project_and_normalize(sine)

    # Stronger antisymmetric
    sine2 = 1.0 + 0.5 * np.sin(2.0 * np.pi * x / n)
    inits['antisym2'] = project_and_normalize(sine2)

    # Cosine modulated (symmetric - keeps middle peak but reduces sidelobes)
    cos_mod = 1.0 + 0.3 * np.cos(4.0 * np.pi * x / n)
    inits['cosmod'] = project_and_normalize(cos_mod)

    # Ramp (maximally antisymmetric)
    ramp = x + 1.0
    inits['ramp'] = project_and_normalize(ramp)

    # Two-bump
    bump = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2) + \
           np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    inits['twobump'] = project_and_normalize(bump)

    # Gaussian
    gauss = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = project_and_normalize(gauss)

    # Random Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)

    # Half-sine window
    half_sine = np.sin(np.pi * x / (n - 1))
    inits['halfsine'] = project_and_normalize(half_sine)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: logsumexp continuation + LP + subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # ===== Phase 1: Quick multi-start scan =====
    print("=== Phase 1: Quick Multi-Start ===")
    t_start = time.monotonic()
    phase1_budget = min(200.0, (deadline - time.monotonic()) * 0.25)
    phase1_deadline = t_start + phase1_budget

    lengths = [768, 1024, 1536, 2048]
    init_names = ['parent', 'antisym', 'antisym2', 'cosmod', 'ramp', 'twobump', 'halfsine']
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_inits(n, parent, rng)
        available = [nm for nm in init_names if nm in inits]
        t_per = max(8.0, (phase1_deadline - time.monotonic() - 15) / (len(lengths) * len(available)))

        for name in available[:5]:
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = inits[name]
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break

            # Quick L-BFGS with moderate smoothness
            a_opt, _ = lgbfgs_soft_max(a0, s=0.02, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    print(f"  Top results: {[(f'{s:.6f}', n, nm) for s, n, nm, _ in scan_results[:5]]}")

    # ===== Phase 2: Logsumexp continuation (main innovation) =====
    print("=== Phase 2: Logsumexp Continuation ===")
    t_start2 = time.monotonic()

    # Pick top 3-4 distinct (length, init) for deep optimization
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt.copy()))
            seen.add(key)
        if len(top_configs) >= 4:
            break

    # Also ensure we have the best
    if best_sequence is not None:
        best_n = len(best_sequence)
        best_key = None
        for score, n, name, a_opt in scan_results:
            if np.allclose(a_opt, best_sequence, atol=1e-6):
                best_key = (n, name)
                break
        if best_key and best_key not in seen:
            top_configs.append((best_n, 'best', best_sequence.copy()))

    remaining_for_p2 = deadline - time.monotonic() - 450.0
    if remaining_for_p2 < 120:
        remaining_for_p2 = 120
    t_per_cont = max(40.0, remaining_for_p2 / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 430:
            break
        t_alloc = min(t_per_cont, deadline - time.monotonic() - 400)
        if t_alloc < 20:
            break

        print(f"  Continuation {i} (n={n}, {name}), budget={t_alloc:.0f}s")
        a_opt, obj_opt = logsumexp_continuation(a0, time_limit=t_alloc,
                                                 s_start=0.04, s_min=0.0003,
                                                 s_factor=0.55, min_time_per_stage=4.0)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    Improved: {score:.6f}")
        else:
            print(f"    Score: {score:.6f} (no improvement)")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: LP refinement =====
    print("=== Phase 3: LP Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 250.0
        if remaining > 30:
            # Run LP on the best, and on a few perturbed versions
            n_best = len(best_sequence)
            lp_configs = [best_sequence.copy()]

            # Add a couple of perturbations
            for pert_scale in [0.005, 0.01]:
                noise = rng.standard_normal(n_best) * pert_scale
                lp_configs.append(project_and_normalize(best_sequence + noise))

            t_per_lp = min(80.0, remaining / (len(lp_configs) + 0.5))

            for li, a0 in enumerate(lp_configs):
                if time.monotonic() > deadline - 200:
                    break
                t_alloc = min(t_per_lp, deadline - time.monotonic() - 180)
                if t_alloc < 10:
                    break

                a_opt, obj_opt = lp_refine(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP {li}: {score:.6f} (improved)")
                else:
                    print(f"  LP {li}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Subgradient + final polish =====
    print("=== Phase 4: Subgradient + Final Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 20:
            n_best = len(best_sequence)

            # Multiple SG configs
            sg_configs = [
                {'lr_init': 0.004, 'momentum': 0.95, 'top_k': 100, 'temp': 0.0001},
                {'lr_init': 0.006, 'momentum': 0.93, 'top_k': 60, 'temp': 0.0002},
                {'lr_init': 0.003, 'momentum': 0.96, 'top_k': 150, 'temp': 0.00008},
            ]

            t_per_sg = remaining / (len(sg_configs) + 2.0)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 60:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 40)
                if t_alloc < 8:
                    break

                a_opt, obj_opt = subgradient_polish(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {ci}: {score:.6f} (improved)")

            # Final quick L-BFGS polish with very small s
            if time.monotonic() < deadline - 20:
                t_alloc = min(40.0, deadline - time.monotonic() - 10)
                if t_alloc > 10:
                    a_opt, _ = lgbfgs_soft_max(best_sequence, s=0.0005, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final L-BFGS: {score:.6f} (improved)")

            # Final perturb + quick re-optimize loop
            if time.monotonic() < deadline - 15:
                n_perturb = max(1, int((deadline - time.monotonic() - 10) / 12))
                for pi in range(min(n_perturb, 8)):
                    if time.monotonic() > deadline - 10:
                        break
                    scale = 0.002 * (1.0 + 0.2 * pi)
                    noise = rng.standard_normal(n_best) * scale
                    # Add smooth component
                    smooth = np.sin(np.pi * np.arange(n_best) / n_best) * scale * 0.5
                    a_pert = project_and_normalize(best_sequence + noise + smooth)

                    t_alloc = min(10.0, deadline - time.monotonic() - 5)
                    if t_alloc < 4:
                        break

                    a_opt, _ = lgbfgs_soft_max(a_pert, s=0.001, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb {pi}: {score:.6f} (improved)")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        n_fb = 768
        best_sequence = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    time_used = time.monotonic() - (deadline - budget_s + 20)
    print(f"Time used: {time_used:.1f}s")

    return [float(x) for x in best_sequence]
