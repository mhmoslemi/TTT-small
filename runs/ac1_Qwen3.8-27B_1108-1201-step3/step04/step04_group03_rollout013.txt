def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                candidates.add(int(rng.integers(0, n)))
            # Edge positions (less contribution to middle peaks)
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 0.98]:
                candidates.add(int(frac * (n - 1)))
            candidates.discard(p)
            candidates = list(candidates)[:40]

            for r in candidates:
                max_delta = a[p] * 0.4
                if max_delta < 1e-14:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 1.0]:
                    delta = max_delta * frac
                    if delta < 1e-14:
                        continue
                    a_new = a.copy()
                    a_new[p] -= delta
                    a_new[r] += delta
                    if a_new[p] < -1e-14:
                        continue
                    a_new = np.maximum(a_new, 0.0)
                    conv_new = fast_convolve(a_new)
                    obj_new = conv_new.max()
                    if obj_new < best_move_obj:
                        best_move_obj = obj_new
                        best_move = (p, r, delta)

        # Also try moving FROM partner
        for ci in top_contribs[:3]:
            p = int(idx[ci])
            partner = k - p
            if partner >= 0 and partner < n and partner != p:
                candidates = set()
                for _ in range(10):
                    candidates.add(int(rng.integers(0, n)))
                for frac in [0.05, 0.15, 0.3, 0.5, 0.9]:
                    r = int(frac * (n - 1))
                    candidates.add(r)
                candidates.discard(partner)
                candidates = list(candidates)[:25]

                for r in candidates:
                    max_delta = a[partner] * 0.4
                    if max_delta < 1e-14:
                        continue
                    for frac in [0.1, 0.2, 0.3, 0.5, 1.0]:
                        delta = max_delta * frac
                        if delta < 1e-14:
                            continue
                        a_new = a.copy()
                        a_new[partner] -= delta
                        a_new[r] += delta
                        if a_new[partner] < -1e-14:
                            continue
                        a_new = np.maximum(a_new, 0.0)
                        conv_new = fast_convolve(a_new)
                        obj_new = conv_new.max()
                        if obj_new < best_move_obj:
                            best_move_obj = obj_new
                            best_move = (partner, r, delta)

        if best_move is not None:
            p_m, r_m, d_m = best_move
            a[p_m] -= d_m
            a[r_m] += d_m
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            cur_obj = conv.max()

            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_f = a.copy()
                no_improve = 0
                T = max(T * 0.995, 1e-7)
            else:
                no_improve += 1
                # SA acceptance
                if cur_obj - M > 0:
                    accept_prob = np.exp(-(cur_obj - M) / max(T, 1e-10))
                    if rng.random() > accept_prob:
                        a = best_f.copy()
                T *= 0.999
        else:
            no_improve += 1
            T *= 0.99

        # Periodic structural perturbation
        if no_improve > 40:
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Gaussian noise
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(best_f + noise)
            elif strategy == 1:
                # Block reversal
                blen = int(rng.integers(n // 8, n // 3))
                start = int(rng.integers(0, n - blen))
                a_new = best_f.copy()
                a_new[start:start + blen] = best_f[start + blen - 1:start - 1:-1]
                a = a_new / a_new.sum()
            elif strategy == 2:
                # Local swap
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 != i2:
                    a = best_f.copy()
                    a[i1], a[i2] = a[i2], a[i1]
                    a = a / a.sum()
                else:
                    a = best_f.copy()
            else:
                # Multi-point perturbation
                a = best_f.copy()
                for _ in range(int(rng.integers(2, 6))):
                    i1 = int(rng.integers(0, n))
                    i2 = int(rng.integers(0, n))
                    if i1 != i2 and a[i1] > 1e-10:
                        amt = a[i1] * rng.uniform(0.05, 0.3)
                        a[i1] -= amt
                        a[i2] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
            no_improve = 0
            T = 0.0005

    return best_f, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.002, mom=0.93, top_k=10, temp=0.0002):
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        tk = min(top_k, len(conv))
        peaks = np.argpartition(conv, -tk)[-tk:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.9998
        cur_temp *= 1.001
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0003)
            no_improve = 0
    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=12, temp_init=0.0004):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.9996
        cur_temp *= 1.002
        if no_improve > 80:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = best_f * np.exp(rng.standard_normal(n) * 0.003)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.01)
            cur_temp = temp_init * 2.0
            no_improve = 0
    return best_f, best_obj


def generate_structured_inits(n, parent, rng):
    inits = []
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    # Cosine edge
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    # Two-hump
    h1 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.15 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    # Parent resampled
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 384

    # === Phase 1: FW-LP on parent (~150s) ===
    print("\n=== Phase 1: FW-LP ===")
    rem = deadline - time.monotonic() - 1200
    if rem > 20 and parent is not None:
        t1 = min(rem, 150)
        print(f"  FW-LP on parent (n={n_work}), t={t1:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
        try_seq(a_opt, "fw_parent")

    # Try a couple of structured inits briefly
    inits = generate_structured_inits(n_work, parent, rng)
    rem = deadline - time.monotonic() - 1100
    for ii in range(1, min(4, len(inits))):
        rem = deadline - time.monotonic() - 1000
        if rem < 15:
            break
        t_alloc = min(rem, 30)
        print(f"  FW-LP on init {ii}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(inits[ii], time_limit=t_alloc)
        try_seq(a_opt, f"init_{ii}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Pair-move search (~600s) ===
    print("\n=== Phase 2: Pair-move Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t2 = min(rem, 600)
            print(f"  Pair-move search, t={t2:.0f}s")
            a_opt, _ = pair_move_search(best_seq, time_limit=t2, rng=rng)
            try_seq(a_opt, "pair_move_1")

        # Second round with fresh RNG
        rem = deadline - time.monotonic() - 350
        if rem > 30 and best_seq is not None:
            rng2 = np.random.default_rng(seed + 12345)
            t2b = min(rem, 300)
            print(f"  Pair-move search (round 2), t={t2b:.0f}s")
            a_opt, _ = pair_move_search(best_seq, time_limit=t2b, rng=rng2)
            try_seq(a_opt, "pair_move_2")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: FW-LP refinement on best found (~250s) ===
    print("\n=== Phase 3: FW-LP Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t3 = min(rem, 250)
            print(f"  Deep FW-LP, t={t3:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t3)
            try_seq(a_opt, "fw_deep")

        # Perturbed FW-LP
        rem = deadline - time.monotonic() - 150
        if rem > 20 and best_seq is not None:
            noise = rng.standard_normal(len(best_seq)) * 0.004
            f_pert = project_simplex(best_seq + noise)
            t3b = min(rem, 80)
            print(f"  FW-LP perturbed, t={t3b:.0f}s")
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t3b)
            try_seq(a_opt, "fw_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Multiplicative weight + Subgradient (~200s) ===
    print("\n=== Phase 4: MW + SG ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t4a = min(rem, 100)
            print(f"  MW optimize, t={t4a:.0f}s")
            a_opt, _ = mult_weight_optimize(best_seq, time_limit=t4a,
                                            lr_init=0.03, top_k=15, temp_init=0.0003)
            try_seq(a_opt, "mw_1")

        rem = deadline - time.monotonic() - 80
        if rem > 15 and best_seq is not None:
            t4b = min(rem, 60)
            print(f"  Subgradient polish, t={t4b:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, time_limit=t4b,
                                          lr_init=0.001, mom=0.94, top_k=12, temp=0.0001)
            try_seq(a_opt, "sg_1")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final fine polish (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        # Pair move with very small steps
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            rng3 = np.random.default_rng(seed + 999)
            t5 = min(rem, 50)
            print(f"  Final pair-move, t={t5:.0f}s")
            a_opt, _ = pair_move_search(best_seq, time_limit=t5, rng=rng3)
            try_seq(a_opt, "final_pair")

        # Very fine subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t5b = min(rem, 30)
            print(f"  Final fine SG, t={t5b:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, time_limit=t5b,
                                          lr_init=0.0003, mom=0.96,
                                          top_k=len(best_seq) // 4, temp=0.00003)
            try_seq(a_opt, "final_sg")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            for pi in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                t_pty = min(rem, 8)
                a_opt, _ = subgradient_polish(f_pert, time_limit=t_pty,
                                              lr_init=0.0002, mom=0.96,
                                              top_k=len(best_seq) // 3, temp=0.00002)
                try_seq(a_opt, f"pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
