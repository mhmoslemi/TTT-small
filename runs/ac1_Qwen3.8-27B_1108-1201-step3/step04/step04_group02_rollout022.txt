3. Coordinate-wise pattern search for fine polishing

import numpy as np
from scipy.optimize import linprog, minimize, project
import time

# Pre-computed structures for speed

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_onto_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(n, f):
    """Build the (2n-1) x n convolution matrix for fixed f."""
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        F[j:j + n, j] = f
    return F

def solve_lp_oracle(f, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    F = build_conv_matrix(n, f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=F, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    """Build the (2n-1) x n convolution matrix for fixed f."""
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        F[j:j + n, j] = f
    return F


def solve_lp_oracle(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_obj, active_set) or (None, None, None)."""
    n = len(f)
    F = build_conv_matrix(n, f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=F, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.success:
            g = res.x
            conv = F @ g
            active = np.where(conv >= M - 1e-8)[0]
            return g, res.fun, active
    except Exception:
        pass
    return None, None, None


def compute_smoothed_grad(f, T, top_k=None):

    else:

    grad = np.zeros(n)

def compute_smoothed_grad_fft(f, T, top_k=None):
    """Faster gradient computation using FFT.
    grad[m] = 2 * (conv(w, f)[m] + w[2m]*f[m] * indicator)
    where w[k] = softmax of conv."""
    n = len(f)
    conv = fast_convolve(f)
    
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(len(conv))
        vals = conv[idx]
        scaled = (vals - vals.max()) / T
        exps = np.exp(scaled)
        w[idx] = exps / exps.sum()
    else:
        scaled = (conv - conv.max()) / T
        exps = np.exp(scaled)
        w = exps / exps.sum()
    
    # Compute sum_k w[k] * f[k-m] for each m (cross-correlation)
    # This equals (w * f_rev)[m + n - 1] in convolution notation
    # Or more directly: let h[m] = sum_{k=m}^{m+n-1} w[k] * f[k-m]
    # h = conv(w, f)[m + n - 1] ... let me just use the direct formula
    
    # h[m] = sum_k w[k] * f[k-m] where k-m >= 0 and k-m < n
    # = sum_{j=0}^{n-1} w[j+m] * f[j] (with w=0 outside [0, 2n-2])
    # This is the cross-correlation: h = (w cross-correlate f)
    # = conv(w, f_flip) shifted appropriately
    fft_size = 1 << (2 * 2 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Ff_flip = np.fft.rfft(f[::-1].astype(np.float64), fft_size)
    # conv(w, f_flip) has length (2n-1) + n - 1 = 3n-1
    # h[m] = sum_j w[j+m] * f[j] = sum_j f[j] * w[j+m]
    # = (f * w)[m + n - 1] ... 
    # Actually let me think again. 
    # h[m] = sum_{j=0}^{n-1} f[j] * w[j+m]
    # Let u = j+m, then h[m] = sum_{u=m}^{m+n-1} f[u-m] * w[u]
    # This is the cross-correlation of f and w at lag m.
    # cross_corr(m) = sum_u f[u] * w[u+m]
    # = conv(f, w_flip)[m + len(w) - 1]
    # Hmm, this is getting confusing. Let me just compute it via FFT:
    # cross_corr_f_w(m) = sum_j f[j] * conj(w[j+m]) for real = sum_j f[j] * w[j+m]
    # = (f * w_rev)[m + (2n-2)]
    w_rev = w[::-1]
    Ff = np.fft.rfft(f, fft_size)
    Wrev = np.fft.rfft(w_rev, fft_size)
    conv_fw = np.fft.irfft(Ff * Wrev, fft_size)
    # conv_fw has length n + (2n-1) - 1 = 3n-1
    # cross_corr(m) = conv_fw[m + (2n-2)]
    h = np.zeros(n)
    offset = (2 * n - 2)
    for m in range(n):
        h[m] = conv_fw[m + offset]
    
    # Second term: w[2m] * f[m] for valid 2m
    t2 = np.zeros(n)
    for m in range(n):
        if 2 * m <= 2 * n - 2:
            t2[m] = w[2 * m] * f[m]
    
    grad = 2.0 * (h + t2)
    return grad


def logsumexp_obj(f, T):
    """Compute (1/T)*logsumexp(T*conv(f,f)) for f on simplex."""
    conv = fast_convolve(f)
    scaled = T * conv
    scaled -= scaled.max()
    return (scaled.max() + np.log(np.exp(scaled).sum())) / T

def sg_descent_fast(f0, time_limit, lr=0.005, top_k=100, temp=0.001, verbose=False):
    """Fast subgradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)
    mom = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Compute subgradient using top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Update with momentum
        v = mom * v - lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = project_simplex(f - 0.5 * lr * sg)
            no_improve += 1
        
        # Adaptive
        lr *= 0.9995
        temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 11 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            lr = 0.005
            temp = 0.001
            no_improve = 0
    
    return best_f, best_obj


def generate_structured_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramps
    r1 = np.arange(1, n + 1, dtype=np.float64)
    r2 = r1[::-1].copy()
    inits.append(r1 / r1.sum())
    inits.append(r2 / r2.sum())
    
    # Parabolic
    p = (x - n / 2) ** 2 + 1
    inits.append(p / p.sum())
    
    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Exponential
    e1 = np.exp(-x / (n / 4.0))
    e2 = np.exp(-x[::-1] / (n / 4.0))
    inits.append(e1 / e1.sum())
    inits.append(e2 / e2.sum())
    
    # Sine
    sn = np.sin(np.pi * x / n) + 0.1
    inits.append(sn / sn.sum())
    
    # Gaussian
    g_val = np.exp(-((x - n / 2) ** 2) / (2 * (n / 6) ** 2))
    inits.append(g_val / g_val.sum())
    
    # Two-peaked
    tp = np.exp(-((x - n * 0.3) ** 2) / (2 * (n / 10) ** 2)) + np.exp(-((x - n * 0.7) ** 2) / (2 * (n / 10) ** 2))
    inits.append(tp / tp.sum())
    
    # Cyclic (Barker-like): use period-based construction
    period = max(3, n // 50)
    cy = np.ones(n)
    for i in range(n):
        cy[i] = 1.0 if (i // period) % 2 == 0 else 0.3
    inits.append(cy / cy.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Parent-based
    if parent is not None:
        if len(parent) != n:
            px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        else:
            px = parent.copy()
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blends
            inits.append((0.7 * px / ps + 0.3 * np.ones(n) / n))
            inits.append((0.7 * px / ps + 0.3 * (r2 / r2.sum())))
    
    return inits


def load_parent_sequence():
    """Try to load the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def try_sequence(a, best_state, label=""):
    """Try a sequence and update best state if improved."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {best_state[0]:.8f})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function combining LP-FW, L-BFGS smoothing, and pattern search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    
    best_score = float('inf')
    best_seq = None
    
    def update_best(a, label=""):
        nonlocal best_score, best_seq
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
    
    # Load parent
    parent = load_parent_sequence()
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}")
    
    print(f"Target: 1.5027, Budget: {budget_s}s")
    print(f"Deadline in {deadline - time.monotonic():.0f}s from now")
    
    # === Phase 1: Quick optimization of parent ===
    print("\n=== Phase 1: Parent Optimization ===")
    if parent is not None:
        t1 = min(120, deadline - time.monotonic() - 500)
        if t1 > 20:
            print(f"  LP-FW on parent, t={t1:.0f}s")
            a_opt, obj = lp_fw_optimize(parent, time_limit=t1,
                                        m_multipliers=(0.97, 0.99, 1.0, 1.02))
            update_best(a_opt, "parent_lp")
            
            # SG follow-up
            rem = min(60, deadline - time.monotonic() - 400)
            if rem > 10:
                a_opt2, _ = sg_descent_fast(a_opt, time_limit=rem,
                                           lr=0.003, top_k=max(32, len(a_opt) // 10), temp=0.0005)
                update_best(a_opt2, "parent_sg")
    
    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length Exploration ===")
    
    lengths = [256, 384, 512]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
        lengths.sort()
    
    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        if parent is not None and n == len(parent):
            continue
        
        inits = generate_structured_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 200
        t_per_init = max(15, rem // (len(inits) * max(1, len(lengths))))
        t_per_init = min(t_per_init, 45)
        
        print(f"  n={n}, {len(inits)} inits, ~{t_per_init:.0f}s each")
        
        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break
            
            a_opt, obj = sg_descent_fast(f_init, time_limit=t_alloc,
                                        lr=0.008, top_k=max(32, n // 15), temp=0.001)
            update_best(a_opt, f"n{n}_i{i}")
            
            # Quick LP polish on best from this init
            rem2 = deadline - time.monotonic() - 80
            if rem2 > 15 and obj < best_score / (2 * n):
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem2, 25),
                                           m_multipliers=(0.98, 1.0, 1.03))
                update_best(a_opt2, f"n{n}_i{i}_lp")
    
    print(f"  After Phase 2: best={best_score:.8f}")
    
    # === Phase 3: Deep optimization ===
    print(f"\n=== Phase 3: Deep Optimization ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Iterate between LP-FW and L-BFGS
        for phase in range(6):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            
            if phase == 0:
                # LP-FW
                t_alloc = min(100, rem // 3)
                print(f"  Deep LP-FW round {phase}, t={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc,
                                           m_multipliers=(0.96, 0.98, 0.99, 1.0, 1.02, 1.05))
                update_best(a_opt, f"deep_lp_{phase}")
            elif phase in [1, 3, 5]:
                # SG descent
                t_alloc = min(80, rem // 3)
                print(f"  Deep SG round {phase}, t={t_alloc:.0f}s")
                a_opt, obj = sg_descent_fast(best_seq, time_limit=t_alloc,
                                            lr=0.003, top_k=max(50, n_best // 8), temp=0.0003)
                update_best(a_opt, f"deep_sg_{phase}")
            else:
                # LP-FW from perturbed start
                t_alloc = min(80, rem // 3)
                f_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
                print(f"  Deep LP-FW perturbed round {phase}, t={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(f_start, time_limit=t_alloc,
                                           m_multipliers=(0.97, 0.99, 1.0, 1.03))
                update_best(a_opt, f"deep_lp_p_{phase}")
        
        # L-BFGS smoothed approach
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            print(f"  L-BFGS smoothed, t={min(120, rem):.0f}s")
            a_opt, obj = lbfgs_smoothed(best_seq, time_limit=min(120, rem),
                                        T_start=30.0, T_end=1500.0)
            update_best(a_opt, "lbfgs")
    
    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Pattern search
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            print(f"  Pattern search, t={min(rem, 60):.0f}s")
            a_opt, obj = pattern_search(best_seq, time_limit=min(rem, 60))
            update_best(a_opt, "pattern1")
        
        # Fine LP-FW
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            print(f"  Fine LP-FW, t={min(rem, 50):.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=min(rem, 50),
                                       m_multipliers=(0.99, 0.995, 1.0, 1.005, 1.01))
            update_best(a_opt, "fine_lp")
        
        # Another pattern search
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            print(f"  Final pattern search, t={min(rem, 30):.0f}s")
            a_opt, obj = pattern_search(best_seq, time_limit=min(rem, 30))
            update_best(a_opt, "pattern2")
        
        # Very fine SG
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, obj = sg_descent_fast(best_seq, time_limit=min(rem, 20),
                                        lr=0.001, top_k=n_best // 3, temp=0.00005)
            update_best(a_opt, "final_sg")
    
    # Fallback
    if best_seq is None:
        print("  WARNING: No valid sequence found, using fallback")
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
