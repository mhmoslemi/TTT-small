    L = 6.0 * n
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.5:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit))
        T = T_start + (T_end - T_start) * progress

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        grad, conv = soft_gradient(y, T, top_k=top_k)

        step = 1.0 / L
        a_new = project_simplex(y - step * grad)
        obj_new = compute_objective(a_new)

        if obj_new >= best_obj and obj_new >= compute_objective(a):
            step = step * 0.5
            a_new = project_simplex(y - step * grad)
            obj_new = compute_objective(a_new)

        t_k = t_new
        a_prev = a
        a = a_new

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        L *= 1.002
        if no_improve > 300:
            break

    return best_a, best_obj


def lp_refine(a0, time_limit, verbose=False):
    """LP-based iterative refinement with adaptive step sizes."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = compute_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no_improve = 120
    perturb_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit - 1.0:
        C = build_conv_matrix(a)

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9))
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': 3.0})
        except Exception:
            break

        if not res.success:
            no_improve += 10
            it += 1
        else:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()

            if g_sum < 1e-10:
                no_improve += 1
                it += 1
            else:
                d = g - a
                norm_d = np.linalg.norm(d)

                if norm_d < 1e-10:
                    no_improve += 1
                    it += 1
                else:
                    # Find best step size via quadratic model + verification
                    conv_ad = C @ g
                    peak_idx = int(np.argmax(conv))
                    da_peak = conv_ad[peak_idx] - M

                    if da_peak < -1e-12:
                        # Compute conv(d,d) at peak
                        js = max(0, peak_idx - n + 1)
                        je = min(n - 1, peak_idx)
                        dd_peak = np.dot(d[js:je + 1],
                                         d[peak_idx - je:peak_idx - js + 1][::-1])
                        if dd_peak > 1e-12:
                            t_quad = min(1.0, -da_peak / (2.0 * dd_peak))
                        else:
                            t_quad = 0.3
                    else:
                        t_quad = 0.1

                    # Try candidates around t_quad
                    candidates = [min(1.0, t_quad), t_quad * 0.7, t_quad * 0.5,
                                  t_quad * 0.3, 0.2, 0.1, 0.05, 0.03]
                    candidates = sorted(set([max(0.01, min(1.0, c)) for c in candidates]),
                                        reverse=True)

                    found_improvement = False
                    for t in candidates:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        st = a_try.sum()
                        if st < 1e-15:
                            continue
                        a_try = a_try / st
                        obj_try = compute_objective(a_try)
                        if obj_try < best_obj - 1e-11:
                            a = a_try
                            conv = fast_convolve(a)
                            M = conv.max()
                            best_obj = obj_try
                            best_a = a.copy()
                            no_improve = 0
                            found_improvement = True
                            break
                        elif obj_try < compute_objective(a) - 1e-11:
                            a = a_try
                            conv = fast_convolve(a)
                            M = conv.max()
                            no_improve = 0
                            found_improvement = True
                            break

                    if not found_improvement:
                        no_improve += 1

        it += 1
        if verbose and it % 200 == 0:
            print(f"      LP iter {it}, obj={best_obj:.8f}, M={M:.6f}, t={time.monotonic()-t0:.1f}s")

        # Handle stagnation
        if no_improve >= max_no_improve:
            scale = 0.0005 + 0.001 * min(perturb_count, 15)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(400, 120 + 40 * perturb_count)
            if perturb_count > 50:
                break

        # Periodic gradient boost
        if it % 200 == 0 and time.monotonic() - t0 < time_limit - 10:
            a_grad, obj_grad = fista_optimize(best_a, time_limit=3.0,
                                               T_start=50.0, T_end=200.0, top_k=40)
            if obj_grad < best_obj:
                best_obj = obj_grad
                best_a = a_grad.copy()
                a = best_a.copy()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.2)

    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] = s1 / s1.sum()

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] = s2 / s2.sum()

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    # Symmetric random
    half = n // 2
    hr = rng.dirichlet(np.ones(half) * 0.3)
    sym = np.zeros(n)
    sym[:half] = hr
    sym[half:] = hr[::-1]
    inits['sym_rand'] = sym / sym.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization combining LP and gradient methods.
    Returns best sequence found within budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, score={evaluate_sequence(p.tolist()):.6f}")
            ps = float(evaluate_sequence(p.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_sequence = p.copy()
    except Exception:
        pass

    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"No parent, starting uniform n=256: score={best_score:.6f}")

    # ============================================================
    # Phase 1: Quick gradient scan (4 minutes)
    # ============================================================
    print("\n=== Phase 1: Gradient Scan ===")
    t_phase1 = 240.0
    lengths = [512, 768, 1024]
    scan_results = []

    t1_start = time.monotonic()
    for n_len in lengths:
        if time.monotonic() > deadline - 600:
            break
        remaining_1 = deadline - time.monotonic() - 600
        if remaining_1 < 30:
            break

        inits = generate_inits(n_len, parent, rng)
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'alternating', 'sym_rand']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]

        tper = max(8.0, min(60.0, remaining_1 / (len(lengths) * 3)))

        for name in init_names:
            if time.monotonic() > deadline - 550:
                break
            t_alloc = min(tper, deadline - time.monotonic() - 500)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = fista_optimize(a0, time_limit=t_alloc,
                                         T_start=20.0, T_end=200.0, top_k=40)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n_len, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FISTA n={n_len} [{name}]: {score:.6f} *BEST*")
                else:
                    print(f"  FISTA n={n_len} [{name}]: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1_start:.0f}s). Best: {best_score:.6f}")

    # ============================================================
    # Phase 2: LP deep optimization (20 minutes)
    # ============================================================
    print("\n=== Phase 2: LP Deep Optimization ===")
    t2_start = time.monotonic()

    # Select top candidates for LP
    lp_targets = []
    seen = set()
    for score, n_len, name, a_opt in scan_results:
        if n_len not in seen:
            lp_targets.append((n_len, a_opt))
            seen.add(n_len)
        if len(lp_targets) >= 3:
            break

    # Always include current best
    if best_sequence is not None:
        bl = len(best_sequence)
        if bl not in seen:
            lp_targets.append((bl, best_sequence.copy()))
            seen.add(bl)
        elif bl in seen:
            # Replace with best if it's the overall best
            for i, (ln, ao) in enumerate(lp_targets):
                if ln == bl:
                    lp_targets[i] = (bl, best_sequence.copy())
                    break

    remaining_2 = deadline - time.monotonic() - 180.0
    if remaining_2 > 60:
        tper_lp = max(60.0, remaining_2 * 0.7 / max(1, len(lp_targets)))

        for i, (n_len, a0) in enumerate(lp_targets):
            if time.monotonic() > deadline - 160:
                break
            t_alloc = min(tper_lp, deadline - time.monotonic() - 140)
            if t_alloc < 30:
                break

            print(f"  LP target {i}: n={n_len}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_refine(a0, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP {i} (n={n_len}): {score:.6f} *BEST*")
                else:
                    print(f"  LP {i} (n={n_len}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2_start:.0f}s). Best: {best_score:.6f}")

    # ============================================================
    # Phase 3: Focused polish on best (5 minutes)
    # ============================================================
    print("\n=== Phase 3: Polish ===")
    t3_start = time.monotonic()
    if best_sequence is not None:
        remaining_3 = deadline - time.monotonic() - 60.0
        if remaining_3 > 30:
            # Multiple LP polish runs with perturbations
            n_polish = min(4, max(1, int(remaining_3 / 60)))
            tper_polish = remaining_3 / n_polish * 0.7

            for pi in range(n_polish):
                if time.monotonic() > deadline - 50:
                    break
                t_alloc = min(tper_polish, deadline - time.monotonic() - 40)
                if t_alloc < 15:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.0005 * (1 + pi)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s0 = a0.sum()
                    if s0 > 1e-15:
                        a0 = a0 / s0
                    else:
                        a0 = best_sequence.copy()

                a_opt, obj = lp_refine(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {pi}: {score:.6f} *BEST*")
                else:
                    print(f"  Polish {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t3_start:.0f}s). Best: {best_score:.6f}")

    # ============================================================
    # Phase 4: Final fine-tuning with gradient (remaining time)
    # ============================================================
    print("\n=== Phase 4: Fine-tuning ===")
    t4_start = time.monotonic()
    if best_sequence is not None:
        remaining_4 = deadline - time.monotonic() - 10.0
        if remaining_4 > 10:
            # Gradient refinement with very sharp smoothing
            a_opt, obj = fista_optimize(best_sequence, time_limit=remaining_4 * 0.6,
                                         T_start=100.0, T_end=500.0, top_k=20)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  GradFine: {score:.6f} *BEST*")
            else:
                print(f"  GradFine: {score:.6f}")

            # One more LP micro-polish
            remaining_4b = deadline - time.monotonic() - 10.0
            if remaining_4b > 15:
                a_opt, obj = lp_refine(best_sequence, time_limit=remaining_4b)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LPMicro: {score:.6f} *BEST*")

    print(f"Phase 4 done ({time.monotonic()-t4_start:.0f}s). Best: {best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
