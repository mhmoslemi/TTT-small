import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005, temp_decay=0.9997,
                eta_decay=0.99995, top_k_frac=0.25, patience=2000, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = eta0
    temp = temp0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Momentum buffer
    sg_mom = np.zeros(n)
    mom_decay = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax weights on active lags
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()
        
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
        
        # Momentum
        sg_mom = mom_decay * sg_mom + (1 - mom_decay) * sg
        sg_max = sg_mom.max()
        if sg_max < 1e-15:
            break
        
        # Multiplicative update
        log_update = -eta * sg_mom / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay
        
        # Adaptive: if stuck, reheat
        if no_improve > patience:
            rng_local = np.random.default_rng(it * 31 + 7)
            perturb = rng_local.standard_normal(n) * 0.003
            a = best_a + perturb
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.3
            temp = max(temp, temp0 * 0.05)
            sg_mom = np.zeros(n)
    
    return best_a.copy(), best_obj

def coord_descent_polish(a0, time_limit, max_rounds=50):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for _round in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Shuffle coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            for factor in [0.7, 0.85, 1.0/0.85, 1.0/0.7]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    break
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    break
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    break
                else:
                    a = best_a.copy()
                    c = fast_convolve(a)
                    break
        # c is already updated if we broke from inner loop
        c = fast_convolve(best_a)
    
    return best_a.copy(), best_obj

def generate_initializations(rng, n, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine edge
    c1 = np.abs(np.cos(np.pi * (x + 0.5) / n)) ** 2 + 0.01
    inits.append(c1 / c1.sum())
    
    # Cosine edge stronger
    c2 = np.abs(np.cos(np.pi * (x + 0.5) / n)) ** 4 + 0.01
    inits.append(c2 / c2.sum())
    
    # Gaussian center
    g = np.exp(-2.0 * ((x - 0.5 * (n - 1)) / n) ** 2)
    inits.append(g / g.sum())
    
    # Linear ramp
    r = (x + 1) / (n * (n + 1) / 2)
    inits.append(r)
    
    # Two-hump
    h = np.exp(-4.0 * ((x - 0.25 * (n-1)) / n) ** 2) + np.exp(-4.0 * ((x - 0.75 * (n-1)) / n) ** 2)
    inits.append(h / h.sum())
    
    # Sine
    s_val = np.abs(np.sin(np.pi * (x + 0.5) / n)) ** 2 + 0.01
    inits.append(s_val / s_val.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        d = rng.dirichlet(np.ones(n) * alpha / n)
        inits.append(d)
    
    # Parent (if same length)
    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)
            # Perturbed parent
            for scale in [0.001, 0.005, 0.01]:
                pp = p / ps + rng.standard_normal(n) * scale
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits.append(pp / pps)
    
    return inits

For a single coordinate change a[i] -> a[i] * factor, the change in the convolution is:

(a*a)[k] = sum_j a[j] * a[k-j]
If a[i] changes by δ:

- Term j=k-i: δ * a[i] (contribution δ * a[i] if k-i ≠ i)
- If k-i = i (i.e., k = 2i): the term is a[i]^2 -> (a[i]+δ)^2, change = 2a[i]δ + δ^2

So Δc[k] = δ * a[k-i] (for j=i term) + δ * a[i] (for j=k-i term) when k-i ≠ i.
= δ * (a[k-i] + a[i]) when k ≠ 2i and 0 ≤ k-i ≤ n-1.
= 2*a[i]*δ + δ^2 when k = 2i.

This allows O(1) update per convolution coefficient affected. The affected coefficients are those where k-i is in [0, n-1], i.e., k in [i, i+n-1]. There are at most n such coefficients.

So for each coordinate change, I can update the convolution in O(n) time (instead of O(n log n) for FFT). This is faster for small n.

But I also need to account for the normalization. After changing a[i], I renormalize: a_new = a / sum(a_new). This changes all coordinates, making the incremental update impossible.

score = 2 * n * max(conv) / (sum)^2

If I track sum and max(conv) incrementally, I can compute the score without full re-evaluation.
