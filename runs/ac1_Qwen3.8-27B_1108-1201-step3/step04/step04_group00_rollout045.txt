"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Log-sum-exp smoothed optimization with Nesterov acceleration
2. LP-based Frank-Wolfe direction finding
3. Multi-length exploration
4. Coordinate descent polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_and_grad(f, temp):
    """Compute log-sum-exp of auto-convolution and gradient w.r.t. f."""
    conv = fast_convolve(f)
    n = len(f)
    max_c = conv.max()
    lse = max_c + temp * np.log(np.sum(np.exp((conv - max_c) / temp)))
    
    # Weights
    w = np.exp((conv - max_c) / temp)
    w /= w.sum()
    
    # Gradient: grad_i = 2 * sum_j f_j * w_{i+j}
    # Using FFT correlation
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    grad = 2.0 * corr[:n]
    
    return lse, grad, conv


def lse_accelerated(f0, time_limit, temp_init, temp_final, 
                    lr_init, momentum_init, verbose=False):
    """
    Nesterov accelerated projected gradient on LSE-smoothed auto-conv.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    y = f.copy()  # extrapolated point
    beta = momentum_init
    cur_temp = temp_init
    cur_lr = lr_init
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute LSE and gradient at y
        lse_val, grad_y, conv_y = lse_and_grad(y, cur_temp)
        
        # Gradient step
        z = y - cur_lr * grad_y
        
        # Project onto simplex
        f_new = project_simplex(z)
        
        # Normalization check
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = best_f.copy()
        else:
            f_new = f_new / s_new
        
        # Update y for next iteration (Nesterov)
        beta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        y = f_new + (beta / beta_new - 1.0) * (f_new - f)
        y = np.maximum(y, 0.0)
        beta = beta_new
        
        # Evaluate actual objective
        conv_new = fast_convolve(f_new)
        actual_obj = conv_new.max()
        
        if actual_obj < best_obj - 1e-12:
            best_obj = actual_obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    LSE iter {it}: obj={best_obj:.10f}, temp={cur_temp:.6f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive schedule
        frac = min(1.0, it / max(1, int(time_limit * 10)))
        cur_temp = temp_init * (temp_final / temp_init) ** frac
        cur_lr = lr_init * (1.0 + it * 0.001) ** (-0.5)
        
        # Reset if stuck

            noise_scale = 0.002 + 0.003 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = f.copy()
            beta = momentum_init
            cur_lr = lr_init * 0.7
            no_improve = 0
        
        # Periodic exact evaluation
        if it % 50 == 0:

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_and_grad(f, temp):
    """Compute log-sum-exp of auto-convolution and gradient w.r.t. f."""
    conv = fast_convolve(f)
    n = len(f)
    max_c = conv.max()
    lse = max_c + temp * np.log(np.sum(np.exp((conv - max_c) / temp)))
    w = np.exp((conv - max_c) / temp)
    w /= w.sum()
    # grad_i = 2 * sum_j f_j * w_{i+j}
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    grad = 2.0 * corr[:n]
    return lse, grad, conv

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        return res.x
    return None

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9998
        temp *= 1.003
        if no_improve > 400:
            rng2 = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng2.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = max(0.001, lr * 2.0)
            no_improve = 0
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def eval_and_update(a, best_score, best_seq, label):
    """Evaluate sequence and return updated best if improved."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score, a.copy()
    return best_score, best_seq
