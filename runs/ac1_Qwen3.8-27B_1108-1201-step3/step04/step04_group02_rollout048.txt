import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a, b=None):
    """Linear auto/cross-convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, p):
    """
    Compute gradient of smooth max: sum_k p[k] * conv(a,a)[k]
    where p are the softmax weights.
    Returns gradient of shape (n,).
    """
    n = len(a)
    grad = np.zeros(n)
    # For each k, contribution is 2 * p[k] * a[k-m] for valid m
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        # a[k - m] for m in [m_lo, m_hi]
        idx = k - np.arange(m_lo, m_hi + 1)
        grad[m_lo:m_hi + 1] += 2.0 * pk * a[idx]
    return grad


def smooth_max_and_grad(a, s):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (value, gradient).
    """
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = s * (conv - max_c)
    # For numerical stability
    if shifted.max() > 700:
        shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    value = max_c + np.log(exp_shifted.sum()) / s  # This is actually (1/s)*log(sum exp(s*conv))
    # Wait: (1/s)*log(sum exp(s*conv[k])) = (1/s)*log(sum exp(s*(conv[k]-max_c)) * exp(s*max_c))
    # = max_c + (1/s)*log(sum exp(s*(conv[k]-max_c)))
    # = max_c + (1/s)*log(sum exp(shifted))
    value = max_c + np.log(exp_shifted.sum()) / s
    grad = compute_smooth_grad(a, p)
    return value, grad

    best_f = f.copy()

    y = f.copy()  # FISTA momentum variable
    t_k = 1.0

        if rem < 0.5:
            break

        val_y, grad_y = smooth_max_and_grad(y, s)

        L = n * s * f.sum()  # crude upper bound on Lipschitz
        # Actually, ||grad(f) - grad(g)|| <= L ||f-g||
        # For smooth max with temperature s, L ~ s * max(||a||_inf) * something

            if val_new <= val_ideal + 1e-10:

            alpha *= 0.5

        f = f_new

def solve_lp_oracle(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

For `a >= 0`, is `conv(a,a)[k] = sum_j a[j]*a[k-j]` convex?

The Hessian is `2*Q_k` where `Q_k[j,i] = 1` if `j+i = k` (and `j,i` in range), else 0.

For `Q_k` to be PSD, we need `x^T Q_k x >= 0` for all `x`. 
`x^T Q_k x = sum_{j+i=k} x[j]*x[i] = sum_j x[j]*x[k-j]` (for valid j).

This is the same as `conv(x, x)[k]`! So `x^T Q_k x = conv(x,x)[k]`. For this to be >= 0 for all `x`, we'd need `conv(x,x)[k] >= 0` for all `x`, which is NOT true (e.g., `x = [1, -1]` gives `conv[1] = -2`).

So the Hessian is not PSD, confirming nonconvexity.

However, on the subspace where `x >= 0`, we have `conv(x,x)[k] >= 0`, so the Hessian is PSD on the non-negative orthant (in the sense that `x^T Q_k x >= 0` for `x >= 0`). But this doesn't make the function convex on the non-negative orthant in the standard sense.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp gradient with FISTA (Nesterov) acceleration
2. LP-based Frank-Wolfe oracle for fine-tuning and escape
3. Multi-phase adaptive optimization with temperature scheduling
4. Diverse structured initializations and peak-shaving

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a, b=None):
    """Linear auto/cross-convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, p):
    """
    Compute gradient of sum_k p[k] * conv(a,a)[k].
    p: softmax weights over convolution positions (length 2n-1).
    Returns gradient of shape (n,).
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        idx = k - np.arange(m_lo, m_hi + 1)
        grad[m_lo:m_hi + 1] += 2.0 * pk * a[idx]
    return grad


def smooth_value_and_grad(a, s):
    """
    Compute (1/s)*log(sum exp(s*conv(a,a))) and its gradient.
    Returns (value, gradient).
    """
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = s * (conv - max_c)
    # Clip for numerical stability
    shifted = np.minimum(shifted, 500.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    value = max_c + np.log(exp_shifted.sum()) / s
    grad = compute_smooth_grad(a, p)
    return value, grad
