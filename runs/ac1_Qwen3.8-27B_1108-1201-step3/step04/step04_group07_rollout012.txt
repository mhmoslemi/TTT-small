"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed gradient descent with
L-BFGS-B, adaptive beta annealing, diverse structured initializations (edge-weighted,
harmonic, chirp), multi-length exploration, and active-set QP polishing.
Key innovations: (1) smooth surrogate via log-sum-exp for efficient BFGS optimization,
(2) edge-weighted and harmonic initializations exploiting convolution structure,
(3) adaptive beta schedule, (4) final Newton-style active-set refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives a * g (linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def compute_grad_lse(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] with respect to a.
    w should be normalized (sum to 1).
    Returns array of length n.
    """
    n = len(a)
    # (a*a)[k] = sum_j a[j]*a[k-j], d/da[j0] = a[k-j0]*1{0<=k-j0<n} + a[j0]*1{0<=k-j0<n, k!=2*j0}
    # Grad[j0] = sum_k w[k] * (a[k-j0]*1{0<=k-j0<n} + a[j0]*1{0<=k-j0<n, k!=2*j0})
    #
    # Term 1: sum_{d=0}^{min(n-1, 2n-2-j0)} a[d] * w[j0+d]  (cross-correlation of w and a)
    # Term 2: a[j0] * (sum_k w[k] - w[2*j0]*1{0<=2*j0<2n-1})
    
    fft_size = 1 << (3 * n - 1).bit_length()
    A_freq = np.fft.rfft(a, fft_size)
    W_freq = np.fft.rfft(w, fft_size)
    # Cross-correlation: ifft(conj(A) * W)[k] = sum_j a[j]*w[j+k] (circular)
    cross = np.fft.irfft(np.conj(A_freq) * W_freq, fft_size)
    term1 = cross[:n]  # term1[j0] = sum_d a[d]*w[d+j0]
    
    # Term 2
    w_sum = w.sum()
    s2 = np.zeros(n)
    idx_2j = 2 * np.arange(n)
    mask = idx_2j < (2 * n - 1)
    s2[mask] = w[idx_2j[mask]]
    term2 = a * (w_sum - s2)
    
    return term1 + term2


def lse_smooth_objective(a, beta, n):
    """Compute LSE-smoothed objective and its gradient."""
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    
    # Objective: 2n * LSE(beta*conv) / (beta * s^2)
    scaled = beta * conv
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    lse_val = max_s + np.log(exp_s.sum())
    obj = 2.0 * n * lse_val / (beta * s * s)
    
    # Weights for gradient
    w = exp_s / exp_s.sum()
    
    # Gradient of LSE(beta*conv) w.r.t. a:
    grad_lse = compute_grad_lse(a, w)
    
    # Full gradient: d/d a [2n * lse / (beta * s^2)]
    # = 2n/(beta) * [grad_lse / s^2 - 2*lse_val * a / s^3]
    # = 2n/(beta*s^2) * [grad_lse - 2*lse_val*a/s]
    grad = (2.0 * n / (beta * s * s)) * (grad_lse - 2.0 * lse_val * a / s)
    
    # Project gradient to respect sum constraint (approximately)
    grad = grad - np.sum(grad) * a / (s * s)
    
    return obj, grad


def exact_objective(a):
    """Compute exact objective 2n*max(conv)/s^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lse_optimize(a0, time_limit, beta_start=200, beta_end=5000, verbose=False):
    """
    Optimize using LSE-smoothed objective with L-BFGS-B and adaptive beta.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = exact_objective(a)
    
    t0 = time.monotonic()
    
    # Adaptive beta schedule: increase beta over time
    n_beta_steps = 8
    beta_schedule = np.geomspace(beta_start, beta_end, n_beta_steps)
    
    steps_per_beta = max(3, int(time_limit / n_beta_steps / 5))
    
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0.0)
            xx = x / x.sum() if x.sum() > 1e-15 else x
            val, gr = lse_smooth_objective(xx, beta, n)
            return val, gr
        
        def obj_only(x):
            val, _ = obj_grad(x)
            return val
        
        res = minimize(
            obj_grad, a, method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': steps_per_beta, 'gtol': 1e-8, 'maxls': 40}
        )
        
        a_trial = np.maximum(res.x, 0.0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial / s_trial
            obj_trial = exact_objective(a_trial)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE beta={beta:.0f}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def gradient_descent_max(a0, time_limit, lr_init=0.003, momentum=0.9, verbose=False):
    """
    Gradient descent on exact max objective with momentum and adaptive step size.
    Uses subgradient of max at current peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = exact_objective(a)
    v = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]
        
        # Subgradient of max at peak_idx: 2 * Q_{peak_idx} @ a
        # (Q_k @ a)[j] = a[k-j] if 0<=k-j<n else 0
        k = peak_idx
        sg = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] = 2.0 * a[k - j]
        
        # Also add constraint gradient for normalization
        sg_norm = sg - np.sum(sg) * a  # project to sum-preserving direction
        
        lr = lr_init / (1.0 + 0.01 * it)
        v = momentum * v - lr * sg_norm
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        obj_new = exact_objective(a_new)
        it += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb
                rng = np.random.default_rng(it * 7)
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                v = np.zeros(n)
                no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    GD iter {it}: obj={best_obj:.8f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj


def qp_polish(a0, time_limit, verbose=False):
    """
    QP-based polishing: at each step, solve a QP to reduce the peak.
    min (a - a_old)^T (a - a_old) s.t. (a*a)[k] <= M*(1-eps) for top peaks, sum(a)=1, a>=0
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = exact_objective(a)
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    top_k = 5
    eps = 0.001
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        M = conv[peaks].max()
        
        # QP: min sum_j (a_new[j] - a[j])^2 
        #     s.t. for each peak k: sum_j C[k,j] * a_new[j] <= M*(1-eps)
        #           sum(a_new) = 1, a_new >= 0
        #
        # The constraint (a_new*a_new)[k] <= M*(1-eps) is quadratic, not linear.
        # Linearize: (a_new*a_new)[k] ≈ (a*a)[k] + 2*sum_j C[k,j]*(a_new[j]-a[j])
        # where C[k,j] = a[k-j] if valid
        # So: 2*sum_j C[k,j]*a_new[j] <= M*(1-eps) - (a*a)[k] + 2*sum_j C[k,j]*a[j]
        #     sum_j C[k,j]*a_new[j] <= (M*(1-eps) - (a*a)[k])/2 + sum_j C[k,j]*a[j]
        
        # Build linearized constraints
        A_ub = np.zeros((top_k + 1, n))
        b_ub = np.zeros(top_k + 1)
        
        for pi, k in enumerate(peaks):
            # C[k, j] = a[k-j] if 0<=k-j<n
            row = np.zeros(n)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                row[j] = a[k - j]
            A_ub[pi] = row
            # Linearized: sum_j row[j] * a_new[j] <= (M*(1-eps) - conv[k])/2 + sum_j row[j]*a[j]
            b_ub[pi] = (M * (1 - eps) - conv[k]) / 2.0 + np.dot(row, a)
        
        # Sum constraint as equality (use two inequalities)
        A_ub[top_k] = np.ones(n)
        b_ub[top_k] = 1.0001
        
        # Objective: min ||a_new - a||^2 -> linear objective -2*a^T a_new (constant drops)
        c = -2.0 * a
        bounds = [(0, None)] * n
        
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        it += 1
        if result.success:
            a_new = np.maximum(result.x, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                obj_new = exact_objective(a_new)
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    # Relax: don't update a, but reduce eps
                    eps = max(0.0001, eps * 0.7)
        else:
            no_improve += 1
            eps = max(0.0001, eps * 0.7)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 13 + 42)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            eps = 0.001
            if it > 200:
                break
        
        if verbose and it % 20 == 0:
            print(f"    QP iter {it}: obj={best_obj:.8f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample sequence to new length via cubic interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    cs = CubicSpline(x_old, seq, bc_type='natural')
    new_seq = cs(x_new)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Edge-weighted (U-shaped)
    edge = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
    inits['edge'] = edge / edge.sum()
    
    # Double-edge
    de = 0.3 + 0.7 * np.cos(4.0 * np.pi * x / n) ** 2
    inits['double_edge'] = np.maximum(de, 0.01)
    inits['double_edge'] = inits['double_edge'] / inits['double_edge'].sum()
    
    # Sine
    sine = 0.6 + 0.4 * np.sin(np.pi * x / n) ** 2
    inits['sine'] = sine / sine.sum()
    
    # Chirp
    chirp = 0.5 + 0.5 * np.cos(2.0 * np.pi * 3.0 * x / n)
    inits['chirp'] = np.maximum(chirp, 0.05)
    inits['chirp'] = inits['chirp'] / inits['chirp'].sum()
    
    # Gaussian edges
    g1 = np.exp(-0.5 * (x / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - n) / (0.15 * n)) ** 2)
    gauss = g1 + g2 + 0.1
    inits['gauss_edges'] = gauss / gauss.sum()
    
    # Smooth random
    ir = np.abs(rng.standard_normal(n)).cumsum()
    ir = ir / ir[-1]
    diff = np.diff(ir, prepend=0)
    inits['smooth_rand'] = np.maximum(diff, 1e-8)
    inits['smooth_rand'] = inits['smooth_rand'] / inits['smooth_rand'].sum()
    
    # Triangular (inverted - peak at edges)
    tri = 1.0 - np.abs(2.0 * x / n - 1.0)
    inits['inv_tri'] = (1.0 - tri + 0.1) 
    inits['inv_tri'] = inits['inv_tri'] / inits['inv_tri'].sum()
    
    return inits

        # Final QP polish

            if obj_qp < obj_ls:

        if obj_ls < best_obj:

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    Uses LSE-smoothed L-BFGS-B, gradient descent, and QP polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, obj={exact_objective(parent):.6f}")
    except Exception:
        pass
    
    # Phase 1: Multi-length exploration
    print("=== Phase 1: Multi-length LSE Optimization ===")
    t_start = time.monotonic()
    
    # Priority lengths
    lengths = [768, 1024, 512, 2048]
    if parent is not None:
        p_len = len(parent)
        if p_len not in lengths:
            lengths.insert(0, p_len)
    
    # Time allocation: give more time to primary lengths
    phase1_budget = min(800.0, (deadline - time.monotonic()) - 400.0)
    time_weights = [0.35, 0.25, 0.20, 0.20]  # weights sum to 1
    time_alloc = []
    for i, n in enumerate(lengths):
        w = time_weights[i] if i < len(time_weights) else 0.15
        time_alloc.append(phase1_budget * w)
    # Normalize
    total_w = sum(time_alloc)
    time_alloc = [t * phase1_budget / total_w for t in time_alloc]
    
    results = []
    for i, n in enumerate(lengths):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(time_alloc[i], deadline - time.monotonic() - 300)
        if t_alloc < 30:
            break
        
        print(f"  Optimizing n={n}, budget={t_alloc:.0f}s")
        a_opt, obj_opt = optimize_length(n, parent, rng, time_limit=t_alloc)
        
        if a_opt is not None and obj_opt < np.inf:
            results.append((obj_opt, n, a_opt.copy()))
            if obj_opt < best_score:
                best_score = obj_opt
                best_sequence = a_opt.copy()
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    best_score = score
                print(f"    n={n}: obj={obj_opt:.6f}, eval={score:.6f}")
    
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on best 2-3 configurations
    print("=== Phase 2: Deep Refinement ===")
    t_start2 = time.monotonic()
    
    results.sort(key=lambda x: x[0])
    top_results = results[:3]
    
    remaining = deadline - time.monotonic() - 150.0
    if remaining > 30 and top_results:
        t_per_deep = remaining / max(1, len(top_results))
        
        for i, (obj_i, n_i, a_i) in enumerate(top_results):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_deep, deadline - time.monotonic() - 100)
            if t_alloc < 15:
                break
            
            print(f"  Deep refine {i} (n={n_i}, start obj={obj_i:.6f})")
            
            # Run LSE with higher beta range
            a_d, obj_d = lse_optimize(a_i, time_limit=t_alloc * 0.4,
                                      beta_start=2000, beta_end=20000)
            if obj_d < obj_i:
                a_i, obj_i = a_d, obj_d
            
            # Gradient descent
            t_rem = deadline - time.monotonic() - 80
            if t_rem > 10:
                a_gd, obj_gd = gradient_descent_max(a_i, time_limit=t_rem * 0.4,
                                                    lr_init=0.001, momentum=0.92)
                if obj_gd < obj_i:
                    a_i, obj_i = a_gd, obj_d
            
            # QP polish
            t_rem2 = deadline - time.monotonic() - 50
            if t_rem2 > 10:
                a_qp, obj_qp = qp_polish(a_i, time_limit=t_rem2)
                if obj_qp < obj_i:
                    a_i, obj_i = a_qp, obj_qp
            
            score = float(evaluate_sequence(a_i.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_i.copy()
                print(f"    Deep {i}: {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # Phase 3: Perturbation search
    print("=== Phase 3: Perturbation Search ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 20:
            n_best = len(best_sequence)
            n_perturb = max(1, int(remaining / 25.0))
            t_per = remaining / n_perturb
            
            for pi in range(n_perturb):
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 10)
                if t_alloc < 8:
                    break
                
                # Structured perturbation
                scale = 0.003 * (1.0 + 0.2 * pi)
                noise = rng.standard_normal(n_best) * scale
                # Add smooth component
                freq = rng.uniform(1, 5)
                smooth = np.sin(2 * np.pi * freq * np.arange(n_best) / n_best) * scale * 0.5
                a_pert = best_sequence + noise + smooth
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a_pert = a_pert / s_pert
                
                # Quick LSE optimization
                a_opt, obj_opt = lse_optimize(a_pert, time_limit=t_alloc * 0.6,
                                              beta_start=5000, beta_end=30000)
                
                # Quick QP
                t_rem = deadline - time.monotonic() - 10
                if t_rem > 5:
                    a_qp, obj_qp = qp_polish(a_opt, time_limit=t_rem)
                    if obj_qp < obj_opt:
                        a_opt, obj_opt = a_qp, obj_qp
                
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
