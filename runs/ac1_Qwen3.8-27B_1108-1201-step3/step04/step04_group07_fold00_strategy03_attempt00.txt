    phase_type = 'LP' if time_budget > 360 else 'SG'
    frequency_budget = 1.0 if phase_type=='LP' else 0.0

    # Prepare exponential cone
    initial_cone = np.ones(768)
    for i in range(30):
        initial_cone = np.log(initial_cone + 1e-3) * (0.9 if time_budget <0 else 1.0)
        if i % 15 ==0:
            initial_cone = initial_cone * (1.0 + 0.03*np.sin(2*np.pi*time_budget/max_time))

    # Primary iterative optimization
    timer_start = time()
    while time() < deadline - 90:
        if phase_type == 'LP':
            # Steps from LP-based method
            step_duration = 1.2
            evolution_rate = 0.002 * step_duration
            seq_temp = best_sequence * (0.9**i) if i<10 else best_sequence * (1.0 + 0.05*np.random.randn(1))
        elif phase_type == 'SG':
            step_duration = 1.0
            # SG step requires higher precision
            delta = np.diff(np.sort(np.random.rand(len(initial_cone))))
            best_sequence = best_sequence + evolution_rate * delta
            best_sequence = np.clip(best_sequence, 0.0, 1000.0)

        # Update the sequence and check score
        best_sequence = np.clip(best_sequence, 0.0, 1000.0)
        score_val = evaluate_sequence(best_sequence.tolist())
        if score_val < frequency_budget * best_score:
            best_score = score_val
            print(f"Iteration {i}: score now {frequency_budget*i:.6f} @ {step_duration}s per step")
        else:
            # Signal confusion or stop condition
            if phase_type == 'LP':
                # Use fine-tuned momentum for learning
                direction = solve_momentum(best_sequence)
                best_sequence = best_sequence * 0.9 + direction * 0.006
            else:
                # Reset phase or break
                if step_duration > 120:
                    step_duration *= 0.99
                else:
                    phase_type = 'LP'

    # Final polish to escape narrow valleys
    final_score = float('inf')
    for iteration in range(8):
        time_remaining = deadline - time()
        if time_remaining < 10:
            break
        best_sequence = np.clip(best_sequence, 0.0, 1000.0)
        score_check = evaluate_sequence(best_sequence.tolist())
        if score_check < final_score and time_remaining > 20:
            final_score = score_check
            print(f"Refined score to {final_score:.6f} @ iteration {iteration} (final)")
        else:
            # Change direction for escape
            if iteration % 3 == 0:
                candidate = generate_sine()
            elif iteration % 3 == 1:
                candidate = generate_gaussian()
            else:
                candidate = generate_random()
            best_sequence = best_sequence * 0.9 + candidate * 0.1

    return best_sequence.tolist()
