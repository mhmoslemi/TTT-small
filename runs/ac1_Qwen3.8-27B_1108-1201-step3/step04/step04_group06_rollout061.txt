def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def gs_init(n, n_iter=50, rng=None):
    """Gerchberg-Saxton inspired initialization."""
    if rng is None:
        rng = np.random.default_rng()
    L = 1 << (2*n - 1 - 1).bit_length()  # next power of 2 >= 2n-1
    a = rng.random(n)
    a = a / a.sum()
    
    for _ in range(n_iter):
        # FFT
        F = np.fft.fft(a, L)
        # Equalize magnitude
        mag = np.abs(F)
        # Target: flat magnitude
        # But DC must be positive and real
        target_mag = np.ones(L) * mag.mean()
        # Preserve phases
        F_new = target_mag * np.exp(1j * np.angle(F))
        # Ensure conjugate symmetry for real signal
        F_new[0] = np.abs(F_new[0])
        for k in range(1, L//2 + 1):
            if k == L - k:
                F_new[k] = np.abs(F_new[k])
            else:
                avg = (F_new[k] + np.conj(F_new[L-k])) / 2
                F_new[k] = avg
                F_new[L-k] = np.conj(avg)
        # IFFT
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = a  # keep old
        else:
            a = a_new / s
    
    return a

def chebyshev_init(n, M=20, rng=None):
    """Initialize using Chebyshev polynomial basis."""
    if rng is None:
        rng = np.random.default_rng()
    x = 2.0 * np.arange(n) / (n - 1) - 1.0
    # Build Chebyshev basis
    basis = np.zeros((M + 1, n))
    for m in range(M + 1):
        basis[m] = eval_chebyt(m, x)
    # Random coefficients (biased towards low order)
    c = rng.standard_normal(M + 1) * np.exp(-np.arange(M + 1) * 0.1)
    a = c @ basis
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit

        top_k = max(top_k_min, int(n * top_k_max_frac * (0.1 + 0.9 * progress)))
        
        # Get top-k peaks
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
        
        # Softmax weights
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
        
        # Update
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0

        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

def polynomial_optimize(n, M=20, time_limit=60, rng=None):

    def objective(params):

        a = np.maximum(a, 0)

        if s < 1e-10:
            return 1e10
        a = a / s
        c = fast_convolve(a)
        return c.max()

                         options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-12})

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. Gerchberg-Saxton spectral equalization for diverse initializations
2. Chebyshev polynomial parameterization for smooth-space exploration
3. Improved exponentiated gradient with adaptive peak focusing

5. Multi-length interpolation search

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import eval_chebyt


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def gs_init(n, n_iter=100, rng=None):
    """Gerchberg-Saxton inspired initialization for flat-spectrum sequences."""
    if rng is None:
        rng = np.random.default_rng()
    L = 1 << (2 * n - 1 - 1).bit_length()
    a = rng.random(n) + 0.01
    a = a / a.sum()

    for it in range(n_iter):
        F = np.fft.fft(a, L)
        mag = np.abs(F)
        mean_mag = mag.mean()
        # Soft equalization: compress dynamic range
        alpha = 0.7 + 0.3 * (it / n_iter)
        new_mag = np.power(np.power(mag, alpha) + np.power(mean_mag, alpha), 1.0 / alpha)
        F_new = new_mag * np.exp(1j * np.angle(F))
        # Enforce conjugate symmetry
        F_new[0] = np.abs(F_new[0])
        for k in range(1, L // 2 + 1):
            if k == L - k:
                F_new[k] = np.abs(F_new[k])
            else:
                avg = (F_new[k] + np.conj(F_new[L - k])) / 2
                F_new[k] = avg
                F_new[L - k] = np.conj(avg)
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s

    return a

def chebyshev_optimize(n, M, time_limit, rng):
    """Optimize using Chebyshev polynomial parameterization."""
    x = 2.0 * np.arange(n) / max(1, n - 1) - 1.0
    basis = np.zeros((M + 1, n))
    for m in range(M + 1):
        basis[m] = eval_chebyt(m, x)

    def obj_and_grad(params):
        a = params @ basis
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        c = fast_convolve(a)
        return c.max()

    best_obj = np.inf
    best_x = None
    t0 = time.monotonic()
    n_starts = 8

    for si in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        # Diverse starting points
        if si == 0:
            x0 = np.zeros(M + 1)
            x0[0] = 1.0
        elif si == 1:
            x0 = rng.standard_normal(M + 1) * 0.05
            x0[0] = 1.0
        elif si == 2:
            x0 = np.zeros(M + 1)
            x0[0] = 1.0
            x0[1] = 0.3
        elif si == 3:
            x0 = np.zeros(M + 1)
            x0[0] = 1.0
            x0[2] = -0.2
        else:
            x0 = rng.standard_normal(M + 1) * np.exp(-np.arange(M + 1) * 0.15)
            x0[0] = 1.0

        per_start_time = max(1.0, (time_limit - (time.monotonic() - t0)) / max(1, n_starts - si))
        result = minimize(obj_and_grad, x0, method='Nelder-Mead',
                          options={'maxiter': min(2000, int(50 + 10 * per_start_time)),
                                   'xatol': 1e-10, 'fatol': 1e-14})
        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x.copy()
            print(f"    Cheb M={M} start {si}: obj={best_obj:.10f}")

    if best_x is not None:
        a = best_x @ basis
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            return a / s
    return np.ones(n) / n


def eg_optimize(a0, time_limit, eta0=1.5, temp0=0.004, temp_decay=0.9998,
                eta_decay=0.99997, top_k_min=2, top_k_max_frac=0.2,
                momentum=0.0, verbose=False):
    """Exponentiated gradient with adaptive top-k and optional momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    prev_log_update = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        top_k = max(top_k_min, int(n * top_k_max_frac * (0.05 + 0.95 * progress)))
        top_k = min(top_k, 2 * n - 1)

        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update_raw = -eta * sg / sg_max
        # Momentum
        if momentum > 0:
            log_update = (1.0 - momentum) * log_update_raw + momentum * prev_log_update
        else:
            log_update = log_update_raw
        prev_log_update = log_update_raw

        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 2000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay
        temp = max(temp, 1e-7)

        if no_improve > 2500:
            rng_local = np.random.default_rng(it * 17 + 3)
            a = best_a + rng_local.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta0 * 0.3, 0.1)
            temp = max(temp, temp0 * 0.005)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def coord_polish(a0, time_limit=40, max_iter=50000):
    """Coordinate descent polishing with adaptive step sizes."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    no_improve = 0
    step_sizes = [0.5, 0.7, 0.85, 0.9, 1.0/0.9, 1.0/0.85, 1.0/0.7, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit and it < max_iter:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        i = int(rng.integers(n))
        if a[i] < 1e-12:
            # Try setting it to a small positive value
            a[i] = 0.001
            s2 = a.sum()
            a_trial = a / s2
            c2 = fast_convolve(a_trial)
            if c2.max() < best_obj - 1e-13:
                a = a_trial
                best_obj = c2.max()
                best_a = a.copy()
                c = c2
                no_improve = 0
            else:
                a[i] = 0.0
            continue

        old_val = a[i]
        improved = False
        for factor in step_sizes:
            a[i] = old_val * factor
            s2 = a.sum()
            if s2 < 1e-15:
                a[i] = old_val
                continue
            a_trial = a / s2
            c2 = fast_convolve(a_trial)

            if obj2 < best_obj - 1e-13:
                a = a_trial
                best_obj = obj2
                best_a = a.copy()
                c = c2
                improved = True
                break
        if not improved:
            a[i] = old_val

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

    return best_a.copy(), best_obj


def pair_polish(a0, time_limit=30, max_iter=20000):
    """Polish by simultaneously adjusting pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if a[i] < 1e-12 or a[j] < 1e-12:
            no_improve += 1
            if no_improve > 300:
                break
            continue

        old_i, old_j = a[i], a[j]
        improved = False
        # Transfer mass between i and j
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9, -0.1, -0.3, -0.5, -0.7, -0.9]:
            new_i = old_i - frac * (old_i - old_j) * 0.5
            new_j = old_j + frac * (old_i - old_j) * 0.5
            if new_i < 0 or new_j < 0:
                continue
            a[i] = new_i
            a[j] = new_j
            s2 = a.sum()
            if s2 < 1e-15:
                a[i], a[j] = old_i, old_j
                continue
            a_trial = a / s2
            c2 = fast_convolve(a_trial)
            obj2 = c2.max()
            if obj2 < best_obj - 1e-13:
                a = a_trial
                best_obj = obj2
                best_a = a.copy()
                c = c2
                improved = True
                break
        if not improved:
            a[i], a[j] = old_i, old_j
            no_improve += 1
            if no_improve > 500:
                break
        else:
            no_improve = 0

    return best_a.copy(), best_obj


def eval_and_update(a, best_score, best_seq):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_score, best_seq, float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:

def resample_sequence(a, n_new):
    """Resample a sequence to a new length via interpolation."""
    n_old = len(a)
    if n_new == n_old:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if best_seq is None:
        parent = np.ones(384) / 384
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        print(f"No parent, uniform n=384, score={best_score:.8f}")

    n = len(best_seq)
    print(f"Starting: n={n}, best={best_score:.8f}")

    # === Phase 1: Primary EG on parent ===
    print(f"\n=== Phase 1: EG on parent (n={n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(500, rem * 0.25)
    if t1 > 30:
        print(f"  Alloc={t1:.0f}s")
        a_opt, _ = eg_optimize(best_seq, time_limit=t1,
                               eta0=2.0, temp0=0.003,
                               temp_decay=0.9997, eta_decay=0.99996,
                               top_k_min=2, top_k_max_frac=0.2)
        best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
        if best_seq is not None:
            print(f"  After EG: best={best_score:.8f}")

    # === Phase 2: GS diverse initializations ===
    print(f"\n=== Phase 2: GS Init ===")
    n_gs_tries = 6
    for gi in range(n_gs_tries):
        rem = deadline - time.monotonic() - (n_gs_tries - gi - 1) * 80 - 200
        if rem < 40:
            break
        alpha_s = 0.3 + 0.1 * gi
        alpha_e = 0.8 + 0.05 * gi
        a_gs = gs_init_varied(n, n_iter=60 + 10 * gi,
                              seed=100 + gi, alpha_start=alpha_s,
                              alpha_end=alpha_e, rng=rng)
        s, seq, sc = eval_and_update(a_gs, best_score, best_seq)
        if sc < best_score + 0.3:
            t_alloc = min(80, rem - 40)
            if t_alloc > 15:
                a_opt, _ = eg_optimize(a_gs, time_limit=t_alloc,
                                       eta0=1.5, temp0=0.005,
                                       temp_decay=0.9998, eta_decay=0.99997,
                                       top_k_min=2, top_k_max_frac=0.15)
                best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
    print(f"  After GS: best={best_score:.8f}")

    # === Phase 3: Chebyshev polynomial optimization ===
    print(f"\n=== Phase 3: Polynomial Optimization ===")
    for M in [12, 20, 35]:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(50, rem / 3)
        if t_alloc < 10:
            break
        print(f"  M={M}, alloc={t_alloc:.0f}s")
        a_poly = chebyshev_optimize(n, M=M, time_limit=t_alloc, rng=rng)
        best_score, best_seq, sc = eval_and_update(a_poly, best_score, best_seq)
        # Polish the polynomial result
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 20:
            a_opt, _ = eg_optimize(a_poly, time_limit=min(rem2, 40),
                                   eta0=1.0, temp0=0.004,
                                   temp_decay=0.9998, eta_decay=0.99997,
                                   top_k_min=2, top_k_max_frac=0.1)
            best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
    print(f"  After Poly: best={best_score:.8f}")

    # === Phase 4: EG with different hyperparameters ===
    print(f"\n=== Phase 4: EG Variations ===")
    configs = [
        {"eta0": 0.3, "temp0": 0.015, "temp_decay": 0.9999,
         "eta_decay": 0.99995, "top_k_min": 5, "top_k_max_frac": 0.4, "momentum": 0.3},
        {"eta0": 5.0, "temp0": 0.001, "temp_decay": 0.9994,
         "eta_decay": 0.9999, "top_k_min": 1, "top_k_max_frac": 0.05, "momentum": 0.0},
        {"eta0": 1.0, "temp0": 0.006, "temp_decay": 0.9997,
         "eta_decay": 0.99998, "top_k_min": 3, "top_k_max_frac": 0.25, "momentum": 0.2},
        {"eta0": 3.0, "temp0": 0.002, "temp_decay": 0.9996,
         "eta_decay": 0.99995, "top_k_min": 1, "top_k_max_frac": 0.1, "momentum": 0.5},
    ]
    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - (len(configs) - ci - 1) * 60 - 150
        if rem < 25:
            break
        t_alloc = min(70, rem)
        a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg)
        best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
    print(f"  After EG var: best={best_score:.8f}")

    # === Phase 5: Coordinate and pair polishing ===
    print(f"\n=== Phase 5: Polishing ===")
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        a_opt, _ = coord_polish(best_seq, time_limit=min(rem, 50))
        best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
        print(f"  After coord: best={best_score:.8f}")

    rem = deadline - time.monotonic() - 80
    if rem > 10:
        a_opt, _ = pair_polish(best_seq, time_limit=min(rem, 40))
        best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
        print(f"  After pair: best={best_score:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    n_base = len(best_seq)
    alt_lengths = [int(n_base * 0.7), int(n_base * 1.3), int(n_base * 1.5),
                   int(n_base * 0.5), int(n_base * 2.0)]
    for n_new in alt_lengths:
        if n_new < 32 or n_new > 4000:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 20:
            break
        a_new = resample_sequence(best_seq, n_new)
        sc = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            print(f"  n={n_new}: {sc:.8f}")
        # Quick EG on new length
        t_alloc = min(35, rem - 10)
        if t_alloc > 5:
            a_opt, _ = eg_optimize(a_new, time_limit=t_alloc,
                                   eta0=1.0, temp0=0.005,
                                   temp_decay=0.9998, eta_decay=0.99997,
                                   top_k_min=2, top_k_max_frac=0.2)
            best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)
    print(f"  After multi-len: best={best_score:.8f}")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final ===")
    rem = deadline - time.monotonic() - 10
    if rem > 5:
        a_opt, _ = eg_optimize(best_seq, time_limit=min(rem, 25),
                               eta0=0.2, temp0=0.0008,
                               temp_decay=0.99998, eta_decay=0.99999,
                               top_k_min=1, top_k_max_frac=0.05)
        best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)

    rem = deadline - time.monotonic() - 5
    if rem > 3:
        a_opt, _ = coord_polish(best_seq, time_limit=min(rem, 15), max_iter=5000)
        best_score, best_seq, _ = eval_and_update(a_opt, best_score, best_seq)

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
