```python
def compute_smooth_gradient(a, tau, top_k=50):
    """Compute approximate gradient of smooth max of convolution."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Use top-k approximation for efficiency
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        if w[idx] < 1e-15:
            continue
        k = int(k)
        # grad[i] += 2 * w[idx] * a[k-i] for 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end] is a[k-i_end] to a[k-i_start]
        # i.e., a[k - i_end : k - i_start + 1] reversed... 
        # Actually: when i = i_start, k-i = k - i_start; when i = i_end, k-i = k - i_end
        # So a[k-i] for i in [i_start, i_end] = a[k-i_start], a[k-i_start-1], ..., a[k-i_end]
        # = a[i_start:n][::-1]... let me just do it directly
        idxs = np.arange(i_start, i_end + 1)
        grad[idxs] += 2.0 * w[idx] * a[k - idxs]
    
    return grad
```

```python
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, tau, top_k=100):
    """Compute smooth approximation of subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    
    return grad, conv

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    # Use overlap-add approach to get better linear convolution properties
    phases = rng.uniform(0, 2 * np.pi, n)
    
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)  # DC = sqrt(n) gives sum(a) = 1 after normalization
    for k in range(1, (n + 1) // 2):
        A[k] = np.exp(1j * phases[k])
        A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = np.exp(1j * phases[n // 2])
        A[n // 2] = 1.0  # Must be real for Hermitian
    
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()

def triangular_init(n):
    """Triangular distribution initialization."""
    x = np.arange(n) / (n - 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()

        pass_count += 1

            if time.monotonic() - t0 > time_limit:

                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                # Renormalize others
                remaining = 1.0 - new_val
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-15:
                    continue
                scale = remaining / old_remaining
                mask = np.ones(n, dtype=bool)
                mask[j] = False

                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:

                old_remaining = 1.0 - old_val

                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using flat-spectrum initialization,
subgradient optimization with adaptive parameters, coordinate descent,
pair-swap local search, and multi-length exploration.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, tau, top_k=100):
    """Compute smooth approximation of subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]

    return grad, conv

    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def triangular_init(n):
    """Triangular distribution initialization."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def gaussian_init(n):
    """Gaussian initialization."""
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / 2)
    return a / a.sum()


def double_peak_init(n, rng):
    """Two Gaussian peaks."""
    c1 = rng.uniform(0.2, 0.4)
    c2 = rng.uniform(0.6, 0.8)
    w = 0.05 + rng.uniform(0, 0.05)
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1)**2) / (2*w**2)) + np.exp(-((x - c2)**2) / (2*w**2))
    return a / a.sum()


def optimize_subgrad(a0, time_limit, verbose=False):
    """Subgradient optimization with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    tau = 80.0
    lr = 0.04
    momentum = 0.82
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Adaptive parameter scheduling
        if no_improve > 400:
            momentum = min(0.95, momentum + 0.005)
            lr = max(0.003, lr * 0.97)
        elif no_improve > 200:
            tau = min(500.0, tau * 1.01)
            momentum = min(0.92, momentum + 0.002)
        elif no_improve > 50:
            tau = max(30.0, tau * 0.995)
            lr = min(0.08, lr * 1.005)
        else:
            tau = min(300.0, tau * 1.005)

        grad, conv = compute_grad(a, tau, top_k=min(150, n))

        # Center gradient for better simplex behavior
        grad_centered = grad - grad.mean()

        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)

        # Evaluate
        obj_new = fast_convolve(a_new).max()

        # Simple line search: try to extend the step
        if obj_new < best_obj:
            a_ext = project_simplex(a + 2.0 * v, target_sum=1.0)
            obj_ext = fast_convolve(a_ext).max()
            if obj_ext < obj_new:
                a_new = a_ext
                obj_new = obj_ext

        a = a_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {iter_count}: obj={best_obj:.10f}, lr={lr:.4f}, tau={tau:.1f}, no_imp={no_improve} ({elapsed:.0f}s)")

        # Restart if stuck
        if no_improve > 600:
            rng_r = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.005 + 0.005 * min(no_improve // 600, 4)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            tau = 60.0
            momentum = 0.82

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 2:
            break

    return best_a, best_obj


def pair_swap_search(a0, time_limit, seed=0):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Focus on non-zero coordinates
    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization, subgradient optimization,
    coordinate descent, pair-swap search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [384, 512, 640, 768, 1024, 1280, 1536]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths.sort(key=lambda x: abs(x - (len(parent) if parent is not None else 640)))

    inits = {}
    for n in target_lengths[:5]:
        best_init = None
        best_init_obj = float('inf')

        # Multiple initialization methods
        methods = [
            ("flat_spec", flat_spectrum_init(n, rng)),
            ("random", random_init(n, rng)),
            ("triangular", triangular_init(n)),
            ("gaussian", gaussian_init(n)),
            ("double_peak", double_peak_init(n, rng)),
        ]

        if parent is not None:
            methods.append(("interp", interpolate_sequence(parent, n)))

        for name, init in methods:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init
            except Exception:
                pass

        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init obj={best_init_obj:.6f}")

    # === Phase 2: Subgradient optimization on top initializations ===
    print(f"\n=== Phase 2: Subgradient Optimization (best={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    total_opt_time = min(500, max(60, (deadline - time.monotonic()) * 0.35))
    n_optimize = min(3, len(sorted_inits))

    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:n_optimize]):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        time_fracs = [0.5, 0.3, 0.2]
        time_alloc = min(total_opt_time * time_fracs[rank], rem)
        print(f"  SG optimize n={n_len} (obj={init_obj:.6f}), t={time_alloc:.0f}s")
        a_opt, obj = optimize_subgrad(init, time_limit=time_alloc, verbose=(rank == 0))
        try_seq(a_opt, f"sg_n{n_len}_r{rank}")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.5, 150)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 4: Pair swap search ===
    print(f"\n=== Phase 4: Pair Swap (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_ps = min(rem * 0.4, 100)
            print(f"  PS t={t_ps:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "pair_swap")

    # === Phase 5: Refined subgradient with perturbations ===
    print(f"\n=== Phase 5: Refined Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem / (4 - p), 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif p == 2:
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            else:
                f_init = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init, target_sum=1.0)

            a_opt, obj = optimize_subgrad(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"sg_refine_{p}")

            if best_seq is not None and p == 0:
                # Quick CD after first refinement
                rem = deadline - time.monotonic() - 50
                if rem > 15:
                    a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 40))
                    try_seq(a_opt, "cd_after_refine")

    # === Phase 6: Try longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for mult in [1.5, 2.0]:
            target_n = int(n_best * mult)
            if target_n <= n_best or target_n > 2560:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                continue

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.5, 100)
            print(f"  Longer n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_subgrad(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"long_n{target_n}")

            rem = deadline - time.monotonic() - 25
            if rem > 15 and best_score < 1.503:
                a_opt, obj = coordinate_descent(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt, f"cd_long_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 15
        for _ in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise, target_sum=1.0)
            a_opt, obj = optimize_subgrad(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
