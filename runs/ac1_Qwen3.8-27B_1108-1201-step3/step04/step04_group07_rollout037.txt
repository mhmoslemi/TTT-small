import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def obj_value(a, n):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def smooth_max_grad(conv, tau):
    """Compute softmax weights for smooth max approximation."""
    scaled = conv * tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w

def get_subgradient(a, conv, w):
    """Compute weighted subgradient of max(conv) using softmax weights."""
    n = len(a)
    # grad of (a*a)[k] w.r.t. a[m] = 2*a[k-m] if 0 <= k-m <= n-1
    # So the total subgradient is sum_k w[k] * 2*a[k-m] for valid k-m
    # This is 2 * (w * a) convolved appropriately
    # Specifically: sg[m] = 2 * sum_k w[k] * a[k-m] (where 0 <= k-m <= n-1)
    # = 2 * (w_full * a)[m] where w_full is w padded/sliced appropriately
    
    # More directly: for each lag k, the contribution to sg[m] is w[k] * 2 * a[k-m]
    # if 0 <= k-m <= n-1
    # So sg[m] = 2 * sum_{k: 0<=k-m<=n-1} w[k] * a[k-m]
    # = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]  (where j = k-m)
    
    # This is like a cross-correlation of w and a
    # sg[m] = 2 * corr(w, a)[m] where corr is defined appropriately
    
    # Let me just compute it directly using the convolution matrix
    # sg = 2 * C(a)^T @ w  where C(a) is the conv matrix
    # Actually: (C(a) @ a) = conv, and d(conv[k])/da[m] = 2*C(a)[k,m]
    # So d(max)/da = sum_k w[k] * 2*C(a)[k,m] = 2 * (w @ C(a))[:]
    
    C = build_conv_matrix(a)
    sg = 2.0 * (w @ C)
    return sg

def adam_step(a, grad, lr, beta1, beta2, eps, m, v, t):
    """One Adam step."""
    n = len(a)
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new, m, v

def adam_optimize(a0, time_limit, tau_init=1000, tau_max=100000, lr=0.001):
    """Adam optimization of smooth max approximation."""
    a = a0.copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v0, conv = obj_value(a, n)
    best_obj = v0
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init
    no_improve = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        w = smooth_max_grad(conv, tau)
        sg = get_subgradient(a, conv, w)
        
        t += 1
        a, m, v = adam_step(a, sg, lr, 0.9, 0.999, 1e-8, m, v, t)
        
        _, conv = obj_value(a, n)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            tau = min(tau_max, tau * 1.05)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.95
                tau = min(tau_max, tau * 1.1)
        
        if no_improve >= 500:
            break
    
    return best_a, best_obj

def lp_optimize(a0, time_limit):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    _, conv = obj_value(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        C = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 0.9995)
        bounds = [(0.0, None)] * n
        
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        
        improved = False
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-7:
                d = g - a
                # Line search
                best_t = 0.0
                best_obj_ls = best_obj
                for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    a_try = np.maximum(a + t_try * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:

                    a_try = a_try / s_try
                    obj_try, _ = obj_value(a_try, n)
                    if obj_try < best_obj_ls:
                        best_obj_ls = obj_try
                        best_t = t_try
                
                if best_t > 0:

                    if s_new > 1e-15:

                    _, conv = obj_value(a, n)

                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0

def simulated_annealing(a0, time_limit, max_iter=50000):

    a = a0.copy()
    n = len(a)
    s = a.sum()
    a = a / s
    
    _, conv = obj_value(a, n)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj

    T_min = 1e-7
    cooling = 0.999

    rng = np.random.default_rng(12345)

        delta = rng.standard_normal(n_perturb) * 0.01
        a_new = a.copy()
        a_new[indices] += delta
        a_new = np.maximum(a_new, 0.0)

        if s_new < 1e-15:

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_new
            current_obj = new_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()

    return best_a, best_obj

def resample_sequence(seq, new_n):
    """Resample a sequence to a different length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses a novel LP formulation (minimize t directly), Adam with log-sum-exp smoothing,
simulated annealing, and peak-shaving coordinate descent.
Key innovation: LP that minimizes the peak of a*g directly, combined with
dual-guided subgradient and adaptive multi-method cycling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_value(a, n):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    scaled = conv * tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w


def weighted_subgradient(w, a):
    """Compute weighted subgradient: sum_k w[k] * d(conv[k])/da."""
    n = len(a)
    C = build_conv_matrix(a)
    sg = 2.0 * (w @ C)
    return sg


def lp_minimize_t(a, time_limit=1.0):
    """
    Solve LP: minimize t s.t. C(a)@g <= t, sum(g)=1, g>=0, t>=0.
    Returns the optimal g direction.
    """
    n = len(a)
    C = build_conv_matrix(a)
    # Variables: g[0..n-1], t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # A_ub @ [g, t] <= b_ub: C(a)@g - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        t_val = result.x[n]
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s > 1e-15:
            g = g / s
        return g, t_val
    return a.copy(), None


def line_search(a, d, current_obj, n, max_evals=12):

    best_t = 0.0
    best_obj = current_obj
    
    for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_try = normalize_nonneg(a + t_try * d)
        obj_try, _ = obj_value(a_try, n)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t_try
    
    # Golden section if interior solution found
    if 0.05 < best_t < 0.95:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.01, best_t - 0.1)
        hi = min(0.99, best_t + 0.1)
        
        for _ in range(5):
            d1 = lo + (1.0 - gr) * (hi - lo)
            d2 = lo + gr * (hi - lo)
            a1 = normalize_nonneg(a + d1 * d)
            a2 = normalize_nonneg(a + d2 * d)
            obj1, _ = obj_value(a1, n)
            obj2, _ = obj_value(a2, n)
            if obj1 < obj2:
                hi = d2
            else:
                lo = d1
            if abs(hi - lo) < 0.01:
                break
        
        mid = (lo + hi) / 2.0
        a_mid = normalize_nonneg(a + mid * d)
        obj_mid, _ = obj_value(a_mid, n)
        if obj_mid < best_obj:
            best_obj = obj_mid
            best_t = mid
    
    return best_t, best_obj

def adam_optimize(a0, time_limit, tau_init=2000, tau_max=500000, lr=0.002):
    """
    Adam optimization using log-sum-exp smooth approximation of max.
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    tau = tau_init
    no_improve = 0
    best_obj, _ = obj_value(a, n)
    best_a = a.copy()
    current_lr = lr
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        w = softmax_weights(conv, tau)
        sg = weighted_subgradient(w, a)
        
        t_count += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * (sg * sg)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize_nonneg(a_new)
        
        _, conv = obj_value(a, n)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            # Increase tau as we converge (sharper max approximation)
            tau = min(tau_max, tau * 1.02)
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                current_lr *= 0.95
                tau = min(tau_max, tau * 1.05)
            if no_improve >= 1000:
                break

def peak_shaving(a0, time_limit):
    """
    Coordinate descent that specifically targets the peak lag.
    Tries to redistribute mass to reduce the maximum convolution value.
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    _, conv = obj_value(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]
        
        # Elements contributing to conv[peak_lag] = sum_i a[i]*a[i+peak_lag]
        i_start = max(0, peak_lag - (n - 1))
        i_end = min(n - 1, n - 1 - peak_lag) if peak_lag < n else 0
        
        # Find the pair (i, j) with i+j related to peak_lag that has largest product
        # For lag k: conv[k] = sum_i a[i] * a[i+k] (0 <= i <= n-1-k)
        if peak_lag < n:
            pairs = []
            for i in range(n - peak_lag):
                j = i + peak_lag
                pairs.append((a[i] * a[j], i, j))
            pairs.sort(reverse=True)
        else:
            # Negative lag (by symmetry, same as lag n-1-peak_lag from the other side)
            eff_lag = 2 * (n - 1) - peak_lag
            if eff_lag < n:
                pairs = []
                for i in range(n - eff_lag):
                    j = i + eff_lag
                    pairs.append((a[i] * a[j], i, j))
                pairs.sort(reverse=True)
            else:
                no_improve += 1
                if no_improve > 100:
                    break
                continue
        
        if not pairs:
            no_improve += 1
            continue
        
        # Try reducing the largest product by moving mass
        improved = False
        for prod, i, j in pairs[:5]:
            if prod < 1e-15:
                break
            
            # Try moving a small amount from the larger element to a random other position
            if a[i] >= a[j]:
                src = i
            else:
                src = j
            
            # Find a target position that doesn't contribute much to the peak
            for _attempt in range(3):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                
                # Try moving delta from src to dst
                delta = a[src] * 0.1
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = normalize_nonneg(a_try)
                
                _, conv_try = obj_value(a_try, n)
                obj_try = 2.0 * n * conv_try.max()
                
                if obj_try < best_obj - 1e-14:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                # Random perturbation to escape
                idx = rng.integers(0, n, size=10)
                a[idx] += rng.standard_normal(10) * 0.01
                a = normalize_nonneg(a)
                no_improve = 0
            if no_improve > 500:
                break
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, n_perturb_frac=0.05):
    """Simulated annealing with adaptive perturbation size."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    
    _, conv = obj_value(a, n)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()

    T_min = 1e-8
    cooling = 0.9995
    base_scale = 0.01
    
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        n_perturb = max(1, int(n * n_perturb_frac))
        indices = rng.choice(n, size=n_perturb, replace=False)
        scale = base_scale * (1.0 + 0.5 * T / T_min * 0.01)
        delta = rng.standard_normal(n_perturb) * scale

        new_obj = 2.0 * n * conv_new.max()
        delta_obj = new_obj - current_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, T_min)):
            a = a_new
            current_obj = new_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        
        T = max(T_min, T * cooling)
        iteration += 1
        
        if iteration % 5000 == 0 and T < T_min * 10:
            # Reheat slightly
            T = T_min * 100
            base_scale *= 0.9
    
    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)
    
    # Various structured inits
    inits['sine3'] = (0.8 + 0.2 * np.sin(6 * np.pi * x / n)) / (0.8 + 0.2 * np.sin(6 * np.pi * x / n)).sum()
    inits['sine7'] = (0.85 + 0.15 * np.sin(14 * np.pi * x / n)) / (0.85 + 0.15 * np.sin(14 * np.pi * x / n)).sum()
    inits['gauss'] = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = inits['gauss'] / inits['gauss'].sum()
    
    # Random Dirichlet with different concentrations
    inits['dirichlet_lo'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['dirichlet_hi'] = rng.dirichlet(np.ones(n) * 5.0)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining novel LP (minimize t), Adam,
    peak shaving, and simulated annealing.
    
    Strategy:
    - Phase 1 (200s): Quick Adam from multiple inits to find good basins
    - Phase 2 (700s): Deep LP with minimize-t formulation
    - Phase 3 (200s): Adam polish with increasing tau
    - Phase 4 (300s): Peak shaving + SA alternating
    - Phase 5 (200s): Final LP + peak shaving
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(p)}, score={parent_score:.6f}")
            best_sequence = parent.copy()
            best_score = parent_score
    except Exception:
        pass
    
    # ===== Phase 1: Quick Adam from multiple inits =====
    print("=== Phase 1: Quick Adam Scout ===")
    t_start = time.monotonic()
    phase1_budget = 200.0
    
    n_target = len(parent) if parent is not None else 768
    lengths_to_try = [n_target]
    # Try a couple nearby lengths
    if n_target > 100:
        lengths_to_try.append(n_target + 96)
        lengths_to_try.append(n_target - 96)
    
    scout_results = []
    total_inits = 0
    for n in lengths_to_try:
        if n < 64:
            continue
        inits = generate_inits(n, parent, rng)
        total_inits += len(inits)
    
    t_per_init = phase1_budget / max(1, total_inits)
    
    for n in lengths_to_try:
        if n < 64 or time.monotonic() > deadline - 1200:
            continue
        inits = generate_inits(n, parent, rng)
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1100:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 1100)
            if t_alloc < 3:
                break
            
            a_opt, obj_opt = adam_optimize(a0, time_limit=t_alloc, tau_init=1000, lr=0.003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scout_results.append((score, len(a_opt), a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scout n={n} {name}: {score:.6f}")
    
    # Sort and keep top
    scout_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # ===== Phase 2: Deep LP =====
    print("=== Phase 2: Deep LP (minimize t) ===")
    t_start2 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            a_opt, obj_opt = deep_lp_optimize(best_sequence, time_limit=remaining, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP: {score:.6f}")
    
    # Try LP from best scout result if different length is better
    for score, n_scout, a_scout in scout_results[:2]:
        if time.monotonic() > deadline - 500:
            break
        if len(a_scout) != len(best_sequence):
            t_alloc = min(100.0, deadline - time.monotonic() - 500)
            if t_alloc > 20:
                a_opt, obj_opt = deep_lp_optimize(a_scout, time_limit=t_alloc)
                s = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_sequence = a_opt.copy()
                    print(f"  LP (alt n={len(a_scout)}): {score:.6f} -> {s:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # ===== Phase 3: Adam Polish =====
    print("=== Phase 3: Adam Polish ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 350
        if remaining > 20:
            # Use high tau for sharp max approximation
            a_opt, obj_opt = adam_optimize(best_sequence, time_limit=remaining,
                                           tau_init=10000, tau_max=1000000, lr=0.001)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Adam polish: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # ===== Phase 4: Peak Shaving + SA =====
    print("=== Phase 4: Peak Shaving + SA ===")
    t_start4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            # Alternate peak shaving and SA
            t_per = remaining // 4 + 1
            for i in range(4):
                if time.monotonic() > deadline - 200:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 200)
                if t_alloc < 10:
                    break
                
                if i % 2 == 0:
                    a_opt, obj_opt = peak_shaving(best_sequence, time_limit=t_alloc)
                else:
                    a_opt, obj_opt = simulated_annealing(best_sequence, time_limit=t_alloc)
                
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  PS/SA {i}: {score:.6f}")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # ===== Phase 5: Final LP + Polish =====
    print("=== Phase 5: Final Refinement ===")
    t_start5 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            # Final LP
            t_lp = remaining * 0.6
            a_opt, obj_opt = deep_lp_optimize(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f}")
            
            # Quick Adam on result
            t_adam = remaining * 0.3
            if t_adam > 5:
                a_opt, obj_opt = adam_optimize(best_sequence, time_limit=t_adam,
                                               tau_init=50000, lr=0.0005)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final Adam: {score:.6f}")
    
    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Emergency fallback
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
