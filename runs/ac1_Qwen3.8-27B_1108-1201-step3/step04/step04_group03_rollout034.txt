def fw_lp_optimize(f0, time_limit, m_exp_init=1.0):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = m_exp_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except:
            res = None
        
        if res is not None and res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
            else:
                g = None
        
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = m_exp_init
            continue
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = m_exp_init
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 1)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = m_exp_init
            no_improve = 0
    
    return best_f, best_obj


def adaptive_mw(f0, time_limit, lr_init=0.04, top_k=20, temp_init=0.0004):

    f = np.maximum(f0.astype(np.float64), 0.0)

        obj = conv.max()

def accelerated_proximal(f0, time_limit, lr=0.005, nesterov=True):
    """
    Nesterov accelerated gradient with projection onto simplex.
    Uses the subgradient of max_k c_k.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()
    t_k = 1.0
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(10, n // 8)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient of max_k c_k at y
        conv_y = fast_convolve(y)
        # Find the argmax (ties broken by averaging)
        max_val = conv_y.max()
        # Use top-k for a smoother subgradient
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        pv = conv_y[peaks]
        # Weight: emphasize the max more
        weights = np.exp((pv - max_val) * 10)
        weights /= weights.sum()
        
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * y[k - idx]
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        
        step = lr / (np.linalg.norm(sg) + 1e-10)
        x_new = project_simplex(y - step * sg)
        
        if not nesterov:
            y = x_new
        else:
            y = x_new + beta * (x_new - f)
            y = project_simplex(y)
            f = x_new
        
        t_k = t_new
        
        conv_f = fast_convolve(f)
        obj = conv_f.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            t_k = 1.0
            no_improve = 0
    
    return best_f, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())
    
    # Two bumps
    b1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits.append(b1 / b1.sum())
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    inits.append(alt / alt.sum())
    
    # If parent available, interpolate
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.02))
    
    return inits


def cosine_annealing_lr(it, total_iters, lr_init, lr_min):
    return lr_min + 0.5 * (lr_init - lr_min) * (1 + np.cos(np.pi * it / total_iters))

Minimize sum_k (c_k - mean)^2 = sum_k c_k^2 - (sum_k c_k)^2/(2n-1)

Since sum_k c_k = 1 (fixed), this is equivalent to minimizing sum_k c_k^2.

L = sum_k c_k^2
dL/da_m = 2 * sum_k c_k * dc_k/da_m = 2 * sum_k c_k * 2 * a_{k-m} = 4 * sum_k c_k * a_{k-m}

sum_k c_k * a_{k-m} = (c * a_rev)_m where a_rev is the reverse of a.

So the gradient is 4 * conv(c, reverse(a)). This is O(n log n) using FFT!

This gives us a very efficient gradient for the "variance minimization" objective. Let's use this!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Quick FW-LP for initial improvement
2. Fast gradient descent on variance + max objective (millions of iterations)
3. Adaptive multiplicative weight polish
4. Multi-length exploration
5. Final micro-polish
Key innovation: leveraging the efficient O(n^2) gradient of the variance objective
to perform far more optimization steps than LP-based methods allow.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def variance_grad(c, a):
    """
    Gradient of sum_k c_k^2 w.r.t. a, where c = auto-conv(a).
    Uses the relation: dL/da_m = 4 * sum_j c[m+j] * a[j]
    """
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = 4.0 * np.dot(c[m:m + n], a)
    return grad


def max_subgradient(f, top_k=20):
    """Subgradient of max_k c_k with top-k smoothing."""
    n = len(f)
    conv = fast_convolve(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    # Exponential weighting favoring the max
    max_pv = pv.max()
    weights = np.exp((pv - max_pv) * 5.0)
    ws = weights.sum()
    if ws < 1e-30:
        weights = np.zeros(len(peaks))
        weights[np.argmax(pv)] = 1.0
    else:
        weights /= ws
    sg = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]
    return sg, conv


def gradient_descent_phase(f0, time_limit, lr_init=0.5, variance_weight=0.5,
                           momentum=0.9, verbose=True):
    """
    Fast gradient descent combining variance and max objectives.
    Can do millions of iterations due to O(n^2) per step.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr_init
    nesterov_t = 1.0
    y = f.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(y)

        # Combined gradient: variance + max
        var_g = variance_grad(conv, y)
        max_g, _ = max_subgradient(y, top_k=max(10, n // 16))

        # Normalize gradients for combining
        var_norm = np.linalg.norm(var_g)
        max_norm = np.linalg.norm(max_g)
        if var_norm < 1e-30:
            var_g = np.zeros(n)
        else:
            var_g = var_g / var_norm
        if max_norm < 1e-30:
            max_g = np.zeros(n)
        else:
            max_g = max_g / max_norm

        combined_g = variance_weight * var_g + (1.0 - variance_weight) * max_g

        # Nesterov acceleration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * nesterov_t * nesterov_t)) / 2.0
        beta = (nesterov_t - 1.0) / t_new

        step = cur_lr / (np.linalg.norm(combined_g) + 1e-10)
        x_new = project_simplex(y - step * combined_g)
        y = x_new + beta * (x_new - f)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = x_new
        f = x_new

        nesterov_t = t_new

        # Check improvement periodically (every 50 iters to save time)
        if it % 50 == 1:
            conv_f = fast_convolve(f)
            obj = conv_f.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        cur_lr *= 0.99999

        if no_improve > 100:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            nesterov_t = 1.0
            cur_lr = lr_init * 0.8
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            res = None

        if res is not None and res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
            else:
                g = None
        else:
            g = None

        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.03, 2.5)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def multiplicative_weight(f0, time_limit, lr_init=0.04, top_k=20, temp_init=0.0004):
    """Adaptive multiplicative weight (mirror descent) optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_update[idx] -= cur_lr * 2.0 * w[i] * f[k - idx]

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9996
        cur_temp *= 1.003

        if no_improve > 120:
            rng = np.random.default_rng((it * 104729) % 1000000)
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.7, 0.01)
            cur_temp = temp_init * 1.5
            no_improve = 0

    return best_f, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c1 / c1.sum())

    # Two bumps at 1/4 and 3/4
    b = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2) + \
        np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    inits.append(b / b.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.01))

    return inits

    gd_configs = [
        {'lr_init': 0.8, 'variance_weight': 0.6, 'momentum': 0.9},
        {'lr_init': 0.5, 'variance_weight': 0.4, 'momentum': 0.92},
        {'lr_init': 1.0, 'variance_weight': 0.7, 'momentum': 0.88},
        {'lr_init': 0.3, 'variance_weight': 0.5, 'momentum': 0.95},
    ]

    rem = deadline - time.monotonic() - 600
    t_per_gd = max(30, rem // len(gd_configs))

    for ci, cfg in enumerate(gd_configs):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_alloc = min(t_per_gd, rem)
        print(f"  GD config {ci} (lr={cfg['lr_init']}, vw={cfg['variance_weight']}), t={t_alloc:.0f}s")
        a_opt, _ = gradient_descent_phase(
            best_seq, time_limit=t_alloc,
            lr_init=cfg['lr_init'], variance_weight=cfg['variance_weight'],
            momentum=cfg['momentum']
        )
        try_seq(a_opt, f"gd_{ci}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Multiplicative Weight (~300s) ===
    print("\n=== Phase 3: Multiplicative Weight ===")
    n_best = len(best_seq)

    mw_configs = [
        {'lr_init': 0.03, 'top_k': 15, 'temp_init': 0.0003},
        {'lr_init': 0.05, 'top_k': 25, 'temp_init': 0.0006},
        {'lr_init': 0.02, 'top_k': 10, 'temp_init': 0.0002},
    ]

    rem = deadline - time.monotonic() - 300
    t_per_mw = max(20, rem // len(mw_configs))

    for ci, cfg in enumerate(mw_configs):
        rem = deadline - time.monotonic() - 200
        if rem < 10:
            break
        t_alloc = min(t_per_mw, rem)
        print(f"  MW {ci} (lr={cfg['lr_init']}), t={t_alloc:.0f}s")
        a_opt, _ = multiplicative_weight(
            best_seq, time_limit=t_alloc,
            lr_init=cfg['lr_init'], top_k=cfg['top_k'],
            temp_init=cfg['temp_init']
        )
        try_seq(a_opt, f"mw_{ci}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (~200s) ===
    print("\n=== Phase 4: Multi-Length ===")
    n_best = len(best_seq)

    other_lengths = [n_base - 32, n_base + 32, n_base + 64]
    for nl in other_lengths:
        if nl < 128 or nl > 1024:
            continue
        rem = deadline - time.monotonic() - 150
        if rem < 20:
            break
        a_resampled = resample_sequence(best_seq, nl)
        score_r = float(evaluate_sequence(a_resampled.tolist()))
        print(f"  Resampled to n={nl}: score={score_r:.8f}")

        # Quick GD on resampled
        t_alloc = min(40, rem // 2)
        a_opt, _ = gradient_descent_phase(a_resampled, time_limit=t_alloc, lr_init=0.5, variance_weight=0.5)
        try_seq(a_opt, f"gd_n{nl}")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    n_best = len(best_seq)

    # GD with very small lr
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_alloc = min(rem, 60)
        print(f"  Fine GD, t={t_alloc:.0f}s")
        a_opt, _ = gradient_descent_phase(best_seq, time_limit=t_alloc, lr_init=0.1, variance_weight=0.5)
        try_seq(a_opt, "fine_gd")

    # Fine MW
    rem = deadline - time.monotonic() - 40
    if rem > 10:
        t_alloc = min(rem, 30)
        print(f"  Fine MW, t={t_alloc:.0f}s")
        a_opt, _ = multiplicative_weight(best_seq, time_limit=t_alloc, lr_init=0.005, top_k=8, temp_init=0.0001)
        try_seq(a_opt, "fine_mw")

    # Micro perturbations + quick GD
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        for _ in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = gradient_descent_phase(f_pert, time_limit=min(rem, 8), lr_init=0.1, variance_weight=0.5)
            try_seq(a_opt, "perturb")

    # Final FW-LP if time remains
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        print(f"  Final FW-LP, t={rem:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=rem)
        try_seq(a_opt, "final_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining FW-LP, fast gradient descent,
    and multiplicative weight optimization with multi-length exploration.
    """
    return main_search(budget_s=budget_s, seed=seed)
