"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage optimization:
1. Exponentiated gradient (multiplicative weights) with adaptive temperature/step-size
2. Softmax + Adam optimization as an alternative parameterization
3. Simulated annealing with coordinate-shift moves
4. Multi-length, multi-init exploration
5. Coordinate-local search polishing
Key improvements over previous: adaptive subgradient focusing, Adam-based
updates in log-space, SA with smart moves, and aggressive length exploration.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def compute_subgradient_top(a, c, top_k, temperature):
    """Compute softmax-weighted subgradient focusing on top-k peaks."""
    n = len(a)
    if top_k < len(c):
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len(c))
        peek = c.copy()

    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    return sg


def eg_adaptive(a0, time_limit, eta0=1.0, temp0=0.003, top_k_frac=0.2):
    """Exponentiated gradient with adaptive temperature and step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy()
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    patience = 5000

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient_top(a, c, top_k, temp)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Check running average
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-13:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
                no_improve = 0

        # Adaptive temperature: increase when many peaks are active
        if it % 200 == 0:
            active = (c > 0.9 * c.max()).sum()
            frac_active = active / len(c)
            target_temp = max(0.0001, temp0 * frac_active * 10)
            temp = 0.9 * temp + 0.1 * min(target_temp, temp0 * 5)

        # Adaptive eta: increase if no progress, decrease if oscillating
        if no_improve > patience // 4:
            eta = max(eta0 * 0.5, eta * 0.998)
        elif no_improve > patience:
            # Restart with perturbation
            rng_local = np.random.default_rng(it * 13 + 3)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0
            temp = temp0
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj / (n * 2.0)  # return max_b for sum=1


def softmax_adam(a0, time_limit, lr=0.05, top_k_frac=0.2):
    """Optimize in log-space using Adam, with softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    z = np.log(a)
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    # Adaptive learning rate
    lr_cur = lr
    base_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        sg_a = compute_subgradient_top(a, c, top_k, 0.002)
        mean_sg = np.sum(a * sg_a)
        sg_z = a * (sg_a - mean_sg)

        m = beta1 * m + (1 - beta1) * sg_z
        v = beta2 * v + (1 - beta2) * (sg_z * sg_z)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr_cur * m_hat / (np.sqrt(v_hat) + eps)

        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Adaptive LR
        if no_improve > 3000:
            lr_cur = base_lr * 0.5
            no_improve = 0
        elif no_improve > 8000:
            # Perturb
            rng_local = np.random.default_rng(t * 7 + 1)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            z = np.log(a)
            c = fast_convolve(a)
            best_obj = c.max()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr_cur = base_lr
            no_improve = 0

    return best_a.copy()


def simulated_annealing(a0, time_limit, t_start=0.01, t_end=0.0001, cooling=0.9995):
    """Simulated annealing with coordinate-shift and perturb moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    cur_obj = best_obj

    rng = np.random.default_rng(42)
    temp = t_start
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        a_cur = a.copy()

        move_type = rng.random()

        if move_type < 0.5:
            # Shift move: move mass from i to j
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            amount = a[i] * rng.uniform(0.05, 0.4)
            a[i] -= amount
            a[j] += amount
        elif move_type < 0.7:
            # Perturb a random block
            start = rng.integers(0, n - 10)
            length = rng.integers(5, min(30, n - start))
            factors = rng.uniform(0.5, 1.5, size=length)
            a[start:start + length] *= factors
        elif move_type < 0.85:
            # Global scale perturb
            factor = rng.uniform(0.7, 1.3)
            half = n // 2
            a[:half] *= factor
            a[half:] /= factor
        else:
            # Multi-point shift
            n_shift = rng.integers(2, 6)
            idx = rng.choice(n, size=n_shift * 2, replace=False)
            amounts = rng.dirichlet(np.ones(n_shift)) * a[idx[::2]].sum() * 0.5
            for k in range(n_shift):
                src = idx[2 * k]
                dst = idx[2 * k + 1]
                amt = amounts[k]
                a[src] = max(0, a[src] - amt)
                a[dst] += amt

        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-10:
            a = a_cur
            continue
        a = a / s_new

        c_new = fast_convolve(a)
        obj_new = c_new.max()
        delta = obj_new - cur_obj

        if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            cur_obj = obj_new
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            a = a_cur
            no_improve += 1

        temp = max(t_end, temp * cooling)

        if no_improve > 10000 and temp <= t_end * 2:
            break

    return best_a.copy()


def local_search(a0, time_limit, n_trials_per_coord=5):
    """Coordinate-wise local search with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(99)
    t0 = time.monotonic()
    no_improve = 0
    factors = [0.5, 0.7, 0.85, 0.9, 1.0 / 0.9, 1.0 / 0.85, 1.0 / 0.7, 1.0 / 0.5]

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        improved = False
        n_perturb = max(4, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-10:
                continue
            for factor in factors:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[i] = old_val

        if improved:
            no_improve = 0
        else:
            no_improve += 1
        a = best_a.copy()

    return best_a.copy()


def generate_inits(parent, rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Parent (if same length)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        # Perturbed parent
        for scale in [0.001, 0.005, 0.02]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    for sigma in [0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma * n)) ** 2)
        inits.append(g / g.sum())

    # Cosine patterns
    for freq in [1, 2, 3, 5]:
        c1 = np.abs(np.cos(np.pi * freq * x / n)) + 0.01
        inits.append(c1 / c1.sum())

    # Sinusoidal (nonnegative via abs)
    for freq in [2, 3, 4, 7, 13]:
        s1 = np.abs(np.sin(2 * np.pi * freq * x / n)) + 0.01
        inits.append(s1 / s1.sum())

    # Random Dirichlet
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Two-point concentrated
    for _ in range(3):
        idx = rng.choice(n, size=rng.integers(2, 8), replace=False)
        r = np.zeros(n)
        r[idx] = 1.0
        inits.append(r / r.sum())

    return inits

                candidate_lengths.add(nl)
        # Also try some specific lengths
        for nl in [64, 96, 128, 160, 192, 256, 320, 384, 448, 512, 640, 768, 1024]:
            if 32 <= nl <= 2048 and nl != n_best:
                candidate_lengths.add(nl)

        candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))

        for nl in candidate_lengths[:8]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(60, rem * 0.08)
            if t_alloc < 10:
                break
            # Interpolate best sequence to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, nl)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-10:
                a_interp = a_interp / s
                a_opt, _ = eg_adaptive(a_interp, time_limit=t_alloc,
                                       eta0=1.0, temp0=0.003, top_k_frac=0.2)
                try_seq(a_opt, f"len_n{nl}")

    # === Phase 7: Local search polish ===
    print(f"\n=== Phase 7: Local Search Polish ===")
    print(f"  Best so far: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t7 = min(80, rem * 0.3)
        if t7 > 10:
            print(f"  Local search (n={len(best_seq)}), alloc={t7:.0f}s")
            a_opt = local_search(best_seq, time_limit=t7)
            try_seq(a_opt, "local_p7")

    # === Phase 8: Final fine EG ===
    print(f"\n=== Phase 8: Final Fine EG ===")
    print(f"  Best so far: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t8 = min(60, rem * 0.6)
            print(f"  Fine EG, alloc={t8:.0f}s")
            a_opt, _ = eg_adaptive(best_seq, time_limit=t8,
                                   eta0=0.1, temp0=0.0005, top_k_frac=0.05)
            try_seq(a_opt, "eg_final")
            # One more local search
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_opt2 = local_search(best_seq, time_limit=min(rem2 - 5, 30))
                try_seq(a_opt2, "local_final")

    # === Phase 9: Adam polish on best ===
    print(f"\n=== Phase 9: Adam Polish ===")
    print(f"  Best so far: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t9 = min(40, rem * 0.7)
            a_opt = softmax_adam(best_seq, time_limit=t9, lr=0.01, top_k_frac=0.05)
            try_seq(a_opt, "adam_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time remaining: {deadline - time.monotonic():.1f}s")
    return [float(x) for x in best_seq]
