def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(v[0], 0.0)])
    
    # Scale to sum = 1 first
    v_scaled = v / target_sum if target_sum != 1.0 else v
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():
        return np.full(n, 1.0 / n) * target_sum
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    result = np.maximum(v_scaled - theta, 0.0)
    return result * target_sum


def solve_lp_fast(f, M, solver='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix efficiently using banded structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=solver)
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    
    # Fallback: greedy heuristic
    g = np.ones(n) / n * M / max(f.max(), 1e-15)
    # Check feasibility
    conv = np.convolve(f, g)
    if conv.max() > M * 1.01:
        g = g * M / conv.max()
    return g

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate-wise local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    delta = 0.01

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            # Try increasing and decreasing a[j] slightly
            for sign in [1, -1]:
                f_trial = f.copy()
                f_trial[j] = max(0, f_trial[j] + sign * delta)
                s_t = f_trial.sum()
                if s_t < 1e-15:
                    continue
                f_trial = f_trial / s_t
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-12:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            # Shrink delta
            delta *= 0.8
            if delta < 1e-7:
                # Add small noise and restart
                rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                delta = 0.005
                no_improve = 0
        else:
            no_improve = 0
            delta = max(delta * 0.95, 1e-6)

    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}. Returns non-negative array summing to target_sum."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([max(v[0], 0.0)] * (1 if target_sum == v[0] else 1), dtype=np.float64) if v[0] > 0 else np.zeros(1)
    # Normalize to sum=1 then scale
    v_scaled = v.astype(np.float64)
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():
        result = np.full(n, 1.0 / n)
    else:
        rho_val = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
        result = np.maximum(v_scaled - theta, 0.0)
        rs = result.sum()
        if rs > 1e-15:
            result = result / rs
        else:
            result = np.full(n, 1.0 / n)
    return result * target_sum


def solve_lp_for_fw(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns (g, dual_y) or (None, None)."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            dual_y = np.zeros(2 * n - 1)
            if res.ineqlin is not None:
                dual_y = np.abs(res.ineqlin)
            return g, dual_y
    except Exception:
        pass
    return None, None


def solve_lp_fallback(f, M):
    """Heuristic fallback if LP solver fails."""
    n = len(f)
    # Greedy: start with uniform, scale down where constraints are violated
    g = np.ones(n)
    for _ in range(50):
        conv = np.convolve(f, g)
        viol = conv - M
        if viol.max() < 1e-10:
            break
        # Reduce g where it contributes most to violations
        contrib = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = min(2 * n - 2, j + n - 1)
            contrib[j] = viol[k_start:k_end + 1].max() * f[j]
        if contrib.max() > 1e-15:
            g -= 0.1 * contrib / contrib.max()
            g = np.maximum(g, 0.0)
        else:
            break
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def fw_lp_core(f0, time_limit, n_gs_steps=25, verbose=False, verbose_every=200):
    """Core Frank-Wolfe with LP oracle and golden-section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while True:
        it += 1
        elapsed = time.monotonic() - t0
        rem = time_limit - elapsed
        if rem < 3.0:
            break

        # Solve LP
        g, dual_y = solve_lp_for_fw(f, M * m_factor)
        if g is None:
            g = solve_lp_fallback(f, M * m_factor)
            if g is None:
                m_factor = min(m_factor * 1.15, 5.0)
                if m_factor > 4.0:
                    rng = np.random.default_rng(it * 7919 + 13)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    conv = fast_convolve(f)
                    M = conv.max()
                    m_factor = 1.0
                    no_improve = 0
                    continue
                continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_factor = min(m_factor * 1.1, 4.0)
            continue
        g = g / g_sum

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 104729 + 7)
                noise_scale = min(0.01, 0.001 * (1.0 + no_improve / 30.0))
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_factor = 1.0
                no_improve = 0
            continue

        # Golden section line search
        lo = 0.0
        hi = 1.0
        for _ in range(n_gs_steps):
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            f1 = np.maximum((1.0 - a1) * f + a1 * g, 0.0)
            s1 = f1.sum()
            v1 = fast_convolve(f1 / s1).max() if s1 > 1e-15 else 1e10
            f2 = np.maximum((1.0 - a2) * f + a2 * g, 0.0)
            s2 = f2.sum()
            v2 = fast_convolve(f2 / s2).max() if s2 > 1e-15 else 1e10
            if v1 < v2:
                hi = a2
            else:
                lo = a1

        best_alpha = (lo + hi) / 2.0
        f_new = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            no_improve += 1
            continue
        f_new = f_new / s_new
        val = fast_convolve(f_new).max()

        if val < M - 1e-15:
            f = f_new
            M = val
            m_factor = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.03, 2.5)

        if verbose and it % verbose_every == 0:
            print(f"      FW iter {it}: M={M:.10f}, best={best_obj:.10f}, no_imp={no_improve}, mfac={m_factor:.3f}")

    return best_f, best_obj


def multi_direction_fw(f0, time_limit, n_directions=3, verbose=False):
    """FW that solves LP with multiple M values to get diverse directions, then picks the best."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while True:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10.0:
            break

        # Solve LP with different M values to get diverse directions
        m_values = [M * 0.95, M * 1.0, M * 1.05, M * 1.15]
        directions = []
        for mv in m_values:
            g, _ = solve_lp_for_fw(f, mv)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                if np.max(np.abs(g - f)) > 1e-10:
                    directions.append(g)
            if len(directions) >= n_directions:
                break

        if not directions:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max()
                no_improve = 0
            continue

        # Try each direction with a quick line search, pick the best
        best_val = M
        best_f_new = None
        best_dir = None
        for g in directions:
            # Quick line search (fewer steps)
            lo2, hi2 = 0.0, 1.0
            for _ in range(15):
                a1 = hi2 - gr * (hi2 - lo2)
                a2 = lo2 + gr * (hi2 - lo2)
                f1 = np.maximum((1.0 - a1) * f + a1 * g, 0.0)
                s1 = f1.sum()
                v1 = fast_convolve(f1 / s1).max() if s1 > 1e-15 else 1e10
                f2 = np.maximum((1.0 - a2) * f + a2 * g, 0.0)
                s2 = f2.sum()
                v2 = fast_convolve(f2 / s2).max() if s2 > 1e-15 else 1e10
                if v1 < v2:
                    hi2 = a2
                else:
                    lo2 = a1
            alpha = (lo2 + hi2) / 2.0
            f_try = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = f_try.sum()
            if st < 1e-15:
                continue
            f_try = f_try / st
            val = fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_f_new = f_try
                best_dir = g

        if best_f_new is not None and best_val < M - 1e-14:
            f = best_f_new
            M = best_val
            no_improve = 0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 53 + 23)
                scale = min(0.008, 0.001 + 0.001 * (no_improve / 40.0))
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                M = fast_convolve(f).max()
                no_improve = 0

        if verbose and it % 50 == 0:
            print(f"      MFW iter {it}: M={M:.10f}, best={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate-wise local search with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    delta = 0.005
    no_improve_rounds = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        # S randomly permute order each round
        perm = rng.permutation(n)
        for j in perm:
            # Try small perturbations
            for sign in [1, -1]:
                f_trial = f.copy()
                f_trial[j] = max(0.0, f_trial[j] + sign * delta)
                s_t = f_trial.sum()
                if s_t < 1e-15:
                    continue
                f_trial = f_trial / s_t
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-13:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                delta = max(delta * 0.9, 1e-7)

        if not improved:
            no_improve_rounds += 1
            delta *= 0.7
            if delta < 1e-8:
                # Restart with noise
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                delta = 0.003
                no_improve_rounds = 0

        if verbose and no_improve_rounds % 50 == 0:
            print(f"      CD: no_improve={no_improve_rounds}, delta={delta:.2e}, obj={best_obj:.10f}")

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr_init=0.002, momentum=0.93, top_k=20, temp_init=0.0003):
    """Subgradient method with momentum, focusing on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Select top-k peaks
        effective_k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -effective_k)[-effective_k:]

        # Softmax weights
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Update with momentum
        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)

        conv_new = fast_convolve(f)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.0005

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-hump
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    two_hump = (g1 + g2 + 0.05)
    inits.append(two_hump / two_hump.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Anti-symmetric perturbation of parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        asym = 0.1 * np.sin(2 * np.pi * x / n)
        px2 = np.maximum(px * (1.0 + asym), 0.0)
        ps2 = px2.sum()
        if ps2 > 1e-10:
            inits.append(px2 / ps2)

    return inits


def evaluate_and_update(a, current_best_score, current_best_seq, label=""):
    """Evaluate sequence and return (should_update, score, normalized_seq)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return False, float('inf'), None
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score):
        print(f"    [{label}] score={score:.10f} (n={len(a)})")
    return (score < current_best_score), score, a_norm


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Enhanced multi-phase search:
    Phase 1: Deep FW-LP with exact line search on parent
    Phase 2: Multi-direction FW for diverse exploration
    Phase 3: Coordinate descent polish
    Phase 4: Multi-length exploration
    Phase 5: Subgradient + annealing restarts
    Phase 6: Final fine polish
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + (total_budget - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: n={len(best_seq)}, score={best_score:.10f}")

    n_cur = len(best_seq)

    # === Phase 1: Deep FW-LP (~500s) ===
    print("\n=== Phase 1: Deep FW-LP (exact line search) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    print(f"  Time allocated: {t1:.0f}s for n={n_cur}")
    try:
        a_opt, _ = fw_lp_core(best_seq, time_limit=t1, verbose=True, verbose_every=200)
        upd, sc, anorm = evaluate_and_update(a_opt, best_score, best_seq, "fw_phase1")
        if upd:
            best_score = sc
            best_seq = anorm
    except Exception as e:
        print(f"  Phase 1 error: {e}")
    print(f"  Phase 1 best: {best_score:.10f}")

    # === Phase 2: Multi-direction FW (~300s) ===
    print("\n=== Phase 2: Multi-Direction FW ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    print(f"  Time allocated: {t2:.0f}s")
    try:
        a_opt, _ = multi_direction_fw(best_seq, time_limit=t2, n_directions=3, verbose=True)
        upd, sc, anorm = evaluate_and_update(a_opt, best_score, best_seq, "mfw_phase2")
        if upd:
            best_score = sc
            best_seq = anorm
            n_cur = len(best_seq)
    except Exception as e:
        print(f"  Phase 2 error: {e}")
    print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: Coordinate Descent (~200s) ===
    print("\n=== Phase 3: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    print(f"  Time allocated: {t3:.0f}s")
    try:
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t3, verbose=True)
        upd, sc, anorm = evaluate_and_update(a_opt, best_score, best_seq, "cd_phase3")
        if upd:
            best_score = sc
            best_seq = anorm
            n_cur = len(best_seq)
    except Exception as e:
        print(f"  Phase 3 error: {e}")
    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Multi-length exploration (~250s) ===
    print("\n=== Phase 4: Multi-Length Exploration ===")
    rem = deadline - time.monotonic()
    t4_total = min(rem * 0.15, 250.0)
    print(f"  Total time allocated: {t4_total:.0f}s")

    lengths_to_try = [320, 448, 512, 384]
    lengths_to_try = [l for l in lengths_to_try if l != n_cur][:4]

    if not lengths_to_try:
        lengths_to_try = [512, 320]

    t_per_len = t4_total / len(lengths_to_try)
    for n_try in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(t_per_len, rem * 0.4)
        if t_alloc < 15:
            continue
        print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
        try:
            x_new = np.linspace(0, 1, n_try)
            x_old = np.linspace(0, 1, len(best_seq))
            a_interp = np.maximum(np.interp(x_new, x_old, best_seq), 0.0)
            a_interp /= a_interp.sum()

            a_opt, _ = fw_lp_core(a_interp, time_limit=t_alloc * 0.7)
            # Quick CD polish
            a_opt2, _ = coordinate_descent_polish(a_opt, time_limit=t_alloc * 0.25)

            for cand in [a_opt, a_opt2]:
                upd, sc, anorm = evaluate_and_update(cand, best_score, best_seq, f"len_{n_try}")
                if upd:
                    best_score = sc
                    best_seq = anorm
                    n_cur = len(best_seq)
        except Exception as e:
            print(f"  Length {n_try} error: {e}")
    print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: Subgradient + Annealing (~200s) ===
    print("\n=== Phase 5: Subgradient + Annealing ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 200.0)
    print(f"  Time allocated: {t5:.0f}s")
    try:
        a_opt, _ = subgradient_polish(best_seq, time_limit=t5, lr_init=0.002, momentum=0.93, top_k=25, temp_init=0.0004)
        upd, sc, anorm = evaluate_and_update(a_opt, best_score, best_seq, "sg_phase5")
        if upd:
            best_score = sc
            best_seq = anorm
            n_cur = len(best_seq)
    except Exception as e:
        print(f"  Phase 5 error: {e}")
    print(f"  Phase 5 best: {best_score:.10f}")

    # === Phase 6: Final polish with perturbation + FW (~remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    print(f"  Remaining: {rem:.0f}s")

    # Run several rounds of: perturb -> FW -> check
    round_time = min(rem * 0.08, 30.0)
    n_rounds = max(1, int(rem * 0.7 / round_time))
    print(f"  {n_rounds} rounds of perturb+FW, {round_time:.0f}s each")

    for ri in range(n_rounds):
        rem = deadline - time.monotonic()
        if rem < round_time + 5:
            break
        t_alloc = min(round_time, rem - 5)
        if t_alloc < 5:
            break
        try:
            # Generate perturbation
            noise_scale = 0.001 * (1.0 + ri * 0.3)
            noise = rng.standard_normal(len(best_seq)) * noise_scale
            f_pert = project_simplex(best_seq + noise)

            a_opt, _ = fw_lp_core(f_pert, time_limit=t_alloc * 0.8)
            # Quick CD
            a_opt2, _ = coordinate_descent_polish(a_opt, time_limit=t_alloc * 0.2)

            for cand in [a_opt, a_opt2]:
                upd, sc, anorm = evaluate_and_update(cand, best_score, best_seq, f"r{ri}")
                if upd:
                    best_score = sc
                    best_seq = anorm
                    n_cur = len(best_seq)
        except Exception as e:
            print(f"  Round {ri} error: {e}")

    # Final: one more CD pass on the best
    rem = deadline - time.monotonic()
    if rem > 10:
        try:
            t_final = min(rem * 0.4, 40.0)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_final)
            upd, sc, anorm = evaluate_and_update(a_opt, best_score, best_seq, "final_cd")
            if upd:
                best_score = sc
                best_seq = anorm
        except Exception:
            pass

    # Ultimate final: tiny FW
    rem = deadline - time.monotonic()
    if rem > 5:
        try:
            a_opt, _ = fw_lp_core(best_seq, time_limit=min(rem * 0.5, 20.0))
            upd, sc, anorm = evaluate_and_update(a_opt, best_score, best_seq, "ultra_final")
            if upd:
                best_score = sc
                best_seq = anorm
        except Exception:
            pass

    # Final verification
    if best_seq is not None:
        final_sc = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_sc) and final_sc < best_score:
            best_score = final_sc

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
