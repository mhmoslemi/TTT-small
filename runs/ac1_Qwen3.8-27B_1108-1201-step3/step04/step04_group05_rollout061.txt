```
def isotonic_regression(v):
    n = len(v)
    # Block starts and ends
    blocks = [(i, i, v[i]) for i in range(n)]  # (start, end, value)
    i = 0
    while i < len(blocks) - 1:
        if blocks[i][2] > blocks[i+1][2]:
            # Merge
            s, e, val = blocks[i]
            s2, e2, val2 = blocks[i+1]
            new_val = (val*(e-s+1) + val2*(e2-s2+1)) / (e2-s+1)
            blocks[i] = (s, e2, new_val)
            blocks.pop(i+1)
        else:
            i += 1
    result = np.zeros(n)
    for s, e, val in blocks:
        result[s:e+1] = val
    return np.maximum(result, 0)
```
