Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient with adaptive softmax-weighted subgradients
2. Coordinate swap local search exploiting convolution structure
3. Multi-length exploration
4. Simulated annealing with FFT-evaluated moves

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def adaptive_eg(a0, time_limit, eta0=1.0, temp0=0.005, top_k=50, verbose=False):
    """
    Exponentiated gradient with adaptive temperature.
    Temperature adapts based on peak separation.
    Uses Polyak-Ruppert averaging.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Polyak-Ruppert average
    avg_a = np.zeros(n)
    max_obj_hist = 0.0
    it = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute top-k peaks
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[indices]
        else:
            indices = np.arange(len(c))
            top_vals = c.copy()
        
        # Adaptive temperature based on peak separation
        sorted_top = np.sort(top_vals)[::-1]
        gap = sorted_top[0] - sorted_top[min(len(sorted_top)-1, len(sorted_top)//2)]
        temp = max(temp0 * 0.1, gap * 0.5)
        
        # Softmax weights
        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(top_vals)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
        
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        # Adaptive learning rate (Polyak-like)
        eta = eta0 / (1 + 0.01 * it)
        
        # Multiplicative update
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        # Polyak-Ruppert average
        max_obj_hist = max(max_obj_hist, obj_new)
        avg_a += a_new / (max_obj_hist + 1e-15)
        avg_a_n = avg_a / avg_a.sum() if avg_a.sum() > 1e-15 else a_new
        
        a = a_new
        c = c_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Also check averaged
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a_n)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a_n.copy()
                no_improve = 0
        
        if no_improve > 5000:
            # Restart with perturbation
            rng = np.random.default_rng(it * 17 + 3)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            avg_a = np.zeros(n)
            max_obj_hist = 0.0
    
    return best_a.copy(), best_obj

Actually, let me think of a simpler but effective local search:

**Swap-based local search:**
1. Compute conv, find top-k peaks
2. For each peak k, the "pressure" on coordinate i is `a[k-i]` (the amount of mass at position k that "sees" coordinate i)
3. We want to move mass from coordinates with high pressure across all peaks to coordinates with low pressure
4. Specifically: for peak k*, coordinate i contributes `a[i]*a[k*-i]` to conv[k*]
5. If we move delta from i to j, the change in conv[k*] is approximately `delta*(a[k*-j] - a[k*-i])`
6. So we want to move from high `a[k*-i]` to low `a[k*-j]`

def coordinate_swap_search(a0, time_limit, top_k_peaks=20, verbose=False):
    """
    Local search: find mass transfers (i -> j) that reduce the max convolution.
    Uses analytical first-order approximation for the change in conv.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top peaks
        if top_k_peaks < len(c):
            peak_indices = np.argpartition(c, -top_k_peaks)[-top_k_peaks:]
        else:
            peak_indices = np.arange(len(c))
        
        # For each coordinate, compute "pressure" = weighted sum of a[peak - coord]
        # over all active peaks
        pressure = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # conv[k] includes a[j] * a[k-j] for j in [js, je]
            # The "pressure" on coordinate j from peak k is a[k-j]
            pressure[js:je+1] += a[k-je:k-js+1][::-1]
        
        # We want to move mass from high-pressure to low-pressure coordinates
        # Sort by pressure
        sorted_coords = np.argsort(pressure)
        
        # Try moving from high pressure to low pressure
        improved = False
        n_tries = 0
        
        # Pick random pairs: high pressure source, low pressure target
        for _ in range(min(1000, n // 4)):
            n_tries += 1
            # Random high-pressure coordinate
            hi_idx = rng.integers(0, n)
            lo_idx = rng.integers(0, n)
            if hi_idx == lo_idx:
                continue
            if pressure[hi_idx] < pressure[lo_idx]:
                hi_idx, lo_idx = lo_idx, hi_idx
            
            if a[hi_idx] < 1e-10:
                continue
            
            # Try a small transfer
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                delta = a[hi_idx] * frac
                if a[hi_idx] - delta < 0:
                    continue
                a[hi_idx] -= delta
                a[lo_idx] += delta
                a = a / a.sum()
                
                c2 = fast_convolve(a)
                obj2 = c2.max()
                
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break
        
        if improved:
            no_improve = 0
            c = fast_convolve(a)
        else:
            no_improve += 1
            if no_improve > 50:
                # Try larger perturbations
                break
    
    return best_a.copy(), best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Exponentiated gradient with adaptive temperature and Polyak averaging
2. Peak-flattening local search with analytical gradient exploitation  
3. Multi-length exploration with diverse initializations
4. Final fine-tuning with coordinate-level optimization
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient_weighted(a, c, weights, peak_indices):
    """Compute weighted subgradient at given peaks."""
    n = len(a)
    sg = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        if w < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w * a[k - je:k - js + 1][::-1]
    return sg

def eg_optimize(a0, time_limit, eta0=1.0, temp_mode='adaptive', top_k_frac=0.3,
                eta_schedule='polyak', verbose=False):
    """
    Exponentiated gradient with adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Polyak-Ruppert
    avg_a = a.copy()
    f0 = best_obj
    it = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Get top peaks
        if top_k < len(c):
            idx_arr = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[idx_arr]
        else:
            idx_arr = np.arange(len(c))
            top_vals = c.copy()
        
        # Adaptive temperature
        if temp_mode == 'adaptive':
            sorted_top = np.sort(top_vals)[::-1]
            if len(sorted_top) > 1:
                gap = sorted_top[0] - sorted_top[len(sorted_top)//2]
            else:
                gap = sorted_top[0]
            temp = max(1e-6, gap * 0.3)
        elif temp_mode == 'decay':
            temp = max(0.0001, 0.01 * (0.999 ** it))
        else:
            temp = 0.005
        
        # Softmax weights
        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(idx_arr))
            w[np.argmax(top_vals)] = 1.0
        else:
            w = w / ws
        
        # Learning rate
        if eta_schedule == 'polyak':
            eta = eta0 * (f0 + best_obj) / (f0 * (it + 1) + 1e-15)
            eta = min(eta, eta0)  # cap at initial
        elif eta_schedule == 'sqrt':
            eta = eta0 / np.sqrt(it + 1)
        else:
            eta = eta0 / (1 + 0.005 * it)
        
        # Subgradient
        sg = compute_subgradient_weighted(a, c, w, idx_arr)
        sg_max = sg.max()

        order = np.argsort(peak_vals)[::-1]

        k_main = int(peak_idx[0])

        # Also compute marginal for second peak to avoid creating new peaks

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the optimal value (given others fixed) minimizes the max of conv.
    We binary search for the optimal value of each coordinate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        # Shuffle coordinates
        coords = rng.permutation(n)
        improved_this_round = False
        
        for i in coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            if a[i] < 1e-15:
                continue
            
            # Try different values for a[i]
            # The effect on conv is quadratic in a[i]
            # conv[k] = a[i]*a[k-i] + rest_k for k where i is in range
            # We want to find a[i] that minimizes max_k conv[k]
            
            # Binary search approach: try values from 0 to 2*a[i]
            old_val = a[i]
            best_local_val = old_val
            best_local_obj = c.max()
            
            for trial_factor in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                new_val = old_val * trial_factor
                if new_val < 0:
                    continue
                a[i] = new_val
                # Renormalize
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_local_obj - 1e-14:
                    best_local_obj = obj2
                    best_local_val = a.copy()
                
                a = a / a.sum()  # ensure normalization consistency
            
            if best_local_obj < c.max() - 1e-14:
                a = best_local_val.copy()
                c = fast_convolve(a)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved_this_round = True
        
        if not improved_this_round:
            break
    
    return best_a.copy(), best_obj

def generate_diverse_inits(rng, n, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Exponential decay (both directions)
    for beta in [0.5, 1.0, 2.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.15*n))**2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    
    # Sine
    s1 = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits.append(s1 / s1.sum())
    
    # Random Dirichlet (various concentrations)
    for alpha in [0.1, 0.5, 1.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # Checkerboard-like
    for alpha in [0.3, 0.5, 0.7]:
        cb = np.where(np.arange(n) % 2 == 0, 1+alpha, 1-alpha)
        inits.append(cb / cb.sum())
    
    # 3-periodic
    for alpha in [0.3, 0.5]:
        p3 = 1 + alpha * np.cos(2*np.pi*np.arange(n)/3)
        inits.append(p3 / p3.sum())
    
    # Parent-based (if available)
    if parent is not None:
        if len(parent) == n:
            for scale in [0.01, 0.05, 0.1]:
                p = parent + rng.standard_normal(n) * scale
                p = np.maximum(p, 0)
                ps = p.sum()
                if ps > 1e-10:
                    inits.append(p / ps)
        else:
            # Interpolate parent to length n
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, parent)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-10:
                inits.append(a_interp / s)
    
    return inits

**Idea: Use the structure of the problem more directly.**

The objective is `2n * max_k conv[k] / (sum a)^2`. Since we normalize `sum a = 1`, we minimize `2n * max_k conv[k]`.

At the optimum, suppose the active set (peaks achieving the max) is `S`. Then we want:
- `conv[k] = M` for all `k in S`
- `conv[k] <= M` for all `k not in S`
- `a[i] >= 0`
- `sum a = 1`

This is a system of equations/inequalities. For a given active set `S`, we can try to solve:
- Minimize `M` subject to `conv[k] <= M` for all k, `a >= 0`, `sum a = 1`

This is a QCQP (quadratically constrained quadratic program). The previous solution tried to solve this with CVXPY, but it was limited to n <= 400 and was slow.

An alternative: use the KKT conditions. At the optimum:
- For active k: `conv[k] = M`
- For inactive k: `conv[k] < M`
- For `a[i] > 0`: the gradient of the Lagrangian w.r.t. `a[i]` is zero
- For `a[i] = 0`: the gradient is >= 0 (complementary slackness)

The gradient of `M = max_k conv[k]` at the active set is a convex combination of the gradients of the active `conv[k]`. Each `grad_a conv[k] = 2 * P_k * a` where `P_k` is the convolution matrix at lag k.

So at the optimum: `sum_{k in S} lambda_k * 2 * P_k * a = nu * 1 + mu`
where `lambda_k >= 0`, `sum lambda_k = 1`, `nu` is the multiplier for `sum a = 1`, and `mu_i >= 0` for `a_i = 0`.

This gives us: for all `i` with `a[i] > 0`:
`sum_{k in S} lambda_k * 2 * (P_k a)[i] = nu`

And for all `i` with `a[i] = 0`:
`sum_{k in S} lambda_k * 2 * (P_k a)[i] >= nu`

This is what the subgradient method is implicitly trying to satisfy. The question is whether we can do better.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase pipeline:
1. Exponentiated gradient with adaptive temperature and Polyak-Ruppert averaging
2. Peak-flattening local search exploiting analytical convolution structure
3. Simulated annealing with convolution-aware mass-transfer moves
4. Projected gradient descent (simplex projection) as an alternative mirror
5. Multi-length exploration and diverse initialization
6. Final coordinate-level polishing
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_weighted_subgradient(a, peak_indices, weights):
    """Compute weighted sum of subgradients at given peaks."""
    n = len(a)
    sg = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        if w < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += 2.0 * w * a[k - je:k - js + 1][::-1]
    return sg


def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    proj = np.maximum(v - theta, 0)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s


def eg_optimize(a0, time_limit, eta0=1.5, temp_mode='adaptive', top_k_frac=0.35,
                eta_schedule='polyak', verbose=False):
    """
    Exponentiated gradient (multiplicative weights) on the simplex
    with adaptive temperature and Polyak-Ruppert averaging.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy()
    f0 = best_obj
    it = 0

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if top_k < len(c):
            idx_arr = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[idx_arr]
        else:
            idx_arr = np.arange(len(c))
            top_vals = c.copy()

        # Adaptive temperature based on peak spread
        if temp_mode == 'adaptive':
            sorted_top = np.sort(top_vals)[::-1]
            gap = sorted_top[0] - sorted_top[max(0, len(sorted_top) // 2)]
            temp = max(1e-7, gap * 0.25)
        elif temp_mode == 'decay':
            temp = max(1e-5, 0.02 * (0.9995 ** it))
        else:
            temp = 0.005

        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(idx_arr))
            w[np.argmax(top_vals)] = 1.0
        else:
            w = w / ws

        # Learning rate schedule
        if eta_schedule == 'polyak':
            eta = eta0 * (f0 + best_obj) / (f0 * (1.0 + float(it)) + 1e-15)
            eta = min(eta, eta0)
        elif eta_schedule == 'sqrt':
            eta = eta0 / np.sqrt(1.0 + float(it))
        else:
            eta = eta0 / (1.0 + 0.003 * it)

        sg = compute_weighted_subgradient(a, idx_arr, w)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        c = c_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 2000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-13:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
                no_improve = 0

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s", flush=True)

        if no_improve > 10000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            avg_a = a.copy()

    return best_a.copy(), best_obj


def projected_gradient_descent(a0, time_limit, eta0=0.1, top_k_frac=0.3, verbose=False):
    """
    Projected gradient descent on the simplex (Euclidean mirror).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if top_k < len(c):
            idx_arr = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[idx_arr]
        else:
            idx_arr = np.arange(len(c))
            top_vals = c.copy()

        # Uniform weights over active peaks (or softmax)
        sorted_top = np.sort(top_vals)[::-1]
        gap = sorted_top[0] - sorted_top[max(0, len(sorted_top) // 2)]
        temp = max(1e-7, gap * 0.4)
        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(idx_arr))
            w[np.argmax(top_vals)] = 1.0
        else:
            w = w / ws

        eta = eta0 / (1.0 + 0.01 * it)

        sg = compute_weighted_subgradient(a, idx_arr, w)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        sg = sg / sg_max

        # Gradient step + projection
        a_step = a - eta * sg
        a_new = proj_simplex(a_step)

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        a = a_new
        c = c_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s", flush=True)

        if no_improve > 8000:
            rng_local = np.random.default_rng(it * 13 + 5)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = proj_simplex(a)
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0

    return best_a.copy(), best_obj


def peak_flatten_search(a0, time_limit, verbose=False):
    """
    Local search targeting the highest convolution peaks.
    Uses analytical marginal contributions to guide mass transfers.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Top peaks
        top_k = min(15, len(c))
        peak_idx = np.argpartition(c, -top_k)[-top_k:]
        peak_vals = c[peak_idx]
        order = np.argsort(peak_vals)[::-1]
        peak_idx = peak_idx[order]
        peak_vals = peak_vals[order]

        # Compute marginal (sensitivity) for main peak
        k_main = int(peak_idx[0])
        js = max(0, k_main - n + 1)
        je = min(n - 1, k_main)
        marginal_main = np.zeros(n)
        if js <= je:
            marginal_main[js:je + 1] = a[k_main - je:k_main - js + 1][::-1]

        # Marginal for second peak (to avoid creating new peaks)
        if len(peak_idx) > 1:
            k_second = int(peak_idx[1])
            js2 = max(0, k_second - n + 1)
            je2 = min(n - 1, k_second)
            marginal_second = np.zeros(n)
            if js2 <= je2:
                marginal_second[js2:je2 + 1] = a[k_second - je2:k_second - js2 + 1][::-1]
        else:
            marginal_second = np.zeros(n)

        # Combined pressure: we want to reduce main peak without increasing second
        pressure = marginal_main + 0.5 * marginal_second

        # Sources: high pressure, Targets: low pressure
        src_mask = pressure > np.median(pressure)
        tgt_mask = pressure < np.median(pressure)
        src_coords = np.where(src_mask)[0]
        tgt_coords = np.where(tgt_mask)[0]

        if len(src_coords) == 0:
            src_coords = np.argsort(pressure)[::-1][:max(1, n // 4)]
        if len(tgt_coords) == 0:
            tgt_coords = np.argsort(pressure)[:max(1, n // 4)]

        improved = False
        n_tries = min(300, n)

        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            src = src_coords[rng.integers(0, len(src_coords))]
            tgt = tgt_coords[rng.integers(0, len(tgt_coords))]
            if src == tgt or a[src] < 1e-12:
                continue

            for frac in [0.03, 0.07, 0.12, 0.2]:
                delta = a[src] * frac
                if delta < 1e-15:
                    continue
                a[src] -= delta
                a[tgt] += delta
                a = a / a.sum()

                c2 = fast_convolve(a)
                obj2 = c2.max()

                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

            if improved:
                break

        if improved:
            no_improve = 0
            c = fast_convolve(a)
        else:
            no_improve += 1
            if no_improve > 80:
                # Perturb and retry
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                ps = a.sum()
                if ps > 1e-10:
                    a = a / ps
                else:
                    a = best_a.copy()
                c = fast_convolve(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a = best_a.copy()
                    c = fast_convolve(a)
                    if no_improve > 200:
                        break

        if verbose and it % 100 == 0:
            print(f"    PeakFlatten iter {it}: obj={best_obj:.10f}", flush=True)

    return best_a.copy(), best_obj


def simulated_annealing(a0, time_limit, verbose=False):
    """
    Simulated annealing with mass-transfer moves between random coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    current_obj = c.max()
    best_obj = current_obj
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    T_start = 0.01 * current_obj
    T_end = 0.0001 * current_obj
    total_iters = max(10000, int(time_limit * 3000))

    no_improve = 0

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / total_iters
        T = T_start * (T_end / T_start) ** progress

        # Move: transfer mass from random i to random j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] < 1e-12:
            continue

        frac = rng.uniform(0.01, 0.15)
        delta = a[i] * frac

        old_i = a[i]
        old_j = a[j]
        a[i] -= delta
        a[j] += delta
        a = a / a.sum()

        c2 = fast_convolve(a)
        new_obj = c2.max()

        delta_obj = new_obj - current_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-15)):
            current_obj = new_obj
            c = c2
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
        else:
            a = best_a.copy() if current_obj > best_obj else a
            a[i] = old_i
            a[j] = old_j
            a = a / a.sum()
            no_improve += 1

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter {it}: best={best_obj:.10f}, T={T:.6f}, elapsed={elapsed:.1f}s", flush=True)

        if no_improve > 50000:
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            current_obj = c.max()
            no_improve = 0

    return best_a.copy(), best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Coordinate-level polishing: for each coordinate, try scaling it
    and keep the best.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(31415)

    while time.monotonic() - t0 < time_limit:
        coords = rng.permutation(n)
        improved_round = False

        for i in coords:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            old_val = a[i]
            best_local_obj = c.max()
            best_local_a = a.copy()

            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = old_val * factor
                a = a / a.sum()
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_local_obj - 1e-14:
                    best_local_obj = obj2
                    best_local_a = a.copy()

            if best_local_obj < c.max() - 1e-14:
                a = best_local_a.copy()
                c = fast_convolve(a)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved_round = True

        if not improved_round:
            # Try adding tiny mass to zero coordinates
            zero_coords = np.where(a < 1e-10)[0]
            if len(zero_coords) > 0:
                for zc in zero_coords[:min(20, len(zero_coords))]:
                    if time.monotonic() - t0 > time_limit:
                        break
                    a[zc] = 1e-6
                    # Take from the largest coordinate
                    max_idx = np.argmax(a)
                    a[max_idx] *= 0.999
                    a = a / a.sum()
                    c2 = fast_convolve(a)
                    if c2.max() < best_obj - 1e-14:
                        best_obj = c2.max()
                        best_a = a.copy()
                    else:
                        a = best_a.copy()

            break

    return best_a.copy(), best_obj


def generate_diverse_inits(rng, n, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine / Sine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    s1 = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits.append(s1 / s1.sum())

    # Checkerboard patterns
    for alpha in [0.3, 0.5, 0.7]:
        cb = np.where(np.arange(n) % 2 == 0, 1 + alpha, max(0.01, 1 - alpha))
        inits.append(cb / cb.sum())

    # 3-periodic
    for alpha in [0.3, 0.5]:
        p3 = 1.0 + alpha * np.cos(2 * np.pi * np.arange(n) / 3.0)
        p3 = np.maximum(p3, 0.01)
        inits.append(p3 / p3.sum())

    # 4-periodic
    p4 = 1.0 + 0.4 * np.cos(2 * np.pi * np.arange(n) / 4.0)
    p4 = np.maximum(p4, 0.01)
    inits.append(p4 / p4.sum())

    # Random Dirichlet
    for alpha in [0.2, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Parent-based perturbations
    if parent is not None:
        if len(parent) == n:
            for scale in [0.01, 0.03, 0.08]:
                p = parent + rng.standard_normal(n) * scale
                p = np.maximum(p, 0)
                ps = p.sum()
                if ps > 1e-10:
                    inits.append(p / ps)
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, parent)
            a_interp = np.maximum(a_interp, 0)
            s2 = a_interp.sum()
            if s2 > 1e-10:
                inits.append(a_interp / s2)

    return inits


def track_best(a, current_best_score, current_best_seq):
    """Evaluate sequence and update best if improved. Returns (score, best_score, best_seq)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf'), current_best_score, current_best_seq
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < current_best_score:
        return score, score, a.copy()
    return score, current_best_score, current_best_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining EG, projected gradient,
    peak-flattening, simulated annealing, and coordinate polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None

    def try_seq(a, label=""):
        nonlocal best_score, best_seq
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384", flush=True)
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.8f}", flush=True)

    # === Phase 1: Primary EG ===
    print("\n=== Phase 1: Primary EG ===", flush=True)
    rem = deadline - time.monotonic() - 700
    if rem > 30:
        t1 = min(450, rem)
        print(f"  EG alloc={t1:.0f}s", flush=True)
        a_opt, _ = eg_optimize(parent, time_limit=t1, eta0=2.0,
                               temp_mode='adaptive', top_k_frac=0.4,
                               eta_schedule='polyak', verbose=True)
        try_seq(a_opt, "eg_p1")

    # === Phase 2: Peak Flattening ===
    print(f"\n=== Phase 2: Peak Flattening ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            t2 = min(180, rem)
            print(f"  alloc={t2:.0f}s", flush=True)
            a_opt, _ = peak_flatten_search(best_seq, time_limit=t2)
            try_seq(a_opt, "peaks")

    # === Phase 3: Projected Gradient Descent ===
    print(f"\n=== Phase 3: Projected Gradient ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t3 = min(120, rem)
            print(f"  alloc={t3:.0f}s", flush=True)
            a_opt, _ = projected_gradient_descent(best_seq, time_limit=t3,
                                                   eta0=0.05, top_k_frac=0.3)
            try_seq(a_opt, "pgd")

    # === Phase 4: EG Variations ===
    print(f"\n=== Phase 4: EG Variations ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)

    configs = [
        {"eta0": 0.5, "temp_mode": 'decay', "top_k_frac": 0.2, "eta_schedule": 'sqrt'},
        {"eta0": 4.0, "temp_mode": 'adaptive', "top_k_frac": 0.5, "eta_schedule": 'linear'},
        {"eta0": 1.0, "temp_mode": 'adaptive', "top_k_frac": 0.6, "eta_schedule": 'polyak'},
    ]

    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(80, rem // max(1, len(configs) - ci))
        if t_alloc < 15:
            break
        a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg, verbose=False)
        try_seq(a_opt, f"eg_v{ci}")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t5 = min(150, rem)
            print(f"  alloc={t5:.0f}s", flush=True)
            a_opt, _ = simulated_annealing(best_seq, time_limit=t5)
            try_seq(a_opt, "sa")

    # === Phase 6: Multi-Length Exploration ===
    print(f"\n=== Phase 6: Multi-Length ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)

    alt_lengths = [320, 448, 512, 640]
    for n_alt in alt_lengths:
        rem = deadline - time.monotonic() - 80
        if rem < 25:
            break
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_alt)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            s2 = a_interp.sum()
            if s2 > 1e-10:
                a_interp = a_interp / s2
            else:
                a_interp = np.ones(n_alt) / n_alt
        else:
            a_interp = np.ones(n_alt) / n_alt

        t_alloc = min(60, rem // 2)
        if t_alloc < 10:
            break
        a_opt, _ = eg_optimize(a_interp, time_limit=t_alloc, eta0=1.5,
                               temp_mode='adaptive', top_k_frac=0.3,
                               eta_schedule='polyak', verbose=False)
        try_seq(a_opt, f"len{n_alt}")

    # === Phase 7: Multi-Init on Best Length ===
    print(f"\n=== Phase 7: Multi-Init ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        inits = generate_diverse_inits(rng, n_best, parent=best_seq)
        # Try a subset of diverse inits
        n_to_try = min(8, len(inits))
        for i in range(1, n_to_try):
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            t_alloc = min(35, rem // max(1, n_to_try - i))
            if t_alloc < 8:
                break
            a_opt, _ = eg_optimize(inits[i], time_limit=t_alloc, eta0=1.0,
                                   temp_mode='adaptive', top_k_frac=0.3,
                                   eta_schedule='polyak', verbose=False)
            try_seq(a_opt, f"init{i}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===", flush=True)
    print(f"  Best: {best_score:.8f}", flush=True)

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            # Coordinate polish
            a_opt, _ = coordinate_polish(best_seq, time_limit=min(rem * 0.6, 30))
            try_seq(a_opt, "coord_polish")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            # Quick peak flatten
            a_opt, _ = peak_flatten_search(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_peaks")

        rem = deadline - time.monotonic() - 2
        if rem > 3:
            # Final fine EG
            a_opt, _ = eg_optimize(best_seq, time_limit=min(rem, 10),
                                   eta0=0.3, temp_mode='adaptive',
                                   top_k_frac=0.5, eta_schedule='linear')
            try_seq(a_opt, "final_eg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
