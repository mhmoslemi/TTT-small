import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)] if v[0] > 0 else [1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - 1) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - 1.0) / np.arange(1, n + 1)) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sg(f, n, temp, top_k):
    conv = fast_convolve(f)
    k_act = min(top_k, len(conv))
    if k_act < len(conv):
        peaks = np.argpartition(conv, -k_act)[-k_act:]
    else:
        peaks = np.arange(len(conv))
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[int(np.argmax(pv))] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg, conv

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0

        it += 1
        sg, _ = compute_sg(f, n, temp, top_k)
        v = mom * v + sg

        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0

def coordinate_descent(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

            if old < 1e-12:

                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False

                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()

                        best_f = f.copy()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_chirp_init(n):
    j = np.arange(n)
    a = 1.0 + np.cos(2.0 * np.pi * j * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_init(n):
    j = np.arange(n)
    a = np.cos(np.pi * j / (2.0 * (n - 1))) ** 2 if n > 1 else np.array([1.0])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_random_init(n, seed=42):
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n))
    return a


def make_beta_init(n, alpha=2.0, beta=2.0):
    j = np.linspace(0, 1, n)
    # Beta distribution pdf (unnormalized)
    a = j ** (alpha - 1) * (1 - j + 1e-10) ** (beta - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
