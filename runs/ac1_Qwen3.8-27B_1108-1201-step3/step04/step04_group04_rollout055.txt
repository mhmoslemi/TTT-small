Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def peak_gradient(a, conv, top_k=20, temp=0.005):
    """Compute softmax-weighted gradient over top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        idx = np.arange(len(conv))
        vals = conv
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
    
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-12:
            continue
        k = int(idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        grad[js] += w[i] * a[k - js]
    return grad

def peak_coordinate_descent(a0, time_limit, seed=0):
    """Peak-directed coordinate descent: transfer mass to reduce the peak lag."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    steps_per_batch = max(100, 200000 // n)
    
    while time.monotonic() - t0 < time_limit:
        # Find peak lag
        peak_k = int(np.argmax(conv))
        
        for step in range(steps_per_batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            # Pick a random position j
            j = rng.integers(0, n)
            # The "mirror" position for the peak lag
            mirror = peak_k - j
            if mirror < 0 or mirror >= n:
                continue
            
            if j == mirror:
                continue
            
            # Try transferring mass from j to mirror and vice versa
            # If a[mirror] < a[x] where x = peak_k - j, then increasing a[j] helps
            # Actually: d(conv[peak_k])/d(a[j]) = a[peak_k - j] = a[mirror]
            # So to reduce conv[peak_k], we want to decrease a[j] if a[mirror] > 0
            # Or transfer from j to some j' where a[peak_k - j'] > a[peak_k - j]
            
            # Simple approach: try small transfers
            a_j = a[j]
            if a_j < 1e-10:
                continue
            
            # Try decreasing a[j] and increasing another element
            j2 = rng.integers(0, n)
            if j2 == j:
                continue
            
            for delta in [0.001, 0.005, 0.01]:
                if a[j] < delta:
                    continue
                a_try = a.copy()
                a_try[j] -= delta
                a_try[j2] += delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine-based
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():

                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:

    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Multi-scale peak-weighted Nesterov gradient descent with adaptive annealing
2. Peak-directed coordinate descent with incremental convolution updates
3. Multi-n exploration and diverse initializations
4. Iterative refinement cycles with perturbation-based restarts
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute the objective value 2n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def peak_gradient(a, conv, top_k, temp):
    """Compute softmax-weighted gradient over top-k peaks of the auto-convolution."""
    n = len(a)
    if top_k >= len(conv):
        idx = np.arange(len(conv))
        vals = conv.copy()
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx].copy()

    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-12:
            continue
        k = int(idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        grad[js] += w[i] * a[k - js]
    return grad


def nesterov_gradient_optimize(a0, time_limit, lr_init, momentum, top_k_init, temp_factor, seed):
    """Nesterov-accelerated gradient descent with adaptive peak weighting and annealing."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    temp_init = M * temp_factor

    v = np.zeros(n)
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    top_k = top_k_init
    no_improve = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = peak_gradient(y, conv_y, top_k=top_k, temp=temp)

        # Momentum update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            continue

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter updates
        lr *= 0.9998
        temp *= 1.003
        mom = min(0.985, mom + 0.00005)

        # Gradually increase top_k for broader peak handling
        if no_improve % 80 == 0 and no_improve > 0:
            top_k = min(300, top_k + 15)

        # Perturbation-based restart
        if no_improve >= 200:
            scale = 0.004 + 0.004 * min(restart_count, 8)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.002, lr_init * 0.6)
            temp = conv.max() * 0.03
            top_k = max(10, top_k_init // 2)
            mom = momentum
            restart_count += 1

                break

    return best_a, best_obj

def symmetric_pair_descent(a0, time_limit, seed):
    """Try symmetric pair modifications around the peak lag."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0).copy()

    while time.monotonic() - t0 < time_limit:
        # Consider top 5 peaks
        top_peaks = np.argpartition(conv, -5)[-5:]

        for _ in range(max(100, 100000 // n)):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            k = int(top_peaks[rng.integers(0, 5)])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start >= j_end:
                continue

            # Pick a pair (j, k-j)
            j = int(rng.integers(j_start, j_end + 1))
            jm = k - j
            if j == jm or jm < 0 or jm >= n:

            a_j = a[j]
            a_jm = a[jm]
            if a_j < 1e-8 and a_jm < 1e-8:

            # Try: decrease the larger, increase the smaller
            if a_j > a_jm:
                src, dst = j, jm
            else:
                src, dst = jm, j

            src_val = a[src]
            if src_val < 1e-8:
                continue

            for frac in [0.01, 0.03, 0.08, 0.15]:
                delta = src_val * frac
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-13:
                    a = a_try
                    conv = conv_try

                    best_a = a.copy()
                    no_improve = 0
                    break
            else:

        if no_improve > 500:
            scale = 0.002
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    inits['sine3'] = (np.sin(3 * np.pi * x / n) ** 2 + 0.01)
    inits['sine3'] /= inits['sine3'].sum()

    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Triangular
    tri = np.minimum(x + 1, n - x)
    inits['triangular'] = tri / tri.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits
