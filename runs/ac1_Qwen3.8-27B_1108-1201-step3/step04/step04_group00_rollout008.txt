    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
with T-annealing, momentum, multi-length exploration, and coordinate descent
polishing. Key insight: use log-sum-exp smooth approximation of max to get
gradients, anneal temperature from smooth to sharp, explore increasing
sequence lengths for more degrees of freedom.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-ish interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use quadratic interpolation for smoother results
    if old_n >= 3:
        from scipy.interpolate import interp1d
        func = interp1d(old_x, f, kind='cubic', fill_value=(f[0], f[-1]), bounds_error=False)
        f_interp = func(new_x)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_grad_smooth(a, T, top_k=None):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*(a*a)[k])) w.r.t. a.
    Returns (smooth_max_value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights (numerically stable)
    M = conv.max()
    scaled = T * (conv - M)
    exp_scaled = np.exp(scaled)
    S = exp_scaled.sum()
    weights = exp_scaled / S
    
    smooth_val = M + np.log(S) / T
    
    # For efficiency, use only top-k weights
    if top_k is not None and top_k < len(weights):
        # Find indices where weights are significant
        threshold_idx = np.argpartition(weights, -(top_k - 1))[-top_k:]
        significant = threshold_idx[weights[threshold_idx] > 1e-14]
    else:
        significant = np.where(weights > 1e-14)[0]
    
    # grad[j] = 2 * sum_k weights[k] * a[k-j]
    grad = np.zeros(n)
    for k in significant:
        w = weights[k]
        if w < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w * a[k - idx]
    
    return smooth_val, grad

def coordinate_descent(a0, time_limit, verbose=False):

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        order = np.arange(n)
        if passes > 1:
            rng = np.random.default_rng(passes * 12345)
            rng.shuffle(order)

            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(a).max()
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale

                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial /= st
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:

            break
        if verbose and passes % 3 == 0:

    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t_cd = min(rem * 0.7, 200)
            print(f"  CD time: {t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "P4_cd")
            
            # Pairwise swap
            rem2 = deadline - time.monotonic() - 20
            if rem2 > 20:
                t_ps = min(rem2 * 0.6, 80)
                print(f"  Pairwise swap time: {t_ps:.0f}s")
                a_opt2, _ = pairwise_swap_optimize(best_seq, time_limit=t_ps, verbose=True)
                try_seq(a_opt2, "P4_swap")
    
    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            # Very fine gradient step
            a_opt, _ = grad_optimize(best_seq, time_limit=min(rem * 0.5, 60),
                                      T_start=200.0, T_end=10000.0,
                                      lr_init=0.005, momentum_init=0.95,
                                      top_k_frac=0.3, verbose=True, label="P5")
            try_seq(a_opt, "P5_fine")
        
        # Tiny perturbations
        n_best = len(best_seq)
        for _ in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            noise_scale = 0.001
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            a_opt, _ = coordinate_descent(f_pert, time_limit=min(rem, 10))
            try_seq(a_opt, "P5_perturb")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
