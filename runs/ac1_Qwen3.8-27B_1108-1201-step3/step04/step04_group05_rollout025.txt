def fast_convolve(a):

def project_simplex(v, tol=1e-15):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

        return np.array([1.0])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton (GS) iterations for flat autocorrelation initialization
2. Smoothed gradient descent (log-sum-exp of peaks) with Adam optimizer

4. Local search with structured perturbations

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gs_initialize(n, n_iter=200, relaxation=1.0, seed=42):
    """Initialize using Gerchberg-Saxton for flat autocorrelation."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * 0.3)
    
    # Zero-pad to FFT-friendly size
    fft_size = 1 << (3 * n - 2).bit_length()
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    for it in range(n_iter):
        # FFT
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Target: flat magnitude (keep DC)
        mag = np.abs(A)
        total_power = np.sum(mag ** 2)
        dc = mag[0]
        non_dc_power = total_power - dc ** 2
        n_bins = len(mag) - 1
        if non_dc_power > 0:
            target_mag = np.sqrt(non_dc_power / n_bins)
        else:
            target_mag = 0
        
        # Relaxation
        new_mag = mag.copy()
        new_mag[1:] = (1 - relaxation) * mag[1:] + relaxation * target_mag
        
        # Reconstruct with new magnitudes, same phases
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_pad_new = np.fft.irfft(A_new, fft_size)
        
        # Enforce non-negativity on support
        a_new = np.maximum(a_pad_new[:n], 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a = a_new
        peak = fast_convolve(a).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
    
    return best_a, best_peak
