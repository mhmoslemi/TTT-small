import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_row(f, k):
    """Return the row of the convolution matrix for lag k: (f*g)[k] = row · g."""
    n = len(f)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    row = np.zeros(n)
    for j in range(js, je + 1):
        row[j] = f[k - j]
    return row


def solve_reduced_lp(f, conv_f, active_lags):
    """Solve reduced LP: min t s.t. t >= conv_f[k] + 2*(B_k·g) for k in active_lags,
    sum(g)=0, g >= -f. Returns direction g or None."""
    n = len(f)
    m = len(active_lags)
    
    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    b_ub = np.zeros(m)
    
    for idx, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j + 1] = 2.0 * f[k - j]
        b_ub[idx] = -conv_f[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    return None
