```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with log-sum-exp approximation of the max, simplex projection, momentum,
and temperature annealing. Combines multiple optimization phases:
smoothed GD, coordinate descent, and local search.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, T):
    """Compute smoothed objective and its gradient.
    Objective: T * log(sum_k exp(conv(a,a)[k] / T))
    Gradient: dObj/da[m] = 2 * sum_k w[k] * a[k-m] where w = softmax(conv/T)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    shifted -= shifted.max()  # numerical stability (redundant since max_c already subtracted, but safe)
    log_sum = max_c / T + np.log(np.sum(np.exp(shifted)))
    obj = T * log_sum
    
    # Softmax weights
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient via cross-correlation
    L = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    corr = np.fft.irfft(A * np.conj(W), L)
    grad = 2.0 * corr[:n]
    
    return obj, grad

    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal temperature
        T = T_start * (T_end / T_start) ** progress
        # Anneal learning rate
        lr = lr_start * (1.0 - 0.5 * progress)
        
        obj, grad = smooth_obj_grad(a, T)
        
        # Momentum update
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        # Evaluate true objective
        conv_new = fast_convolve(a_new)
        true_obj = conv_new.max()
        
        if true_obj < best_obj - 1e-15:
            best_obj = true_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Restart if stuck
        if no_improve > 200:
            v = np.zeros(n)
            no_improve = 0
            # Small perturbation
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.001)
        
        if verbose and it % 500 == 0:
            print(f"    SGD iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Try a random coordinate
        m = rng.integers(0, n)
        
        # Optimize a[m] given all other coordinates fixed
        # conv[k] = sum_j a[j]*a[k-j]
        # dconv[k]/da[m] = 2*a[k-m] (for valid k-m)
        # We want to minimize max_k conv[k] subject to a >= 0, sum = 1
        # This is equivalent to: find a[m] that minimizes max_k conv[k]
        # conv[k] is linear in a[m] (for fixed other coordinates)
        # conv[k] = base[k] + a[m] * (2 * a[k-m]) where base[k] = conv[k] - a[m]*2*a[k-m]
        
        # Actually, let's just do a 1D search
        a_test = a.copy()
        # Try a range of values for a[m]
        # Current value: a[m]
        # We can increase a[m] and decrease others proportionally, or just vary a[m] and renormalize
        
        # Simpler: try several multipliers for a[m]
        best_val = best_obj
        best_am = a[m]
        
        for mult in [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5, 0.3, 2.0]:
            a_test[m] = a[m] * mult
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test = a_test / st
            val = fast_convolve(a_test).max()
            if val < best_val:
                best_val = val
                best_am = a[m] * mult
        
        if best_val < best_obj - 1e-14:
            a[m] = best_am
            a = a / a.sum()
            best_obj = best_val
            best_a = a.copy()
    
    return best_a, best_obj


def local_search(f0, time_limit, rng, num_perturbations=200):
    """Try random perturbations and keep the best."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    scale = 0.003

    while time.monotonic() - t0 < time_limit:
        for _ in range(num_perturbations):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()
        scale *= 0.97
        if scale < 1e-6:
            scale = 0.002

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # U-shape
    edge_w = 1.0 / (0.5 * np.abs(x - 0.5 * (n - 1)) / n + 0.05)
    inits.append(edge_w / edge_w.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    
    # Two peaks
    peak1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    two = peak1 + peak2
    inits.append(two / two.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smoothed gradient descent with log-sum-exp approximation of max
2. Mirror descent (multiplicative weight updates) as alternative
3. Spectral flatness optimization (Wiener-Khinchin insight)
4. Temperature annealing with adaptive learning rate
5. Multiple diverse initializations and lengths
6. Local search polishing
The key innovation is using the smooth gradient (via cross-correlation FFT)
for efficient optimization, combined with spectral domain insights.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, T):
    """Compute smoothed objective (log-sum-exp) and its gradient.
    
    Objective: T * log(sum_k exp(conv(a,a)[k] / T))
    Gradient uses cross-correlation via FFT for O(n log n) computation.
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    log_sum = max_c / T + np.log(np.sum(np.exp(shifted)))
    obj = T * log_sum

    w = np.exp(shifted)
    w /= w.sum()

    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * cross_correlation(a, w)[m]
    L = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    corr = np.fft.irfft(A * np.conj(W), L)
    grad = 2.0 * corr[:n]

    return obj, grad


def spectral_flatness_grad(a, T):
    """Optimize spectral flatness: minimize max of circular autocorrelation.
    
    Uses the Wiener-Khinchin theorem: circular autocorrelation = IFFT(|FFT(a)|^2).
    The peak of circular autocorrelation approximates peak of linear autocorrelation.
    """
    n = len(a)
    # Circular autocorrelation
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = np.abs(A[:n]) ** 2  # power spectrum
    circ_ac = np.fft.irfft(A * np.conj(A), fft_size)[:n]  # circular autocorrelation
    
    max_c = circ_ac.max()
    shifted = (circ_ac - max_c) / T
    obj = T * (max_c / T + np.log(np.sum(np.exp(shifted))))
    
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient of circular autocorrelation at lag k w.r.t. a[m]:
    # d(circ_ac[k])/da[m] = a[(k-m) mod n] + a[(m-k) mod n] = 2*a[(k-m) mod n]
    # So grad[m] = sum_k w[k] * 2*a[(k-m) % n]
    # This is 2 * circular cross-correlation
    W = np.fft.rfft(w, fft_size)
    grad_full = np.fft.irfft(A * np.conj(W), fft_size)[:n]
    grad = 2.0 * grad_full

    return obj, grad


def smooth_gd_optimize(a0, time_limit, T_start=0.01, T_end=0.0001,
                       lr_start=0.005, momentum=0.9, use_mirror=False,
                       use_spectral=False, verbose=False, seed=0):
    """Smoothed gradient descent with temperature annealing and momentum.
    
    Supports two update rules:
    - Projected GD: a_new = project_simplex(a - lr * v)
    - Mirror descent: a_new = a * exp(-lr * grad) / sum(a * exp(-lr * grad))
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_lr = lr_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Cosine annealing for temperature
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        # Linear decay for learning rate
        lr = lr_start * (1.0 - 0.7 * progress)

        if use_spectral:
            obj, grad = spectral_flatness_grad(a, T)
        else:
            obj, grad = smooth_obj_grad(a, T)

        if use_mirror:
            # Mirror descent (multiplicative updates)
            scaled = -lr * grad
            scaled -= scaled.max()  # numerical stability
            updates = np.exp(scaled)
            a_new = a * updates
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-30:
                a_new = a.copy()
            else:
                a_new = a_new / a_new_sum
        else:
            # Projected GD with momentum
            v = momentum * v + grad
            a_new = a - lr * v
            a_new = project_simplex(a_new)

        # Evaluate true objective
        conv_new = fast_convolve(a_new)
        true_obj = conv_new.max()

        if true_obj < best_obj - 1e-15:
            best_obj = true_obj
            best_a = a_new.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        a = a_new

        # Reheating: if stuck, perturb and reduce momentum
        if no_improve > 150:
            v = np.zeros(n)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            lr = best_lr * 0.8

        if verbose and it % 1000 == 0:
            print(f"    iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def multi_start_optimize(a0, time_limit, n_starts=5, verbose=False):
    """Run smooth GD from multiple perturbed starting points."""
    rng = np.random.default_rng(99)
    n = len(a0)
    t0 = time.monotonic()
    
    best_a = a0.copy()
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    best_obj = fast_convolve(best_a).max()

    t_per = (time_limit - 5) / n_starts

    for i in range(n_starts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Create perturbed start
        if i == 0:
            a_start = a0.copy()
        else:
            scale = 0.005 * (1 + i * 0.5)
            a_start = project_simplex(a0 + rng.standard_normal(n) * scale)
        
        # Vary parameters per start
        T_s = 0.01 * (1.0 + 0.3 * i)
        T_e = 0.0001 * (0.5 + 0.1 * i)
        lr_s = 0.004 * (1.0 + 0.2 * i)
        mom = 0.9 + 0.01 * (i % 3)
        use_mirror = (i % 3 == 0)
        rem = time_limit - (time.monotonic() - t0) - 3
        t_alloc = min(t_per, rem)
        if t_alloc < 10:
            continue
            
        a_opt, obj = smooth_gd_optimize(a_start, time_limit=t_alloc,
                                         T_start=T_s, T_end=T_e,
                                         lr_start=lr_s, momentum=mom,
                                         use_mirror=use_mirror,
                                         verbose=verbose, seed=100 + i)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a, best_obj


def local_search(f0, time_limit, rng, num_perturbations=300):
    """Random perturbation local search with adaptive scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    scale = 0.003

    while time.monotonic() - t0 < time_limit:
        batch_end = min(time.monotonic() + 5.0, t0 + time_limit)
        while time.monotonic() < batch_end:
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()
        scale *= 0.95
        if scale < 1e-7:
            scale = 0.001

    return best_f, best_obj


def coordinate_refine(a0, time_limit, rng):
    """Refine by trying to adjust pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Pick two random coordinates to swap mass between
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Try transferring mass from one to the other
        for delta in [0.001, 0.005, 0.01, 0.02]:
            a_test = best_a.copy()
            a_test[i] -= delta
            a_test[j] += delta
            if a_test[i] < 0 or a_test[j] < 0:
                continue
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test = a_test / st
            obj = fast_convolve(a_test).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_test.copy()

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.1, 0.2, 0.35]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # U-shape (edges emphasized)
    edge_w = 1.0 / (0.3 * np.abs(x - 0.5 * (n - 1)) / n + 0.03)
    inits.append(edge_w / edge_w.sum())

    # Cosine bump
    c1 = np.cos(np.pi * (x / n - 0.5)) ** 2 + 0.1
    inits.append(c1 / c1.sum())

    # Two-peak
    p1 = np.exp(-((x - 0.2 * n) / (0.08 * n)) ** 2)
    p2 = np.exp(-((x - 0.8 * n) / (0.08 * n)) ** 2)
    two = p1 + p2
    inits.append(two / two.sum())

    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    # Simple low-pass via repeated averaging
    for _ in range(3):
        noise = 0.5 * (noise + np.roll(noise, 1))
        noise = 0.5 * (noise + np.roll(noise, -1))
    noise = np.abs(noise)
    inits.append(noise / noise.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Diverse initialization across lengths
    Phase 2: Multi-start smoothed gradient descent (primary)
    Phase 3: Refined optimization with lower temperature
    Phase 4: Spectral domain optimization
    Phase 5: Local search and coordinate refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent))
        n_parent = len(parent)
    else:
        n_parent = 384
        parent = np.ones(n_parent) / n_parent
        candidates.append(("uniform", parent))

    # Try several lengths
    lengths = [n_parent]
    for n_try in [256, 320, 384, 448, 512, 640, 768, 1024]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(l for l in lengths if 128 <= l <= 1024))

    for n_try in lengths:
        if time.monotonic() > deadline - 400:
            break
        inits = generate_inits(n_try, parent, rng)
        for i, ini in enumerate(inits[:5]):
            candidates.append((f"n{n_try}_{i}", ini))

    for label, cand in candidates:
        if time.monotonic() > deadline - 350:
            break
        try_seq(cand, label)

    if best_seq is None:
        best_seq = parent.copy()
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.8f} (n={n_best})")

    # === Phase 2: Multi-start Smoothed GD (main optimizer) ===
    print("\n=== Phase 2: Multi-start Smoothed GD ===")
    remaining = deadline - time.monotonic() - 150
    t_p2 = min(remaining * 0.45, 700)

    if t_p2 > 50:
        print(f"  Running multi-start SGD (n={n_best}), budget={t_p2:.0f}s")
        a_opt, obj = multi_start_optimize(best_seq, time_limit=t_p2, n_starts=5, verbose=True)
        try_seq(a_opt, "multi_sgd")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Refined single-run optimization ===
    print("\n=== Phase 3: Refined Optimization ===")
    remaining = deadline - time.monotonic() - 100
    t_p3 = min(remaining * 0.3, 400)

    if t_p3 > 30:
        n_best = len(best_seq)
        # Run 1: Low temperature, high momentum
        t1 = min(t_p3 * 0.4, 200)
        a_opt, obj = smooth_gd_optimize(best_seq, time_limit=t1,
                                         T_start=0.003, T_end=0.0002,
                                         lr_start=0.002, momentum=0.93, seed=200)
        try_seq(a_opt, "refine1")

        # Run 2: Mirror descent
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t2 = min(rem * 0.4, 150)
            a_opt, obj = smooth_gd_optimize(best_seq, time_limit=t2,
                                             T_start=0.005, T_end=0.0003,
                                             lr_start=0.003, momentum=0.9,
                                             use_mirror=True, seed=300)
            try_seq(a_opt, "mirror")

        # Run 3: Very fine
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t3 = min(rem * 0.5, 100)
            a_opt, obj = smooth_gd_optimize(best_seq, time_limit=t3,
                                             T_start=0.001, T_end=0.00005,
                                             lr_start=0.001, momentum=0.95, seed=400)
            try_seq(a_opt, "refine2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Spectral domain optimization ===
    print("\n=== Phase 4: Spectral Optimization ===")
    remaining = deadline - time.monotonic() - 80
    if remaining > 30:
        n_best = len(best_seq)
        t_sp = min(remaining * 0.4, 200)
        a_opt, obj = smooth_gd_optimize(best_seq, time_limit=t_sp,
                                         T_start=0.005, T_end=0.0003,
                                         lr_start=0.003, momentum=0.9,
                                         use_spectral=True, seed=500)
        try_seq(a_opt, "spectral")

        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, obj = smooth_gd_optimize(best_seq, time_limit=min(rem * 0.5, 100),
                                             T_start=0.002, T_end=0.0001,
                                             lr_start=0.001, momentum=0.94,
                                             use_spectral=True, seed=600)
            try_seq(a_opt, "spectral_fine")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Local search and coordinate refinement ===
    print("\n=== Phase 5: Local Search & Polish ===")
    remaining = deadline - time.monotonic() - 40
    if remaining > 15:
        n_best = len(best_seq)

        # Local search
        t_ls = min(remaining * 0.45, 100)
        a_opt, _ = local_search(best_seq, time_limit=t_ls, rng=rng)
        try_seq(a_opt, "local")

        # Coordinate refinement
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cr = min(rem * 0.4, 60)
            a_opt, _ = coordinate_refine(best_seq, time_limit=t_cr, rng=rng)
            try_seq(a_opt, "coord")

        # Final SGD polish
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = smooth_gd_optimize(best_seq, time_limit=min(rem * 0.5, 40),
                                             T_start=0.0005, T_end=0.00005,
                                             lr_start=0.0005, momentum=0.97, seed=700)
            try_seq(a_opt, "final_sgd")

    # === Phase 6: Try alternative lengths ===
    rem = deadline - time.monotonic() - 20
    if rem > 30 and best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [l for l in [n_best - 32, n_best + 32, n_best + 64, 512, 448]
                       if l >= 128 and l != n_best]
        for n_try in alt_lengths[:3]:
            if rem < 15:
                break
            a_try = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            a_try = a_try / a_try.sum()
            t_alloc = min(rem * 0.4, 45)
            a_opt, obj = smooth_gd_optimize(a_try, time_limit=t_alloc,
                                             T_start=0.005, T_end=0.0003,
                                             lr_start=0.003, momentum=0.9, seed=800)
            try_seq(a_opt, f"len{n_try}")
            rem = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
```
