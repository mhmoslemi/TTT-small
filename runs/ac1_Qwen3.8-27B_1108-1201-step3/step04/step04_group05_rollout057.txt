import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gs_optimize(a0, time_limit, r_start=0.3, r_end=0.95, n_warmup=100):
    """
    Gerchberg-Saxton style optimization.
    Damps non-DC spectral components to flatten the auto-convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    L = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive damping ratio
        frac = min(1.0, it / max(1, n_warmup))
        r = r_start + (r_end - r_start) * frac
        
        # FFT
        A = np.fft.fft(a, L)
        
        # Damp non-DC components
        damp = np.ones(L)
        damp[1:] = r
        # Also damp the "other side" more aggressively for symmetry
        # Actually, let's use a frequency-dependent damping
        # damp[k] = r^min(k, L-k) for a smoother roll-off
        # For speed, just use uniform r
        
        A_damped = A * np.sqrt(damp)
        
        # IFFT
        a_pad = np.fft.ifft(A_damped).real
        a_new = np.maximum(a_pad[:n], 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def gs_with_peak_shaving(a0, time_limit, r=0.7, top_k=20):
    """
    GS combined with direct peak shaving in the time domain.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    L = 1 << (2 * n - 1).bit_length()
    freq_idx = np.arange(L)
    freq_dist = np.minimum(freq_idx, L - freq_idx).astype(np.float64)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    conv = fast_convolve(a)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # GS step
        damp = np.power(r, freq_dist)
        damp[0] = 1.0
        A = np.fft.fft(a, L)
        A_damped = A * np.sqrt(damp)
        a_pad = np.fft.ifft(A_damped).real
        a_new = np.maximum(a_pad[:n], 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Peak shaving step
        conv = fast_convolve(a_new)
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        # Build subgradient
        sg = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += a_new[k - idx]
        
        # Apply with small step
        a_shaved = a_new - 0.01 * sg
        a_shaved = np.maximum(a_shaved, 0)
        ss = a_shaved.sum()
        if ss > 1e-15:
            a_shaved = a_shaved / ss
            obj = fast_convolve(a_shaved).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_shaved.copy()
            a = a_shaved
        else:
            a = a_new
        
        # Update best
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def adaptive_subgrad(f0, time_limit, lr_start=0.005, lr_end=0.0001, 
                     mom_start=0.9, mom_end=0.98, top_k_start=10, top_k_end=50,
                     restart_interval=200, temp=0.0001):
    """
    Adaptive subgradient descent with:
    - Learning rate decay
    - Momentum increase
    - Expanding top-k peak set
    - Periodic restarts
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / (time_limit * 100))  # rough progress estimate
        
        # Adaptive parameters
        lr = lr_start * (lr_end / lr_start) ** frac
        mom = mom_start + (mom_end - mom_start) * frac
        top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Softmax weights
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Momentum update
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
        
        # Periodic restart
        if it % restart_interval == 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
    
    return best_f, best_obj


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for all k, g >= -f, sum(g)=0."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None

        # Line search

                    hi = mid

        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.001 + no_improve * 0.00005))
            no_improve = 0

def coordinate_descent(f0, time_limit, n_sweeps_max=100):

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = f[i]
            if old_val < 1e-15:
                continue
            
            # Determine the range of a[i] to search
            # The contribution of a[i] to c[k] for k in [i, i+n-1] is:
            #   delta_c[k] = 2*(new - old)*a[k-i] (for k != 2i)
            #   delta_c[2i] = new^2 - old^2
            # We want to minimize max(c + delta_c)
            
            # Upper bound: if we set a[i] = 0, the change is:
            #   delta_c[k] = -2*old*a[k-i]
            #   delta_c[2i] = -old^2
            # This can't increase any c[k], so a[i]=0 is always at least as good
            # as a[i]=old in terms of max(c)... wait, that's not right.
            # Setting a[i]=0 reduces c[k] for k where a[k-i] > 0.
            # But it also changes the normalization.
            
            # Actually, since we're working with normalized f (sum=1),
            # changing f[i] requires renormalizing. This complicates things.
            
            # Simpler approach: just try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5, old_val * 2.0]
            best_candidate = old_val
            best_candidate_obj = None
            
            for cand in candidates:
                if cand < 0:
                    continue
                f[i] = cand
                s_new = f.sum()
                if s_new < 1e-15:
                    f[i] = old_val
                    continue
                f_norm = f / s_new
                obj = fast_convolve(f_norm).max()
                f[i] = old_val  # restore
                
                if best_candidate_obj is None or obj < best_candidate_obj:
                    best_candidate_obj = obj
                    best_candidate = cand
            
            if best_candidate != old_val:
                f[i] = best_candidate
                s_new = f.sum()
                if s_new > 1e-15:
                    f = f / s_new
                    conv = fast_convolve(f)
                    obj = conv.max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def coordinate_descent_fast(f0, time_limit):
    """
    Faster coordinate descent using incremental convolution updates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old = f[i]
            if old < 1e-12:
                continue
            
            # Try reducing f[i] (this reduces peaks it contributes to)
            # New contribution to c[k] for k in [i, i+n-1]:
            #   For new value v: delta = (v - old) * a[k-i] * 2 (for k != 2i)
            #   For k = 2i: delta = v^2 - old^2
            
            # The peaks that a[i] contributes to most are where a[k-i] is large
            # Try v = 0 (remove) and v = old * 0.1 (shrink)
            
            # Try v = 0
            v = 0.0
            # Update conv
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            
            # Save old conv values
            old_conv = conv[k_min:k_max+1].copy()
            
            for k in range(k_min, k_max + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        conv[k] = old_conv[k - k_min] - old ** 2
                    else:
                        conv[k] = old_conv[k - k_min] - 2 * old * f[ki]
            
            # Renormalize
            s_new = f.sum() - old  # since v=0
            if s_new > 1e-15:
                scale = s / s_new
                conv_scaled = conv * scale * scale
                obj = conv_scaled.max()
                
                if obj < best_obj - 1e-14:
                    f[i] = 0.0
                    f = f / s_new
                    conv = fast_convolve(f)  # recompute to avoid errors
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True
                else:
                    conv = old_conv  # restore... wait this is wrong
                    # Need to restore all affected entries
                    conv[k_min:k_max+1] = old_conv
            else:
                conv[k_min:k_max+1] = old_conv
        
        if n_passes > 20 and not improved:
            break
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    # Various shapes
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Linear ramp (non-symmetric!)
    ramp_up = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_up / ramp_up.sum())
    
    ramp_down = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_down / ramp_down.sum())
    
    # Exponential
    exp_up = np.exp(np.linspace(0, 3, n))
    inits.append(exp_up / exp_up.sum())
    exp_down = np.exp(np.linspace(3, 0, n))
    inits.append(exp_down / exp_down.sum())
    
    # Sine
    sin_seq = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits.append(sin_seq / sin_seq.sum())
    
    # Chirp-like
    chirp = np.abs(np.sin(2 * np.pi * x * x / (4 * n))) + 0.1
    inits.append(chirp / chirp.sum())
    
    # Two humps (non-symmetric)
    h1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
