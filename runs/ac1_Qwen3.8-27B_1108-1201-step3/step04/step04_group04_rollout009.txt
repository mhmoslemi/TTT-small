```python
    n = len(v)
    # Project onto {x >= 0, sum(x) = 1}
    # Find t such that sum(max(v_i - t, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1] + 1  # 1-indexed
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)
```

1. At point a, compute gradient g = ∇f(a) (or subgradient).
2. Solve the linear minimize problem: min_{s in S} <g, s>.
3. Move: a_new = (1-t) * a + t * s, where t is chosen by line search.

In our case, S is the probability simplex {a >= 0, sum(a) = 1}. The linear step min_{s in S} <g, s> is simply: s = e_min where min is the index of the minimum component of g. (Because the minimum of a linear function over the simplex is achieved at a vertex.)

Wait, that's not quite right. The minimum of <g, s> = sum_i g[i]*s[i] over {s >= 0, sum(s) = 1} is achieved by putting all mass on the index with the smallest g[i]. So s = e_{argmin g}.

This is very fast! But it might be too aggressive (always moving toward a vertex). We can use a step size t < 1 to soften this.

Actually, for non-convex problems, Frank-Wolfe doesn't have the same guarantees, but it can still be useful as a heuristic.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient descent with adaptive active set
2. LP-based direction finding with backtracking step size
3. Multi-length search with parent interpolation
4. Peak-specific coordinate polishing
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1] + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def compute_active_subgradient(a, conv, n, eps=1e-4):
    """
    Compute subgradient of max_k conv[k] using active set {k : conv[k] > M*(1-eps)}.
    Returns gradient vector of length n.
    """
    M = conv.max()
    active = np.where(conv > M * (1.0 - eps))[0]
    
    grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += 2.0 * a[k - idxs]
    
    grad /= len(active)
    return grad


def compute_exact_subgradient(a, conv, n):
    """Compute subgradient using all k where conv[k] == max (within tolerance)."""
    M = conv.max()
    tol = 1e-10
    active = np.where(np.abs(conv - M) < tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += 2.0 * a[k - idxs]
    
    grad /= len(active)
    return grad


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. C@g <= M, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, 
                         eps_active=1e-3, lp_every=50):
    """
    Subgradient descent with momentum and periodic LP steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    mom = momentum
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        grad = compute_active_subgradient(a, conv, n, eps=eps_active)
        
        # Nesterov momentum update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a + 0.5 * v  # half step
            a = project_simplex(a)
            no_improve += 1
        
        it += 1
        
        # Adaptive
        lr *= 0.999
        mom = min(0.97, mom + 0.001)
        
        # Periodic LP step
        if it % lp_every == 0 and time.monotonic() - t0 < time_limit - 5:
            M = conv.max() * 1.0001
            g, success = lp_direction(a, M)
            if success:
                # Try step toward g
                for t in [0.3, 0.2, 0.1, 0.05]:
                    a_try = (1 - t) * a + t * g
                    a_try = project_simplex(a_try)
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                        break
        
        if no_improve > 300:
            # Perturb and restart
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 100, 5)
            a = best_a + rng.standard_normal(n) * scale
            a = project_simplex(a)
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0
        
        if no_improve > 1000:
            break
    
    return best_a, best_obj


def peak_shave_polish(a0, time_limit, n_peaks=5):
    """
    Peak-specific coordinate descent: identify top peaks and try to reduce them
    by adjusting the elements that contribute most.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Identify top peaks
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            # Find elements contributing most to this peak
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            contributions = a[idxs] * a[k - idxs]
            
            # Try reducing the largest contributor
            max_contrib_idx = np.argmax(contributions)
            i = idxs[max_contrib_idx]
            j = k - i
            
            if i == j or i >= n or j >= n:
                continue
            
            # Try moving mass from the larger of a[i], a[j] to a random other position
            if a[i] > a[j]:
                src = i
            else:
                src = j
            
            if a[src] < 1e-6:
                continue
            
            # Try various targets and amounts
            for _ in range(20):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = a[src] * frac
                    if delta < 1e-8:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    a_try = project_simplex(a_try)
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-10:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        conv = conv_try
                        no_improve = 0
                        break
                else:
                    no_improve += 1
                if no_improve == 0:
                    break
            if no_improve == 0:
                break
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine'] = s1 / s1.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits
```

For `project_simplex`, the algorithm is:
- To project v onto {x >= 0, sum(x) = 1}:
  - We want to find x that minimizes ||x - v||^2 subject to x >= 0, sum(x) = 1.
  - By KKT: x_i = max(v_i - theta, 0) for some theta.
  - sum_i max(v_i - theta, 0) = 1.
  - Sort v in decreasing order: u_1 >= u_2 >= ... >= u_n.
  - Find the largest rho such that u_rho > (sum_{i=1}^{rho} u_i - 1) / rho.
  - Then theta = (sum_{i=1}^{rho} u_i - 1) / rho.
  - x_i = max(v_i - theta, 0).

Let me verify my implementation:
```python

# This is equivalent to u[i+1] > (sum_{j=1}^{i+1} u_j - 1) / (i+1)
# So rho_candidates[-1] + 1 = rho (the largest valid rho)
```

Now let me think about potential issues:

**Alternative idea: Use the fact that the problem is related to minimizing the L-infinity norm of the autocorrelation.**

In signal processing, sequences with low peak autocorrelation are well-studied. Some known good sequences:
- Barker sequences (short, binary)
- Zadoff-Chu sequences (complex, but their magnitude is 1)
- m-sequences (binary, good autocorrelation properties)
- Computer-optimized sequences (like the ones in the "Golay complementary" family)

For real, non-negative sequences, the problem is different from the complex case. But we can try to use known structures as initialization.

```python
import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1} via water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1] + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def compute_active_subgradient(a, conv, n, eps=1e-3):
    """
    Compute subgradient of max_k conv[k] using active set {k : conv[k] > M*(1-eps)}.
    Returns gradient vector of length n.
    """
    M = conv.max()
    if M <= 0:
        return np.zeros(n)
    threshold = M * (1.0 - eps)
    active = np.where(conv > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += 2.0 * a[k - idxs]

    grad /= len(active)
    return grad


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. C@g <= M, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    if n > 2000:
        return None, False
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def spectral_init(n, rng, num_iter=50):
    """
    Generate a sequence with approximately flat power spectrum using
    alternating projections (Gerchberg-Saxton style).
    Returns a non-negative sequence summing to 1.
    """
    # Start with random phase
    phases = rng.uniform(0, 2 * np.pi, n)
    a = np.ones(n) / n

    for _ in range(num_iter):
        # Compute DFT
        A = np.fft.fft(a)
        # Set target magnitudes: flat except DC
        mag = np.ones(n)
        mag[0] = 1.0  # DC = sum(a) = 1
        phases = np.angle(A)
        # Reconstruct
        A_new = mag * np.exp(1j * phases)
        a_new = np.real(np.fft.ifft(A_new))
        # Project onto simplex
        a = project_simplex(a_new)

    return a


def subgradient_optimize(a0, time_limit, lr_init=0.008, momentum=0.92,
                         eps_active=5e-4, lp_every=40, max_its=10000):
    """
    Subgradient descent with Nesterov momentum, adaptive active set,
    and periodic LP direction steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    mom = momentum
    no_improve = 0
    it = 0
    eps = eps_active

    while time.monotonic() - t0 < time_limit and it < max_its:
        # Adaptive active set: tighten over time
        effective_eps = eps * (0.995 ** it)
        effective_eps = max(effective_eps, 1e-6)

        # Compute subgradient
        grad = compute_active_subgradient(a, conv, n, eps=effective_eps)

        # Nesterov momentum
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            a = project_simplex(a + 0.5 * v)

        it += 1

        # Adaptive learning rate
        lr *= 0.9992
        mom = min(0.97, mom + 0.0005)

        # Periodic LP step
        if it % lp_every == 0 and n <= 1200 and time.monotonic() - t0 < time_limit - 5:
            M = conv.max() * 1.0001
            g, success = lp_direction(a, M)
            if success:
                for t in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05]:
                    a_try = project_simplex((1 - t) * a + t * g)
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-13:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                        break

        # Stagnation handling
        if no_improve > 200:
            rng_local = np.random.default_rng(it * 31 + 7 + no_improve)
            scale = 0.001 + 0.003 * min(no_improve // 100, 5)
            a = project_simplex(best_a + rng_local.standard_normal(n) * scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0
            if it > 3000:
                break

    return best_a, best_obj


def lp_optimize_deep(a0, time_limit, verbose=False):
    """
    Deep LP-based optimization with adaptive tightness and restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    tight = 1.0001

    while time.monotonic() - t0 < time_limit:
        M = conv.max() * tight
        g, success = lp_direction(a, M)

        if success:
            improved = False
            for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
                a_try = project_simplex((1 - t) * a + t * g)
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-13:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
        else:
            no_improve += 1

        it += 1

        if verbose and it % 5 == 0:
            print(f"    LP iter {it}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        # Adapt tightness: make constraints tighter over time
        if no_improve % 5 == 0:
            tight = max(1.00001, tight * 0.99)

        if no_improve >= 25:
            # Perturb and restart
            rng_local = np.random.default_rng(it * 13 + no_improve * 7)
            scale = 0.001 + 0.002 * min(no_improve // 5, 5)
            a = project_simplex(best_a + rng_local.standard_normal(n) * scale)
            conv = fast_convolve(a)
            no_improve = 0
            tight = 1.0001

            if it > 400:
                break

    return best_a, best_obj


def peak_shave_polish(a0, time_limit, n_peaks=3):
    """
    Peak-specific coordinate descent: identify top peaks and try to reduce them
    by adjusting the elements that contribute most to each peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    total_tries = 0

    while time.monotonic() - t0 < time_limit:
        # Identify top peaks
        if n_peaks >= len(conv):
            top_peaks = np.argsort(conv)[::-1]
        else:
            top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Find elements contributing to this peak
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            if len(idxs) == 0:
                continue

            contributions = a[idxs] * a[k - idxs]
            
            # Try adjusting top contributors
            order = np.argsort(contributions)[::-1]
            for oi in range(min(3, len(order))):
                i = idxs[order[oi]]
                j = k - i
                if i == j or i >= n or j >= n:
                    continue
                src = i if a[i] >= a[j] else j
                if a[src] < 1e-7:
                    continue

                # Try moving mass from src to random dst
                for _ in range(10):
                    dst = rng.integers(0, n)
                    if dst == src:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        delta = a[src] * frac
                        if delta < 1e-9:
                            continue
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[dst] += delta
                        a_try = project_simplex(a_try)
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        total_tries += 1
                        if obj_try < best_obj - 1e-10:
                            a = a_try
                            best_obj = obj_try
                            best_a = a.copy()
                            conv = conv_try
                            no_improve = 0
                            break
                    else:
                        no_improve += 1
                    if no_improve == 0:
                        break
                if no_improve == 0:
                    break

        if no_improve > 1000 or total_tries > 50000:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine (peaks at edges)
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine'] = s1 / s1.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Spectral (flat spectrum approximation)
    try:
        inits['spectral'] = spectral_init(n, rng, num_iter=30)
    except Exception:
        pass

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

        t_burst = min(10.0, deadline - time.monotonic() - 2)
        if t_burst > 2:
            a_opt, obj = subgradient_optimize(best_sequence, time_limit=t_burst,
                                               lr_init=0.001, momentum=0.97,
                                               eps_active=1e-5, lp_every=100)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final Burst: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
