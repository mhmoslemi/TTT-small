"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Analytical initialization (Gaussian, cosine, beta distributions)
2. Exponentiated gradient with cosine-annealed temperature, momentum, and adaptive top-k
3. Multi-length exploration
4. Fine polishing with near-greedy updates
The key improvement over the previous approach is the use of cosine annealing 
for smoother temperature decay, momentum on subgradients for more stable updates,
and a continuation strategy on top-k (from large to small) for better exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        vals = c[indices]
    else:
        indices = np.arange(len_c)
        vals = c.copy()
    
    scaled = vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-14:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # sg[j] += 2*w * a[k-j] for j in [js, je]
        sg[js:je+1] += 2.0 * w[idx] * a[k-je:k-js+1][::-1]
    
    return sg


def eg_optimize(a0, time_limit, eta_max=1.0, eta_min=0.05,
                temp_max=0.01, temp_min=0.0001,
                momentum=0.9, top_k_start=None, top_k_end=5,
                verbose=False):
    """EG with cosine annealing, momentum, and adaptive top-k."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
    n = len(a)
    len_c = 2 * n - 1
    
    if top_k_start is None:
        top_k_start = len_c
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    sg_mom = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        # Cosine annealing schedule
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        eta = eta_min + (eta_max - eta_min) * cos_val
        temp = temp_min + (temp_max - temp_min) * cos_val
        top_k = max(top_k_end, int(top_k_start + (top_k_end - top_k_start) * progress))
        
        it += 1
        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max > 1e-15:
            sg = sg / sg_max
            # Momentum
            sg_mom = momentum * sg_mom + (1 - momentum) * sg
        else:
            break
        
        # Multiplicative update
        log_update = -eta * sg_mom
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, temp={temp:.6f}, top_k={top_k}")
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        # Periodic restart on stagnation
        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 13 + 7)
            noise_scale = 0.001 * (1 + no_improve / 10000)
            a = best_a + rng_local.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            sg_mom = np.zeros(n)
    
    return best_a.copy(), best_obj


def generate_analytical_inits(n, rng):
    """Generate diverse analytical initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x / (n - 1)  # normalized to [0, 1]
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian with various widths
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((xm - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (xm - 0.5)))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())
    
    # Beta-like distributions
    for alpha, beta in [(2, 2), (3, 3), (4, 4), (2, 3), (3, 2), (1.5, 1.5), (5, 5)]:
        # Approximate beta distribution
        b = (xm + 0.01) ** (alpha - 1) * (1 - xm + 0.01) ** (beta - 1)
        inits.append(b / b.sum())
    
    # Power law decay from center
    for p in [0.5, 1.0, 1.5, 2.0]:
        d = np.abs(xm - 0.5)
        pw = (1 - d * 2) ** p
        pw = np.maximum(pw, 1e-10)
        inits.append(pw / pw.sum())
    
    # Edge-concentrated
    edge = np.minimum(xm, 1 - xm)
    for p in [1, 2, 0.5]:
        e = edge ** p
        inits.append(e / e.sum())
    
    # Bimodal
    for sep in [0.2, 0.3, 0.4]:
        bi = np.exp(-0.5 * ((xm - 0.5 + sep) / 0.08) ** 2) + np.exp(-0.5 * ((xm - 0.5 - sep) / 0.08) ** 2)
        inits.append(bi / bi.sum())
    
    # Random smooth (using spline-like interpolation)
    for _ in range(5):
        n_ctrl = rng.integers(3, 8)
        ctrl_x = np.sort(rng.uniform(0, 1, n_ctrl))
        ctrl_x = np.concatenate([[0], ctrl_x, [1]])
        ctrl_y = rng.uniform(0.1, 1.0, len(ctrl_x))
        r = np.interp(xm, ctrl_x, ctrl_y)
        inits.append(r / r.sum())
    
    return inits


def try_length(a_best, n_new, method='interp'):
    """Resize a sequence to a new length."""
    n_old = len(a_best)
    if n_new == n_old:
        return a_best.copy()
    if method == 'interp':
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, n_new)
        a_new = np.interp(x_new, x_old, a_best)
    else:
        # Padding or truncation
        a_new = np.zeros(n_new)
        m = min(n_old, n_new)
        a_new[:m] = a_best[:m]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_perturb = max(4, n // 8)
        coords = rng.choice(n, size=min(n_perturb, n), replace=False)
        improved = False
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a.copy(), best_obj

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)

    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        vals = c[indices]
    else:
        indices = np.arange(len_c)
        vals = c.copy()

    scaled = vals / max(temperature, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-14:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def eg_optimize(a0, time_limit, eta_max=1.0, eta_min=0.05,
                temp_max=0.01, temp_min=0.0001,
                momentum=0.9, top_k_start=None, top_k_end=5,
                verbose=False, restart_interval=2000):
    """EG with cosine annealing, momentum, and adaptive top-k."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    len_c = 2 * n - 1

    if top_k_start is None:
        top_k_start = len_c

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    sg_mom = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-10))
        cos_val = 0.5 * (1.0 + np.cos(np.pi * progress))
        eta = eta_min + (eta_max - eta_min) * cos_val
        temp = temp_min + (temp_max - temp_min) * cos_val
        top_k = max(top_k_end, int(round(top_k_start + (top_k_end - top_k_start) * progress)))

        it += 1
        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max > 1e-15:
            sg = sg / sg_max
            sg_mom = momentum * sg_mom + (1.0 - momentum) * sg
        else:
            break

        log_update = -eta * sg_mom
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, temp={temp:.6f}, tk={top_k}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if no_improve > restart_interval:
            rng_local = np.random.default_rng(it * 17 + 3)
            noise_scale = 0.002 * min(1.0, no_improve / (2 * restart_interval))
            a = best_a + rng_local.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            sg_mom = np.zeros(n)

    return best_a.copy(), best_obj


def fourier_basis_optimize(n, time_limit, n_modes=12):
    """Optimize a sequence parameterized by Fourier cosine modes."""
    x = np.arange(n, dtype=np.float64)
    # Precompute basis matrix: B[j, k] = cos(2*pi*k*j/n)
    modes = np.arange(n_modes + 1)
    B = np.cos(2.0 * np.pi * np.outer(x, modes) / n)  # (n, n_modes+1)

    def objective(params):
        # a[j] = max(0, B[j] @ params)^2
        raw = B @ params
        a = np.maximum(raw, 0.0) ** 2
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        c = fast_convolve(a)
        return 2.0 * n * c.max()

    # Initial guess: mostly DC + slight modulation
    params0 = np.zeros(n_modes + 1)
    params0[0] = 1.0
    params0[1] = 0.1

    best_result = None
    best_obj_val = 1e10

    # Try multiple starting points
    rng = np.random.default_rng(99)
    starts = [params0.copy()]
    for _ in range(8):
        p = params0.copy()
        for k in range(1, n_modes + 1):
            p[k] = rng.uniform(-0.3, 0.3)
        p[0] = 1.0 + rng.uniform(-0.2, 0.2)
        starts.append(p)

    t0 = time.monotonic()
    for si, p0 in enumerate(starts):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_max = rem * 0.7
        try:
            res = sp_minimize(objective, p0, method='Nelder-Mead',
                              options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10,
                                       'adaptive': True})
            val = res.fun
            if val < best_obj_val and np.isfinite(val):
                best_obj_val = val
                best_result = res.x
        except Exception:
            pass

    if best_result is None:
        return None

    raw = B @ best_result
    a = np.maximum(raw, 0.0) ** 2
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s


def generate_analytical_inits(n, rng):
    """Generate diverse analytical initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x / max(n - 1, 1)  # [0, 1]

    inits.append(np.ones(n) / n)

    # Gaussian with various widths
    for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5]:
        g = np.exp(-0.5 * ((xm - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Raised cosine (full)
    rc = 0.5 * (1 + np.cos(2 * np.pi * (xm - 0.5)))
    rc = np.maximum(rc, 1e-12)
    inits.append(rc / rc.sum())

    # Half raised cosine (truncated)
    for frac in [0.5, 0.6, 0.7, 0.8]:
        h = np.zeros(n)
        m = int(n * frac)
        center = (m - 1) / 2.0
        for j in range(m):
            h[j] = 0.5 * (1 + np.cos(np.pi * (j - center) / (m / 2.0)))
        h = np.maximum(h, 1e-12)
        inits.append(h / h.sum())

    # Beta-like
    for alpha, beta in [(2, 2), (3, 3), (4, 4), (1.5, 1.5), (2.5, 2.5), (6, 6)]:
        b = (xm + 0.01) ** (alpha - 1) * (1.0 - xm + 0.01) ** (beta - 1)
        inits.append(b / b.sum())

    # Power decay from center
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        d = 2.0 * np.abs(xm - 0.5)
        pw = (1.0 - d) ** p
        pw = np.maximum(pw, 1e-12)
        inits.append(pw / pw.sum())

    # Bimodal
    for sep in [0.15, 0.2, 0.25, 0.3]:
        for wid in [0.05, 0.08, 0.12]:
            bi = (np.exp(-0.5 * ((xm - 0.5 + sep) / wid) ** 2) +
                  np.exp(-0.5 * ((xm - 0.5 - sep) / wid) ** 2))
            inits.append(bi / bi.sum())

    # Smooth random
    for _ in range(4):
        n_ctrl = int(rng.integers(3, 8))
        ctrl_x = np.sort(rng.uniform(0, 1, n_ctrl))
        ctrl_x = np.concatenate([[0.0], ctrl_x, [1.0]])
        ctrl_y = rng.uniform(0.1, 1.0, len(ctrl_x))
        r = np.interp(xm, ctrl_x, ctrl_y)
        inits.append(r / r.sum())

    return inits


def resize_sequence(a_best, n_new):
    """Resize a sequence to a new length via interpolation."""
    n_old = len(a_best)
    if n_new == n_old:
        return a_best.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_best)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def local_search(a0, time_limit, n_factors=8):
    """Local search with multiplicative perturbations on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    factors = [0.6, 0.75, 0.85, 0.9, 1.0/0.9, 1.0/0.85, 1.0/0.75, 1.0/0.6]
    if n_factors < len(factors):
        factors = factors[:n_factors]

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_perturb = max(8, n // 6)
        coords = rng.choice(n, size=min(n_perturb, n), replace=False)
        improved = False

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break

    return best_a.copy(), best_obj

        if rem > 10:
            a_opt, _ = eg_optimize(best_seq, time_limit=min(rem * 0.6, 40),
                                   eta_max=0.3, eta_min=0.01,
                                   temp_max=0.001, temp_min=0.00001,
                                   momentum=0.95, top_k_start=n_best,
                                   top_k_end=1)
            try_seq(a_opt, "eg_final")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = local_search(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "local_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
