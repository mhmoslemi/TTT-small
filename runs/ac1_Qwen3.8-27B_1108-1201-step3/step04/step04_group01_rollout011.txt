def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - j] for j in js)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    """LSE-smoothed objective and gradient w.r.t. z (softmax parameterization)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    f = exp_z / exp_z.sum()

    conv = fast_convolve(f)
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    exp_bc = np.exp(beta_conv - max_bc)
    lse_val = max_bc + np.log(exp_bc.sum())
    w = exp_bc / exp_bc.sum()

    # Gradient w.r.t. f: grad_f[i] = 2 * sum_m w[m+i] * f[m]
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    cross = np.fft.irfft(W * F, fft_size)
    grad_f = 2.0 * cross[n - 1:n - 1 + n]

    # Convert to grad_z via softmax Jacobian: grad_z = f * (grad_f - f·grad_f)
    dot = np.dot(grad_f, f)
    grad_z = f * (grad_f - dot)

    return lse_val, grad_z

            f_safe = np.maximum(best_f, 1e-30)
            z = np.log(f_safe / f_safe.sum()) + rng.standard_normal(n) * 0.01
            beta = max(beta * 0.7, beta_start)
            stall_count = 0

        if beta > beta_end:
            # One more pass at max beta then stop
            beta = beta_end
            if time.monotonic() - t0 > time_limit * 0.8:
                break

        elapsed = time.monotonic() - t0
        if int(elapsed) % 30 < 1:
            print(f"    LSE-LBFGS: obj={best_obj:.10f}, beta={beta:.1f}, t={elapsed:.0f}s")

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.15, 4.0)
            if m_exp > 3.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 40:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Adaptive line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 5.0)

        if no_improve > 50:
            # Perturbation restart
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, n_peaks=48):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.004
    momentum = 0.93
    v = np.zeros(n)
    no_improve = 0
    temp = 0.0005

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -min(n_peaks, len(conv)))[-min(n_peaks, len(conv)):]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.94
            if no_improve > 250:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.004
                temp *= 3.0
                no_improve = 0

        temp *= 1.004

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining FW-LP, L-BFGS-B with LSE,
    and subgradient polishing for minimizing the peak auto-convolution ratio.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 90.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # Phase 1: FW-LP warm start
    print("\n=== Phase 1: FW-LP ===")
    t_alloc = min(180, max(30, deadline - time.monotonic() - 700))
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Phase 2: L-BFGS-B with LSE (primary optimizer)
    print(f"\n=== Phase 2: L-BFGS-B LSE (main) ===")

    # Phase 3: FW-LP refinement
    print(f"\n=== Phase 3: FW-LP Refinement ===")
    t_alloc = min(180, max(20, deadline - time.monotonic() - 350))

        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(current, time_limit=t_alloc)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Phase 4: Subgradient polish
    print(f"\n=== Phase 4: SG Polish ===")
    t_alloc = min(120, max(15, deadline - time.monotonic() - 180))

        n_best = len(current)
        a_opt, _ = subgradient_polish(current, time_limit=t_alloc,
                                       n_peaks=max(24, n_best // 3))

    # Phase 5: Final L-BFGS-B at high beta
    current = best_tracker[1] if best_tracker[1] is not None else parent
    print(f"\n=== Phase 5: Final L-BFGS-B (high beta) ===")
    t_alloc = min(130, max(15, deadline - time.monotonic() - 80))

        print(f"  Allocating {t_alloc:.0f}s")

        return [1.0 / 448.0] * 448
