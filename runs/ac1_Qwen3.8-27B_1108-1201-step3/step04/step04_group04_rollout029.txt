def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def lp_direction(a, M_factor=1.0):
    """
    Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0.
    Returns g or None if infeasible/no improvement.
    """
    n = len(a)
    M = fast_convolve(a).max() * M_factor
    
    # Build constraint matrix
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = result.x
            if g.sum() > a.sum() * 1.001:  # Must be better
                return g
    except Exception:
        pass
    return None

def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    x_short = np.linspace(0, 1, n_short)
    x_target = np.linspace(0, 1, n_target)
    a_ext = np.interp(x_target, x_short, a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP direction finding with exact quadratic step-size optimization
2. Subgradient descent with adaptive multi-peak weighting
3. Multi-scale search (try different n values)
4. Warm start from parent sequence

Key insight: For the linear interpolation a_t = (1-t)*a + t*g, the autocorrelation
is exactly quadratic in t, allowing optimal step-size selection in O(n) time.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.success:
            g = result.x
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_step(a, conv_aa):
    """
    One LP-based step with optimal step size.
    Returns (a_new, improved, new_conv).
    """
    n = len(a)
    M = conv_aa.max()
    
    g = lp_solve(a, M)
    if g is None:
        return a.copy(), False, conv_aa
    
    if g.sum() <= a.sum() * (1 + 1e-6):
        return a.copy(), False, conv_aa
    
    # Compute cross-convolutions
    conv_ag = fast_cross_conv(a, g)
    conv_gg = fast_convolve(g)
    
    # Quadratic model: for a_t = (1-t)*a + t*g
    # conv(a_t, a_t) = (1-t)^2 * conv_aa + 2*(1-t)*t * conv_ag + t^2 * conv_gg
    # = conv_aa + t*(2*conv_ag - 2*conv_aa) + t^2*(conv_aa - 2*conv_ag + conv_gg)
    
    A_k = conv_aa
    B_k = 2.0 * (conv_ag - conv_aa)
    C_k = conv_aa - 2.0 * conv_ag + conv_gg
    
    # Find t that minimizes max_k (A_k + B_k*t + C_k*t^2)
    # Use grid search (the function is convex in t)
    t_grid = np.linspace(0.0, 1.0, 200)
    best_t = 0.0
    best_val = M
    
    # Vectorized: compute all values at once
    # Shape: (200, 2n-1)
    vals = A_k[None, :] + B_k[None, :] * t_grid[:, None] + C_k[None, :] * t_grid[:, None] ** 2
    max_vals = vals.max(axis=1)
    
    best_idx = np.argmin(max_vals)
    best_t = t_grid[best_idx]
    best_val = max_vals[best_idx]
    
    if best_t < 0.01 or best_val >= M * (1 - 1e-8):
        return a.copy(), False, conv_aa
    
    a_new = (1 - best_t) * a + best_t * g
    # No need to clip (both a and g are non-negative)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    
    conv_new = fast_convolve(a_new)
    return a_new, True, conv_new


def subgradient_step(a, conv, top_k=30, lr=0.01):
    """
    One subgradient step on the peak of the autocorrelation.
    """
    n = len(a)
    
    # Find top-k peaks
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    peak_vals = conv[peak_indices]
    
    # Softmax weights (focus on the max)
    max_peak = peak_vals.max()
    temp = max_peak * 0.005 + 1e-10
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peak_indices))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        if w[idx] < 1e-12:
            continue
        k = int(peak_indices[idx])
        # d(conv[k])/d(a[i]) = a[k-i] (if valid) + a[i-k] (if valid)
        # First term: i in [max(0, k-n+1), min(n-1, k)]
        i1 = max(0, k - n + 1)
        i2 = min(n - 1, k)
        if i1 <= i2:
            idxs = np.arange(i1, i2 + 1)
            grad[idxs] += w[idx] * a[k - idxs]
        # Second term: i in [max(0, k), min(n-1, k+n-1)] = [k, n-1] if k < n
        if k < n:
            idxs2 = np.arange(k, n)
            grad[idxs2] += w[idx] * a[idxs2 - k]
    
    # Project onto simplex tangent (zero-mean)
    grad -= grad.mean()
    
    # Step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    else:
        a_new = a.copy()
    
    return a_new


def optimize(a0, time_limit, use_lp=True, lp_every=200):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
