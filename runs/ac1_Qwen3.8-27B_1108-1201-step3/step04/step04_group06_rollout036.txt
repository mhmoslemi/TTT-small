def fast_autoconv(a):

    conv = np.fft.irfft(PR * A, fft_size)[:conv_len]
    return conv[n-1:2*n-1][::-1]

def smoothed_grad(a, c, T):
    """Gradient of T*log(sum(exp(c/T))) wrt a."""
    n = len(a)
    scaled = c / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p /= p_sum
    r = cross_corr_fft(p, a)
    return 2.0 * r

def eg_optimize(a0, time_limit, T0=0.05, T_min=0.001, T_decay=0.9995,
                eta0=1.0, eta_decay=0.99995, symmetric=False, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if symmetric:
        half = (n + 1) // 2
        # a[j] = a[n-1-j], so we only store first half
        # For even n: a[0..n/2-1] determines a[n/2..n-1]
        # For odd n: a[0..(n-1)/2] with middle being free
        
    c = fast_autoconv(a)
    best_obj = c.max()
    best_a = a.copy()
    
    T = T0
    eta = eta0
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad = smoothed_grad(a, c, T)
        
        # For symmetric case, project gradient to symmetric subspace
        if symmetric:
            grad_sym = 0.5 * (grad + grad[::-1])
            grad = grad_sym
        
        gm = grad.max()
        if gm < 1e-15:
            break
        
        a_new = a * np.exp(-eta * grad / gm)
        a_new = np.maximum(a_new, 1e-300)
        
        if symmetric:
            a_new = 0.5 * (a_new + a_new[::-1])
        
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_autoconv(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        T = max(T * T_decay, T_min)
        eta *= eta_decay
        
        if no_improve > 3000:
            # Perturbation restart
            rng = np.random.default_rng(it)
            pert = rng.standard_normal(n) * 0.001
            a = best_a + pert
            a = np.maximum(a, 0)
            if symmetric:
                a = 0.5 * (a + a[::-1])
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            c = fast_autoconv(a)
            no_improve = 0
            eta = max(eta0 * 0.1, 0.01)
            T = max(T, T0 * 0.1)
    
    if verbose:
        print(f"  EG done: {it} iters, obj={best_obj:.10f}")
    return best_a, best_obj
