def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient_topk(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient using top-k peaks."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    
    return sg

def smooth_objective_and_grad(x, beta):
    """Log-sum-exp smoothed objective and gradient with softmax parameterization."""
    n = len(x)
    
    # Softmax
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a_sum = a.sum()
    a = a / a_sum
    
    # Convolution
    c = fast_convolve(a)
    
    # Log-sum-exp
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    log_sum_exp = c_max + np.log(exp_bc.sum())
    F = log_sum_exp / beta
    
    # Softmax weights p
    p = exp_bc / exp_bc.sum()
    
    # Gradient of F wrt a: g[i] = 2 * sum_k p[k] * a[k-i] (valid k-i)
    # = 2 * cross_corr[i] where cross_corr = np.correlate(p, a, 'full')[0:n]
    cross = np.correlate(p, a, mode='full')[:n]
    g = 2.0 * cross  # dF/da
    
    # Gradient wrt x (softmax parameterization):
    # dF/dx_j = a_j * (g_j - sum_i g_i * a_i)
    dot_ga = np.dot(g, a)
    grad_x = a * (g - dot_ga)
    
    return F, grad_x

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    
    # Initialize x from a (inverse softmax)
    x = np.log(np.maximum(a, 1e-30))
    
    t0 = time.monotonic()
    best_F = np.inf
    best_a = a.copy()
    
    # Anneal beta from beta_start to beta_end
    n_anneal_steps = 5
    for step in range(n_anneal_steps):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        frac = step / max(1, n_anneal_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        step_time = rem / (n_anneal_steps - step)
        
        result = minimize(
            lambda xx: smooth_objective_and_grad(xx, beta)[0],
            x,
            jac=lambda xx: smooth_objective_and_grad(xx, beta)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'maxfun': 5000, 'gtol': 1e-8}
        )
        
        # Convert back
        x_new = result.x
        x_shift = x_new - x_new.max()
        a_new = np.exp(x_shift)
        a_new = a_new / a_new.sum()
        
        c_new = fast_convolve(a_new)

            best_a = a_new.copy()
        
        x = x_new
        
        if verbose:
            print(f"    Smooth opt step {step}: beta={beta:.1f}, F={F_new:.10f}")
    
    return best_a

    return best_a.copy(), best_obj


def smooth_objective_and_grad(x, beta, reg_strength):
    """Log-sum-exp smoothed objective and gradient with softmax parameterization."""
    n = len(x)

    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()

    c = fast_convolve(a)

    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    exp_bc_sum = exp_bc.sum()
    F = (c_max + np.log(exp_bc_sum)) / beta

    if reg_strength > 0:
        F += 0.5 * reg_strength * np.dot(x, x)

    p = exp_bc / exp_bc_sum

    cross = np.correlate(p, a, mode='full')[:n]
    g = 2.0 * cross

    dot_ga = np.dot(g, a)
    grad_x = a * (g - dot_ga)

    if reg_strength > 0:
        grad_x += reg_strength * x

    return F, grad_x


def smooth_optimize(a0, time_limit, beta_start=30.0, beta_end=300.0,
                    reg_strength=1e-5, verbose=False):
    """Optimize using log-sum-exp smoothing with L-BFGS and annealed beta."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s

    x = np.log(np.maximum(a, 1e-30))
    x = x - x.max()  # center

    t0 = time.monotonic()
    best_F = np.inf
    best_a = a.copy()

    n_anneal_steps = 6
    for step in range(n_anneal_steps):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        frac = step / max(1, n_anneal_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        step_time = rem / (n_anneal_steps - step)

        def obj_func(xx):
            return smooth_objective_and_grad(xx, beta, reg_strength)[0]

        def grad_func(xx):
            return smooth_objective_and_grad(xx, beta, reg_strength)[1]

        max_iter = int(max(50, step_time * 3))
        try:
            result = minimize(
                obj_func, x, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'maxfun': max_iter * 10, 'gtol': 1e-10}
            )
            x = result.x
        except Exception:
            pass

        x_shift = x - x.max()
        a_new = np.exp(x_shift)
        a_new = a_new / a_new.sum()

        c_new = fast_convolve(a_new)
        F_actual = c_new.max()

        if F_actual < best_F:
            best_F = F_actual
            best_a = a_new.copy()

        x = x - x.mean()  # re-center

        if verbose:
            print(f"    Smooth step {step}: beta={beta:.1f}, max_c={F_actual:.10f}, "
                  f"score={2*n*F_actual:.8f}")

    return best_a


def coordinate_descent_polish(a0, time_limit, max_rounds=500):
    """Coordinate descent with multiplicative factors, using FFT for speed."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    no_improve = 0
    it = 0

    factors = [0.5, 0.7, 0.85, 0.92, 1.0/0.92, 1.0/0.85, 1.0/0.7, 1.0/0.5]

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False

        n_perturb = max(4, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-18:
                continue

            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    break
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    break
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a.copy(), best_obj


def generate_diverse_inits(a0, rng, n):
    """Generate diverse initializations."""
    inits = [a0.copy() if len(a0) == n else np.ones(n) / n]

    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay
    for beta in [0.5, 1.5, 4.0, 8.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / (n + 1))) + 0.1
    inits.append(c1 / c1.sum())

    # Bimodal
    for center_frac in [0.25, 0.75]:
        mu = center_frac * n
        b = np.exp(-0.5 * ((x - mu) / (0.1 * n)) ** 2)
        b2 = np.exp(-0.5 * ((x - (n - 1 - mu)) / (0.1 * n)) ** 2)
        inits.append((b + b2) / (b + b2).sum())

    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Perturbed parent
    if len(a0) == n:
        for scale in [0.001, 0.005, 0.02, 0.05]:
            p = a0 + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits


def resample_sequence(a0, n_new):
    """Resample a sequence to a new length using interpolation."""
    n_old = len(a0)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a0)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Combines exponentiated gradient with adaptive temperature,
    smooth L-BFGS optimization with annealed log-sum-exp, coordinate descent,
    and multi-initialization exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384")
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.10f}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Primary EG optimization (long run) ===
    print("\n=== Phase 1: Primary EG (adaptive temp) ===")
    t1 = min(500, deadline - time.monotonic() - 800)
    if t1 > 30:
        print(f"  EG on parent n={n}, alloc={t1:.0f}s")
        a_opt, _ = eg_optimize(parent, time_limit=t1,
                               eta0=1.5, temp0=0.002,
                               temp_decay=0.9996, eta_decay=0.99995,
                               top_k_frac=0.3, adaptive_temp=True,
                               momentum=0.3, verbose=True)
        try_seq(a_opt, "eg_p1")

    # === Phase 2: Smooth L-BFGS optimization ===
    print(f"\n=== Phase 2: Smooth L-BFGS ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(350, rem)
            print(f"  Smooth opt, alloc={t_alloc:.0f}s")
            a_smooth = smooth_optimize(best_seq, time_limit=t_alloc,
                                       beta_start=30.0, beta_end=400.0,
                                       reg_strength=1e-6, verbose=True)
            try_seq(a_smooth, "smooth1")

    # === Phase 3: EG variations with different hyperparameters ===
    print(f"\n=== Phase 3: EG Variations ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        configs = [
            {"eta0": 0.5, "temp0": 0.01, "temp_decay": 0.9999, "eta_decay": 0.9999,
             "momentum": 0.5, "top_k_frac": 0.5},
            {"eta0": 3.0, "temp0": 0.001, "temp_decay": 0.9995, "eta_decay": 0.9998,
             "momentum": 0.0, "top_k_frac": 0.15},
            {"eta0": 1.0, "temp0": 0.005, "temp_decay": 0.9997, "eta_decay": 0.99995,
             "momentum": 0.4, "top_k_frac": 0.4},
            {"eta0": 2.0, "temp0": 0.003, "temp_decay": 0.9998, "eta_decay": 0.9999,
             "momentum": 0.2, "top_k_frac": 0.25},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(100, rem / max(1, len(configs) - ci))
            if t_alloc < 10:
                break
            print(f"  EG config {ci}, alloc={t_alloc:.0f}s")
            a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"eg_cfg{ci}")

    # === Phase 4: Multi-init exploration ===
    print(f"\n=== Phase 4: Multi-Init ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        inits = generate_diverse_inits(best_seq, rng, n)
        for i, f_init in enumerate(inits[1:6]):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(40, rem / max(1, 6 - i))
            if t_alloc < 8:
                break
            a_opt, _ = eg_optimize(f_init, time_limit=t_alloc,
                                   eta0=1.0, temp0=0.01,
                                   temp_decay=0.9997, eta_decay=0.99995,
                                   top_k_frac=0.3, adaptive_temp=True,
                                   momentum=0.2)
            try_seq(a_opt, f"init{i}")

    # === Phase 5: Second smooth optimization from current best ===
    print(f"\n=== Phase 5: Smooth L-BFGS (round 2) ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(250, rem)
            print(f"  Smooth opt round 2, alloc={t_alloc:.0f}s")
            a_smooth = smooth_optimize(best_seq, time_limit=t_alloc,
                                       beta_start=50.0, beta_end=500.0,
                                       reg_strength=1e-7, verbose=True)
            try_seq(a_smooth, "smooth2")

    # === Phase 6: Coordinate descent polishing ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "coord")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different Lengths ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        alt_lengths = [n, n + 2, n - 2, n + 4, n - 4, n + 8, n - 8]
        alt_lengths = [l for l in alt_lengths if 64 <= l <= 1024]
        for li, n_new in enumerate(alt_lengths[1:5]):
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            a_resampled = resample_sequence(best_seq, n_new)
            t_alloc = min(30, rem / max(1, 5 - li))
            if t_alloc < 5:
                break
            a_opt, _ = eg_optimize(a_resampled, time_limit=t_alloc,
                                   eta0=1.0, temp0=0.005,
                                   temp_decay=0.9997, eta_decay=0.99995,
                                   top_k_frac=0.3, adaptive_temp=True,
                                   momentum=0.3)
            try_seq(a_opt, f"len_n{n_new}")

    # === Phase 8: Final fine EG + smooth ===
    print(f"\n=== Phase 8: Final Polish ===")
    print(f"  Best so far: {best_score:.10f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = eg_optimize(best_seq, time_limit=min(rem, 30),
                                   eta0=0.3, temp0=0.0005,
                                   temp_decay=0.99995, eta_decay=0.99999,
                                   top_k_frac=0.5, adaptive_temp=True,
                                   momentum=0.1)
            try_seq(a_opt, "eg_final")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_smooth = smooth_optimize(best_seq, time_limit=min(rem, 15),
                                       beta_start=100.0, beta_end=500.0,
                                       reg_strength=1e-7, verbose=False)
            try_seq(a_smooth, "smooth_final")

        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 5),
                                                 max_rounds=50)
            try_seq(a_opt, "coord_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"Time remaining: {deadline - time.monotonic():.1f}s")
    return [float(x) for x in best_seq]
