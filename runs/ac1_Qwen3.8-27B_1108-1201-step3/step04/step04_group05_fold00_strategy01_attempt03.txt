def project_simplex(v, tol=1e-8):
    """Projects onto the probability simplex with non-negativity.
    
    Parameters:
        v (numpy.ndarray): Input vector to project.
        
    Returns:
        simplex projection
    """
    n = len(v)
    u = np.sort(v)
    cssv = np.cumsum(u)
    
    if np.sum(u) < tol:
        return np.ones(n)/n
    
    # Find the threshold index
    indices = np.arange(1, n+1)
    cond = u > (cssv - 1.0) / indices
    
    if not np.any(cond):
        return np.ones(n)/n
    
    last_non_negative = np.max(indices[cond])
    theta = (cssv[last_non_negative-1] - 1.0) / float(last_non_negative)
    
    # Adjust the vector
    v_simplex = np.maximum(v - theta, 0.0)
    v_simplex_sum = np.sum(v_simplex)
    
    if v_simplex_sum < 1.0:
        # Normalize if necessary
        v_simplex = v_simplex/float(v_simplex_sum < 1e-8)
    
    return v_simplex
