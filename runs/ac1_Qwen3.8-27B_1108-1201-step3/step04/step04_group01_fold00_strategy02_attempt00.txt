def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # v in R^n
    # Project onto simplex: {x | sum(x) = 1, x >=0}
    m = len(v)
    u = np.sort(v)[::-1]  # descending order
    cssv = np.cumsum(u)
    # Find the largest tau such that tau * (len(constraint) - index) + cssv[index] > 1 + index? Not exactly, but standard approach
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.argwhere(cond)[:, 0])  # Largest index where the condition holds
    theta = (cssv[rho] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Faster convolution using FFT."""
    n = len(a)
    # Double the length to FFT
    new_length = (2 * n) - 1
    fft_size = 1 << (new_length).bit_length()  # Ensure power of two
    F = rfft(a, fft_size)
    conv = irfft(F * F.conj(), fft_size)
    return conv[:new_length]

def compute_convolution_peak(f):
    """Compute the position of max convolution without full evaluation."""
    n = len(f)
    conv = fast_convolve(f)
    return np.argmax(conv)

    """Update simplex using projected gradients."""
