For a palindromic sequence, A(x) = x^{(n-1)/2} * B(x) where B has only even/odd powers (depending on n). Then A(x)^2 = x^{n-1} * B(x)^2, and the coefficients of B(x)^2 relate to the autocorrelation.

So for symmetric sequences, the problem is equivalent to making the power spectrum flat!

This suggests: try to make the sequence symmetric and then flatten its power spectrum.

**Algorithm for symmetric sequences:**

1. Start with a symmetric sequence a (a[i] = a[n-1-i])
2. Compute the power spectrum |A(k)|^2
3. Try to make it flat: multiply by 1/sqrt(|A(k)|^2) in the frequency domain (like a Wiener filter)
4. Take IFFT, clip to non-negative, and re-symmetrize
5. Repeat

This is similar to the "wavelet shrinkage" or "iterative reweighted" approaches in signal processing.

Actually, this is very similar to the algorithm for designing flat-top windows! The flat-top window is designed to have a very flat frequency response (in the main lobe).

def spectral_flatten(a, n_iters=100):
    a = a.copy()
    n = len(a)
    # Make symmetric
    a = (a + a[::-1]) / 2
    
    for it in range(n_iters):
        # Compute power spectrum
        fft_size = 1 << (2*n).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2
        
        # Flattening: divide by sqrt(power) to make power more uniform
        # Add epsilon to avoid division by zero
        eps = 1e-10
        target = np.median(power[power > eps])
        A_flat = A * np.sqrt(target / (power + eps))

        a_new = np.fft.irfft(A_flat, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = (a_new + a_new[::-1]) / 2  # re-symmetrize
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        a = a_new
    
    return a

**Approach: Multiplicative Update / Proportional Fitting**

Inspired by iterative proportional fitting (IPF) or the multiplicative weights algorithm:

At each iteration, update a to reduce the peaks:

where grad_j is the "responsibility" of a[j] for the excess peaks.

For a given peak at position k with value c_k > t (target), the "responsibility" of a[j] is proportional to a[k-j] (the contribution of a[j] to c_k).

This is a multiplicative update that reduces elements that contribute to excess peaks.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_value(a):
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        C[j:end, j] = a[:end - j]
    return C


def compute_lse_gradient(a, c, temp, top_k=50):
    n = len(a)
    kc = len(c)
    kk = min(top_k, kc)
    peaks = np.argpartition(c, -kk)[-kk:]
    peak_vals = c[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(kk)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(kk):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad

def lp_solve(a, M, C=None):

def chirp_window_init(n, beta=0.5, eps=0.5, window='raised_cosine'):
    x = np.arange(n, dtype=np.float64)
    if window == 'raised_cosine':
        w = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2
    elif window == 'parabolic':
        t = 2.0 * x / (n - 1) - 1.0
        w = 1.0 - 0.5 * t * t
    elif window == 'quartic':
        t = 2.0 * x / (n - 1) - 1.0
        w = 1.0 - 0.5 * t ** 4
    else:
        w = np.ones(n)
    chirp = np.cos(2.0 * np.pi * beta * x * x / n)
    a = w * (1.0 + eps * chirp)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def spectral_perturb(a, rng, strength=0.1):
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = rng.uniform(0, 2 * np.pi, size=len(A))
    magnitude = rng.uniform(1 - strength, 1 + strength, size=len(A))
    A_new = A * magnitude * np.exp(1j * phase)
    A_new[0] = A[0]
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_value(a):
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def normalize_a(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        C[j:end, j] = a[:end - j]
    return C


def compute_lse_gradient(a, c, temp, top_k=50):
    n = len(a)
    kc = len(c)
    kk = min(top_k, kc)
    peaks = np.argpartition(c, -kk)[-kk:]
    peak_vals = c[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(kk)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(kk):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def gradient_optimize(a0, time_limit, lr_init=0.04, momentum=0.9, top_k=40, temp_init=0.8):
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    mom = momentum
    lr = lr_init
    temp = temp_init
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        grad = compute_lse_gradient(a, c, temp, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a_new / s
        c = fast_convolve(a)
        cur_obj = 2.0 * n * c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9998
        temp = max(0.01, temp * 0.9995)
        mom = min(0.97, mom + 0.00005)
        if no_improve > 300:
            break
    return best_a, best_obj


def lp_solve(a, M, C=None):
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step(a, conv, C=None):
    n = len(a)
    M = conv.max()
    if C is None:
        C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)
    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False
    best_a = a.copy()
    best_obj = 2.0 * n * M
    best_t = 0.0
    candidates = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t
            break
    if best_t > 0:
        return best_a, True
    return a.copy(), False


def lp_optimize(a0, time_limit, max_no_improve=60):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    C = build_conv_matrix(a)
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    lp_fail_streak = 0
    rng_local = np.random.default_rng(42 + n)
    while time.monotonic() - t0 < time_limit:
        a_new, improved = lp_step(a, conv, C)
        if improved:
            a = a_new
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a)
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
            lp_fail_streak = 0
        else:
            no_improve += 1
            lp_fail_streak += 1
        iteration += 1
        if no_improve >= max_no_improve:
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng_local.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 < 1e-15:
                a = best_a.copy()
            else:
                a = a / s2
            conv = fast_convolve(a)
            C = build_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)
            if perturb_count > 30:
                break
        if lp_fail_streak > 20:
            a, obj = gradient_optimize(best_a, time_limit=1.5, lr_init=0.01,
                                       top_k=40, temp_init=0.5)
            conv = fast_convolve(a)
            C = build_conv_matrix(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            lp_fail_streak = 0
            no_improve = 0
    return best_a, best_obj


def chirp_window_init(n, beta, eps, window_type):
    x = np.arange(n, dtype=np.float64)
    if n <= 1:
        return np.ones(n)
    if window_type == 'raised_cosine':
        w = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2
    elif window_type == 'parabolic':
        t = 2.0 * x / (n - 1) - 1.0
        w = 1.0 - 0.5 * t * t
    elif window_type == 'quartic':
        t = 2.0 * x / (n - 1) - 1.0
        w = 1.0 - 0.5 * t ** 4
    else:
        w = np.ones(n)
    chirp = np.cos(2.0 * np.pi * beta * x * x / n)
    a = w * (1.0 + eps * chirp)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spectral_perturb(a, rng, strength):
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = rng.uniform(0, 2 * np.pi, size=len(A))
    magnitude = rng.uniform(1 - strength, 1 + strength, size=len(A))
    A_new = A * magnitude * np.exp(1j * phase)
    A_new[0] = A[0]
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def structural_perturb(a, rng, strength):
    n = len(a)
    a_new = a.copy()
    n_zeros = max(1, int(n * 0.02))
    zero_idx = rng.choice(n, size=n_zeros, replace=False)
    a_new[zero_idx] = 0.0
    n_boost = max(1, int(n * 0.05))
    boost_idx = rng.choice(n, size=n_boost, replace=False)
    a_new[boost_idx] *= np.maximum(1.0 + strength * rng.standard_normal(n_boost), 0.1)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
