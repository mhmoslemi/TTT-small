def uniform_crossover(parent1, parent2, index):
    """Perform uniform crossover on parent1 and parent2 at position index."""
    n = len(parent1)
    child = np.zeros(n)

            child[i] = parent1[i]
        else:
            child[i] = parent2[i]

def rotate_cross(parent1, parent2):
    """Perform rotate crossover to center and shift distribution."""
    # 重新中心化原理序列
    ind = np.argmax(parent1)
    shifted = np.roll(parent1, -ind)
    # 处理边界
    while shifted[0] < 0:
        shifted[0] = 0
    while shifted[-1] > 0:
