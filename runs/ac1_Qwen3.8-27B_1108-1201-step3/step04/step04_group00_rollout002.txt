def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved and pass_count > 2:
            break
        if pass_count > 200:
            break

    return best_a, best_score


def subgradient_averaged(a, time_limit, lr0=0.01, decay=0.9995, top_k=30, verbose=False):
    """Averaged subgradient method (Nemirovski-Yudin style) with entropy mirror."""
    n = len(a)
    a = normalize(a.copy())
    conv = fast_convolve(a)
    best_score = compute_score(a)
    best_a = a.copy()

    # Averaging: maintain running average
    avg_a = np.zeros(n)
    avg_count = 0
    cur_lr = lr0

    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1

        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Softmax weights on peaks
        temp = max(1e-6, pv.max() - pv.min()) * 0.5
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Mirror descent step (multiplicative)
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        # Update average
        avg_a += a
        avg_count += 1
        avg_candidate = avg_a / avg_count

        conv = fast_convolve(a)
        sc = compute_score(a)
        sc_avg = compute_score(avg_candidate)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg
            best_a = avg_candidate.copy()

        cur_lr *= decay

        if verbose and step % 1000 == 0:
            print(f"    SG step {step}: score={best_score:.10f}, lr={cur_lr:.6f}")

        # Restart if stuck
        if step % 5000 == 0 and cur_lr < lr0 * 0.1:
            cur_lr = lr0 * 0.5
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            avg_a = a.copy()
            avg_count = 1

    return best_a, best_score


def pair_swap_local(a, time_limit, n_pers=200, verbose=False):
    """Random pair swap / transfer local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_pers):
            if time.monotonic() - t0 > time_limit:
                break

            i, j = rng.integers(0, n, 2)
            if i == j:
                continue

            # Transfer from j to i (or vice versa)
            # Try several transfer amounts
            a_j = a[j]
            a_i = a[i]
            if a_j < 1e-12 and a_i < 1e-12:
                continue

            best_delta = 0.0
            best_sc = best_score

            # Try transferring small fractions of a_j to a_i
            fracs = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
            for f in fracs:
                delta = a_j * f
                new_a = a.copy()
                new_a[j] = a_j - delta
                new_a[i] = a_i + delta
                if new_a[j] < 0:
                    continue
                conv = fast_convolve(new_a)
                sc = 2.0 * n * conv.max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = delta

            if best_delta != 0.0:
                a[j] -= best_delta
                a[i] += best_delta
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    """Cardinal sine shaped sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    """Gaussian-shaped sequence centered."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def triangular_sequence(n):
    """Triangular (tent) sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def optimize_sequence(a, time_limit, verbose=False):
    """Multi-stage optimization: SG -> CD -> pair_swap -> CD."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Stage 1: Subgradient (exploratory)
    t1 = min(time_limit * 0.3, 120)
    if time_limit - (time.monotonic() - t0) > t1 + 30:
        a_sg, sc_sg = subgradient_averaged(a, time_limit=t1, lr0=0.008,
                                           decay=0.9997, top_k=max(20, n // 8))
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
            a = a_sg.copy()
        if verbose:
            print(f"      After SG: {best_score:.10f}")

    # Stage 2: Precise coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.4, 180)
    if rem > t2 + 20:
        a_cd, sc_cd = coord_descent_precise(a, time_limit=t2, verbose=verbose)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
        if verbose:
            print(f"      After CD: {best_score:.10f}")

    # Stage 3: Pair swap
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.3, 80)
    if rem > t3 + 15:
        a_ps, sc_ps = pair_swap_local(a, time_limit=t3, n_pers=150)
        if sc_ps < best_score:
            best_score = sc_ps
            best_a = a_ps.copy()
            a = a_ps.copy()
        if verbose:
            print(f"      After PS: {best_score:.10f}")

    # Stage 4: Final CD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_final, sc_final = coord_descent_precise(a, time_limit=min(rem, 120))
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        if verbose:
            print(f"      After final CD: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining diverse initialization, precise
    coordinate descent, averaged subgradient, and pair-swap local search.
    Explores multiple sequence lengths and structured starting points.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []

    # Geometric sequences
    for r in [0.995, 0.997, 0.998, 0.999, 0.9995]:
        for n in [384, 512, 768, 1024, 1280]:
            starts.append((geometric_sequence(n, r), f"geom_r{r}_n{n}"))

    # Cosine
    for n in [384, 512, 768, 1024]:
        starts.append((cosine_sequence(n), f"cos_n{n}"))

    # Gaussian
    for n in [512, 768, 1024]:
        for sf in [0.2, 0.3, 0.4, 0.5]:
            starts.append((gaussian_sequence(n, sf), f"gauss_sf{sf}_n{n}"))

    # Triangular
    for n in [512, 768, 1024]:
        starts.append((triangular_sequence(n), f"tri_n{n}"))

    # From parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Random smooth
    for n in [512, 768, 1024]:
        a = rng.random(n)
        # Smooth
        k = 15
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        starts.append((normalize(a), f"rand_smooth_n{n}"))

    # Evaluate all starts (quick)
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    # Show top starts
    for i, (sc, a, label) in enumerate(start_scores[:5]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Optimize top diverse starting points ===
    print("\n=== Phase 1: Optimize diverse starts ===")

    # Pick diverse high-quality starts (not all similar)
    selected = []
    selected_labels = set()
    for sc, a, label in start_scores:
        if len(selected) >= 6:
            break
        # Skip if already selected a start of same type+length
        base = label.rsplit('_', 1)[0] if '_' in label else label
        if base not in selected_labels:
            selected.append((a, label))
            selected_labels.add(base)

    # Ensure parent is included if it's good
    parent_included = any(label == "parent" for _, label in selected)
    if not parent_included:
        for sc, a, label in start_scores:
            if label == "parent" and sc < best_score * 1.05:
                selected.append((a, label))
                break

    # Allocate time to each
    n_selected = len(selected)
    t_per = min((deadline - time.monotonic() - 300) / max(1, n_selected), 250)

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = optimize_sequence(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration on best ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        for target_n in [384, 512, 640, 768, 896, 1024, 1280, 1536]:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 40:
                break
            t_alloc = min(rem * 0.5, 120)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = optimize_sequence(a_interp, time_limit=t_alloc)
            if sc < best_score:
                try_seq(a_opt, f"dim_{target_n}")

    # === Phase 3: Aggressive local search on best ===
    print(f"\n=== Phase 3: Aggressive local search (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            # Very precise CD
            t_cd = min(rem * 0.5, 200)
            print(f"  Final CD: t={t_cd:.0f}s")
            a_final, sc = coord_descent_precise(best_a, time_limit=t_cd, verbose=True)
            try_seq(a_final, "final_cd")

            rem = deadline - time.monotonic() - 40
            if rem > 15:
                # Pair swap
                t_ps = min(rem * 0.7, 100)
                a_ps, sc = pair_swap_local(best_a, time_limit=t_ps, n_pers=300)
                try_seq(a_ps, "final_ps")

            rem = deadline - time.monotonic() - 20
            if rem > 10:
                # Tiny perturbations + re-optimize
                n_best = len(best_a)
                for trial in range(3):
                    rem = deadline - time.monotonic() - 15
                    if rem < 10:
                        break
                    noise = rng.standard_normal(n_best) * 0.002
                    a_pert = project_simplex(best_a + noise)
                    a_opt, sc = optimize_sequence(a_pert, time_limit=min(rem, 40))
                    try_seq(a_opt, f"perturb_{trial}")

    # === Final verification ===
    if best_a is None:
        best_a = cosine_sequence(768)
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
