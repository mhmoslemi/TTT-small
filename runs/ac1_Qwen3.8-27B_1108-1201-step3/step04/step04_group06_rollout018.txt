    n = len(a0)
    x = cp.Variable(n)
    # Build the matrix M_k
    # (M_k)_{j,l} = 1 if j + l = k
    M = cp.Expression  # Build as a constant
    M_data = np.zeros((n, n))
    for j in range(n):
        l = k - j
        if 0 <= l < n:
            M_data[j, l] = 1.0
    objective = cp.Minimize(cp.quad_form(x, cp.psd_wrap(M_data)))
    constraints = [cp.sum(x) == 1, x >= 0]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)  # or another solver
    if x.value is not None:
        return np.maximum(x.value, 0)

def fft_convolve(a):
    """Compute linear auto-convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_logsumexp_obj_and_grad(w, beta):
    """Compute log-sum-exp objective and gradient w.r.t. w parameters.
    where a = softmax(w)."""
    # Compute a = softmax(w)
    w_max = w.max()
    exp_w = np.exp(w - w_max)
    a = exp_w / exp_w.sum()
    n = len(a)
    
    # Compute convolution
    c = fft_convolve(a)
    
    # Compute log-sum-exp
    bc = beta * c
    bc_max = bc.max()
    log_sum_exp = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
    F = log_sum_exp / beta
    
    # Compute weights
    p = np.exp(bc - bc_max)
    p /= p.sum()
    
    # Compute gradient w.r.t. a: g_a = 2 * cross_correlate(p, a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    g_a = 2.0 * np.fft.irfft(np.conj(P) * A, fft_size)[:n]
    
    # Compute s = sum_i g_a_i * a_i = 2 * sum_k p_k * c_k
    s = 2.0 * np.dot(p, c)
    
    # Gradient w.r.t. w: g_w = a * (g_a - s)
    g_w = a * (g_a - s)
    
    return F, g_w, a, c, p

def lbfgs_optimize(w_init, beta_schedule, max_iter_per_beta=200):
    """Run L-BFGS with beta annealing."""
    w = w_init.copy()
    best_F = float('inf')
    best_w = w.copy()
    best_a = None
    
    for beta in beta_schedule:
        def fun(w_local):
            F, _, a, c, _ = compute_logsumexp_obj_and_grad(w_local, beta)
            return F
        
        def jac(w_local):
            _, g_w, a, c, _ = compute_logsumexp_obj_and_grad(w_local, beta)
            return g_w
        
        try:
            result = minimize(fun, w, jac=jac, method='L-BFGS-B',
                            options={'maxiter': max_iter_per_beta, 
                                   'ftol': 1e-15, 'gtol': 1e-10})
            w = result.x
            F_final, _, a_final, c_final, _ = compute_logsumexp_obj_and_grad(w, beta)
            if F_final < best_F:
                best_F = F_final
                best_w = w.copy()
                best_a = a_final.copy()
        except Exception as e:
            print(f"  L-BFGS error at beta={beta}: {e}")
            # Try with smaller w
            w = w / (1 + np.abs(w).max())
    
    return best_w, best_a, best_F
