def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, conv, T):
    """Compute gradient of log-sum-exp(T; conv(a,a)) w.r.t. a."""
    n = len(a)
    # Softmax weights
    shifted = conv / T
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()

    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    grad_full = np.fft.irfft(2.0 * P * A, fft_size)
    return grad_full[:n]

def solve_lp(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g or None if solution fails.
    """
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: A[k, j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        A_ub[row_start:row_end, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)
    # Softmax over convolution values
    shifted = conv / T
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()

    # Linear convolution of p (len 2n-1) with flip(a) (len n) has length 3n-2
    flip_a = a[::-1]
    size = (2 * n - 1) + (n - 1)  # 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    FA = np.fft.rfft(flip_a, fft_size)
    conv_full = np.fft.irfft(P * FA, fft_size)

    # grad_i = 2 * conv_full[n-1 + i] for i = 0..n-1
    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    return grad


def smooth_descent(a0, time_limit, T_init=None, T_min=1e-9,
                   mom=0.93, lr_init=0.01, verbose=False):
    """
    Projected gradient descent on log-sum-exp smoothed objective
    with Nesterov momentum and T-annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    if T_init is None:
        T_init = max(best_obj / 4.0, 1e-5)

    T = T_init
    lr = lr_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        grad = smooth_grad_fft(a, conv, T)

        # Nesterov momentum
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        # Backtracking if we went too far
        if obj_new > best_obj * (1.0 + 2e-4) and it > 30:
            v = 0.5 * v
            a_new = project_simplex(a + v)
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()

        if obj_new < best_obj - 5e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        # Anneal
        T = max(T * 0.9997, T_min)
        lr *= 0.9999

        # Escaping mechanism
        if no_improve > 400:
            stall_count += 1
            if stall_count > 5:
                # Larger perturbation
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                T = max(T, best_obj / 10.0)
                lr = lr_init * 0.8
                stall_count = 0
            else:
                rng = np.random.default_rng(it * 13 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                T = max(T, best_obj / 15.0)
                lr = lr_init * 0.6

            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def solve_lp_with_dual(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_y) or (None, None).
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            dual_y = None
            try:
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    dual_y = getattr(res.ineqlin, 'marginals', None)
            except Exception:
                pass
            return res.x, dual_y
    except Exception:
        pass
    return None, None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and binary search line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 5:
            break

        g, _ = solve_lp_with_dual(f, M * m_exp)
        if g is None:
            m_exp *= 1.3
            if m_exp > 12.0:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 150:
                break
            m_exp = min(m_exp * 1.01, 3.0)
            continue

        # Golden section line search
        best_alpha = 0.0
        best_val = M
        phi = (np.sqrt(5) - 1) / 2
        lo, hi = 0.0, 1.0
        for _ in range(20):
            if hi - lo < 1e-7:
                break
            a1 = hi - phi * (hi - lo)
            a2 = lo + phi * (hi - lo)
            f1 = np.maximum((1 - a1) * f + a1 * g, 0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            f2 = np.maximum((1 - a2) * f + a2 * g, 0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            if v1 < v2:
                hi = a2
            else:
                lo = a1
        best_alpha = (lo + hi) / 2.0

        # Check endpoints
        ft = np.maximum((1 - best_alpha) * f + best_alpha * g, 0)
        ft = ft / ft.sum()
        best_val = fast_convolve(ft).max()
        val_at_1 = fast_convolve(g).max()
        if val_at_1 < best_val:
            best_alpha = 1.0
            best_val = val_at_1
        val_at_0 = fast_convolve(f).max()
        if val_at_0 < best_val:
            best_alpha = 0.0
            best_val = val_at_0

        if best_alpha > 1e-12:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    # Triangle
    tri = np.minimum(x, n - 1 - x).astype(float) + 1.0
    inits.append(tri / tri.sum())

    # Cosine bell
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Gaussian
    sigma = n / 3.5
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    inits.append(g / g.sum())

    # Flat top with smooth edges
    edge = n * 0.15
    ft = np.ones(n)
    ft[:int(edge)] = np.linspace(0.3, 1.0, int(edge))
    ft[-int(edge):] = np.linspace(1.0, 0.3, int(edge))
    inits.append(ft / ft.sum())

    # W-shaped (two peaks)
    w = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.12)) ** 2) + \
        np.exp(-0.5 * ((x - n * 0.75) / (n * 0.12)) ** 2) + 0.01
    inits.append(w / w.sum())

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.01))
            inits.append(project_simplex(0.9 * (px / ps) + 0.1 * (np.ones(n) / n)))

    return inits


def propose_candidate(budget_s=1800, **kwargs):
    """
    Main search: hybrid smooth gradient + Frank-Wolfe with adaptive time allocation.
    """
    rng = np.random.default_rng(77)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)}) (was {old:.10f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
    except Exception:
        pass

    n = 384
    if parent is not None:
        n = len(parent)

    # === Phase 1: Smooth gradient on parent (fast, many iterations) ===
    print(f"\n=== Phase 1: Smooth Gradient on Parent (n={n}) ===")
    if parent is not None:
        rem = deadline - time.monotonic() - 600
        t1 = max(60, min(rem, 250))
        print(f"  t={t1:.0f}s, T_init=adaptive")
        a_opt, obj = smooth_descent(parent, time_limit=t1,
                                    T_init=obj / 3.0 if obj > 0 else 0.01,
                                    T_min=1e-9, mom=0.92, lr_init=0.015)
        try_seq(a_opt, "smooth_p1")

    # === Phase 2: FW with LP on parent (global direction) ===
    print(f"\n=== Phase 2: FW-LP on Parent ===")
    if parent is not None:
        rem = deadline - time.monotonic() - 500
        t2 = max(60, min(rem, 250))
        print(f"  t={t2:.0f}s")
        a_opt, obj = fw_optimize(parent, time_limit=t2)
        try_seq(a_opt, "fw_p2")

    # === Phase 3: Smooth polish on current best ===
    print(f"\n=== Phase 3: Smooth Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t3 = max(30, min(rem, 120))
        print(f"  t={t3:.0f}s")
        a_opt, obj = smooth_descent(best_seq, time_limit=t3,
                                    T_init=0.002, T_min=1e-10,
                                    mom=0.96, lr_init=0.004)
        try_seq(a_opt, "smooth_p3")

    # === Phase 4: FW deep runs with different starts ===
    print(f"\n=== Phase 4: FW Deep ===")
    if best_seq is not None:
        for p in range(4):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t4 = min(rem // (4 - p), 90)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n) * 0.004)
            elif p == 2:
                f_init = project_simplex(0.92 * best_seq + 0.08 * (np.ones(n) / n))
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n) * 0.01)

            print(f"  FW deep {p}, t={t4:.0f}s")
            a_opt, obj = fw_optimize(f_init, time_limit=t4)
            try_seq(a_opt, f"fw_deep{p}")

    # === Phase 5: Exploration with different n and inits ===
    print(f"\n=== Phase 5: Exploration ===")
    explore_lengths = [n]
    if n != 512:
        explore_lengths.append(512)
    if n != 256:
        explore_lengths.append(256)

    for n_try in explore_lengths:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n and parent is not None:
            continue

        inits = generate_inits(n_try, parent, rng)
        rem = deadline - time.monotonic() - 100
        n_to_try = min(len(inits), 5)
        t_per = max(20, min(50, rem / max(1, n_to_try)))

        for i in range(n_to_try):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 60)
            if t_alloc < 15:
                break
            a_opt, obj = smooth_descent(inits[i], time_limit=t_alloc,
                                        T_init=0.02, T_min=1e-8,
                                        mom=0.9, lr_init=0.03)
            try_seq(a_opt, f"explore_n{n_try}_i{i}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        # Repeated short smooth + FW alternations
        for round_num in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            t_alloc = min(rem // (6 - round_num), 30)
            if t_alloc < 5:
                break

            if round_num % 2 == 0:
                # Smooth
                a_opt, _ = smooth_descent(best_seq, time_limit=t_alloc,
                                          T_init=0.0005, T_min=1e-11,
                                          mom=0.97, lr_init=0.002)
                try_seq(a_opt, f"final_smooth{round_num}")
            else:
                # FW
                a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_fw{round_num}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFINAL SCORE (fallback): {best_score:.10f}")

    return [float(x) for x in best_seq]
