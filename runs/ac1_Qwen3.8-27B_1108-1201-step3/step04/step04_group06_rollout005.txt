"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization on a
log-softmax parameterization (ensuring non-negativity), combined with
soft-max subgradient smoothing, frequency-domain spectral equalization,
conjugate gradient polishing, and multi-scale exploration. The key
innovation is using second-order optimization (L-BFGS) with analytic
gradients in the log-domain, which converges much faster than the
first-order multiplicative weights used previously.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_exact(a, c, n):
    """Compute gradient of max(c) w.r.t. a at the peak location."""
    k_star = int(np.argmax(c))
    g_a = np.zeros(n)
    for i in range(n):
        if 0 <= k_star - i < n:
            g_a[i] = 2.0 * a[k_star - i]
    return g_a


def compute_grad_soft(a, c, n, beta):
    """Compute soft-max weighted gradient."""
    len_c = len(c)
    top_k = max(5, int(n * 0.3))
    if top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
    else:
        indices = np.arange(len_c)
    
    peek = c[indices]
    scaled = beta * peek
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    g_a = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        for i in range(n):
            if 0 <= k - i < n:
                g_a[i] += 2.0 * w[idx] * a[k - i]
    return g_a


def softmax_obj(x, n, beta):
    """Objective function for L-BFGS in log-domain with soft-max."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    c = fast_convolve(a)
    if beta == np.inf:
        return c.max()
    else:
        scaled = beta * c
        scaled -= scaled.max()
        return (1.0 / beta) * np.log(np.sum(np.exp(scaled)))


def softmax_grad(x, n, beta):
    """Gradient for L-BFGS in log-domain."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    c = fast_convolve(a)
    
    if beta == np.inf:
        g_a = compute_grad_exact(a, c, n)
    else:
        g_a = compute_grad_soft(a, c, n, beta)
    
    # Chain rule: df/dx[l] = 2*n*a[l]*(g_a[l] - dot(g_a, a))
    dot_ga_a = np.dot(g_a, a)
    gx = 2.0 * n * a * (g_a - dot_ga_a)
    return gx


def l_bfgs_optimize(a0, beta_seq, max_iter, time_limit):
    """Run L-BFGS with annealing beta (soft-max temperature)."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    x0 = np.log(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t_start = time.monotonic()
    current_x = x0.copy()
    
    for beta in beta_seq:
        if time.monotonic() - t_start > time_limit:
            break
        
        # Define local functions for scipy
        def obj_func(x_local):
            return softmax_obj(x_local, n, beta)
        
        def grad_func(x_local):
            return softmax_grad(x_local, n, beta)
        
        try:
            result = minimize(
                obj_func, current_x, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxls': 50, 'maxcor': 20}
            )
            if result.x is not None:
                current_x = result.x
                x_shifted = current_x - current_x.max()
                ex = np.exp(x_shifted)
                a_new = ex / ex.sum()
                a_new = np.maximum(a_new, 0)
                c_new = fast_convolve(a_new)
                obj_new = c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
        except Exception:
            pass
        
        x_shifted = current_x - current_x.max()
        ex = np.exp(x_shifted)
        current_a = ex / ex.sum()
    
    # Final check
    x_shifted = current_x - current_x.max()
    ex = np.exp(x_shifted)
    a_final = ex / ex.sum()
    c_final = fast_convolve(a_final)
    if c_final.max() < best_obj:
        best_obj = c_final.max()
        best_a = a_final.copy()
    
    return best_a, best_obj


def spectral_equalize(a0, n_iters, time_limit):
    """Frequency-domain equalization: flatten the power spectrum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Zero-pad for linear convolution
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        
        # Target: flat power spectrum (except DC)
        # DC component:
        dc = power[0]
        # Non-DC target:
        target = dc / (fft_size - 1)  # uniform target
        
        # Compute the "excess" spectrum
        excess = np.maximum(power - target, 0)
        
        # Gradient in time domain: the excess in frequency corresponds to
        # a time-domain signal that we subtract
        # d(max_c)/da ≈ ifft(excess * A_conj / |A|^2) (roughly)
        # More directly: we want to reduce the peaks in c
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Time-domain gradient
        g = np.zeros(fft_size)
        for i in range(n):
            if 0 <= k_star - i < n:
                g[i] = 2.0 * a[k_star - i]
        
        # Convert to frequency domain
        G = rfft(g, fft_size)
        
        # Update: reduce components where spectrum is high
        # a_new = a - step * gradient (projected to non-negative)
        step = 0.01
        a_new = a - step * g[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        if c_new.max() < best_obj:
            best_obj = c_new.max()
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step *= 0.5
            a_new = a - step * g[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                c_new = fast_convolve(a_new)
                if c_new.max() < best_obj:
                    best_obj = c_new.max()
                    best_a = a_new.copy()
                    a = a_new
    
    return best_a, best_obj


def conjugate_gradient_opt(a0, time_limit, max_iter=5000):
    """Conjugate gradient method in log-domain (Fletcher-Reeves)."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s
    n = len(a)
    x = np.log(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t_start = time.monotonic()
    
    def get_ax(x_cur):
        x_s = x_cur - x_cur.max()
        ex = np.exp(x_s)
        return ex / ex.sum()
    
    def get_obj_grad(x_cur):
        a_cur = get_ax(x_cur)
        c_cur = fast_convolve(a_cur)
        obj_val = c_cur.max()
        g_a = compute_grad_exact(a_cur, c_cur, n)
        dot_ga = np.dot(g_a, a_cur)
        gx = 2.0 * n * a_cur * (g_a - dot_ga)
        return obj_val, gx
    
    # Initial gradient
    f0, g = get_obj_grad(x)
    p = -g.copy()
    
    for it in range(max_iter):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Line search (backtracking)
        alpha = 0.1
        x_trial = x + alpha * p
        f_trial = softmax_obj(x_trial, n, np.inf)
        
        ls_count = 0
        while f_trial > f0 and alpha > 1e-10 and ls_count < 30:
            alpha *= 0.5
            x_trial = x + alpha * p
            f_trial = softmax_obj(x_trial, n, np.inf)
            ls_count += 1
        
        if alpha < 1e-10:
            break
        
        x = x_trial
        a_cur = get_ax(x)
        c_cur = fast_convolve(a_cur)
        if c_cur.max() < best_obj:
            best_obj = c_cur.max()
            best_a = a_cur.copy()
        
        f_new, g_new = get_obj_grad(x)
        
        # Check convergence
        if np.max(np.abs(g_new)) < 1e-10:
            break
        
        # Fletcher-Reeves update
        gamma = np.dot(g_new, g_new) / max(np.dot(g, g), 1e-30)
        p = -g_new + gamma * p
        g = g_new
        f0 = f_new
    
    return best_a, best_obj


def local_search(a0, time_limit, rng):
    """Random coordinate descent with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t_start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t_start < time_limit:
        if no_improve > 100:
            break
        
        # Pick random coordinates
        n_perturb = max(1, n // 32)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        improved = False
        for i in coords:
            if a[i] < 1e-15:
                continue
            old_val = a[i]
            
            for factor in [0.5, 0.7, 0.85, 1.18, 1.43, 2.0, 0.6, 0.9, 1.11, 1.67]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    break
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
                else:
                    a[i] = old_val
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(rc / rc.sum())
    
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hann / hann.sum())
    
    # Gaussian (various widths)
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sigma_frac * n)) ** 2)
        inits.append(g / g.sum())
    
    # Bimodal (two peaks)
    for sep in [0.2, 0.3, 0.4]:
        bip = (np.exp(-0.5 * ((x - mid * sep) / (0.1 * n)) ** 2) +
               np.exp(-0.5 * ((x - mid * (2 - sep)) / (0.1 * n)) ** 2))
        inits.append(bip / bip.sum())
    
    # Polynomial (degree 2, 4)
    xn = x / max(n - 1, 1)
    for deg in [2, 4, 6]:
        p = 1.0 + deg * (xn - 0.5) ** 2
        inits.append(1.0 / p / np.sum(1.0 / p))
    
    # Edge-concentrated
    edge = np.minimum(x, n - 1 - x).astype(float) + 1
    inits.append(1.0 / edge / np.sum(1.0 / edge))
    
    # Alternating (high-frequency)
    alt = np.where(np.arange(n) % 2 == 0, 2.0, 1.0)
    inits.append(alt / alt.sum())
    
    # Random Dirichlet with various concentrations
    for alpha in [0.1, 0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    return inits


def try_resample(best_a, n_new, method='linear'):
    """Resample a sequence to a new length."""
    n_old = len(best_a)
    if n_new <= 1 or n_old <= 1:
        return None
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, best_a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
        return a_new
    return None

            t_alloc = min(60, rem * 0.7)
            print(f"  Local search, alloc={t_alloc:.0f}s")
            a_opt, _ = local_search(best_seq, time_limit=t_alloc, rng=rng)
            try_seq(a_opt, "local1")
            
            # Another local search with fresh rng
            rem2 = deadline - time.monotonic() - 30
            if rem2 > 10:
                rng2 = np.random.default_rng(seed + 999)
                a_opt2, _ = local_search(best_seq, time_limit=min(30, rem2 * 0.5), rng=rng2)
                try_seq(a_opt2, "local2")
    
    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different Lengths ===")
    print(f"  Best so far: {best_score:.8f}")
    
    if best_seq is not None:
        alt_lengths = []
        for mult in [0.5, 0.75, 1.5, 2.0, 0.6, 1.2]:
            n_new = int(n * mult)
            if 64 <= n_new <= 2048:
                alt_lengths.append(n_new)
        # Also try some fixed lengths
        for nl in [128, 192, 256, 384, 512, 640, 768, 1024]:
            if nl not in alt_lengths:
                alt_lengths.append(nl)
        
        # Sort by proximity to current n (try closest first)
        alt_lengths = sorted(set(alt_lengths), key=lambda x: abs(x - n))
        
        for nl in alt_lengths[:6]:
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            a_res = try_resample(best_seq, nl)
            if a_res is None:
                continue
            t_alloc = min(40, rem / 2)
            if t_alloc < 10:
                break
            a_opt, _ = l_bfgs_optimize(a_res, [100, np.inf], max_iter=100, time_limit=t_alloc)
            try_seq(a_opt, f"len_n{nl}")
    
    # === Phase 8: Final fine L-BFGS ===
    print(f"\n=== Phase 8: Final Polish ===")
    print(f"  Best so far: {best_score:.8f}")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(50, rem * 0.6)
            print(f"  Final L-BFGS, alloc={t_alloc:.0f}s")
            a_opt, _ = l_bfgs_optimize(best_seq, [np.inf], max_iter=1000, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfgs")
            
            # Final local search
            rem2 = deadline - time.monotonic() - 5
            if rem2 > 5:
                rng3 = np.random.default_rng(seed + 7777)
                a_opt2, _ = local_search(best_seq, time_limit=min(20, rem2), rng=rng3)
                try_seq(a_opt2, "final_local")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
