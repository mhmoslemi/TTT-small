"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_gradient(f, conv, tau):
    """
    Compute gradient of log-sum-exp of conv / tau, which is a smooth approximation
    to max(conv). Returns (value, gradient).
    """
    n = len(f)
    # Only need to consider k = 0, ..., n-1 due to symmetry
    # But for safety, use all
    scaled = conv / tau
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    lse = max_s + np.log(exp_s.sum())
    
    # Gradient of lse with respect to f
    weights = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # d conv[k] / d f[i] = f[k-i] for i in [js, je]
        grad[idx] += 2.0 * weights[k] * f[k - idx]
    
    return lse * tau, grad  # value is tau * lse ≈ smoothed max, grad is gradient of smoothed max


def smoothed_descent(f0, time_limit, tau0=0.001, tau_decay=0.999, lr0=0.01, mom=0.9):
    """
    Smoothed gradient descent with adaptive temperature and momentum.
    Minimizes smoothed max(conv(f,f)) on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    tau = tau0
    lr = lr0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        it += 1
        lse_val, grad = logsumexp_gradient(f, conv, tau)
        
        # Momentum step
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Check exact objective
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        
        # Adaptive parameters
        tau *= tau_decay
        lr *= 0.9995
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            tau = max(tau0 * 0.5, tau * 2)
            lr = lr0 * 0.5
            no_improve = 0
    
    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_smoothed_conv(f, tau):
    """
    Compute log-sum-exp smoothed max of conv(f,f) and its gradient.
    Returns (smoothed_value, gradient).
    """
    conv = fast_convolve(f)
    n = len(f)
    scaled = conv / tau
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    value = tau * (max_s + np.log(exp_s.sum()))
    
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * p[k] * f[k - idx]
    
    return value, grad, conv

    """Build sparse constraint matrix for conv(f,g) constraints."""

    inits.append(h / h.sum())
    
    # Exponential edge-concentrated
    e = np.exp(2.0 * np.abs(x - n / 2) / (n / 2))
    inits.append(e / e.sum())
    
    # Exponential center-concentrated
    ec = np.exp(-3.0 * (x - n / 2) ** 2 / (n * n / 4))
    inits.append(ec / ec.sum())
    
    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Sine
    s = np.sin(np.pi * (x + 0.5) / n) + 0.05
    inits.append(s / s.sum())
    
    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Power law edge
    p = (1.0 + np.abs(x - n / 2) / (n / 2)) ** 2
    inits.append(p / p.sum())
    
    # Bimodal
    b1 = np.exp(-((x - n * 0.2) ** 2) / (n * 0.05))
    b2 = np.exp(-((x - n * 0.8) ** 2) / (n * 0.05))
    bim = b1 + b2 + 0.01
    inits.append(bim / bim.sum())
    
    # Parent-based (if available)
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px_norm = px / ps
            inits.append(px_norm)
            # Blend parent with U-shape
            blend = 0.6 * px_norm + 0.4 * (u / u.sum())
            inits.append(blend / blend.sum())
            # Blend parent with uniform
            blend2 = 0.7 * px_norm + 0.3 * (np.ones(n) / n)
            inits.append(blend2 / blend2.sum())
            # Blend parent with random
            r = rng.dirichlet(np.ones(n) * 0.3)
            blend3 = 0.5 * px_norm + 0.5 * r
            inits.append(blend3 / blend3.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining smoothed descent,
    LP Frank-Wolfe, and coordinate descent with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f}) at t={elapsed:.0f}s")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, obj={fast_convolve(parent).max() * 2 * len(parent):.8f}")
    except Exception:
        pass
    
    # Determine sequence lengths to try
    lengths = [384]
    if parent is not None:
        if len(parent) not in lengths:
            lengths.insert(0, len(parent))
    lengths.extend([512, 256, 250, 400])
    lengths = list(dict.fromkeys(lengths))  # deduplicate preserving order
    
    print(f"=== Multi-phase optimization ===")
    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline at t+{deadline - t_start:.0f}s")
    
    # === Phase 1: Quick exploration with smoothed descent ===
    print("\n--- Phase 1: Smoothed Descent Exploration ---")
    
    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        
        inits = generate_diverse_inits(n, parent, rng)
        n_inits = len(inits)
        
        # Allocate time: more for smaller sets, less for larger
        t_alloc_total = min(80, (deadline - time.monotonic() - 150) / max(1, len(lengths) - lengths.index(n)))
        t_per_init = max(8, t_alloc_total / n_inits)
        
        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 5:
                break
            
            a_opt, _ = smoothed_descent(
                f_init, time_limit=t_alloc,
                tau0=0.003, tau_end=0.0001,
                lr0=0.03, mom=0.82,
                n_restarts=3, restart_every=40
            )
            try_seq(a_opt, f"sd_n{n}_i{i}")
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: LP refinement on top candidates ===
    print("\n--- Phase 2: LP Frank-Wolfe Refinement ---")
    
    if best_seq is not None:
        # Refine the best with FW
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            a_opt, _ = fw_refine(best_seq, time_limit=min(rem, 120), verbose=True)
            try_seq(a_opt, "fw_best")
        
        # Also try FW from a few perturbed starts
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            if p == 0:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif p == 1:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best))
            else:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (best_seq[::-1]))
            
            a_opt, _ = fw_refine(f_init, time_limit=min(rem, 60))
            try_seq(a_opt, f"fw_pert{p}")
    
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: Coordinate descent polishing ===
    print("\n--- Phase 3: Coordinate Descent ---")
    
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt, _ = coord_descent(best_seq, time_limit=min(rem, 100), n_pairs=3000)
            try_seq(a_opt, "coord1")
        
        # Another round after perturbation
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            a_opt, _ = coord_descent(f_pert, time_limit=min(rem, 60), n_pairs=2000)
            try_seq(a_opt, "coord2")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Deep LP + smoothed cycle ===
    print("\n--- Phase 4: Deep Refinement Cycle ---")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Alternating LW passes
        for cycle in range(5):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            
            # Smoothed descent pass
            t_sd = min(rem * 0.4, 40)
            if t_sd > 5:
                a_opt, _ = smoothed_descent(
                    best_seq, time_limit=t_sd,
                    tau0=0.001 / (cycle + 1), tau_end=0.00002,
                    lr0=0.01 / (1 + 0.3 * cycle), mom=0.88,
                    n_restarts=2, restart_every=60
                )
                try_seq(a_opt, f"cycle{cycle}_sd")
            
            # FW pass
            rem = deadline - time.monotonic() - 15
            t_fw = min(rem * 0.5, 50)
            if t_fw > 10:
                a_opt, _ = fw_refine(best_seq, time_limit=t_fw)
                try_seq(a_opt, f"cycle{cycle}_fw")
            
            # Coord descent pass
            rem = deadline - time.monotonic() - 10
            t_cd = min(rem * 0.6, 30)
            if t_cd > 8:
                a_opt, _ = coord_descent(best_seq, time_limit=t_cd, n_pairs=1500)
                try_seq(a_opt, f"cycle{cycle}_cd")
    
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Final micro-polish ===
    print("\n--- Phase 5: Final Polish ---")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Very fine coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coord_descent(best_seq, time_limit=min(rem, 20), n_pairs=2000)
            try_seq(a_opt, "final_cd")
        
        # Micro perturbations + polish
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            for m in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 4:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = smoothed_descent(
                    f_pert, time_limit=min(rem, 8),
                    tau0=0.0005, tau_end=0.00005,
                    lr0=0.005, mom=0.92,
                    n_restarts=1, restart_every=50
                )
                try_seq(a_opt, f"micro{m}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    total_time = time.monotonic() - t_start
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.1f}s")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
