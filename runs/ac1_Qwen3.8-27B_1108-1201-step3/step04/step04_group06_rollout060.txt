The objective is: f(a) = (1/τ) * log(sum_k exp(τ * c[k])) where c = conv(a,a).

The gradient df/dy[i] = sum_j (df/da[j]) * (da[j]/dy[i])

where da[j]/dy[i] = a[j] * (δ(j,i) - a[i])

So: df/dy[i] = sum_j (df/da[j]) * a[j] * (δ(j,i) - a[i])

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Exponentiated gradient with adaptive temperature and momentum
2. L-BFGS on log-sum-exp smoothed objective (scipy)
3. Multi-length exploration
4. Local search polishing with structured perturbations

import numpy as np
import time
import scipy.optimize
import scipy.special

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def compute_subgradient_full(a, c, tau):
    """Compute full softmax-weighted subgradient of max(c)."""
    n = len(a)
    w = softmax(tau * c)
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je+1] += 2.0 * w[k] * a[k-je:k-js+1][::-1]
    return g

def compute_subgradient_tophat(a, c, top_k):
    """Compute subgradient using top-k lags with uniform weights."""
    n = len(a)
    indices = np.argpartition(c, -top_k)[-top_k:]
    g = np.zeros(n)
    w = 1.0 / top_k
    for k in indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je+1] += 2.0 * w * a[k-je:k-js+1][::-1]
    return g

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    mu = 0.0
    sorted_v = np.sort(v)[::-1]
    csv = np.cumsum(sorted_v)
    rho = np.where(sorted_v * (np.arange(1, n+1) - 1) > (csv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    mu = (csv[rho] - 1) / (rho + 1)
    return np.maximum(v - mu, 0)

def eg_optimize_adaptive(a0, time_limit, eta0=1.0, tau0=1.0, tau_max=100.0,
                         momentum=0.7, verbose=False):
    """
    Exponentiated gradient with adaptive temperature and momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    tau = tau0
    g_ema = np.zeros(n)  # momentum buffer
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive temperature: increase tau when the top values are spread out
        top_10 = np.sort(c)[-min(10, len(c)):]
        spread = (top_10[0] - top_10[-1]) / (top_10[-1] + 1e-300)
        # If spread is large, use lower tau (more uniform weighting)
        # If spread is small (one dominant peak), use higher tau
        tau_target = np.clip(1.0 / max(spread, 0.1), 0.5, tau_max)
        tau = 0.95 * tau + 0.05 * tau_target
        
        # Compute subgradient
        sg = compute_subgradient_full(a, c, tau)
        
        # Momentum
        g_ema = momentum * g_ema + (1 - momentum) * sg
        g_ema_norm = np.linalg.norm(g_ema)
        if g_ema_norm < 1e-15:
            break
        
        # Multiplicative update
        log_update = -eta * g_ema / g_ema_norm
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Simulated annealing: occasionally accept worse solutions
            if no_improve > 50 and np.random.random() < 0.1:
                a = a_new  # accept anyway
            else:
                a = a_new
        
        c = c_new
        
        # Periodic check of running average
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
                no_improve = 0
        
        # Decay eta
        eta *= 0.9999
        
        # Stagnation: restart
        if no_improve > 2000:
            perturbation = np.random.standard_normal(n) * 0.001
            a = best_a + perturbation
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            g_ema = np.zeros(n)
            tau = tau0
            avg_a = a.copy()
            count = 0
    
    return best_a.copy(), best_obj

def lbsfgs_smoothed(a0, time_limit, tau=5.0, maxiter=2000):
    """
    L-BFGS optimization on the log-sum-exp smoothed objective
    using softmax parameterization.
    """
    n = len(a0)
    
    def objective_and_grad(y):
        # Softmax parameterization
        y_shifted = y - y.max()
        a = np.exp(y_shifted)
        a = a / a.sum()
        
        # Compute convolution
        c = fast_convolve(a)
        
        # Log-sum-exp smoothed objective
        tau_c = tau * c
        max_tc = tau_c.max()
        lse = max_tc + np.log(np.sum(np.exp(tau_c - max_tc)))
        obj = lse / tau
        
        # Gradient
        w = softmax(tau_c)  # shape (2n-1,)
        # g[j] = sum_k w[k] * 2*a[k-j] (with bounds)
        g = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += 2.0 * w[k] * a[k-je:k-js+1][::-1]
        
        # Chain rule for softmax: dy[i] = a[i] * (g[i] - sum(a*g))
        ag = np.dot(a, g)
        grad = a * (g - ag)
        
        return obj, grad
    
    # Initialize y from a0
    a_init = np.maximum(a0, 1e-10)
    y0 = np.log(a_init / a_init.sum())
    
    result = scipy.optimize.minimize(
        lambda y: objective_and_grad(y)[0],
        y0,
        jac=lambda y: objective_and_grad(y)[1],
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'maxfun': time_limit * 10,
                 'ftol': 1e-15, 'gtol': 1e-8, 'disp': False}
    )
    
    y_opt = result.x
    y_shifted = y_opt - y_opt.max()
    a_opt = np.exp(y_shifted)
    a_opt = a_opt / a_opt.sum()
    
    return a_opt

def local_search_structured(a0, time_limit, verbose=False):
    """
    Local search with structured perturbations:
    - Sinusoidal perturbations at various frequencies
    - Edge modification
    - Block scaling
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 100:
        improved = False
        
        # Try various structured perturbations
        perturbations = []
        
        # Sinusoidal perturbations
        for freq in [1, 2, 3, 5, 7, 10, 15, 20]:
            if freq < n:
                for amp in [0.001, 0.005, 0.01, 0.02]:
                    phase = rng.random() * 2 * np.pi
                    pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                    perturbations.append(pert)
        
        # Edge modification
        for edge_frac in [0.05, 0.1, 0.2, 0.3]:
            edge_size = int(n * edge_frac)
            if edge_size > 0:
                for delta in [0.01, 0.02, 0.05]:
                    pert = np.zeros(n)
                    pert[:edge_size] -= delta
                    pert[n-edge_size:] -= delta
                    perturbations.append(pert)
        
        # Block scaling
        for n_blocks in [4, 8, 16, 32]:
            for scale in [0.9, 0.95, 1.05, 1.1]:
                block_size = n // n_blocks
                pert = np.zeros(n)
                for b in range(n_blocks):
                    start = b * block_size
                    end = start + block_size if b < n_blocks - 1 else n
                    pert[start:end] = (scale - 1) * a[start:end].mean()
                perturbations.append(pert)
        
        # Random
        for _ in range(5):
            perturbations.append(rng.standard_normal(n) * 0.005)
        
        # Try perturbations
        for pert in perturbations:
            if time.monotonic() - t0 > time_limit:
                break
            
            a_new = a + pert
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                c = c_new
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a.copy(), best_obj

def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective as a function of a[i] (with others fixed,
    after renormalization) is a smooth function that we can minimize.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small positive value
                for new_val in [1e-4, 1e-3, 0.01/n]:
                    a[i] = new_val
                    s2 = a.sum()
                    a_norm = a / s2
                    c2 = fast_convolve(a_norm)
                    if c2.max() < best_obj - 1e-12:
                        best_obj = c2.max()
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        improved = True
                        break
                a[i] = old_val
                continue
            
            # Try multiplicative factors
            for factor in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                if c2.max() < best_obj - 1e-12:
                    best_obj = c2.max()
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
            
            a[i] = old_val if not improved else a[i]  # keep if improved
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()  # reset to best
            c = fast_convolve(a)
    
    return best_a.copy(), best_obj

def generate_diverse_inits(best_a, n, seed):
    """Generate diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # From best (resampled)
    if len(best_a) != n:
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n)
        b = np.interp(x_new, x_old, best_a)
        b = np.maximum(b, 0)
        if b.sum() > 1e-10:
            inits.append(b / b.sum())
    else:
        inits.append(best_a.copy())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangle
    tri = 1.0 - 2.0 * np.abs(x / (n-1) - 0.5)
    inits.append(tri / tri.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits.append(g / g.sum())
    
    # Uniform middle, small edges
    u = np.ones(n)
    edge = int(n * 0.1)
    u[:edge] = 0.1
    u[-edge:] = 0.1
    inits.append(u / u.sum())
    
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / (n-1)))
    cw = np.maximum(cw, 0.01)
    inits.append(cw / cw.sum())
    
    # Multi-peak
    for npeaks in [2, 3, 4]:
        peaks = np.random.choice(n, npeaks, replace=False)
        m = np.zeros(n)
        for p in peaks:
            m += np.exp(-0.5 * ((x - p) / (n/10))**2)
        inits.append(m / m.sum())
    
    # Palindromic random
    half = n // 2
    h = rng.dirichlet(np.ones(half) * 0.5)
    p = np.zeros(n)
    p[:half] = h
    if n % 2 == 0:
        p[half:] = h[::-1]
    else:
        p[half] = 0.1
        p[half+1:] = h[:half-1][::-1]
    p = np.maximum(p, 0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    
    # Random Dirichlet
    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(r)
    
    # Perturbed best
    if len(best_a) == n:
        for scale in [0.001, 0.01, 0.05]:
            p = best_a + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
1. Exponentiated gradient with adaptive temperature, momentum, and simulated annealing
2. L-BFGS on log-sum-exp smoothed objective via softmax parameterization
3. Subgradient bundle method for boundary exploration
4. Structured local search (sinusoidal, block, edge perturbations)
5. Coordinate descent for fine polishing
6. Multi-length and multi-initialization exploration
"""

import numpy as np
import time
import scipy.optimize
import scipy.special
from functools import partial


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return np.array([a[0] * a[0]])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        out = np.zeros_like(x)
        out[x.argmax()] = 1.0
        return out
    return e / s


def compute_weighted_subgradient(a, c, weights):
    """
    Compute weighted sum of subgradients.
    weights: array of shape (2n-1,) with weights for each lag.
    Returns gradient of sum_k weights[k] * c[k] w.r.t. a.
    """
    n = len(a)
    g = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je + 1] += 2.0 * wk * a[k - je:k - js + 1][::-1]
    return g


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v)
    cond = sorted_v * np.arange(1, n + 1) > (css - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    mu = (css[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - mu, 0.0)
    # Ensure it sums to 1 (numerical fix)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def eg_optimize(a0, time_limit, eta0=1.0, tau0=1.0, tau_max=50.0,
                tau_min=0.1, momentum=0.5, anneal_prob=0.05,
                anneal_start=100, anneal_decay=0.99, verbose=False):
    """
    Exponentiated gradient with adaptive temperature, momentum, and annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    tau = tau0
    g_ema = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    anneal_p = anneal_prob

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive temperature
        top_5 = np.sort(c)[-5:]
        if len(top_5) >= 5:
            if top_5[0] > 2.0 * top_5[4]:
                tau = min(tau * 1.05, tau_max)
            elif top_5[0] < 1.05 * top_5[4]:
                tau = max(tau * 0.98, tau_min)

        # Compute subgradient with softmax weighting
        w = softmax_fn(tau * c)
        sg = compute_weighted_subgradient(a, c, w)

        # Momentum
        g_ema = momentum * g_ema + (1.0 - momentum) * sg
        g_norm = np.linalg.norm(g_ema)
        if g_norm < 1e-15:
            break

        # Multiplicative update
        log_update = -eta * g_ema / g_norm
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Simulated annealing: occasionally accept worse to escape
        # (In EG, we always "accept" since the update is deterministic,
        #  but we can add noise)
        if no_improve > anneal_start:
            if np.random.random() < anneal_p:
                noise_scale = 0.005 * (1.0 + no_improve / 1000.0)
                a = a + np.random.standard_normal(n) * noise_scale
                a = np.maximum(a, 0)
                ps = a.sum()
                if ps > 1e-10:
                    a = a / ps
                c = fast_convolve(a)

        # Periodic check of running average
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-13:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
                no_improve = 0

        # Decay
        eta *= 0.99995
        anneal_p *= anneal_decay

        # Stagnation restart
        if no_improve > 3000:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            g_ema = np.zeros(n)
            tau = max(tau0, tau_min)
            avg_a = a.copy()
            count = 0
            anneal_p = anneal_prob

        if verbose and it % 5000 == 0:
            print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, tau={tau:.4f}")

    return best_a.copy(), best_obj


def _lbfgs_compute(y, tau):
    """Compute objective and gradient for L-BFGS on smoothed objective."""
    n = len(y)
    y_shifted = y - y.max()
    a = np.exp(y_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return 1e10, np.zeros(n)
    a = a / a_sum

    c = fast_convolve(a)
    tau_c = tau * c

    # Log-sum-exp for numerical stability
    max_tc = tau_c.max()
    lse = max_tc + np.log(np.sum(np.exp(tau_c - max_tc)))
    obj = lse / tau

    # Gradient
    w = softmax_fn(tau_c)
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je + 1] += 2.0 * w[k] * a[k - je:k - js + 1][::-1]

    # Chain rule for softmax parameterization
    ag = np.dot(a, g)
    grad = a * (g - ag)

    return obj, grad


def _lbfgs_f(y, tau):
    obj, _ = _lbfgs_compute(y, tau)
    return obj


def _lbfgs_grad(y, tau):
    _, grad = _lbfgs_compute(y, tau)
    return grad


def lbfgs_smoothed_optimize(a0, time_limit, tau=5.0, maxiter=2000):
    """L-BFGS optimization on log-sum-exp smoothed objective."""
    n = len(a0)
    a_init = np.maximum(a0, 1e-10)
    a_init = a_init / a_init.sum()
    y0 = np.log(a_init)
    y0 = y0 - y0.mean()  # center for numerical stability

    f = partial(_lbfgs_f, tau=tau)
    g = partial(_lbfgs_grad, tau=tau)

    t0 = time.monotonic()

    # Custom callback to enforce time limit
    stop_flag = [False]

    def callback(xk):
        if time.monotonic() - t0 > time_limit:
            stop_flag[0] = True
            return True
        return False

    try:
        result = scipy.optimize.minimize(
            f, y0, jac=g, method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': maxiter * 10,
                     'ftol': 1e-18, 'gtol': 1e-10, 'disp': False,
                     'maxls': 20}
        )
        y_opt = result.x
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a0.copy()

    y_shifted = y_opt - y_opt.max()
    a_opt = np.exp(y_shifted)
    a_opt = a_opt / a_opt.sum()

    return a_opt
