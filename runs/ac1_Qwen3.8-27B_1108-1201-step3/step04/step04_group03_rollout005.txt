"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
incremental convolution updates, combined with multiplicative weight
global moves and random restarts for escaping local minima.
Key innovation: exact 1D coordinate optimization via bisection on the
subgradient of the max-of-linear-functions objective.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_descent(a0, time_limit, restart_every=200, anneal_start_temp=0.002,
                  anneal_rate=0.999, seed=0):
    """
    Coordinate descent minimizing max(conv(a,a)) on the simplex.
    Uses incremental convolution updates. Periodic annealing restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    cur_temp = anneal_start_temp
    sweep_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep_count += 1

        for i in range(n):
            v = a[i]
            if v < 1e-18:
                continue

            # Affected range: k in [i, min(2n-2, i+n-1)]
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)

            # Unaffected max
            c_unaff = 0.0
            if k_start > 0:
                c_unaff = max(c_unaff, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                c_unaff = max(c_unaff, float(conv[k_end + 1:].max()))

            # Build linear terms
            k_vals = np.arange(k_start, k_end + 1)
            k_off = k_vals - i

            quad_mask = (k_vals == 2 * i)
            lin_mask = ~quad_mask

            A = np.full(len(k_vals), -np.inf)
            B = np.zeros(len(k_vals))

            if lin_mask.any():
                kl = k_vals[lin_mask]
                ol = k_off[lin_mask]
                A[lin_mask] = conv[kl] - 2.0 * v * a[ol]
                B[lin_mask] = 2.0 * a[ol]

            Q = None
            if quad_mask.any():
                Q = conv[2 * i] - v * v

            x_opt = find_optimal_coord(A, B, c_unaff, Q, v, n)

            # Annealing: with small probability, accept a worse x
            if cur_temp > 1e-8 and x_opt > 0:
                r = rng.random()
                if r < cur_temp * 0.01:
                    x_opt = v * (1.0 + rng.standard_normal() * 0.5 * max(1.0, cur_temp * 1000))
                    x_opt = max(0.0, x_opt)

            delta = x_opt - v
            if abs(delta) < 1e-18:
                continue

            # Update conv incrementally
            if lin_mask.any():
                conv[k_vals[lin_mask]] += 2.0 * delta * a[k_off[lin_mask]]
            if quad_mask.any():
                conv[2 * i] += x_opt * x_opt - v * v

            a[i] = x_opt

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
            conv = fast_convolve(a)

        cur_max = conv.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= anneal_rate

        # Periodic restart
        if no_improve >= restart_every or no_improve % (restart_every * 3) == 0:
            if time.monotonic() - t0 > time_limit * 0.8:
                break
            noise_scale = max(0.001, cur_temp * 50)
            a = best_a * np.exp(rng.standard_normal(n) * noise_scale)
            a = np.maximum(a, 0.0)
            s = a.sum()

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            ii = np.arange(js, je + 1)
            log_update[ii] -= lr * w[idx] * 2.0 * a[k - ii]

        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-25)
        a = a / a.sum()

        cur_conv = fast_convolve(a).max()
        if cur_conv < best_conv:
            best_conv = cur_conv
            best_a = a.copy()

    return best_a


def try_sequence(a, best_state):
    """Evaluate a sequence and update best state if improved."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a = a / s
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge emphasis
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(rc / rc.sum())

    # Bimodal
    bm = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(bm / bm.sum())

    # Hamming-like (high at edges)
    ham = (x.astype(int) ^ (n - 1 - x.astype(int))).clip(min=1)
    inits.append(ham / ham.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase coordinate descent with annealing,
    multiplicative weight global moves, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = [float('inf'), None]

    def try_seq(a, label=""):
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_state[0]:
            best_state[0] = score
            best_state[1] = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent_init")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is not None:
        n_best = len(parent)
    else:
        n_best = 384
        parent = np.ones(n_best) / n_best

    # Phase 1: Quick coordinate descent on parent and a few inits (~300s)
    print("\n=== Phase 1: Quick Coordinate Descent ===")
    rem = deadline - time.monotonic()

    if parent is not None:
        t1 = min(120, rem * 0.2)
        print(f"  CD on parent (n={n_best}), t={t1:.0f}s")
        a_opt, _ = coord_descent(parent, time_limit=t1, restart_every=100, seed=1)
        try_seq(a_opt, "cd_parent")

    # Try a couple other inits
    inits = generate_inits(n_best, parent, rng)
    for fi in [0, 2, 4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(60, rem * 0.1)
        if t_alloc < 15:
            break
        print(f"  CD on init_{fi}, t={t_alloc:.0f}s")
        a_opt, _ = coord_descent(inits[fi], time_limit=t_alloc, restart_every=80, seed=10 + fi)
        try_seq(a_opt, f"cd_init_{fi}")

    print(f"  Phase 1 best: {best_state[0]:.8f}")

    # Phase 2: Deep coordinate descent on best (~600s)
    print("\n=== Phase 2: Deep Coordinate Descent ===")
    if best_state[1] is not None:
        rem = deadline - time.monotonic()
        t2 = min(500, rem * 0.4)
        print(f"  Deep CD, t={t2:.0f}s")
        a_opt, _ = coord_descent(best_state[1], time_limit=t2,
                                  restart_every=300, anneal_start_temp=0.001,
                                  anneal_rate=0.9995, seed=100)
        try_seq(a_opt, "deep_cd")

    print(f"  Phase 2 best: {best_state[0]:.8f}")

    # Phase 3: Interleave CD and MW global moves (~400s)
    print("\n=== Phase 3: CD + MW Interleaving ===")
    if best_state[1] is not None:
        current = best_state[1].copy()
        rem = deadline - time.monotonic()
        block_time = min(80, rem * 0.1)
        max_blocks = int(rem / block_time)

        for block in range(max_blocks):
            rem = deadline - time.monotonic()
            if rem < 30:
                break

            # MW step for global move
            mw_t = min(10, rem * 0.05)
            if rem > 60:
                current = mult_weight_step(current, lr=0.02, top_k=15,
                                           temp=0.0003 + block * 0.0001,
                                           n_iter=max(20, int(mw_t * 10)))
                try_seq(current, f"mw_block_{block}")

            # CD refinement
            cd_t = min(block_time - 5, rem * 0.3)
            if cd_t > 10:
                a_opt, _ = coord_descent(current, time_limit=cd_t,
                                         restart_every=200, anneal_start_temp=0.0005,
                                         anneal_rate=0.999, seed=200 + block)
                try_seq(a_opt, f"cd_block_{block}")
                if best_state[1] is not None:
                    current = best_state[1].copy()

    print(f"  Phase 3 best: {best_state[0]:.8f}")

    # Phase 4: More CD with different hyperparameters (~250s)
    print("\n=== Phase 4: Diverse CD Attempts ===")
    if best_state[1] is not None:
        configs = [
            {'restart_every': 150, 'anneal_start_temp': 0.003, 'anneal_rate': 0.998, 'seed': 500},
            {'restart_every': 500, 'anneal_start_temp': 0.0005, 'anneal_rate': 0.9998, 'seed': 600},
            {'restart_every': 100, 'anneal_start_temp': 0.005, 'anneal_rate': 0.997, 'seed': 700},
        ]

        rem = deadline - time.monotonic()
        per_cfg = max(20, rem / (len(configs) + 1))

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(per_cfg, rem * 0.5)
            if t_alloc < 15:
                break
            print(f"  CD config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(best_state[1], time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"cd_cfg_{ci}")

    print(f"  Phase 4 best: {best_state[0]:.8f}")

    # Phase 5: Final polish with very fine CD (~150s)
    print("\n=== Phase 5: Final Polish ===")
    if best_state[1] is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t5 = min(120, rem * 0.7)
            print(f"  Fine CD, t={t5:.0f}s")
            a_opt, _ = coord_descent(best_state[1], time_limit=t5,
                                     restart_every=1000, anneal_start_temp=0.0001,
                                     anneal_rate=0.9999, seed=999)
            try_seq(a_opt, "final_cd")

        # Tiny perturbation attempts
        rem = deadline - time.monotonic()
        if rem > 20:
            for pi in range(3):
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                noise = rng.standard_normal(len(best_state[1])) * 0.001
                perturbed = project_simplex(best_state[1] + noise)
                t_alloc = min(30, rem * 0.5)
                a_opt, _ = coord_descent(perturbed, time_limit=t_alloc,
                                         restart_every=500, anneal_start_temp=0.0002,
                                         anneal_rate=0.9995, seed=1000 + pi)
                try_seq(a_opt, f"perturb_{pi}")

    # Fallback
    if best_state[1] is None:
        best_state[1] = np.ones(384) / 384
        best_state[0] = float(evaluate_sequence(best_state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_state[1].tolist()))
    if np.isfinite(fs) and fs < best_state[0]:
        best_state[0] = fs

    print(f"\nFINAL SCORE: {best_state[0]:.8f} (n={len(best_state[1])})")
    return [float(x) for x in best_state[1]]
