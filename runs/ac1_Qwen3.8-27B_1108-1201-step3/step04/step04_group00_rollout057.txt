def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_stable(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

    """Compute h_i = sum_{j=0}^{n-1} f[j] * w[i+j] for i=0..n-1.
    w has length 2n-1, f has length n.
    """
    n = len(f)
    # h = cross-correlation of f and w (valid part)
    # h_i = sum_j f[j] * w[i+j]

    w_rev = w[::-1]
    conv = fftconvolve(f, w_rev)  # length n + (2n-1) - 1 = 3n-2
    # h[i] = conv[2n-2-i]
    h = conv[n-1:2*n-1][::-1]
    return h[:n]  # ensure length n

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k most active k, g >= 0."""
    conv_f = fast_convolve(f)
    # Get top-k most active constraints
    if top_k < 2*n-1:
        active = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        active = np.arange(2*n-1)
    
    # Build constraint matrix for active constraints only
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    A_ub = np.zeros((len(active), n))
    for i, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[j]  # Wait, this should be f[k-j]

    A_ub = np.zeros((len(active), n))

    return None

def nesterov_gd(x0, n, time_limit, eps_init=0.1, eps_min=0.001, eta_init=0.01):
    """Nesterov-accelerated gradient descent on the smoothed objective."""
    x = x0.copy()
    y = x.copy()
    
    conv_y = fast_convolve(softmax_stable(y))
    best_obj = conv_y.max()
    best_x = x.copy()
    
    t0 = time.monotonic()
    it = 0
    eps = eps_init
    eta = eta_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Update i (momentum)
        beta = (1 + np.sqrt(1 + 4 * it * it)) / 2  # Nesterov momentum
        y = x + (beta - 1) / (beta + 1) * (y - x)
        
        # Compute gradient at y
        f = softmax_stable(y)
        conv = fast_convolve(f)
        
        # Softmax weights
        conv_shifted = (conv - conv.max()) / eps
        w = np.exp(conv_shifted)
        w /= w.sum()
        
        # h_i = sum_j f[j] * w[i+j]
        h = compute_h_fft(w, f)
        
        # grad_f = 2 * h
        grad_f = 2.0 * h
        
        # grad_x = f * (grad_f - sum(f * grad_f))
        dot = np.dot(f, grad_f)
        grad_x = f * (grad_f - dot)
        
        # Update x
        # Adaptive step size: use backtracking
        # Or fixed with decay
        x_new = y - eta * grad_x
        
        # Optional: project back (but softmax already handles this)
        x = x_new
        
        # Evaluate
        f_new = softmax_stable(x)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_x = x.copy()
        
        # Decay epsilon and eta
        if it % 100 == 0:
            eps = max(eps * 0.98, eps_min)
            eta *= 0.999
        
        if it % 1000 == 0:
            print(f"  NGD iter {it}: obj={obj:.10f}, best={best_obj:.10f}")
    
    return best_x, best_obj

**Approach: Iterative reweighting / power iteration**

The KKT conditions at the optimum (for the problem min max_k (a*a)[k] s.t. a >= 0, sum(a) = 1) are:
- There exist multipliers λ_k >= 0 for the active constraints
- sum_k λ_k * 2 * a_{k-j} = μ for all j with a_j > 0
- sum_k λ_k * 2 * a_{k-j} <= μ for all j with a_j = 0

This suggests an iterative algorithm:
1. Start with a
2. Compute conv = a * a, find M = max(conv)
3. Set λ_k = 1 for all k where conv[k] = M (or use a soft version)
4. Update: a_j <- a_j / (sum_k λ_k * 2 * a_{k-j}) (multiplicative update)
5. Normalize: a <- a / sum(a)
6. Repeat

This is similar to the "power method" for eigenvalue problems or the "sink scaling" in network flow. It's related to the matrix scaling / iterative proportional fitting (IPF) algorithm.

Let me think about this more carefully. The update is:
a_j <- a_j / (2 * sum_k λ_k * a_{k-j})

With λ_k = 1 for active k (where conv[k] = M), this becomes:
a_j <- a_j / (2 * sum_{k active} a_{k-j})

This is a multiplicative update that reduces the values of a_j that are "responsible" for the peaks.

Actually, this is very similar to the **mirror descent** / **multiplicative weights** approach! The Bregman divergence for the entropy is the KL divergence, and the update for mirror descent on the simplex is:
a_j <- a_j * exp(-η * g_j) / Z

where g_j is the subgradient and Z is a normalizing constant.

For the subgradient of max_k (a*a)[k]:
g_j = 2 * sum_{k in S} a_{k-j} / |S|

where S is the set of active indices (where (a*a)[k] = max).

So the mirror descent update is:
a_j <- a_j * exp(-η * 2 * sum_{k in S} a_{k-j} / |S|) / Z

This is exactly the multiplicative update I described above (in the limit of small η).

Let me combine these ideas:

**Algorithm: Multiplicative Weight Updates with Soft Active Set**

1. f = initial sequence (sum to 1)
2. Repeat:
   a. conv = f * f
   b. M = max(conv)
   c. Compute weights: w_k = exp((conv[k] - M) / τ) where τ is a temperature
   d. h_j = sum_k w_k * f[k-j] (cross-correlation)
   e. Update: f_j <- f_j * exp(-η * h_j) / Z (or f_j <- f_j / (1 + η * h_j) for a rational approximation)
   f. Normalize
3. Return best f found

The temperature τ controls how "soft" the active set is. Small τ -> only the most active constraints. Large τ -> all constraints weighted.

The step size η controls how aggressive the update is.

This is essentially mirror descent with a smoothed subgradient, which should have good convergence properties.

            w /= ws
        
        h = compute_h_fft(w, f)
        
        h_max = h.max()
        if h_max < 1e-15:
            break
        
        f_new = f * np.exp(-eta * h / h_max)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f_new = f_new / sn
        
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        if it % 50 == 0:
            tau *= 0.97
            eta *= 0.998
            if tau < 0.001:
                tau = 0.001 + 0.001 * (it % 10)
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            best_obj_check = conv.max()
            if best_obj_check < best_obj:
                best_obj = best_obj_check
                best_f = f.copy()
            no_improve = 0
            tau = max(tau, 0.02)
            eta = min(eta * 1.2, 2.0)
        
        if verbose and it % 500 == 0:
            print(f"    MD iter {it}: obj={obj:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def nesterov_gd(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient descent with softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Initialize x such that softmax(x) = f
    x = np.log(f + 1e-30)
    x -= x.mean()
    
    best_obj = fast_convolve(softmax_stable(x)).max()
    best_x = x.copy()
    
    t0 = time.monotonic()
    it = 0
    eta = 0.1
    eps = 0.05
    momentum = 0.9
    y = x.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f_y = softmax_stable(y)
        conv = fast_convolve(f_y)
        
        shifted = (conv - conv.max()) / eps
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        h = compute_h_fft(w, f_y)
        grad_f = 2.0 * h
        dot = np.dot(f_y, grad_f)
        grad_x = f_y * (grad_f - dot)
        
        y = y + momentum * (x - y + eta * grad_x)
        # Actually standard Nesterov: y = x + beta*(x - x_prev) - eta*grad
        # Let's use a simpler momentum approach
        y = x + momentum * (x - y + eta * grad_x)
        # Hmm, let me use the standard form:
        # v = momentum * v - eta * grad
        # x = x + v
        # y = x + momentum * v (lookahead)
        
        x_new = x + momentum * (x - y) - eta * grad_x
        y_new = x_new + momentum * (x_new - x)
        
        # Actually let me just use plain momentum:
        # v = momentum * v + grad
        # x = x - eta * v
        
        x = x_new
        y = y_new
        
        # Evaluate
        f_x = softmax_stable(x)
        conv_x = fast_convolve(f_x)
        obj = conv_x.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_x = x.copy()
        
        # Adapt
        if it % 200 == 0:
            eta *= 0.995
            eps *= 0.99
            if eps < 0.005:
                eps = 0.005
        
        if verbose and it % 1000 == 0:
            print(f"    NGD iter {it}: obj={obj:.10f}, best={best_obj:.10f}")
    
    return best_x, best_obj


def momentum_gd(f0, time_limit, verbose=False):
    """Momentum gradient descent with softmax parameterization and adaptive step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    x = np.log(f + 1e-30)
    x -= x.mean()
    v = np.zeros(n)
    
    best_obj = fast_convolve(softmax_stable(x)).max()
    best_x = x.copy()
    
    t0 = time.monotonic()
    it = 0
    eta = 0.05
    beta = 0.9
    eps = 0.03
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f_y = softmax_stable(x)
        conv = fast_convolve(f_y)
        M = conv.max()
        
        shifted = (conv - M) / eps
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        h = compute_h_fft(w, f_y)
        grad_f = 2.0 * h
        dot = np.dot(f_y, grad_f)
        grad_x = f_y * (grad_f - dot)
        
        v = beta * v + grad_x
        x = x - eta * v
        
        f_x = softmax_stable(x)
        conv_x = fast_convolve(f_x)
        obj = conv_x.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            x = best_x + rng.standard_normal(n) * 0.01
            v = np.zeros(n)
            no_improve = 0
            eta = max(eta * 0.5, 0.005)
            eps = max(eps * 0.7, 0.005)
        
        if it % 500 == 0:
            eta *= 0.997
            eps = max(eps * 0.995, 0.003)
        
        if verbose and it % 2000 == 0:
            print(f"    MoGD iter {it}: obj={obj:.10f}, best={best_obj:.10f}, eta={eta:.4f}")
    
    return best_x, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False, top_k=300):
    """LP-based Frank-Wolfe with reduced constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = lp_direction_reduced(f, M * m_exp, n, top_k=top_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 80:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 80, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def fine_subgradient(f0, time_limit, lr=0.002, mom=0.93, top_k=100, temp=0.0001):
    """Fine subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        cur_temp *= 1.002
        
        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining mirror descent, LP-FW, momentum GD,
    coordinate descent, and subgradient polishing with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    parent = get_parent()
    
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = parent.copy()
        best_score = p_score
    else:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    n0 = len(parent)
    
    # === Phase 1: Mirror descent (fast, many iterations) ===
    t_remaining = deadline - time.monotonic()
    t1 = min(350.0, t_remaining * 0.25)
    print(f"\n=== Phase 1: Mirror Descent (t={t1:.0f}s) ===")
    if t1 > 20:
        a_opt, _ = mirror_descent(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "mirror")
    
    # === Phase 2: Momentum GD with softmax parameterization ===
    base = best_seq if best_seq is not None else parent
    t_remaining = deadline - time.monotonic()
    t2 = min(450.0, t_remaining * 0.30)
    print(f"\n=== Phase 2: Momentum GD (t={t2:.0f}s) ===")
    if t2 > 20:
        x_opt, _ = momentum_gd(base, time_limit=t2, verbose=True)
        f_opt = softmax_stable(x_opt)
        try_seq(f_opt, "momgd")
    
    # === Phase 3: LP-FW on best ===
    base = best_seq if best_seq is not None else parent
    t_remaining = deadline - time.monotonic()
    t3 = min(400.0, t_remaining * 0.30)
    print(f"\n=== Phase 3: LP-FW (t={t3:.0f}s) ===")
    if t3 > 20:
        n_best = len(base)
        # Run LP-FW with multiple top_k values
        t3a = t3 * 0.6
        a_opt, _ = lp_fw_optimize(base, time_limit=t3a, verbose=True, top_k=300)
        try_seq(a_opt, "lpfw_300")
        
        t_remaining = deadline - time.monotonic()
        if t_remaining > 30:
            t3b = min(t3a * 0.5, t_remaining * 0.5)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3b, verbose=True, top_k=500)
            try_seq(a_opt, "lpfw_500")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    lengths_to_try = []
    n_best = len(base)
    for delta in [-64, -32, 32, 64, 128]:
        new_n = n_best + delta
        if 256 <= new_n <= 2048:
            lengths_to_try.append(new_n)
    
    t_remaining = deadline - time.monotonic()
    per_length = min(80.0, t_remaining / max(1, len(lengths_to_try) + 1))
    
    for target_n in lengths_to_try:
        t_remaining = deadline - time.monotonic()
        if t_remaining < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        # Quick mirror descent
        t_alloc = min(per_length, t_remaining * 0.4)
        if t_alloc > 10:
            a_opt, _ = mirror_descent(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"md_n{target_n}")
        
        t_remaining = deadline - time.monotonic()
        if t_remaining > 30 and best_seq is not None and len(best_seq) != target_n:
            # If this length improved, do LP-FW on it
            pass
    
    # === Phase 5: Coordinate descent on best ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        t_remaining = deadline - time.monotonic()
        t5 = min(150.0, t_remaining * 0.3)
        if t5 > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=t5)

    # === Phase 6: Fine subgradient polishing ===

                                         lr=0.0015, mom=0.94,

    # === Phase 7: Final LP-FW + perturbations ===
    print(f"\n=== Phase 7: Final Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        t_remaining = deadline - time.monotonic()

            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t7, verbose=True, top_k=200)
            try_seq(a_opt, "final_lpfw")
        
        # Tiny perturbations with quick polish
        n_best = len(best_seq)
        t_remaining = deadline - time.monotonic()
        for p in range(5):
            t_remaining = deadline - time.monotonic()
            if t_remaining < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(15.0, t_remaining * 0.3)
            if t_alloc > 3:
                a_opt, _ = fine_subgradient(f_pert, time_limit=t_alloc,
                                             lr=0.001, mom=0.95,
                                             top_k=max(20, n_best // 4),
                                             temp=0.00003)
                try_seq(a_opt, f"perturb_{p}")
        
        # Final mirror descent on best
        t_remaining = deadline - time.monotonic()
        if t_remaining > 15:
            a_opt, _ = mirror_descent(best_seq, time_limit=min(30.0, t_remaining * 0.5))
            try_seq(a_opt, "final_md")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
