"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with incremental convolution updates, multiple random restarts, and
adaptive shuffling to escape local minima. Key innovation: exact 1D
subproblem solution via bisection on the convex subgradient, combined
with O(n) incremental convolution maintenance per coordinate step.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def solve_1d_subproblem(f, g, quad_idx):
    """
    Minimize max_k(f[k] + g[k]*x + (x^2 if k==quad_idx else 0)) over x >= 0.
    f, g: arrays of length (2n-1). quad_idx: index of quadratic term or -1.
    Returns the optimal x.
    """
    n = len(f)

    def obj(x):
        vals = f + g * x
        if quad_idx >= 0:
            vals = vals.copy()
            vals[quad_idx] += x * x
        return np.max(vals)

    # Check if x=0 is optimal
    max_f = np.max(f)
    active_at_0 = np.where(f >= max_f - 1e-13)[0]
    active_g = g[active_at_0]
    # If quad is active at x=0, its derivative is 0
    quad_active_at_0 = (quad_idx >= 0 and f[quad_idx] >= max_f - 1e-13)
    all_nonpos = np.all(active_g <= 1e-12)
    if all_nonpos:
        return 0.0

    # Find upper bound
    x_hi = max(1.0, 2.0 * max_f / (np.max(np.abs(g)) + 1e-15))
    x_hi = min(x_hi, 10.0)

    res = minimize_scalar(obj, bounds=(0.0, x_hi), method='bounded',
                          options={'xatol': 1e-15})
    return max(0.0, res.x)


def coord_descent(a_init, max_time, seed=42, shuffle=True, verbose=False):
    """
    Coordinate descent minimizing max_conv(a) subject to a>=0, sum(a)=1.
    Uses exact 1D subproblem solving with incremental convolution updates.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve_passes = 0
    order = np.arange(n)

    pass_num = 0
    while time.monotonic() - t0 < max_time:
        pass_num += 1

        if shuffle:
            if pass_num % 5 == 0:
                rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > max_time:
                break

            old_ai = a[i]

            # Compute g vector: g[i+j] = 2*a[j] for j in [0,n)
            g = np.zeros(2 * n - 1)
            g[i:i + n] = 2.0 * a

            # Compute f vector: f = conv - old_ai * g, then fix quad term
            f = conv - old_ai * g
            quad_idx = 2 * i
            if quad_idx < 2 * n - 1:
                f[quad_idx] = conv[quad_idx] - old_ai * old_ai
                g[quad_idx] = 0.0
            else:
                quad_idx = -1

            # Solve 1D subproblem
            new_ai = solve_1d_subproblem(f, g, quad_idx)

            delta = new_ai - old_ai
            if abs(delta) > 1e-16:
                # Incremental convolution update
                conv[i:i + n] += 2.0 * delta * a
                if quad_idx >= 0:
                    conv[quad_idx] -= delta * delta
                a[i] = new_ai

        # End of pass: recompute and renormalize
        conv = fast_convolve(a)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1

        if verbose and pass_num % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pass {pass_num}, obj={obj:.10f}, elapsed={elapsed:.1f}s")

        if no_improve_passes > 30:
            break

    return best_a, best_obj

    """Coordinate descent with periodic random restarts."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a, best_obj = coord_descent(a, max_time / (n_restarts + 1), seed=seed, verbose=verbose)

    for r in range(n_restarts):
        remaining = max_time - (r + 1) * (max_time / (n_restarts + 1)) * (r + 1) / (r + 1)
        elapsed = (r + 1) * (max_time / (n_restarts + 1))
        rem = max_time - elapsed
        if rem < 5:
            break

        # Perturb the best solution
        noise_scale = 0.01 + 0.01 * r
        a_pert = best_a + rng.standard_normal(n) * noise_scale
        a_pert = np.maximum(a_pert, 0.0)
        ps = a_pert.sum()
        if ps < 1e-15:
            a_pert = best_a.copy()
        else:
            a_pert = a_pert / ps

        t_chunk = min(rem * 0.7, max_time / (n_restarts + 1))
        a_opt, obj = coord_descent(a_pert, t_chunk, seed=seed + 1000 * (r + 1))
        if obj < best_obj:
            best_a = a_opt
            best_obj = obj
            if verbose:
                print(f"    Restart {r}: improved to {obj:.10f}")

    return best_a, best_obj


def block_coord_descent(a_init, max_time, block_size=4, seed=42, verbose=False):
    """
    Block coordinate descent: optimize small blocks of coordinates simultaneously.
    Uses 2D or small-dim optimization for each block.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    pass_num = 0

    while time.monotonic() - t0 < max_time:
        pass_num += 1
        improved = False

        # Process blocks
        for start in range(0, n - block_size + 1, block_size):
            if time.monotonic() - t0 > max_time:
                break

            idx = np.arange(start, start + block_size)
            a_block = a[idx].copy()
            old_block = a_block.copy()

            # Compute the part of conv that depends on the block
            # For simplicity, do a few coordinate descent steps within the block
            for it in range(3):
                for ii in range(block_size):
                    i = idx[ii]
                    old_ai = a[i]

                    g = np.zeros(2 * n - 1)
                    g[i:i + n] = 2.0 * a
                    f = conv - old_ai * g
                    quad_idx = 2 * i
                    if quad_idx < 2 * n - 1:
                        f[quad_idx] = conv[quad_idx] - old_ai * old_ai
                        g[quad_idx] = 0.0
                    else:
                        quad_idx = -1

                    new_ai = solve_1d_subproblem(f, g, quad_idx)
                    delta = new_ai - old_ai
                    if abs(delta) > 1e-16:
                        conv[i:i + n] += 2.0 * delta * a
                        if quad_idx >= 0:
                            conv[quad_idx] -= delta * delta
                        a[i] = new_ai
                        improved = True

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if not improved and pass_num > 5:
            break

    return best_a, best_obj


def generate_diverse_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(w / w.sum())

    # Gaussian centered
    mu = 0.5 * (n - 1)
    sigma = 0.25 * n
    g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c / c.sum())

    # Two bumps
    b = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
        np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(b / b.sum())

    # Random exponential
    e = rng.exponential(1.0, size=n)
    inits.append(e / e.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def peak_spread_init(n, rng):
    """
    Initialize by trying to make the auto-convolution as flat as possible.
    Start from uniform and do a few rounds of peak reduction.
    """
    a = np.ones(n) / n
    conv = fast_convolve(a)

    # Identify peaks and try to spread
    for _ in range(20):
        # Find peak location
        k_peak = np.argmax(conv)
        # The peak at k_peak is sum of a[i]*a[k_peak-i]
        # Reduce the largest contributing pair
        contribs = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contribs[i] += a[i] * a[j]
        # Reduce the largest contributor
        i_max = np.argmax(contribs)
        # Move mass from i_max to a random other position
        i_dest = rng.integers(0, n)
        if i_dest == i_max:
            i_dest = (i_max + n // 2) % n
        move = 0.1 * a[i_max]
        a[i_max] -= move
        a[i_dest] += move
        a = np.maximum(a, 0)
        a = a / a.sum()
        conv = fast_convolve(a)

    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-strategy search:
    1. Generate diverse initializations
    2. Quick coordinate descent on each
    3. Select best, do deep coordinate descent
    4. Block coordinate descent refinement
    5. Random restarts with perturbation
    6. Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p.copy()
            # Normalize to sum=1 for optimization
            p_norm = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, norm_score={evaluate_sequence(p_norm.tolist()):.8f}")
            try_seq(parent, "parent_raw")
            try_seq(p_norm, "parent_norm")
    except Exception as e:
        print(f"Could not load parent: {e}")

    n = 384
    if parent is not None and len(parent) in [256, 384, 512, 640]:
        n = len(parent)

    # === Phase 1: Quick exploration (~200s) ===
    print(f"\n=== Phase 1: Quick Exploration (n={n}) ===")
    t_phase1 = min(200, (deadline - time.monotonic()) * 0.15)

    inits = generate_diverse_inits(n, parent if parent is not None else None, rng)
    inits.append(peak_spread_init(n, rng))

    t_per_init = t_phase1 / len(inits)
    for i, init in enumerate(inits):
        rem = deadline - time.monotonic() - (len(inits) - i - 1) * 5
        if rem < 5:
            break
        t_alloc = min(t_per_init, rem)
        a_opt, _ = coord_descent(init, t_alloc, seed=seed + i, shuffle=True)
        try_seq(a_opt, f"init_{i}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep coordinate descent (~600s) ===
    print(f"\n=== Phase 2: Deep Coordinate Descent ===")
    if best_seq is not None:
        t_phase2 = min(600, (deadline - time.monotonic()) * 0.45)
        # Use normalized version
        a_work = best_seq / best_seq.sum()
        a_opt, obj = coord_descent(a_work, t_phase2, seed=seed + 100, shuffle=True)
        try_seq(a_opt, "deep_cd")

        # Try from parent with longer time
        rem = deadline - time.monotonic() - 600
        if rem > 30 and parent is not None:
            p_norm = parent / parent.sum()
            if len(p_norm) == n:
                t_alloc = min(rem * 0.5, 100)
                a_opt2, _ = coord_descent(p_norm, t_alloc, seed=seed + 200, shuffle=True)
                try_seq(a_opt2, "parent_cd")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Random restarts + CD (~400s) ===
    print(f"\n=== Phase 3: Random Restarts ===")
    if best_seq is not None:
        t_phase3 = min(400, (deadline - time.monotonic()) * 0.35)
        a_work = best_seq / best_seq.sum()
        a_opt, _ = coord_descent_with_restarts(
            a_work, t_phase3, n_restarts=4, seed=seed + 300, verbose=False
        )
        try_seq(a_opt, "restarts")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Block coordinate descent (~250s) ===
    print(f"\n=== Phase 4: Block CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_phase4 = min(250, rem)
            a_work = best_seq / best_seq.sum()
            a_opt, _ = block_coord_descent(a_work, t_phase4, block_size=3, seed=seed + 400)
            try_seq(a_opt, "block_cd")

            # Try with different block size
            rem = deadline - time.monotonic() - 50
            if rem > 20:
                a_opt2, _ = block_coord_descent(a_work, min(rem, 100), block_size=5, seed=seed + 500)
                try_seq(a_opt2, "block_cd5")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Fine polishing (~150s) ===
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_work = best_seq / best_seq.sum()
            # Very careful coordinate descent with no shuffle (deterministic)
            t_polish = min(rem * 0.6, 100)
            a_opt, _ = coord_descent(a_work, t_polish, seed=seed + 600, shuffle=False)
            try_seq(a_opt, "polish_noshuffle")

            # Shuffled polish
            rem = deadline - time.monotonic() - 15
            if rem > 10:
                a_opt2, _ = coord_descent(a_work, min(rem, 60), seed=seed + 700, shuffle=True)
                try_seq(a_opt2, "polish_shuffle")

            # Tiny perturbation + CD
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                for attempt in range(3):
                    rem = deadline - time.monotonic() - 5
                    if rem < 8:
                        break
                    noise = rng.standard_normal(n) * 0.003
                    a_pert = np.maximum(a_work + noise, 0)
                    ps = a_pert.sum()
                    if ps > 1e-15:
                        a_pert = a_pert / ps
                        a_opt3, _ = coord_descent(a_pert, min(rem, 30), seed=seed + 800 + attempt)
                        try_seq(a_opt3, f"perturb_{attempt}")

    # === Phase 6: Try different lengths if time allows ===
    print(f"\n=== Phase 6: Different Lengths ===")
    rem = deadline - time.monotonic() - 10
    if rem > 60:
        for n_try in [512, 256, 640]:
            rem = deadline - time.monotonic() - 5
            if rem < 20:
                break
            # Interpolate parent to new length or use uniform
            if parent is not None:
                px = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, len(parent)), parent)
                px = np.maximum(px, 0)
                ps = px.sum()
                if ps > 1e-10:
                    px = px / ps
            else:
                px = np.ones(n_try) / n_try

            t_alloc = min(rem * 0.5, 50)
            a_opt, _ = coord_descent(px, t_alloc, seed=seed + 900 + n_try, shuffle=True)
            try_seq(a_opt, f"n{n_try}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
