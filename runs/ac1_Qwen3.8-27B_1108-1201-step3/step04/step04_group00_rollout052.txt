from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)

    for j in range(n):

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def nesterov_sg_polish(f0, time_limit, lr=0.002, mom=0.9, top_k=30, temp=0.00005):

        k_actual = min(top_k, len(conv))

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:

            w[np.argmax(pv)] = 1.0

            if w[i] < 1e-13:

        f_new = project_simplex(y_new + v)

        beta = beta_new
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

            best_obj = co

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.12, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7 + n)
            scale = 0.002 + 0.003 * min(no_improve // 80, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def nesterov_sg_polish(f0, time_limit, lr=0.002, mom=0.9, top_k=30, temp=0.00005):
    """Nesterov-accelerated subgradient polishing with softmax peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    beta = mom
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = y + (beta / beta_new) * (y - f)
        v = mom * v - cur_lr * sg
        f_new = project_simplex(y_new + v)

        y = y_new
        beta = beta_new
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.99985
        cur_temp *= 1.002

        if no_improve > 250:
            rng = np.random.default_rng(it * 100003 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            beta = mom
            cur_lr = max(lr * 0.5, 0.0003)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def peak_targeted_refine(f0, time_limit, n_peaks=5, lr_decay=0.999):
    """Target the top peaks individually with small gradient steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        cur_peak = conv.max()

        if cur_peak < best_obj - 1e-14:
            best_obj = cur_peak
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 100:
            break

        # Get top peaks
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        # For each peak, take a small negative gradient step
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad = np.zeros(n)
            grad[idx] = 2.0 * f[k - idx]

            # Try multiple step sizes
            best_f_step = f.copy()
            best_val_step = cur_peak
            for alpha in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
                ft = project_simplex(f - alpha * grad)
                val = fast_convolve(ft).max()
                if val < best_val_step:
                    best_val_step = val
                    best_f_step = ft

            if best_val_step < cur_peak - 1e-15:
                f = best_f_step.copy()
            else:
                f = best_f_step.copy()

        lr *= lr_decay

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - n / 2.0) / sigma) ** 2)
    g /= g.sum()
    inits.append(g)

    # Skewed (front-loaded)
    skew = np.exp(-x / (n * 0.3))
    skew /= skew.sum()
    inits.append(skew)

    # Skewed (back-loaded)
    skew2 = np.exp(-(n - 1 - x) / (n * 0.3))
    skew2 /= skew2.sum()
    inits.append(skew2)

    # Bimodal (peaks at 1/4 and 3/4)
    bim = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    bim /= bim.sum()
    inits.append(bim)

    # Sine
    sine = np.abs(np.sin(np.pi * (x + 0.5) / n)) + 0.05
    sine /= sine.sum()
    inits.append(sine)

    # Random
    for _ in range(2):
        r = rng.random(n)
        r /= r.sum()
        inits.append(r)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: enhanced LP-FW, Nesterov SG, peak-targeted
    refinement, multi-length exploration, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    # === Phase 1: Long LP-FW on parent ===
    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(250, deadline - time.monotonic() - 500)
    if t1 > 40:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Nesterov SG + Peak refine ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            print(f"\n=== Phase 2: SG + Peak refine (best={best_score:.6f}) ===")
            t_sg = min(rem * 0.4, 100)
            a_opt, _ = nesterov_sg_polish(best_seq, time_limit=t_sg,
                                          lr=0.003, mom=0.9,
                                          top_k=max(20, len(best_seq) // 8),
                                          temp=0.00004)
            try_seq(a_opt, "sg1")

            rem = deadline - time.monotonic() - 350
            if rem > 20:
                a_opt, _ = peak_targeted_refine(best_seq, time_limit=min(rem * 0.4, 60),
                                                n_peaks=5, lr_decay=0.998)
                try_seq(a_opt, "peak_refine")

            rem = deadline - time.monotonic() - 300
            if rem > 15:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.5, 60))
                try_seq(a_opt, "lpfw_p2")

    # === Phase 3: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
        base = best_seq
        base_n = len(base)

        candidate_lengths = set()
        for factor in [0.75, 1.0, 1.25, 1.5, 0.6]:
            ln = int(base_n * factor)
            if 64 <= ln <= 2048:
                candidate_lengths.add(ln)
        for ln in [512, 768, 1024, 1280, 1536]:
            if 64 <= ln <= 2048:
                candidate_lengths.add(ln)

        # Sort by priority: closer to base_n first, then longer
        candidate_lengths = sorted(candidate_lengths,
                                   key=lambda x: (abs(x - base_n), -x))

        for target_n in candidate_lengths[:3]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            if target_n == len(base):
                continue

            f_interp = interpolate_sequence(base, target_n)
            t_alloc = min(rem * 0.45, 120)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"lpfw_n{target_n}")

            rem = deadline - time.monotonic() - 120
            if rem > 15:
                a_opt, _ = nesterov_sg_polish(a_opt, time_limit=min(rem * 0.4, 50),
                                              lr=0.002, mom=0.9,
                                              top_k=max(15, target_n // 10),
                                              temp=0.00003)
                try_seq(a_opt, f"sg_n{target_n}")

    # === Phase 4: Diverse starts at best length ===
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60

        if rem > 30:
            print(f"\n=== Phase 4: Diverse restarts n={n_best} (best={best_score:.6f}) ===")
            inits = generate_diverse_inits(n_best, rng)

            for i, init in enumerate(inits[:5]):
                rem = deadline - time.monotonic() - 40
                if rem < 15:
                    break

                # Blend with best: vary the blend ratio
                ratios = [0.7, 0.5, 0.3]
                ratio = ratios[i % len(ratios)]
                blend = ratio * best_seq + (1.0 - ratio) * init
                blend = project_simplex(blend)

                t_alloc = min(rem * 0.3, 50)
                a_opt, _ = lp_fw_optimize(blend, time_limit=t_alloc)
                try_seq(a_opt, f"diverse_{i}_r{ratio}")

                rem = deadline - time.monotonic() - 35
                if rem > 10:
                    a_opt, _ = nesterov_sg_polish(a_opt, time_limit=min(rem * 0.5, 25),
                                                  lr=0.0015, mom=0.9,
                                                  top_k=max(10, n_best // 8),
                                                  temp=0.00003)
                    try_seq(a_opt, f"sg_div_{i}")

    # === Phase 5: Final deep optimization ===
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20

        print(f"\n=== Phase 5: Final deep opt (best={best_score:.6f}) ===")

        if rem > 15:
            t_sg = min(rem * 0.4, 60)
            a_opt, _ = nesterov_sg_polish(best_seq, time_limit=t_sg,
                                          lr=0.001, mom=0.92,
                                          top_k=max(10, n_best // 6),
                                          temp=0.00002)
            try_seq(a_opt, "final_sg")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = peak_targeted_refine(best_seq, time_limit=min(rem * 0.5, 30),
                                            n_peaks=8, lr_decay=0.999)
            try_seq(a_opt, "final_peak")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.6, 30))
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations
        for _ in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_sg_polish(f_pert, time_limit=min(rem - 3, 15),
                                          lr=0.0008, mom=0.93,
                                          top_k=max(10, n_best // 6),
                                          temp=0.00001)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
