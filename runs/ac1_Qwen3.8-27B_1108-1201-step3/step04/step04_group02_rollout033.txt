def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

def compute_sg(f, conv, n_peaks=64, temp=0.005):
    """Subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if n_peaks >= len(conv):
        top = np.arange(len(conv))
    else:
        top = np.argpartition(conv, -n_peaks)[-n_peaks:]
    vals = conv[top] / temp
    vals -= vals.max()
    w = np.exp(vals)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(top))
        w[np.argmax(conv[top])] = 1.0
    else:
        w /= ws

def lp_oracle(f, M, timeout=20):
    """LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def spectral_init(n, seed=0):
    """Generate a non-negative sequence from random phase (flat spectrum init)."""
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0, 2 * np.pi, n)
    signal = np.ones(n) * np.exp(1j * phases)
    a = np.abs(np.fft.irfft(signal, n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    r = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(r / r.sum())
    e = np.exp(-x / (n / 2.5))
    inits.append(e / e.sum())
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.02
    inits.append(c / c.sum())
    s = np.sin(np.pi * (x + 1) / (n + 1)) + 0.1
    inits.append(s / s.sum())
    # Spectral
    for sd in range(3):
        inits.append(spectral_init(n, seed=sd * 100 + 7))
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            inits.append(px)
            inits.append(project_simplex(0.7 * px + 0.3 * np.ones(n) / n))
            inits.append(project_simplex(0.5 * px + 0.5 * (r / r.sum())))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining subgradient descent,
    LP Frank-Wolfe, and fine polishing with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Optimize parent heavily ===
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t_avail = deadline - time.monotonic() - 300
        t_fw = min(200, t_avail * 0.3)
        t_sg = min(150, t_avail * 0.25)

        if t_fw > 20:
            print(f"  FW-LP on parent n={n_p}, t={t_fw:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t_fw)
            try_seq(a_opt, "parent_fw")

        if best_seq is not None and len(best_seq) == n_p:
            rem = deadline - time.monotonic() - 250
            if rem > 20:
                print(f"  SG on parent, t={min(rem, t_sg):.0f}s")
                a_opt2, _ = sg_optimize(best_seq, time_limit=min(rem, t_sg),
                                        lr0=0.005, n_pk=n_p // 6, temp0=0.004)
                try_seq(a_opt2, "parent_sg")

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length Exploration ===")
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for ln in [256, 384, 512, 768]:
        if ln not in lengths:
            lengths.append(ln)

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        if parent is not None and n == len(parent):
            continue

        inits = structured_inits(n, rng, parent)
        rem = deadline - time.monotonic() - 150
        t_per = max(15, min(40, rem / (len(inits) * 2)))

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 130:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break
            a_opt, _ = sg_optimize(f_init, time_limit=t_alloc,
                                   lr0=0.01, n_pk=max(32, n // 10), temp0=0.01)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # FW-LP deep
        for p in range(5):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (5 - p), 100)
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif p == 2:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (best_seq[::-1]))
            elif p == 3:
                f_init = spectral_init(n_best, seed=p * 50 + 3)
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)

            if rem > 50:
                a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
                try_seq(a_opt, f"deep_fw_{p}")

        # SG deep
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem // (4 - p), 60)
            if p == 0:
                f_init = best_seq.copy()
                cfg = {"lr0": 0.003, "n_pk": n_best // 5, "temp0": 0.002}
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
                cfg = {"lr0": 0.006, "n_pk": n_best // 4, "temp0": 0.005}
            elif p == 2:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best))
                cfg = {"lr0": 0.002, "n_pk": n_best // 6, "temp0": 0.001}
            else:
                f_init = spectral_init(n_best, seed=99)
                cfg = {"lr0": 0.008, "n_pk": n_best // 4, "temp0": 0.008}

            a_opt, _ = sg_optimize(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{p}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = fine_polish(best_seq, time_limit=min(rem, 50),
                                   lr=0.0008, mom=0.96, n_pk=n_best // 3, temp=0.0003)
            try_seq(a_opt, "polish1")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "polish_fw")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = fine_polish(best_seq, time_limit=min(rem, 30),
                                   lr=0.0003, mom=0.97, n_pk=n_best // 2, temp=0.0001)
            try_seq(a_opt, "polish2")

        # Micro perturbations
        rem = deadline - time.monotonic() - 8
        for _ in range(3):
            if rem < 5:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            a_opt, _ = fine_polish(f_pert, time_limit=min(rem, 10),
                                   lr=0.0004, mom=0.97, n_pk=n_best // 3, temp=0.0001)
            try_seq(a_opt, "micro")
            rem = deadline - time.monotonic() - 6

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
