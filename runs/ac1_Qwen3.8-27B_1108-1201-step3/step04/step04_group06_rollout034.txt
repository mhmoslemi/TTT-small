Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with softmax parameterization
and temperature annealing, combined with EG polishing and local search.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def compute_smooth_obj_grad(z, T):
    """Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. z.
    
    a = softmax(z), b = conv(a, a)
    objective = logsumexp(T*b) / T
    """
    n = len(z)
    a = softmax(z)
    
    # Compute b = conv(a, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)[:size]
    
    # logsumexp(T*b) / T
    log_sum_exp = logsumexp(T * b)
    obj = log_sum_exp / T
    
    # Softmax weights
    w = np.exp(T * b - log_sum_exp)
    
    # h_j = sum_k w_k * a_{k-j}
    # h = conv(w, a_reversed)[n-1 : 2*n-1]
    w_fft = np.fft.rfft(w, fft_size * 2)  # need larger FFT for this convolution
    a_rev_fft = np.fft.rfft(a[::-1], fft_size * 2)
    conv_w_ar = np.fft.irfft(w_fft * a_rev_fft, fft_size * 2)[:size + n - 1]
    h = conv_w_ar[n-1 : 2*n-1]
    
    # m = w · b
    m = np.dot(w, b)
    
    # Gradient = 2*a*(h - m)
    grad = 2.0 * a * (h - m)
    
    return obj, grad


def lbfqg_optimize(a0, time_limit=300, T_initial=10, T_final=0.1, n_stages=5):
    """Optimize using L-BFGS with temperature annealing."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    # Initialize z from a: z = log(a) + small noise (for a_i > 0)
    z = np.zeros(n)
    positive = a > 1e-10
    z[positive] = np.log(a[positive])
    z[~positive] = -10.0
    
    best_obj = np.inf
    best_a = a.copy()
    
    t0 = time.monotonic()
    T_schedule = np.geomspace(T_initial, T_final, n_stages)
    
    for si, T in enumerate(T_schedule):
        stage_time = min(60, time_limit - (time.monotonic() - t0))
        if stage_time < 5:
            break
        
        def obj_fun(z_in):
            obj, _ = compute_smooth_obj_grad(z_in, T)
            return obj
        
        def grad_fun(z_in):
            _, grad = compute_smooth_obj_grad(z_in, T)
            return grad
        
        # Use L-BFGS-B
        try:
            result = minimize(obj_fun, z, method='L-BFGS-B', jac=grad_fun,
                             options={'maxiter': 500, 'maxfun': 2000,
                                     'gtol': 1e-8, 'iprint': 0})
            
            a_new = softmax(result.x)
            # Evaluate true max
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            A = np.fft.rfft(a_new, fft_size)
            b_new = np.fft.irfft(A * A, fft_size)[:size]
            true_obj = b_new.max()
            
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_new.copy()
            
            z = result.x
            print(f"    L-BFGS stage {si+1}/{n_stages} (T={T:.3f}): "
                  f"smooth={result.fun:.8f}, true_max={true_obj:.8f}")
        except Exception as e:
            print(f"    L-BFGS stage {si+1} failed: {e}")
    
    return best_a.copy(), best_obj


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                temp_decay=0.9997, eta_decay=0.99995, top_k_frac=0.25):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_eg_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    
    return sg

        # Try random coordinate pairs

                j = (j + 1) % n

                    best_a = a_try.copy()
                    improved = True

                break

        else:
            no_improve += 1

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with softmax parameterization
and temperature annealing as the core optimizer, combined with exponentiated gradient
(EG) polishing, local search, and multi-length exploration. The key innovation is the
use of L-BFGS on the log-softmax parameterization with analytic gradients computed
via FFT, which leverages curvature information for faster convergence than pure
first-order methods.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_stable(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_smooth_obj_grad(z, T):
    """Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. z.
    
    a = softmax(z), b = conv(a, a)
    objective = logsumexp(T*b) / T
    gradient = 2 * a * (h - m) where h_j = sum_k w_k * a_{k-j}, m = w·b
    """
    n = len(z)
    a = softmax_stable(z)
    
    # Compute b = conv(a, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)[:size]
    
    # logsumexp(T*b) / T
    lse = logsumexp(T * b)
    obj = lse / T
    
    # Softmax weights w_k = exp(T*b_k) / sum(exp(T*b))
    w = np.exp(T * b - lse)
    
    # h_j = sum_k w_k * a_{k-j}
    # h[n-1:2n-1] of conv(w, a_reversed)
    a_rev = a[::-1]
    conv_size = size + n - 1  # (2n-1) + n - 1 = 3n-2
    fft_size2 = 1 << (conv_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size2)
    AR = np.fft.rfft(a_rev, fft_size2)
    conv_w_ar = np.fft.irfft(W * AR, fft_size2)[:conv_size]
    h = conv_w_ar[n - 1:2 * n - 1]
    
    # m = w · b (weighted average of b)
    m = np.dot(w, b)
    
    # Gradient = 2*a*(h - m)
    grad = 2.0 * a * (h - m)
    
    return obj, grad

    if s < 1e-15:

        s = 1.0
    a = a / s

    z = np.zeros(n)
    positive = a > 1e-12
    z[positive] = np.log(a[positive])
    z[~positive] = -20.0
    
    best_obj = np.inf
    best_a = a.copy()

        if stage_budget < 3:

            val, _ = compute_smooth_obj_grad(z_in, T)
            return val

            _, g = compute_smooth_obj_grad(z_in, T)
            return g

def compute_eg_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    
    return sg


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                temp_decay=0.9997, eta_decay=0.99995, top_k_frac=0.25, verbose=False):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = eta0
    temp = temp0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_eg_subgradient(a, c, temp, top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay
        
        if no_improve > 8000:
            break
        
        # Periodic stagnation restart
        if no_improve > 3000:
            rng_local = np.random.default_rng(it * 13 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            eta = max(eta, eta0 * 0.05)
            temp = max(temp, 0.001)
    
    return best_a.copy(), best_obj


def local_search(a0, time_limit, rng, step_sizes=None):
    """Local search with multiplicative perturbations on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    if step_sizes is None:
        step_sizes = [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 80:
        improved = False
        n_perturb = max(5, n // 20)
        coords = rng.choice(n, size=min(n_perturb, n), replace=False)
        
        for i in coords:
            if a[i] < 1e-12:
                continue
            
            for factor in step_sizes:
                a_try = best_a.copy()
                a_try[i] *= factor
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = fast_convolve(a_try)
                if c_try.max() < best_obj - 1e-13:
                    best_obj = c_try.max()
                    best_a = a_try.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a.copy(), best_obj


def coordinate_descent(a0, time_limit, rng, n_sweeps=10):
    """Cyclic coordinate descent trying multiplicative changes on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    factors = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if best_a[i] < 1e-12:
                continue
            
            best_factor = None
            best_c = None
            for f in factors:
                a_try = best_a.copy()
                a_try[i] *= f
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = fast_convolve(a_try)
                if c_try.max() < best_obj - 1e-14:
                    best_obj = c_try.max()
                    best_a = a_try.copy()
                    best_factor = f
                    break
    
    return best_a.copy(), best_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xc = (x + 0.5) / n  # centered in [0, 1]
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * xc))
    inits.append(rc / rc.sum())
    
    # Parabolic (edge-concentrated)
    par = (xc - 0.5) ** 2
    inits.append(par / par.sum())
    
    # Gaussian center
    g = np.exp(-20 * (xc - 0.5) ** 2)
    inits.append(g / g.sum())
    
    # Bipolar (peaks at 1/4 and 3/4)
    bp = np.exp(-50 * ((xc - 0.25) ** 2 + (xc - 0.75) ** 2))
    inits.append(bp / bp.sum())
    
    # Cosine power
    for p in [0.5, 2.0, 4.0]:
        cp = np.abs(np.cos(np.pi * xc)) ** p
        inits.append(cp / cp.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # Exponential decay
    for beta in [1.0, 3.0, 10.0]:
        e = np.exp(-beta * xc)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (1 - xc))
        inits.append(e2 / e2.sum())
    
    return inits


def track_best(a, best_seq, best_score, label=""):
    """Evaluate sequence and update best if improved. Returns (best_seq, best_score, score)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_seq, best_score, float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a.copy(), score, score
    return best_seq, best_score, score
