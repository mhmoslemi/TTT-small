import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_conv_gradient(f, k):
    """Compute gradient of conv(f,f)[k] w.r.t. f.
    grad[i] = 2*f[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(f)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * f[k - idx]
    return grad

def lse_optimize(f0, time_limit, beta_start=15.0, beta_end=5000.0, verbose=True):
    """Log-sum-exp smoothed gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient: 2 * cross_correlation(w, f) of length n
        # grad[i] = 2 * sum_m w[i+m] * f[m]
        f_flipped = f[::-1].copy()
        total_len = (2 * n - 1) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_wf = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_wf[n - 1:n - 1 + n]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 3.0 / n)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 1.5 / n
                beta = min(beta * 1.5, beta_end)
                momentum = 0.90
                no_improve = 0

        # Ramp up beta
        if beta < beta_end:
            beta *= 1.004

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def mirror_descent_polish(f0, time_limit, lr_init=0.05, decay=0.9995, top_k=20, verbose=True):
    """Mirror descent (multiplicative weights) for peak reduction.
    Uses a: a_new = a * exp(-lr * grad) / Z
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        current_max = conv.max()

        # Find top peaks
        k_top = min(top_k, len(conv))
        peak_indices = np.argpartition(conv, -k_top)[-k_top:]

        # Compute averaged subgradient over top peaks
        # Weight peaks by their value (softmax-like)
        peak_vals = conv[peak_indices]
        # Use a sharp softmax to focus on the highest peaks
        temp = (peak_vals.max() - peak_vals.min()) * 0.1 + 1e-10
        w_peaks = np.exp((peak_vals - peak_vals.max()) / temp)
        w_peaks = w_peaks / w_peaks.sum()

        grad = np.zeros(n)
        for i, k in enumerate(peak_indices):
            if w_peaks[i] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w_peaks[i] * 2.0 * f[k - idx]

        # Multiplicative update
        log_f = np.log(f)
        log_f_new = log_f - lr * grad
        # Normalize
        log_f_new -= np.logaddexp.reduce(log_f_new)  # log-sum-exp normalization
        f_new = np.exp(log_f_new)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.01, 1.0)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.95
            if no_improve > 100:
                f = best_f.copy()
                lr = lr_init * 0.5
                no_improve = 0

        lr *= decay

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it}, obj={best_obj:.12f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def peak_equalize(f0, time_limit, n_peaks=10, verbose=True):
    """Custom peak equalization: try to make the top peaks of conv equal and minimal.
    Uses a custom step that pushes mass away from peak-contributing coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        current_max = conv.max()

        # Find the top peaks
        k = min(n_peaks, len(conv))
        peak_indices = np.argsort(conv)[-k:]

        # For each peak, compute which coordinates contribute most
        # and build a "pressure" vector
        pressure = np.zeros(n)
        for idx_k in peak_indices:
            kk = int(idx_k)
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            # The contribution of f[i] to conv[kk] is 2*f[kk-i]
            # To reduce conv[kk], we should reduce f[i] where f[kk-i] is large
            contributions = 2.0 * f[kk - idx]
            pressure[idx] += contributions

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Step: reduce where pressure is high
        f_new = project_simplex(f - lr * pressure)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.92
            if no_improve > 80:
                f = best_f.copy()
                lr = 0.01
                no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PE it={it}, obj={best_obj:.12f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_iter=2000, verbose=True):
    """Coordinate descent: try updating individual pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick two random coordinates to transfer mass between
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try transferring a small amount from i to j (or vice versa)
        delta = f[i] * 0.05 * (0.5 + rng.random() * 0.5)

        # Direction: from larger to smaller, or random
        if f[i] > f[j]:
            f_new = f.copy()
            f_new[i] -= delta
            f_new[j] += delta
        else:
            f_new = f.copy()
            f_new[j] -= delta
            f_new[i] += delta

        if f_new[i] < 0 or f_new[j] < 0:
            continue

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Try a bigger perturbation
            idx = rng.integers(0, n)
            f_new = f.copy()
            f_new[idx] *= 0.5
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD it={it}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse.compute_multiple_peaks(f, n_peaks=5, beta=100.0):
    """Compute weighted gradient focusing on multiple peaks."""
    n = len(f)
    conv = fast_convolve(f)
    peak_indices = np.argsort(conv)[-n_peaks:]
    peak_vals = conv[peak_indices]

    # Sharp softmax on peak values
    w = np.exp((peak_vals - peak_vals.max()) * beta)
    w = w / w.sum()

    grad = np.zeros(n)
    for i, k in enumerate(peak_indices):
        kk = int(k)
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx = np.arange(js, je + 1)
        grad[idx] += w[i] * 2.0 * f[kk - idx]
    return grad


def lse_optimize_v2(f0, time_limit, beta_start=20.0, beta_end=8000.0, verbose=True):
    """Improved LSE with multiple peak focus and adaptive momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / n
    momentum = 0.93
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Full LSE gradient
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient via FFT
        f_flipped = f[::-1].copy()
        total_len = (2 * n - 1) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_wf = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_wf[n - 1:n - 1 + n]

        # Combine with multi-peak gradient for better focus
        peak_grad = lse_compute_multiple_peaks(f, n_peaks=5, beta=50.0)
        grad = 0.7 * grad + 0.3 * peak_grad

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Adaptive learning rate based on progress
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 6:
                lr *= 0.94
                momentum = max(0.86, momentum - 0.002)
            if no_improve > 50:
                # Try a random restart with increased beta
                f = best_f.copy()
                v = np.zeros(n)
                lr = 2.0 / n
                beta = min(beta * 2.0, beta_end)
                momentum = 0.90
                no_improve = 0

        beta = min(beta * 1.005, beta_end)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE2 it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_compute_multiple_peaks(f, n_peaks=5, beta=100.0):
    """Compute weighted gradient focusing on multiple peaks."""
    n = len(f)
    conv = fast_convolve(f)
    k = min(n_peaks, len(conv))
    peak_indices = np.argsort(conv)[-k:]
    peak_vals = conv[peak_indices]

    w = np.exp((peak_vals - peak_vals.max()) * beta)
    w = w / w.sum()

    grad = np.zeros(n)
    for i, kk in enumerate(peak_indices):
        kk = int(kk)
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx = np.arange(js, je + 1)
        grad[idx] += w[i] * 2.0 * f[kk - idx]
    return grad


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def diversified_restart(best_f, n, rng, magnitude=0.01):
    """Create a diversified version of best_f."""
    f = best_f.copy()
    # Random perturbation
    f = f + rng.standard_normal(n) * magnitude
    f = project_simplex(f)
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: FW-LP on parent
    Phase 2: LSE gradient polishing (v2)
    Phase 3: Mirror descent
    Phase 4: Peak equalization
    Phase 5: Coordinate descent
    Phase 6: Alternating FW-LP and LSE
    Phase 7: Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            par_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={par_score:.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting optimization, n={n_parent}")

    # === Phase 1: FW-LP on parent ===
    print("\n=== Phase 1: FW-LP on parent ===")
    t_remaining = deadline - time.monotonic()
    t_alloc = min(300, t_remaining * 0.20)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(parent, time_limit=t_alloc, verbose=True)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE v2 polishing ===
    print("\n=== Phase 2: LSE v2 Polishing ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(400, t_remaining * 0.25)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_optimize_v2(current_best, time_limit=t_alloc,
                                     beta_start=20.0, beta_end=10000.0, verbose=True)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Mirror descent ===
    print("\n=== Phase 3: Mirror Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(200, t_remaining * 0.15)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = mirror_descent_polish(current_best, time_limit=t_alloc,
                                           lr_init=0.03, decay=0.9997,
                                           top_k=30, verbose=True)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Peak equalization ===
    print("\n=== Phase 4: Peak Equalization ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(150, t_remaining * 0.12)
    if t_alloc > 15:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = peak_equalize(current_best, time_limit=t_alloc,
                                   n_peaks=15, verbose=True)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Coordinate descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(150, t_remaining * 0.10)
    if t_alloc > 15:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = coordinate_descent_polish(current_best, time_limit=t_alloc,
                                               verbose=True)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Alternating FW-LP and LSE ===
    print("\n=== Phase 6: Alternating Refinement ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()

        for round_i in range(4):

            if t_remaining < 30:

            t_alloc = min(60, t_remaining / (5 - round_i))
            if t_alloc < 20:

            current_best = best_tracker[1] if best_tracker[1] is not None else parent
            # Alternate: even rounds do FW-LP, odd rounds do LSE

                a_opt, _ = fw_lp_optimize(current_best, time_limit=t_alloc, verbose=False)

                                           beta_start=50.0, beta_end=20000.0, verbose=False)
            try_sequence(a_opt, best_tracker)

    print("\n=== Phase 8: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 30

    alt_lengths = [384, 512, 320]
    for n_alt in alt_lengths:
        if n_alt == n_best:
            continue
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 50:
            break
        t_alloc = min(60, t_remaining)
        if t_alloc < 30:
            break

        print(f"  Trying n={n_alt} ({t_alloc:.0f}s)")
        # Interpolate to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick optimization: FW-LP then LSE
        t1 = t_alloc * 0.5
        t2 = t_alloc * 0.5
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t1, verbose=False)
        try_sequence(a_opt, best_tracker)

        current_alt = best_tracker[1] if best_tracker[1] is not None and len(best_tracker[1]) == n_alt else a_opt
        if len(current_alt) == n_alt:
            a_opt, _ = lse_optimize_v2(current_alt, time_limit=t2,
                                       beta_start=20.0, beta_end=5000.0, verbose=False)
            try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
