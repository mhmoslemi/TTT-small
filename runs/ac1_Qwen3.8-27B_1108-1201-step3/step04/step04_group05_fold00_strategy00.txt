    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([np.clip(v[0], 0, 1000.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) * (1 + np.arange(n))
    theta = (cssv - 1.0) / cssv[:, np.newaxis]
    w = np.clip(theta, 0, 1)
    w = (w * (np.sum(w) > 1e-10)) + (1 - np.sum(w)) * np.ones(n) / n
    return np.maximum(v - np.minimum(w, theta).sum(), 0)
