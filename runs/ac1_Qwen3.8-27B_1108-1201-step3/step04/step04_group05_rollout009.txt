"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using adaptive accelerated subgradient
with shape-based restarts, LP-based escape directions, and multi-scale search.
Key innovations: Nesterov momentum with adaptive temperature, parametric shape
perturbations for restarts, and convolution-aware coordinate descent for final polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_rows(f, peak_indices):
    """Build rows of conv matrix for given peak indices. Returns list of (js, je, f_slice)."""
    n = len(f)
    result = []
    for k in peak_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # d(conv[k])/d(f[j]) = 2*f[k-j] for j in [js, je]
        # f[k-j] for j in [js,je] is f[k-je : k-js+1] reversed
        f_slice = f[k - je : k - js + 1][::-1]
        result.append((js, je, f_slice))
    return result


def compute_subgrad_vec(f, conv, peaks, weights, n):
    """Compute weighted subgradient of max conv at given peaks."""
    sg = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = peaks[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # f[k-j] for j in [js, je]: indices k-je down to k-js
        f_vals = f[k - je : k - js + 1][::-1]
        sg[js:je+1] += 2.0 * w * f_vals
    return sg


def solve_escape_lp(f, conv, top_k=20):
    """Solve LP to find a direction that reduces peak convolution.
    Minimize t s.t. conv[k] + 2*(B*g)[k] <= t for top-k peaks,
    g >= -f, sum(g) = 0. Returns g or None."""
    n = len(f)
    if top_k >= len(conv):
        top_k = len(conv)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0

    for ii, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # (B*g)[k] = sum_j g[j] * f[k-j] for j in [js, je]
        # So column j of A_ub[ii, :] for j in [js,je] is f[k-j]
        f_vals = f[k - je : k - js + 1][::-1]
        A_ub[ii, js+1:je+2] = 2.0 * f_vals

    b_ub = -conv[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})
    if res.success:
        return res.x[1:]
    return None


def shape_perturbation(f, rng, shape_type=None):
    """Perturb towards a parametric shape to escape local minima."""
    n = len(f)
    if shape_type is None:
        shape_type = rng.integers(0, 5)

    x = np.arange(n, dtype=np.float64)
    xc = 0.5 * (n - 1)

    if shape_type == 0:
        # Asymmetric Gaussian
        center = xc * rng.uniform(0.3, 0.7)
        sigma = n * rng.uniform(0.15, 0.35)
        shape = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif shape_type == 1:
        # Bimodal
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        s = n * rng.uniform(0.08, 0.2)
        shape = np.exp(-0.5 * ((x - c1) / s) ** 2) + np.exp(-0.5 * ((x - c2) / s) ** 2)
    elif shape_type == 2:
        # Cosine window with phase
        phase = rng.uniform(0, np.pi)
        freq = rng.uniform(0.5, 2.0)
        shape = np.abs(np.cos(np.pi * freq * x / n + phase)) + 0.1
    elif shape_type == 3:
        # Power law (edge concentration)
        alpha = rng.uniform(1.5, 4.0)
        dist = np.abs(x - xc) / (0.5 * n)
        shape = dist ** alpha + 0.05
    else:
        # Inverse Gaussian (center concentration)
        sigma = n * rng.uniform(0.1, 0.25)
        shape = 1.0 / (1.0 + 0.5 * ((x - xc) / sigma) ** 2)

    shape = np.maximum(shape, 0)
    s = shape.sum()
    if s < 1e-10:
        shape = np.ones(n)
        s = n
    shape = shape / s

    # Blend with current
    t = rng.uniform(0.3, 0.7)
    f_new = (1 - t) * f + t * shape
    f_new = np.maximum(f_new, 0)
    fs = f_new.sum()
    if fs < 1e-15:
        return f.copy()
    return f_new / fs


def accelerated_subgrad(f0, time_limit, top_k=50, initial_temp=0.001,
                        initial_lr=0.005, mom=0.9, verbose=False):
    """Accelerated subgradient with adaptive temperature and Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    y = f.copy()  # Nesterov ghost

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = initial_lr
    cur_temp = initial_temp
    cur_mom = mom
    it = 0
    rng = np.random.default_rng(777)

    # Adaptive parameters
    lr_decay = 0.9995
    temp_growth = 1.003
    patience = 200

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute peaks and weights
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks].astype(np.float64)
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = compute_subgrad_vec(y, conv, peaks, w, n)

        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        f_new = y + v
        f_new = project_simplex(f_new)

        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    ASubGrad iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1

        # Update Nesterov ghost
        if no_improve > patience:
            # Restart: jump towards best + shape perturbation
            shape = shape_perturbation(best_f, rng)
            y = shape
            f_new = shape
            v = np.zeros(n)
            cur_lr = max(cur_lr * 1.5, initial_lr * 0.5)
            no_improve = 0
            # Re-evaluate
            conv = fast_convolve(f_new)
            obj = conv.max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_new.copy()
        else:
            # Nesterov interpolation
            beta = (1.0 + np.sqrt(1.0 + 4.0 * cur_mom**2)) / 2.0
            y = f_new + (cur_mom / beta) * (f_new - f)
            f = f_new

        # Adaptive parameter updates
        cur_lr *= lr_decay
        cur_temp *= temp_growth

        if cur_lr < 1e-5:
            cur_lr = initial_lr * 0.3
            cur_temp = initial_temp * 2.0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, rng):
    """Coordinate descent with smart coordinate selection based on convolution sensitivity."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    step_size = 0.01
    patience = 0

    while time.monotonic() - t0 < time_limit:
        # Find the peak
        peak_idx = np.argmax(conv)

        # For the peak at position k, the gradient w.r.t. f[j] is 2*f[k-j]
        # We want to reduce f[j] where f[k-j] is large (j contributes most to peak)
        # Or increase f[j] where f[k-j] is small (j contributes least)

        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        f_at_peak = f[peak_idx - je : peak_idx - js + 1][::-1]

        # Choose coordinate to modify: the one with highest f[j] * f[k-j]
        # (largest contribution to the peak)
        contribs = f[js:je+1] * f_at_peak
        if contribs.max() < 1e-15:
            break

        # Try reducing the max contributor and increasing a min contributor
        j_reduce = js + np.argmax(contribs)
        j_increase = js + np.argmin(contribs)

        if j_reduce == j_increase:
            continue

        # Try a few step sizes
        best_local = None
        best_local_obj = best_obj

        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            delta = f[j_reduce] * ss
            if delta > f[j_reduce]:
                delta = f[j_reduce] * 0.5
            if delta < 1e-10:
                continue

            f_trial = f.copy()
            f_trial[j_reduce] -= delta
            f_trial[j_increase] += delta
            f_trial = project_simplex(f_trial)

            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()

            if obj_trial < best_local_obj:
                best_local_obj = obj_trial
                best_local = f_trial.copy()

        if best_local is not None:
            f = best_local
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1

        if patience > 100:
            # Perturb and restart
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            step_size *= 0.8
            patience = 0
            if step_size < 1e-4:
                step_size = 0.005

    return best_f, best_obj


def lp_escape_optimize(f0, time_limit, verbose=False):
    """Optimize using LP-based escape directions with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(31415)
    top_k = max(10, n // 10)

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_escape_lp(f, conv, top_k=top_k)
        if g is None:
            no_improve += 10
            if no_improve > 50:
                f = shape_perturbation(best_f, rng)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = shape_perturbation(best_f, rng)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            f_new = (1 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            # Fine search
            lo = max(0.01, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                f_new = (1 - mid) * f + mid * target
                f_new = np.maximum(f_new, 0)
                sn = f_new.sum()
                if sn < 1e-15:
                    break
                f_new = f_new / sn
                val = fast_convolve(f_new).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0)
            f = f / f.sum()

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:

                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = shape_perturbation(best_f, rng, shape_type=rng.integers(0, 5))
            no_improve = 0

    return best_f, best_obj


def generate_parametric_inits(n, rng):

            center = frac * (n - 1)
            sigma = sigma_frac * n
            g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
            g = np.maximum(g, 0)
            if g.sum() > 1e-10:
                inits.append(g / g.sum())

    # Bimodal
    for c1_frac, c2_frac in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7), (0.15, 0.85)]:
        for s_frac in [0.08, 0.12, 0.18, 0.25]:

            s = s_frac * n
            b = np.exp(-0.5 * ((x - c1) / s) ** 2) + np.exp(-0.5 * ((x - c2) / s) ** 2)
            b = np.maximum(b, 0)
            if b.sum() > 1e-10:
                inits.append(b / b.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        for phase in [0, np.pi / 2, np.pi / 3]:
            c = np.abs(np.cos(np.pi * freq * x / n + phase)) + 0.1
            inits.append(c / c.sum())

    # Edge concentration

        dist = np.abs(x - xc) / (0.5 * n)
        e = dist ** alpha + 0.05
        inits.append(e / e.sum())

    # Random (but structured)
    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(r)

    return inits

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Multi-length exploration ===
    print("=== Phase 1: Exploration ===")
    t_start = time.monotonic()
    phase1_time = min(80.0, (deadline - time.monotonic()) * 0.05)

    lengths_to_try = [192, 256, 320, 384, 448, 512, 640]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            lengths_to_try.insert(0, pn)

    candidates = []

    for n in lengths_to_try:
        inits = generate_parametric_inits(n, rng)
        # Sample a subset for speed
        if len(inits) > 12:
            idx = rng.choice(len(inits), 12, replace=False)
            inits = [inits[i] for i in idx]

        for i, ini in enumerate(inits):
            candidates.append((n, f"p1_{n}_{i}", ini))

    # Also add parent if available
    if parent is not None:
        candidates.insert(0, (len(parent), "parent", parent))

    # Quick evaluate and optimize top candidates
    per_candidate_time = max(2.0, phase1_time / max(1, len(candidates)))

    for n, label, ini in candidates:
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(per_candidate_time, 8.0)

        # Quick eval
        score = try_seq(ini, f"init_{label}")

        # Quick optimize
        if t_alloc > 3:
            f_opt, _ = quick_optimize(ini, t_alloc, n)
            try_seq(f_opt, f"opt_{label}")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 2: Accelerated Subgradient (primary optimizer) ===
    print("\n=== Phase 2: Accelerated Subgradient ===")
    n_best = len(best_seq) if best_seq is not None else 384
    remaining = deadline - time.monotonic() - 300
    t_p2 = min(remaining * 0.5, 800)

    if t_p2 > 30:
        print(f"  Running ASubGrad on n={n_best}, budget={t_p2:.0f}s")

        # Run multiple times with different hyperparameters
        configs = [
            {'top_k': max(30, n_best // 6), 'lr': 0.004, 'temp': 0.0008, 'mom': 0.9},
            {'top_k': max(20, n_best // 10), 'lr': 0.002, 'temp': 0.0003, 'mom': 0.93},
            {'top_k': max(40, n_best // 4), 'lr': 0.006, 'temp': 0.0015, 'mom': 0.85},
        ]

        per_config_time = t_p2 / len(configs)

        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(per_config_time, 300)
            print(f"  Config {ci}: top_k={cfg['top_k']}, lr={cfg['lr']}, temp={cfg['temp']}")

            f_opt, obj = accelerated_subgrad(
                best_seq, time_limit=t_alloc * 0.8,
                top_k=cfg['top_k'], initial_lr=cfg['lr'],
                initial_temp=cfg['temp'], mom=cfg['mom'], verbose=True
            )
            try_seq(f_opt, f"asubgrad_{ci}")

            # From the best, try again with perturbation
            rem = deadline - time.monotonic() - 200
            if rem > 20 and ci < len(configs) - 1:
                f_pert = shape_perturbation(best_seq, rng, shape_type=ci)
                f_opt2, _ = accelerated_subgrad(
                    f_pert, time_limit=min(rem * 0.4, 100),
                    top_k=cfg['top_k'], initial_lr=cfg['lr'] * 0.5,
                    initial_temp=cfg['temp'] * 2
                )
                try_seq(f_opt2, f"asubgrad_pert_{ci}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: LP-based escape ===
    print("\n=== Phase 3: LP Escape ===")
    remaining = deadline - time.monotonic() - 200
    t_p3 = min(remaining * 0.4, 400)

    if t_p3 > 20:
        print(f"  Running LP escape on n={n_best}, budget={t_p3:.0f}s")
        f_opt, _ = lp_escape_optimize(best_seq, time_limit=t_p3 * 0.7, verbose=True)
        try_seq(f_opt, "lp_escape")

        # From perturbed start
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            f_pert = shape_perturbation(best_seq, rng, shape_type=1)  # bimodal
            f_opt2, _ = lp_escape_optimize(f_pert, time_limit=min(rem * 0.5, 150))
            try_seq(f_opt2, "lp_escape_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Coordinate Descent ===
    print("\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 100
    t_p4 = min(remaining * 0.4, 250)

    if t_p4 > 15:
        print(f"  Running coordinate descent, budget={t_p4:.0f}s")
        f_opt, _ = coordinate_descent(best_seq, time_limit=t_p4 * 0.8, rng=rng)
        try_seq(f_opt, "coord_desc")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 40
    t_p5 = min(remaining * 0.8, 200)

    if t_p5 > 10:
        # Fine subgradient
        t_fine = min(t_p5 * 0.5, 100)
        f_opt, _ = accelerated_subgrad(
            best_seq, time_limit=t_fine,
            top_k=max(10, n_best // 20),
            initial_lr=0.0005, initial_temp=0.0001, mom=0.95
        )
        try_seq(f_opt, "final_sg")

        # Tiny local search
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_ls = min(rem * 0.6, 60)
            rng_ls = np.random.default_rng(999)
            best_f = best_seq.copy()
            t0 = time.monotonic()
            conv = fast_convolve(best_f)
            best_obj = conv.max()
            scale = 0.002

            while time.monotonic() - t0 < t_ls:
                for _ in range(200):
                    if time.monotonic() - t0 > t_ls:
                        break
                    noise = rng_ls.standard_normal(n_best) * scale
                    f_trial = project_simplex(best_f + noise)
                    obj_trial = fast_convolve(f_trial).max()
                    if obj_trial < best_obj - 1e-15:
                        best_obj = obj_trial
                        best_f = f_trial.copy()
                scale *= 0.9
                if scale < 1e-5:
                    scale = 0.001
                    best_f = project_simplex(best_f + rng_ls.standard_normal(n_best) * 0.005)
                    best_obj = fast_convolve(best_f).max()

            try_seq(best_f, "local_search")

        # Final LP polish
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            f_opt, _ = lp_escape_optimize(best_seq, time_limit=min(rem * 0.6, 40))
            try_seq(f_opt, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
