        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * (0.9999 ** t)
        cur_lr = max(cur_lr, lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 100 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.999,
                            temp_init=0.008, temp_decay=0.9995, top_k=None,
                            verbose=False):
    """Mirror descent (multiplicative weights) with soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)

        # Mirror descent: multiplicative update
        log_update = -cur_lr * sg
        log_update -= log_update.max()  # numerical stability
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-7)
        cur_lr *= lr_decay
        cur_lr = max(cur_lr, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 100 == 0:
                print(f"    Mirror iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            multipliers = [0.3, 0.5, 0.7, 0.8, 1.2, 1.4, 1.7, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization as a sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse initializations, Adam/mirror descent,
    coordinate descent polishing, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform init n=512")

    n_parent = len(parent)

    # === Phase 1: Diverse initializations at parent length ===
    print(f"\n=== Phase 1: Diverse initializations (n={n_parent}) ===")
    t_phase1_end = time.monotonic() + 150

    initializations = {
        'parent': parent.copy(),
        'gauss_015': make_gaussian_init(n_parent, 0.5, 0.15),
        'gauss_025': make_gaussian_init(n_parent, 0.5, 0.25),
        'bump': make_symmetric_init(n_parent, 'bump'),
        'triangle': make_symmetric_init(n_parent, 'triangle'),
        'smooth_rand': make_smooth_random_init(n_parent, rng, 5),
    }

    # Also try perturbed parent
    initializations['parent_pert'] = project_simplex(parent + rng.standard_normal(n_parent) * 0.01)
    # Convex combination
    initializations['parent_uniform'] = project_simplex(0.7 * parent + 0.3 * np.ones(n_parent) / n_parent)

    phase1_best = None
    phase1_best_obj = float('inf')

    for name, init in initializations.items():
        if time.monotonic() > t_phase1_end:
            break
        rem = t_phase1_end - time.monotonic()
        if rem < 10:
            break
        t_alloc = min(rem * 0.45, 25)

        print(f"  Trying init '{name}'...")
        a_opt, obj = adam_optimize(init, time_limit=t_alloc,
                                   lr=0.004, temp_init=0.006,
                                   temp_decay=0.9993, top_k=max(20, n_parent // 10))
        try_seq(a_opt, f"p1_{name}")
        if obj < phase1_best_obj:
            phase1_best_obj = obj
            phase1_best = a_opt.copy()

    if phase1_best is not None:
        try_seq(phase1_best, "p1_best")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Extended Adam on best with different hyperparameters ===
    print(f"\n=== Phase 2: Extended optimization (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 400
    n_base = len(base)

    configs = [
        {'lr': 0.006, 'temp_init': 0.005, 'temp_decay': 0.9994, 'top_k': max(20, n_base // 12)},
        {'lr': 0.003, 'temp_init': 0.010, 'temp_decay': 0.9996, 'top_k': max(30, n_base // 8)},
        {'lr': 0.008, 'temp_init': 0.004, 'temp_decay': 0.9992, 'top_k': max(15, n_base // 15)},
        {'lr': 0.005, 'temp_init': 0.007, 'temp_decay': 0.9995, 'top_k': None},
    ]

    for ci, cfg in enumerate(configs):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.7, 120)

        print(f"  Adam config {ci} (lr={cfg['lr']}, temp={cfg['temp_init']}), t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc, verbose=False, **cfg)
        try_seq(a_opt, f"p2_adam_{ci}")

        # Also try mirror descent with same config
        rem = t_phase2_end - time.monotonic()
        if rem < 20:
            break
        t_alloc_md = min(rem * 0.5, 60)
        a_opt, _ = mirror_descent_optimize(base, time_limit=t_alloc_md,
                                           lr_init=cfg['lr'] * 2,
                                           temp_init=cfg['temp_init'],
                                           temp_decay=cfg['temp_decay'],
                                           top_k=cfg['top_k'])
        try_seq(a_opt, f"p2_mirror_{ci}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 350

    target_lengths = [512, 640, 768, 896, 1024, 1280]
    # Filter to reasonable ones
    target_lengths = [t for t in target_lengths if t != n_base and 128 <= t <= 2048]
    # Sort by distance to current best length
    target_lengths.sort(key=lambda x: abs(x - n_base))
    target_lengths = target_lengths[:4]  # Try at most 4 new lengths

    for target_n in target_lengths:
        rem = t_phase3_end - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.55, 100)

        print(f"  Length n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc,
                                 lr=0.005, temp_init=0.006,
                                 temp_decay=0.9994, top_k=max(20, target_n // 12))
        try_seq(a_opt, f"p3_n{target_n}")

        # Quick mirror descent too
        rem = t_phase3_end - time.monotonic()
        if rem > 20:
            a_opt, _ = mirror_descent_optimize(f_interp, time_limit=min(rem * 0.3, 40),
                                               lr_init=0.01, temp_init=0.006,
                                               temp_decay=0.9994)
            try_seq(a_opt, f"p3_mirror_n{target_n}")

    # === Phase 4: Deep optimization on current best ===
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 300
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple Adam runs with different starts
        starts = [
            ('direct', best_seq.copy()),
            ('noise1', project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)),
            ('noise2', project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)),
            ('mirror_mix', project_simplex(0.8 * best_seq + 0.2 * make_gaussian_init(n_best, 0.5, 0.2))),
        ]

        for si, (sname, sinit) in enumerate(starts):
            rem = t_phase4_end - time.monotonic()
            if rem < 25:
                break
            t_alloc = min(rem * 0.6, 80)

            print(f"  Deep start '{sname}', t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(sinit, time_limit=t_alloc,
                                     lr=0.003, temp_init=0.004,
                                     temp_decay=0.9996, top_k=max(15, n_best // 16))
            try_seq(a_opt, f"p4_{sname}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 150
    if best_seq is not None:
        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.8, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_cd")

            # Another CD pass from different start
            rem = t_phase5_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                n_best = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                a_opt, _ = coordinate_descent(perturbed, time_limit=min(rem * 0.7, 40))
                try_seq(a_opt, "p5_cd_pert")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine Adam with low temperature
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem * 0.6, 60)
            print(f"  Fine Adam, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc,
                                     lr=0.001, temp_init=0.002,
                                     temp_decay=0.9998, top_k=max(10, n_best // 25))
            try_seq(a_opt, "p6_fine")

        # Tiny perturbations + quick optimize
        rem = deadline - time.monotonic() - 10
        for pi in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = adam_optimize(f_pert, time_limit=min(rem * 0.5, 20),
                                     lr=0.002, temp_init=0.003,
                                     temp_decay=0.9997, top_k=max(10, n_best // 20))
            try_seq(a_opt, f"p6_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
