def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)
    # This is just scaling, not a true projection. For a true projection:
    # We need to solve: min ||x - v||^2 s.t. x >= 0, sum(x) = target_sum
    # This has a closed-form solution using a threshold.

def project_simplex_true(v, target_sum=1.0):

            cur_obj = 2.0 * n * fast_convolve(a).max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Various structured inits
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps
    
    return inits
