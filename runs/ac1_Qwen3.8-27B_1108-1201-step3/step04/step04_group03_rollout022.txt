"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using second-order Newton optimization
with adaptive active set identification. Key innovations:
1. Exact Hessian computation exploiting the Hankel structure of the auto-convolution
2. Adaptive active set that expands/contracts based on peak proximity
3. Sum-preserving Newton directions with projection for non-negativity
4. Log-sum-exp smoothing as a fallback for non-smooth max transitions
5. Multiple restarts with diverse initializations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Scale to target sum
    v_scaled = v * (target_sum / max(v.sum(), 1e-30))
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v_scaled - theta, 0.0)


def compute_gradient(a, K):
    """Compute gradient of (1/|K|) sum_{k in K} q_k(a) where q_k = conv(a,a)[k].
    
    ∂q_k/∂a_m = 2*a[k-m] if 0 <= k-m < n, else 0
    """
    n = len(a)
    grad = np.zeros(n)
    for k in K:
        k = int(k)
        # For each m, ∂q_k/∂a_m = 2*a[k-m] if 0 <= k-m < n
        # This means: grad[m] += 2 * a[k-m] for m in range(max(0,k-n+1), min(n, k+1))
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            # a[k-m] for m in [m_lo, m_hi] is a[k-m_lo] down to a[k-m_hi]
            # = a[k-m_lo], a[k-m_lo-1], ..., a[k-m_hi]
            # = a[np.arange(k-m_lo, k-m_hi-1, -1)]
            idx = np.arange(k - m_hi, k - m_lo + 1)  # this is k-m for m in [m_lo, m_hi]
            # Wait, let me redo this
            # m goes from m_lo to m_hi
            # k-m goes from k-m_lo down to k-m_hi
            # We need a[k-m] for each m
            km = k - np.arange(m_lo, m_hi + 1)  # array of k-m values
            grad[m_lo:m_hi+1] += 2.0 * a[km]
    grad /= len(K)
    return grad


def compute_hessian_diag(K, n):
    """Returns the anti-diagonal values of the Hessian.
    
    H[m,l] = (2/|K|) if m+l is in K, else 0.
    So the k-th anti-diagonal (where m+l=k) has value 2/|K| if k in K, else 0.
    """
    K_set = set(int(k) for k in K)
    # Anti-diagonal k has value 2/|K| if k in K_set
    # Return the c vector where c[k] = H[m,k-m] for any valid m
    c = np.zeros(2 * n - 1)
    for k in K_set:
        if 0 <= k < 2 * n - 1:
            c[k] = 2.0 / len(K)
    return c


def build_hessian(c, n):
    """Build the full Hessian matrix from anti-diagonal values c.
    
    H[m,l] = c[m+l]
    """
    H = np.zeros((n, n))
    for k in range(2 * n - 1):
        if c[k] != 0:
            # H[m, k-m] = c[k] for valid m
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                m = np.arange(m_lo, m_hi + 1)
                l = k - m
                H[m, l] = c[k]
    return H


def newton_optimize(a0, time_limit, verbose=True, max_iter=10000):
    """Newton optimization with adaptive active set.
    
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    mu = 0.01  # regularization
    eps_active = 0.01  # fraction of max for active set threshold
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Adaptive active set
        threshold = max_val - eps_active * max_val
        K = np.where(conv >= threshold)[0]
        
        if len(K) == 0:
            K = np.array([np.argmax(conv)])
        elif len(K) > 100:
            # Limit active set size
            K = np.argpartition(conv, -100)[-100:]
        
        # Compute gradient
        grad = compute_gradient(a, K)
        
        # Make gradient sum-zero (to preserve sum(a)=1)
        grad -= grad.mean()
        
        # Compute Hessian
        c = compute_hessian_diag(K, n)
        H = build_hessian(c, n)
        
        # Also make H "sum-zero" consistent: H @ ones = 0? Not necessarily.
        # We'll handle the constraint via the direction.
        
        # Regularized Newton step
        H_reg = H + mu * np.eye(n)
        # Ensure positive definite
        try:
            d = np.linalg.solve(H_reg, -grad)
        except np.linalg.LinAlgError:
            # Add more regularization
            H_reg = H + (mu * 10) * np.eye(n)
            try:
                d = np.linalg.solve(H_reg, -grad)
            except:
                d = -grad / (np.linalg.norm(grad) + 1e-10)
        
        # Make d sum-zero
        d -= d.mean()
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_new = a + alpha * d
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_new = a_new / s
            
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            
            if obj_new < max_val - 1e-15:
                a = a_new
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if improved:
            if max_val < best_obj - 1e-12:
                best_obj = max_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    current_score = 2 * n * best_obj
                    print(f"    Newton it={it}: best_obj={best_obj:.10f}, score={current_score:.8f}, |K|={len(K)}, mu={mu:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 20:
            mu *= 0.5
            eps_active = max(eps_active * 0.8, 0.001)
            no_improve = 0
        
        if no_improve > 100:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.001 + 0.002 * min(1.0, no_improve / 500.0)
            a = best_a * np.exp(rng.standard_normal(n) * noise_scale)
            a = np.maximum(a, 0)
            a = a / a.sum()
            mu = 0.01
            eps_active = 0.01
            no_improve = 0
        
        # Gradually decrease mu
        if it > 50:
            mu = max(0.001, mu * 0.999)
    
    return best_a, best_obj
