def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_minmax_lp(f):
    """Build A_ub, b_ub for the min-max LP: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for m in range(n):
            km = k - m
            if 0 <= km < n:
                A_ub[k, m] = f[km]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    return A_ub, b_ub


def solve_minmax_lp(f, t_ub=None):
    """Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0. Returns g."""
    n = len(f)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub, b_ub = build_minmax_lp(f)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]
    if t_ub is not None:
        bounds[-1] = (-1e10, t_ub)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None

def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n)))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    ew = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(ew / ew.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    # Sine squared
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits.append(s1 / s1.sum())
    # Two peaks
    p1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits.append(p1 / p1.sum())
    # Bowl (inverted gaussian)
    bw = 1.0 - 0.8 * np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(bw / bw.sum())
    # Linear ramp
    r1 = np.linspace(0.1, 1.0, n)
    inits.append(r1 / r1.sum())
    # Zigzag
    zz = np.abs(np.sin(np.pi * x * 3.0 / n)) + 0.1
    inits.append(zz / zz.sum())
    return inits


def upsample(f, n_new):
    """Upsample sequence to new length using linear interpolation."""
    if len(f) == n_new:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return f_new / s

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_fallback")

    n_p = len(parent)

    # === Phase 1: Quick exploration with multiple inits (~100s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase1 = min(100, deadline - time.monotonic() - 1500)
    if t_phase1 > 20:
        inits = generate_structured_inits(n_p, rng)
        # Add upsampled parent at different lengths
        for n_try in [n_p]:
            t_remaining = t_phase1 - (time.monotonic() - (deadline - float(budget_s) + 20 - t_phase1))
            for ii, init in enumerate(inits[:4]):
                rem = deadline - time.monotonic() - 1380
                if rem < 10:
                    break
                t_alloc = min(rem, 25)
                a_opt, _ = fw_minmax_lp(init, time_limit=t_alloc)
                try_seq(a_opt, f"fw_init{ii}")

    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: Deep FW min-max LP (~500s) ===
    print("\n=== Phase 2: Deep FW Min-Max LP ===")
    if best_seq is not None:
        t2 = min(500, deadline - time.monotonic() - 900)
        if t2 > 30:
            print(f"  Running FW mm-LP for {t2:.0f}s")
            a_opt, _ = fw_minmax_lp(best_seq, time_limit=t2)
            try_seq(a_opt, "fw_deep")

            # Perturbed restart
            rem = deadline - time.monotonic() - 500
            if rem > 30:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                f_p = project_simplex(best_seq + noise)
                a_opt2, _ = fw_minmax_lp(f_p, time_limit=min(rem, 200))
                try_seq(a_opt2, "fw_perturbed")

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Log-sum-exp Natural Gradient (~400s) ===
    print("\n=== Phase 3: LogSumExp NG ===")
    if best_seq is not None:
        ls_configs = [
            {'beta_init': 2.0, 'beta_end': 200.0, 'lr_init': 0.02, 'lr_end': 0.002},
            {'beta_init': 5.0, 'beta_end': 400.0, 'lr_init': 0.01, 'lr_end': 0.001},
            {'beta_init': 1.0, 'beta_end': 100.0, 'lr_init': 0.03, 'lr_end': 0.005},
        ]
        rem = deadline - time.monotonic() - 500
        t_per = max(15, rem / len(ls_configs))
        for ci, cfg in enumerate(ls_configs):
            rem = deadline - time.monotonic() - 300
            if rem < 10:
                break
            t_alloc = min(t_per, rem)
            print(f"  LS-NG config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = logsumexp_ng(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"lsg_ng{ci}")

    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Coordinate Descent (~200s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem, 180)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: Multi-length exploration (~200s) ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        for n_try in [256, 512]:
            if n_try == len(best_seq):
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            f_interp = upsample(best_seq, n_try)
            try_seq(f_interp, f"interp_n{n_try}")

            t_alloc = min(rem - 10, 80)
            if t_alloc > 10:
                a_opt, _ = fw_minmax_lp(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"fw_n{n_try}")

                rem2 = deadline - time.monotonic() - 60
                if rem2 > 15:
                    a_opt2, _ = logsumexp_ng(a_opt, time_limit=min(rem2, 40),
                                             beta_init=5.0, beta_end=100.0,
                                             lr_init=0.01, lr_end=0.002)
                    try_seq(a_opt2, f"lsg_n{n_try}")

    print(f"  After Phase 5: best={best_score:.8f}")

    # === Phase 6: Final polish (~80s) ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_half = max(10, rem / 2)
            # Quick FW
            a_opt, _ = fw_minmax_lp(best_seq, time_limit=t_half)
            try_seq(a_opt, "polish_fw")

            rem = deadline - time.monotonic() - 15
            if rem > 10:
                # Quick LS-NG with high beta
                a_opt2, _ = logsumexp_ng(best_seq, time_limit=rem,
                                         beta_init=20.0, beta_end=200.0,
                                         lr_init=0.005, lr_end=0.0005)
                try_seq(a_opt2, "polish_lsg")

        # Tiny random restarts
        rem = deadline - time.monotonic() - 10
        if rem > 10 and best_seq is not None:
            for _ in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_p = project_simplex(best_seq + noise)
                a_opt, _ = fw_minmax_lp(f_p, time_limit=min(rem, 8))
                try_seq(a_opt, "tiny_restart")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        print(f"\nFINAL SCORE: {final_score:.8f} (n={len(best_seq)})")
    else:
        print(f"\nWARNING: Final score is {final_score}, using fallback")
        best_seq = parent if parent is not None else np.ones(384) / 384

    return [float(x) for x in best_seq]
