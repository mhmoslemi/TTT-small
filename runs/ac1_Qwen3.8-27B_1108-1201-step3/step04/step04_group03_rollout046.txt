    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def ratio_coord_descent(a0, time_limit, n_candidates=12, verbose=False):
    """
    Ratio-based coordinate descent.
    Directly minimizes F(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses incremental convolution updates for speed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    else:
        a = a / S
        S = 1.0
    n = len(a)
    
    b = fast_convolve(a)
    best_F = 2.0 * n * b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    sweep = 0
    last_print = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        S = a.sum()
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        
        for j in order:
            j = int(j)
            v = a[j]
            S = a.sum()
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            
            # B0: max of b outside affected range
            B0 = 0.0
            if k_lo > 0:
                B0 = max(B0, float(b[:k_lo].max()))
            if k_hi < 2 * n - 2:
                B0 = max(B0, float(b[k_hi + 1:].max()))
            
            has_quad = (2 * j >= k_lo) and (2 * j <= k_hi)
            affected = np.arange(k_lo, k_hi + 1)
            
            if has_quad:
                lin_mask = affected != 2 * j
                lin_ks = affected[lin_mask]
                lin_a = a[lin_ks - j]
                lin_b = b[lin_ks]
            else:
                lin_ks = affected
                lin_a = a[lin_ks - j]
                lin_b = b[lin_ks]
            
            # Current F
            cur_max_b = B0
            if len(lin_ks) > 0:
                cur_max_b = max(cur_max_b, float(lin_b.max()))
            if has_quad:
                cur_max_b = max(cur_max_b, float(b[2 * j]))
            best_F_j = 2.0 * n * cur_max_b / (S * S)
            best_w = v
            
            # Generate candidates
            if v < 1e-12:
                base = S / n
                cands = np.array([0.0, 0.1*base, 0.5*base, base, 2.0*base])
            else:
                cands = np.array([
                    0.0,
                    0.2*v, 0.4*v, 0.6*v, 0.8*v,
                    1.2*v, 1.5*v, 2.0*v, 3.0*v,
                    S/n * 0.5, S/n, S/n * 2.0
                ])
            
            for w in cands:
                w = float(w)
                if w < -1e-15 or w > 100 * S:
                    continue
                w = max(0.0, w)
                if abs(w - v) < 1e-15:
                    continue
                t = w - v
                S_new = S + t
                if S_new < 0.01:
                    continue
                
                # Compute new max b
                new_max_b = B0
                if len(lin_ks) > 0:
                    new_lin_max = float((lin_b + 2.0 * t * lin_a).max())
                    if new_lin_max > new_max_b:
                        new_max_b = new_lin_max
                if has_quad:
                    qv = float(b[2 * j] + t * (2.0 * v + t))
                    if qv > new_max_b:
                        new_max_b = qv
                
                F_cand = 2.0 * n * new_max_b / (S_new * S_new)
                if F_cand < best_F_j - 1e-15:
                    best_F_j = F_cand
                    best_w = w
            
            if abs(best_w - v) > 1e-15:
                t = best_w - v
                if len(lin_ks) > 0:
                    b[lin_ks] += 2.0 * t * lin_a
                if has_quad:
                    b[2 * j] += t * (2.0 * v + t)
                a[j] = best_w
                S += t
                improved = True
        
        # Check improvement
        S = a.sum()
        F_cur = 2.0 * n * float(b.max()) / (S * S)
        if F_cur < best_F - 1e-12:
            best_F = F_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and sweep % 100 == 0:
            print(f"    CoordDescent sweep {sweep}: F={best_F:.8f}, no_improve={no_improve}")
        
        if no_improve > 8:
            # Perturb from best
            noise_scale = 0.005 + 0.005 * min(1.0, no_improve / 20.0)
            a = best_a * np.exp(rng.standard_normal(n) * noise_scale)
            a = a / a.sum()
            b = fast_convolve(a)
            no_improve = 0
    
    return best_a, best_F
