def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            grad = grad - grad.mean()
            
            return val, grad
        
        # Run L-BFGS-B
        try:
            res = minimize(func_and_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 200, 'maxfun': 500, 
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxls': 20})
            if res.x is not None:
                x = np.maximum(res.x, 0.0)
                sx = x.sum()
                if sx > 1e-15:
                    x = x / sx
                    obj = fast_convolve(x).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = x.copy()
                        a = x.copy()
        except Exception as e:
            pass
        
        # Increase temperature
        T *= 3.0
        if T > T_final:
            T = T_final
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, 2.0 * n * best_obj


def subgradient_nesterov(a0, time_limit, lr=0.005, mom=0.9, top_k=100):
    """
    Subgradient method with Nesterov acceleration on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    y = a.copy()  # Nesterov lookahead
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov step: compute gradient at y
        c = fast_convolve(y)
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        
        # Uniform weights on peaks (or softmax with small temp)
        pv = c[peaks]
        temp = 0.001
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * n * w[i] * y[k - idx]
        
        # Project subgradient onto tangent space
        sg = sg - sg.mean()
        
        # Update
        v = mom * v + cur_lr * sg
        a_new = project_simplex(a - v)
        
        # Nesterov: y = a_new + (beta_{it+1} - 1) / (1 + beta_{it+1}) * (a_new - a)
        # where beta_t = (1 + sqrt(1 + 4*t^2)) / 2 ... or simpler: beta = (t+2)/(t+1)
        beta = (it + 2.0) / (it + 1.0)
        y = a_new + (1.0 / beta - 1.0) * (a_new - a)
        a = a_new
        
        # Re-evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            cur_lr *= 0.8
            no_improve = 0
        
        # Restart if stuck
        if no_improve > 2000:
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0001)
            no_improve = 0
    
    return best_a, 2.0 * n * best_obj


def coordinate_descent_v2(a0, time_limit):
    """
    Coordinate descent: for each coordinate, find the optimal value
    by 1D search (the objective as a function of one coordinate is 
    piecewise quadratic, so we can find the minimum analytically or by 
    golden section search).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            # If we change a[j] to new_val and scale the rest to maintain sum=1:
            # a'[j] = new_val
            # a'[k] = a[k] * (1 - new_val) / (1 - old_val) for k != j
            # 
            # The new convolution:
            # c'[k] = a'[k] + 2*sum_{j'<k} a'[j']*a'[k-j'] + ... 
            # This is complicated. Let's just do golden section search.
            
            # Objective as function of new_val (with rest scaled):
            # Let alpha = (1 - new_val) / (1 - old_val)
            # a'[j] = new_val, a'[k] = alpha * a[k] for k != j
            # 
            # conv(a', a')[k] = sum_i a'[i] * a'[k-i]
            # = sum_{i != j} sum_{k-i != j} a'[i]*a'[k-i] + a'[j]*a'[k-j] + a'[k-j]*a'[j] - a'[j]*a'[j] (if k-j = j)
            # This is getting complex. Let's use numerical search.
            
            def obj_of(val):
                if val < 0 or val > 1 - 1e-10:
                    return float('inf')
                denom = 1.0 - old_val
                if denom < 1e-10:
                    return float('inf')
                alpha = (1.0 - val) / denom
                a_trial = a.copy()
                a_trial[j] = val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= alpha
                c = fast_convolve(a_trial)
                return c.max()
            
            # Golden section search on [0, min(old_val*3, 0.5)]
            lo = max(0.0, old_val * 0.3)
            hi = min(old_val * 2.0, 0.5)
            if lo >= hi:
                lo, hi = 0.0, old_val
            
            # Check if old value is good
            obj_old = obj_of(old_val)
            if obj_old < best_obj:
                best_obj = obj_old
                improved = True
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_of(a_pt)
            fb = obj_of(b_pt)
            
            for _ in range(40):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_of(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_of(b_pt)
            
            best_val = (lo + hi) / 2.0
            best_obj_1d = min(fa, fb, obj_old)
            
            if best_obj_1d < best_obj - 1e-14:
                # Update
                denom = 1.0 - old_val
                if denom > 1e-10:
                    alpha = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= alpha
                    a = a / a.sum()
                best_obj = best_obj_1d
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, 2.0 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(auto_conv(a))/(sum(a))^2 using a multi-phase strategy:
1. Diverse initialization (analytic forms, parent interpolation, random)
2. Soft-max L-BFGS-B with animated temperature (smooth optimization)
3. Subgradient with Nesterov acceleration (sharp optimization)
4. Efficient coordinate descent with golden-section search (polishing)
5. Final micro-perturbation search
Key innovation: soft-max smoothing enables L-BFGS-B to find basins,
then subgradient + CD polish to convergence.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, top_k=50, temp=0.001):
    """
    Compute subgradient of 2n*max(c) w.r.t. a using weighted peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    k_actual = min(top_k, len(c))
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    
    pv = c[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Build weight vector p of length 2n-1
    p = np.zeros(2 * n - 1)
    for i in range(len(peaks)):
        if w[i] > 1e-14:
            p[int(peaks[i])] += w[i]
    
    # Gradient: 4n * (p cross-corr a)[m] for m=0..n-1
    # (p cross-corr a)[m] = sum_t p[t] * a[t-m] = ifft(fft(p)*conj(fft(a)))[m]
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    grad = 4.0 * n * np.fft.irfft(P * np.conj(A), N)[:n]
    
    return grad


def subgradient_nesterov(a0, time_limit, lr=0.003, mom=0.9, top_k=80, temp=0.001):
    """
    Subgradient method with Nesterov acceleration on the probability simplex.
    Minimizes 2n*max(auto_conv(a)) subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        c = fast_convolve(y)
        grad = compute_subgradient(y, c, top_k=top_k, temp=cur_temp)
        
        # Project gradient to tangent space (zero mean)
        grad = grad - grad.mean()
        
        # Nesterov momentum update
        v = mom * v + cur_lr * grad
        a_new = project_simplex(a - v)
        
        # Nesterov extrapolation
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta_prev = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1))) / 2.0 if it > 1 else 1.0
        gamma = (1.0 - beta) / (1.0 - beta_prev) if beta_prev < 1.0 else 0.0
        y = a_new + gamma * (a_new - a)
        a = a_new
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 300:
            cur_lr *= 0.7
            no_improve = 0
        
        # Gradually increase temperature (sharper peaks)
        cur_temp = max(0.0001, cur_temp * 0.999)
        
        # Restart if stuck
        if no_improve > 3000:
            rng = np.random.default_rng(it * 7 + 3)
            scale = 0.002 + 0.003 * min(no_improve // 5000, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            no_improve = 0
    
    return best_a, 2.0 * n * best_obj

    if T_vals is None:
        T_vals = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    N = 1 << (3 * n - 2).bit_length()
    
    for T in T_vals:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        def func_and_grad(x):
            x = np.maximum(x, 1e-15)
            sx = x.sum()
            x = x / sx
            
            c = fast_convolve(x)
            max_c = c.max()
            exp_c = np.exp(T * (c - max_c))
            sum_exp = exp_c.sum()
            
            if sum_exp < 1e-300:
                val = 2.0 * n * max_c
                p = np.zeros(2 * n - 1)
                p[int(np.argmax(c))] = 1.0
            else:
                log_sum_exp = max_c + np.log(sum_exp) / T
                val = 2.0 * n * log_sum_exp
                p = np.zeros(2 * n - 1)
                p[:len(exp_c)] = exp_c / sum_exp
            
            # Gradient: 4n * (p cross-corr x)
            P_fft = np.fft.rfft(p, N)
            A_fft = np.fft.rfft(x, N)
            grad = 4.0 * n * np.fft.irfft(P_fft * np.conj(A_fft), N)[:n]
            
            # Project to tangent space
            grad = grad - grad.mean()
            
            return val, grad
        
        try:
            max_iter = max(50, int((time_limit * 0.12) / max(0.01, (time.monotonic() - t0) / max(1, len(T_vals)))))
            max_iter = min(max_iter, 300)
            
            res = minimize(func_and_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                   'ftol': 1e-14, 'gtol': 1e-8,
                                   'maxls': 30, 'iprint': -1})
            if res.x is not None:
                x = np.maximum(res.x, 0.0)
                sx = x.sum()
                if sx > 1e-15:
                    x = x / sx
                    obj = fast_convolve(x).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_a = x.copy()
                        a = x.copy()
        except Exception:
            pass
        
        if time.monotonic() - t0 > time_limit * 0.85:
            break
    
    return best_a, 2.0 * n * best_obj


def coordinate_descent(a0, time_limit):
    """
    Coordinate descent with golden-section search per coordinate.
    For coordinate j, tries to find optimal value while scaling others to maintain sum=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-13:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            # Search range for new value of a[j]
            lo = max(0.0, old_val * 0.2)
            hi = min(old_val * 3.0, 0.5)
            if lo >= hi:
                lo, hi = 0.0, min(old_val * 1.5, 0.5)
            
            if hi - lo < 1e-12:
                continue
            
            def obj_of(val):
                if val < 0 or val > 1 - 1e-10:
                    return float('inf')
                alpha = (1.0 - val) / denom
                a_trial = a.copy()
                a_trial[j] = val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= alpha
                c = fast_convolve(a_trial)
                return c.max()
            
            # Golden section search (limit iterations for speed)
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_of(a_pt)
            fb = obj_of(b_pt)
            best_val_1d = old_val
            best_obj_1d = obj_of(old_val)
            
            if fa < best_obj_1d:
                best_obj_1d = fa
                best_val_1d = a_pt
            if fb < best_obj_1d:
                best_obj_1d = fb
                best_val_1d = b_pt
            
            for _ in range(30):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_of(a_pt)
                    if fa < best_obj_1d:
                        best_obj_1d = fa
                        best_val_1d = a_pt
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_of(b_pt)
                    if fb < best_obj_1d:
                        best_obj_1d = fb
                        best_val_1d = b_pt
            
            if best_obj_1d < best_obj - 1e-13:
                alpha = (1.0 - best_val_1d) / denom
                a[j] = best_val_1d
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= alpha
                a = a / a.sum()
                best_obj = best_obj_1d
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, 2.0 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
    Combines soft-max L-BFGS-B, Nesterov subgradient, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0 or a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    # === Phase 0: Generate diverse candidates ===
    print("=== Phase 0: Diverse initialization ===")
    candidates = []
    
    if parent is not None:
        candidates.append((parent.copy(), "parent"))
        # Interpolate parent to other lengths
        for target_n in [512, 640, 800, 1024, 1280]:
            if target_n != len(parent):
                interp = interpolate_sequence(parent, target_n)
                candidates.append((interp, f"parent_n{target_n}"))
    
    # Analytic initializations
    analytic_specs = [
        (384, "uniform"),
        (512, "uniform"),
        (640, "uniform"),
        (768, "uniform"),
        (1024, "uniform"),
        (512, "sine2"),
        (640, "sine2"),
        (768, "sine2"),
        (1024, "sine2"),
        (512, "gauss"),
        (768, "gauss"),
        (1024, "gauss"),
        (512, "cos2"),
        (768, "cos2"),
    ]
    
    for n_t, kind in analytic_specs:
        if time.monotonic() > deadline - 500:
            break
        if kind == "uniform":
            a = np.ones(n_t) / n_t
        elif kind == "sine2":
            a = np.sin(np.pi * np.arange(n_t) / (n_t - 1)) ** 2
        elif kind == "gauss":
            x = np.arange(n_t) - n_t / 2.0
            a = np.exp(-0.5 * (x / (n_t / 5.0)) ** 2)
        elif kind == "cos2":
            a = (0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_t) / n_t)) ** 2
        else:
            a = np.ones(n_t) / n_t
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        candidates.append((a, f"{kind}_{n_t}"))
    
    # Random structured initializations
    for n_t in [512, 768, 1024]:
        if time.monotonic() > deadline - 450:
            break
        # Low-frequency random
        frequencies = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.zeros(n_t)
        for f, ph in zip(frequencies, phases):
            a += np.sin(2 * np.pi * f * np.arange(n_t) / n_t + ph)
        a = np.abs(a) + 0.1
        a /= a.sum()
        candidates.append((a, f"randstruct_{n_t}"))
    
    print(f"  {len(candidates)} candidates generated")
    
    # === Phase 1: Quick screening ===
    print("\n=== Phase 1: Quick screening ===")
    t_per = max(8.0, 350.0 / max(1, len(candidates)))
    screened = []
    
    for a_init, label in candidates:
        rem = deadline - time.monotonic() - 500
        if rem < 5:
            break
        t_alloc = min(t_per, rem * 0.5)
        if t_alloc < 4:
            continue
        
        a_opt, _ = subgradient_nesterov(a_init, time_limit=t_alloc,
                                         lr=0.004, mom=0.9, top_k=50, temp=0.002)
        score = try_seq(a_opt, f"sg_{label}")
        if np.isfinite(score):
            screened.append((a_opt.copy(), score, label))
    
    # Sort and keep top candidates
    screened.sort(key=lambda x: x[1])
    top_n = min(5, len(screened))
    top_candidates = [(s[0], s[2]) for s in screened[:top_n]]
    
    print(f"  Top {top_n} candidates after screening")
    for _, sc, lb in screened[:top_n]:
        print(f"    {lb}: {sc:.6f}")
    
    # === Phase 2: Soft-max L-BFGS-B on top candidates ===
    print(f"\n=== Phase 2: L-BFGS-B optimization (best={best_score:.6f}) ===")
    for a_init, label in top_candidates:
        rem = deadline - time.monotonic() - 350
        if rem < 15:
            break
        t_alloc = min(rem * 0.35, 120)
        print(f"  L-BFGS-B [{label}] n={len(a_init)}, t={t_alloc:.0f}s")
        a_opt, _ = soft_max_lbfgsb(a_init, time_limit=t_alloc,
                                     T_vals=[30, 80, 200, 500])
        try_seq(a_opt, f"lbfgsb_{label}")
    
    # === Phase 3: Deep subgradient on best ===
    print(f"\n=== Phase 3: Deep subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        variants = [
            ("direct", best_seq.copy()),
            ("noise1", project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)),
            ("noise2", project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)),
            ("rev_mix", project_simplex(0.75 * best_seq + 0.25 * best_seq[::-1])),
            ("flat_mix", project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)),
        ]
        
        for label, a_init in variants:
            rem = deadline - time.monotonic() - 250
            if rem < 12:
                break
            t_alloc = min(rem * 0.4, 150)
            a_opt, _ = subgradient_nesterov(a_init, time_limit=t_alloc,
                                             lr=0.002, mom=0.92, top_k=100, temp=0.0005)
            try_seq(a_opt, f"deep_sg_{label}")
    
    # === Phase 4: L-BFGS-B on the current best ===
    print(f"\n=== Phase 4: Final L-BFGS-B (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.6, 150)
            print(f"  L-BFGS-B final t={t_alloc:.0f}s")
            a_opt, _ = soft_max_lbfgsb(best_seq, time_limit=t_alloc,
                                         T_vals=[100, 300, 800, 2000])
            try_seq(a_opt, "lbfgsb_final")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_cd = min(rem * 0.7, 180)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")
            
            # Subgradient after CD
            rem = deadline - time.monotonic() - 60
            if rem > 10:
                a_opt, _ = subgradient_nesterov(best_seq, time_limit=min(rem, 60),
                                                 lr=0.001, mom=0.95, top_k=100, temp=0.0002)
                try_seq(a_opt, "sg_after_cd")
    
    # === Phase 6: Final micro-polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Fine subgradient
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = subgradient_nesterov(best_seq, time_limit=min(rem, 50),
                                             lr=0.0005, mom=0.96, top_k=200, temp=0.0001)
            try_seq(a_opt, "sg_fine")
        
        # Tiny perturbations with quick optimize
        for i in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * i)
            a_pert = project_simplex(best_seq + noise)
            a_opt, _ = subgradient_nesterov(a_pert, time_limit=min(rem, 12),
                                             lr=0.001, mom=0.95, top_k=100, temp=0.0002)
            try_seq(a_opt, f"perturb_{i}")
        
        # One more coordinate descent pass on the polished result
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
