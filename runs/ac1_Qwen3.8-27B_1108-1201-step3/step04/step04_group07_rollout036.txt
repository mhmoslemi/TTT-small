a_new[i] = a[i] * exp(-lr * grad[i]) / Z

def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    n = len(a)

        if i + k < n:

        if i - k >= 0:
            g += a[i - k]
        grad[i] = g
    return grad

Great, so the formula is:
grad[i] = 2 * a[k-i] if 0 <= k-i < n, else 0

This is very simple! And it can be computed efficiently.

For a batch of lags with weights w_k:
grad[i] = 2 * sum_{k: 0<=k-i<n} w_k * a[k-i]
         = 2 * sum_{j: 0<=j<n} w_{i+j} * a[j]  (substituting j = k-i, so k = i+j)
         = 2 * (w * a)[i]  (where * denotes convolution!)

Wait, that's the cross-correlation of w and a! So the gradient is 2 * (w ⊛ a) where ⊛ denotes cross-correlation, which can be computed via FFT!

This is a huge simplification. Let me formalize:

The subgradient of max_k conv[k](a) at point a, using weights w on the active set, is:
grad = 2 * (w ⊛ a) (cross-correlation, computable via FFT in O(n log n))

Where w[k] is a probability distribution over the active lags.

This means each gradient step is O(n log n) instead of O(n * num_active)!

Now let me design the algorithm:

1. Compute conv = conv(a, a) via FFT: O(n log n)
2. Identify active set: top-K lags (or lags within epsilon of max): O(n)
3. Compute weights w: O(n) (e.g., softmax over active lags)
4. Compute gradient: grad = 2 * cross_correlate(w, a) via FFT: O(n log n)
5. Update: a_new = a - lr * grad, project onto {a >= 0, sum = 1}: O(n)

Total per iteration: O(n log n)!

This is much faster than the LP approach for many iterations.

Let me also think about the step size. For the subgradient method, the step size should be decreasing. A common choice is lr_t = c / sqrt(t) (for subgradient) or lr_t = 1/t (for gradient).

But since we're minimizing a non-smooth function (max of smooth functions), the subgradient method is appropriate. We can also use the "mirror descent" version which is often faster:

a_new[i] = a[i] * exp(-lr * grad[i]) / Z

This is the multiplicative weights update (a.k.a. exponentiated gradient), which is the mirror descent with entropy regularization. It naturally keeps a >= 0.

3. Use coordinate descent for final polish

Let me also think about the step size for the gradient method. The objective is:
f(a) = 2n * max_k conv_k(a) / (sum a)^2

If we normalize sum(a) = 1, then f(a) = 2n * max_k conv_k(a).

The "Lipschitz constant" of max_k conv_k is related to the maximum eigenvalue of the Hessian, which is O(n) (since each conv_k is a quadratic form with O(n) non-zero entries).

So a reasonable step size is lr ~ 1/n or smaller.

Actually, for the mirror descent version, the step size is less sensitive. Let me use an adaptive step size:
- Start with lr = 0.1 / n
- If the objective increases, halve lr
- If it decreases consistently, slowly increase lr

"""
Hybrid optimizer: FFT-accelerated subgradient + mirror descent + LP + coordinate descent.
Key innovation: O(n log n) per gradient step using the cross-correlation trick.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def subgradient_update(a, conv, top_k=50, temp=0.01, lr=0.01):
    """One mirror descent step using subgradient of max(conv)."""
    n = len(a)
    # Find active lags
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    
    # Softmax weights
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Build weight vector
    w_full = np.zeros(len(conv))
    w_full[peaks] = w
    
    # Gradient = 2 * cross_correlate(w_full, a)
    # cross_correlate(w, a)[i] = sum_j w[i+j] * a[j] = (w * a_flipped)[i] ... 
    # Actually, cross-correlation: (w ⊛ a)[i] = sum_j w[i+j] * a[j]
    # This is the same as: (FFT(w) * conj(FFT(a)))^{-1} ... 
    # Or: if we define the cross-correlation as sum_j w[j] * a[i+j], that's different.
    #
    # Let me be careful. We want:
    # grad[i] = 2 * sum_k w[k] * a[k-i] * [0 <= k-i < n]
    # Let j = k-i, so k = i+j:
    # grad[i] = 2 * sum_j w[i+j] * a[j] * [0<=j<n, 0<=i+j<2n-1]
    #
    # This is the cross-correlation of w with a, evaluated at lag i.
    # In numpy terms: np.correlate(w, a, mode='full')[i + n - 1]
    # Or via FFT: ifft(conj(fft(w)) * fft(a)) ... let me think.
    #
    # Actually, let's define it more carefully.
    # We have w defined on [0, 2n-2] (the convolution lags) and a on [0, n-1].
    # We want: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (for valid i+j)
    #
    # This is equivalent to: grad = 2 * ifft(conj(fft(w_padded)) * fft(a_padded)) ... 
    # Hmm, let me just use the direct formula or np.correlate.
    #
    # np.correlate(w, a, mode='same') or 'full'
    # np.correlate(w, a, mode='full')[i] = sum_j w[i+j] * conj(a[j]) = sum_j w[i+j] * a[j] (real)
    # So grad[i] = 2 * np.correlate(w, a, mode='full')[i] for i in [0, n-1]
    
    # Using FFT for efficiency:
    # cross-correlation via FFT:
    # corr = ifft(fft(w) * conj(fft(a)))
    # But we need to be careful about the definition.
    #
    # Let's use: corr[i] = sum_j w[j] * a[j+i] (this is one convention)
    # vs. what we want: sum_j w[i+j] * a[j] = sum_j w[j'] * a[j'-i] where j' = j+i
    # = sum_j w[j'] * a[j'-i] which is cross-correlation at lag i.
    
    # Simplest: just compute it directly for now (O(n*k) which is fine for moderate n)
    # Or use FFT:
    # 
    # What we want: G[i] = sum_j w[i+j] * a[j] for j=0..n-1 (and i+j < 2n-1)
    # Let's pad a to length 2n-1 (with zeros) and compute cross-correlation.
    #
    # Actually, the simplest FFT approach:
    # G = ifft(conj(fft(a, size)) * fft(w, size)) ... no, this isn't right either.
    #
    # Let me just use the direct computation since it's O(n * k) and k is small:
    grad = np.zeros(n)
    for idx in range(k):
        wi = w[idx]
        if wi < 1e-15:
            continue
        kk = int(peaks[idx])
        # grad[i] += 2 * wi * a[kk - i] for valid i
        # a[kk-i] is valid when 0 <= kk-i < n, i.e., kk-n+1 <= i <= kk
        i_start = max(0, kk - n + 1)
        i_end = min(n - 1, kk)
        if i_start <= i_end:
            grad[i_start:i_end+1] += 2.0 * wi * a[kk - np.arange(i_start, i_end+1)]
    
    # Mirror descent (multiplicative update)
    # a_new[i] = a[i] * exp(-lr * grad[i]) / Z
    # To avoid overflow, use log-space:
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= lr * grad
    log_a -= log_a.max()  # for numerical stability
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


def lp_direction(a, M, tol=1e-8):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: C @ g <= M where C[i,j] = a[j] if j <= i < j+n else 0
    # conv(a, g)[i] = sum_j a[j] * g[i-j] (for valid j)
    # As a matrix: C[i, j] = a[i-j] if 0 <= i-j < n else 0 ... 
    # Actually: conv(a,g)[i] = sum_{j: 0<=j<n, 0<=i-j<n} a[i-j] * g[j]
    # So C[i, j] = a[i-j] if 0 <= i-j < n, else 0
    
    # For efficiency, use FFT-based approach:
    # We need to find g that maximizes sum(g) subject to:
    #   for all i in [0, 2n-2]: sum_j C[i,j] * g[j] <= M
    #   g[j] >= 0
    #
    # The constraint matrix C is (2n-1) x n, which is large but structured (each row
    # is a shifted version of a).
    #
    # For n=768: C is 1535 x 768. This is manageable for HiGHS.
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    # Build C (this is the expensive part)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        row_start = i
        row_end = min(2 * n - 1, i + n)
        C[row_start:row_end, i] = a[:row_end - row_start]
    
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def coordinate_descent(a, max_sweeps=5, n_candidates=8):
    """Coordinate descent polish."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    conv = fast_convolve(a)
    current_max = conv.max()
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Effect of changing a[i] on conv:
            # delta conv[k] = delta * (a[k-i] if 0<=k-i<n else 0) * 2
            # Wait, from our earlier derivation:
            # d(conv[k])/d(a[i]) = 2 * a[k-i] if 0 <= k-i < n, else 0
            
            effect = np.zeros(2 * n - 1)
            # a[k-i] is valid for k-i in [0, n-1], i.e., k in [i, i+n-1]
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)
            effect[k_start:k_end+1] = 2.0 * a[:k_end - k_start + 1]
            
            # Try candidate values for a[i]
            old = a[i]
            if old < 1e-15:
                candidates = [0.0, 0.5 * np.median(a[np.arange(n) != i]) if n > 1 else 0.0,
                             np.mean(a)]
            else:
                candidates = [old * 0.5, old * 0.7, old * 1.3, old * 1.5,
                             old + 0.01 * old, old - 0.01 * old,
                             np.mean(a), 0.0]
            
            best_val = old
            best_max = current_max
            
            for c_val in candidates:
                c_val = max(0.0, c_val)
                delta = c_val - old
                if abs(delta) < 1e-15:
                    continue
                new_conv_max = (conv + delta * effect).max()
                if new_conv_max < best_max - 1e-15:
                    best_max = new_conv_max
                    best_val = c_val
            
            if best_val != old:
                a[i] = best_val
                # Update conv incrementally
                delta = best_val - old
                conv += delta * effect
                # Also need to update the effect for future i's... 
                # Actually, changing a[i] changes effect[k] for future coordinates j != i.
                # For simplicity, let's recompute effect each time (which we already do).
                current_max = conv.max()
                improved = True
        
        if not improved:
            break
    
    # Normalize
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def get_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Parent (resampled)
    if parent is not None:
        inits['parent'] = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        inits['parent'] = np.maximum(inits['parent'], 0)
        s = inits['parent'].sum()
        if s > 1e-15:
            inits['parent'] /= s
    
    # Linear ramp (asymmetric - this is key!)
    ramp = np.linspace(0.1, 1.0, n)
    inits['ramp'] = ramp / ramp.sum()
    
    # Reverse ramp
    inits['ramp_rev'] = np.linspace(1.0, 0.1, n) / n
    
    # Exponential decay
    exp_decay = np.exp(-2.0 * x / n)
    inits['exp'] = exp_decay / exp_decay.sum()
    
    # Exponential growth
    exp_grow = np.exp(2.0 * x / n)
    inits['exp_grow'] = exp_grow / exp_grow.sum()
    
    # Sine
    sine = 0.5 + 0.5 * np.sin(2.0 * np.pi * x / n - np.pi/2)
    inits['sine'] = sine / sine.sum()
    
    # Two-lobe (asymmetric)
    lobe = 0.3 + 0.7 * np.exp(-0.5 * ((x - 0.3*n) / (0.15*n))**2) + 0.3 * np.exp(-0.5 * ((x - 0.8*n) / (0.1*n))**2)
    inits['asym_lobe'] = lobe / lobe.sum()
    
    # Gaussian (asymmetric - off-center)
    g = np.exp(-0.5 * ((x - 0.4*n) / (0.2*n))**2)
    inits['gauss_off'] = g / g.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Power law
    power = (x + 1) ** 2  # quadratic growth
    inits['power'] = power / power.sum()
    
    return inits


def gradient_optimize(a0, time_limit, top_k=50, temp_init=0.005, lr_init=None, name=""):
    """Mirror descent with subgradient for max(conv) minimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr_init is None:
        lr_init = 5.0 / n
    
    lr = lr_init
    temp = temp_init
    
    conv = fast_convolve(a)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    patience = 5000
    
    while time.monotonic() - t0 < time_limit:
        a = subgradient_update(a, conv, top_k=top_k, temp=temp, lr=lr)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        iteration += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve % 1000 == 0:
            lr *= 0.95
