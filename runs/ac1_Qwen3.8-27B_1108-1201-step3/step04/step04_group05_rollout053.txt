"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a continuation method with
log-sum-exp smoothing, L-BFGS in softmax parameterization, multiplicative
weights (mirror descent), and LP-based Frank-Wolfe for large jumps.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_a(a, conv, temp):
    """Compute gradient of log-sum-exp smoothed objective w.r.t. a."""
    n = len(a)
    # Softmax weights over conv lags
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # This is cross-correlation of w (len 2n-1) with a (len n), first n values
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    return 2.0 * cross, w


def lse_objective_and_grad_b(b, temp):
    """Compute LSE smoothed objective and gradient in b-space (softmax param)."""
    # Softmax
    b_shifted = b - b.max()
    a = np.exp(b_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return 1e300, np.zeros_like(b)
    a = a / a_sum
    
    conv = fast_convolve(a)
    
    # LSE value
    scaled = conv / temp
    scaled -= scaled.max()
    lse_val = temp * (scaled.max() + np.log(np.exp(scaled).sum()))
    
    # Gradient in a-space
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * cross
    
    # Gradient in b-space: a * (grad_a - <grad_a, a>)
    ga_dot_a = np.dot(grad_a, a)
    grad_b = a * (grad_a - ga_dot_a)
    
    return lse_val, grad_b


def continuation_lse(a0, time_limit, verbose=False):
    """Continuation method: start with smooth LSE, anneal temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv0 = fast_convolve(a)
    max0 = conv0.max()
    
    # Temperature schedule
    t_start = max0 * 0.5
    t_end = max0 * 1e-5
    num_phases = 12
    
    temps = np.geomspace(t_start, max(t_end, 1e-8), num_phases)
    
    # Initial b
    a_safe = np.maximum(a, 1e-15)
    b = np.log(a_safe / a_safe.sum())
    
    best_obj = max0
    best_a = a.copy()
    t_total = time.monotonic()
    
    for phase, temp in enumerate(temps):
        if time.monotonic() - t_total > time_limit - 5:
            break
        
        # Run L-BFGS at this temperature
        time_per_phase = (time_limit - (time.monotonic() - t_total)) / (num_phases - phase)
        
        def func_and_grad(b_vec):
            val, grad = lse_objective_and_grad_b(b_vec, temp)
            return val, grad
        
        try:
            res = minimize(func_and_grad, b, method='L-BFGS-B',
                          jac=True,  # grad is True since func_and_grad returns (f, g)
                          options={'maxiter': 200, 'maxfun': 500, 'factr': 1e7})
            if res.success or res.x is not None:
                b = res.x
                a_new = np.exp(b - b.max())
                a_new /= a_new.sum()
                conv_new = fast_convolve(a_new)
                obj_new = conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
        
        if verbose:
            print(f"    Phase {phase}: t={temp:.2e}, obj={best_obj:.10f}")
    
    return best_a, best_obj

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)

    n = len(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross, w


def lse_obj_grad_b(b_vec, temp):
    """LSE objective and gradient in b-space (softmax parameterization)."""
    b_shifted = b_vec - b_vec.max()
    a = np.exp(b_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return 1e300, np.zeros_like(b_vec)
    a = a / a_sum

    conv = fast_convolve(a)
    scaled = conv / temp
    scaled_max = scaled.max()
    exp_s = np.exp(scaled - scaled_max)
    lse_val = temp * (scaled_max + np.log(exp_s.sum()))

    w = exp_s / exp_s.sum()

    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * cross

    ga_dot_a = np.dot(grad_a, a)
    grad_b = a * (grad_a - ga_dot_a)

    return lse_val, grad_b


def continuation_lse(a0, time_limit, verbose=False):
    """Continuation method: LSE smoothing with L-BFGS in b-space, annealing temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv0 = fast_convolve(a)
    max0 = conv0.max()

    t_start = max0 * 0.3
    t_end = max0 * 1e-6
    num_phases = 15

    temps = np.geomspace(t_start, max(t_end, 1e-10), num_phases)

    a_safe = np.maximum(a, 1e-15)
    b = np.log(a_safe) - np.log(a_safe.sum())

    best_obj = max0
    best_a = a.copy()
    t0 = time.monotonic()

    for phase, temp in enumerate(temps):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        time_per_phase = remaining / (num_phases - phase)
        maxiter = max(50, int(300 * time_per_phase / 30))

        try:
            res = minimize(lse_obj_grad_b, b, args=(temp,),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                                    'factr': 1e8, 'gtol': 1e-8})
            if res.x is not None:
                b = res.x
                a_new = np.exp(b - b.max())
                a_new = np.maximum(a_new, 0)
                a_new_sum = a_new.sum()
                if a_new_sum > 1e-15:
                    a_new = a_new / a_new_sum
                    conv_new = fast_convolve(a_new)
                    obj_new = conv_new.max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

        if verbose:
            print(f"      ContLSE phase {phase}: t={temp:.2e}, obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a, best_obj


def mirror_descent_lse(a0, time_limit, lr_init=0.05, temp_init_factor=0.01, temp_end_factor=1e-5, decay=0.9995, verbose=False):
    """Mirror descent (multiplicative weights) with LSE smoothing and temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv0 = fast_convolve(a)
    max0 = conv0.max()

    temp = max0 * temp_init_factor
    temp_end = max0 * temp_end_factor
    lr = lr_init

    best_obj = max0
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        cur_obj = conv.max()

        grad_a, w = compute_lse_grad_a(a, conv, temp)

        # Multiplicative update
        log_update = -lr * grad_a
        # Clip to avoid overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a = best_a.copy()
            no_improve += 1
        else:
            a_new = a_new / a_new_sum
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
                # Slightly increase lr if improving
                lr = min(lr * 1.002, 0.5)
            else:
                no_improve += 1
                # Decrease lr if not improving
                lr = max(lr * 0.99, 0.001)

        # Anneal temperature
        if temp > temp_end:
            temp *= decay
            if temp < temp_end:
                temp = temp_end

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
            temp = max(temp * 2, max0 * 0.001)

        if verbose and it % 500 == 0:
            print(f"      MirrDesc it {it}: obj={best_obj:.10f}, t={temp:.2e}, lr={lr:.4f}")

    return best_a, best_obj


def solve_linearized_lp(f, conv_f, top_k=None):
    """Solve linearized LP for Frank-Wolfe direction."""
    n = len(f)
    if top_k is None or top_k >= len(conv_f):
        peaks = np.arange(len(conv_f))
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
        peaks = np.sort(peaks)

    B = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    num_constraints = len(peaks)
    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve max-sum LP for Frank-Wolfe direction."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def fw_optimize(f0, time_limit, use_linearized=True, reduced_k=30, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()

        if use_linearized:
            g = solve_linearized_lp(f, conv, top_k=reduced_k)
        else:
            g = solve_maxsum_lp(f, current_obj * m_exp)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                else:
                    g = None

        if g is None:
            no_improve += 5
            if not use_linearized:
                m_exp = min(m_exp * 1.15, 5.0)
            if no_improve > 80:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                no_improve = 0
                m_exp = 1.0
            continue

        target = f + g if use_linearized else g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * target
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                m_exp = 1.0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + no_improve * 0.0002))
            no_improve = 0
            m_exp = 1.0

        if verbose and it % 50 == 0:
            print(f"      FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(a0, time_limit, rng, num_sweeps=None, verbose=False):
    """Coordinate-wise descent: try adjusting each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Try different values for a[i]
            old_val = a[i]
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                          best_a[i] * 0.8, best_a[i] * 1.2]
            # Add a small random perturbation
            candidates.append(old_val + rng.standard_normal() * 0.002)

            best_trial = old_val
            best_trial_obj = best_obj

            for cv in candidates:
                cv = max(0.0, cv)
                a[i] = cv
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s
                obj = fast_convolve(a_norm).max()
                if obj < best_trial_obj:
                    best_trial_obj = obj
                    best_trial = cv

            a[i] = best_trial
            s = a.sum()
            if s > 1e-15:
                a = a / s
                if best_trial_obj < best_obj:
                    best_obj = best_trial_obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            print(f"      CoordDesc sweep {sweep}: obj={best_obj:.10f}")

        # Periodic restart
        if sweep % 20 == 0:
            noise = rng.standard_normal(n) * 0.003
            a_pert = project_simplex(best_a + noise)
            obj_pert = fast_convolve(a_pert).max()
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
            a = best_a.copy()

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # Asymmetric: concentrated at one end with exponential decay
    for decay_rate in [0.02, 0.05, 0.1, 0.2]:
        e = np.exp(-decay_rate * x)
        inits.append(e / e.sum())

    # Cosine
    c1 = np.abs(np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits.append(rc / rc.sum())

    # Two bumps
    for pos in [0.25, 0.35]:
        b1 = np.exp(-0.5 * ((x - pos * n) / (0.1 * n)) ** 2)
        b2 = np.exp(-0.5 * ((x - (1 - pos) * n) / (0.1 * n)) ** 2)
        two_bump = b1 + b2 + 0.01
        inits.append(two_bump / two_bump.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining:
    1. Diverse initialization
    2. Continuation method (LSE + L-BFGS in b-space)
    3. Mirror descent with LSE smoothing
    4. LP-based Frank-Wolfe
    5. Coordinate descent
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a.astype(np.float64), 0.0)
        if a.sum() < 1e-15:
            return float('inf')
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            pscore = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={pscore:.10f}")
            if np.isfinite(pscore):
                best_score = pscore
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t1_start = time.monotonic()

    n_parent = len(parent) if parent is not None else 384

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent))

    inits = generate_inits(n_parent, parent, rng)
    for i, ini in enumerate(inits[:8]):
        candidates.append((f"init_{i}", ini))

    # Quick eval
    for label, cand in candidates:
        if time.monotonic() > deadline - 1000:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback_uniform")

    t1_elapsed = time.monotonic() - t1_start
    print(f"  Phase 1 done: {t1_elapsed:.1f}s, best={best_score:.10f}")

    # === Phase 2: Continuation LSE + L-BFGS ===
    print("\n=== Phase 2: Continuation LSE + L-BFGS ===")
    t2_start = time.monotonic()
    remaining = deadline - time.monotonic() - 500
    t2_alloc = min(remaining * 0.35, 600)

    if t2_alloc > 20:
        print(f"  Running continuation LSE, budget={t2_alloc:.0f}s")
        a_opt, _ = continuation_lse(best_seq, time_limit=t2_alloc * 0.6, verbose=True)
        try_seq(a_opt, "cont_lse")

        # Run again from the improved point
        rem = deadline - time.monotonic() - 450
        if rem > 20:
            a_opt, _ = continuation_lse(best_seq, time_limit=min(rem * 0.5, 200), verbose=False)
            try_seq(a_opt, "cont_lse_2")

    print(f"  Phase 2 done: {time.monotonic() - t2_start:.1f}s, best={best_score:.10f}")

    # === Phase 3: Mirror Descent ===
    print("\n=== Phase 3: Mirror Descent ===")
    t3_start = time.monotonic()
    remaining = deadline - time.monotonic() - 400
    t3_alloc = min(remaining * 0.3, 400)

    if t3_alloc > 20:
        # Multiple mirror descent runs with different parameters
        configs = [
            {'lr_init': 0.03, 'temp_init_factor': 0.02, 'temp_end_factor': 1e-5, 'decay': 0.999},
            {'lr_init': 0.01, 'temp_init_factor': 0.005, 'temp_end_factor': 5e-6, 'decay': 0.9995},
            {'lr_init': 0.05, 'temp_init_factor': 0.05, 'temp_end_factor': 1e-4, 'decay': 0.998},
        ]

        t_per_config = t3_alloc / len(configs)
        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 300:
                break
            print(f"  MirrDesc config {ci}, budget={t_per_config:.0f}s")
            a_opt, _ = mirror_descent_lse(best_seq, time_limit=t_per_config * 0.8, **cfg, verbose=False)
            try_seq(a_opt, f"mirr_desc_{ci}")

    print(f"  Phase 3 done: {time.monotonic() - t3_start:.1f}s, best={best_score:.10f}")

    # === Phase 4: LP-based Frank-Wolfe ===
    print("\n=== Phase 4: LP Frank-Wolfe ===")
    t4_start = time.monotonic()
    remaining = deadline - time.monotonic() - 250
    t4_alloc = min(remaining * 0.3, 300)

    if t4_alloc > 20:
        # Linearized LP FW
        t_fw1 = min(t4_alloc * 0.5, 150)
        print(f"  Linearized LP FW, budget={t_fw1:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_fw1, use_linearized=True, reduced_k=30)
        try_seq(a_opt, "lp_fw_lin")

        # Max-sum LP FW
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_fw2 = min(rem * 0.5, 100)
            print(f"  Max-sum LP FW, budget={t_fw2:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_fw2, use_linearized=False)
            try_seq(a_opt, "lp_fw_max")

    print(f"  Phase 4 done: {time.monotonic() - t4_start:.1f}s, best={best_score:.10f}")

    # === Phase 5: Coordinate Descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t5_start = time.monotonic()
    remaining = deadline - time.monotonic() - 150
    t5_alloc = min(remaining * 0.4, 200)

    if t5_alloc > 15:
        print(f"  Coordinate descent, budget={t5_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t5_alloc, rng=rng, verbose=False)
        try_seq(a_opt, "coord_desc")

    print(f"  Phase 5 done: {time.monotonic() - t5_start:.1f}s, best={best_score:.10f}")

    # === Phase 6: Final polish and multi-length ===
    print("\n=== Phase 6: Final Polish ===")
    t6_start = time.monotonic()
    remaining = deadline - time.monotonic() - 30

    # Quick mirror descent polish
    if remaining > 30:
        t_polish = min(remaining * 0.3, 60)
        a_opt, _ = mirror_descent_lse(best_seq, time_limit=t_polish,
                                       lr_init=0.005, temp_init_factor=0.001,
                                       temp_end_factor=1e-6, decay=0.9998)
        try_seq(a_opt, "final_polish")

    # Try nearby lengths
    rem = deadline - time.monotonic() - 20
    n_best = len(best_seq) if best_seq is not None else 384
    for n_try in [n_best - 16, n_best + 16, n_best - 32, n_best + 32]:
        if n_try < 64 or rem < 15:
            continue
        px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            # Quick optimization at this length
            t_small = min(rem * 0.3, 30)
            a_opt, _ = mirror_descent_lse(px, time_limit=t_small,
                                           lr_init=0.01, temp_init_factor=0.005,
                                           temp_end_factor=1e-5, decay=0.999)
            try_seq(a_opt, f"len_{n_try}")
            rem = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
