def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_subgradient(a, c, temperature, top_k=None):
    """Compute softmax-weighted subgradient of max(c) using top-k peaks."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = peek / max(temperature, 1e-30)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    
    return sg

def nesterov_eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                         temp_decay=0.9997, eta_decay=0.99995,
                         top_k_frac=0.25, momentum0=0.9,
                         momentum_decay=0.99999, verbose=False):
    """Nesterov-accelerated exponentiated gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Nesterov momentum
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead point
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    
    eta = eta0
    temp = temp0
    momentum = momentum0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov: compute gradient at lookahead point y
        sg = compute_smooth_subgradient(y, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        # Update momentum
        v = momentum * v - eta * sg / sg_max
        
        # Nesterov step: y = a + v (in log-space for multiplicative weights)
        # Actually, let's do it in the multiplicative weights framework
        # a_new[i] proportional to a[i] * exp(v[i])
        log_update = v
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        # Update Nesterov lookahead
        y = a_new + (momentum / (1 - momentum)) * (a_new - a) if momentum > 0.5 else a_new
        # Actually simpler: y = a_new + momentum * (a_new - a)
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 1e-300)
        ys = y.sum()
        if ys > 1e-300:
            y = y / ys
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    NE iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        # Periodic check of running average
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        eta *= eta_decay
        temp *= temp_decay
        momentum = min(momentum * momentum_decay, 0.99)
        
        # Stagnation: restart
        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta, eta0 * 0.05)
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0
            v = np.zeros(n)
            y = a.copy()
            momentum = momentum0 * 0.5
    
    return best_a.copy(), best_obj

        p = 1.0 + 0.3*np.cos(2*np.pi*f1*x/n) + 0.3*np.cos(2*np.pi*f2*x/n)
        p = np.maximum(p, 0.01)
        inits.append(p / p.sum())
    
    # Random smooth (low-pass filtered noise)
    for _ in range(3):
        noise = rng.standard_normal(n)

    return inits

def local_search_polish(a0, time_limit, max_rounds=200):
    """Local search with line search on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        
        n_perturb = max(1, n // 8)

        for i in coords:

            if old_val < 1e-15:
                continue
            
            # Try a range of multiplicative factors
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3,

def solve_qcqp_cvxpy(a0, time_limit=120):
    """Solve QCQP using CVXPY."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    a_norm = a / s
    n = len(a)
    
    if n > 300:
        return None
    
    t_start = time.monotonic()
    
    try:
        x = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [
            cp.sum(x) == 1,
            x >= 0
        ]
        
        for k in range(2 * n - 1):
            expr = cp.Constant(0)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                expr = expr + x[j] * x[k - j]
            constraints.append(expr <= t)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                t_check = time.monotonic()
                if prob.status is None or prob.status not in ['optimal', 'optimal_inaccurate']:
                    prob.solve(solver=solver_name)
                else:
                    break
                solve_time = time.monotonic() - t_check
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    print(f"    QCQP {solver_name}: t={solve_time:.1f}s")
                    break
            except Exception:
                continue
        
        if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
            result = np.maximum(x.value, 0)
            rs = result.sum()
            if rs > 1e-10:
                result = result / rs
                return result
    except Exception as e:
        print(f"    QCQP failed: {e}")
    
    return None

    len_c = len(c)

    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()

    scaled = peek / max(temperature, 1e-30)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def compute_lse_gradient(a, c, temperature):
    """Compute gradient of log-sum-exp(c/temperature) w.r.t. a."""
    n = len(a)
    scaled = c / max(temperature, 1e-30)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-15:
            continue
        k = int(idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.004,
                temp_decay=0.9997, eta_decay=0.99995,
                top_k_frac=0.3, use_momentum=False, momentum0=0.85,
                lse_mode=False, verbose=False):
    """Exponentiated gradient with optional momentum and LSE mode."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0

    v = np.zeros(n) if use_momentum else None
    mom = momentum0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if lse_mode:
            sg = compute_lse_gradient(a, c, temperature=temp)
        else:
            sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        if use_momentum and v is not None:
            v = mom * v - eta * sg / sg_max
            log_update = v
        else:
            log_update = -eta * sg / sg_max

        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}, mom={mom:.3f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay
        if use_momentum:
            mom = min(mom * 0.99998, 0.99)

        if no_improve > 2500:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta, eta0 * 0.05)
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0
            if v is not None:
                v = np.zeros(n)
            mom = momentum0 * 0.5

    return best_a.copy(), best_obj


def projected_grad_optimize(a0, time_limit, lr0=0.01, lr_decay=0.9999,
                            lse_temp0=0.01, lse_temp_decay=0.9995,
                            verbose=False):
    """Projected gradient descent with LSE-smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy()
    count = 0
    lr = lr0
    lse_temp = lse_temp0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad = compute_lse_gradient(a, c, temperature=lse_temp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_new = a - lr * grad / grad_norm
        a_new = project_simplex(a_new)

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    PG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, lse_temp={lse_temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        lr *= lr_decay
        lse_temp *= lse_temp_decay

        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 17 + 3)
            a = best_a + rng_local.standard_normal(n) * 0.005
            a = project_simplex(a)
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            lr = lr0 * 0.1
            lse_temp = max(lse_temp, lse_temp0 * 0.1)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def local_search_polish(a0, time_limit, max_rounds=300):
    """Local search with multiplicative perturbations on random coordinate subsets."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    no_improve = 0

    factors = [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 0.95,
               1.05, 1.1, 1.25, 1.5, 2.0, 2.5, 3.0]

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False

        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue

            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a.copy(), best_obj


def solve_qcqp_cvxpy(a0, time_limit=120):
    """Solve the QCQP using CVXPY for small n."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    a_norm = a / s
    n = len(a)

    if n > 250:
        return None

    try:
        x = cp.Variable(n)
        t = cp.Variable()

        constraints = [
            cp.sum(x) == 1,
            x >= 0
        ]

        for k in range(2 * n - 1):
            expr = cp.Constant(0.0)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                expr = expr + x[j] * x[k - j]
            constraints.append(expr <= t)

        prob = cp.Problem(cp.Minimize(t), constraints)

        for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                if prob.status not in ['optimal', 'optimal_inaccurate']:
                    prob.solve(solver=solver_name)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    break
            except Exception:
                continue

        if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
            result = np.maximum(x.value, 0)
            rs = result.sum()
            if rs > 1e-10:
                result = result / rs
                return result
    except Exception as e:
        print(f"    QCQP failed: {e}")

    return None


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Parent (if same length)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    # Uniform
    inits.append(np.ones(n) / n)

    # Smooth random (low-pass filtered)
    for _ in range(4):
        noise = rng.standard_normal(n)
        spec = np.fft.rfft(noise)
        cutoff = max(2, n // 8)
        spec[cutoff:] = 0
        filtered = np.fft.irfft(spec, n)
        a = 1.0 + 0.4 * filtered / (np.abs(filtered).max() + 1e-10)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    # Exponential decay (both directions)
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Multi-frequency
    for f1, f2 in [(2, 5), (3, 7), (1, 4)]:
        p = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x / n) + 0.2 * np.cos(2 * np.pi * f2 * x / n)
        p = np.maximum(p, 0.01)
        inits.append(p / p.sum())

    # Dirichlet
    for alpha in [0.1, 0.3, 0.5]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Perturbed parent
    if parent is not None and len(parent) == n:
        for scale in [0.002, 0.01, 0.05]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
