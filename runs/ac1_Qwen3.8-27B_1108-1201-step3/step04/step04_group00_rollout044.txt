def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:

def project_simplex(v):

            if old_val < 1e-13:
                continue

            cur_peak = float(fast_convolve(f).max())
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = float(fast_convolve(f_trial).max())
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=40, temp=0.00004):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        # Ensure it's different enough
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            continue
            
        f = f_new
        conv = fast_convolve(f)
        co = float(conv.max())

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 300:
            rng = np.random.default_rng((it * 7 + 3) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, spread=None):
    """Create a Gaussian-like initial sequence."""
    if spread is None:
        spread = n / 6.0
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / spread) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_triangular(n):
    """Create a triangular initial sequence."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(x - center) / (center + 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_topped(n, width_frac=0.6):
    """Create a flat-topped (raised cosine) initial sequence."""
    x = np.arange(n) / (n - 1)
    a = np.ones(n)
    edge = (1.0 - width_frac) / 2.0
    mask_left = x < edge
    mask_right = x > 1.0 - edge
    a[mask_left] = 0.5 * (1 + np.cos(np.pi * x[mask_left] / edge))
    a[mask_right] = 0.5 * (1 + np.cos(np.pi * (1.0 - x[mask_right]) / edge))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: fast LP-FW with top-k constraints on multiple
    sequence lengths, diverse initialization, multi-stage polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}, t={time.monotonic() :.0f})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)

    print(f"Initial best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 1: LP-FW on parent length ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    t1 = min(250, deadline - time.monotonic() - 600)
    if t1 > 40:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, top_k=50, frac=0.93,
                                  momentum=0.25, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Try different initializations at parent length ===
    print(f"\n=== Phase 2: Diverse init at n={n_parent} ===")
    inits = [
        ("gauss", make_gaussian(n_parent, n_parent / 5.0)),
        ("tri", make_triangular(n_parent)),
        ("flat", make_flat_topped(n_parent, 0.7)),
        ("flat_narrow", make_flat_topped(n_parent, 0.4)),
    ]
    for label, init in inits:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(rem * 0.4, 80)
        a_opt, _ = lp_fw_optimize(init, time_limit=t_alloc, top_k=40, frac=0.94,
                                  momentum=0.2, verbose=False)
        try_seq(a_opt, f"init_{label}")

    # === Phase 3: Longer sequences ===
    print(f"\n=== Phase 3: Longer sequences ===")
    base = best_seq.copy()
    
    for target_n in [1024, 1280]:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 180)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, top_k=60, frac=0.93,
                                  momentum=0.25, verbose=True)
        try_seq(a_opt, f"long_n{target_n}")

    # === Phase 4: Deep optimization on current best ===
    print(f"\n=== Phase 4: Deep Optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 5 - p), 150)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Reverse
                f_init = best_seq[::-1].copy()
            elif p == 3:
                # Mix with uniform
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Different top_k and frac
                f_init = best_seq.copy()

            tk = 50 if p != 4 else 80
            fc = 0.93 if p != 4 else 0.90
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, top_k=tk, frac=fc,
                                      momentum=0.3, verbose=(p == 0))
            try_seq(a_opt, f"deep_{p}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_cd = min(rem * 0.6, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: Subgradient polishing ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_sg = min(rem * 0.7, 100)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0015, mom=0.94,
                                 top_k=max(20, n_best // 6), temp=0.00003)
            try_seq(a_opt, "sg")

            # Another round with different params
            rem = deadline - time.monotonic() - 40
            if rem > 10:
                a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 50),
                                     lr=0.0008, mom=0.96,
                                     top_k=max(10, n_best // 10), temp=0.00001)
                try_seq(a_opt, "sg_fine")

    # === Phase 7: Final LP-FW on polished best ===
    print(f"\n=== Phase 7: Final LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 60),
                                      top_k=80, frac=0.90, momentum=0.4)
            try_seq(a_opt, "final_lpfw")

    # === Phase 8: Tiny perturbations and quick polish ===
    print(f"\n=== Phase 8: Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 15)
            a_opt, _ = sg_polish(f_pert, time_limit=t_alloc,
                                 lr=0.001, mom=0.95,
                                 top_k=max(15, n_best // 8), temp=0.00002)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
