Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w):
    """Compute gradient of weighted sum of convolution peaks.
    
    w: weight for each lag k
    Returns grad[j] = sum_k w_k * ∂b_k/∂a_j
    """
    n = len(a)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_idx = np.arange(js, je + 1)
        contrib = 2.0 * a[k - j_idx]
        self_mask = (2 * j_idx == k)
        if self_mask.any():
            contrib[self_mask] = a[j_idx[self_mask]]
        grad[j_idx] += wk * contrib
    return grad


def softmax_mirror_descent(f0, time_limit, beta_start=200, beta_end=20000,

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            # Try multiplicative changes
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def ensemble_average(solutions, weights=None):
    """Average multiple solutions, all must have same length."""
    n = len(solutions[0])
    if weights is None:
        weights = np.ones(len(solutions)) / len(solutions)
    result = np.zeros(n)
    for sol, w in zip(solutions, weights):
        result += w * sol
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

- `A` has the constraint that `ifft(A)` is real and non-negative.
- We want `max(ifft(A.^2))` to be small.

This is a complex optimization problem. Let me think of a simpler approach.

**Alternative: Direct time-domain optimization with a different parameterization**

Instead of the simplex constraint, use a soft parameterization:
`a_j = exp(θ_j) / sum_i exp(θ_i)` (softmax)

This automatically ensures `a_j > 0` and `sum(a) = 1`. The optimization is over `θ ∈ R^n`.

The objective: `f(θ) = 2n * max_k (a(θ) * a(θ))_k`

We can use gradient descent on `θ` with the softmax-smoothed max.

The gradient: `∂f/∂θ_j = (df/da) * (∂a/∂θ)`
where `∂a_i/∂θ_j = a_i (δ_{ij} - a_j)` (Jacobian of softmax).

This might be more effective than mirror descent because it doesn't have the "sticking to zero" problem.

The O(n^2) Jacobian is a problem for large n. Let me think of a way around this.

Actually, `∂a_i/∂θ_j = a_i (δ_{ij} - a_j)`. So:
`grad_theta_j = sum_i grad_a_i * a_i (δ_{ij} - a_j) = a_j * sum_i grad_a_i * a_i - a_j * grad_a_j`
`= a_j * (dot(grad_a, a) - grad_a_j)`

This is O(n) per component, O(n^2) total. For n=768, that's about 600K operations, which is fine.

Actually wait, that's O(n) total since we just need `dot(grad_a, a)` (O(n)) and then `grad_theta = a * (dot - grad_a)` (O(n)).

Let me redo: 
`grad_theta_j = a_j * (sum_i grad_a_i * a_i - grad_a_j)`

Let `c = sum_i grad_a_i * a_i` (a scalar, O(n)).
Then `grad_theta = a * (c - grad_a)` (O(n)).

Great, so the full gradient computation is O(n + cost_of_convolution).

Let me implement this properly. This is essentially "natural gradient descent" or "Fisher scoring" on the simplex.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Natural gradient descent (softmax parameterization) with smooth-max
2. LP-based Frank-Wolfe for refinement
3. Coordinate descent polishing
4. Multi-length exploration and ensemble averaging
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(v):
    """Stable softmax."""
    v_shifted = v - v.max()
    e = np.exp(v_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_conv_gradient(a, weights):
    """Compute gradient of sum_k weights[k] * b_k where b = a*a.
    
    grad[j] = sum_k weights[k] * (a_{k-j} + a_{j-k}) [with a_i=0 outside [0,n-1]]
             = sum_k weights[k] * 2*a_{k-j} - sum_k weights[k] * a_{k-j} * [2j==k]
    Actually: grad[j] = sum_k weights[k] * (a_{k-j} + a_{j-k})
    For j-k < 0: a_{j-k} = 0 (out of range)
    For k-j < 0: a_{k-j} = 0 (out of range)
    So effectively: grad[j] = sum_{k: 0<=k-j<n} weights[k] * a_{k-j} + sum_{k: 0<=j-k<n} weights[k] * a_{j-k}
    First sum: cross-correlation of weights and a
    Second sum: convolution of weights and a
    """
    n = len(a)
    L = 2 * n - 1
    fft_size = 1 << (L - 1).bit_length()
    
    # First term: sum_k w[k] * a[k-j] for valid k,j
    # This is (w * a_flipped)[j + offset] or cross-correlation
    # Let's compute via FFT
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # Cross-correlation: sum_k w[k]*a[k-j] = ifft(conj(W) * A) ... 
    # Actually: sum_k w[k] * a[k-j] = (w ⋆ a)[j] (cross-correlation)
    # In FFT: cross-correlation = ifft(conj(FFT(w)) * FFT(a))
    corr1 = np.fft.irfft(np.conj(W) * A, fft_size)
    
    # Second term: sum_k w[k] * a[j-k] = (w * a)[j] (convolution)
    conv2 = np.fft.irfft(W * A, fft_size)
    
    grad = (corr1[:n] + conv2[:n]).real
    
    # But we need to handle the self-term correctly
    # When k = 2j, the term a_{k-j} = a_j is counted once, not twice
    # So we need to subtract a_j * w[2j] for even k=2j
    for j in range(n):
        k = 2 * j
        if k < L:
            grad[j] -= weights[k] * a[j]
    
    return grad


def natural_gradient_descent(f0, time_limit, beta_start=100, beta_end=50000,
                             lr_start=0.01, verbose=False):
    """Natural gradient descent with softmax parameterization and smooth max."""
    n = len(f0)
    f_init = np.maximum(f0, 1e-15)
    f_init = f_init / f_init.sum()
    theta = np.log(f_init)
    
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Progressively increase beta
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.5)
        beta = max(beta, 50.0)
        
        a = softmax(theta)
        b = fast_convolve(a)
        cur_max = b.max()
        
        # Check best
        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_f = a.copy()
            no_improve = 0
        
        # Softmax weights for smooth max
        scaled = beta * b
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = exp_vals / w_sum
        
        # Gradient of smooth max w.r.t. a
        grad_a = compute_conv_gradient(a, w)
        
        # Chain rule through softmax: grad_theta_j = a_j * (sum_i grad_a_i * a_i - grad_a_j)
        c = np.dot(grad_a, a)
        grad_theta = a * (c - grad_a)
        
        # Gradient descent on theta
        theta -= lr * grad_theta
        
        # Adaptive learning rate
        if cur_max < best_obj:
            lr = min(lr * 1.002, lr_start * 3)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr = max(lr * 0.8, lr_start * 0.1)
            if no_improve > 300:
                # Restart
                rng = np.random.default_rng(it * 3 + 1)
                f_pert = f_init + rng.standard_normal(n) * 0.003
                f_pert = project_simplex(f_pert)
                theta = np.log(np.maximum(f_pert, 1e-15))
                no_improve = 0
                lr = lr_start
        
        if verbose and it % 500 == 0:
            print(f"    NGD iter {it}: obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.6f}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
