"""
Optimization of  O(a) = 2*n*max(a*a) / (sum a)^2  for a >= 0.

Key ideas (differs from the previous LP/top-k-subgradient program):
  * Explicit, fully analytic gradient of the log-sum-exp (LSE) smooth
    surrogate  phi_T(a) = (1/T) log sum_k exp(T (a*a)_k).
    Using c = a*a and p = softmax(T*c) one can show
        grad phi_T = 2 * np.correlate(p, a, mode='valid')
    which is exact and O(n^2) (or FFT) per step -- no top-k truncation.
  * Adam with backtracking line-search performed on the *true* (non-smooth)
    objective, plus exponential temperature annealing from high T (flat,
    smooth) to low T (sharp, peak-reducing).  This drives the local optimum
    much tighter than a pure top-k subgradient.
  * Periodic classical LP "peak-equalisation" kick (max sum g s.t. a*g <= M)
    to make non-local redistribution moves that pure gradient can't.
  * Basin-hopping (diversify + re-optimise) at the end so we can escape the
    basin the parent construction sits in -- this is the main mechanism for
    beating 1.5059.
  * Larger lengths (up to 2048) are explored because the bound improves with n.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ---------------- low level helpers (top level, no closures/lambdas) -------

def _fft_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def true_obj(a):
    """Return (O, c) for the true objective with a assumed ~ nonneg."""
    n = a.size
    c = _fft_conv(a)
    s = a.sum()
    if s <= 1e-15:
        return np.inf, c
    return 2.0 * n * c.max() / (s * s), c


def project(a):
    """Project onto a>=0, sum a = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-15:
        return a / a.size
    return a / s


def lse_grad(c, T):
    """Return (phi_T, p, grad) where grad = 2*correlate(p,a) is passed in a."""
    tc = T * c
    tc = tc - tc.max()
    e = np.exp(tc)
    p = e / e.sum()
    phi = (tc.max() + np.log(e.sum())) / T
    return phi, p


def grad_of_phi(c, a, T):
    _, p = lse_grad(c, T)
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g


def _build_conv_matrix(a):
    n = a.size
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_direction(a, M):
    """max sum(g) s.t. a*g <= M, g>=0 ; return normalised g or None."""
    n = a.size
    cvec = -np.ones(n)
    A_ub = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 1e-12:
        return None
    return g / s


# ---------------- main optimizer (LSE-Adam + LP kick + backtracking) -------

def optimize(a0, n, time_limit, seed=0,
             do_lp=True, lp_every=10,
             T0=10.0, T1=0.4, lr=0.7,
             b1=0.90, b2=0.999, eps=1e-8,
             stag=140, perturb=0.0015, verbose=False):
    """Combined LSE-Adam with periodic LP equalisation and basin escape."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    trend_min = max(1.0, time_limit)

    a = project(a0.astype(np.float64).copy())
    c = _fft_conv(a)
    best_O, c = true_obj(a)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / trend_min)
        if frac < 0.55:
            T = T0
        else:
            T = T1 + (T0 - T1) * (1.0 - (frac - 0.55) / 0.45)

        # ---- LSE-Adam step ----
        c = _fft_conv(a)
        g = grad_of_phi(c, a, T)
        it += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mhat = m / (1.0 - b1 ** it)
        vhat = v / (1.0 - b2 ** it)
        u = mhat / (np.sqrt(vhat) + eps)
        unorm = np.linalg.norm(u)
        if unorm > 1e-15:
            u = u / unorm
        else:
            u = np.zeros(n)

        # backtracking on true objective
        cur_O = true_obj(a)[0]
        chosen = None
        for alpha in (lr, lr * 0.5, lr * 0.25, lr * 0.125):
            cand = project(a - alpha * u)
            if np.abs(cand - a).sum() < 1e-15:
                continue
            o = true_obj(cand)[0]
            if o < cur_O - 1e-14:
                chosen = (cand, o)
                break
        if chosen is None:
            a = project(a - lr * 0.125 * u)
            o = true_obj(a)[0]
        else:
            a, o = chosen

        # ---- periodic LP equalisation kick ----
        if do_lp and (step % lp_every == 0):
            M = _fft_conv(a).max()
            gg = lp_direction(a, M * 1.0)
            if gg is not None:
                lg = None
                base = true_obj(a)[0]
                for tstep in (0.4, 0.2, 0.1):
                    cand = project((1.0 - tstep) * a + tstep * gg)
                    oo = true_obj(cand)[0]
                    if oo < base - 1e-14:
                        lg = (cand, oo)
                        break
                if lg is not None:
                    a, o = lg

        step += 1

        if o < best_O - 1e-14:
            best_O = o
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and step % 200 == 0:
            print("    opt iter %d n=%d obj=%.7f T=%.3f t=%.1fs"
                  % (step, n, best_O, T, elapsed))

        if no_improve >= stag:
            noise = rng.standard_normal(n) * perturb
            a = project(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_O

def _resample(seq, new_n):
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    xo = np.linspace(0.0, 1.0, old_n)
    xn = np.linspace(0.0, 1.0, new_n)
    out = np.interp(xn, xo, seq)
    return project(out)


def make_inits(n, parent, rng):
    x = np.arange(n, dtype=np.float64)
    out = {}
    if parent is not None:
        out['parent'] = _resample(parent, n)
    out['uniform'] = np.ones(n) / n
    rc = 0.5 * (1.0 + np.cos(np.pi * x / (n - 1)))
    out['rcos'] = project(rc)
    two = 0.6 + 0.4 * np.cos(4.0 * np.pi * x / n)
    out['two'] = project(two)
    four = 0.7 + 0.3 * np.cos(8.0 * np.pi * x / n)
    out['four'] = project(four)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    out['gauss'] = project(g)
    out['rand'] = project(rng.dirichlet(np.ones(n) * 0.5))
    out['symrand'] = project(np.abs(rng.standard_normal(n)))
    return out

            consider(aopt, "scout n=%d %s" % (n, nm))
    print("Phase1 done best=%.6f (%.0fs)" % (best_score, time.monotonic() - t_start))

    # ---- Phase 2: deep polish on top distinct-length configs ----
    print("=== Phase 2: deep ===")
    pool.sort(key=lambda z: z[0])
    seen = set()
    top = []
    for s, n, a in pool:
        if n not in seen:
            top.append((n, a.copy()))
            seen.add(n)
        if len(top) >= 3:
            break
    p2_dead = deadline - 300.0
    for i, (n, a0) in enumerate(top):
        if time.monotonic() > p2_dead - 30:
            break
        ta = max(20.0, min(220.0, p2_dead - time.monotonic() - 20))
        aopt, _ = optimize(a0, n, time_limit=ta, seed=seed + 100 + n,
                           do_lp=enable_lp, lp_every=8,
                           T0=8.0, T1=0.4, lr=0.6, stag=180, perturb=0.0015,
                           verbose=True)
        consider(aopt, "deep n=%d" % n)
    print("Phase2 done best=%.6f (%.0fs)" % (best_score, time.monotonic() - t_start))

    # ---- Phase 3: basin-hopping around the global best (escape basin) ----
    print("=== Phase 3: basin hopping ===")
    if best_seq is None:
        best_seq = project(np.ones(768))
        best_score = float(true_obj(best_seq)[0])
    p3_dead = deadline - 15.0
    base = best_seq
    base_n = base.size
    radius = 0.0015
    hop = 0
    while time.monotonic() < p3_dead - 20:
        # smooth + random perturbation, growing radius
        x = np.arange(base_n)
        sm = np.sin(2.0 * np.pi * rng.uniform(0.5, 3.0) * x / base_n) * radius * 0.5
        sm += np.sin(2.0 * np.pi * rng.uniform(0.5, 3.0) * x / base_n) * radius * 0.5
        noise = rng.standard_normal(base_n) * radius
        pert = project(base + noise + sm)
        ta = max(12.0, min(80.0, p3_dead - time.monotonic() - 12))
        aopt, _ = optimize(pert, base_n, time_limit=ta, seed=seed + 300 + hop,
                           do_lp=enable_lp, lp_every=10,
                           T0=6.0, T1=0.35, lr=0.55, stag=160, perturb=0.001)
        before = best_score
        consider(aopt, "hop%d n=%d" % (hop, base_n))
        hop += 1
        radius *= 1.06
        if best_seq is not base:  # improved: refresh base
            if float(true_obj(aopt)[0]) < before:
                base = aopt.copy()
                base_n = base.size
        if hop % 8 == 0:
            radius = 0.0015  # reset to explore many small basins
    print("Phase3 done best=%.6f (%.0fs)" % (best_score, time.monotonic() - t_start))

    # ---- Phase 4: final tight micro-polish on the very best ----
    print("=== Phase 4: final micro-polish ===")
    ta = max(20.0, deadline - time.monotonic() - 8.0)
    if ta > 15:
        aopt, _ = optimize(best_seq, best_seq.size, time_limit=ta,
                           seed=seed + 999, do_lp=enable_lp, lp_every=6,
                           T0=4.0, T1=0.25, lr=0.4, stag=220, perturb=0.0008,
                           verbose=True)
        consider(aopt, "final")

    # ---- final verification with the real evaluator ----
    final = [float(x) for x in best_seq]
    try:
        fs = float(evaluate_sequence(final))
        if not np.isfinite(fs):
            fs = float(true_obj(best_seq)[0])
    except Exception:
        fs = float(true_obj(best_seq)[0])
    print("FINAL n=%d evaluator_score=%.6f internal=%.6f time=%.0fs"
          % (len(final), fs, float(true_obj(best_seq)[0]), time.monotonic() - t_start))
    return final
