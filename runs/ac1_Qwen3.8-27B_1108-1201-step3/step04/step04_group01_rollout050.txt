def project_simplex(v):
    n = len(v)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Gradient of LSE smoothed objective via FFT cross-correlation."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_subgrad_max(f):
    """Compute subgradient of max(conv(f,f)). Returns gradient at argmax."""
    n = len(f)
    conv = fast_convolve(f)
    k_star = int(np.argmax(conv))
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.zeros(2 * n - 1)
    W[k_star] = 1.0
    Wf = np.fft.rfft(W, fft_size)
    Ff = np.fft.rfft(f_flipped, fft_size)
    conv_result = np.fft.irfft(Wf * Ff, fft_size)
    return 2.0 * conv_result[n - 1:n - 1 + n], conv


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP oracle."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, A_cache):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle for global structure optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        A_cache = build_lp_matrix(f)
        g = solve_lp(f, M * m_exp, A_cache)

                m_exp = 1.0

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search: coarse then fine

                if st < 1e-15:

                ft = ft / st

                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)

def lse_optimize(f0, time_limit, beta_start=15.0, beta_end=20000.0, verbose=False):
    """
    LSE smoothed gradient descent with Nesterov momentum, adaptive beta,
    and hybrid subgradient at high beta for final convergence.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    # Nesterov momentum
    lr = 2.0 / n
    momentum = 0.90
    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead point
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    use_subgrad = 0  # fraction of steps using exact subgradient

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at lookahead point y
        conv_y = fast_convolve(y)

        # At very high beta, blend with exact subgradient
        if beta > 500 and it % 2 == 0:
            sg, _ = compute_subgrad_max(y)
            # LSE grad
            bc = beta * conv_y
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            w = exp_bc / exp_bc.sum()
            lse_grad = compute_lse_grad(w, y)
            blend = min(0.5, (beta - 500) / 5000.0)
            grad = (1 - blend) * lse_grad + blend * sg
        else:
            bc = beta * conv_y
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            w = exp_bc / exp_bc.sum()
            grad = compute_lse_grad(w, y)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Nesterov update
        v = momentum * v - lr * grad
        y_new = project_simplex(f + v)  # lookahead from current f

        # Actually for Nesterov: y = f + momentum * (f - f_prev)
        # Simpler: use classical momentum with lookahead
        f_new = project_simplex(f + v)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            y = f_new + v  # Nesterov lookahead
            no_improve = 0
            # Tiny lr increase on progress
            lr = min(lr * 1.001, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.94
                momentum = max(0.82, momentum - 0.004)
            if no_improve > 120:
                # Restart with perturbation
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                y = f.copy()
                lr = max(1.5 / n, 0.001)
                beta = min(beta * 1.4, beta_end)
                momentum = 0.88
                no_improve = 0

        # Adaptive beta increase
        if no_improve < 20:
            beta *= 1.006  # increase faster when making progress
        else:
            beta *= 1.002  # slower when stuck

        beta = min(beta, beta_end)

        if verbose and it % 1000 == 0:
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coord_descent(f0, time_limit, verbose=False):
    """
    Cyclic coordinate descent with multiple trial steps per coordinate.
    Efficient because changing one element only affects a subset of conv values.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Precompute trial deltas
    deltas_coarse = np.array([0.005, 0.01, 0.02, 0.04, -0.005, -0.01, -0.02, -0.04])
    deltas_fine = np.array([0.001, 0.002, 0.005, -0.001, -0.002, -0.005])
    deltas_very_fine = np.array([0.0003, 0.0006, 0.001, -0.0003, -0.0006, -0.001])

    level = 0  # 0=coarse, 1=fine, 2=very_fine
    all_deltas = [deltas_coarse, deltas_fine, deltas_very_fine]

    while time.monotonic() - t0 < time_limit:
        improved = False
        deltas = all_deltas[level]

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            f_i = f[i]
            best_val = conv.max()
            best_d = 0.0

            for d in deltas:
                new_val = f_i + d
                if new_val < -1e-10:
                    continue
                new_val = max(0.0, new_val)
                if abs(new_val - f_i) < 1e-12:
                    continue
                f[i] = new_val
                s = f.sum()
                if s < 1e-15:
                    f[i] = f_i
                    continue
                f = f / s
                conv = fast_convolve(f)
                val = conv.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = new_val
                    # Save best f
                    f_best = f.copy()
                else:
                    f[i] = f_i
                    f = f / s
                    conv = fast_convolve(f)

            if best_d != 0.0:
                f = f_best
                f = f / f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-15:
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True

        if not improved:
            level = min(level + 1, 2)
            if level == 2:
                # Try random perturbations
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 10000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                level = 0

    return best_f, best_obj
