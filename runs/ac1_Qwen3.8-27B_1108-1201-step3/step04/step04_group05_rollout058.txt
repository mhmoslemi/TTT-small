"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient with Nesterov momentum and adaptive step size
2. Trust-region LP refinement on active peaks for precise local steps
3. Coordinate descent polishing for final refinement
4. Multiple restarts with diverse initializations and length exploration
Key insight: objective is homogeneous of degree 0, so we fix sum(a)=1 and minimize 2n*max(a*a).

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_subgrad(a, conv, temp):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    peaks = conv
    scaled = peaks / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(peaks)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for k in range(len(peaks)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[k] * 2.0 * a[k - idx]
    return sg


def subgrad_optimize(a0, time_limit, rng, lr=0.002, mom=0.92, temp_init=0.001,

    best_a = a.copy()

    it = 0

        # Nesterov: evaluate gradient at lookahead point
        a_look = a + mom * v
        a_look = np.maximum(a_look, 0)

            a_look = a_look / a_look_sum

def build_conv_matrix_rows(f, peaks):
    """Build rows of the convolution matrix for specific peak indices."""
    n = len(f)
    rows = np.zeros((len(peaks), n))
    for i, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        rows[i, idx] = f[k - idx]
    return rows


def trust_region_lp(a, conv, time_per_lp=0.5, max_active=20, radius=0.02):
    """Solve trust-region LP: minimize linearized peak over active set."""
    n = len(a)
    M = conv.max()
    # Find active set: peaks within 5% of max
    threshold = M - 0.05 * M
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        return None
    if len(active) > max_active:
        # Keep the top max_active
        active = np.argsort(conv)[::-1][:max_active]

    t0 = time.monotonic()

    # Variables: [t, d_0, ..., d_{n-1}]
    # minimize t
    # s.t. 2 * (rows @ d)[i] <= t  for each active lag i
    #      sum(d) = 0
    #      -a <= d <= a*2  (so a+d >= 0)
    #      ||d||_1 <= radius * sum(a)  (approximate trust region)

    c = np.zeros(n + 1)
    c[0] = 1.0

    rows = build_conv_matrix_rows(a, active)
    n_active = len(active)

    A_ub = np.zeros((2 * n_active, n + 1))
    b_ub = np.zeros(2 * n_active)
    # 2*rows@d <= t  =>  -t + 2*rows@d <= 0
    A_ub[:n_active, 0] = -1.0
    A_ub[:n_active, 1:] = 2.0 * rows
    b_ub[:n_active] = 0.0
    # -2*rows@d <= t  =>  -t - 2*rows@d <= 0  (symmetrize for stability)
    A_ub[n_active:, 0] = -1.0
    A_ub[n_active:, 1:] = -2.0 * rows
    b_ub[n_active:] = 0.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-a[i], 2.0 * a[i]) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_per_lp, 'presolve': True})

    if time.monotonic() - t0 > time_per_lp:
        return None

    if res.success and res.x is not None:
        d = res.x[1:]
        if np.max(np.abs(d)) > 1e-10:
            return d
    return None


def lp_refine(a, time_limit, rng, max_active=15, radius=0.03):
    """Iterative LP refinement using trust-region subproblems."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        current_obj = conv.max()

        d = trust_region_lp(a, conv, time_per_lp=0.3, max_active=max_active, radius=radius)
        if d is None:
            no_improve += 3
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Line search along a + alpha*d
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            val = fast_convolve(a_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = a + best_alpha * d
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 20:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a, time_limit, rng, max_passes=5):
    """Cycle through coordinates, optimizing each one via 1D search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()

    else:

    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            old_val = a[j]
            # Try a few candidates for a[j]
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                          old_val + 0.001, old_val - 0.001, 0.0, old_val * 2.0]
            best_local_obj = best_obj
            best_local_val = old_val
            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                a[j] = cv
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = old_val
                    continue
                a_norm = a / s_new
                obj = fast_convolve(a_norm).max()
                if obj < best_local_obj - 1e-14:
                    best_local_obj = obj
                    best_local_val = cv
            a[j] = best_local_val
            if best_local_val != old_val:
                improved = True
                a = a / a.sum()
                best_obj = best_local_obj
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def generate_structured_inits(n, parent, rng):

        center = 0.5 * (n - 1)
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        gs = g.sum()
        if gs > 1e-10:
            inits.append(g / gs)

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    inits.append(u / u.sum())

    sn = np.sin(np.pi * (x + 0.5) / n) + 0.1
    inits.append(sn / sn.sum())

    # Power-law decay from center
    d = np.abs(x - 0.5 * (n - 1)) / (0.5 * n)
    p = (1 - d) ** 3 + 0.05
    inits.append(p / p.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    if parent is not None:

        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.02))

        a_opt, _ = subgrad_optimize(best_seq, time_limit=t4_budget * 0.5, rng=rng,
                                     lr=0.0005, mom=0.96, temp_init=0.0002,
                                     temp_end=0.000005, decay=0.9998)
        try_seq(a_opt, "sg_fine1")

        rem = deadline - time.monotonic() - 100
        if rem > 20:
            a_opt, _ = subgrad_optimize(best_seq, time_limit=min(rem * 0.6, 120), rng=rng,
                                         lr=0.0003, mom=0.97, temp_init=0.0001,
                                         temp_end=0.000003, decay=0.9999)
            try_seq(a_opt, "sg_fine2")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Coordinate descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 60
    t5_budget = min(remaining * 0.5, 150)

    if t5_budget > 10:
        print(f"  Running coordinate descent, budget={t5_budget:.0f}s")
        n_cur = len(best_seq)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t5_budget, rng=rng, max_passes=3)
        try_seq(a_opt, "coord_desc")

    # === Phase 6: Final alternation and length exploration ===
    print("\n=== Phase 6: Final Attempts ===")
    remaining = deadline - time.monotonic() - 30

    if remaining > 20 and best_seq is not None:
        n_cur = len(best_seq)
        # Try nearby lengths
        for n_try in [n_cur - 4, n_cur + 4]:
            if n_try < 64 or remaining < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_cur), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                rem_each = min(remaining * 0.3, 40)
                a_opt, _ = subgrad_optimize(px, time_limit=rem_each, rng=rng,
                                             lr=0.001, mom=0.94, temp_init=0.0005,
                                             temp_end=0.00005, decay=0.9995)
                try_seq(a_opt, f"len{n_try}")
                remaining = deadline - time.monotonic() - 20

        # One more LP+SG cycle on best
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            a_opt, _ = lp_refine(best_seq, time_limit=min(rem * 0.4, 60), rng=rng,
                                  max_active=10, radius=0.01)
            try_seq(a_opt, "final_lp")
            rem2 = deadline - time.monotonic() - 15
            if rem2 > 15:
                a_opt, _ = subgrad_optimize(best_seq, time_limit=rem2, rng=rng,
                                             lr=0.0003, mom=0.96, temp_init=0.0001,
                                             temp_end=0.00001, decay=0.9998)
                try_seq(a_opt, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
