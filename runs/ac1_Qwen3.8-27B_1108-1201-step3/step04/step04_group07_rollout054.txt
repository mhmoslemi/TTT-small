def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_banded(a):
    """Build the convolution matrix C where (C@g)[k] = conv(a,g)[k].
    Uses the fact that C is banded for efficiency."""
    n = len(a)
    # C has shape (2n-1, n), C[k,j] = a[k-j] if 0<=k-j<n else 0
    # This is a Toeplitz-like matrix
    # We'll build it as a dense matrix but exploit sparsity
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def obj_normalized(a):
    """Objective for normalized a (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def lp_direction(a, M):

def resample(seq, new_n):
    """Resample sequence to new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-12:
        new_seq = new_seq / s
    return new_seq

def fast_convolve(a):
    n = len(a)

    A = np.fft.rfft(a, fft_size)

        return None, 1.0, None
    dual = None

def resample(seq, nn):
    on = len(seq)
    if on == nn:
        return seq.copy()
    xo = np.linspace(0, 1, on)

    ns = np.interp(xn, xo, seq)
    ns = np.maximum(ns, 0.0)
    s = ns.sum()
    if s > 1e-12:
        ns = ns / s
    return ns

"""
LP-driven auto-convolution minimizer with multi-length progressive refinement.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using iterative LP direction search,
dual-guided subgradient polishing, and adaptive multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsz = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsz)
    return np.fft.irfft(A * A, fsz)[:size]


def conv_matrix(a):
    """Build C such that (C@g)[k] = full_conv(a,g)[k]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def obj(a):
    """Return (2n*max(conv(a,a)), conv) for normalized a."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max(), c


def solve_lp(a, M):
    """LP: max sum(g) s.t. conv(a,g)<=M, g>=0. Returns (g_norm, g_sum)."""
    n = len(a)
    C = conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bnd = [(0.0, None)] * n
    r = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bnd, method='highs')
    if not r.success:
        return None, 1.0
    g = np.maximum(r.x, 0.0)
    gs = g.sum()
    if gs < 1e-12:
        return None, 1.0
    return g / gs, gs
