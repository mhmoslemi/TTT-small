Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel multi-phase approach:
1. Pressure-balance multiplicative updates (all peaks above threshold)
2. Coordinate descent with exact 1D minimization
3. FW-LP for global structure

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def pressure_balance_optimize(f0, time_limit, eta_init=0.02, threshold=0.995, decay=0.9995):
    """
    Pressure-balance multiplicative update.
    For each position i, compute its "pressure" (contribution to peaks).
    Reduce high-pressure positions, increase low-pressure ones.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    eta = eta_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Identify peaks above threshold
        threshold_val = M * threshold
        peak_mask = conv >= threshold_val
        peak_indices = np.where(peak_mask)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute pressure on each position
        pressure = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # conv[k] = sum_j a[j]*a[k-j]
            # Position i contributes a[i]*a[k-i] to conv[k]
            j = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(j, je + 1)
            # pressure[i] += a[k-i]
            pressure[idx] += f[k - idx]

        # Normalize pressure
        p_mean = pressure.mean()
        p_std = pressure.std() + 1e-15

        # Multiplicative update: reduce high pressure, increase low pressure
        # Use a soft version to avoid zeroing out
        log_update = -eta * (pressure - p_mean) / p_std
        # Clip to prevent extreme updates
        log_update = np.clip(log_update, -2.0, 2.0)
        
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv_new = fast_convolve(f)
        obj = conv_new.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If stuck, try a larger step
            if no_improve % 50 == 0:
                eta = min(eta * 1.2, 0.1)

        eta *= decay

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            eta = eta_init * 0.5
            no_improve = 0

    return best_f, best_obj

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel multi-phase approach:
1. Fourier-domain (IFT/Gerchberg-Saxton) optimization for spectral flattening
2. Pressure-balance multiplicative updates with backtracking and all-lag weighting
3. FW-LP for structural optimization

Key innovations: IFT-based spectral flattening, smooth all-lag pressure with
temperature annealing, and adaptive backtracking line search.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    col = np.arange(n)
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fourier_optimize(f0, time_limit, eta_init=0.25):
    """
    Fourier-domain (Gerchberg-Saxton / IFT) optimization.
    Alternates between:
    - Frequency domain: flatten the power spectrum |A|^2
    - Time domain: project onto {a >= 0, sum(a) = 1}
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    eta = eta_init
    it = 0

    # Target flat power
    target = 1.0 / N  # since sum(a) = 1, total power = 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        a_padded = np.zeros(N)
        a_padded[:n] = f

        A = np.fft.fft(a_padded)
        power = np.abs(A) ** 2

        # Flatten: move toward target
        power_new = (1.0 - eta) * power + eta * target
        # Keep DC bin fixed (it encodes the sum)
        power_new[0] = power[0]

        phase = np.angle(A)
        mag = np.sqrt(np.maximum(power_new, 0.0))
        A_new = mag * np.exp(1j * phase)

        f_new_padded = np.fft.ifft(A_new).real
        f_new = f_new_padded[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            no_improve += 10
            eta *= 0.95
            continue
        f_new = f_new / s_new

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        eta = max(0.005, eta * 0.9995)

        if no_improve > 300:
            rng = np.random.default_rng(it + 12345)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            eta = eta_init * 0.7
            no_improve = 0

    return best_f, best_obj


def fourier_optimize_v2(f0, time_limit):
    """

    Uses a "water-filling" approach: only reduce peaks above target,
    and redistribute to valleys below target.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

        a_padded[:n] = f

        A = np.fft.fft(a_padded)
        power = np.abs(A) ** 2

        # Water-filling: reduce peaks, boost valleys
        target = 1.0 / N
        peak_mask = power > target * 1.01
        valley_mask = power < target * 0.99

        if peak_mask.any() and valley_mask.any():
            excess = (power[peak_mask] - target).sum() * 0.3
            deficit = (target - power[valley_mask]).sum()
            transfer = min(excess, deficit) * 0.5

            if transfer > 1e-15:
                power[peak_mask] -= transfer / peak_mask.sum()
                power[valley_mask] += transfer / valley_mask.sum()

        # Also apply a gentle flattening to everything
        power = 0.95 * power + 0.05 * target
        power[0] = np.abs(A[0]) ** 2  # keep DC

        phase = np.angle(A)
        mag = np.sqrt(np.maximum(power, 0.0))
        A_new = mag * np.exp(1j * phase)

        f_new_padded = np.fft.ifft(A_new).real
        f_new = f_new_padded[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            no_improve += 10
            continue
        f_new = f_new / s_new

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0

        f = f_new

def pressure_optimize(f0, time_limit, eta_init=0.015, temp_init=0.0002):
    """
    Pressure-balance multiplicative update with all-lag exponential weighting
    and backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    eta = eta_init
    temp = temp_init
    it = 0
    momentum = np.zeros(n)
    mom_coeff = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute smooth weights for all lags
        # w[k] = exp((conv[k] - M) / temp), normalized
        diffs = (conv - M) / temp
        diffs = np.maximum(diffs, -50.0)  # avoid underflow
        w = np.exp(diffs)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(conv_len)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute pressure: P[i] = sum_k w[k] * a[k-i] (for valid k-i)
        pressure = np.zeros(n)
        for k in range(conv_len):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += w[k] * f[k - idx]

        # Apply momentum
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * pressure
        p_mean = momentum.mean()
        p_std = momentum.std() + 1e-15

        # Log-update direction
        log_dir = -momentum / p_std  # negative pressure direction

        # Backtracking line search
        accepted = False
        for shrink in range(8):
            step = eta * (0.7 ** shrink)
            lr = np.clip(step * log_dir, -1.5, 1.5)
            f_try = f * np.exp(lr)
            f_try = np.maximum(f_try, 1e-25)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try
            obj_try = fast_convolve(f_try).max()

            if obj_try < M - 1e-16:
                f = f_try
                conv = fast_convolve(f)
                M = conv.max()
                accepted = True
                break

        if accepted:
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            eta = max(eta * 0.9, eta_init * 0.1)

        if accepted:
            eta = min(eta * 1.02, eta_init * 2.0)
        else:
            eta = max(eta * 0.95, eta_init * 0.2)

        # Slowly increase temperature (broader pressure)
        temp = min(temp * 1.001, temp_init * 10.0)

        if no_improve > 150:
            rng = np.random.default_rng(it * 7 + 42)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            momentum = np.zeros(n)
            eta = eta_init * 0.6
            temp = temp_init
            no_improve = 0

    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=20, temp_init=0.0004):
    """
    Multiplicative weight (mirror descent) with top-k peaks and softmax weighting.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = (pv - M) / cur_temp
        scaled = np.maximum(scaled, -50.0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_update[idx] -= cur_lr * w[i] * f[k - idx]

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.001

        if no_improve > 120:

            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.008)
            cur_temp = temp_init * 1.5
            no_improve = 0

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.002, mom=0.93, top_k=15):
    """Subgradient optimization with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        tk = max(3, n // 25)
        if top_k > 0:
            tk = top_k
        if tk >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -tk)[-tk:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        tw = weights.sum()
        if tw > 0:
            weights /= tw

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)

        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997

        if no_improve > 180:
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0004)
            no_improve = 0

    return best_f, best_obj


def fine_polish(f0, time_limit):
    """Very fine local search with tiny steps in multiple random directions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    scale = 0.001
    while time.monotonic() - t0 < time_limit:
        # Try multiple small perturbations
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            # Random direction on simplex
            d = rng.standard_normal(n)
            d = d - d.mean()  # keep sum roughly constant
            d = d / (np.linalg.norm(d) + 1e-15)

            f_try = best_f + scale * d
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try

            obj = fast_convolve(f_try).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f_try.copy()

        # Occasionally try a larger step
        if rng.random() < 0.05:
            d = rng.standard_normal(n)
            d = d - d.mean()
            d = d / (np.linalg.norm(d) + 1e-15)
            f_try = best_f + scale * 5.0 * d
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try > 1e-15:
                f_try = f_try / s_try
                obj = fast_convolve(f_try).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_f = f_try.copy()

        # Slowly reduce scale
        scale *= 0.98
        if scale < 1e-5:
            scale = 0.001
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)

    return best_f, best_obj


def genetic_search(f0, time_limit, pop_size=30):
    """
    Genetic algorithm for global exploration.
    Uses blend crossover and Gaussian mutation.
    """
    n = len(f0)
    rng = np.random.default_rng(42)

    # Initialize population
    population = [f0.copy()]
    for _ in range(pop_size - 1):
        # Mix of perturbed parent and random
        if rng.random() < 0.5:
            individual = f0 * np.exp(rng.standard_normal(n) * 0.05)
        else:
            individual = rng.dirichlet(np.ones(n) * rng.uniform(0.3, 2.0))
        individual = np.maximum(individual, 0)
        individual = individual / individual.sum()
        population.append(individual)

    # Score
    scores = [fast_convolve(p).max() for p in population]
    best_idx = np.argmin(scores)
    best_f = population[best_idx].copy()
    best_obj = scores[best_idx]

    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        # Selection: keep top half
        order = np.argsort(scores)
        parents = [population[i] for i in order[:pop_size // 2]]
        best_obj_gen = scores[order[0]]
        if best_obj_gen < best_obj - 1e-15:
            best_obj = best_obj_gen
            best_f = population[order[0]].copy()

        # Generate new population
        new_pop = [best_f.copy()]  # elitism
        for _ in range(pop_size - 1):
            # Crossover: blend
            p1 = parents[rng.integers(len(parents))]
            p2 = parents[rng.integers(len(parents))]
            alpha = rng.uniform(0.2, 0.8)
            child = alpha * p1 + (1.0 - alpha) * p2
            child = np.maximum(child, 0)
            cs = child.sum()
            if cs > 1e-15:
                child = child / cs

            # Mutation
            mut_scale = 0.01 + 0.02 * (gen % 10) / 10.0
            child = child * np.exp(rng.standard_normal(n) * mut_scale)
            child = np.maximum(child, 0)
            cs = child.sum()
            if cs > 1e-15:
                child = child / cs

            new_pop.append(child)

        population = new_pop
        scores = [fast_convolve(p).max() for p in population]

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining:
    - Fourier-domain IFT optimization
    - Pressure-balance multiplicative updates
    - FW-LP structural optimization
    - Subgradient and fine local search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = np.ones(384) / 384

    n_use = len(parent) if parent is not None else 384

    # === Phase 1: Fourier-domain optimization (~300s) ===
    print("\n=== Phase 1: Fourier IFT Optimization ===")
    rem = deadline - time.monotonic() - 1200
    if rem > 30:
        t1 = min(rem, 200)
        print(f"  Fourier v1, t={t1:.0f}s")
        a_opt, _ = fourier_optimize(parent, time_limit=t1, eta_init=0.2)
        try_seq(a_opt, "fourier_v1")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 1000
        if rem > 20:
            t1b = min(rem, 120)
            print(f"  Fourier v2, t={t1b:.0f}s")
            a_opt, _ = fourier_optimize_v2(best_seq, time_limit=t1b)
            try_seq(a_opt, "fourier_v2")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Pressure-balance optimization (~400s) ===
    print("\n=== Phase 2: Pressure-Balance Optimization ===")
    if best_seq is not None:
        configs = [
            {'eta_init': 0.015, 'temp_init': 0.0002},
            {'eta_init': 0.01, 'temp_init': 0.0005},
            {'eta_init': 0.02, 'temp_init': 0.0001},
        ]
        rem = deadline - time.monotonic() - 600
        t_per = max(20, rem // len(configs))
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 500
            if rem < 15:
                break
            t_alloc = min(t_per, rem)
            print(f"  Pressure config {ci} (eta={cfg['eta_init']}), t={t_alloc:.0f}s")
            a_opt, _ = pressure_optimize(best_seq, time_limit=t_alloc,
                                         eta_init=cfg['eta_init'],
                                         temp_init=cfg['temp_init'])
            try_seq(a_opt, f"pressure_{ci}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: FW-LP (~300s) ===
    print("\n=== Phase 3: FW-LP ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem, 250)
            print(f"  FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp")

            # Perturbed start
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                noise = rng.standard_normal(len(best_seq)) * 0.004
                f_pert = project_simplex(best_seq + noise)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(rem, 100))
                try_seq(a_opt2, "fw_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Subgradient + MW (~250s) ===
    print("\n=== Phase 4: Subgradient + MW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 15:
            t_alloc = min(rem, 120)
            print(f"  SG, t={t_alloc:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_alloc, lr=0.002, mom=0.93, top_k=15)
            try_seq(a_opt, "sg")

        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem, 100)
            print(f"  MW, t={t_alloc:.0f}s")
            a_opt, _ = mult_weight_optimize(best_seq, time_limit=t_alloc,
                                            lr_init=0.03, top_k=15, temp_init=0.0003)
            try_seq(a_opt, "mw")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Genetic search for exploration (~150s) ===
    print("\n=== Phase 5: Genetic Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  GA, t={t_alloc:.0f}s")
            a_opt, _ = genetic_search(best_seq, time_limit=t_alloc, pop_size=20)
            try_seq(a_opt, "ga")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Fine polish (~100s) ===
    print("\n=== Phase 6: Fine Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_alloc = min(rem, 60)
            print(f"  Fine polish, t={t_alloc:.0f}s")
            a_opt, _ = fine_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fine")

        # One more Fourier pass
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final Fourier, t={t_alloc:.0f}s")
            a_opt, _ = fourier_optimize(best_seq, time_limit=t_alloc, eta_init=0.1)
            try_seq(a_opt, "final_fourier")

        # Final pressure
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 20)
            a_opt, _ = pressure_optimize(best_seq, time_limit=t_alloc,
                                         eta_init=0.005, temp_init=0.0001)
            try_seq(a_opt, "final_pressure")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
