"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad_fully_differentiable(weights, f):
    """
    Compute gradient of LSE objective w.r.t. f.
    grad[i] = 2 * sum_{m} weights[i+m] * f[m]  (cross-correlation style)
    Uses FFT for efficiency.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def topk_subgradient_polish(f0, time_limit, top_k=16, lr_init=0.005, momentum=0.93):
    """
    Subgradient method using top-k peaks of the convolution.
    Uses backtracking line search for step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_mom = momentum
    v = np.zeros(n)
    temp = 0.01  # temperature for weight softness

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        f_max = conv.max()

        # Select top-k peaks
        k_select = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_select)[-k_select:]
        pv = conv[peaks]

        # Softmax weights over top-k peaks
        scaled = (pv - f_max) / temp  # relative to max
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute subgradient as weighted average
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.93
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
                scale = 0.001
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                cur_lr = lr_init
                no_improve = 0

        temp *= 1.003  # gradually sharpen peak selection

    return best_f, best_obj

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Coarse + fine line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        m_exp = 1.0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 13 + 5)
                scale = 0.002
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def optimize_length(n, time_budget, seed=42):
    """Run the full optimization pipeline for a given length."""
    rng = np.random.default_rng(seed)

    # Initialization: mix of uniform and random
    candidates = []
    candidates.append(np.ones(n) / n)
    candidates.append(project_simplex(rng.standard_normal(n) + n * 0.1))
    # Triangular-like init
    x = np.linspace(-1, 1, n)
    tri = np.maximum(1 - np.abs(x), 0)
    tri = tri / tri.sum()
    candidates.append(tri)

    best_f = None
    best_obj = float('inf')

    for cand in candidates[:2]:
        if best_f is not None and best_obj < 1.51:
            break
        f_opt, obj = adam_lse_optimize(cand, time_limit=min(100, time_budget * 0.3),
                                        beta_start=20.0, beta_end=5000.0)
        try:
            score = float(evaluate_sequence(f_opt.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_obj:
            best_obj = score
            best_f = f_opt

    if best_f is None:
        best_f = np.ones(n) / n
        best_obj = 2.0

    # Continue with best candidate using more time
    remaining = time_budget - 100
    if remaining > 20:
        f_opt, obj = adam_lse_optimize(best_f, time_limit=max(50, remaining * 0.6),
                                        beta_start=50.0, beta_end=15000.0)
        try:
            score = float(evaluate_sequence(f_opt.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_obj:
            best_obj = score
            best_f = f_opt

    # Top-k subgradient polish
    remaining = time_budget - 200
    if remaining > 15:
        f_opt, obj = topk_subgradient_polish(best_f, time_limit=max(10, remaining * 0.4),
                                              top_k=max(8, n // 30))
        try:
            score = float(evaluate_sequence(f_opt.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_obj:
            best_obj = score
            best_f = f_opt

    # Short FW-LP
    remaining = time_budget - 250
    if remaining > 10:
        f_opt, obj = fw_lp_optimize(best_f, time_limit=max(10, remaining * 0.3))
        try:
            score = float(evaluate_sequence(f_opt.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_obj:
            best_obj = score
            best_f = f_opt

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase, multi-length optimization:
    1. Optimize parent length with Adam-LSE + top-k subgradient + FW-LP
    2. Try alternative lengths with shorter budgets
    3. Return best sequence found
    """
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            print(f"Parent: n={n_parent}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")
        n_parent = 448
        parent = np.ones(n_parent) / n_parent
        try_sequence(parent, best_tracker)

    # === Phase 1: Main optimization on parent length ===
    print(f"\n=== Phase 1: Main optimization (n={n_parent}) ===")
    t_alloc = min(700, total_deadline - time.monotonic() - 400)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        f_opt, obj = optimize_length(n_parent, time_budget=max(30, t_alloc), seed=seed)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Additional LSE run with different hyperparams ===
    print(f"\n=== Phase 2: Different hyperparams ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)
    t_alloc = min(300, total_deadline - time.monotonic() - 200)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        f_opt, obj = adam_lse_optimize(current_best, time_limit=int(t_alloc * 0.7),
                                        beta_start=10.0, beta_end=20000.0)
        try_sequence(f_opt, best_tracker)

        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        t_rem = total_deadline - time.monotonic() - 50
        if t_rem > 20:
            f_opt, obj = topk_subgradient_polish(current_best, time_limit=int(min(t_rem * 0.5, 80)),
                                                  top_k=max(4, n_best // 50))
            try_sequence(f_opt, best_tracker)

    # === Phase 3: Try alternative lengths ===
    print(f"\n=== Phase 3: Alternative lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    alt_lengths = [384, 512, 400, 480]
    alt_lengths = [x for x in alt_lengths if x != n_best][:3]

    for i, n_alt in enumerate(alt_lengths):
        t_remaining = total_deadline - time.monotonic() - 30
        if t_remaining < 50:
            break
        t_alloc = max(40, min(150, t_remaining // (len(alt_lengths) - i)))
        print(f"  Trying n={n_alt}, budget={t_alloc:.0f}s")

        # Interpolate
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        f_interp = f_interp / f_interp.sum()

        f_opt, obj = optimize_length(n_alt, time_budget=int(t_alloc), seed=seed + i * 100)
        try_sequence(f_opt, best_tracker)
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        n_best = len(current_best)

    # === Final: quick top-k polish on overall best ===
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = total_deadline - time.monotonic() - 10
    if t_remaining > 10:
        print(f"\n=== Final polish ({t_remaining:.0f}s) ===")
        f_opt, obj = topk_subgradient_polish(current_best, time_limit=int(t_remaining * 0.8),
                                              top_k=max(4, len(current_best) // 40),
                                              lr_init=0.003)
        try_sequence(f_opt, best_tracker)

    # === Return ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(448) / 448
    print(f"\nFALLBACK: score={evaluate_sequence(fallback.tolist()):.8f}")
    return [float(x) for x in fallback]
