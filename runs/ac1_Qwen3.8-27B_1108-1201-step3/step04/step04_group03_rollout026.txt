def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def push_pull_optimize(f0, time_limit, verbose=False):
    """
    Targeted peak shaving: for each top peak, move mass from contributing
    positions to non-contributing positions using optimal transport.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]
        
        # For each peak, compute push-pull direction
        direction = np.zeros(n)
        
        for peak_pos in peaks:
            k = int(peak_pos)
            # Contributing positions: j such that both j and k-j are in [0, n-1]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            contributing = np.arange(j_min, j_max + 1)
            
            # Non-contributing positions
            all_pos = np.arange(n)
            non_contributing = np.setdiff1d(all_pos, contributing)
            
            if len(non_contributing) == 0:
                continue
            
            # Weight: how much each contributing position "causes" the peak
            # grad_k[j] = 2*f[k-j]
            weights = 2.0 * f[k - contributing]
            total_w = weights.sum()
            if total_w < 1e-15:
                continue
            
            # Normalize weights
            weights = weights / total_w
            
            # Source: contributing positions (weighted)
            # Destination: non-contributing positions (uniform)
            # Move small amount of mass
            amount = lr * f[contributing].max() * len(contributing) / n
            
            for i, j in enumerate(contributing):
                direction[j] -= weights[i] * amount
            
            # Distribute to non-contributing positions
            # Prefer positions that are "far" from the peak (less likely to create new peaks)
            for j in non_contributing:
                # Distance from peak in convolution
                # Position j will affect convolution positions [j, j+n-1]
                # Peak is at k, so j affects k iff j <= k <= j+n-1
                # i.e., k-n+1 <= j <= k
                # We already excluded those, so all non-contributing are safe for this peak
                # But they might affect OTHER peaks. Use uniform for simplicity.
                direction[j] += amount / len(non_contributing)
        
        # Apply update
        f_new = f + direction
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Reduce learning rate and restart
                lr *= 0.5
                if lr < 0.0001:
                    # Random restart
                    rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    lr = 0.01
                    no_improve = 0
        
        # Anneal learning rate
        if no_improve == 0:
            lr *= 0.999
        
    return best_f, best_obj

def multi_peak_fw(f0, time_limit, verbose=False):
    """
    Frank-Wolfe with LP oracle, improved M relaxation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:

    it = 0
    m_multiplier = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1

        if g is None:
            m_multiplier *= 1.2
            if m_multiplier > 10.0:
                m_multiplier = 1.0
                continue
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_multiplier = min(m_multiplier * 1.1, 5.0)
                if m_multiplier > 5.0:
                    break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)

            for _ in range(8):

                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)

                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_multiplier = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()

            scale = 0.001 + 0.003 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_multiplier = 1.0
            no_improve = 0
    
    return best_f, best_obj

def adaptive_sg(f0, time_limit, verbose=False):
    """
    Adaptive subgradient with momentum, varying the number of peaks tracked
    and the weighting scheme over time.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    mom = 0.9
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Adaptive number of peaks
        phase = min(1.0, it / 5000.0)
        top_k = int(3 + 20 * phase)
        top_k = min(top_k, 2 * n - 1)
        
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        
        # Exponential weighting (sharper over time)
        sharpness = 1.0 + 3.0 * phase
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        total_w = 0.0
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** sharpness
        total_w = weights.sum()
        if total_w > 0:
            weights /= total_w
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            idx = np.arange(j_min, j_max + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]
        
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve % 100 == 0:
            lr *= 0.9
        if no_improve > 300:
            rng = np.random.default_rng(it * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = 0.003
            mom = 0.9
            no_improve = 0
    
    return best_f, best_obj

def simulated_anneal(f0, time_limit, verbose=False):
    """
    Simulated annealing with structured moves: push-pull, rotation, and
    local redistribution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    cur_obj = conv.max()
    best_obj = cur_obj
    best_f = f.copy()
    
    t0 = time.monotonic()
    T = 0.01
    T_min = 0.00001
    cooling = 0.9995
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        peak_pos = int(np.argmax(conv))
        
        # Choose move type
        move_type = rng.integers(0, 4)
        
        if move_type == 0:
            # Push-pull: move mass from peak-contributing to non-contributing
            k = peak_pos
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            contributing = np.arange(j_min, j_max + 1)
            non_contributing = np.setdiff1d(np.arange(n), contributing)
            
            if len(contributing) > 0 and len(non_contributing) > 0:
                # Pick a contributing position to decrease
                src = contributing[rng.integers(0, len(contributing))]
                # Pick a non-contributing position to increase
                dst = non_contributing[rng.integers(0, len(non_contributing))]
                amount = min(f[src], T * 0.1)
                f_new = f.copy()
                f_new[src] -= amount
                f_new[dst] += amount
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
                
                obj_new = fast_convolve(f_new).max()
                delta = obj_new - cur_obj
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    f = f_new
                    cur_obj = obj_new
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_f = f.copy()
        
        elif move_type == 1:
            # Gaussian perturbation
            scale = T * 0.05
            f_new = f + rng.standard_normal(n) * scale
            f_new = project_simplex(f_new)
            obj_new = fast_convolve(f_new).max()
            delta = obj_new - cur_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                f = f_new
                cur_obj = obj_new
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_f = f.copy()
        
        elif move_type == 2:
            # Local swap: exchange mass between two nearby positions
            i = rng.integers(0, n - 1)
            j = i + rng.integers(1, min(10, n - i))
            amount = min(f[i], f[j]) * rng.uniform(0.1, 0.5)
            f_new = f.copy()
            f_new[i] += amount
            f_new[j] -= amount
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            
            obj_new = fast_convolve(f_new).max()
            delta = obj_new - cur_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                f = f_new
                cur_obj = obj_new
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_f = f.copy()
        
        else:
            # "Mirror" move: try to make f more or less symmetric
            # The center element of conv is sum_j f[j]*f[n-1-j]
            # If we make f more symmetric, this increases; less symmetric, decreases
            # But we want to DECREASE the overall max, so try asymmetric moves
            axis = n // 2
            i = rng.integers(0, axis)
            j = n - 1 - i
            if i != j:
                amount = min(f[i], f[j]) * rng.uniform(0.05, 0.3)
                if rng.random() < 0.5:
                    f_new = f.copy()
                    f_new[i] += amount
                    f_new[j] -= amount
                else:
                    f_new = f.copy()
                    f_new[i] -= amount
                    f_new[j] += amount
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
                
                obj_new = fast_convolve(f_new).max()
                delta = obj_new - cur_obj
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    f = f_new
                    cur_obj = obj_new
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_f = f.copy()

    return best_f, best_obj

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Dirichlet with various concentrations
    for alpha in [0.3, 0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** rng.uniform(0.5, 3.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian bump
    center = rng.uniform(0.2 * n, 0.8 * n)
    width = rng.uniform(0.05 * n, 0.3 * n)
    g = np.exp(-0.5 * ((x - center) / width) ** 2)
    inits.append(g / g.sum())
    
    # Cosine-based
    freq = rng.uniform(0.5, 5.0)
    c = np.abs(np.cos(np.pi * freq * x / n)) + 0.05
    inits.append(c / c.sum())
    
    # Random smooth
    rough = rng.standard_normal(n)
    # Smooth with a few iterations of averaging
    for _ in range(rng.integers(1, 10)):
        rough = (rough + np.roll(rough, 1)) / 2
    rough = np.maximum(rough, 0)
    if rough.sum() > 1e-10:
        inits.append(rough / rough.sum())
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            for t in [0.3, 0.5, 0.7]:
                mixed = (1 - t) * np.ones(n) / n + t * px / ps
                inits.append(mixed / mixed.sum())
    
    return inits

Let me think about what's really limiting performance:
- The convolution has 2n-1 = 767 elements for n=384
- We want all of them to be ≤ M
- The "active" constraints (those at the max) determine the solution
- At the optimum, typically several constraints are active (the KKT conditions)

A better approach might be to:
1. Identify which constraints are active at the current solution
2. Solve the KKT system directly for the active set
3. Update the active set based on violations

This is essentially the "active set" method for non-smooth optimization. Let me implement this:

**Active Set Method:**
1. Start with current solution a
2. Find the top K peaks (active constraints)
3. For the active set A, solve the system:
   - sum_j a[j] = 1
   - a[j] >= 0
   - (a*a)[k] = M for k in A (the peaks should be equal at optimum)
   - minimize M
   
   This is a system of |A| + 1 equations in n + 1 unknowns (a[0],...,a[n-1], M). Since n >> |A|, we need to also minimize some regularizer or use the KKT conditions.

Actually, the KKT conditions say:
- For a[j] > 0: 2*sum_{k in A} mu_k * a[k-j] = lambda (constant)
- For a[j] = 0: 2*sum_{k in A} mu_k * a[k-j] <= lambda

This is a coupled system that's hard to solve directly. Let me think of another way.

Alternative: **Iterative Proportional Fitting (IPF) / Sinkhorn-like approach**

We want to find a such that (a*a)[k] ≈ M for all "important" k. This is like a "dual" of the problem: given target values for the convolution, find a.

For the special case where we want (a*a)[k] = c for all k (perfectly flat), this is equivalent to finding a "costas" sequence, which doesn't exist for even lengths in the binary case.

In the continuous case, we can use an iterative approach:
1. Start with a
2. Compute c = (a*a)[k] for all k
3. Update: a_new[j] = a[j] / sqrt(sum_k c[k] * indicator(j contributes to k))
   (This is like a multiplicative weight update)

def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0
            no_improve = 0

    return best_f, best_obj
