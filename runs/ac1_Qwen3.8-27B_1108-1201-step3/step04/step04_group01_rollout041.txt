def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent (ternary search per coordinate)
2. Nesterov-accelerated LSE gradient descent
3. Frank-Wolfe with LP oracle
4. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE-smoothed objective."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[k - jj] for jj in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def coord_desc_one_sweep(a, order=None, n_iters=30):
    """One full sweep of coordinate descent.
    Returns updated a and the max convolution value.
    """
    n = len(a)
    if order is None:
        order = np.random.permutation(n)
    
    conv = fast_convolve(a)
    
    for j in order:
        a_old = a[j]
        m_lo = max(0, j)
        m_hi = min(2 * n - 2, j + n - 1)
        
        # Compute unaffected max
        if m_lo > 0 or m_hi < 2 * n - 2:
            unaffected = np.ones(2 * n - 1, dtype=bool)
            unaffected[m_lo:m_hi + 1] = False
            if unaffected.any():
                unaffected_max = np.max(conv[unaffected])
            else:
                unaffected_max = 0.0
        else:
            unaffected_max = 0.0
        
        # Ternary search for optimal x
        lo = 0.0
        hi = max(2.0 * a_old + 1.0 / n, 2.0 / n)
        
        best_x = a_old
        # Initial evaluation
        if m_lo <= m_hi:
            offsets = np.arange(m_lo, m_hi + 1) - j
            vals = conv[m_lo:m_hi + 1].copy()
            if m_lo <= 2 * j <= m_hi:
                vals[2 * j - m_lo] = conv[2 * j] - a_old * a_old + a_old * a_old  # = conv[2*j]
            best_val = max(np.max(vals), unaffected_max)
        else:
            best_val = unaffected_max
        
        for _ in range(n_iters):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            
            # Evaluate at m1
            delta1 = m1 - a_old
            if m_lo <= m_hi:
                offsets = np.arange(m_lo, m_hi + 1) - j
                vals1 = conv[m_lo:m_hi + 1] + 2.0 * delta1 * a[offsets]
                if m_lo <= 2 * j <= m_hi:
                    vals1 = vals1.copy()
                    vals1[2 * j - m_lo] = conv[2 * j] - a_old * a_old + m1 * m1
                v1 = max(np.max(vals1), unaffected_max)
            else:
                v1 = unaffected_max
            
            # Evaluate at m2
            delta2 = m2 - a_old
            if m_lo <= m_hi:
                vals2 = conv[m_lo:m_hi + 1] + 2.0 * delta2 * a[offsets]
                if m_lo <= 2 * j <= m_hi:
                    vals2 = vals2.copy()
                    vals2[2 * j - m_lo] = conv[2 * j] - a_old * a_old + m2 * m2
                v2 = max(np.max(vals2), unaffected_max)
            else:
                v2 = unaffected_max
            
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        
        # Final value at midpoint
        x_star = (lo + hi) / 2.0
        delta_s = x_star - a_old
        if m_lo <= m_hi:
            offsets = np.arange(m_lo, m_hi + 1) - j
            vals_s = conv[m_lo:m_hi + 1] + 2.0 * delta_s * a[offsets]
            if m_lo <= 2 * j <= m_hi:
                vals_s = vals_s.copy()
                vals_s[2 * j - m_lo] = conv[2 * j] - a_old * a_old + x_star * x_star
            v_s = max(np.max(vals_s), unaffected_max)
        else:
            v_s = unaffected_max
        
        if v_s < best_val:
            best_val = v_s
            best_x = x_star
        
        a[j] = best_x
    
    conv = fast_convolve(a)
    return a, conv


def coord_desc_optimize(a0, time_limit, sweeps_per_check=3):
    """Coordinate descent optimization with periodic checking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    sweep_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Random order each sweep
        order = np.random.permutation(n)
        
        for _ in range(sweeps_per_check):
            if time.monotonic() - t0 > time_limit:
                break
            a, conv = coord_desc_one_sweep(a, order, n_iters=25)
            sweep_count += 1
        
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        if no_improve > 20:
            # Add small perturbation
            rng = np.random.default_rng(sweep_count)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(a)
            no_improve = 0
    
    return best_a, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rhos = np.arange(1, n + 1)
    cond = u * rhos > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - jj] for jj in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def ternary_search_coord(a, j, conv, unaffected_max, n_iter=35):
    """Find optimal a[j] via ternary search. Returns (optimal_x, optimal_val)."""
    n = len(a)
    a_old = a[j]
    m_lo = max(0, j)
    m_hi = min(2 * n - 2, j + n - 1)
    
    if m_lo > m_hi:
        return a_old, conv.max()
    
    offsets = np.arange(m_lo, m_hi + 1) - j
    a_slice = a[offsets]  # a[m_lo-j : m_hi-j+1] = a[0 : n] basically
    
    has_quad = (m_lo <= 2 * j <= m_hi)
    quad_idx = 2 * j - m_lo if has_quad else -1

    quad_base = base[quad_idx] - a_old * a_old if has_quad else 0.0

            vals = vals.copy()
            vals[quad_idx] = quad_base + x * x
        else:
            vals = base + 2.0 * delta * a_slice
        return max(np.max(vals), unaffected_max)
    
    best_x = a_old
    best_val = eval_at(a_old)

        else:

    v_star = eval_at(x_star)
    if v_star < best_val:
        best_val = v_star
        best_x = x_star
    
    return best_x, best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LSE gradient descent with adaptive beta
2. Exact coordinate descent (ternary search per coordinate)
3. Frank-Wolfe with LP oracle for global direction
4. U-shaped spectral initialization + multi-length exploration

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rhos = np.arange(1, n + 1)
    cond = u * rhos > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - jj] for jj in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 12})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lse_nesterov(f0, time_limit, beta_start=20.0, beta_end=5000.0):
    """Nesterov-accelerated LSE gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    no_improve = 0

    t_nest = 1.0
    y = f.copy()
    x_prev = f.copy()
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        grad2 = compute_lse_grad(w2, y_new)

        conv_new = fast_convolve(x_new)

            no_improve = 0

        else:
            no_improve += 1

        y = np.maximum(y, 0.0)

        if it % 300 == 0:
