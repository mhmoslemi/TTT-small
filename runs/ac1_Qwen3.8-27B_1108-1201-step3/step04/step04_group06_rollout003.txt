"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a refined exponentiated gradient
method with log-sum-exp smoothing, FFT-accelerated subgradients, adaptive
step sizes, interleaved coordinate descent, and multi-scale exploration.
Key improvement over previous: uses proper cross-correlation via FFT for O(n
log n) subgradient computation, log-sum-exp (not top-k) for smoother
optimization landscape, and a dedicated coordinate descent phase.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve(a, b):
    """Full linear convolution via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_sg_ksm(a, c, temperature):
    """
    Compute subgradient of log-sum-exp smoothed max(c) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) efficiency.
    sg[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]  where w = softmax(c/temp)
    """
    n = len(a)
    len_c = len(c)

    # Softmax weights
    scaled = c / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len_c)
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum

    # Compute sg[j] = 2 * dot(a, w[j:j+n]) using cross-correlation via FFT
    # R[j] = sum_m a[m]*w[m+j] = (a * flip(w))[2n-2-j]
    b_rev = w[::-1]
    conv = fast_convolve(a, b_rev)
    # conv has length n + (2n-1) - 1 = 3n-2
    # We need conv[2n-2-j] for j=0..n-1, i.e., conv[n-1 : 2n-1] reversed
    sg = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return sg


def compute_sg_topk(a, c, top_k):
    """Compute subgradient using top-k peaks (hard selection)."""
    n = len(a)
    len_c = len(c)

    if top_k >= len_c:
        indices = np.arange(len_c)
    else:
        indices = np.argpartition(c, -top_k)[-top_k:]
    weights = np.ones(len(indices)) / len(indices)

    w_full = np.zeros(len_c)
    for idx in range(len(indices)):
        w_full[int(indices[idx])] = weights[idx]

    # Same FFT-based cross-correlation
    b_rev = w_full[::-1]
    conv = fast_convolve(a, b_rev)
    sg = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return sg


def identify_active_set(c, threshold_frac=0.999):
    """Identify lags where b[k] is near the maximum."""
    max_c = c.max()
    threshold = max_c * threshold_frac
    active = np.where(c >= threshold)[0]
    return active

def coordinate_descent_polish(a0, time_limit, max_rounds=5000):
    """

    best_obj = c.max()

    no_improve = 0

        improved = False

        for i in coords:
            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Try a few multiplicative factors
            factors = [0.7, 0.85, 0.95, 1.0/0.95, 1.0/0.85, 1.0/0.7]

                obj2 = c2.max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True

def block_coordinate_search(a0, time_limit, block_size=8):
    """

    a = np.maximum(a0.astype(np.float64), 0.0)

        improved = False

        if block_sum < 1e-12:
            continue

        # Try random perturbations of the block
        for trial in range(20):
            pert = rng.standard_normal(block_size) * 0.1

            if nb_sum < 1e-12:
                continue
            # Scale to keep total sum the same
            new_block = new_block * (block_sum / nb_sum)

            a[start:end] = new_block
            s2 = a.sum()
            if s2 < 1e-15:

            a = a / s2

def generate_inits(rng, n, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay (both directions)
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine window
    c1 = np.abs(np.cos(np.pi * 0.5 * x / n)) + 0.01
    inits.append(c1 / c1.sum())

    # Two peaks (asymmetric)
    for frac in [0.3, 0.4, 0.6]:
        p1 = np.exp(-0.5 * ((x - frac * n) / (0.1 * n)) ** 2)
        p2 = np.exp(-0.5 * ((x - (1 - frac) * n) / (0.15 * n)) ** 2)
        combined = p1 + 0.7 * p2
        inits.append(combined / combined.sum())

    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Sine-based
    for freq in [2, 3, 5]:
        s1 = np.abs(np.sin(np.pi * freq * x / n)) + 0.05
        inits.append(s1 / s1.sum())

    # From parent (if available and same length)
    if parent is not None and len(parent) == n:
        p = parent.astype(np.float64)
        p = p / p.sum()
        inits.append(p)
        for scale in [0.002, 0.01, 0.05]:
            pert = p + rng.standard_normal(n) * scale
            pert = np.maximum(pert, 0)
            ps = pert.sum()
            if ps > 1e-10:
                inits.append(pert / ps)

    return inits


def resample_sequence(a, n_new):
    """Resample a sequence to a new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new
