def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_convolution(c_base, delta_idx, delta_val):

def smooth_objective_and_grad(x, n, T):
    """
    Compute smooth objective and gradient for the reduced problem.
    x: array of length n-1 (the first n-1 elements)
    The last element is 1 - sum(x) to satisfy sum(a) = 1.
    Returns (objective, gradient w.r.t. x of length n-1).
    """
    a = np.empty(n)
    a[:n-1] = x
    a[n-1] = 1.0 - x.sum()
    
    # If last element is negative, add large penalty
    if a[n-1] < -1e-8:
        # Return a large value and gradient pushing a[n-1] up (i.e., x_i down)
        penalty = 1e6 * (a[n-1] ** 2)
        grad_pen = 2 * 1e6 * a[n-1] * (-np.ones(n-1))
        return penalty, grad_pen
    
    a = np.maximum(a, 0.0)
    
    c = fast_convolve(a)
    
    # Log-sum-exp smoothing
    c_max = c.max()
    if T < 1e-12:
        # Use exact max with subgradient
        obj = 2.0 * n * c_max
        # Subgradient: find all maximizers
        mask = c >= c_max - 1e-10
        w = np.zeros(len(c))
        w[mask] = 1.0 / mask.sum()
    else:
        scaled = (c - c_max) / T
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        log_sum = np.log(exp_scaled.sum()) + max_scaled
        obj = 2.0 * n * (c_max + T * log_sum)
        w = exp_scaled / exp_scaled.sum()
    
    # Gradient: d(obj)/d(a[i]) = 2n * sum_k w[k] * 2*a[k-i] (for valid k-i)
    # Using cross-correlation: grad_a[i] = 4n * sum_j a[j] * w[i+j]
    # which is 4n * (w cross-correlate a) at index i
    # For efficiency, use the direct sum
    grad_a = np.zeros(n)
    for i in range(n):
        # sum over k where 0 <= k-i < n, i.e., k from i to i+n-1
        k_start = i
        k_end = min(2 * n - 1, i + n - 1)
        if k_start <= k_end:

            # = sum_{j=0}^{n-1} w[i+j] * a[j] (with bounds)
            j_end = min(n - 1, k_end - i)
            grad_a[i] = 2.0 * np.dot(w[k_start:k_end + 1], a[0:j_end + 1])
    
    grad_obj = 2.0 * n * grad_a
    
    # Reduce: gradient w.r.t. x (first n-1 elements)

    grad_x = grad_obj[:n-1] - grad_obj[n-1]
    
    return obj, grad_x


def adam_optimize(a0, n, T_start, T_end, n_iter, lr_start=0.01, lr_end=0.0001,
                  beta1=0.8, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimizer for the smooth objective."""
    x = a0[:n-1].copy()

    v = np.zeros(n-1)

    a_best = np.append(x, 1.0 - x.sum())
    
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        obj, grad = smooth_objective_and_grad(x, n, T)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_x = x.copy()
            a_best = np.append(x, 1.0 - x.sum())
            if verbose and it % 500 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.maximum(x, 0.0)

    return a_best, best_obj


def lfgs_optimize(a0, n, T, max_iter=500, verbose=False):

            c = fast_convolve(a)
            no_improve = 0
            eta = max(eta, eta0 * 0.05)
            temp = max(temp, temp0 * 0.01)
    
    return best_a.copy(), best_obj


def generate_diverse_inits(rng, n, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        # Perturbed versions
        for scale in [0.001, 0.005, 0.01, 0.02, 0.05]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Exponential decay
    for beta in [0.5, 1.0, 2.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma * n)) ** 2)
        inits.append(g / g.sum())
    
    # Cosine
    for freq in [1, 2, 3]:
        c1 = np.abs(np.cos(np.pi * freq * x / n)) + 0.1
        inits.append(c1 / c1.sum())
    
    # Power law
    for p in [0.5, 2.0, 3.0]:
        pw = (x + 1.0) ** p
        inits.append(pw / pw.sum())
        pw2 = (n - x + 1.0) ** p
        inits.append(pw2 / pw2.sum())
    
    # Random Dirichlet
    for alpha in [0.3, 1.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha * 0.1)
        inits.append(r)
    
    # Sinusoidal modulation
    for freq in [2, 5, 10]:
        s1 = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        s1 = np.maximum(s1, 0.01)
        inits.append(s1 / s1.sum())
    
    return inits


def parametric_optimize(a0, n, time_limit, max_degree=10, verbose=False):
    """
    Try parametric fit: fit a polynomial to the sequence and optimize coefficients.
    This reduces dimensionality and may escape local minima.
    """
    a = a0.copy()
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    n_fit = len(a)
    x = np.linspace(0, 1, n_fit)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n
    t0 = time.monotonic()
    
    for degree in [3, 5, 7, max_degree]:
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        
        # Fit polynomial
        coeffs = np.polyfit(x, a, degree)
        
        # Optimize coefficients to minimize the objective
        def poly_obj(vec):
            a_try = np.polyval(vec, x)
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                return 1e10
            a_try = a_try / st
            c = fast_convolve(a_try)
            return 2.0 * n * c.max()
        
        # Use Nelder-Mead to optimize coefficients
        try:
            result = minimize(poly_obj, coeffs, method='Nelder-Mead',
                             options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-10})
            if result.fun < best_obj:
                a_try = np.polyval(result.x, x)
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try = a_try / st
                    best_obj = result.fun
                    best_a = a_try.copy()
                    if verbose:
                        print(f"    Parametric (deg={degree}): obj={best_obj:.10f}")
        except Exception:
            pass
    
    return best_a, best_obj


def resample_and_optimize(a0, n_new, time_limit, verbose=False):
    """Resample sequence to new length and optimize."""
    n_old = len(a0)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        a_new = np.ones(n_new) / n_new
    else:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function combining multiple optimization strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384")
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
    
    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.8f}")
    
    # === Phase 1: L-BFGS-B with moderate smoothing ===
    print(f"\n=== Phase 1: L-BFGS-B (T=0.01) ===")
    t1 = min(180, deadline - time.monotonic() - 800)
    if t1 > 30:
        print(f"  Alloc={t1:.0f}s")
        a_opt, _ = lfgs_optimize(parent, n, T=0.01, max_iter=300, verbose=True)
        try_seq(a_opt, "lbfgs_T01")
    
    # === Phase 2: L-BFGS-B with fine smoothing ===
    print(f"\n=== Phase 2: L-BFGS-B (T=0.001) ===")
    print(f"  Best so far: {best_score:.8f}")
    t2 = min(180, deadline - time.monotonic() - 650)
    if t2 > 20 and best_seq is not None:
        print(f"  Alloc={t2:.0f}s")
        a_opt, _ = lfgs_optimize(best_seq, n, T=0.001, max_iter=500, verbose=True)
        try_seq(a_opt, "lbfgs_T001")
    
    # === Phase 3: Adam optimizer ===
    print(f"\n=== Phase 3: Adam ===")
    print(f"  Best so far: {best_score:.8f}")
    t3 = min(250, deadline - time.monotonic() - 500)
    if t3 > 30 and best_seq is not None:
        print(f"  Alloc={t3:.0f}s")
        # Estimate iterations: each iter is O(n) for eval + O(n) for grad
        # For n=384, about 1000 iters per second roughly
        n_iter = int(t3 * 300)
        a_opt, _ = adam_optimize(best_seq, n, T_start=0.003, T_end=0.0001,
                                  n_iter=n_iter, lr_start=0.05, lr_end=0.001,
                                  verbose=True)
        try_seq(a_opt, "adam")
    
    # === Phase 4: Multiplicative weights (improved) ===
    print(f"\n=== Phase 4: Multiplicative Weights ===")

    t4 = min(250, deadline - time.monotonic() - 300)
    if t4 > 30 and best_seq is not None:

                                                    eta0=1.0, temp0=0.002,
                                                    temp_decay=0.9996, eta_decay=0.9999,

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    print(f"  Best so far: {best_score:.8f}")
    t5 = min(200, deadline - time.monotonic() - 150)
    if t5 > 20 and best_seq is not None:
        print(f"  Alloc={t5:.0f}s")
        # Randomize coordinate order seed
        np.random.seed(int(time.time()) % 10000)

                if rem2 > 20:

                                                                temp_decay=0.9998, eta_decay=0.99995,

    print(f"  Best so far: {best_score:.8f}")
    t7 = deadline - time.monotonic() - 80
    if t7 > 30 and best_seq is not None:
        for n_new in [n + 64, n - 64, n + 128, 512, 256]:

                continue
            rem = deadline - time.monotonic() - 60

            a_resampled = resample_and_optimize(best_seq, n_new, 0)

        try_seq(a_opt, "lbfgs_fine")
        
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            np.random.seed(42)
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30), max_sweeps=100, verbose=False)
            try_seq(a_opt, "coord_fine")
    
    # === Phase 9: Very fine multiplicative weights ===
    print(f"\n=== Phase 9: Fine MW ===")
    print(f"  Best so far: {best_score:.8f}")
    t9 = deadline - time.monotonic() - 10
    if t9 > 10 and best_seq is not None:

                                                    eta0=0.1, temp0=0.0003,
                                                    temp_decay=0.99995, eta_decay=0.99999,
                                                    top_k_frac=0.8, verbose=False)
        try_seq(a_opt, "mw_fine")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
