                mutated = np.maximum(mutated, 0)
                if mutated.sum() > 1e-10:
                    mutated = mutated / mutated.sum()
                population.append(mutated)

            # 删除重复项并保持种群大小稳定。
            population = list(set(tuple(seq) for seq in population))

        if best_sequence is None:
            initial = np.ones(base) / base
            score_val = evaluate_sequence(initial.tolist())
            best_score = score_val
            best_sequence = initial

        print(f"Optimization completed with best objective {best_score:.6f}")
        return best_sequence.tolist()

    except Exception as e:
        print(f"Error in propose_candidate: {e}")
