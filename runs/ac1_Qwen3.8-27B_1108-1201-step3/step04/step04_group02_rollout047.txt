Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient descent with adaptive tau and BB step sizes

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_lse(a, c, tau, top_k=None):
    """Compute subgradient of LSE-smoothed max of auto-convolution."""
    n = len(a)
    conv_len = 2 * n - 1
    
    # Softmax weights
    max_c = c.max()
    scaled = tau * (c - max_c)
    w = np.exp(scaled)
    
    if top_k is not None and top_k < conv_len:
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(conv_len)
        w[idx] = np.exp(tau * (c[idx] - max_c))
    
    wsum = w.sum()
    if wsum < 1e-30:
        # Degenerate: use hard max
        k_star = c.argmax()
        w = np.zeros(conv_len)
        w[k_star] = 1.0
    else:
        w /= wsum
    
    # g[i] = 2 * sum_j w[j+i] * a[j] (cross-correlation)
    fft_size = 1 << (conv_len + n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cc = np.fft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * cc[:n]
