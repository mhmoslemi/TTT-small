"""
Optimized step function search using symmetric LP-based optimization with
adaptive constraint selection, multiple length exploration, and structured
perturbation. Key improvements: symmetry enforcement, active constraint
selection, better initialization, and adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_and_conv(a):
    """Compute objective and convolution for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    obj = 2.0 * n * max_conv
    return obj, conv, max_conv


def make_symmetric(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def build_conv_matrix_symmetric(a, n):
    """
    Build matrix for symmetric variables.
    If a is symmetric, a[i] = x[i] for i in [0, n//2].
    The convolution (a*g) where both a and g are symmetric can be
    expressed in terms of the first-half variables.
    
    Actually, let's just use the full matrix but with fewer variables.
    """
    # For simplicity, just use the full n-variable formulation
    # Symmetry will be enforced after each step
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M, n, active_only=False, threshold=0.85):
    """
    Solve LP: max sum(g) s.t. C@a g <= M, g >= 0.
    Optionally use only active constraints.
    """
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    
    if active_only:
        # Only include constraints where C @ a is close to M
        current_conv = C @ a
        mask = current_conv >= threshold * M
        C_sub = C[mask]
        b_sub = np.full(mask.sum(), M)
    else:
        C_sub = C
        b_sub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=C_sub, b_ub=b_sub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        return g, g.sum()
    return None, 0.0

    """Adaptive line search."""
    best_t = 0.0

def sg_polish(a0, time_limit, lr_init=0.004, momentum=0.95, top_k=50, temp=0.0002):
    """Subgradient polish with symmetry."""
    a = make_symmetric(np.maximum(a0.astype(np.float64), 0.0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    obj, conv, _ = obj_and_conv(a)
    best_obj = obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient from top-k peaks
        top_k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        scaled = (peak_vals - peak_vals.min()) / max(peak_vals.max() - peak_vals.min(), 1e-15)
        w = np.exp(scaled / temp)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]
        
        # Symmetrize gradient
        sg = make_symmetric(sg)
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = make_symmetric(np.maximum(a_new, 0.0))
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        
        obj, conv, _ = obj_and_conv(a)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve % 100 == 0:
            lr *= 0.99
            temp *= 1.01
        
        if no_improve >= 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            noise = make_symmetric(rng.standard_normal(n) * 0.002)
            a = make_symmetric(np.maximum(best_a + noise, 0.0))
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            obj, conv, _ = obj_and_conv(a)
            no_improve = 0
    
    return best_a, best_obj


def get_initializations(n, parent=None, rng=None):
    """Generate diverse symmetric initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - mid) / n))
    inits['raised_cos'] = rc / rc.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - mid) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Sine window
    sw = np.sin(np.pi * (x + 0.5) / n)
    inits['sine'] = sw / sw.sum()
    
    # Flat-top (Kaiser-like)
    # Use a polynomial approximation
    t = (x - mid) / (n / 2)
    flat = 1.0 - 0.5 * t**4
    flat = np.maximum(flat, 0.01)
    inits['flattop'] = flat / flat.sum()
    
    # Parent (resampled)
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        new_seq = np.interp(np.linspace(0, 1, n), x_old, parent)
        new_seq = make_symmetric(np.maximum(new_seq, 0.0))
        s = new_seq.sum()
        if s > 1e-15:
            inits['parent'] = new_seq / s
    
    return inits
