```python
import scipy.signal

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(a, T):
    """Compute gradient of soft-max approximation to max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    shifted = T * conv - T * conv.max()
    w = np.exp(shifted)
    w /= w.sum()

    # This is 2 * correlate(w, a, 'valid')
    g = 2.0 * scipy.signal.correlate(w, a, mode='valid')
    return g, conv.max()
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using soft-max gradient descent
with Adam optimizer, multi-temperature annealing, pair-swap local search,
and multi-length exploration.
"""

import numpy as np
import scipy.signal
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, T, top_k=None):
    """Compute gradient of soft-max approximation to max(conv(a,a)).
    
    Returns (gradient, current_max).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    if top_k is not None and top_k < len(conv):
        # Use top-k for efficiency
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        shifted = T * vals - T * vals.max()
        w_sub = np.exp(shifted)
        w_sub /= w_sub.sum()
        
        # Build full weight vector (sparse)
        w = np.zeros(len(conv))
        w[idx] = w_sub
    else:
        shifted = T * conv - T * conv.max()
        w = np.exp(shifted)
        w /= w.sum()
    
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    g = 2.0 * scipy.signal.correlate(w, a, mode='valid')
    return g, conv.max()

    v = np.zeros(n)

    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp
```

But actually, I can make pair swap much faster by computing the change in the convolution analytically:
- If I change a[i] by +d and a[j] by -d, the convolution changes by:
  - conv_new[k] = conv_old[k] + d*(a[k-i] + a[i-k]) - d*(a[k-j] + a[j-k])

```python
def pair_swap_fast(f, conv, i, j, d):
    """Compute new max of convolution after swapping mass d from j to i."""
    n = len(f)
    # Change in conv: +d*(a[k-i] + a[i-k]) - d*(a[k-j] + a[j-k])
    delta = np.zeros(2*n-1)
    # +d * a[k-i] for valid k-i (0 <= k-i < n, i.e., i <= k <= i+n-1)
    delta[i:i+n] += d * f[:n]  # a[k-i] for k = i to i+n-1
    # +d * a[i-k] for valid i-k (0 <= i-k < n, i.e., i-n+1 <= k <= i)
    if i > 0:
        delta[:i+1] += d * f[:i+1][::-1]  # a[i-k] for k=0 to i
    # -d * a[k-j]
    delta[j:j+n] -= d * f[:n]
    # -d * a[j-k]
    if j > 0:
        delta[:j+1] -= d * f[:j+1][::-1]
    
    conv_new = conv + delta
    return conv_new.max()
```

Wait, but I also need to update f[i] and f[j]. Let me be more careful.

If f[i] increases by d and f[j] decreases by d:
- The new f is f' where f'[i] = f[i]+d, f'[j] = f[j]-d
- conv'[k] = conv[k] + d*(f[k-i] + f[i-k]) - d*(f[k-j] + f[j-k])
  (using the ORIGINAL f for the cross terms, since the O(d^2) terms are negligible for small d)

Actually, for exact computation:
conv'[k] = sum_m f'[m] * f'[k-m]
= sum_m (f[m] + d*delta[m-i] - d*delta[m-j]) * (f[k-m] + d*delta[k-m-i] - d*delta[k-m-j])
= conv[k] + d*sum_m delta[m-i]*f[k-m] - d*sum_m delta[m-j]*f[k-m] 
  + d*sum_m f[m]*delta[k-m-i] - d*sum_m f[m]*delta[k-m-j]
  + O(d^2)

The O(d^2) terms are:
d^2 * delta[k-2i] - d^2*delta[k-i-j] - d^2*delta[k-j-i] + d^2*delta[k-2j]

For small d, we can ignore these. So:
conv'[k] ≈ conv[k] + d*(f[k-i] + f[i-k] - f[k-j] - f[j-k])

where f[x] = 0 for x < 0 or x >= n.

This is clean and computable in O(n) per swap trial.

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent with Adam
2. Fast pair-swap local search with O(n) delta computation
3. Multi-length exploration via interpolation
4. Multi-temperature annealing and periodic restarts
"""

import numpy as np
import scipy.signal
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, T, top_k=None):
    n = len(a)
    conv = fast_convolve(a)
    conv_len = len(conv)
    
    if top_k is not None and top_k < conv_len:
        k = min(top_k, conv_len)
        idx = np.argpartition(conv, -k)[-k:]
        vals = conv[idx]
        shifted = T * vals - T * vals.max()
        w_sub = np.exp(shifted)
        w_sub /= w_sub.sum()
        w = np.zeros(conv_len)
        w[idx] = w_sub
    else:
        shifted = T * conv - T * conv.max()
        w = np.exp(shifted)
        w /= w.sum()
    
    g = 2.0 * scipy.signal.correlate(w, a, mode='valid')
    return g, conv.max()


def softmax_adam_optimize(f0, time_limit, T_init=200, T_max=5000,
                          lr_init=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                          top_k=100, restart_interval=300, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    restart_scale = 0.01
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        g, cur_obj = compute_gradient(f, T, top_k=top_k)
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        new_obj = fast_convolve(f_new).max()
        f = f_new
        
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= restart_interval:
            rng = np.random.default_rng(t * 7919 + n)
            f = project_simplex(best_f + rng.standard_normal(n) * restart_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            T = min(T * 1.5, T_max)
            restart_scale = max(restart_scale * 0.5, 0.0005)
            no_improve = 0
        
        if t % 200 == 0:
            T = min(T * 1.08, T_max)
            lr *= 0.98
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam iter={t}, obj={best_obj:.10f}, T={T:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_obj


def pair_swap_search(f0, time_limit, n_trials=100000, verbose=False):

    n = len(f)

    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    improvements = 0
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:

                    best_d = mid
                    hi_b = mid

                    lo_b = mid

        best_val_neg = best_obj

            d = -frac * f[i]

            if val < best_val_neg:
                best_val_neg = val
                best_d_neg = d
        
        if best_d_neg < 0:
            lo_b, hi_b = max(-f[i], best_d_neg - 0.1*f[i]), min(0.0, best_d_neg + 0.1*f[i])
            for _ in range(8):

                    best_d_neg = mid
                    lo_b = mid
                else:
                    hi_b = mid

            d_best = best_d

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```
