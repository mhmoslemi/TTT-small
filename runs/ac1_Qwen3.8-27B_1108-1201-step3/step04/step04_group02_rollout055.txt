```python
import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_objective(f):
    """Compute the objective value (assuming sum(f) = 1)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def smoothed_max(conv, temp):
    """Compute smoothed max using log-sum-exp."""
    m = conv.max()
    return m + temp * np.log(np.sum(np.exp((conv - m) / temp)))


def compute_smooth_subgradient(f, conv, temp, top_k=None):
    """Compute subgradient of smoothed max of convolution."""
    n = len(f)
    if top_k is not None and top_k < len(conv):
        # Only consider top-k peaks for efficiency
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        m = pv.max()
        scaled = (pv - m) / temp
        w = np.exp(scaled)
        wsum = w.sum()
        w /= wsum
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
    else:
        m = conv.max()
        scaled = (conv - m) / temp
        w = np.exp(scaled)
        wsum = w.sum()
        w /= wsum

            je = min(n - 1, k)

    return sg

def coordinate_descent(f0, time_limit, epochs=10, n_sweep=16):
    """Full coordinate descent: cycle through all coordinates multiple times."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / S
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for epoch in range(epochs):
        if time.monotonic() - t0 > time_limit - 1:
            break
        improved = False
        for m in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break
            f_new, conv_new = coordinate_descent_step(f, conv, m, n_sweep=n_sweep)
            if not np.allclose(f_new, f):
                f = f_new
                conv = conv_new
                improved = True
            obj = 2.0 * n * conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
        if not improved:
            break
    
    return best_f, best_obj
```

```python
    alpha = np.zeros(conv_len)
    base = conv.copy()
    
    km_min = max(0, m)  # k-m >= 0 => k >= m
    km_max = min(conv_len, m + n)  # k-m < n => k < m+n
    
    for k in range(km_min, km_max):
        if k == 2 * m:
            base[k] = conv[k] - a_m * a_m
            # alpha[k] = 0, beta = 1
        else:
            idx = k - m
            base[k] = conv[k] - 2 * a_m * f[idx]
            alpha[k] = 2 * f[idx]
    
    # Now objective(t) = 2n * max(base + alpha*t + beta*t^2) / (S - a_m + t)^2
    # where beta[2m] = 1, all other beta = 0
    
    S_rem = S - a_m  # sum of all other elements
    
    def obj(t):
        if t < 0:
            return np.inf
        vals = base + alpha * t
        if 2 * m < conv_len:
            vals[2*m] += t * t
        new_S = S_rem + t
        if new_S < 1e-10:
            return np.inf
        return 2.0 * n * vals.max() / (new_S * new_S)
    
    # Search for optimal t in [0, hi]
    hi = max(10 * a_m + 0.01, 0.5)
    
    # Grid scan
    ts = np.linspace(0, hi, n_sweep)
    vals_obj = np.array([obj(t) for t in ts])
    best_idx = np.argmin(vals_obj)
    best_t = ts[best_idx]
    best_val = vals_obj[best_idx]
    
    # Golden section refinement
    gr = (np.sqrt(5) - 1) / 2
    if best_idx > 0:
        lo = ts[best_idx - 1]
    else:
        lo = 0.0
    if best_idx < n_sweep - 1:
        hi_g = ts[best_idx + 1]
    else:
        hi_g = hi
    
    a_g = lo
    b_g = hi_g
    c_g = b_g - gr * (b_g - a_g)
    d_g = a_g + gr * (b_g - a_g)
    fc = obj(c_g)
    fd = obj(d_g)
    
    for _ in range(40):
        if b_g - a_g < 1e-10:
            break
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr * (b_g - a_g)
            fc = obj(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr * (b_g - a_g)
            fd = obj(d_g)
    
    best_t = (a_g + b_g) / 2
    best_val = obj(best_t)
    
    # Check if improvement
    current_val = 2.0 * n * conv.max() / (S * S)
    if best_val < current_val - 1e-12:
        f[m] = best_t
        S = f.sum()
        f = f / S
        conv = fast_convolve(f)
    
    return f, conv
```

This is much faster because each `obj(t)` evaluation is O(n) (just a vector operation), not O(n log n) (FFT).

Wait, but `vals.max()` is O(2n) which is fine. And we do this ~50 times per coordinate, so total per epoch is O(n * 50 * n) = O(n^2 * 50). For n=384, that's about 7.4M operations per epoch. Very fast!

The advantage is that it stays strictly in the interior (all components positive) and can make large relative changes to small components.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with O(n) per-evaluation 1D optimization
2. Mirror descent (exponentiated gradient) with adaptive temperature annealing
3. Population-based local search with diverse structured initializations
4. LP-based Frank-Wolfe for occasional global moves
5. Multi-length exploration (256-768)
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective assuming sum(f) = 1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def coord_descent_epoch(f, conv, n_sweep=32, random_order=False, rng=None):
    """One full epoch of coordinate descent."""
    n = len(f)
    if random_order and rng is not None:
        order = rng.permutation(n)
    else:
        order = range(n)
    
    improved = False
    for m in order:
        a_m = f[m]
        S = 1.0  # f is normalized
        conv_len = 2 * n - 1
        
        # Precompute base and alpha
        base = conv.copy()
        alpha = np.zeros(conv_len)
        
        km_lo = m
        km_hi = min(conv_len, m + n)
        
        for k in range(km_lo, km_hi):
            idx = k - m
            if k == 2 * m:
                base[k] = conv[k] - a_m * a_m
            else:
                base[k] = conv[k] - 2.0 * a_m * f[idx]
                alpha[k] = 2.0 * f[idx]
        
        S_rem = 1.0 - a_m
        
        # 1D optimization
        def obj_t(t):
            vals = base + alpha * t
            if 2 * m < conv_len:
                vals[2 * m] += t * t
            ns = S_rem + t
            if ns < 1e-10:
                return np.inf
            return 2.0 * n * vals.max() / (ns * ns)
        
        hi = max(10.0 * a_m + 0.05, 0.3)
        ts = np.linspace(0, hi, n_sweep)
        objs = np.array([obj_t(t) for t in ts])
        best_idx = np.argmin(objs)
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = ts[max(0, best_idx - 1)]
        hi_g = ts[min(n_sweep - 1, best_idx + 1)]
        
        a_g, b_g = lo, hi_g
        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        fc, fd = obj_t(c_g), obj_t(d_g)
        
        for _ in range(35):
            if b_g - a_g < 1e-11:
                break
            if fc < fd:
                b_g, d_g, fd = d_g, c_g, fc
                c_g = b_g - gr * (b_g - a_g)
                fc = obj_t(c_g)
            else:
                a_g, c_g, fc = c_g, d_g, fd
                d_g = a_g + gr * (b_g - a_g)
                fd = obj_t(d_g)
        
        best_t = (a_g + b_g) / 2.0
        if best_t < 0:
            best_t = 0.0
        new_obj = obj_t(best_t)
        cur_obj = 2.0 * n * conv.max()
        
        if new_obj < cur_obj - 1e-10:
            f[m] = best_t
            S_new = f.sum()
            f = f / S_new
            conv = fast_convolve(f)
            improved = True
    
    return f, conv, improved


def coordinate_descent(f0, time_limit, epochs=50, n_sweep=32, random_order=False, seed=0):
    """Full coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for ep in range(epochs):
        if time.monotonic() - t0 > time_limit - 2:
            break
        f, conv, improved = coord_descent_epoch(f, conv, n_sweep=n_sweep,
                                                 random_order=random_order, rng=rng)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5 and not improved:
            break
    
    return best_f, best_obj


def mirror_descent(f0, time_limit, initial_temp=0.1, final_temp=0.001,
                   decay_rate=0.995, n_sweep=32, cd_every=20, seed=0):
    """Mirror descent (exponentiated gradient) with periodic coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    temp = initial_temp
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time.monotonic()  # actually time_limit - (time.monotonic()-t0)
        if rem < 2:
            break
        
        # Compute subgradient (softmax-weighted over all peaks)
        m = conv.max()
        scaled = (conv - m) / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            # Pure max
            peaks = np.where(conv >= m - 1e-10)[0]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / len(peaks)
        else:
            w /= wsum
        
        # Subgradient
        sg = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[k] * f[k - idx]
        
        # Mirror descent step: f_new ∝ f * exp(-eta * sg)
        # Choose eta adaptively
        eta = 1.0 / (np.max(np.abs(sg)) + 1e-8)
        eta = min(eta, 0.5)  # Cap step size
        
        log_f = np.log(np.maximum(f, 1e-30))
        log_f_new = log_f - eta * sg
        f_new = np.exp(log_f_new - log_f_new.max())
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        temp *= decay_rate
        
        # Periodic coordinate descent
        if cd_every > 0 and it % cd_every == 0:
            f_cd, obj_cd = coordinate_descent(f, time_limit=min(30, time_limit - (time.monotonic()-t0)-2),
                                               epochs=3, n_sweep=n_sweep, seed=it)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_f = f_cd.copy()
                f = f_cd
                conv = fast_convolve(f)
        
        # Restart if stuck
        if no_improve > 100:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(f)
            temp = initial_temp * 0.5
            no_improve = 0
    
    # Final coordinate descent polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        f_cd, obj_cd = coordinate_descent(best_f, time_limit=max(5, rem - 2),
                                          epochs=10, n_sweep=64, seed=12345)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_f = f_cd.copy()
    
    return best_f, best_obj


def solve_lp_quick(f, M, time_limit=10):
    """Quick LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix sparsely
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # This is a (2n-1) x n matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if f[j] < 1e-15:
            continue
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_step(f, M, time_limit=15):
    """One Frank-Wolfe step with LP oracle."""
    n = len(f)
    g = solve_lp_quick(f, M, time_limit)
    if g is None or g.sum() < 1e-10:
        return None
    g = g / g.sum()
    return g


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())
    
    q = (x - n / 2.0) ** 2 + 1.0
    inits.append(q / q.sum())
    
    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())
    
    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - n/2) / (n/6)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine^2
    c = np.cos(np.pi * (x / (n-1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Tukey/Hanning window
    tw = 0.5 * (1 + np.cos(2 * np.pi * x / (n-1))) + 0.01
    inits.append(tw / tw.sum())
    
    # Blackman
    bw = 0.42 + 0.5 * np.cos(2*np.pi*x/(n-1)) + 0.08 * np.cos(4*np.pi*x/(n-1))
    bw = np.maximum(bw, 0.01)
    inits.append(bw / bw.sum())
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with O(n) per-evaluation 1D optimization
2. Mirror descent (exponentiated gradient) with adaptive temperature annealing

import numpy as np

from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, temp, top_k=None):
    """Compute subgradient of smoothed max of auto-convolution."""
    n = len(f)
    cl = len(conv)
    if top_k is not None and top_k < cl:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    else:
        peaks = np.arange(cl)
        pv = conv
    
    m = pv.max()
    scaled = (pv - m) / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def coord_descent_epoch(f, conv, n_sweep=32, order=None):
    """One full epoch of vectorized coordinate descent. f must be sum=1."""
    n = len(f)
    conv_len = 2 * n - 1
    
    if order is None:
        order = range(n)
    
    improved = False
    for m in order:
        a_m = f[m]
        if a_m < 1e-15:
            # Try setting it to a small positive value
            hi_try = 0.05
        else:
            hi_try = max(10.0 * a_m + 0.05, 0.3)
        
        # Vectorized computation of base and alpha
        km_lo = m
        km_hi = min(conv_len, m + n)
        
        base = conv.copy()
        alpha = np.zeros(conv_len)
        
        segment_len = km_hi - km_lo
        if segment_len > 0:
            idxs = np.arange(km_lo, km_hi)
            f_idxs = idxs - m  # 0 to segment_len-1
            base[km_lo:km_hi] = conv[km_lo:km_hi] - 2.0 * a_m * f[f_idxs]
            alpha[km_lo:km_hi] = 2.0 * f[f_idxs]
            
            # Special case: k = 2m
            if km_lo <= 2 * m < km_hi:
                pos = 2 * m
                base[pos] = conv[pos] - a_m * a_m
                alpha[pos] = 0.0
        
        S_rem = 1.0 - a_m
        
        # 1D objective: 2n * max(base + alpha*t + beta*t^2) / (S_rem + t)^2
        # beta is 1 at position 2m, 0 elsewhere
        has_quad = (2 * m < conv_len)
        quad_pos = 2 * m
        
        def obj_t(t):
            vals = base + alpha * t
            if has_quad:
                vals[quad_pos] += t * t
            ns = S_rem + t
            if ns < 1e-10:
                return np.inf
            return 2.0 * n * vals.max() / (ns * ns)
        
        # Grid scan
        ts = np.linspace(0.0, hi_try, n_sweep)
        objs = np.empty(n_sweep)
        for i in range(n_sweep):
            objs[i] = obj_t(ts[i])
        
        best_idx = int(np.argmin(objs))
        
        # Golden section refinement
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = ts[max(0, best_idx - 1)]
        hi_g = ts[min(n_sweep - 1, best_idx + 1)]
        
        a_g, b_g = lo, hi_g
        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        fc = obj_t(c_g)
        fd = obj_t(d_g)
        
        for _ in range(40):
            if b_g - a_g < 1e-12:

            if fc < fd:
                b_g = d_g
                d_g = c_g
                fd = fc
                c_g = b_g - gr * (b_g - a_g)
                fc = obj_t(c_g)
            else:
                a_g = c_g
                c_g = d_g
                fc = fd
                d_g = a_g + gr * (b_g - a_g)
                fd = obj_t(d_g)
        
        best_t = (a_g + b_g) / 2.0
        if best_t < 0:
            best_t = 0.0
        new_obj = obj_t(best_t)
        cur_obj = 2.0 * n * conv.max()
        
        if new_obj < cur_obj - 1e-10:
            f[m] = best_t
            S_new = f.sum()
            if S_new > 1e-10:
                f = f / S_new
                conv = fast_convolve(f)
                improved = True
    
    return f, conv, improved

def solve_lp_quick(f, M, time_limit=8):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if f[j] < 1e-15:
            continue
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True,
                      'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

                f_init = project_simplex(0.85 * f_start + 0.15 * (np.ones(n_best) / n_best))
            
            f_opt, obj_opt = mirror_descent(f_init, time_limit=t_alloc,
                                             initial_temp=0.08,
                                             decay_rate=0.995,
                                             cd_every=30, n_sweep=36,
                                             seed=300 + run)
            try_seq(f_opt, f"mirror_{run}")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: LP-based escape + CD ===
    print("\n=== Phase 4: LP Escape + CD ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        f_cur = best_seq / best_seq.sum()
        conv_cur = fast_convolve(f_cur)
        M_cur = conv_cur.max()
        
        for it in range(min(15, max(3, int((deadline - time.monotonic() - 80) // 40)))):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            
            # Try LP oracle
            g = solve_lp_quick(f_cur, M_cur * 1.002, time_limit=8)
            
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                best_mix = None
                best_mix_obj = np.inf
                for alpha in [0.2, 0.4, 0.6, 0.8]:
                    f_mix = project_simplex((1 - alpha) * f_cur + alpha * g)
                    f_cdm, obj_cdm = coordinate_descent(f_mix, time_limit=min(20, rem - 10),
                                                         epochs=8, n_sweep=36,
                                                         random_order=True,
                                                         seed=it * 13 + alpha * 100)
                    if obj_cdm < best_mix_obj:
                        best_mix_obj = obj_cdm
                        best_mix = f_cdm
                
                if best_mix is not None:
                    try_seq(best_mix, f"lp_cd_{it}")
                    f_cur = best_mix
                    M_cur = fast_convolve(f_cur).max()
            else:
                # Fallback: just CD
                f_opt, obj_opt = coordinate_descent(f_cur, time_limit=min(25, rem - 10),
                                                     epochs=12, n_sweep=40,
                                                     random_order=True,
                                                     seed=it * 7 + 3)
                try_seq(f_opt, f"cd_escape_{it}")
                f_cur = f_opt
    
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Final deep polish ===
    print("\n=== Phase 5: Final Polish ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        f_start = best_seq / best_seq.sum()
        
        # Very fine CD
        rem = deadline - time.monotonic() - 25
        if rem > 20:
            f_opt, obj_opt = coordinate_descent(f_start, time_limit=min(rem, 100),
                                                 epochs=80, n_sweep=72,
                                                 random_order=True, seed=9999)
            try_seq(f_opt, "final_cd_fine")
        
        # Mirror descent with low temp
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            f_start2 = best_seq / best_seq.sum()
            f_opt, obj_opt = mirror_descent(f_start2, time_limit=min(rem, 50),
                                             initial_temp=0.02,
                                             decay_rate=0.997,
                                             cd_every=15, n_sweep=48,
                                             seed=8888)
            try_seq(f_opt, "final_mirror")
        
        # Micro perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 12:
            for _ in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 8:
                    break
                f_pert = project_simplex(best_seq / best_seq.sum() +
                                         rng.standard_normal(n_best) * 0.004)
                f_opt, obj_opt = coordinate_descent(f_pert,
                                                     time_limit=min(rem, 15),
                                                     epochs=15, n_sweep=56,
                                                     random_order=True, seed=444)
                try_seq(f_opt, "micro")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
