"""
Spectral gradient descent with softmax smoothing, coordinate descent, 
and LP-FW hybrid for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_gradient(a, tau):
    """
    Compute the gradient of softmax(conv(a,a)/tau) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    
    # Softmax weights
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient: grad[j] = 2/tau * sum_m w[j+m] * a[m]
    # This is a cross-correlation, computable via FFT
    L = 1 << (size + n - 2 - 1).bit_length()  # >= size + n - 1
    W = np.fft.rfft(w, L)
    A_pad = np.fft.rfft(a, L)
    corr = np.fft.irfft(W * np.conj(A_pad), L)[:n]
    
    grad = 2.0 * corr / tau
    return grad, conv, w


def spectral_gd(a0, time_limit, tau_init=0.01, tau_rate=0.999, lr_init=0.01, 
                momentum=0.9, verbose=False):
    """
    Spectral gradient descent with momentum and adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv, w = spectral_gradient(a, tau)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # Still move, but track best
            no_improve += 1

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None


def lp_fw_step(f, M, n, line_search=True):
    """Single LP-FW step."""
    g = solve_lp_direction(f, M, n)
    if g is None:
        return None, None
    
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, None
    g = g / g_sum
    
    if line_search:
        best_alpha = 0.0
        best_val = fast_convolve(f).max()
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            f_t = project_simplex(f + alpha * (g - f))
            val = fast_convolve(f_t).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        return f + best_alpha * (g - f), best_val
    else:
        return f + 0.3 * (g - f), None


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value
    that minimizes the peak convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            # Try several values for coordinate j
            # Normalize: set a[j] = x, scale others so sum = 1
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5,
                         old_val + 0.01, old_val - 0.01]
            # Also try extremes
            if old_val > 0.01:
                candidates.append(0.0)
            candidates.append(0.5)
            
            best_x = old_val
            best_obj_j = best_obj
            
            for x in candidates:
                if x < 0 or x > 0.999:
                    continue
                f_trial = f.copy()
                f_trial[j] = x
                # Rescale others
                others_sum = 1.0 - x
                others_original_sum = 1.0 - old_val
                if others_original_sum > 1e-10:
                    scale = others_sum / others_original_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                
                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_x = x
            
            if best_x != old_val:
                f[j] = best_x
                others_sum = 1.0 - best_x
                others_original_sum = 1.0 - old_val
                if others_original_sum > 1e-10:
                    scale = others_sum / others_original_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                f = np.maximum(f, 0)
                f /= f.sum()
                
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_gradient(a, tau):
    """
    Compute gradient of softmax(conv(a,a)/tau) w.r.t. a using FFT.
    Returns (gradient, convolution).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    conv = np.maximum(conv, 0.0)

    # Softmax weights over the convolution
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # Cross-correlation g[j] = sum_m w[j+m] * a[m] via FFT
    L = 1 << (3 * n - 3).bit_length()
    W = np.fft.rfft(w, L)
    A_pad = np.fft.rfft(a, L)
    corr = np.fft.irfft(np.conj(W) * A_pad, L)[:n]

    grad = 2.0 * corr / tau
    return grad, conv


def spectral_gd(a0, time_limit, tau_init=0.01, tau_decay=0.999, lr_init=0.01,
                momentum=0.9, verbose=False):
    """Spectral gradient descent with momentum, adaptive tau, and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = spectral_gradient(a, tau)

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        tau *= tau_decay
        lr = lr_init * (0.9995 ** it)

        if no_improve > 300:
            seed_val = (it * 2654435761 + n) & 0x7FFFFFFF
            rng = np.random.default_rng(seed_val)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            no_improve = 0
            tau = max(tau, 0.008)

        if verbose and it % 200 == 0:
            print(f"    SpecGD it={it}: obj={best_obj:.10f} tau={tau:.5f}")

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None
