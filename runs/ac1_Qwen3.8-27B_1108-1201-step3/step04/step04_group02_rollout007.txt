`(w ⋆_N f)[j] = sum_{m=0}^{N-1} w[(j+m) mod N] * f[m]`

If N >= len(w) + len(f) - 1 = 3n-2, then the circular correlation equals the linear correlation for the valid range.

Specifically, `(w ⋆ f)[j] = (w ⋆_N f)[j]` for j = 0, ..., n-1 (since j+m < N for all valid m).

So: `corr = ifft(fft(w, N) * conj(fft(f, N)))[:n]`

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def soft_gradient(f, beta):
    """Gradient of (1/beta)*log(sum exp(beta*conv(f,f)[k]))."""
    n = len(f)
    conv = fast_convolve(f)
    mx = conv.max()
    shifted = beta * (conv - mx)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    
    # grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * f[m]
    # Cross-correlation via FFT
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(weights, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)[:n]
    return 2.0 * corr


def lp_direction(f, M, time_limit=5.0):
    """Find direction via LP: max sum(g) s.t. conv(f,g)[k]<=M, g>=0."""
    n = len(f)
    if n > 2000:
        return None  # Too large for LP
    
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
