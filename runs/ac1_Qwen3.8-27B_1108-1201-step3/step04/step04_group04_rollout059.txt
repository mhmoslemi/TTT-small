def softmax_grad(a, conv, top_k=50, temp=0.01):

    nk = 2 * n - 1

        peaks = np.argpartition(conv, -top_k)[-top_k:]

    scaled = cv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(cv)] = 1.0

        w /= w_sum

        grad[js] += w[idx] * (a[k - js] + a[js])

def init_uniform(n):
    return np.ones(n) / n

def init_monotonic(n):
    a = np.arange(1, n+1, dtype=np.float64)
    return a / a.sum()

def init_chirp(n, c=1.0):
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(2 * np.pi * c * j * (j + 1) / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def init_freq_chirp(n, c=1.0):
    # Chirp in frequency domain
    k = np.arange(n // 2 + 1)
    # Random magnitudes with chirp phase
    magnitudes = 1.0 + 0.5 * np.random.default_rng(42).random(n // 2 + 1)
    phases = np.pi * c * k * k / n
    A = magnitudes * np.exp(1j * phases)
    # Make it conjugate symmetric for real result
    if n % 2 == 0:
        A_full = np.concatenate([A, A[-2:0:-1]])
    else:
        A_full = np.concatenate([A, A[-1:0:-1]])
    a = np.fft.irfft(A_full, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_random(n, rng):
    a = rng.dirichlet(np.ones(n) * 0.5)
    return a

def init_parent_interp(parent, n):
    if parent is None or len(parent) == 0:
        return np.ones(n) / n
    p = np.maximum(parent, 0).astype(np.float64)
    ps = p.sum()
    if ps < 1e-10:
        return np.ones(n) / n
    p = p / ps
    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_combined(n, rng):
    """Combine monotonic and chirp."""
    j = np.arange(n, dtype=np.float64)
    mono = j / (n - 1) if n > 1 else np.ones(1)
    c = rng.uniform(0.5, 3.0)
    chirp = 1.0 + 0.5 * np.cos(2 * np.pi * c * j * (j + 1) / n)
    a = (1.0 + mono) * np.maximum(chirp, 0.1)

import numpy as np

def fast_conv(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_grad(a, conv, top_k, temp):
    """Compute softmax-weighted gradient of max(conv)."""
    n = len(a)
    nk = 2 * n - 1
    if top_k >= nk:
        peaks = np.arange(nk)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    cv = conv[peaks]
    scaled = cv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(cv)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        grad[js] += w[idx] * (a[k - js] + a[js])
    return grad


def exact_subgrad(a, conv):
    """Compute subgradient at the single maximum peak."""
    n = len(a)
    k = int(np.argmax(conv))
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    js = np.arange(j_start, j_end + 1)
    grad[js] = a[k - js] + a[js]
    return grad
