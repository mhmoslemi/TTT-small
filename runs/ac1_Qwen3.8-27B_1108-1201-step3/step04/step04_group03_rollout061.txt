import numpy as np

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def convolve_fft(a, n):
    """Linear auto-convolution via FFT."""
    N = 2 * n - 1
    size = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:N]

def cross_corr_for_grad(a, w, n):
    """Compute g[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1.
    This is the cross-correlation needed for the gradient."""
    N = 2 * n - 1
    size = 1 << (3 * n).bit_length()  # enough for the convolution
    # g[i] = sum_j a[j] * w[i+j]
    # This is (a_rev * w)[n-1+i] where a_rev = a[::-1]
    a_rev = a[::-1]
    result = np.fft.irfft(np.fft.rfft(a_rev, size) * np.fft.rfft(w, size))
    return result[n-1:2*n-1]

def objective_lp(a, n, p):
    """Compute sum(b[k]^p) where b = conv(a,a)."""
    b = convolve_fft(a, n)
    return np.sum(b**p)

def grad_lp(a, n, p):
    """Gradient of sum(b^p) w.r.t. a."""
    b = convolve_fft(a, n)
    w = p * b**(p-1)
    g = cross_corr_for_grad(a, w, n)
    return 2 * g  # factor of 2 from symmetry in db/da

    b = convolve_fft(a, n)

    tb -= tb.max()
    return (1/t) * np.log(np.sum(np.exp(tb)))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

def convolve_fft(a, n):
    """Linear auto-convolution via FFT."""
    N = 2 * n - 1
    size = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:N]


def cross_corr_grad(a, w, n):
    """Compute g[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1 via FFT."""
    size = 1 << (3 * n).bit_length()
    a_rev = a[::-1]
    result = np.fft.irfft(np.fft.rfft(a_rev, size) * np.fft.rfft(w, size))
    return result[n - 1:2 * n - 1]


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_score(a, n):
    """Compute the objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    b = convolve_fft(a, n)
    return 2.0 * n * b.max(), b


def grad_lp(a, b, n, p):
    """Gradient of sum(b^p) w.r.t. a, given b=conv(a,a)."""
    if p == 1:
        w = np.ones_like(b)
    else:
        w = p * np.power(np.maximum(b, 0.0), p - 1)
    g = cross_corr_grad(a, w, n)
    return 2.0 * g


def grad_softmax_smooth(a, b, n, t):
    """Gradient of (1/t)*log(sum(exp(t*b))) w.r.t. a."""
    tb = t * b
    tb -= tb.max()
    w = np.exp(tb)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    g = cross_corr_grad(a, w, n)
    return 2.0 * g


def softmax_func(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def peak_chase_optimize(a_init, n, budget, lr_base=0.002, seed=42,
                        top_k=5, mom=0.9):
    """Peak-chasing: use exact gradient of the top peak(s)."""
    rng = np.random.default_rng(seed)
    a = project_simplex(a_init.copy())
    best_score = float('inf')
    best_a = a.copy()
    velocity = np.zeros(n)
    t0 = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < budget:
        step += 1
        lr = lr_base / (1.0 + 0.001 * step)

        b = convolve_fft(a, n)
        # Find top-k peaks
        if top_k >= len(b):
            peak_idx = np.arange(len(b))
            weights = np.ones(len(b)) / len(b)
        else:
            peak_idx = np.argpartition(b, -top_k)[-top_k:]
            # Weight by value (normalized)
            pv = b[peak_idx]
            weights = pv / pv.sum()

        # Gradient of weighted sum of peaks
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            k = int(peak_idx[i])
            w_val = weights[i]
            # db[k]/da[j] = 2*a[k-j] for valid k-j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w_val * 2.0 * a[k - j]

        # Momentum + step
        velocity = mom * velocity - lr * grad
        a_new = project_simplex(a + velocity)

        score, _ = eval_score(a_new, n)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 10000:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            velocity = np.zeros(n)
            no_improve = 0

    return best_a, best_score


def lbfgs_smooth_optimize(a_init, n, budget, t=200.0, seed=42, max_iter=2000):

    a = project_simplex(a_init.copy())
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    z = z - z.mean()  # center for numerical stability

    n

    def objective_and_grad(z_vec):
        a_local = softmax_func(z_vec)
        b = convolve_fft(a_local, n)
        # Soft-max value (for L-BFGS to minimize)
        tb = t * b
        tb_max = tb.max()
        log_sum = tb_max + np.log(np.sum(np.exp(tb - tb_max)))
        val = (1.0 / t) * log_sum
        # Gradient w.r.t. a
        exp_tb = np.exp(tb - tb_max)
        w = exp_tb / exp_tb.sum()
        g_a = 2.0 * cross_corr_grad(a_local, w, n)
        # Chain rule: d/dz = a * (g_a - dot(a, g_a))
        mean_g = np.dot(a_local, g_a)
        g_z = a_local * (g_a - mean_g)
        return val, g_z

    best_score = float('inf')
    best_a = a.copy()

    # First evaluate initial
    score_init, _ = eval_score(a, n)
    best_score = score_init

    t_start = time.monotonic()
    n_restarts = 0

    while time.monotonic() - t_start < budget and n_restarts < 10:
        n_restarts += 1
        # Run L-BFGS
        try
            res = scipy_minimize(objective_and_grad, z, method='L-BFGS-B',
                                 jac=True, options={'maxiter': max_iter,
                                                    'maxfun': 5000,
                                                    'gtol': 1e-8})
            a_opt = softmax_func(res.x)
            score, _ = eval_score(a_opt, n)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_opt.copy()
        except Exception:
            pass

        # Check time
        if time.monotonic() - t_start > budget * 0.8:
            break

        # Restart with perturbation
        noise = rng.standard_normal(n) * 0.02
        a_pert = project_simplex(best_a + noise)
        a_pert = np.maximum(a_pert, 1e-15)
        z = np.log(a_pert)
        z = z - z.mean()

    return best_a, best_score


def coord_descent_polish(a_init, n, budget, seed=42):
    """Coordinate descent with grid search for final polish."""
    rng = np.random.default_rng(seed)
    a = project_simplex(a_init.copy())
    best_score, b = eval_score(a, n)
    best_a = a.copy()
    t0 = time.monotonic()

    n_grid = 12
    grid_fracs = np.linspace(0.0, 2.5, n_grid)

    order = rng.permutation(n)
    idx = 0

    while time.monotonic() - t0 < budget:
        j = int(order[idx % n])
        idx += 1

        a_j_orig = a[j]
        if a_j_orig < 1e-12:
            continue

        # Compute b without a[j] contribution
        # b[k] = b[k] - 2*a[j]*a[k-j] (for valid k-j) - a[j]^2 if k=2j
        b_save = b.copy()
        # Remove a[j] contribution
        for k in range(n):
            if 0 <= k - j < n:
                b_save[k] -= 2.0 * a_j_orig * a[k - j]
        if 2 * j < 2 * n - 1:
            b_save[2 * j] += a_j_orig * a[k if False else 0]  # placeholder

        # Actually let me just recompute b for each candidate
        S_base = 1.0 - a_j_orig
        a[j] = 0.0

        best_local = best_score
        best_a_j = a_j_orig

        for frac in grid_fracs:
            a_j_new = a_j_orig * frac
            a[j] = a_j_new
            S = S_base + a_j_new
            if S < 1e-15:
                continue
            a_norm = a / S
            b_cand = convolve_fft(a_norm, n)
            score_cand = 2.0 * n * b_cand.max()
            if score_cand < best_local:
                best_local = score_cand
                best_a_j = a_j_new

        a[j] = best_a_j
        a = a / a.sum()
        score, b = eval_score(a, n)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def generate_diverse_inits(n, parent, rng, num_inits=8):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        inits.append(parent.copy())
        # Perturbed parent
        for scale in [0.01, 0.05, 0.1]:
            inits.append(project_simplex(parent + rng.standard_normal(n) * scale))

    # Structured inits
    x = np.arange(n, dtype=np.float64)
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Edge-concentrated
    e = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 1.5
    inits.append(e / e.sum())
    # Cosine
    c = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n)) + 0.01
    inits.append(c / c.sum())
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    return inits
