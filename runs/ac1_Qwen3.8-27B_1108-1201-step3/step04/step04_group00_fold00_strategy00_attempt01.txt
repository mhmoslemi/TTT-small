def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size.bit_length() - 1)
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    Conv = np.fft.irfft(A * A, fft_size)
    return Conv[:size]
