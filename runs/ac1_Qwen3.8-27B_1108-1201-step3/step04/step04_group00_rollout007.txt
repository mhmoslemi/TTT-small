def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] for valid indices."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):

    a_safe = np.maximum(a, 1e-300)

def soft_max_obj_grad(z, T, n):
    """
    Compute the soft-max smoothed objective and its gradient w.r.t. z.
    Objective: (2n/T) * log(sum_k exp(T * b_k)) where b = conv(a,a)
    and a = softmax(z).
    """

    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights for each lag
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a_i = (2n) * sum_k w_k * a_{k-i} = (2n) * cross_correlation(w, a)_i
    grad_a = 2.0 * n * cross_corr(w, a)
    
    # Chain rule: grad_z_j = a_j * (grad_a_j - a_j * sum_i grad_a_i * a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def peak_shave(a, n_peaks=5, n_steps=100, time_limit=30.0):
    """
    Peak shaving: iteratively reduce the highest peaks in the convolution
    by adjusting the sequence elements that contribute most to those peaks.
    """
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top n_peaks (excluding edges which are just squares)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = [k for k in peak_indices if b[k] > best_obj * 0.95]
        
        if not peak_indices:
            break
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find which pairs (i, k-i) contribute most to b[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            # Reduce the largest contribution
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i
            
            # Try reducing a[i] or a[j]
            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                # Reduce by a small factor
                factor = 0.95
                new_val = old_val * factor
                # Redistribute the difference to other elements proportionally
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                mask = np.ones(n, dtype=bool)
                mask[target] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[target] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-12:
                    a = a_trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit=60.0):
    """Fast coordinate descent cycling through the best multipliers."""
    a = a0.copy().astype(np.float64)
    s = a.sum()

    n = len(a)
    best_a = a.copy()

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit=60.0, lr=0.005, mom=0.9, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = b.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.01
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_a, best_obj
