For k near 0: conv[k] ≈ c^2 * (k+1) * f(0)^2 (approximately)
For k near 2n-2: similar
For k in the middle: conv[k] ≈ c^2 * n * ∫ f(x) f(x + (k-(n-1))/n) dx ... this is getting complicated.

The key point is: the center term involves f(x) * f(1-x), not f(x)^2. So if f is "anti-symmetric" (f(x) is large where f(1-x) is small), the center term can be much less than c^2 * n * ∫f^2.

For example, if f(x) = x^α (increasing), then f(x)*f(1-x) = x^α (1-x)^α, and ∫ x^α(1-x)^α dx = B(α+1, α+1) = Γ(α+1)^2/Γ(2α+2).

While (∫f)^2 = (1/(α+1))^2.

So the center ratio is: n * c^2 * B(α+1,α+1) / (c*n/(α+1))^2 = (α+1)^2 * B(α+1,α+1) = (α+1)^2 * Γ(α+1)^2/Γ(2α+2)

For α=0 (uniform): 1 * Γ(1)^2/Γ(2) = 1*1/1 = 1. Objective = 2*1 = 2. ✓
For α=1: 4 * Γ(2)^2/Γ(4) = 4*1/6 = 2/3. 
For α=2: 9 * Γ(3)^2/Γ(6) = 9*4/120 = 3/10 = 0.3

But we also need to check the edge terms! For f(x) = x^α:
conv[0] ≈ c^2 * f(0)^2 = c^2 * 0 = 0 (for α > 0)
conv[2n-2] ≈ c^2 * f(1)^2 = c^2 * 1 = 1/n^2

And intermediate terms: for k between 0 and n-1, the number of terms is k+1, and each term is approximately c^2 * f(i/n) * f((i+k)/n).

For the "power law" f(x) = x^α with large α, the sequence is concentrated near x=1 (the right edge). Then:

- conv[n-1] = sum a[i]*a[n-1-i] involves products of small*a[large index] which could be small
- conv[2n-2] = a[n-1]^2 could be large if the last element is large

So there's a trade-off. The optimal α balances these.

where grad_log[i] = grad[i] * a[i] (chain rule)

    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s
    n = len(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    temp = temp_init
    lr_current = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        grad = compute_gradient_softmax(a, conv, top_k=top_k, temp=temp)
        
        # Adam update in log-space
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Multiplicative update: a <- a / (a + lr * grad_normalized)
        # Or equivalently: log(a) <- log(a) - lr * grad
        # a <- a * exp(-lr * grad / (sqrt(v_hat) + eps))
        update = lr_current * m_hat / (np.sqrt(v_hat) + eps)
        # Clip the update to prevent too large changes
        update = np.clip(update, -0.5, 0.5)
        a_new = a * np.exp(-update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal
        temp *= 1.005
        lr_current *= 0.9998
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using successive linear programming:
At each step, solve min t s.t. C(a)@g <= t, sum(g)=1, g>=0 to find the
ideal direction, then line-search towards it. Combined with softmax-gradient
polishing for local refinement and multi-n exploration.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives conv(a, g) (full linear)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_minmax(C):
    """Solve min t s.t. C@g <= t, sum(g)=1, g>=0. Returns (g_opt, t_opt) or (None, inf)."""
    n = C.shape[1]
    # Variables: g[0..n-1], t  (total n+1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    # Inequality: C@g - t <= 0  for each of (2n-1) rows
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success:
            g_opt = np.maximum(result.x[:n], 0)
            gs = g_opt.sum()
            if gs > 1e-15:
                g_opt = g_opt / gs
            t_opt = result.x[n]
            return g_opt, t_opt
    except Exception:
        pass
    return None, np.inf


def lp_maxsum(C, M):
    """Solve max sum(g) s.t. C@g <= M, g>=0. Returns (g_opt, g_sum, success)."""
    n = C.shape[1]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            g_opt = np.maximum(result.x, 0)
            g_sum = g_opt.sum()
            if g_sum > 1e-10:
                return g_opt, g_sum, True
    except Exception:

    return None, 0.0, False

    n = len(a)

    scaled = peak_vals / max(temp, 1e-10)

    if w_sum < 1e-15:

        w[np.argmax(peak_vals)] = 1.0

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    stall_threshold = 40
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        # Build conv matrix from current a
        C = build_conv_matrix(a)

        # Solve min-max LP
        g_opt, t_opt = lp_minmax(C)

        if g_opt is not None and t_opt < best_obj / (2.0 * n) + 1e-10:
            # Found a direction that should reduce the peak
            a_try, obj_try = line_search_lp(a, g_opt, n)
            if obj_try < best_obj - 1e-12:
                a = a_try
                conv = fast_convolve(a)
                best_obj = min(best_obj, obj_try)
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            print(f"    LP-cycle iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        # Handle stagnation with perturbation
        if no_improve >= stall_threshold:
            rng = np.random.default_rng(iteration * 31 + perturb_count * 7 + 42)
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            no_improve = 0
            perturb_count += 1
            stall_threshold = min(100, 40 + 10 * perturb_count)

            if perturb_count > 50:
                # Try gradient-based escape
                a = _gradient_escape(a, n, time_limit=3.0, rng=rng)
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                perturb_count = 0
                stall_threshold = 40

    return best_a, best_obj


def _gradient_escape(a, n, time_limit=3.0, rng=None):
    """Short gradient-based escape from local minimum."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = a.copy()
    t0 = time.monotonic()
    lr = 0.01
    temp = 0.005
    v = np.zeros(n)
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_soft_grad(a, conv, top_k=30, temp=temp)
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
        lr *= 0.99
        temp *= 1.01
    return a


def gradient_polish(a0, time_limit, top_k=50, temp_init=0.003, lr_init=0.008):
    """Adam-based gradient polishing in log-space (multiplicative updates)."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init

    while time.monotonic() - t0 < time_limit:
        t_count += 1
        conv = fast_convolve(a)
        # Use exact gradient at max + soft for nearby peaks
        grad = compute_soft_grad(a, conv, top_k=top_k, temp=temp)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        # Multiplicative update (log-space)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -0.3, 0.3)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-10)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            break

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        lr *= 0.9995
        temp *= 1.004

        if no_improve > 300:
            break

    return best_a, best_obj


def coordinate_polish(a0, time_limit, rng=None):
    """Coordinate descent: adjust pairs of elements to reduce peak."""
    if rng is None:
        rng = np.random.default_rng(99)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    deltas = [0.001, 0.003, 0.005, 0.01, -0.001, -0.003, -0.005, -0.01]

    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            improved = False
            for d in deltas:
                if a[i] + d < 0 or a[j] - d < 0:
                    continue
                a[i] += d
                a[j] -= d
                # Quick check with partial conv update
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
                else:
                    a[i] -= d
                    a[j] += d
            if not improved:
                no_improve += 1

        if no_improve > 500:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    # Power-law shapes (different exponents)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        pw = (x + 1.0) ** alpha
        inits[f'pow{alpha}'] = pw / pw.sum()
        # Mirrored version
        pw2 = (n - x) ** alpha
        inits[f'powR{alpha}'] = pw2 / pw2.sum()
        # Symmetric
        pw3 = (x + 1.0) ** alpha + (n - x) ** alpha
        inits[f'powS{alpha}'] = pw3 / pw3.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Sinc-like
    for freq in [1, 2, 3]:
        val = np.sin(np.pi * freq * x / n) ** 2 + 0.05
        inits[f'sinc{freq}'] = val / val.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Epanechnikov kernel shape
    val = np.maximum(0, 1 - (2 * x / n - 1) ** 2)
    val[val < 0.01] = 0.01
    inits['epan'] = val / val.sum()

    return inits


def extend_to_length(a, n_target, rng=None):
    """Extend/interpolate sequence to target length."""
    n_src = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_src), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Uses successive LP (min-max formulation) as primary optimizer,
    with gradient polishing and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick multi-n scan (first ~150s)
    # =========================================================
    print("=== Phase 1: Multi-n Scan ===")
    t_start = time.monotonic()
    phase1_time = 150.0

    scan_n_values = [512, 640, 768, 896, 1024, 1280]
    scan_results = []

    for n in scan_n_values:
        if time.monotonic() > deadline - 1400:
            break
        phase1_remaining = min(phase1_time, deadline - time.monotonic() - 1300)
        if phase1_remaining < 10:
            break

        inits = generate_initializations(n, parent, rng)
        # Prioritize parent and structured inits
        priority = ['parent', 'uniform', 'pow1.0', 'powS1.0', 'pow0.5', 'bimodal', 'pow2.0', 'sinc1']
        init_names = [nm for nm in priority if nm in inits][:5]

        t_per = max(3.0, phase1_remaining / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() > deadline - 1350:
                break
            t_alloc = min(t_per, 15.0)
            if t_alloc < 2:
                break

            a0 = inits[name]

            # Quick LP optimization
            a_opt, obj = lp_optimize_cycle(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan n={n}, {name}: {score:.6f} *NEW BEST*")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep LP optimization on top candidates
    # =========================================================
    print("=== Phase 2: Deep LP Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (diverse by n)
    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results[:10]:
        if n not in seen_n:
            top_configs.append((n, a_opt, score))
            seen_n.add(n)
        if len(top_configs) >= 4:
            break

    # Ensure parent's n is included
    if parent is not None and len(parent) not in seen_n:
        top_configs.append((len(parent), parent.copy(), best_score))

    remaining = deadline - time.monotonic() - 300.0
    n_cand = max(1, len(top_configs))
    t_per_deep = max(30.0, remaining * 0.65 / n_cand)

    for i, (n, a0, init_score) in enumerate(top_configs):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 200)
        if t_alloc < 20:
            break

        print(f"  Deep LP {i}: n={n}, init_score={init_score:.6f}, t_alloc={t_alloc:.0f}s")

        # Run LP optimization
        a_opt, obj = lp_optimize_cycle(a0, time_limit=t_alloc, verbose=True)

        # Quick gradient polish
        t_polish = min(10.0, (deadline - time.monotonic() - 200) * 0.1)
        if t_polish > 3:
            a_opt2, obj2 = gradient_polish(a_opt, time_limit=t_polish, top_k=40, temp_init=0.002, lr_init=0.006)
            if obj2 < obj:
                a_opt = a_opt2
                obj = obj2

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  Deep {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Extended LP on the single best with variations
    # =========================================================
    print("=== Phase 3: Extended LP on Best ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        n_best = len(best_sequence)

        # Multiple LP runs with different perturbation strengths
        perturb_scales = [0.0, 0.002, 0.005, 0.01, 0.003, 0.007]
        t_per_run = max(15.0, remaining * 0.5 / len(perturb_scales))

        for pi, scale in enumerate(perturb_scales):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per_run, deadline - time.monotonic() - 120)
            if t_alloc < 10:
                break

            if scale == 0:
                a0 = best_sequence.copy()
            else:
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()

            a_opt, obj = lp_optimize_cycle(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Ext LP {pi} (scale={scale}): {score:.6f} *NEW BEST*")
                else:
                    print(f"  Ext LP {pi} (scale={scale}): {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient + coordinate polishing
    # =========================================================
    print("=== Phase 4: Polishing ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        n_best = len(best_sequence)

        # Gradient polish (multiple configs)
        grad_configs = [
            {'top_k': 30, 'temp_init': 0.002, 'lr_init': 0.01},
            {'top_k': 50, 'temp_init': 0.005, 'lr_init': 0.005},
            {'top_k': 80, 'temp_init': 0.01, 'lr_init': 0.003},
            {'top_k': 20, 'temp_init': 0.001, 'lr_init': 0.015},
        ]

        t_per_grad = min(20.0, remaining * 0.3 / len(grad_configs))
        for gi, cfg in enumerate(grad_configs):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(t_per_grad, deadline - time.monotonic() - 60)
            if t_alloc < 5:
                break

            a_opt, obj = gradient_polish(best_sequence, time_limit=t_alloc, **cfg)
            if obj < 2.0 * n_best * fast_convolve(best_sequence).max():
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad {gi}: {score:.6f} *NEW BEST*")

        # Coordinate polish
        remaining = deadline - time.monotonic() - 20.0
        t_coord = min(remaining * 0.4, 40.0)
        if t_coord > 5:
            a_opt, obj = coordinate_polish(best_sequence, time_limit=t_coord, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(30.0, deadline - time.monotonic() - 5)
        if t_final > 10:
            a_opt, obj = lp_optimize_cycle(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")

        # One last gradient pass
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, obj = gradient_polish(best_sequence, time_limit=remaining,
                                         top_k=30, temp_init=0.001, lr_init=0.003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final Grad: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
