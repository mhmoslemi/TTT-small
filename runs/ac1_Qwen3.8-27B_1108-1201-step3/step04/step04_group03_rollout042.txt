import time
from scipy.optimize import minimize

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def compute_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_and_grad(x, T=0.0):
    """Objective and its gradient for L-BFGS-B."""
    a = softmax(x)
    n = len(a)
    C = compute_conv(a)
    if T > 0:
        # Log-sum-exp smoothing
        val = T * np.log(np.sum(np.exp(C / T)))
        # Gradient of log-sum-exp
        weights = np.exp(C / T) / np.sum(np.exp(C / T))
    else:
        k_star = np.argmax(C)
        val = C[k_star]
        weights = np.zeros(len(C))
        # For subgradient: use all argmax indices
        max_val = C[k_star]
        argmax_idx = np.where(C == max_val)[0]
        weights[argmax_idx] = 1.0 / len(argmax_idx)
    
    objective = 2.0 * n * val  # Since sum(a) = 1, the denominator is 1
    
    # Gradient
    # d(obj)/d(x_j) = 2n * sum_k weights[k] * d(C_k)/d(s_j) * ds_j/dx_j
    # d(C_k)/d(s_j) = 2*s_{k-j} (or s_j if k=2j)
    # ds_j/dx_j = s_j * (1 - s_j)
    
    grad_s = np.zeros(n)
    for k in range(len(C)):
        if weights[k] < 1e-15:
            continue
        wk = weights[k]
        for j in range(n):
            if 0 <= k - j < n:
                if k - j == j:  # k = 2j
                    grad_s[j] += wk * s[j]
                else:
                    grad_s[j] += wk * 2.0 * s[k - j]
    
    grad = 2.0 * n * grad_s * a * (1.0 - a)
    
    # Since we're on the simplex (via softmax), we need to project the gradient
    # Actually, L-BFGS-B on x directly should work since softmax handles the constraint.
    
    return objective, grad

def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def softmax(x):

    e = np.exp(x)
    return e / e.sum()

def logsumexp(x, T):

    x_max = x_scaled.max()
    return T * (x_max + np.log(np.sum(np.exp(x_scaled - x_max))))

    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        a = a_new
        Pa = hankel_matvec(p, a)
        grad = 2.0 * Pa
        
        # Adaptive step size
        # eta = 1 / (2 * ||P||_2) approximately
        pass

            S_total = a.sum()  # Current sum = S + aj_old
            
            # Grid search
            t_candidates = np.concatenate([
                np.linspace(0, aj_old * 3, 50),
                np.linspace(aj_old * 0.3, aj_old * 1.5, 50)
            ])
            t_candidates = t_candidates[t_candidates >= 0]
            
            for t in t_candidates:
                tj = t
                a[j] = tj
                S_new = S + tj
                if S_new < 1e-10:
                    continue
                C_new = C_base.copy()
                for k in range(len(C_new)):
                    if k == 2 * j:
                        C_new[k] += tj * tj
                    else:
                        C_new[k] += tj * e[k]
                obj_val = 2.0 * n * C_new.max() / (S_new ** 2)
                if obj_val < best_val:
                    best_val = obj_val
                    best_t = tj
            
            a[j] = best_t
            if abs(best_t - aj_old) > 1e-12:
                improved = True
            
            # Update best
            C = fast_conv(a)
            obj = 2.0 * n * C.max() / (a.sum()**2)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj
