def spectral_flatten(a, n_iter=1000, beta=0.5):
    """
    Iterative spectral flattening.
    a: input sequence (normalized to sum 1)
    beta: flattening strength (0 = no change, 1 = full flattening)
    """
    n = len(a)
    # Zero-pad to at least 2n-1
    fft_size = 1 << (2*n - 1 - 1).bit_length()
    
    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2
        
        # Flattening: new_power = power^(1-beta) * (mean(power))^beta
        # Or: new_power = power / (power + eps) or similar
        # Simplest: new_amplitude = 1/sqrt(power) (full flattening)
        # Or partial: new_amplitude = power^(-beta/2)
        
        # Use a soft inverse
        pmax = power.max()
        pmin = power.min() + 1e-15
        
        # Geometric mean as target
        pmean = np.exp(np.log(power + 1e-30).mean())
        
        # New amplitude: interpolate between current and flat
        new_power = (power ** (1-beta)) * (pmean ** beta)
        new_amplitude = np.sqrt(new_power)
        
        new_A = new_amplitude * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:

        a_new = a_new / s
        a = a_new
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative spectral flattening
(Gerchberg-Saxton style) combined with LP-based Frank-Wolfe refinement.
The spectral flattening operates in the frequency domain to reduce the peak
of the auto-convolution, while the LP oracle provides targeted direction
updates for fine-tuning.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, n_iter=5000, beta=0.5, relax=0.5, time_limit=None, 
                     start_time=None, verbose=False):
    """
    Iterative spectral flattening (Gerchberg-Saxton style).
    Flattens the power spectrum to reduce peak auto-convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    if start_time is None:
        start_time = time.monotonic()
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        
        # Forward FFT
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Flattening: reduce dynamic range
        # new_power = power * (mean_power / power)^beta = power^(1-beta) * mean_power^beta
        p_mean = power.mean()
        # Avoid division by zero
        ratio = np.maximum(power, 1e-30) / p_mean
        new_power = power * (1.0 / ratio) ** beta
        # Equivalently: new_power = power^(1-beta) * p_mean^beta
        
        new_amplitude = np.sqrt(new_power)
        phase = np.angle(A)
        new_A = new_amplitude * np.exp(1j * phase)
        
        # Inverse FFT
        a_raw = np.fft.irfft(new_A, fft_size)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_raw, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        # Relaxation
        a = (1.0 - relax) * a + relax * a_new
        
        # Check improvement
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 1000 == 0:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def build_conv_matrix_rows(f, lags):
    """Build rows of the convolution matrix for specific lags."""
    n = len(f)
    rows = np.zeros((len(lags), n))
    for i, k in enumerate(lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            rows[i, idx] = f[k - idx]
    return rows


def solve_linearized_lp(f, conv_f, top_k=None):
    """Solve linearized LP for Frank-Wolfe direction."""
    n = len(f)
    if top_k is None or top_k >= len(conv_f):
        lags = np.arange(len(conv_f))
    else:
        lags = np.argpartition(conv_f, -top_k)[-top_k:]
    
    B = build_conv_matrix_rows(f, lags)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((len(lags), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f[lags]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def fw_refine(f0, time_limit, top_k=30, verbose=False):
    """Frank-Wolfe refinement with linearized LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        g = solve_linearized_lp(f, conv, top_k=top_k)
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1.0 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.001 * no_improve))
            no_improve = 0
    
    return best_f, best_obj


def subgrad_refine(f0, time_limit, lr=0.002, mom=0.9, top_k=40, temp=0.0001):
    """Subgradient descent refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.range(js, je + 1) if hasattr(np, 'range') else np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9995
        temp *= 1.003
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def local_search(f0, time_limit, rng, num_batch=100):
    """Random perturbation local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    scale = 0.005
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(num_batch):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()
        
        scale *= 0.97
        if scale < 1e-5:
            scale = 0.003
    
    return best_f, best_obj


def gen_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    inits.append(rng.dirichlet(np.ones(n) * 3.0))
    
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.2, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())
    
    # Sine window
    sw = np.sin(np.pi * (x + 1) / (n + 1)) ** 2 + 0.01
    inits.append(sw / sw.sum())
    
    # If parent available, interpolate
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
