import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_symmetric_simplex(v):
    """Projection onto symmetric simplex: a[i] = a[n-1-i], sum(a)=1, a>=0."""
    n = len(v)
    half = (n + 1) // 2
    # Average symmetric pairs
    sym = np.zeros(n)
    for i in range(half):
        j = n - 1 - i
        sym[i] = (v[i] + v[j]) / 2
        sym[j] = sym[i]
    # Now project onto simplex with the symmetry constraint
    # The effective variables are sym[0], ..., sym[half-1] (with the middle one only once)
    # sum(sym) = sum_{i=0}^{half-1} sym[i] * (2 if i != n-1-i else 1)
    # Let's work with the "effective" variables
    eff = np.zeros(half)
    weights = np.zeros(half)
    for i in range(half):
        j = n - 1 - i
        if i == j:
            eff[i] = sym[i]
            weights[i] = 1
        else:
            eff[i] = sym[i]
            weights[i] = 2
    
    # We need sum(eff * weights) = 1, eff >= 0
    # This is a weighted simplex projection
    # Use binary search on the threshold
    lo = -1e15
    hi = max(eff) + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.maximum(eff - mid, 0)
        s = (proj * weights).sum()
        if s > 1:
            lo = mid
        else:
            hi = mid
    
    mid = (lo + hi) / 2
    proj = np.maximum(eff - mid, 0)
    # Rescale to get exact sum = 1
    s = (proj * weights).sum()
    if s > 0:
        proj = proj * (1.0 / s)
    
    result = np.zeros(n)
    for i in range(half):
        j = n - 1 - i
        result[i] = proj[i]
        result[j] = proj[i]
    return result


def compute_conv_grad(f):
    """Compute subgradient of max(conv(f,f)) with respect to f.
    Returns the vector of subgradients at the maximizing position.
    grad[i] = 2 * sum_{k where conv_k is max} f[k-i] for valid k.
    """
    n = len(f)
    conv = fast_convolve(f)
    max_val = conv.max()
    # Find all positions where conv is at max (within tolerance)
    mask = conv >= max_val - 1e-10
    # grad[i] = 2 * sum_{k in mask} f[k-i] for valid k
    # This is 2 * cross_correlation(mask, f)
    mask_arr = mask.astype(np.float64)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    M = np.fft.rfft(mask_arr, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_result = np.fft.irfft(M * F, fft_size)
    return 2.0 * conv_result[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None, time_limit=30):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def power_sum_optimize(f0, time_limit, p_start=4.0, p_end=200.0, lr_start=2.0, momentum=0.9):

    best_f = f.copy()

    lr = lr_start / n

    t0 = time.monotonic()

    it = 0
    best_lr = lr
    p_multiplier = 1.005

        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        b = conv_y
        b_p = np.sum(b ** p)
        if b_p < 1e-30:

        obj_val = b_p ** (1.0 / p)

        b_scaled = b / b.max()
        w = b_scaled ** (p - 1)
        w_sum = w.sum()
        if w_sum < 1e-15:

            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        w_arr = w.astype(np.float64)

        conv_grad = np.fft.irfft(W * Y, fft_size)

        # f_new = f + v_new
        # y = f_new + (momentum * (1 + momentum)) * v_new  (Nesterov lookahead)
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Nesterov lookahead for next iteration
        y = f_new + momentum * (f_new - f)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 4.0 / n)

            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(lr_start / n, 0.001)

        if it % 1000 == 0:

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    else:

    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_start / n
    v = np.zeros(n)
    y = f.copy()

    no_improve = 0
    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit:

        conv_y = fast_convolve(y)

def target_peak_optimize(f0, time_limit, top_k=20):
    """Optimize by targeting the top-k peak convolution values."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    momentum_v = 0.9
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Weight by peak values (exponential weighting)
        pv = conv[peaks]
        scaled = (pv - pv.min()) / (pv.max() - pv.min() + 1e-15)
        w = np.exp(10 * scaled)
        w = w / w.sum()
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = momentum_v * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.005
                no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. FW-LP with golden-section line search for global structure
2. LSE (log-sum-exp) smoothed gradient descent with Nesterov momentum
3. Power-sum smoothed optimization (alternative smoothing to LSE)
4. Target-peak subgradient optimization
5. Symmetric-constrained FW-LP
6. Adaptive alternation of all methods in remaining time
7. Exploration of alternative sequence lengths
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            ks = k - js
            rows.extend(list(ks // 1 * 0 + [k] * len(js)))
            cols.extend(js.tolist())
            vals.extend(f[ks].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_grad_from_weights(w, f):
    """Compute 2 * cross_correlation(w, f) of length n using FFT.
    w has length 2n-1, f has length n.
    Result[i] = 2 * sum_{m} w[i+m] * f[m]
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def fw_lp_optimize(f0, time_limit, lp_timeout=15):
    """Frank-Wolfe with LP oracle and golden-section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    A_cache = build_lp_matrix(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_direction(f, M * m_exp, A_cache, time_limit=lp_timeout)
        if g is None:
            m_exp = min(m_exp * 1.12, 5.0)
            if m_exp >= 4.0:
                rng = np.random.default_rng(it * 7 + 3)
                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 35:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search: coarse grid + golden section
        best_alpha = 0.0
        best_val = M
        alphas = np.array([0.03, 0.06, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0])
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Find bracket
            lo_a = 0.0
            hi_a = 1.0
            for i, a in enumerate(alphas):
                if abs(a - best_alpha) < 1e-9:
                    lo_a = alphas[i - 1] if i > 0 else 0.0
                    hi_a = alphas[i + 1] if i < len(alphas) - 1 else 1.0
                    break

            # Golden section refinement
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi_a - gr * (hi_a - lo_a)
            d = lo_a + gr * (hi_a - lo_a)
            ft_c = np.maximum((1.0 - c) * f + c * g, 0.0)
            ft_c = ft_c / ft_c.sum()
            val_c = fast_convolve(ft_c).max()
            ft_d = np.maximum((1.0 - d) * f + d * g, 0.0)
            ft_d = ft_d / ft_d.sum()
            val_d = fast_convolve(ft_d).max()

            for _ in range(8):
                if val_c < val_d:
                    hi_a = d
                    d = c
                    val_d = val_c
                    c = hi_a - gr * (hi_a - lo_a)
                    ft_c = np.maximum((1.0 - c) * f + c * g, 0.0)
                    ft_c = ft_c / ft_c.sum()
                    val_c = fast_convolve(ft_c).max()
                else:
                    lo_a = c
                    c = d
                    val_c = val_d
                    d = lo_a + gr * (hi_a - lo_a)
                    ft_d = np.maximum((1.0 - d) * f + d * g, 0.0)
                    ft_d = ft_d / ft_d.sum()
                    val_d = fast_convolve(ft_d).max()

                best_mid = c if val_c < val_d else d
                best_mid_val = min(val_c, val_d)
                if best_mid_val < best_val:
                    best_val = best_mid_val
                    best_alpha = best_mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 17 + 42)
            scale = 0.001 + 0.004 * min(1.0, (no_improve - 50) / 60.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            A_cache = build_lp_matrix(f)

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=15.0, beta_end=5000.0, lr_start=2.0, momentum=0.92):
    """LSE smoothed gradient descent with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_start / n
    v = np.zeros(n)
    y = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_grad_from_weights(w, y)

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        y = f_new + momentum * (f_new - f)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if it % 20 == 0:
                lr = min(lr * 1.0015, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.94
            if no_improve > 90:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(lr_start / n, 0.0005)
                beta = min(beta * 1.4, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def power_sum_optimize(f0, time_limit, p_start=8.0, p_end=300.0, lr_start=2.0, momentum=0.90):
    """Power-sum smoothed optimization: minimize (sum b_k^p)^(1/p)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = p_start
    lr = lr_start / n
    v = np.zeros(n)
    y = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_convolve(y)
        b_max = conv_y.max()
        if b_max < 1e-30:
            break

        # Weight: w_k proportional to b_k^(p-1)
        b_norm = conv_y / b_max
        log_w = (p - 1.0) * np.log(b_norm + 1e-30)
        log_w_max = log_w.max()
        w = np.exp(log_w - log_w_max)
        w = w / w.sum()

        grad = compute_grad_from_weights(w, y)

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        y = f_new + momentum * (f_new - f)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if it % 20 == 0:
                lr = min(lr * 1.001, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.94
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(lr_start / n, 0.0005)
                p = min(p * 1.3, p_end)
                no_improve = 0

        if p < p_end:
            p *= 1.005

        if it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS it={it}, obj={best_obj:.12f}, p={p:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def target_peak_optimize(f0, time_limit, top_k=30):
    """Subgradient method targeting the top-k peak convolution values."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.003
    mom = 0.92
    v = np.zeros(n)
    temp = 0.00005

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.95
            if no_improve > 250:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.003
                temp *= 1.5
                no_improve = 0

        temp *= 1.003

    return best_f, best_obj


def symmetric_optimize(f0, time_limit, lp_timeout=15):
    """FW-LP with symmetry constraint a[i] = a[n-1-i]."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f = (f + f[::-1]) / 2.0
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        M = conv.max()
        A_cache = build_lp_matrix(f)
        g = solve_lp_direction(f, M, A_cache, time_limit=lp_timeout)

        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        g = (g + g[::-1]) / 2.0
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = (ft + ft[::-1]) / 2.0
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = (f + f[::-1]) / 2.0
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 13 + 5)
            noise = rng.standard_normal(n)
            noise = (noise + noise[::-1]) / 2.0
            f = best_f + noise * 0.002
            f = (f + f[::-1]) / 2.0
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FW-LP, LSE gradient descent, power-sum smoothing, target-peak,
    and symmetric optimization in an adaptive alternating fashion.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception:
        print("No parent available, using uniform initialization")

    if parent is None:
        parent = np.ones(448) / 448.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: FW-LP warm start ===
    print("\n=== Phase 1: FW-LP warm start ===")
    t_rem = deadline - time.monotonic()
    t_alloc = min(120, t_rem * 0.08)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc, lp_timeout=12)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE polishing ===
    print("\n=== Phase 2: LSE Polishing ===")
    current_best = best_tracker[1].copy()
    t_rem = deadline - time.monotonic()

        a_opt, _ = lse_optimize(current_best, time_limit=t_alloc,

        try_sequence(a_opt, best_tracker)

    # === Phase 3: Power-sum optimization ===
    print("\n=== Phase 3: Power-sum Optimization ===")
    current_best = best_tracker[1].copy()
    t_rem = deadline - time.monotonic()
    t_alloc = min(200, t_rem * 0.15)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = power_sum_optimize(current_best, time_limit=t_alloc,
                                      p_start=8.0, p_end=300.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: FW-LP refinement ===
    print("\n=== Phase 4: FW-LP Refinement ===")
    current_best = best_tracker[1].copy()
    t_rem = deadline - time.monotonic()
    t_alloc = min(200, t_rem * 0.15)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")

        try_sequence(a_opt, best_tracker)

    # === Phase 5: Target-peak optimization ===
    print("\n=== Phase 5: Target-Peak ===")
    current_best = best_tracker[1].copy()
    n_cur = len(current_best)
    t_rem = deadline - time.monotonic()
    t_alloc = min(120, t_rem * 0.10)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = target_peak_optimize(current_best, time_limit=t_alloc,

    # === Phase 6: Symmetric optimization ===
    print("\n=== Phase 6: Symmetric FW-LP ===")
    current_best = best_tracker[1].copy()
    t_rem = deadline - time.monotonic()
    t_alloc = min(80, t_rem * 0.08)

    # === Phase 7: Adaptive alternation ===
    print("\n=== Phase 7: Adaptive Alternation ===")
    current_best = best_tracker[1].copy()
    round_id = 0
    stagnant_rounds = 0

    while time.monotonic() < deadline - 30:
        round_id += 1
        t_rem = deadline - time.monotonic() - 30
        if t_rem < 30:

                try_sequence(a_opt, best_tracker)

        if best_tracker[0] < score_before - 1e-10:
            current_best = best_tracker[1].copy()
            stagnant_rounds = 0
        else:
            stagnant_rounds += 1
            if stagnant_rounds > 6:
                # Perturb and try a different method
                rng = np.random.default_rng(round_id * 101 + 7)
                current_best = best_tracker[1].copy()
                noise = rng.standard_normal(len(current_best)) * 0.003
                current_best = project_simplex(current_best + noise)
                stagnant_rounds = 0

    # === Phase 8: Try alternative lengths ===
    print("\n=== Phase 8: Alternative Lengths ===")
    t_rem = deadline - time.monotonic() - 30
    if t_rem > 100:
        current_best = best_tracker[1].copy()
        alt_lengths = [384, 512, 576]
        if n_parent in alt_lengths:
            alt_lengths.remove(n_parent)

        for n_alt in alt_lengths:
            t_rem = deadline - time.monotonic() - 30
            if t_rem < 50:
                break
            t_alloc = min(60, t_rem // max(1, len(alt_lengths)))

            # Interpolate to new length
            x_old = np.linspace(0, 1, len(current_best))
            x_new = np.linspace(0, 1, n_alt)
            f_interp = np.interp(x_new, x_old, current_best)
            f_interp = np.maximum(f_interp, 0.0)
            if f_interp.sum() > 1e-10:
                f_interp = f_interp / f_interp.sum()

            t_fw = t_alloc * 0.5
            t_lse = t_alloc * 0.5
            a_opt, _ = fw_lp_optimize(f_interp, time_limit=t_fw, lp_timeout=8)
            try_sequence(a_opt, best_tracker)

            cur = best_tracker[1] if best_tracker[1] is not None and len(best_tracker[1]) == n_alt else a_opt
            a_opt, _ = lse_optimize(cur, time_limit=t_lse, beta_start=20.0, beta_end=3000.0)
            try_sequence(a_opt, best_tracker)

    # === Final ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(448) / 448.0
    print(f"\nFALLBACK: n=448, score={evaluate_sequence(fallback.tolist()):.8f}")
    return [float(x) for x in fallback]
