```python
def project_feasible(h):
    h_clamped = np.clip(h, 0, 1)
    total = h_clamped.sum()
    if np.isclose(total, len(h)/2, atol=1e-6):

    # Redistribute excess/deficit
    diff = len(h)/2 - total
    h_clamped += diff * (h > 0) / (h > 0).sum()
    return h_clamped
```
