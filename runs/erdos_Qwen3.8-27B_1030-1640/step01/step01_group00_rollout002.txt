def find_active_direction(h, G, c, top_k, bound_scale):
    """Find a direction d that decreases the top_k components of c.
    Returns d or None if no direction found."""
    n = h.size
    # Sort components descending
    order = np.argsort(c)[::-1]
    active = order[:top_k]
    A = G[active, :]  # (k, n)
    
    # We want: A @ d <= -eps (all active decrease)
    #           sum(d) = 0
    #           -bound_i <= d_i <= bound_i
    
    # Use LP: minimize -t
    # subject to: A @ d - t <= 0 (i.e., A@d <= t, minimize t, we want t<0)
    #             sum(d) = 0
    #             -bound <= d <= bound
    
    # Actually simpler: minimize max(A@d) subject to sum(d)=0, bounds on d
    # This is a LP with variables d (n vars) and t (1 var), total n+1 vars
    
    bound = np.zeros(n)
    for i in range(n):
        bound[i] = bound_scale * max(0.01, min(h[i], 1-h[i]))
    
    # Use scipy linprog
    # Variables: [d, t]
    # Minimize: t (we want t < 0)
    # Constraints:
    #   A @ d - t <= 0  (A@d <= t)
    #   sum(d) = 0
    #   -bound <= d <= bound
    #   t is free (or bounded by some reasonable value)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A @ d - t <= 0
    A_ub = np.hstack([A, -np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-bound[i], bound[i]) for i in range(n)] + [(-1.0, 1.0)]
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                               A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                               method='highs')
        if res.success and res.x[-1] < -1e-6:
            d = res.x[:n].copy()
            # Ensure sum is exactly 0
            d -= np.mean(d)
            return d
    except Exception:
        pass
    
    # Fallback: use negative mean active gradient projected to sum-zero
    g_mean = np.mean(A, axis=0)  # (n,)
    g_mean -= np.mean(g_mean)  # sum zero
    # Scale to fit bounds
    if np.max(np.abs(g_mean)) > 0:
        scale = bound_scale / np.max(np.abs(g_mean))
        d = g_mean * min(1.0, scale)
    else:
        return None
    # Check if it's a descent direction
    worst = np.max(A @ d)
    if worst < -1e-8:
        return d
    return None

    alphas = [a for a in alphas if a > 1e-10 and a <= alpha_max]
    best_alpha = 0.0
    best_F = F_current
    best_h = h
    for a in alphas:
        h_new = project_box_sum(h + a * d, n)
        F_new = exact_value(h_new)
        if F_new < best_F - 1e-12:
            best_F = F_new
            best_alpha = a
            best_h = h_new
    if best_alpha == 0.0:
        return h, F_current
    return best_h, best_F

def active_descent(h, deadline, need=25.0, max_rounds=300):
    h = project_box_sum(h, h.size)
    n = h.size
    F = exact_value(h)
    best_h = h.copy()
    best_F = F
    k_schedule = [1, 2, 3, 4, 5, 7, 10, 14]
    stall = 0
    last_improve_iter = 0
    
    for it in range(max_rounds):
        if deadline is not None and (deadline - time.monotonic()) < need:
            break
        
        c = corr_components(h)
        G = grad_matrix(h)
        improved = False
        
        for k in k_schedule:
            if deadline is not None and (deadline - time.monotonic()) < need:
                break
            if (it + k) % 3 != 0 and k > 5:
                continue
            
            d, t_lp = find_direction_lp(G, c, k, h, bound_scale=0.04)
            if d is None:
                d = gradient_direction(G, c, k, h, bound_scale=0.04)
            
            if np.max(np.abs(d)) < 1e-10:
                continue
            
            h_new, F_new = line_search(h, d, F)
            if F_new < F - 1e-11:
                h = h_new
                F = F_new
                if F < best_F:
                    best_F = F
                    best_h = h.copy()
                improved = True
                c = corr_components(h)
                G = grad_matrix(h)
                if it - last_improve_iter == 0:
                    last_improve_iter = it
                break
        
        if not improved:
            stall += 1
            if stall >= 4:
                d = gradient_direction(G, c, min(3, n), h, bound_scale=0.02)
                if np.max(np.abs(d)) > 1e-10:
                    h_new, F_new = line_search_fine(h, d, F, 0.3)
                    if F_new < F - 1e-11:
                        h = h_new
                        F = F_new
                        if F < best_F:
                            best_F = F
                            best_h = h.copy()
                        improved = True
                if not improved and stall >= 8:
                    rnd = np.random.RandomState(it * 7 + 13)
                    h_pert = project_box_sum(h + 0.015 * rnd.standard_normal(n), n)
                    F_pert = exact_value(h_pert)
                    if F_pert < F + 0.005:
                        h = h_pert
                        F = F_pert
                    stall = 0
                elif stall >= 6:
                    stall = 0
    
    return best_h, best_F

def resample(h, n_new):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0 or n_new <= 0:
        return None
    if n_new == n:
        return project_box_sum(h, n)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def binary_search_pairs(h, F, max_flips=2000, deadline=None, need=20.0):
    h = project_box_sum(h, h.size)
    n = h.size
    best_h = h.copy()
    best_F = F
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_F
    
    c = corr_components(best_h)
    n_c = c.size
    for _ in range(max_flips):
        if deadline is not None and (deadline - time.monotonic()) < need:
            break
        best_delta = 1e-10
        best_flip = None
        for i in ones[:60]:
            for j in zeros[:60]:
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                F2 = exact_value(h2)
                delta = F2 - F
                if delta < best_delta:
                    best_delta = delta
                    best_flip = (i, j)
        if best_flip is not None and best_delta < -1e-10:
            i, j = best_flip
            h = h.copy()
            h[i] = 0.0
            h[j] = 1.0
            h = project_box_sum(h, n)
            F = exact_value(h)
            if F < best_F:
                best_F = F
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            break
    return best_h, best_F

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(1000 * (int(seed) if seed is not None else 0) + 17)
    
    best = {"h": None, "val": math.inf, "n": 0}
    
    def consider(h_cand):
        if h_cand is None:
            return
        h_cand = project_box_sum(h_cand, int(h_cand.size))
        if h_cand.size <= 0:
            return
        v = exact_value(h_cand)
        if not math.isfinite(v):
            return
        if v < best["val"] - 1e-12:
            best["h"] = h_cand.copy()
            best["val"] = float(v)
            best["n"] = int(h_cand.size)
        elif abs(v - best["val"]) <= 1e-12 and h_cand.size < best["n"]:
            best["h"] = h_cand.copy()
            best["val"] = float(v)
            best["n"] = int(h_cand.size)
    
    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).ravel()
        if hinit.size > 0 and np.all(np.isfinite(hinit)):
            hinit = project_box_sum(hinit, hinit.size)
            consider(hinit)
    
    if best["h"] is None:
        consider(np.full(45, 0.5))
    if best["h"] is None:
        h_fb = project_box_sum(np.full(45, 0.5), 45)
        return h_fb, exact_value(h_fb), 45
    
    # Phase 1: Deep active-set descent on current best
    if deadline - time.monotonic() > 60:
        h1, v1 = active_descent(best["h"].copy(), deadline, need=40, max_rounds=400)
        consider(h1)
    
    # Phase 2: Try nearby resolutions
    if best["h"] is not None and deadline - time.monotonic() > 50:
        base_n = best["n"]
        for dn in [-2, -1, 1, 2, -3, 3, -4, 4]:
            if deadline - time.monotonic() < 35:
                break
            nn = base_n + dn
            if nn < 20 or nn > 120:
                continue
            hr = resample(best["h"], nn)
            if hr is None:
                continue
            consider(hr)
            if exact_value(hr) < best["val"] + 0.02:
                if deadline - time.monotonic() > 40:
                    h2, v2 = active_descent(hr.copy(), deadline, need=30, max_rounds=200)
                    consider(h2)
    
    # Phase 3: More active-set descent rounds with different parameters
    if best["h"] is not None and deadline - time.monotonic() > 40:
        h2, v2 = active_descent(best["h"].copy(), deadline, need=30, max_rounds=300)
        consider(h2)
    
    # Phase 4: Random perturbations + re-optimize
    if best["h"] is not None and deadline - time.monotonic() > 50:
        for attempt in range(4):
            if deadline - time.monotonic() < 35:
                break
            n_base = best["n"]
            sigmas = [0.01, 0.02, 0.03, 0.05]
            sig = sigmas[attempt % len(sigmas)]
            h_pert = project_box_sum(best["h"] + sig * rng.standard_normal(n_base), n_base)
            consider(h_pert)
            F_pert = exact_value(h_pert)
            if F_pert < best["val"] + 0.03 and deadline - time.monotonic() > 40:
                h3, v3 = active_descent(h_pert.copy(), deadline, need=25, max_rounds=150)
                consider(h3)
    
    # Phase 5: Binary local search (if solution is near-binary)
    if best["h"] is not None and deadline - time.monotonic() > 30:
        h_bin = best["h"].copy()
        frac_above = np.sum(h_bin > 0.3)
        frac_below = np.sum(h_bin < 0.7)
        if frac_above <= best["n"] * 0.55 and frac_below <= best["n"] * 0.55:
            h4, v4 = binary_search_pairs(h_bin, exact_value(h_bin),
                                         max_flips=500, deadline=deadline, need=20)
            consider(h4)
    
    # Phase 6: Final polishing - very small active sets, fine line search
    if best["h"] is not None and deadline - time.monotonic() > 20:
        h_pol = best["h"].copy()
        n_pol = h_pol.size
        F_pol = exact_value(h_pol)
        for k in [1, 2, 3]:
            if deadline - time.monotonic() < 15:
                break
            for _ in range(50):
                if deadline - time.monotonic() < 15:
                    break
                c = corr_components(h_pol)
                G = grad_matrix(h_pol)
                order = np.argsort(c)[::-1]
                active = order[:k]
                A = G[active, :]
                d = gradient_direction(G, c, k, h_pol, bound_scale=0.01)
                if np.max(np.abs(d)) < 1e-12:
                    break
                d_lp, _ = find_direction_lp(G, c, k, h_pol, bound_scale=0.01)
                if d_lp is not None:
                    d = d_lp
                h_new, F_new = line_search_fine(h_pol, d, F_pol, 0.2)
                if F_new < F_pol - 1e-12:
                    h_pol = h_new
                    F_pol = F_new
                else:
                    break
        consider(h_pol)
    
    final_h = best["h"]
    if final_h is None:
        final_h = project_box_sum(np.full(45, 0.5), 45)
    final_h = project_box_sum(final_h, int(final_h.size))
    final_val = exact_value(final_h)
    if not math.isfinite(final_val) or final_h.size == 0:
        final_h = project_box_sum(np.full(45, 0.5), 45)
        final_val = exact_value(final_h)
    return final_h, float(final_val), int(final_h.size)
