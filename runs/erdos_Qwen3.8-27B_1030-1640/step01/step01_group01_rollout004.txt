def lp_minimax_step(h0, radius, method='highs'):
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * n
    c0, G = correl_lags_and_grads(h0)
    
    # LP: minimize t
    # s.t. G_m · h - t <= G_m · h0 - c_m  for each m
    #      sum(h) = S
    #      h0[i] - r <= h[i] <= h0[i] + r
    #      0 <= h[i] <= 1
    #      0 <= t <= 2
    
    # A_ub: L rows, n+1 columns
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c0
    
    # A_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Objective
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub)) + [(0.0, 2.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method, options={'presolve': True})
        if res.success and res.x is not None:
            h_lp = res.x[:n]
            t_lp = res.x[n]
            return h_lp, t_lp
    except Exception:
        pass
    
    # Try alternative methods
    for alt in ['highs', 'highs-ds', 'revised simplex', 'interior-point']:
        if alt == method:
            continue
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method=alt)
            if res.success and res.x is not None:
                return res.x[:n], res.x[n]
        except Exception:
            continue
    
    return None, None

import time
import math
import numpy as np
from scipy.optimize import linprog

def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return float('inf')
    return float(np.max(vals))

def adjust_sum(h, S, tol=1e-13):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(n * 4 + 20):
        r = S - float(np.sum(h))
        if abs(r) <= tol:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            if slack[idx] <= 1e-15:
                break
            h[idx] += min(r, slack[idx])
        else:
            idx = int(np.argmin(h))
            if h[idx] <= 1e-15:
                break
            h[idx] -= min(-r, h[idx])
    np.clip(h, 0.0, 1.0, out=h)
    return h

def proj_box_sum(v):
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * n
    if not np.all(np.isfinite(v)):
        return adjust_sum(np.full(n, 0.5), S)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s < S:
            hi = mid
        else:
            lo = mid
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum(p, S)

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        h = np.array([0.5, 0.5])
        return adjust_sum(h, 1.0)
    S = 0.5 * n
    if not np.all(np.isfinite(h)):
        return adjust_sum(np.full(n, 0.5), S)
    if np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12):
        np.clip(h, 0.0, 1.0, out=h)
        if abs(float(np.sum(h)) - S) <= 1e-6:
            return adjust_sum(h, S)
    return proj_box_sum(h)

def correl_grads(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / n
    L = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n), dtype=float)
    for k in range(L):
        shift = k - (n - 1)
        g = np.zeros(n)
        if shift >= 0:
            if n - shift > 0:
                g[shift:] = 1.0 - h[:n - shift]
                g[:n - shift] -= h[shift:]
        else:
            s = -shift
            if n - s > 0:
                g[:n - s] = 1.0 - h[s:]
                g[s:] -= h[:n - s]
        G[k] = g * dx
    return c, G

def lp_step(h0, radius, method='highs'):
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * n
    c0, G = correl_grads(h0)
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c0 + 1e-12
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    for i in range(n):
        if lb[i] > ub[i]:
            lb[i] = ub[i] = h0[i]
    bounds = list(zip(lb, ub)) + [(0.0, 2.0)]
    methods_to_try = [method] + ['highs', 'highs-ds', 'revised simplex', 'interior-point']
    seen = set()
    for m in methods_to_try:
        if m in seen:
            continue
        seen.add(m)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method=m, options={'presolve': True})
            if res.success and res.x is not None and np.all(np.isfinite(res.x)):
                return res.x[:n], float(res.x[n])
        except Exception:
            continue
    return None, None

def local_polish(h0, max_iter=80, init_r=0.03, max_r=0.08, min_r=2e-5,

    c = exact_c5(h)
    n = h.size

    fail_count = 0

    best_c = c

            break
        if radius < min_r or fail_count > 12:
            break

        if h_lp is None:
            radius *= 0.55

            continue
        h_lp_feas = ensure_feasible(h_lp)
        delta = h_lp_feas - h
        dnorm = float(np.max(np.abs(delta)))
        if dnorm < 1e-10:
            radius *= 0.5
            fail_count += 1
            continue
        accepted = False
        best_a_c = c
        best_a_h = h
        for alpha in alphas:
            h_try = ensure_feasible(h + alpha * delta)
            c_try = exact_c5(h_try)
            if c_try < best_a_c - EPS_OBJ:
                best_a_c = c_try
                best_a_h = h_try
                accepted = True
        if accepted:
            h = best_a_h
            c = best_a_c
            if c < best_c - EPS_OBJ or (abs(c - best_c) <= EPS_OBJ and h.size < best_h.size):
                best_h = h.copy()
                best_c = c
            fail_count = 0
            if dnorm > 0.6 * radius:

        else:

def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return h
    return ensure_feasible(0.5 * (h + h[::-1]))

def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(max(4, int(n_new)))
    if n_old == 0:
        return ensure_feasible(np.full(n_new, 0.5))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

    elif mode == 2:

        if cr < best_c + 0.012 and time.monotonic() < t_end - reserve - 100:
            h3, c3 = local_polish(hr, max_iter=25, init_r=0.02, max_r=0.05,
                                  min_r=1e-5, stop_time=t_end, reserve=reserve + 20)
            try_update(h3, c3, h3.size)
            hs3 = symmetrize(h3)
            cs3 = exact_c5(hs3)
            try_update(hs3, cs3, hs3.size)
        attempts += 1

    # Phase 4: Fine polish on best
    if best_h is not None and time.monotonic() < t_end - reserve - 20:
        h4, c4 = local_polish(best_h, max_iter=60, init_r=0.015, max_r=0.04,
                              min_r=5e-6, stop_time=t_end, reserve=reserve + 5)
        try_update(h4, c4, h4.size)
        hs4 = symmetrize(h4)
        cs4 = exact_c5(hs4)
        try_update(hs4, cs4, hs4.size)

    # Phase 5: Ultra-fine polish
    if best_h is not None and time.monotonic() < t_end - reserve - 10:
        h5, c5 = local_polish(best_h, max_iter=40, init_r=0.006, max_r=0.02,
                              min_r=3e-6, stop_time=t_end, reserve=reserve + 3)
        try_update(h5, c5, h5.size)
        hs5 = symmetrize(h5)
        cs5 = exact_c5(hs5)
        try_update(hs5, cs5, hs5.size)

    # Final safety
    if best_h is None:
        best_h = ensure_feasible(np.full(64, 0.5))
    h_final = ensure_feasible(best_h)
    c_final = exact_c5(h_final)
    if not np.isfinite(c_final) or c_final > 0.5:
        h_final = ensure_feasible(np.full(64, 0.5))
        c_final = exact_c5(h_final)
    return h_final, float(c_final), int(h_final.size)
