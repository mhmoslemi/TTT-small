    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # term1: (1-h)[j+s] for valid j
    if s >= 0:

def find_direction(h, active_shifts, eps=0.001):
    n = len(h)
    G = np.array([grad_row(h, s) for s in active_shifts])
    d = cp.Variable(n)
    constraints = [
        G @ d <= -eps * np.ones(len(active_shifts)),
        cp.sum(d) == 0,
        d >= -h,
        d <= 1 - h,

    obj = cp.Minimize(0.5 * cp.sum_squares(d))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.OSQP)  # or ECOS, SCS
    if prob.status in ('optimal', 'optimal_inaccurate'):

    return None

def active_descent(h, max_iters=500, deadline=None):
    h = clean_feasible(h.copy(), len(h))
    n = len(h)
    best_h = h.copy()
    best_val = exact_value(h)
    K = 8  # number of active lags
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        c = components(h)
        F = float(c.max())
        if F < best_val:
            best_val = F
            best_h = h.copy()
        
        # Get active set: top K lags
        order = np.argsort(c)[::-1]
        active_idx = order[:K]
        # Convert to shifts
        active_shifts = active_idx - (n - 1)
        
        # Compute gradient of average active component
        G = np.array([grad_row(h, s) for s in active_shifts])
        g = G.mean(axis=0)
        g = g - g.mean()  # null space of sum constraint
        
        # Line search: try several step sizes
        # Direction is -g (descent)
        # Find max step
        mask_pos = g < 0  # where -g > 0, i.e., h increases
        mask_neg = g > 0  # where -g < 0, i.e., h decreases
        
        alpha_max = 1.0
        if np.any(mask_pos):
            alpha_max = min(alpha_max, ((1-h)[mask_pos] / (-g[mask_pos])).min())
        if np.any(mask_neg):
            alpha_max = min(alpha_max, (h[mask_neg] / g[mask_neg]).min())
        alpha_max = max(alpha_max, 0)
        
        # Try several step sizes
        candidates = [0.5, 1.0, 2.0, 3.0, 5.0]
        candidates = [a * alpha_max for a in [0.01, 0.05, 0.1, 0.3, 0.5, 0.8, 1.0]]
        
        best_alpha = 0
        best_F = F
        for alpha in sorted(set(candidates)):
            if alpha <= 0 or alpha > alpha_max:
                continue
            h_trial = np.clip(h + alpha * (-g), 0, 1)
            h_trial = clean_feasible(h_trial, n)
            F_trial = exact_value(h_trial)
            if F_trial < best_F - 1e-10:
                best_F = F_trial
                best_alpha = alpha
        
        if best_alpha > 0:
            h = np.clip(h - best_alpha * g, 0, 1)
            h = clean_feasible(h, n)
        else:
            # Stuck - try random perturbation
            break
    
    return best_h, best_val

For k < 0: ∂c_k/∂h[j] = (1-h[j-k]) * I(j-k < n) - h[j+k] * I(j ≥ -k)

Or equivalently:
For k ≥ 0: ∂c_k/∂h[j] = (1-h[j-k]) * I(0 ≤ j-k < n) - h[j+k] * I(0 ≤ j+k < n)
For k < 0: ∂c_k/∂h[j] = (1-h[j-k]) * I(0 ≤ j-k < n) - h[j+k] * I(0 ≤ j+k < n)

Wait... let me check. For k≥0, j≥k means j-k≥0. And j-k<n means j<n+k, which is always true since j<n. So I(j≥k) = I(0≤j-k<n) for k≥0. ✓
And j+k<n is the same as 0≤j+k<n (since j+k≥0 for j≥0, k≥0). ✓

For k<0: j-k<n means j<n+k. Since k<0, n+k<n, so this is a non-trivial upper bound.
And j≥-k means j+k≥0. So 0≤j+k<n. ✓
And 0≤j-k<n: since k<0, j-k=j+|k|≥0 always. And j-k<n means j<n+k. Same condition. ✓

So the unified formula is:
**∂c_k/∂h[j] = (1-h[j-k]) * I(0 ≤ j-k < n) - h[j+k] * I(0 ≤ j+k < n)**

This is elegant! And it's the same formula for all k.

Great, so the gradient computation is:
```python
def compute_grad(h, k):
    """Gradient of component at shift k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j - k < n:
            g += (1.0 - h[j - k])
        if 0 <= j + k < n:
            g -= h[j + k]
        grad[j] = g
    return grad * dx
```

This is O(n) per lag, which is fine.

Now let me think about efficiency. For each iteration, I need:
1. Compute all components: O(n²) via np.correlate
2. Find active set: O(n log n)
3. Compute gradients for active lags: O(K*n)
4. Line search: ~7 evaluations × O(n²)

Total per iteration: O(n²) dominated by the correlation. For n=45, that's ~2000 ops for the correlation plus ~7*2000 for line search = ~16000 ops. In Python with numpy, each np.correlate call on n=45 arrays takes maybe 10-20 microseconds. So each iteration might take 100-200 microseconds. We can do ~500,000 iterations in 1100 seconds. More than enough.

Now let me think about the QP escape. When the simple gradient method stalls (no improvement for several iterations), I'll try a QP:

min (1/2) ||d||²
s.t. G_A * d ≤ -ε (reduce active components)
     sum(d) = 0
     -h ≤ d ≤ 1-h

I'll use cvxpy with ECOS or SCS solver for this. It should be fast for small n.

Actually, let me reconsider. Instead of a separate QP phase, I can make the line search smarter. The issue with the simple gradient method is that it might pick a direction that reduces the target lag but increases other lags, resulting in no net improvement.

A better approach: use the **average gradient of all active lags** as the direction. This naturally balances the reduction across active lags.

g_avg = (1/|A|) * sum_{k in A} grad_k

Then d = project(-g_avg) onto the feasible direction space (sum=0, box).

This is what I had before. Let me stick with this and add the QP as a fallback.

Let me also add a **restart mechanism**: if stuck for 50 iterations, perturb h slightly (add small noise, reproject).

Here's my final algorithm:

```
Phase 1: Active descent (up to 5000 iterations or time limit)
  - Each iteration: compute active set, average gradient, line search
  - Restart with perturbation every 200 stagnant iterations

Phase 2: QP escape (up to 20 QP calls)
  - When stuck, solve QP to find a better direction
  - Use the QP solution for line search

Phase 3: Grid size exploration
  - Resample to n±1, n±2, n±3, n±5
  - Run active descent on each
  - Keep best

Phase 4: Final polish
  - Run a few more iterations with smaller K (top 3 lags)
  - Try palindromic constraint
```

import time
import math
import numpy as np
import cvxpy as cp

def project_feasible(y, n):
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y[np.isnan(y) | np.isinf(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            add = min(diff, 1.0 - x[idx])
            x[idx] += add
        else:
            idx = int(np.argmax(x))
            rem = min(-diff, x[idx])
            x[idx] -= rem
    return x

def eval_components(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_max(h):
    n = len(h)
    if n <= 0:
        return math.inf
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_grad(h, k):
    n = len(h)
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jm = j - k
        if 0 <= jm < n:
            g += (1.0 - h[jm])
        jp = j + k
        if 0 <= jp < n:
            g -= h[jp]
        grad[j] = g
    return grad * dx

def line_search(h, direction, F0, n, max_evals=12):
    alpha_max = 1e6
    pos = direction < 0
    neg = direction > 0
    if np.any(pos):
        a_max = float(np.min((1.0 - h)[pos] / (-direction[pos])))
        alpha_max = min(alpha_max, a_max)
    if np.any(neg):
        a_max = float(np.min(h[neg] / direction[neg]))
        alpha_max = min(alpha_max, a_max)
    alpha_max = max(alpha_max * 0.999, 0.0)
    if alpha_max <= 0:
        return h, F0, 0.0
    fracs = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]
    best_alpha = 0.0
    best_F = F0
    best_h = h
    for f in fracs:
        alpha = f * alpha_max
        if alpha <= 0:
            continue
        ht = h + alpha * direction
        ht = project_feasible(ht, n)
        Ft = eval_max(ht)
        if Ft < best_F - 1e-12:
            best_F = Ft
            best_alpha = alpha
            best_h = ht
    if best_alpha == 0.0:
        return h, F0, 0.0
    lo_a = 0.0
    hi_a = best_alpha * 2.0 if best_alpha * 2.0 < alpha_max else alpha_max
    if hi_a > alpha_max:
        hi_a = alpha_max
    for _ in range(15):
        mid = 0.5 * (lo_a + hi_a)
        if mid <= best_alpha:
            lo_a = mid
        else:
            ht = h + mid * direction
            ht = project_feasible(ht, n)
            Ft = eval_max(ht)
            if Ft < best_F - 1e-12:
                best_F = Ft
                best_alpha = mid
                best_h = ht
                hi_a = mid
            else:
                lo_a = mid
        if hi_a - lo_a < 1e-6 * alpha_max:
            break
    return best_h, best_F, best_alpha

def qp_direction(h, active_shifts, eps, n):
    G = np.array([compute_grad(h, s) for s in active_shifts])
    p = len(active_shifts)
    d_var = cp.Variable(n)
    constraints = [
        G @ d_var <= -eps * np.ones(p),
        cp.sum(d_var) == 0,
        d_var >= -h,
        d_var <= 1.0 - h,
    ]
    obj = cp.Minimize(0.5 * cp.sum_squares(d_var))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.ECOS, cp.SCS, cp.OSQP]:
        try:
            prob.solve(solver=solver, max_time=5)
            if prob.status in ('optimal', 'optimal_inaccurate') and d_var.value is not None:
                d_val = np.asarray(d_var.value, dtype=float).reshape(-1)
                if d_val.size == n and np.all(np.isfinite(d_val)):
                    return d_val
        except Exception:
            continue
    return None

def active_descent(h, deadline, max_iters=3000, K=8, verbose=False):
    h = project_feasible(h.copy(), len(h))
    n = len(h)
    best_h = h.copy()
    best_val = eval_max(h)
    stagnant = 0
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        c = eval_components(h)
        F = float(c.max())
        if F < best_val - 1e-12:
            best_val = F
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 300:
            rng = np.random.default_rng(it)
            h = project_feasible(h + 0.01 * rng.standard_normal(n), n)
            stagnant = 0
            continue

        order = np.argsort(c)[::-1]
        k_act = min(K, n)
        active_idx = order[:k_act]
        active_shifts = active_idx - (n - 1)

        G = np.array([compute_grad(h, s) for s in active_shifts])
        g = G.mean(axis=0)
        g = g - g.mean()

        if np.linalg.norm(g) < 1e-14:
            break

        direction = -g
        h, F_new, alpha = line_search(h, direction, F, n)

        if alpha == 0.0 and k_act > 3:
            g2 = compute_grad(h, int(active_shifts[0]))
            g2 = g2 - g2.mean()
            if np.linalg.norm(g2) > 1e-14:
                direction2 = -g2
                h, F_new2, alpha2 = line_search(h, direction2, F, n)
                if alpha2 > 0:
                    F_new = F_new2

        if stagnant > 150 and it % 50 == 0:
            eps = 0.0005 * max(0.1, abs(best_val))
            d_q = qp_direction(h, active_shifts.tolist(), eps, n)
            if d_q is not None and np.linalg.norm(d_q) > 1e-8:
                h_q, F_q, a_q = line_search(h, d_q, F, n)
                if a_q > 0 and F_q < F:
                    h = h_q

    return best_h, best_val

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return project_feasible(h.copy(), n)
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_feasible(y, n_new)

def symmetrize(h):
    n = len(h)
    h_sym = 0.5 * (h + h[::-1])
    return project_feasible(h_sym, n)

def try_symmetric_descent(h, deadline, max_iters=1500, K=6):
    n = len(h)
    h_s = symmetrize(h)
    best_hs = h_s.copy()
    best_vs = eval_max(h_s)
    half = n // 2
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        c = eval_components(h_s)
        F = float(c.max())
        if F < best_vs - 1e-12:
            best_vs = F
            best_hs = h_s.copy()
        order = np.argsort(c)[::-1]
        active_shifts = (order[:min(K, n)] - (n - 1)).astype(int)
        G = np.array([compute_grad(h_s, s) for s in active_shifts])
        g = G.mean(axis=0)
        g = g - g.mean()
        g = 0.5 * (g + g[::-1])
        g = g - g.mean()
        if np.linalg.norm(g) < 1e-14:
            break
        direction = -g
        h_s, F_new, alpha = line_search(h_s, direction, F, n)
        if alpha > 0:
            h_s = project_feasible(0.5 * (h_s + h_s[::-1]), n)
    return project_feasible(0.5 * (best_hs + best_hs[::-1]), n), best_vs
