    n = len(h)
    G = compute_all_lags(h)
    # Find the active set (lags achieving the max)
    Gmax = np.max(G)

                grad[i + k] += -dx * h[i]
        else:

            for i in range(l, n):
                grad[i] += dx * (1 - h[i - l])
                grad[i - l] += -dx * h[i]
    # Normalize
    grad /= len(np.where(active)[0])
    # Step
    y = h - alpha * grad
    return project_h(y, target), grad
