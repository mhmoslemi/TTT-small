import time
import math
import numpy as np
from scipy.optimize import linprog

def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))

def compute_c_and_G(h):
    """Compute correlation values and gradient matrix."""
    n = h.size
    L = 2 * n - 1
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    # For each lag m, shift = m - (n-1)
    for m in range(L):
        s = m - (n - 1)
        for j in range(n):
            # dc/dh[j] = dx * [I(j+s in [0,n-1])*(1-h[j+s]) - I(j-s in [0,n-1])*h[j-s]]
            g = 0.0
            if 0 <= j + s <= n - 1:
                g += 1.0 - h[j + s]
            if 0 <= j - s <= n - 1:
                g -= h[j - s]
            G[m, j] = dx * g
    return c, G

def project_box_sum(v, target_sum=None):
    """Project v onto {0<=x<=1, sum(x)=target_sum}."""
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    if target_sum is None:
        target_sum = 0.5 * float(n)
    # Binary search for lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    lo = v.min() - 1.0 - 1e-9
    hi = v.max() + 1.0 + 1e-9
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    r = target_sum - p.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.argmax(1.0 - p)
            p[idx] = min(1.0, p[idx] + r)
        else:
            idx = np.argmin(p)
            p[idx] = max(0.0, p[idx] + r)
    # Second pass for tiny residual
    for _ in range(100):
        r = target_sum - p.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            idxs = np.where(p < 1.0 - 1e-12)[0]
            if len(idxs) == 0:
                break
            idx = idxs[0]
            p[idx] = min(1.0, p[idx] + r)
        else:
            idxs = np.where(p > 1e-12)[0]
            if len(idxs) == 0:
                break
            idx = idxs[0]
            p[idx] = max(0.0, p[idx] + r)
    return p

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        h = np.array([0.5, 0.5])
        return project_box_sum(h)
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5)
    return project_box_sum(h, 0.5 * float(n))

def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return project_box_sum(0.5 * (h + h[::-1]), 0.5 * float(h.size))

def resample_h(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(n_new)
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

def lp_minimax_step(h0, radius, margin=1e-9):
    """Solve one LP minimax step. Returns (h_candidate, t_val) or (None, None)."""
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * float(n)
    
    c, G = compute_c_and_G(h0)
    
    # A_ub: G @ h - t <= G @ h0 - c - margin (wait, + margin for feasibility)
    # Actually: t >= c + G@(h-h0) => G@h - t <= G@h0 - c
    # For numerical safety, add margin: G@h - t <= G@h0 - c + margin
    # This makes the constraint slightly relaxed at h0, ensuring feasibility
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    
    b_ub = G @ h0 - c + margin
    
    # Equality: sum(h) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, 2.0))  # t
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = res.x[:n]
            t_val = res.x[n]
            return h_new, t_val
    except Exception:
        pass
    return None, None

def local_lp_optimize(h0, c0, max_iter=100, init_radius=0.03, stop_time=None):
    """Iterative LP minimax optimization with line search."""
    n = h0.size
    r = init_radius
    min_r = 1e-5
    max_r = 0.12
    current_h = h0.copy()
    current_c = c0
    no_improve_count = 0
    
    for it in range(max_iter):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        
        h_lp, t_lp = lp_minimax_step(current_h, r)
        if h_lp is None:
            r *= 0.5
            no_improve_count += 1
            if no_improve_count > 8:
                break
            if r < min_r:
                break
            continue
        
        h_lp = ensure_feasible(h_lp)
        
        # Line search
        delta = h_lp - current_h
        dnorm = np.max(np.abs(delta))
        if dnorm < 1e-10:
            r *= 0.5
            no_improve_count += 1
            if no_improve_count > 5 or r < min_r:
                break
            continue
        
        best_candidate = None
        best_candidate_c = current_c
        
        alphas = [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]
        for alpha in alphas:
            h_test = ensure_feasible(current_h + alpha * delta)
            c_test = exact_c5(h_test)
            if c_test < best_candidate_c - 1e-11:
                best_candidate_c = c_test
                best_candidate = h_test
            if c_test <= current_c - 1e-10:
                break  # Good enough, use this alpha
        
        if best_candidate is not None:
            current_h = best_candidate
            current_c = best_candidate_c
            no_improve_count = 0
            # Grow radius slightly on success
            r = min(max_r, r * 1.2)
        else:
            r *= 0.5
            no_improve_count += 1
            if no_improve_count > 8:
                break
            if r < min_r:
                break
    
    return current_h, current_c

import time

import numpy as np
from scipy.optimize import linprog


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0 or not np.all(np.isfinite(vals)):
        return float('inf')
    return float(np.max(vals))


def compute_c_and_G(h):
    n = h.size
    L = 2 * n - 1
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.where(np.isfinite(c), c, 0.0)
    G = np.zeros((L, n))
    for m in range(L):
        s = m - (n - 1)
        for j in range(n):
            g = 0.0
            js = j + s
            if 0 <= js <= n - 1:
                g += 1.0 - h[js]
            jm = j - s
            if 0 <= jm <= n - 1:
                g -= h[jm]
            G[m, j] = dx * g
    return c, G


def project_box_sum(v, target_sum=None):
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    if target_sum is None:
        target_sum = 0.5 * float(n)
    if not np.all(np.isfinite(v)):
        v = np.full(n, 0.5)
    lo = float(v.min()) - 1.0 - 1e-9
    hi = float(v.max()) + 1.0 + 1e-9
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target_sum - float(p.sum())
    if abs(r) > 1e-13:
        if r > 0:
            idx = int(np.argmax(1.0 - p))
            p[idx] = min(1.0, p[idx] + r)
        else:
            idx = int(np.argmin(p))
            p[idx] = max(0.0, p[idx] + r)
    for _ in range(200):
        r = target_sum - float(p.sum())
        if abs(r) < 1e-15:
            break
        if r > 0:
            idxs = np.where(p < 1.0 - 1e-14)[0]
            if idxs.size == 0:
                break
            idx = int(idxs[0])
            p[idx] = min(1.0, p[idx] + r)

            idxs = np.where(p > 1e-14)[0]
            if idxs.size == 0:
                break
            idx = int(idxs[0])
            p[idx] = max(0.0, p[idx] + r)
    return p


def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        h = np.array([0.5, 0.5])
        return project_box_sum(h)
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5)
    return project_box_sum(h, 0.5 * float(n))


def symmetrize_h(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return project_box_sum(0.5 * (h + h[::-1]), 0.5 * float(h.size))


def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(max(2, n_new))
    if n_old == 0:
        return project_box_sum(np.full(n_new, 0.5), 0.5 * float(n_new))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return ensure_feasible(y)


def lp_minimax_step(h0, radius, margin=1e-9):
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * float(n)
    c, G = compute_c_and_G(h0)
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c + margin
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs-ds')
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None
