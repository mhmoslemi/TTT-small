def active_set_gradient_descent(h0, n_active=5, lr=None, momentum=0.7, max_iter=500, time_limit=60):
    """Active-set gradient descent with momentum."""
    h = project_feasible(h0.copy())
    n = h.size
    t0 = time.monotonic()
    
    if lr is None:
        lr = 0.01
    
    v = np.zeros(n, dtype=float)
    best_c = exact_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = compute_overlaps(h)
        c_max = np.max(c)
        
        if c_max < best_c - 1e-12:
            best_c = c_max
            best_h = h.copy()
        
        # Find active set: top n_active shifts
        active = np.argsort(c)[-n_active:]
        
        # Compute gradient
        g = grad_max_overlap(h, active)
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = project_feasible(h + v)
        
        # Line search (backtracking)
        c_new = exact_c5(h_new)
        step_factor = 1.0
        for _ in range(10):
            if c_new < c_max + 1e-10:
                break
            step_factor *= 0.5
            h_new = project_feasible(h + step_factor * v)
            c_new = exact_c5(h_new)
        
        h = h_new
        c_current = exact_c5(h)
        
        if c_current < best_c - 1e-12:
            best_c = c_current
            best_h = h.copy()
        
        # Adapt learning rate
        if it % 50 == 0:
            if c_current > best_c:
                lr *= 0.7
            else:
                lr = min(lr * 1.1, 0.1)
    
    return best_h, best_c

def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0:
        return float('inf')
    return float(np.max(vals))

def compute_overlaps(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(v):
    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * n
    if not np.all(np.isfinite(v)):
        v = np.full(n, 0.5)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s < S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(h)
    if abs(r) > 1e-13:
        if r > 0:
            idxs = np.where(h < 1.0 - 1e-14)[0]
        else:
            idxs = np.where(h > 1e-14)[0]
        if idxs.size > 0:
            idx = idxs[0]
            delta = min(abs(r), abs(r) / 1.0) if r > 0 else min(-r, h[idx])
            if r > 0:
                h[idx] = min(1.0, h[idx] + r)

def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    if n_old == n_new:
        return project_feasible(h.copy())
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(y)

def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return project_feasible(0.5 * (h + h[::-1]))

def gradoverlap_single(h, s):
    """Gradient of overlap at shift s (can be negative). Returns array of length n."""
    n = h.size
    dx = 2.0 / float(n)
    g = np.zeros(n)
    if s >= 0:
        end = n - 1 - s
        if end < 0:
            return g
        # d/dh[j] = dx * [1_{j<=end} - 1_{j<=end}*h[j+s] - 1_{j>=s}*h[j-s]]
        # For j in [0, min(s-1, end)]: dx * (1 - h[j+s])
        # For j in [s, end] (if s <= end): dx * (1 - h[j+s] - h[j-s])
        # For j in [end+1, n-1] but j>=s: dx * (-h[j-s])
        # Simplified: j in [0, end] gets (1-h[j+s]), j in [s, n-1] gets -h[j-s]
        grad_first = end + 1
        g[0:grad_first] += dx * (1.0 - h[s:s + grad_first])
        if s < n:
            g[s:n] += dx * (-h[0:n - s])
    else:
        t = -s  # t > 0
        # overlap = dx * sum_{i=t}^{n-1} h[i]*(1-h[i-t])
        # d/dh[j]: j in [t, n-1] gets dx*(1-h[j-t]), j in [0, n-1-t] gets dx*(-h[j+t])
        g[t:n] += dx * (1.0 - h[0:n - t])
        g[0:n - t] += dx * (-h[t:n])
    return g

def active_gradient(h, active_shifts):
    """Average gradient over active shifts. active_shifts are actual shift values (not correlation indices)."""
    n = h.size
    grad = np.zeros(n)
    for s in active_shifts:
        grad += gradoverlap_single(h, int(s))
    grad /= len(active_shifts)
    return grad

    n = h.size
    t0 = time.monotonic()

            best_h = h.copy()
            patience = 0
        else:
            patience += 1

            lr *= 0.8
            momentum = max(0.3, momentum * 0.95)
            if lr < 0.001:
                break
            patience = 0
        
        # Find active shifts (top n_active)
        idx_sorted = np.argsort(c)[-n_active_max:]
        active_shifts = [(int(i) - n + 1) for i in idx_sorted]
        
        # Compute gradient
        g = active_gradient(h, active_shifts)
        
        # Momentum step
        v = momentum * v - lr * g
        h_new = project_feasible(h + v)
        
        # Backtracking line search
        alpha = 1.0
        c_new = exact_c5(h_new)
        for _ in range(12):
            if c_new < c_max + 1e-12:
                break
            alpha *= 0.5
            h_new = project_feasible(h + alpha * v)
            c_new = exact_c5(h_new)

            if c_new < c_max - 1e-4:
                lr = min(lr * 1.15, 0.08)
            elif c_new > c_max + 1e-6:

    return best_h, best_c

def coordinate_refine(h0, time_limit=20.0):

            if time.monotonic() - t0 > time_limit:
                break
            for j in range(i + step, n, step):
                if abs(i - j) < step:
                    continue
                # Try transferring mass from i to j and vice versa
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer < 1e-6:
                    continue

                    if amt < 1e-8:
                        continue
                    h_try = h.copy()

                    h_try[j] += amt
                    c_try = exact_c5(h_try)
                    if c_try < best_c - 1e-10:
                        h = h_try
                        best_h = h.copy()
                        best_c = c_try
                        improved = True
                        break
                # Also try reverse
                max_transfer2 = min(h[j], 1.0 - h[i])
                if max_transfer2 < 1e-6:
