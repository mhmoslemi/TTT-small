    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx, w in zip(active_idx, weights):
        if w < 1e-15:
            continue
        s = k_idx - (n - 1)
        # (1-h[j+s]) for valid j+s
        js = np.arange(n) + s

        term1 = np.zeros(n)
        term1[valid_pos] = 1.0 - h[js[valid_pos]]

        term2[valid_neg] = h[js2[valid_neg]]

    Returns gradient of sum_k weights[k] * C_{s_k}

def binary_swap_delta(b, p, q, n):
    s = np.arange(-(n - 1), n)
    idx1 = p - s
    idx2 = q - s
    idx3 = q + s
    idx4 = p + s
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    v3 = (idx3 >= 0) & (idx3 < n)
    v4 = (idx4 >= 0) & (idx4 < n)
    bp_s = np.zeros(2 * n - 1)

    bq_s = np.zeros(2 * n - 1)
    bq_s[v2] = b[idx2[v2]]
    bq_ps = np.zeros(2 * n - 1)

    bp_ps = np.zeros(2 * n - 1)
    bp_ps[v4] = b[idx4[v4]]
    return (bp_s - bq_s) + (bq_ps - bp_ps)

            b[zeros_idx[i]] = 1.0
    elif n_ones > target:
        ones_idx = np.where(b == 1)[0]
        rng.shuffle(ones_idx)
        for i in range(min(n_ones - target, len(ones_idx))):
            b[ones_idx[i]] = 0.0
    return b.astype(float)

def make_balanced_binary(n, rng):
    b = np.zeros(n)
    idx = rng.permutation(n)
    b[idx[:n // 2]] = 1.0
    return b

def make_binary_from_parent(parent_h, n, rng):
    x_old = (np.arange(len(parent_h)) + 0.5) * 2.0 / len(parent_h)
    x_new = (np.arange(n) + 0.5) * 2.0 / n
    interp = np.interp(x_new, x_old, parent_h)
    # Threshold to binary
    sorted_vals = np.sort(interp)[::-1]
    threshold = sorted_vals[n // 2 - 1] if n // 2 > 0 else 0
    b = (interp >= threshold).astype(float)
    # Fix count if needed
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones < target:
        zeros_idx = np.where(b == 0)[0]
        # Pick highest-valued zeros
        order = np.argsort(-interp[zeros_idx])
        for i in range(min(target - n_ones, len(zeros_idx))):
            b[zeros_idx[order[i]]] = 1.0
    elif n_ones > target:
        ones_idx = np.where(b == 1)[0]
        order = np.argsort(interp[ones_idx])
        for i in range(min(n_ones - target, len(ones_idx))):
            b[ones_idx[order[i]]] = 0.0
    return b

def binary_to_continuous(b, eps):
    return eps + (1.0 - 2.0 * eps) * b

def resample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def full_polish(h, n, time_limit, rng):
    """Multi-stage polish: active set, pair search, LSE."""
    f = evaluate_c5(h, n)
    t0 = time.time()
    frac1 = 0.45
    frac2 = 0.30
    frac3 = 0.25
    t1 = (time_limit - 1.0) * frac1
    t2 = (time_limit - 1.0) * frac2
    t3 = (time_limit - 1.0) * frac3

    h, f = active_optimize(h, n, min(t1, max(1, time_limit - (time.time() - t0) - 1)), tol=1e-4)
    h, f = pair_local_search(h, n, min(t2, max(1, time_limit - (time.time() - t0) - 1)), rng)
    h, f = lse_polish(h, n, min(t3, max(1, time_limit - (time.time() - t0) - 1)))
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    def tleft():
        return budget_s - 8.0 - (time.time() - t_start)

    # Initialize from parent
    n0 = len(initial_h_values)
    parent = np.array(initial_h_values, dtype=float)
    parent = project_feasible(parent, n0)
    best_h = parent.copy()
    best_c5 = evaluate_c5(parent, n0)

    # Fallback
    fallback_h = 0.5 * np.ones(n0)
    fallback_c5 = 0.5

    if best_c5 > fallback_c5:
        best_h = fallback_h
        best_c5 = fallback_c5

    if tleft() < 10:
        return best_h, best_c5, n0

    # === PHASE 1: Binary annealing at multiple even resolutions ===
    # Use even n for binary search
    n0_even = n0 if n0 % 2 == 0 else n0 + 1
    binary_resolutions = []
    for dn in [0, -2, 2, -4, 4, 6, -6, 8, 10, 12]:
        cand = n0_even + dn
        if 30 <= cand <= 100 and cand % 2 == 0 and cand not in binary_resolutions:
            binary_resolutions.append(cand)
    binary_resolutions = binary_resolutions[:6]

    golden = 0.6180339887498949
    slopes = [golden, 1.0 / golden, 0.38196601125, 0.5, 2.0/3.0, 3.0/5.0]

    binary_candidates = []  # (b, n, raw_max)
    for n_b in binary_resolutions:
        if tleft() < 30:
            break
        n_chains = 3 if n_b <= n0_even + 4 else 2
        time_per_chain = max(2, min(20, tleft() / (n_chains * max(1, len(binary_resolutions) - binary_resolutions.index(n_b) + 1))))
        for ch in range(n_chains):
            if tleft() < 5:
                break
            if ch == 0:
                b_init = make_binary_from_parent(parent, n_b, rng)
            elif ch == 1:
                b_init = make_sturmian(n_b, slopes[rng.integers(len(slopes))], rng)
            else:
                b_init = make_balanced_binary(n_b, rng)
            b_best, raw_max = binary_anneal(b_init, n_b, time_per_chain, rng)
            binary_candidates.append((b_best, n_b, raw_max))

    # === PHASE 2: Convert binary to continuous and polish ===
    # Sort by raw_max / n (proxy for C5)
    binary_candidates.sort(key=lambda x: x[2] * 2.0 / x[1])

    n_polish = min(len(binary_candidates), 8)
    candidates_to_polish = []

    # Also add parent and perturbed parent
    candidates_to_polish.append((parent.copy(), n0))

    for b, n_b, raw_max in binary_candidates[:n_polish]:
        for eps in [0.05, 0.1, 0.2]:
            h_cont = binary_to_continuous(b, eps)
            h_cont = project_feasible(h_cont, n_b)
            candidates_to_polish.append((h_cont, n_b))

    # Add some random perturbations of parent
    for scale in [0.05, 0.15, 0.3]:
        h_pert = project_feasible(parent + rng.normal(0, scale, n0), n0)
        candidates_to_polish.append((h_pert, n0))

    # Remove duplicates by (n, rounded tuple) - just limit total
    candidates_to_polish = candidates_to_polish[:15]

    # Polish each candidate
    for h_cand, n_cand in candidates_to_polish:
        if tleft() < 30:
            break
        c5_cand = evaluate_c5(h_cand, n_cand)
        # Only polish if promising
        if c5_cand > best_c5 + 0.02:
            continue
        polish_time = min(40, max(5, tleft() / max(1, len(candidates_to_polish) - candidates_to_polish.index((h_cand, n_cand)))))
        h_pol, f_pol = full_polish(h_cand, n_cand, polish_time, rng)
        if f_pol < best_c5:
            best_c5 = f_pol
            best_h = h_pol.copy()

    # === PHASE 3: Deeper polish on best ===
    if tleft() > 50:
        nb = len(best_h)
        polish_time = min(200, tleft() * 0.4)
        best_h, best_c5 = full_polish(best_h, nb, polish_time, rng)

    # === PHASE 4: Multi-resolution refinement ===
    if tleft() > 60:
        nb = len(best_h)
        for n_try in [nb - 2, nb + 2, nb - 4, nb + 4]:
            if tleft() < 30:
                break
            if n_try < 30 or n_try > 100:
                continue
            h_res = resample(best_h, nb, n_try)
            h_res = project_feasible(h_res, n_try)
            c5_res = evaluate_c5(h_res, n_try)
            if c5_res < best_c5 + 0.01:
                polish_time = min(30, max(5, tleft() / 2))
                h_pol, f_pol = full_polish(h_res, n_try, polish_time, rng)
                if f_pol < best_c5:
                    best_c5 = f_pol
                    best_h = h_pol.copy()

    # === PHASE 5: Final intensive polish ===
    if tleft() > 30:
        nb = len(best_h)
        # More aggressive active set optimization
        favicon = 1e-5
        best_h, f1 = active_optimize(best_h, nb, min(30, tleft() * 0.5), tol=favicon, max_iter=5000)
        if f1 < best_c5:
            best_c5 = f1
        best_h, f2 = pair_local_search(best_h, nb, min(20, tleft() * 0.3), rng)
        if f2 < best_c5:
            best_c5 = f2
            best_h = best_h.copy()
        if tleft() > 10:
            best_h, f3 = lse_polish(best_h, nb, min(15, tleft() * 0.5))
            if f3 < best_c5:
                best_c5 = f3

    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h, len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Ensure we return something valid
    if not np.isfinite(best_c5) or best_c5 > fallback_c5:
        return fallback_h, fallback_c5, len(fallback_h)

    return best_h, float(best_c5), int(len(best_h))
