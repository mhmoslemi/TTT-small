# --- Core utilities ---

def project_box_sum(y, n):
    """Project y onto {x in [0,1]^n : sum(x) = n/2}"""
    y = np.clip(np.asarray(y, dtype=float).reshape(-1), 0.0, 1.0)
    if y.size != n:
        y = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmax(1-x)) if diff > 0 else int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))
    return x

def c5_components(h):
    """Return all overlap components c_k for shift k."""
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_max(h):
    return float(np.max(c5_components(h)))

def c5_grad_matrix(h):
    """G[m, j] = dc_m/dh[j] for each lag m and variable j."""
    n = len(h)
    dx = 2.0 / n
    L = 2*n - 1
    G = np.zeros((L, n))
    for m in range(L):
        s = n - 1 - m  # shift in index terms
        i0 = max(0, -s)
        i1 = min(n-1, n-1-s)
        if i0 > i1: continue
        i = np.arange(i0, i1+1)
        j = i + s
        G[m, i] += dx * (1.0 - h[j])
        G[m, j] -= dx * h[i]
    return G

# --- Optimization ---

def soft_max(c, tau):
    """Log-sum-exp approximation of max(c)."""
    cmax = c.max()
    z = (c - cmax) / tau
    ez = np.exp(z)
    s = ez.sum()
    if s < 1e-300:
        return cmax, np.eye(len(c))[np.argmax(c)]
    return cmax + tau * np.log(s/len(c)), ez/s

def soft_max_grad(c, tau):
    """Return F and gradient w.r.t. c (weights)."""
    cmax = c.max()
    z = (c - cmax) / tau
    ez = np.exp(z)
    s = ez.sum()
    if s < 1e-300:
        w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
        return cmax, w
    F = cmax + tau * (np.log(s) - np.log(len(c)))
    return F, ez/s

def projected_gradient_step(h, tau, alpha, n_steps=100):
    """Run projected gradient descent on soft_max_tau(c(h))."""
    h = h.copy()
    dx = 2.0 / len(h)
    for _ in range(n_steps):
        c = c5_components(h)
        F, w = soft_max_grad(c, tau)
        G = c5_grad_matrix(h)
        grad = G.T @ w
        # Projected gradient step
        h_new = project_box_sum(h - alpha * grad, len(h))
        c_new = c5_components(h_new)
        F_new = soft_max(c_new, tau)[0]
        if F_new < F:
            h = h_new
            alpha = min(alpha * 1.1, 4.0)
        else:
            alpha *= 0.5
            if alpha < 1e-8:
                break
    return h

    """Solve epigraph with only the top n_active constraints."""
    n = len(h0)
    h0 = project_box_sum(h0, n)

    cmax = c0.max()

    def obj(x):
        return x[-1]  # minimize t
    
    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    # Constraints
    def eq_fun(x):
        return x[:-1].sum() - 0.5*n
    
    def eq_jac(x):
        J = np.zeros((1, len(x)))
        J[0, :-1] = 1.0
        return J
    
    def ineq_funs(x):
        """t - c_k >= 0 for active k's"""
        h = np.clip(x[:-1], 0.0, 1.0)
        c = c5_components(h)
        t = x[-1]
        return t - c[idx]
    
    def ineq_jacs(x):
        h = np.clip(x[:-1], 0.0, 1.0)
        G = c5_grad_matrix(h)
        hlen = len(h)
        J = np.zeros((n_active, hlen + 1))
        for i, k in enumerate(idx):
            J[i, :hlen] = -G[k]
            J[i, hlen] = 1.0
        return J
    
    x0 = np.append(h0.copy(), cmax + 0.001)
    bounds = [(0.0, 1.0)]*n + [(0.0, t_ub)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_funs, 'jac': ineq_jacs}
    ]
    
    try:

                                constraints=cons, method='SLSQP',
                                options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.fun < cmax:
            return project_box_sum(res.x[:-1], n)
    except:
        pass
    return None

import time
import math
import numpy as np
from scipy import optimize


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if y.size != n:
        y = np.full(n, 0.5)
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * n
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))
    return x


def c5_components(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def c5_max_val(h):
    return float(np.max(c5_components(h)))


def c5_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for m in range(L):
        s = n - 1 - m
        i0 = max(0, -s)
        i1 = min(n - 1, n - 1 - s)
        if i0 > i1:
            continue
        i = np.arange(i0, i1 + 1)
        j = i + s
        G[m, i] += dx * (1.0 - h[j])
        G[m, j] -= dx * h[i]
    return G


def soft_max_val(c, tau):
    cmax = float(c.max())
    z = (c - cmax) / tau
    ez = np.exp(z)
    s = float(ez.sum())
    if s < 1e-300:
        return cmax
    return cmax + tau * (math.log(s) - math.log(len(c)))


def soft_max_weights(c, tau):
    cmax = float(c.max())
    z = (c - cmax) / tau
    ez = np.exp(z)
    s = float(ez.sum())
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return ez / s


def pg_step(h, tau, alpha, n_steps, momentum=0.0, deadline=None, need=20.0):
    h = h.copy().astype(float)
    n = len(h)
    v_prev = None
    for _ in range(n_steps):
        if deadline is not None and (deadline - time.monotonic()) < need:
            break
        c = c5_components(h)
        w = soft_max_weights(c, tau)
        G = c5_grad_matrix(h)
        grad = G.T.dot(w)
        if not np.all(np.isfinite(grad)):
            break
        if momentum > 0.0 and v_prev is not None:
            step = grad + momentum * v_prev
        else:
            step = grad
        v_prev = step.copy()
        h_new = project_box_sum(h - alpha * step, n)
        c_new = c5_components(h_new)
        F_new = soft_max_val(c_new, tau)
        c_old_F = soft_max_val(c, tau)
        if F_new < c_old_F - 1e-15:
            h = h_new
            alpha = min(alpha * 1.08, 3.0)
        else:
            alpha *= 0.5
            if alpha < 1e-9:
                break
    return h


def epigraph_solve(h0, n_active, t_ub, maxiter=80, ftol=1e-11):
    n = len(h0)
    h0 = project_box_sum(h0, n)
    c0 = c5_components(h0)
    cmax = float(c0.max())
    idx = np.argsort(c0)[-min(n_active, len(c0)):]
    k = len(idx)

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return float(x[:-1].sum() - 0.5 * n)

    def eq_jac(x):
        J = np.zeros((1, len(x)))
        J[0, :-1] = 1.0
        return J

    def ineq_fun(x):
        h = np.clip(x[:-1], 0.0, 1.0)
        c = c5_components(h)
        return x[-1] - c[idx]

    def ineq_jac(x):
        h = np.clip(x[:-1], 0.0, 1.0)
        G = c5_grad_matrix(h)
        J = np.zeros((k, n + 1))
        for i, m in enumerate(idx):
            J[i, :n] = -G[m]
            J[i, n] = 1.0
        return J

    x0 = np.empty(n + 1)
    x0[:n] = h0
    x0[n] = cmax + 0.002
    bounds = [(0.0, 1.0)] * n + [(0.0, t_ub)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, bounds=bounds,
                                constraints=cons, method='SLSQP',
                                options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x[:n], n)
    except Exception:
        pass
    return None


def epigraph_solve2(h0, n_active, t_ub, maxiter=80):

def pair_polish(h, n_pairs_per_sweep, n_d, deadline, need=15.0, max_sweeps=5):
    """Pair coordinate descent polish."""
    h = project_box_sum(h.copy(), len(h))
    n = len(h)
    c = c5_components(h)
    best_val = float(c.max())
    G = c5_grad_matrix(h)

    for sweep in range(max_sweeps):
        if (deadline - time.monotonic()) < need:
            break
        pairs = []
        for _ in range(n_pairs_per_sweep):
            i = int(np.random.randint(n))
            j = int(np.random.randint(n))
            if i != j:
                if i > j:
                    i, j = j, i
                pairs.append((i, j))
        # Deduplicate
        pairs = list(set(pairs))

        for (i, j) in pairs:
            if (deadline - time.monotonic()) < need:
                break
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            # Evaluate at n_d points
            deltas = np.linspace(d_lo + 1e-9, d_hi - 1e-9, n_d)
            best_d = 0.0
            best_f = best_val
            for d in deltas:
                h[i] += d
                h[j] -= d
                c_new = c5_components(h)
                f = float(c_new.max())
                if f < best_f:
                    best_f = f
                    best_d = d
                h[i] -= d
                h[j] += d
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                c = c5_components(h)
                if float(c.max()) < best_val:
                    best_val = float(c.max())
    return project_box_sum(h, n), best_val
