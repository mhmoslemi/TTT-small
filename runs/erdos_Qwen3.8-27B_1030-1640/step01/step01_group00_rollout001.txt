    for m in range(len(c)):
        kk = n - 1 - m
        i0 = max(0, -kk)
        i1 = min(n - 1, n - 1 - kk)
        if i0 > i1:
            continue
        i = np.arange(i0, i1 + 1)
        j = i + kk
        g[i] += w[m] * dx * (1.0 - x[j])
        g[j] -= w[m] * dx * x[i]
    return g

def lse_eq_fun(x):
    n = int(_TASK['n'])
    return float(np.sum(x) - 0.5 * n)

def lse_eq_jac(x):
    n = int(_TASK['n'])
    return np.ones((1, n))

def lse_slsqp_refine(h, tau, deadline, maxiter=60):
    n = len(h)
    _TASK['n'] = n
    _TASK['tau'] = tau
    try:
        res = optimize.minimize(
            lse_obj, h.copy(), jac=lse_grad,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lse_eq_fun, 'jac': lse_eq_jac}],

            if x.size == n and np.all(np.isfinite(x)):

    except Exception:
        pass
    return None

Let me also think about a key improvement: **gradient-based coordinate descent**. Instead of a grid search, I can use the derivative of the objective w.r.t. h[p] to do a Newton or gradient step in 1D.

For the 1D subproblem: minimize f(x) = max_m (A_m * x + B_m) over x ∈ [0,1].

The subgradient of f at x is: avg of A_m for all m that achieve the max.
The minimum is where 0 is in the subgradient, i.e., where the active A_m's straddle 0.

I can find this by:
1. Evaluating f and its subgradient at the current x
2. If all active A_m > 0, decrease x
3. If all active A_m < 0, increase x
4. Otherwise, x is optimal

This is much faster than a grid search! Let me implement it:
