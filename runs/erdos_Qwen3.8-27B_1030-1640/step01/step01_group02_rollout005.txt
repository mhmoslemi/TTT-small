```python
def binary_delta_vec(b, n, p, q):
    s = np.arange(-(n-1), n)  # all lags
    delta = np.zeros(2*n-1)
    
    # p going 1->0:
    ps = p + s
    mask_ps = (ps >= 0) & (ps < n)
    delta[mask_ps] += b[ps[mask_ps]] - 1.0
    
    pm = p - s
    mask_pm = (pm >= 0) & (pm < n)
    delta[mask_pm] += b[pm[mask_pm]]
    
    # q going 0->1:
    qs = q + s
    mask_qs = (qs >= 0) & (qs < n)
    delta[mask_qs] += 1.0 - b[qs[mask_qs]]
    
    qm = q - s
    mask_qm = (qm >= 0) & (qm < n)
    delta[mask_qm] += -b[qm[mask_qm]]
    
    return delta
```

```python
def make_balanced_binary(n, rng, slope=None, offset=0.0):
    if slope is None:
        slope = rng.uniform(0.3, 0.7)
    x = np.arange(n, dtype=float)
    vals = slope * x + offset + rng.normal(0, 0.1, n)
    threshold = np.sort(vals)[n // 2]

        deficit = n // 2 - int(b.sum())
        to_add = zero_idx[:deficit]

def compute_grad_lse(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            cnt = n - s
            grad[:cnt] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:cnt]
        else:
            a = -s
            cnt = n - a
            grad[:cnt] += wk * (1.0 - h[a:])
            grad[a:] -= wk * h[:cnt]
    return grad * dx

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam
```

```python
import numpy as np
import time

def task_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def binary_delta_vec(b, n, p, q, s_vals):
    delta = np.zeros(2 * n - 1)
    ps = p + s_vals
    mask = (ps >= 0) & (ps < n)
    if np.any(mask):
        delta[mask] += b[ps[mask]] - 1.0
    pm = p - s_vals
    mask = (pm >= 0) & (pm < n)
    if np.any(mask):
        delta[mask] += b[pm[mask]]
    qs = q + s_vals
    mask = (qs >= 0) & (qs < n)
    if np.any(mask):
        delta[mask] += 1.0 - b[qs[mask]]
    qm = q - s_vals
    mask = (qm >= 0) & (qm < n)
    if np.any(mask):
        delta[mask] += -b[qm[mask]]
    return delta

            p = ones[rng.integers(len(ones))]
            q = zeros[rng.integers(len(zeros))]
            delta = binary_delta_vec(b, n, p, q, s_vals)
            nm = (corr + delta).max()
            if nm < best_new_max:
                best_new_max = nm
                best_delta = delta
                best_p, best_q = p, q

        if best_new_max <= cur_max + 1e-12:
            accept = True
        else:
            diff = best_new_max - cur_max
            if diff < T * 30:
                accept = rng.random() < np.exp(-diff / T)
            else:
                accept = False

        if accept:
            b[best_p] = 0.0
            b[best_q] = 1.0
            corr += best_delta
            cur_max = best_new_max
            ones.remove(best_p)
            ones.append(best_q)
            zeros.remove(best_q)
            zeros.append(best_p)
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cool_rate
        if T < 0.03:
            T = 0.03
        if no_improve > 8000:
            T = T_start * 0.4
            no_improve = 0

    return best_b, best_max

def make_random_binary(n, rng):
    b = np.zeros(n)
    b[rng.choice(n, n // 2, replace=False)] = 1.0
    return b

def make_sturmian_binary(n, rng):
    alpha = rng.uniform(0.4, 0.6)
    offset = rng.uniform(0, 1)
    x = np.arange(n, dtype=float)
    vals = np.floor(alpha * (x + offset)) - np.floor(alpha * x)
    b = vals.astype(float)
    s = int(b.sum())
    if s > n // 2:
        one_idx = np.where(b == 1.0)[0]
        b[one_idx[:s - n // 2]] = 0.0
    elif s < n // 2:
        zero_idx = np.where(b == 0.0)[0]
        b[zero_idx[:n // 2 - s]] = 1.0
    return b

def make_modular_binary(n, rng):
    mod = rng.integers(n // 3, 2 * n // 3 + 1)
    b = np.zeros(n)
    b[np.arange(0, n, mod)] = 1.0
    b[b >= 1.0] = 1.0
    s = int(b.sum())
    if s > n // 2:
        one_idx = np.where(b == 1.0)[0]
        b[one_idx[:s - n // 2]] = 0.0
    elif s < n // 2:
        zero_idx = np.where(b == 0.0)[0]
        b[zero_idx[:n // 2 - s]] = 1.0
    return b

def eps_transform(b, eps):
    return eps + b * (1.0 - 2.0 * eps)

def resample_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def compute_grad_lse(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            cnt = n - s
            grad[:cnt] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:cnt]
        else:
            a = -s
            cnt = n - a
            grad[:cnt] += wk * (1.0 - h[a:])
            grad[a:] -= wk * h[:cnt]
    return grad * dx

def lse_value(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def pg_optimize(h, n, lam, max_iter=400):
    h = h.copy()
    f = lse_value(h, lam, n)
    alpha = 1.0
    stall = 0
    for it in range(max_iter):
        g = compute_grad_lse(h, lam, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        found = False
        a = alpha
        for _ in range(30):
            h_new = project_feasible(h - a * g, n)
            f_new = lse_value(h_new, lam, n)
            if f_new < f - 1e-15:
                found = True
                break
            a *= 0.5
        if not found:
            stall += 1
            alpha = max(a, 0.001)
            if stall > 20:
                break
            continue
        stall = 0
        h = h_new
        f = f_new
        alpha = min(a * 1.5, 10.0)
    return h

def active_descent(h, n, max_iter=300):
    h = h.copy()
    cur_val = task_exact(h)
    for it in range(max_iter):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        tol = max(1e-5, m * 1e-4)
        active_mask = corr >= m - tol
        if active_mask.sum() == 0:
            active_mask = corr >= m - 1e-3
        w = np.zeros(len(corr))
        w[active_mask] = 1.0 / active_mask.sum()
        grad = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            if s >= 0:
                cnt = n - s
                grad[:cnt] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:cnt]
            else:
                a = -s
                cnt = n - a
                grad[:cnt] += wk * (1.0 - h[a:])
                grad[a:] -= wk * h[:cnt]
        grad *= dx
        grad = grad - grad.mean()
        found = False
        a = 0.3
        for _ in range(15):
            h_new = project_feasible(h - a * grad, n)
            f_new = task_exact(h_new)
            if f_new < cur_val - 1e-14:
                found = True
                break
            a *= 0.5
        if not found:
            break
        h = h_new
        cur_val = f_new
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def time_left():
        return budget_s - 6.0 - (time.time() - t0)

    n0 = len(initial_h_values)
    h0 = project_feasible(np.array(initial_h_values, dtype=float), n0)
    best_h = h0.copy()
    best_c5 = task_exact(h0)

    fallback_h = np.full(64, 0.5)
    fallback_c5 = 0.5

    # Phase 1: Binary SA at multiple even resolutions
    sa_deadline = t0 + budget_s * 0.45
    n_candidates = [40, 42, 44, 46, 48, 50, 52, 54, 56, 58, 60, 64]
    n_candidates = [n for n in n_candidates if n <= n0 + 20]
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time.time() > sa_deadline:
            break
        remaining = sa_deadline - time.time()
        if remaining < 3:
            break
        # Time per n value: proportional to 1/sqrt(n)
        n_time = min(remaining * 0.6, 40 * (44.0 / n) ** 0.5)
        n_time = max(n_time, 2.0)

        n_starts = max(1, min(4, int(n_time / 4)))
        for si in range(n_starts):
            if time.time() > sa_deadline:
                break
            per_start = n_time / n_starts

            if si == 0:
                h_res = resample_to(best_h, len(best_h), n)
                b = (h_res >= 0.5).astype(float)
                s = int(b.sum())
                if s > n // 2:
                    oi = np.where(b == 1.0)[0]
                    b[oi[:s - n // 2]] = 0.0
                elif s < n // 2:
                    zi = np.where(b == 0.0)[0]
                    b[zi[:n // 2 - s]] = 1.0
            elif si == 1:
                b = make_sturmian_binary(n, rng)
            elif si == 2:
                b = make_modular_binary(n, rng)
            else:
                b = make_random_binary(n, rng)

            T0 = 1.5 + 0.3 * si
            b_best, raw_best = binary_sa(b, n, per_start, rng, T_start=T0)
            c5_bin = raw_best * 2.0 / n
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = b_best.copy()

    # Phase 2: Continuous polishing
    cont_deadline = t0 + budget_s * 0.82
    nb = len(best_h)

    # Try polynomial epsilon transforms
    for eps in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15]:
        if time.time() > cont_deadline - 20:
            break
        if np.all(np.isclose(best_h, 0) | np.isclose(best_h, 1)):
            h_start = eps_transform(best_h, eps)
        else:
            h_start = project_feasible(best_h + eps * rng.normal(0, 0.3, nb), nb)
        h_start = project_feasible(h_start, nb)

        sl = max(3, (cont_deadline - 20 - time.time()) / 7)
        h_opt = pg_optimize(h_start, nb, 30, max_iter=200)
        h_opt = pg_optimize(h_opt, nb, 200, max_iter=200)
        h_opt = pg_optimize(h_opt, nb, 1000, max_iter=200)
        h_opt = pg_optimize(h_opt, nb, 5000, max_iter=150)
        h_opt = project_feasible(h_opt, nb)
        c5 = task_exact(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Active descent
    if time.time() < cont_deadline - 5:
        nb = len(best_h)
        h_opt = active_descent(best_h, nb, max_iter=200)
        h_opt = project_feasible(h_opt, nb)
        c5 = task_exact(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 3: Multi-resolution continuous
    res_deadline = t0 + budget_s * 0.92
    for n_try in [44, 46, 48, 50, 52, 54, 56, 58, 60, 64, 70, 76]:
        if time.time() > res_deadline:
            break
        if n_try == len(best_h):
            continue
        h_res = resample_to(best_h, len(best_h), n_try)
        h_res = project_feasible(h_res, n_try)
        r_time = max(3, (res_deadline - time.time()) * 0.25)
        h_res = pg_optimize(h_res, n_try, 50, max_iter=150)
        h_res = pg_optimize(h_res, n_try, 500, max_iter=150)
        h_res = pg_optimize(h_res, n_try, 3000, max_iter=100)
        h_res = project_feasible(h_res, n_try)
        c5 = task_exact(h_res)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res.copy()

    # Phase 4: Final refinement with high lambda
    final_deadline = t0 + budget_s - 6
    nb = len(best_h)
    for lam in [200, 1000, 5000, 15000, 40000]:
        if time.time() > final_deadline - 2:
            break
        h_opt = pg_optimize(best_h, nb, lam, max_iter=300)
        h_opt = project_feasible(h_opt, nb)
        c5 = task_exact(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Active descent final
    if time.time() < final_deadline - 3:
        h_opt = active_descent(best_h, nb, max_iter=200)
        h_opt = project_feasible(h_opt, nb)
        c5 = task_exact(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = task_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    if not np.all(np.isfinite(best_h)) or not np.isfinite(best_c5):
        return fallback_h, fallback_c5, len(fallback_h)
    if best_c5 > 0.5:
        return fallback_h, fallback_c5, len(fallback_h)

    return best_h, best_c5, len(best_h)
```
