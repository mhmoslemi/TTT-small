    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = G @ h0 - F0 + margin
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = []
    for i in range(n):
        lb = max(0.0, h0[i] - radius)
        ub = min(1.0, h0[i] + radius)
        if lb > ub:
            lb = ub = h0[i]
        bounds.append((lb, ub))
    bounds.append((0.0, 2.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success or res.status == 0:
            h_new = res.x[:n]
            t_new = res.x[-1]
            return ensure_feasible(h_new), float(t_new), True
    except Exception:
        pass
    
    # Try cvxopt as fallback
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='cvxopt')
        if res.success or res.status == 0:
            h_new = res.x[:n]
            t_new = res.x[-1]
            return ensure_feasible(h_new), float(t_new), True
    except Exception:
        pass
    
    return None, None, False

def local_lp_polish(h_start, max_iter=200, init_radius=0.05, stop_time=None):
    """Iterative LP-based local polish."""
    h = ensure_feasible(h_start.copy())
    c = exact_c5(h)
    n = h.size
    radius = init_radius
    min_radius = 0.002
    max_radius = 0.15
    no_improve_count = 0
    
    for it in range(max_iter):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        
        h_new, t_new, ok = lp_minimax_step(h, radius)
        if not ok or h_new is None:
            radius = max(min_radius, radius * 0.5)
            no_improve_count += 1
            if no_improve_count > 5:
                break
            continue
        
        # Line search
        direction = h_new - h
        max_step = np.max(np.abs(direction))
        if max_step < 1e-10:
            no_improve_count += 1
            radius = max(min_radius, radius * 0.5)
            if no_improve_count > 5:
                break
            continue
        
        accepted = False
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05]:
            h_try = ensure_feasible(h + alpha * direction)
            c_try = exact_c5(h_try)
            if c_try < c - 1e-12:
                h = h_try
                c = c_try
                accepted = True
                break
        
        if accepted:
            no_improve_count = 0
            # Increase radius slightly on success
            if max_step >= 0.8 * radius:
                radius = min(max_radius, radius * 1.3)
        else:
            no_improve_count += 1
            radius = max(min_radius, radius * 0.5)
            if no_improve_count > 8:
                # Try a perturbation
                rng = np.random.default_rng(it + 42)
                h = ensure_feasible(h + rng.normal(0, 0.01, n))
                c = exact_c5(h)
                no_improve_count = 0
                radius = min_radius * 2
    
    return h, c

def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    if n_old == 0:
        return ensure_feasible(np.full(max(2, n_new), 0.5))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    return ensure_feasible(0.5 * (h + h[::-1]))

def get_parent():
    try:
        ph = globals().get('initial_h_values')
    except Exception:
        ph = None
    if ph is None:
        return None
    arr = np.asarray(ph, dtype=float).reshape(-1)
    if arr.size < 2 or not np.all(np.isfinite(arr)):
        return None
    return ensure_feasible(arr)

def make_random_candidate(base, n, mode, rng):
    n = int(max(24, n))
    if base is not None:
        b = base
        if b.size != n:
            b = resample_h(b, n)
        b = ensure_feasible(b)
    else:
        b = np.full(n, 0.5)
    
    if mode == 0:  # noise
        scale = float(rng.uniform(0.003, 0.08))
        h = b + rng.normal(0, scale, n)
    elif mode == 1:  # wave
        x = (np.arange(n) + 0.5) * (2.0 / n)
        k1 = int(rng.integers(1, 8))
        k2 = int(rng.integers(1, 12))
        a1 = float(rng.uniform(0.2, 0.45))
        a2 = float(rng.uniform(0.05, 0.2))
        p1 = float(rng.uniform(0, 2 * math.pi))
        p2 = float(rng.uniform(0, 2 * math.pi))
        h = 0.5 + a1 * np.sin(math.pi * k1 * x + p1) + a2 * np.sin(math.pi * k2 * x + p2)
    elif mode == 2:  # block
        nb = int(rng.integers(4, min(13, n)))
        cuts = np.sort(rng.uniform(0, 2, nb + 1))
        cuts[0] = 0
        cuts[-1] = 2
        vals = rng.uniform(0.1, 0.9, nb)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        idx = np.searchsorted(cuts, x, side='right') - 1
        idx = np.clip(idx, 0, nb - 1)
        h = vals[idx]
    elif mode == 3:  # threshold
        th = b + rng.normal(0, 0.02, n)
        p = float(rng.uniform(0.4, 0.6))
        hi = float(rng.uniform(0.7, 0.95))
        lo = float(rng.uniform(0.05, 0.3))
        h = np.where(th > p, hi, lo)
    else:  # binary-ish
        h = (rng.random(n) < 0.5).astype(float)
    
    return ensure_feasible(h)
