```python
import numpy as np
from scipy.optimize import linprog

def build_A(n, fixed):
    """Build constraint matrix for LP. fixed is the 'other' vector."""
    dx = 2.0 / n
    L = 2 * n - 1
    A = np.zeros((L, n))
    for k in range(L):
        shift = n - 1 - k
        if shift >= 0:
            m = n - shift
            if m > 0:
                A[k, :m] = fixed[shift:shift+m] * dx
        else:
            m = n + shift  # = n - (-shift)
            if m > 0:
                A[k, -shift:] = fixed[:m] * dx
    return A

def solve_lp(n, A, fixed_vals=None):
    """Solve the LP: min t s.t. A@x <= t, 0<=x<=1, sum(x)=n/2"""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    L = A.shape[0]
    A_ub = np.hstack([A, -np.ones((L, 1))])
    b_ub = np.zeros(L)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.clip(res.x[:n], 0.0, 1.0), res.x[-1]
    return None, None

def project_sum(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2}."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    n = len(h)
    if n <= 0:
        return math.inf
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

    h_best = h.copy()

        h_new, t_val = solve_lp(n, A)

            break
        y_new = project_sum(y_new, n)
        
        # Check
        h_candidate = project_sum(1.0 - y_new, n)
        val = exact_value(h_candidate)
        if val < t_best:
            t_best = val
            h_best = h_candidate.copy()
```

The subgradient of f(h) = max_k g_k(h) at h:
- Compute all g_k(h)
- Let K = {k : g_k(h) = max}

- The steepest descent subgradient is the one with minimum norm

For the steepest descent subgradient, I solve:

```python
def smooth_grad(h, T=0.01):
    """Compute smooth gradient of f(h) = max_k g_k(h) using softmax."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    
    # Compute all g_k
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Softmax weights
    max_g = np.max(corr)
    w = np.exp((corr - max_g) / T)
    w = w / np.sum(w)
    
    # Compute gradient: grad[j] = sum_k w_k * ∂g_k/∂h[j]
    # ∂g_k/∂h[j] = (1-h[j+s_k])*dx (if j+s_k valid) - h[j-s_k]*dx (if j-s_k valid)
    grad = np.zeros(n)
    for k in range(L):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        # First term: (1-h[j+s])*dx for valid j+s
        j_start1 = max(0, -s)
        j_end1 = min(n-1, n-1-s)
        if j_start1 <= j_end1:
            grad[j_start1:j_end1+1] += w[k] * (1.0 - h[j_start1+s:j_end1+s+1]) * dx
        # Second term: -h[j-s]*dx for valid j-s
        j_start2 = max(0, s)
        j_end2 = min(n-1, s+n-1)
        if j_start2 <= j_end2:
            grad[j_start2:j_end2+1] -= w[k] * h[j_start2-s:j_end2-s+1] * dx
    
    return grad, max_g + T * np.log(np.sum(np.exp((corr - max_g) / T)) / L)
```

```python
import time
import math
import numpy as np
from scipy.optimize import linprog


def build_A_for_h(n, y):
    """A[idx, i] = y[i-s]*dx where s=idx-(n-1), valid if 0<=i-s<n.
    Used when optimizing h with y fixed."""
    dx = 2.0 / n
    L = 2 * n - 1
    A = np.zeros((L, n))
    for idx in range(L):
        s = idx - (n - 1)
        i_start = max(0, s)
        i_end = min(n - 1, s + n - 1)
        if i_start <= i_end:
            js = i_start - s
            A[idx, i_start:i_end + 1] = y[js:js + (i_end - i_start + 1)] * dx
    return A


def build_A_for_y(n, h):
    """A[idx, j] = h[j+s]*dx where s=idx-(n-1), valid if 0<=j+s<n.
    Used when optimizing y with h fixed."""
    dx = 2.0 / n
    L = 2 * n - 1
    A = np.zeros((L, n))
    for idx in range(L):
        s = idx - (n - 1)
        j_start = max(0, -s)
        j_end = min(n - 1, n - 1 - s)
        if j_start <= j_end:
            hs = j_start + s
            A[idx, j_start:j_end + 1] = h[hs:hs + (j_end - j_start + 1)] * dx
    return A


def solve_lp(n, A):
    """Solve: min t s.t. A@x <= t, 0<=x<=1, sum(x)=n/2."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    L = A.shape[0]
    A_ub = np.hstack([A, -np.ones((L, 1))])
    b_ub = np.zeros(L)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.clip(res.x[:n], 0.0, 1.0), float(res.x[-1])
    return None, None


def project_sum(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2}."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(100):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def exact_value(h):
    n = len(h)
    if n <= 0:
        return math.inf
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))


def smooth_grad(h, T):
    """Smooth approximation gradient and value of f(h)=max_k g_k(h)."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_g = float(np.max(corr))
    z = (corr - max_g) / T
    ez = np.exp(z)
    ss = float(np.sum(ez))
    if ss <= 0 or not np.isfinite(ss):
        w = np.zeros(L)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = ez / ss
    F = max_g + T * (math.log(ss) - math.log(float(L)))
    grad = np.zeros(n)
    for k in range(L):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        # First term: (1-h[j+s])*dx for valid j+s in [0,n-1]
        j1s = max(0, -s)
        j1e = min(n - 1, n - 1 - s)
        if j1s <= j1e:
            grad[j1s:j1e + 1] += w[k] * (1.0 - h[j1s + s:j1e + s + 1]) * dx
        # Second term: -h[j-s]*dx for valid j-s in [0,n-1]
        j2s = max(0, s)
        j2e = min(n - 1, s + n - 1)
        if j2s <= j2e:
            grad[j2s:j2e + 1] -= w[k] * h[j2s - s:j2e - s + 1] * dx
    return F, grad


def lp_cycle(h0, max_iters=500, deadline=None, need_s=15.0):
    """Alternating LP linearization iteration."""
    n = len(h0)
    h = project_sum(h0.copy(), n)
    h_best = h.copy()
    v_best = exact_value(h)
    no_improve = 0
    stall_limit = 30

    for it in range(max_iters):
        if deadline is not None and (float(deadline) - time.monotonic()) < need_s:
            break

        if it % 2 == 0:
            y = 1.0 - h
            A = build_A_for_h(n, y)
            h_new, _ = solve_lp(n, A)
            if h_new is None:
                break
            h_new = project_sum(h_new, n)
        else:
            A = build_A_for_y(n, h)
            y_new, _ = solve_lp(n, A)
            if y_new is None:
                break
            y_new = project_sum(y_new, n)
            h_new = project_sum(1.0 - y_new, n)

        v = exact_value(h_new)
        if v < v_best - 1e-14:
            v_best = v
            h_best = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if np.max(np.abs(h_new - h)) < 1e-13:
            h = h_new
            break

        if no_improve > stall_limit:
            break

        # Damping
        alpha = 1.0 if no_improve == 0 else (0.7 if no_improve < 10 else 0.4)
        h = (1.0 - alpha) * h + alpha * h_new
        h = project_sum(h, n)

    return h_best, v_best


def subgrad_polish(h0, deadline, need_s=10.0, max_iters=300):
    """Smooth subgradient descent with backtracking."""
    n = len(h0)
    h = project_sum(h0.copy(), n)
    h_best = h.copy()
    v_best = exact_value(h)
    T = 0.02
    lr = 0.05
    stall = 0

    for it in range(max_iters):
        if deadline is not None and (float(deadline) - time.monotonic()) < need_s:
            break

        F, g = smooth_grad(h, T)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-10:
            break

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(15):
            h_trial = project_sum(h - alpha * g, n)
            v_trial = exact_value(h_trial)
            if v_trial < v_best - 1e-14:
                h = h_trial
                h_best = h.copy()
                v_best = v_trial
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break

        if improved:
            lr = min(0.5, alpha * 1.3)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 25 or lr < 1e-9:
                break

        # Anneal temperature
        T *= 0.97
        if T < 0.002:
            T = 0.002

    return h_best, v_best


def resample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_sum(y, n_new)


def random_init(n, rng, mode=0):
    if mode == 0:
        h = np.full(n, 0.5)
        h += 0.3 * rng.standard_normal(n)
    elif mode == 1:
        h = np.zeros(n)
        k = n // 2
        idx = rng.choice(n, k, replace=False)
        h[idx] = 1.0
        h += 0.05 * rng.standard_normal(n)
    else:
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        h = 0.5 + 0.35 * np.sin(2 * t + rng.uniform(0, 2 * np.pi))
        h += 0.03 * rng.standard_normal(n)
    return project_sum(h, n)

    # Phase 5: Final subgradient polish on best
    if deadline - time.monotonic() > 25:
        h_sg2, v_sg2 = subgrad_polish(best_h, deadline, need_s=15.0, max_iters=300)
        if v_sg2 < best_v:
            best_v = v_sg2
            best_h = h_sg2.copy()

    # Phase 6: One more LP pass
    if deadline - time.monotonic() > 30:
        h_lp3, v_lp3 = lp_cycle(best_h, max_iters=200, deadline=deadline, need_s=15.0)
        if v_lp3 < best_v:
            best_v = v_lp3
            best_h = h_lp3.copy()

    # Phase 7: Try a few random restarts if time allows
    if deadline - time.monotonic() > 100:
        for attempt in range(3):
            nn = n_base
            h_rand = random_init(nn, rng, attempt)
            h_rl, v_rl = lp_cycle(h_rand, max_iters=150, deadline=deadline, need_s=50.0)
            if v_rl < best_v:
                best_v = v_rl
                best_h = h_rl.copy()
            if deadline - time.monotonic() < 60:
                break

    # Final cleanup
    best_h = project_sum(best_h, len(best_h))
    best_v = exact_value(best_h)

    return best_h, float(best_v), int(best_h.size)
```
