import numpy as np
import time

def project_feasible(x, n):
    """Project x onto {h in [0,1]^n : sum(h) = n/2} using bisection."""
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny residual
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        interior = (result > 1e-8) & (result < 1 - 1e-8)
        if interior.sum() > 0:
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def evaluate_c5(h, n):
    """Compute C5 = max_k corr[k] * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def lse_value(h, lam, n):
    """Log-sum-exp approximation of max(corr*dx)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def lse_grad(h, lam, n):

def pg_optimize(h, n, lam, max_iter=500, alpha_init=1.0):
    """Projected gradient descent with LSE smoothing."""
    h = h.copy()
    f = lse_value(h, lam, n)
    alpha = alpha_init
    stall = 0
    for it in range(max_iter):
        g = lse_grad(h, lam, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        found = False
        a = alpha
        for _ in range(30):
            h_new = project_feasible(h - a * g, n)
            f_new = lse_value(h_new, lam, n)
            if f_new < f - 1e-15:
                found = True
                break
            a *= 0.5
        if not found:
            stall += 1
            alpha = max(a, 0.001)
            if stall > 20:
                break
            continue
        stall = 0
        h = h_new
        f = f_new
        alpha = min(a * 1.5, 10.0)
    return h

def pairwise_cd(h, n, max_sweeps=100, t_check=None, t_start=None):
    """Exact pairwise coordinate descent.
    For each pair (i,j), try to minimize the exact C5 by changing h[i]+=d, h[j]-=d.
    """
    h = h.copy()
    current_c5, _ = evaluate_c5(h, n)
    
    for sweep in range(max_sweeps):
        if t_check is not None and t_start is not None:
            if time.time() - t_start > t_check:
                break
        
        improved = False
        # Try all pairs (or a random subset for speed)
        pairs = np.random.permutation(n * (n-1) // 2)
        # Actually for n=45, that's ~1000 pairs, each with a line search
        # Let's do a random subset per sweep
        n_pairs = min(500, n * (n-1) // 2)
        
        for _ in range(n_pairs):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            
            # Line search: find optimal d such that h[i]+d, h[j]-d stay in [0,1]
            # d in [max(-h[i], h[j]-1), min(1-h[i], h[j])]
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            
            # Try a few values of d
            # First try the midpoint and perpendicularly
            best_d = 0.0
            best_val = current_c5
            
            # Coarse grid
            for frac in np.linspace(0, 1, 11):
                d_test = d_lo + frac * (d_hi - d_lo)
                if abs(d_test) < 1e-12:
                    continue
                h[i] += d_test
                h[j] -= d_test
                val, _ = evaluate_c5(h, n)
                h[i] -= d_test
                h[j] += d_test
                if val < best_val:
                    best_val = val
                    best_d = d_test
            
            if best_d != 0.0 and best_val < current_c5 - 1e-14:
                h[i] += best_d
                h[j] -= best_d
                current_c5 = best_val
                improved = True
        
        if not improved:
            break
    
    return h, current_c5

def binary_local_search(h_binary, n, max_iter=5000, t_check=None, t_start=None, temperature=0.01):
    """Local search on binary sequence: flip pairs of bits."""
    h = h_binary.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    current_c5, _ = evaluate_c5(h, n)
    
    for it in range(max_iter):
        if t_check is not None and t_start is not None:
            if time.time() - t_start > t_check:
                break
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Flip: h[i] 1->0, h[j] 0->1
        h[i] = 0.0
        h[j] = 1.0
        new_c5, _ = evaluate_c5(h, n)
        
        if new_c5 < current_c5 or (np.random.random() < np.exp(-(new_c5 - current_c5) / temperature)):
            current_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h, current_c5

def make_binary_from_continuous(h, n):
    """Convert continuous h to binary by thresholding with exact count."""
    threshold = np.sort(h)[n // 2]
    h_bin = (h >= threshold).astype(float)
    # Fix count if needed
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Randomly set some 1s to 0s (from the lowest-valued ones)
        one_indices = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[one_indices])
        for idx in order[:n_ones - target]:
            h_bin[one_indices[idx]] = 0.0
    elif n_ones < target:
        zero_indices = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zero_indices])
        for idx in order[:target - n_ones]:
            h_bin[zero_indices[idx]] = 1.0
    return h_bin

def resample_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

            h[j] = 0.0

import numpy as np
import time

def project_feasible(x, n):
    target = n / 2.0
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        interior = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = int(interior.sum())
        if cnt > 0:
            result[interior] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def corr_grad_single_lag(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            j1 = np.arange(s, n)
            grad[j1] += (1.0 - h[j1 - s])
            j2 = np.arange(0, n - s)
            grad[j2] -= h[j2 + s]
    else:
        t = -s
        j1 = np.arange(0, n - t)
        grad[j1] += (1.0 - h[j1 + t])
        j2 = np.arange(t, n)
        grad[j2] -= h[j2 - t]
    return grad * (2.0 / n)

def subgrad_optimize(h, n, max_iter=800, alpha_init=0.15):
    h = h.copy()
    c5, _ = evaluate_c5(h, n)
    alpha = alpha_init
    prev_g = None
    for it in range(max_iter):
        _, corr = evaluate_c5(h, n)
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-9 / (2.0 / n))[0]
        g = np.zeros(n)
        for k_idx in active:
            s = k_idx - (n - 1)
            g += corr_grad_single_lag(h, s, n)
        g /= len(active)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        # BB step size if possible
        if prev_g is not None:
            d = g - prev_g
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                h_new_tests = h - alpha * g
                alpha_bb = alpha * 0.8 + 0.5  # fallback
        found = False
        a = alpha
        for _ in range(30):
            h_new = project_feasible(h - a * g, n)
            c5_new, _ = evaluate_c5(h_new, n)
            if c5_new < c5 - 1e-14:
                found = True
                break
            a *= 0.5
        if not found:
            alpha = max(a, 0.002)
            if alpha < 0.003:
                break
            continue
        prev_g = g.copy()
        h = h_new
        c5 = c5_new
        alpha = min(a * 1.3, 2.0)
    return h, c5

def lse_optimize(h, n, lam, max_iter=300, alpha_init=0.5):
    h = h.copy()
    dx = 2.0 / n
    def lse_f(hh):
        c = np.correlate(hh, 1.0 - hh, mode="full") * dx
        m = np.max(c)
        sc = lam * (c - m)
        ms = np.max(sc)
        if ms > 400:
            return m + ms / lam
        return m + (ms + np.log(np.mean(np.exp(sc - ms)))) / lam
    def lse_g(hh):
        c = np.correlate(hh, 1.0 - hh, mode="full") * dx
        m = np.max(c)
        sc = lam * (c - m)
        ms = np.max(sc)
        if ms > 400:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            ev = np.exp(sc - ms)
            w = ev / ev.sum()
        g = np.zeros(n)
        for k_idx in range(len(c)):
            wk = w[k_idx]
            if wk < 1e-14:
                continue
            s = k_idx - (n - 1)
            g += wk * corr_grad_single_lag(hh, s, n)
        return g
    f = lse_f(h)
    g = lse_g(h)
    alpha = alpha_init
    stall = 0
    for it in range(max_iter):
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        found = False
        a = alpha
        for _ in range(25):
            h_new = project_feasible(h - a * g, n)
            f_new = lse_f(h_new)
            if f_new < f - 1e-14:
                found = True
                break
            a *= 0.5
        if not found:
            stall += 1
            alpha = max(a, 0.001)
            if stall > 15:
                break
            continue
        stall = 0
        h = h_new
        f = f_new
        g = lse_g(h)
        alpha = min(a * 1.4, 5.0)
    return h

def pairwise_cd(h, n, rng, max_sweeps=100, deadline=None, n_pairs_per_sweep=300):
    h = h.copy()
    c5, _ = evaluate_c5(h, n)
    fracs = np.array([0.15, 0.3, 0.5, 0.7, 0.85])
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(n_pairs_per_sweep):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            best_d = 0.0
            best_val = c5
            for frac in fracs:
                d_test = d_lo + frac * (d_hi - d_lo)
                if abs(d_test) < 1e-13:
                    continue
                h[i] += d_test
                h[j] -= d_test
                val, _ = evaluate_c5(h, n)
                if val < best_val:
                    best_val = val
                    best_d = d_test
                h[i] -= d_test
                h[j] += d_test
            if best_d != 0.0 and best_val < c5 - 1e-14:
                h[i] += best_d
                h[j] -= best_d
                c5 = best_val
                improved = True
        if not improved:
            break
    return h, c5

def binary_anneal(h_bin, n, rng, max_iter=4000, deadline=None, temp_init=0.01):
    h = h_bin.copy()
    c5, _ = evaluate_c5(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best_h = h.copy()
    best_c5 = c5
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        t_curr = temp_init * (0.9995 ** it)
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5, _ = evaluate_c5(h, n)
        delta = new_c5 - c5
        if delta < 0 or (t_curr > 1e-10 and rng.random() < np.exp(-delta / t_curr)):
            c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def make_binary(h, n, rng):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def resample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def make_structured_init(n, rng, pattern='alt'):
    h = np.zeros(n)
    if pattern == 'alt':
        h[::2] = 1.0
    elif pattern == 'blocks':
        block = max(2, n // 10)
        for i in range(0, n, block):
            h[i:i+block//2] = 1.0
    else:
        h[rng.choice(n, size=n//2, replace=False)] = 1.0
    # Verify count
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h == 1.0)[0]
        to_flip = rng.choice(one_idx, size=n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h == 0.0)[0]
        to_flip = rng.choice(zero_idx, size=target - n_ones, replace=False)
        h[to_flip] = 1.0
    return h
