```python
import time
import math
import numpy as np
from scipy.optimize import linprog

def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n < 2:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0:
        return float('inf')
    return float(np.max(vals))

def compute_correlations(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradients(h):
    n = h.size
    dx = 2.0 / float(n)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    for k in range(L):
        shift = n - 1 - k
        start = max(0, -shift)
        end = min(n - 1, n - 1 - shift)
        if start > end:
            continue
        i = np.arange(start, end + 1)
        j = i + shift
        if shift == 0:
            G[k, i] = dx * (1.0 - 2.0 * h[i])
        else:
            G[k, i] += dx * (1.0 - h[j])
            G[k, j] += -dx * h[i]
    return G

def lp_minimax_step(h0, radius, margin=1e-10):
    h0 = np.asarray(h0, dtype=float).reshape(-1)
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * float(n)
    dx = 2.0 / float(n)
    
    c0 = compute_correlations(h0)
    G = compute_gradients(h0)
    
    # LP: minimize t
    # s.t. G[k,:] @ h - t <= G[k,:] @ h0 - c0[k] + margin
    #      sum(h) = n/2
    #      lb[i] <= h[i] <= ub[i]
    #      -1 <= t <= 2
    
    A_ub = np.zeros((L, n + 1), dtype=float)
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c0 + margin
    
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = []
    for i in range(n):
        lb = max(0.0, h0[i] - radius)
        ub = min(1.0, h0[i] + radius)
        if lb > ub:
            lb = ub = h0[i]
        bounds.append((lb, ub))
    bounds.append((-1.0, 2.0))  # t
    
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            h_new = res.x[:n]
            if np.all(np.isfinite(h_new)):
                return h_new
    except Exception:
        pass
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs-ds')
        if res.status == 0 and res.x is not None:
            h_new = res.x[:n]
            if np.all(np.isfinite(h_new)):
                return h_new
    except Exception:
        pass
    
    return None

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    S = 0.5 * float(n)
    if n == 0:
        return np.array([0.5, 0.5])
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)
    np.clip(h, 0.0, 1.0, out=h)
    # Adjust sum to exactly S
    for _ in range(10 * n + 50):
        r = S - float(np.sum(h))
        if abs(r) < 1e-12:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            if slack[idx] <= 0:
                break
            h[idx] += min(r, slack[idx])
        else:
            idx = int(np.argmin(h))
            if h[idx] <= 0:
                break
            h[idx] -= min(-r, h[idx])
    np.clip(h, 0.0, 1.0, out=h)
    return h

def local_search(h0, max_iter=200, init_radius=0.02, max_radius=0.06,
                 min_radius=0.0005, stop_time=None):
    h = ensure_feasible(h0)
    c = exact_c5(h)
    n = h.size
    radius = init_radius
    stall = 0
    best_local = h.copy()
    best_local_c = c
    
    for it in range(max_iter):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        if radius < min_radius:
            break
        
        h_lp = lp_minimax_step(h, radius)
        if h_lp is None:
            radius *= 0.5
            stall += 1
            if stall > 10:
                break
            continue
        
        h_lp = ensure_feasible(h_lp)
        delta = h_lp - h
        if np.max(np.abs(delta)) < 1e-9:
            radius *= 0.5
            stall += 1
            continue
        
        # Line search
        alphas = [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03]
        best_alpha = None
        best_c_trial = c
        best_h_trial = h.copy()
        
        for alpha in alphas:
            h_trial = ensure_feasible(h + alpha * delta)
            c_trial = exact_c5(h_trial)
            if c_trial < best_c_trial - 1e-12:
                best_c_trial = c_trial
                best_alpha = alpha
                best_h_trial = h_trial
        
        if best_alpha is not None:
            h = best_h_trial
            c = best_c_trial
            if c < best_local_c:
                best_local_c = c
                best_local = h.copy()
            stall = 0
            # Increase radius slightly
            if best_alpha >= 0.8:
                radius = min(max_radius, radius * 1.3)
            else:
                radius = min(max_radius, radius * 1.1)
        else:
            radius *= 0.6
            stall += 1
            if stall > 15:
                break
    
    return best_local, best_local_c

def symmetrize(h):
    return ensure_feasible(0.5 * (h + h[::-1]))

def resample(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = max(2, int(n_new))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

def make_random_candidate(n, mode, rng, base=None):
    n = max(2, int(n))
    if base is not None:
        b = np.asarray(base, dtype=float).reshape(-1)
        if b.size != n:
            b = resample(b, n)
        b = ensure_feasible(b)
    else:
        b = np.full(n, 0.5, dtype=float)
    
    if mode == 'noise':
        scale = float(rng.uniform(0.005, 0.06))
        h = b + rng.normal(0, scale, n)
    elif mode == 'wave':
        x = (np.arange(n) + 0.5) * (2.0 / n)
        k1 = int(rng.integers(1, 6))
        k2 = int(rng.integers(1, 10))
        a1 = float(rng.uniform(0.15, 0.4))
        a2 = float(rng.uniform(0.03, 0.2))
        ph1 = float(rng.uniform(0, 2 * math.pi))
        ph2 = float(rng.uniform(0, 2 * math.pi))
        h = 0.5 + a1 * np.sin(math.pi * k1 * x + ph1) + a2 * np.sin(math.pi * k2 * x + ph2)
    elif mode == 'block':
        nb = int(rng.integers(3, 12))
        cuts = np.sort(rng.uniform(0, 2, nb - 1))
        cuts = np.concatenate([[0], cuts, [2]])
        vals = rng.uniform(0.1, 0.9, nb)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        idx = np.searchsorted(cuts, x, side='right') - 1
        idx = np.clip(idx, 0, nb - 1)
        h = vals[idx]
    elif mode == 'threshold':
        th = b + rng.normal(0, 0.02, n)
        p = float(rng.uniform(0.35, 0.65))
        hi = float(rng.uniform(0.6, 0.95))
        lo = float(rng.uniform(0.05, 0.4))
        h = np.where(th > p, hi, lo)
    else:
        h = b.copy()
    
    return ensure_feasible(h)
```

def compute_gradients(h):

    L = 2 * n - 1

        else:
            G[k, i] += dx * (1.0 - h[j])

    return G
