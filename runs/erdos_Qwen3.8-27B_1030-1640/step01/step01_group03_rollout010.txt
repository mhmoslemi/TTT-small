def compute_corr(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_grad_for_lag(h, k, dx):
    """Compute gradient of correlation at lag k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
        for i in range(n - k):
            grad[i] += dx * (1.0 - h[i + k])
            grad[i + k] += -dx * h[i]
    else:
        l = -k
        for i in range(n - l):
            grad[i + l] += dx * (1.0 - h[i])
            grad[i] += -dx * h[i + l]
    return grad

def compute_soft_grad(h, theta):
    """Compute soft-max weighted gradient."""
    n = len(h)
    dx = 2.0 / n
    G = compute_corr(h)
    Gmax = np.max(G)
    exp_G = np.exp((G - Gmax) / theta)
    w = exp_G / np.sum(exp_G)
    offset = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - offset
        g = compute_grad_for_lag(h, k, dx)
        grad += wm * g
    return Gmax + theta * np.log(np.sum(exp_G)), grad

def nesterov_smooth(h, theta_schedule, max_iters=200):
    """Smooth max descent with Nesterov momentum."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    
    for theta in theta_schedule:
        mu = 0.01 / (theta + 1e-8)  # learning rate
        for it in range(max_iters):
            # Nesterov step
            y = h_cur + beta * v
            obj, grad = compute_soft_grad(y, theta)
            if not np.isfinite(obj):
                break
            # Line search
            alpha = mu
            accepted = False
            for _ in range(30):
                h_new = h_cur - alpha * grad
                h_proj = project_h(h_new, target)
                c_t = evaluate_C(h_proj)
                if c_t < best_c + theta * 0.1:
                    v = -alpha * grad + beta * v
                    h_cur = h_proj
                    best_c = min(best_c, c_t)
                    if c_t < best_c + 1e-15:
                        best_h = h_cur.copy()
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-16:
                    break
            if not accepted:
                break
    return best_h, best_c

import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def dA_dh(h, i):
    """Compute d(corr_k)/d(h[i]) for all lags k, returned as vector of length 2n-1."""
    n = len(h)
    dx = 2.0 / n
    result = np.zeros(2 * n - 1)
    offset = n - 1
    # First factor: (1-h[i+k]) for k where 0 <= i+k < n
    k_lo = -i
    k_hi = n - 1 - i
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        vals = 1.0 - h[i + ks]
        result[ks + offset] += dx * vals
    # Second factor: -h[i-k] for k where 0 <= i-k < n
    k_lo2 = i - n + 1
    k_hi2 = i
    if k_lo2 <= k_hi2:
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        vals2 = -h[i - ks2]
        result[ks2 + offset] += dx * vals2
    return result

def find_optimal_t(A0, slopes, T):
    """Find t in [0,T] minimizing max_k(A0[k] + slopes[k]*t)."""
    # Check endpoints first
    f0 = np.max(A0)
    fT = np.max(A0 + slopes * T)
    if fT >= f0:
        # Minimum might be at 0 or in between
        # Check slope at 0
        v0 = A0
        vmax0 = np.max(v0)
        active0 = v0 >= vmax0 - 1e-14
        if np.min(slopes[active0]) > 1e-14:
            return 0.0  # increasing at 0, min at 0
        # Check slope at T
        vT = A0 + slopes * T
        vmaxT = np.max(vT)
        activeT = vT >= vmaxT - 1e-14
        if np.max(slopes[activeT]) < -1e-14:
            return T  # still decreasing at T, min at T
    else:
        # f(T) < f(0), minimum is likely interior or at T
        vT = A0 + slopes * T
        vmaxT = np.max(vT)
        activeT = vT >= vmaxT - 1e-14
        if np.max(slopes[activeT]) < -1e-14:
            return T
    
    # Binary search
    lo, hi = 0.0, T
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = A0 + slopes * mid
        vmax = np.max(vals)
        active = vals >= vmax - 1e-14
        s_min = np.min(slopes[active])
        s_max = np.max(slopes[active])
        if s_min > 0:
            hi = mid
        elif s_max < 0:
            lo = mid
        else:
            break
    t_opt = (lo + hi) / 2.0
    return t_opt

def pair_sweep(h, max_sweeps=200, t0=None, time_limit=None):
    """Exhaustive pair sweep with exact 1D optimization."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_c = evaluate_C(h_cur)
    dx = 2.0 / n

        if t0 is not None and time_limit is not None:

        improved = False
        A0 = compute_corr(h_cur)
        
        for i in range(n):
            for j in range(i + 1, n):
                # Direction 1: h[i] -= t, h[j] += t
                T1 = min(h_cur[i], 1.0 - h_cur[j])
                if T1 > 1e-10:
                    gi = dA_dh(h_cur, i)
                    gj = dA_dh(h_cur, j)
                    slopes = gj - gi
                    t_opt = find_optimal_t(A0, slopes, T1)
                    if t_opt > 1e-8:
                        h_new = h_cur.copy()
                        h_new[i] -= t_opt
                        h_new[j] += t_opt
                        c_new = evaluate_C(h_new)
                        if c_new < best_c - 1e-11:
                            h_cur = h_new
                            best_c = c_new
                            improved = True
                            A0 = compute_corr(h_cur)
                    # Direction 2: h[i] += t, h[j] -= t
                    T2 = min(1.0 - h_cur[i], h_cur[j])
                    if T2 > 1e-10 and abs(T1 - T2) > 1e-10:

                                h_cur = h_new2
                                best_c = c_new2
                                improved = True
                                A0 = compute_corr(h_cur)
        if not improved:

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def dA_dh(h, i):
    n = len(h)
    dx = 2.0 / n
    result = np.zeros(2 * n - 1)
    offset = n - 1
    k_lo = -i
    k_hi = n - 1 - i
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        result[ks + offset] += dx * (1.0 - h[i + ks])
    k_lo2 = i - n + 1
    k_hi2 = i
    if k_lo2 <= k_hi2:
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        result[ks2 + offset] += dx * (-h[i - ks2])
    return result

def precompute_all_grads(h):
    n = len(h)
    grads = np.zeros((n, 2 * n - 1))
    for i in range(n):
        grads[i] = dA_dh(h, i)
    return grads

def find_optimal_t(A0, slopes, T):
    if T < 1e-12:
        return 0.0
    f0 = float(np.max(A0))
    fT = float(np.max(A0 + slopes * T))
    if fT >= f0:
        vmax0 = np.max(A0)
        active0 = A0 >= vmax0 - 1e-14
        if len(active0) > 0 and np.min(slopes[active0]) > 1e-12:
            return 0.0
    vT = A0 + slopes * T
    vmaxT = np.max(vT)
    activeT = vT >= vmaxT - 1e-14
    if len(activeT) > 0 and np.max(slopes[activeT]) < -1e-12:
        return T
    lo = 0.0
    hi = T
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = A0 + slopes * mid
        vmax = float(np.max(vals))
        active = vals >= vmax - 1e-13
        if len(active) == 0:
            active = vals >= vmax - 1e-10
        s_min = float(np.min(slopes[active]))
        s_max = float(np.max(slopes[active]))
        if s_min > 0:
            hi = mid
        elif s_max < 0:
            lo = mid
        else:
            break
    t_opt = (lo + hi) * 0.5
    return t_opt

def pair_sweep(h, max_sweeps, t0, time_limit):
    n = len(h)
    h_cur = h.copy()
    best_c = evaluate_C(h_cur)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit - 20:
            break
        improved = False
        A0 = compute_corr(h_cur)
        all_grads = precompute_all_grads(h_cur)
        for i in range(n):
            for j in range(i + 1, n):
                T1 = min(h_cur[i], 1.0 - h_cur[j])
                if T1 > 1e-10:
                    slopes = all_grads[j] - all_grads[i]
                    t_opt = find_optimal_t(A0, slopes, T1)
                    if t_opt > 1e-9:
                        h_new = h_cur.copy()
                        h_new[i] -= t_opt
                        h_new[j] += t_opt
                        c_new = evaluate_C(h_new)
                        if c_new < best_c - 1e-11:
                            h_cur = h_new
                            best_c = c_new
                            improved = True
                            A0 = compute_corr(h_cur)
                            all_grads = precompute_all_grads(h_cur)
                T2 = min(1.0 - h_cur[i], h_cur[j])
                if T2 > 1e-10:
                    slopes2 = all_grads[i] - all_grads[j]
                    t_opt2 = find_optimal_t(A0, slopes2, T2)
                    if t_opt2 > 1e-9:
                        h_new = h_cur.copy()
                        h_new[i] += t_opt2
                        h_new[j] -= t_opt2
                        c_new = evaluate_C(h_new)
                        if c_new < best_c - 1e-11:
                            h_cur = h_new
                            best_c = c_new
                            improved = True
                            A0 = compute_corr(h_cur)
                            all_grads = precompute_all_grads(h_cur)
        if not improved:
            break
    return h_cur, best_c

def compute_soft_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    G = np.correlate(h, 1.0 - h, mode="full") * dx
    Gmax = float(np.max(G))
    exp_G = np.exp((G - Gmax) / theta)
    w = exp_G / np.sum(exp_G)
    offset = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - offset
        if k >= 0:
            grad[:n - k] += wm * dx * (1.0 - h[k:])
            grad[k:] += wm * dx * (-h[:n - k])
        else:
            l = -k
            grad[l:] += wm * dx * (1.0 - h[:n - l])
            grad[:n - l] += wm * dx * (-h[l:])
    soft_obj = Gmax + theta * np.log(float(np.sum(exp_G)))
    return soft_obj, grad

def smooth_descent(h, theta_schedule, max_iters, t0, time_limit):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    v = np.zeros(n)
    beta = 0.7
    for theta in theta_schedule:
        if time.time() - t0 > time_limit - 30:
            break
        alpha_init = 0.05 / (theta + 1e-8)
        for it in range(max_iters):
            if time.time() - t0 > time_limit - 30:
                break
            y = h_cur + beta * v
            y = np.clip(y, 0.0, 1.0)
            obj, grad = compute_soft_grad(y, theta)
            if not np.isfinite(obj):
                break
            gnorm = float(np.max(np.abs(grad)))
            if gnorm < 1e-14:
                beta *= 0.5
                continue
            alpha = alpha_init / (gnorm + 1e-10)
            accepted = False
            for _ in range(30):
                h_new = h_cur - alpha * grad
                h_proj = project_h(h_new, target)
                c_t = evaluate_C(h_proj)
                if c_t <= best_c + theta * 0.05:
                    v_new = -alpha * grad + beta * v
                    h_cur = h_proj
                    if c_t < best_c - 1e-13:
                        best_c = c_t
                        best_h = h_cur.copy()
                    v = v_new
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-16:
                    break
            if not accepted:
                beta *= 0.8
                if beta < 0.1:
                    break
    return best_h, best_c

def interpolate_h(h, new_n):
    n = len(h)
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, new_n / 2.0)

def perturb_h(h, rng, eps):
    n = len(h)
    h_p = h.copy()
    mask = (h_p > 0.05) & (h_p < 0.95)
    if np.sum(mask) > 0:
        h_p[mask] += eps * (rng.random(np.sum(mask)) - 0.5)
    return project_h(h_p, n / 2.0)

def block_sweep(h, block_sizes, t0, time_limit, max_sweeps=20):
    n = len(h)
    h_cur = h.copy()
    best_c = evaluate_C(h_cur)
    for bsize in block_sizes:
        if time.time() - t0 > time_limit - 30:
            break
        for sweep in range(max_sweeps):
            if time.time() - t0 > time_limit - 30:
                break
            improved = False
            A0 = compute_corr(h_cur)
            for i in range(0, n - bsize + 1, bsize):
                for j in range(i + bsize, n - bsize + 1, bsize):
                    block_i = np.arange(i, i + bsize)
                    block_j = np.arange(j, j + bsize)
                    T1 = min(float(np.min(h_cur[block_i])), 1.0 - float(np.max(h_cur[block_j])))
                    if T1 > 1e-9:
                        delta = np.zeros(n)
                        delta[block_i] = -1.0 / bsize
                        delta[block_j] = 1.0 / bsize
                        slopes = np.zeros(2 * n - 1)
                        for idx in range(n):
                            slopes += delta[idx] * dA_dh(h_cur, idx)
                        t_opt = find_optimal_t(A0, slopes, T1)
                        if t_opt > 1e-8:
                            h_new = h_cur + t_opt * delta
                            h_new = project_h(h_new, n / 2.0)
                            c_new = evaluate_C(h_new)
                            if c_new < best_c - 1e-11:
                                h_cur = h_new
                                best_c = c_new
                                improved = True
                                A0 = compute_corr(h_cur)
            if not improved:
                break
    return h_cur, best_c
