def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n < 2:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0 or not np.all(np.isfinite(vals)):
        return float('inf')
    return float(np.max(vals))

def exact_c5_components(h):
    """Return all correlation components (not just the max)."""
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def adjust_sum_exact(h, S):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    if n == 0:
        return h
    for _ in range(n + 50):
        r = float(S) - float(np.sum(h))
        if abs(r) <= 1e-13:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            if slack[idx] <= 0:
                break
            delta = min(r, float(slack[idx]))
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmin(h))
            if h[idx] <= 0:
                break
            delta = min(-r, float(h[idx]))
            if delta <= 0:
                break
            h[idx] -= delta
    np.clip(h, 0.0, 1.0, out=h)
    # Final fine adjustment
    for _ in range(1000):
        r = float(S) - float(np.sum(h))
        if abs(r) <= 1e-15:
            break
        if r > 0:
            idxs = np.where(h < 1.0 - 1e-14)[0]
            if idxs.size == 0:

            idx = int(idxs[0])

        else:

            if idxs.size == 0:
                break
            idx = int(idxs[0])

    return h

def project_box_sum(v):
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * float(n)
    if not np.all(np.isfinite(v)):

    # Bisection on lambda: sum(clip(v - lambda, 0, 1)) = S
    lo = float(v.min()) - 1.0 - 1e-12
    hi = float(v.max()) + 1.0 + 1e-12
    for _ in range(200):

            lo = mid
        else:
            hi = mid
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum_exact(p, S)

def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = max(2, int(n_new))

    y = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))

def correlation_gradients(h):
    """Compute gradients of all correlation components.
    Returns G of shape (2n-1, n) where G[k,j] = partial c[k] / partial h[j].
    """
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    for k in range(L):
        shift = k - (n - 1)
        for j in range(n):
            g = 0.0
            # Term 1: [1 - h[j-shift]] if 0 <= j-shift < n
            if 0 <= j - shift < n:
                g += (1.0 - h[j - shift])
            # Term 2: -h[j+shift] if 0 <= j+shift < n
            if 0 <= j + shift < n:
                g -= h[j + shift]
            G[k, j] = dx * g
    return G

    if c < best_c - 1e-10:
        return h, c, n
    if abs(c - best_c) <= 1e-8 and n < best_n:
        return h, c, n
    return best, best_c, best_n

def lp_minimax_step(h0, radius, max_time=None):
    """One LP minimax step. Returns (h_new, success)."""
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * float(n)
    
    # Compute correlation values and gradients
    c_vals = exact_c5_components(h0)  # length L
    G = correlation_gradients(h0)  # shape (L, n)
    
    # Build LP:
    # Variables: x = [h_0, ..., h_{n-1}, t]  (n+1 variables)
    # Minimize: t (so c_obj = [0,...,0, 1])
    # Constraints:
    #   G[k,:]·h - t <= G[k,:]·h0 - c_vals[k]  for each k
    #   sum(h) = S
    #   max(0, h0_j - r) <= h_j <= min(1, h0_j + r)
    #   0 <= t <= 2
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: G·h - t <= G·h0 - c_vals
    # A_ub has shape (L, n+1), b_ub has shape (L,)
    A_ub = np.hstack([G, -np.ones((L, 1))])
    b_ub = G @ h0 - c_vals
    
    # Equality constraint: sum(h) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    # Ensure lb <= ub
    for j in range(n):
        if lb[j] > ub[j]:
            lb[j] = ub[j] = h0[j]
    bounds = list(zip(lb, ub)) + [(-0.01, 2.0)]
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time if max_time else 30.0,
                               'presolve': True})
    except Exception:
        return None, False
    
    if res.status != 0:
        return None, False
    
    h_new = res.x[:n]
    h_new = ensure_feasible(h_new)
    return h_new, True

def lp_minimax_polish(h_seed, max_iter=100, init_radius=0.02, min_radius=1e-4,
                      max_radius=0.08, stop_time=None, verbose=False):
    """Iterative LP minimax polishing."""
    current_h = ensure_feasible(h_seed)
    current_c = exact_c5(current_h)
    n = current_h.size
    radius = init_radius
    no_improve_count = 0
    
    for it in range(max_iter):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        
        # Try LP step
        h_new, success = lp_minimax_step(current_h, radius, max_time=15.0)
        
        if not success or h_new is None:
            no_improve_count += 1
            radius = max(min_radius, radius * 0.5)
            if no_improve_count > 5:
                break
            continue
        
        # Line search
        delta = h_new - current_h
        delta_norm = float(np.max(np.abs(delta)))
        if delta_norm < 1e-9:
            no_improve_count += 1
            radius = max(min_radius, radius * 0.7)
            if no_improve_count > 5:
                break
            continue
        
        best_alpha = None
        best_h_ls = None
        best_c_ls = current_c
        
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.025]:
            h_try = ensure_feasible(current_h + alpha * delta)
            c_try = exact_c5(h_try)
            if np.isfinite(c_try) and c_try < best_c_ls - 1e-12:
                best_c_ls = c_try
                best_alpha = alpha
                best_h_ls = h_try
        
        if best_h_ls is not None:
            current_h = best_h_ls
            current_c = best_c_ls
            no_improve_count = 0
            # Maybe increase radius slightly
            if best_alpha >= 0.7:
                radius = min(max_radius, radius * 1.3)
        else:
            no_improve_count += 1
            radius = max(min_radius, radius * 0.5)
            if no_improve_count > 5:
                break
    
    return current_h, current_c

        p = float(rng.uniform(0.35, 0.65))
        hi = float(rng.uniform(0.6, 0.95))
        lo = float(rng.uniform(0.05, 0.4))
        h = np.where(th > p, hi, lo)
    else:
