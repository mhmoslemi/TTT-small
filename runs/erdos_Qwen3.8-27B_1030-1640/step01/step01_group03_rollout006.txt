                        grad[j - l] += wm * (-h_y[j])

            grad = grad - np.mean(grad)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            alpha = dx / (theta * gnorm)  # heuristic step size

            h_new = project_h(h_new, target)
            c_new = evaluate_C(h_new)

                h_cur = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_cur.copy()
            else:
                # Reject, reduce momentum
                beta *= 0.5
                if beta < 0.01:
                    break
