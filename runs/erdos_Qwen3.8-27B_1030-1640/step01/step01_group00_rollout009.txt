def project_box_sum(y, n):
    """Project y onto {0 <= x <= 1, sum(x) = n/2}"""
    y = np.asarray(y, dtype=float).reshape(-1)
    if y.size != n:
        y = np.full(n, 0.5)
    y = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    
    # Bisection on the threshold
    lo, hi = max(y.min() - 1, -1), min(y.max() + 1, 2)
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1 - x)
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = np.argmax(x)
            x[idx] = max(0.0, x[idx] + diff)
    return x

import time
import math
import numpy as np
from scipy import optimize

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1 - x)
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = np.argmax(x)
            x[idx] = max(0.0, x[idx] + diff)
    return x

def antisym_project(h, n):
    """Project onto anti-symmetric subspace h[i] = 1-h[n-1-i], [0,1], sum=n/2."""
    h_new = np.zeros(n)
    mid = n // 2
    for i in range(mid):
        v = 0.5 * (h[i] + 1.0 - h[n-1-i])
        h_new[i] = np.clip(v, 0, 1)
        h_new[n-1-i] = 1.0 - h_new[i]
    if n % 2 == 1:
        h_new[mid] = 0.5
    return h_new

def compute_c_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    c = np.zeros(L)
    G = np.zeros((L, n))
    for m in range(L):
        p = m - (n - 1)
        k_min = max(0, -p)
        k_max = min(n, n - p) - 1
        for k in range(k_min, k_max + 1):
            k2 = k + p
            c[m] += h[k] * (1.0 - h[k2]) * dx
            G[m, k] += dx * (1.0 - h[k2])
            G[m, k2] -= dx * h[k]
    return c, G

def lse_hessian_free_grad(h, tau):
    """Compute LSE objective and its gradient."""
    c, G = compute_c_G(h)
    L = len(c)
    maxc = c.max()
    z = (c - maxc) / tau
    ez = np.exp(z)
    ss = ez.sum()
    if ss < 1e-300:
        w = np.zeros(L)
        w[c.argmax()] = 1.0
        F = maxc
    else:
        F = maxc + tau * (np.log(ss) - np.log(L))
        w = ez / ss
    grad = G.T @ w
    return F, grad

def pg_antisym(h, n, tau, max_iter, deadline, min_time=30):

    h_cur = h.copy()
    F, g = lse_hessian_free_grad(h_cur, tau)
    # Anti-symmetric gradient
    g_as = g.copy()

        g_as[n-1-i] = -g_as[i]
    if n % 2 == 1:

    alpha = min(alpha, 2.0)

    alpha = 1.0 / (np.linalg.norm(g) + 1e-8)

            F = F_new
            _, g = lse_hessian_free_grad(h_cur, tau)
            alpha = min(alpha * 1.15, 8.0)
            stall = 0
        else:
            alpha *= 0.5
            stall += 1
            if alpha < 1e-8 or stall > 50:

def slqp_polish(h, n, tau, maxiter=30, deadline=None, min_time=30):
    """SLSQP polish for the LSE objective."""
    c, G = compute_c_G(h)
    cmax = c.max()
    if deadline is not None and time.monotonic() > deadline - min_time:
        return h
    
    def obj(x):
        F, _ = lse_hessian_free_grad(x, tau)
        return F

        _, g = lse_hessian_free_grad(x, tau)
        return g

        return x.sum() - n / 2.0

        return np.ones((1, n))

            val_new = compute_c_G(h_new)[0].max()
            val_old = compute_c_G(h)[0].max()
            if val_new <= val_old + 1e-10:
                return h_new
    except:
        pass
    return h

For our case, A A^T = 2I (each row has two 1's, and rows are orthogonal).

For component n-1-i:

d[i] = (g[n-1-i] - g[i]) / 2 for i < n/2
d[n-1-i] = -d[i]

This is the direction of steepest descent on the constraint manifold.

Then the step is: h_new = h - alpha * d
And we project h_new onto {h: h[i]+h[n-1-i]=1, 0≤h≤1}.

The projection onto {h[i]+h[n-1-i]=1, 0≤h≤1} is:
For each pair (i, n-1-i):
  s = h_new[i] + h_new[n-1-i]
  We need to find (a, b) with a+b=1, 0≤a≤1, 0≤b≤1, minimizing (a-h_new[i])^2 + (b-h_new[n-1-i])^2
  = minimize (a - h_new[i])^2 + (1-a - h_new[n-1-i])^2
  Let t = h_new[i] + h_new[n-1-i]
  Optimal a = (h_new[i] + 1 - h_new[n-1-i]) / 2 = (1 + 2h_new[i] - t) / 2
  Clip a to [0,1]
  b = 1 - a

import time
import math
import numpy as np
from scipy import optimize

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    if y.size != n:
        y = np.full(n, 0.5)
    y = np.array(y, dtype=float, copy=True)
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * float(n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + diff)
    return x

def antisym_project(h, n):
    h_new = np.zeros(n)
    mid = n // 2
    for i in range(mid):
        v = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
        h_new[i] = float(np.clip(v, 0.0, 1.0))
        h_new[n - 1 - i] = 1.0 - h_new[i]
    if n % 2 == 1:
        h_new[mid] = 0.5
    return h_new

def compute_c_G(h):
    n = len(h)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    c = np.zeros(L)
    G = np.zeros((L, n))
    for m in range(L):
        p = m - (n - 1)
        k_min = max(0, -p)
        k_max = min(n, n - p) - 1
        for k in range(k_min, k_max + 1):
            k2 = k + p
            c[m] += h[k] * (1.0 - h[k2]) * dx
            G[m, k] += dx * (1.0 - h[k2])
            G[m, k2] -= dx * h[k]
    return c, G

def exact_max_overlap(h):
    c, _ = compute_c_G(h)
    return float(np.max(c))

def lse_fg(h, tau):
    c, G = compute_c_G(h)
    L = len(c)
    maxc = float(np.max(c))
    z = (c - maxc) / tau
    ez = np.exp(z)
    ss = float(np.sum(ez))
    if ss < 1e-300:
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        F = maxc
    else:
        F = maxc + tau * (math.log(ss) - math.log(L))
        w = ez / ss
    grad = G.T.dot(w)
    return F, np.asarray(grad, dtype=float).reshape(-1)

def anti_grad(g, n):
    mid = n // 2
    d = np.zeros(n)
    for i in range(mid):
        d[i] = 0.5 * (g[n - 1 - i] - g[i])
        d[n - 1 - i] = -d[i]
    if n % 2 == 1:
        d[mid] = 0.0
    return d

        return h

        _, g = lse_fg(x, tau)

    except Exception:
        pass

    h_cur = h.copy()
    dx = 2.0 / float(n)
    for _ in range(200):
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        cur_max = float(np.max(c))

        best_pair = None
        ones = np.where(h_cur > 0.95)[0]
