            x = np.asarray(res.x, dtype=float).reshape(-1)
            if x.size == n and np.all(np.isfinite(x)):
                return clean_feasible(x, n)
    except Exception:
        pass
    return None


def consider_candidate(best, h):
    h = np.asarray(h, dtype=float)
    if h.size <= 0:
        return best
    h = clean_feasible(h, int(h.size))
    val = exact_value(h)
    if not math.isfinite(val):
        return best
    cur = float(best.get("val", math.inf))
    curn = int(best.get("n", 10**9))
    if val < cur - 1e-10 or (abs(val - cur) <= 1e-10 and int(h.size) < curn):
        best["h"] = h.copy()
        best["val"] = float(val)
        best["n"] = int(h.size)
    return best


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    n_new = int(n_new)
    if n <= 0 or n_new <= 0 or n_new == n:
        return clean_feasible(h, max(n, n_new))
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)


def make_cos_init(n, a1=0.3, a2=0.15, a3=0.05):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + a1 * np.cos(np.pi * x) + a2 * np.cos(2 * np.pi * x) + a3 * np.cos(3 * np.pi * x)
    return project_box_sum(h, n)


def make_random_init(n, rng, mode=0):
    if mode == 0:
        h = np.full(n, 0.5)
        h += 0.12 * rng.standard_normal(n)
    elif mode == 1:
        h = np.zeros(n)
        k = n // 2
        idx = rng.choice(n, k, replace=False)
        h[idx] = 1.0
        if n % 2 == 1:
            h[rng.integers(n)] = 0.5
    elif mode == 2:
        x = (np.arange(n) + 0.5) * (2.0 / n)
        freq = int(rng.integers(1, 6))
        phase = rng.uniform(0, 2 * math.pi)
        amp = rng.uniform(0.2, 0.45)
        h = 0.5 + amp * np.cos(freq * np.pi * x + phase)
        h += 0.03 * rng.standard_normal(n)
    else:
        h = 0.5 + rng.uniform(-0.3, 0.3) * rng.standard_normal(n)
    return project_box_sum(h, n)


def local_optimize(h0, deadline, need=30.0):
    n = len(h0)
    h = clean_feasible(h0, n)
    c0 = exact_value(h)
    taus = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.001, 0.0005]
    for idx, tau in enumerate(taus):
        if (deadline - time.monotonic()) < need + 5.0:
            break
        iters = max(1000, min(20000, int(8000000 / max(n, 1))))
        hf = fista_minimize(h, tau, max_iter=iters, deadline=deadline, need=need)
        if hf is not None:
            hf = clean_feasible(hf, n)
            cf = exact_value(hf)
            if math.isfinite(cf) and cf < c0 + 0.01:
                h = hf
                c0 = cf
        if (deadline - time.monotonic()) < need + 5.0:
            break
        hp = pg_minimize(h, tau, max_iter=2000, deadline=deadline, need=need)
        if hp is not None:
            hp = clean_feasible(hp, n)
            cp = exact_value(hp)
            if math.isfinite(cp) and cp < c0 + 0.005:
                h = hp
                c0 = cp
    # SLSQP polish with small tau
    for tau in [0.002, 0.001, 0.0005]:
        if (deadline - time.monotonic()) < need + 3.0:
            break
        hs = slqp_minimize(h, tau, maxiter=50, deadline=deadline, need=need)
        if hs is not None:
            hs = clean_feasible(hs, n)
            cs = exact_value(hs)
            if math.isfinite(cs) and cs < c0:
                h = hs
                c0 = cs
    # Final FISTA with tiny tau
    if (deadline - time.monotonic()) > need + 3.0:
        hf = fista_minimize(h, 0.0003, max_iter=30000, deadline=deadline, need=need)
        if hf is not None:
            hf = clean_feasible(hf, n)
            cf = exact_value(hf)
            if math.isfinite(cf) and cf < c0:
                h = hf
                c0 = cf
    return h, c0


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(1000 * int(seed) + 7)
    best = {"h": None, "val": math.inf, "n": 0}

    # Get initial solution
    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0:
            hinit = clean_feasible(hinit, len(hinit))
            consider_candidate(best, hinit)

    if best["h"] is None:
        consider_candidate(best, make_cos_init(45))

    if best["h"] is None:
        h_out = make_cos_init(45)
        return h_out, exact_value(h_out), len(h_out)

    n_init = int(best["n"])

    # Phase 1: Polish the initial solution
    if (deadline - time.monotonic()) > 60.0:
        h_pol, c_pol = local_optimize(best["h"], deadline, need=50.0)
        consider_candidate(best, h_pol)

    # Phase 2: Try different n values
    n_candidates = list(range(max(20, n_init - 15), min(90, n_init + 25)))
    # Ensure we include some spread
    for extra_n in [24, 30, 36, 42, 48, 54, 60, 66, 72, 80]:
        if extra_n not in n_candidates and 20 <= extra_n <= 90:
            n_candidates.append(extra_n)
    n_candidates = sorted(set(n_candidates))

    tried_n = 0
    for nn in n_candidates:
        if (deadline - time.monotonic()) < 50.0:
            break
        if tried_n >= 12:
            break
        # Resample from best
        hr = resample_h(best["h"], nn)
        if hr is not None:
            crc = exact_value(hr)
            consider_candidate(best, hr)

        # Also try a cosine init at this size
        hc = make_cos_init(nn)
        chc = exact_value(hc)
        consider_candidate(best, hc)

        # Pick the better of the two for local optimization
        h_try = hr if (math.isfinite(crc) and (not math.isfinite(chc) or crc <= chc)) else hc
        h_try = clean_feasible(h_try, nn)
        c_try = exact_value(h_try)

        if math.isfinite(c_try) and c_try < float(best["val"]) + 0.04:
            if (deadline - time.monotonic()) > 50.0:
                ho, co = local_optimize(h_try, deadline, need=45.0)
                consider_candidate(best, ho)
        tried_n += 1

    # Phase 3: Random restarts with perturbation of best
    n_best = int(best["n"])
    for attempt in range(8):
        if (deadline - time.monotonic()) < 45.0:
            break
        h_pert = best["h"].copy()
        sig = [0.01, 0.02, 0.04, 0.08][attempt % 4]
        h_pert += sig * rng.standard_normal(n_best)
        h_pert = project_box_sum(h_pert, n_best)
        c_pert = exact_value(h_pert)
        consider_candidate(best, h_pert)
        if math.isfinite(c_pert) and c_pert < float(best["val"]) + 0.02:
            if (deadline - time.monotonic()) > 45.0:
                ho, co = local_optimize(h_pert, deadline, need=40.0)
                consider_candidate(best, ho)

    # Phase 4: Final aggressive polish
    if best["h"] is not None and (deadline - time.monotonic()) > 20.0:
        h_final, c_final = local_optimize(best["h"], deadline, need=10.0)
        consider_candidate(best, h_final)

    # Additional fine-tuning: try tiny perturbations and re-optimize
    if best["h"] is not None and (deadline - time.monotonic()) > 15.0:
        n_f = len(best["h"])
        for _ in range(3):
            if (deadline - time.monotonic()) < 12.0:
                break
            h_tiny = best["h"] + 0.003 * rng.standard_normal(n_f)
            h_tiny = project_box_sum(h_tiny, n_f)
            h_tiny = fista_minimize(h_tiny, 0.0002, max_iter=20000, deadline=deadline, need=10.0)
            h_tiny = clean_feasible(h_tiny, n_f)
            consider_candidate(best, h_tiny)

    final_h = best["h"]
    if final_h is None:
        final_h = make_cos_init(45)
    final_h = clean_feasible(final_h, len(final_h))
    final_val = exact_value(final_h)
    if not math.isfinite(final_val):
        final_h = make_cos_init(45)
        final_h = clean_feasible(final_h, len(final_h))
        final_val = exact_value(final_h)
    return final_h, float(final_val), int(final_h.size)
