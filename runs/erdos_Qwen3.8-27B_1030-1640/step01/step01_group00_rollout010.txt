import time

def project_polytope(y, n):
    y = np.array(y, dtype=float).reshape(-1)
    if y.size != n:
        z = np.zeros(n)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * n
    lo, hi = y.min() - 0.5, y.max() + 0.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if mask.any():
            x[mask] += diff / mask.sum()
            np.clip(x, 0, 1, out=x)
    if abs(x.sum() - target) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] = np.clip(x[idx] + (target - x.sum()), 0, 1)
    return x


def eval_c(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def eval_max(h):
    c = eval_c(h)
    return float(np.max(c))


def grad_c_k(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    s = k_idx - (n - 1)
    g = np.zeros(n)
    j = np.arange(n)
    va = (j - s >= 0) & (j - s < n)
    g[va] += (1.0 - h[j[va] - s])
    vb = (j + s >= 0) & (j + s < n)
    g[vb] -= h[j[vb] + s]
    return dx * g


def lse_FG(h, tau):
    n = len(h)
    c = eval_c(h)
    cm = c.max()
    z = (c - cm) / tau
    ez = np.exp(z)
    ss = ez.sum()
    if ss < 1e-300:
        F = cm
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        F = cm + tau * (math.log(ss) - math.log(len(c)))
        w = ez / ss
    g = np.zeros(n)
    idx = np.where(w > 1e-12)[0]
    for m in idx:
        g += w[m] * grad_c_k(h, m)
    return float(F), g


def lse_pen_objective(h, tau, mu):
    F, g = lse_FG(h, tau)
    s = h.sum() - 0.5 * len(h)
    return F + 0.5 * mu * s * s, g + mu * s * np.ones(len(h))


def run_lse_lbfgs(h0, tau, mu=300.0, maxiter=120):
    n = len(h0)
    h0 = project_polytope(h0, n)
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        return lse_pen_objective(h, tau, mu)[0]

    def jac(h):
        return lse_pen_objective(h, tau, mu)[1]

    try:
        r = optimize.minimize(obj, h0, jac, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-9})
        if r.x is not None and np.all(np.isfinite(r.x)):
            return project_polytope(r.x, n)
    except Exception:
        pass
    return None


def run_lse_slqp(h0, tau, maxiter=50):
    n = len(h0)
    h0 = project_polytope(h0, n)

    def obj(h):
        return lse_FG(h, tau)[0]

    def jac(h):
        return lse_FG(h, tau)[1]

    def eqfun(h):
        return float(h.sum() - 0.5 * n)

    def eqjac(h):
        return np.ones((1, n))

    try:
        r = optimize.minimize(obj, h0, jac, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': eqfun, 'jac': eqjac}],
                              options={'maxiter': maxiter, 'ftol': 1e-10})
        if r.x is not None and np.all(np.isfinite(r.x)):
            return project_polytope(r.x, n)
    except Exception:
        pass
    return None


def run_epigraph_top(h0, t_init, num_top=10, maxiter=50):

    h0 = project_polytope(h0, n)
    c = eval_c(h0)
    top_idx = np.argsort(c)[::-1][:num_top]

    x0 = np.empty(n + 1)
    x0[:-1] = h0
    x0[-1] = t_init

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def epi_obj(x):
        return float(x[-1])

    def epi_jac(x):
        g = np.zeros(x.size)
        g[-1] = 1.0
        return g

    def epi_eq(x):
        return float(x[:-1].sum() - 0.5 * n)

    def epi_eqjac(x):
        J = np.zeros((1, x.size))
        J[0, :-1] = 1.0
        return J

    def epi_ineq(x):

    consider(best, hi)
    consider(best, sym_candidate(hi))
    consider(best, binary_init(len(hi), rng))

    # Phase 1: Main optimization from best initial
    if deadline - t0 > 60.0:
        h_opt = optimize_round(best['h'], deadline, need=40.0)
        consider(best, h_opt)

    # Phase 2: Try different n values
    cur_n = best['n']
    n_candidates = sorted(set([cur_n - 4, cur_n - 2, cur_n - 1, cur_n + 1, cur_n + 2,
                               cur_n + 4, cur_n + 6, cur_n + 8, cur_n + 12]))
    for n_t in n_candidates:
        if n_t < 20 or n_t > 90:
            continue
        if deadline - time.monotonic() < 35.0:
            break
        hr = resample(best['h'], n_t)
        consider(best, hr)
        h_opt = optimize_round(hr, deadline, need=25.0)
        consider(best, h_opt)

    # Phase 3: Perturbation-based local search
    attempts = 0
    while deadline - time.monotonic() > 35.0 and attempts < 10:
        n = best['n']
        h_p = best['h'].copy()
        npert = int(rng.integers(2, 6))
        idx_p = rng.choice(n, npert, replace=False)
        h_p[idx_p] += rng.uniform(-0.06, 0.06, npert)
        h_p = project_polytope(h_p, n)
        consider(best, h_p)
        h_opt = optimize_round(h_p, deadline, need=25.0)
        consider(best, h_opt)
        attempts += 1

    # Phase 4: Try binary-style starts at different n
    for n_b in [best['n'] - 2, best['n'] + 2, best['n'] + 5]:
        if n_b < 20 or n_b > 80:
            continue
        if deadline - time.monotonic() < 30.0:
            break
        hb = binary_init(n_b, rng)
        consider(best, hb)
        h_opt = optimize_round(hb, deadline, need=20.0)
        consider(best, h_opt)

    # Phase 5: Final polish on best
    if best['h'] is not None and deadline - time.monotonic() > 15.0:
        for tau in [0.0004, 0.0002, 0.00008]:
            if deadline - time.monotonic() < 8.0:
                break
            hf = run_lse_slqp(best['h'], tau, maxiter=60)
            consider(best, hf)
        if deadline - time.monotonic() > 10.0:
            hv = best['val']
            he = run_epigraph_top(best['h'], hv + 0.001, num_top=12, maxiter=55)
            consider(best, he)

    final_h = best['h']
    if final_h is None:
        final_h = np.full(45, 0.5)
    final_h = project_polytope(final_h, len(final_h))
    final_val = eval_max(final_h)
    if not np.isfinite(final_val):
        final_h = np.full(45, 0.5)
        final_val = eval_max(final_h)
    return final_h, float(final_val), int(len(final_h))
