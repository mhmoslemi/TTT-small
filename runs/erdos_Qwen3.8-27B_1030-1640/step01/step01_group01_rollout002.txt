  Second term: 1{0≤i+k<n} means i < n-k. So for i in [0, n-k-1]: contribution is -h[i+k].
  
  So: G[m, k:n] += dx * (1-h[0:n-k])
       G[m, 0:n-k] -= dx * h[k:n]
  
  Note: for i in [k, n-k-1] (if k < n/2), both terms apply.

For k < 0, let kk = -k > 0:
  G[m,i] = dx * [(1-h[i+kk]) * 1{0≤i+kk<n} - h[i-kk] * 1{0≤i-kk<n}]
  
  First term: i+kk < n => i < n-kk. Contribution: (1-h[i+kk]).
  Second term: i-kk ≥ 0 => i ≥ kk. Contribution: -h[i-kk].
  
  So: G[m, 0:n-kk] += dx * (1-h[kk:n])
       G[m, kk:n] -= dx * h[0:n-kk]

Let me verify with k=0 (m=n-1):
  G[m, 0:n] += dx * (1-h[0:n]) → G[m,i] = dx*(1-h[i])
  G[m, 0:n] -= dx * h[0:n] → G[m,i] -= dx*h[i]
  Total: G[m,i] = dx*(1-2h[i]) ✓

For k=1, n=4 (m = n-1+1 = 4, so m=4):
  G[4, 1:4] += dx*(1-h[0:3]) → G[4,1]+=dx*(1-h[0]), G[4,2]+=dx*(1-h[1]), G[4,3]+=dx*(1-h[2])
  G[4, 0:3] -= dx*h[1:4] → G[4,0]-=dx*h[1], G[4,1]-=dx*h[2], G[4,2]-=dx*h[3]
  
  So: G[4,0] = -dx*h[1]
      G[4,1] = dx*(1-h[0]) - dx*h[2]
      G[4,2] = dx*(1-h[1]) - dx*h[3]
      G[4,3] = dx*(1-h[2])
  
  Let me verify: shift k=1, c = dx*(h[1]*(1-h[0]) + h[2]*(1-h[1]) + h[3]*(1-h[2]))
  ∂c/∂h[0] = -dx*h[1] ✓
  ∂c/∂h[1] = dx*(1-h[0]) - dx*h[2] ✓
  ∂c/∂h[2] = dx*(1-h[1]) - dx*h[3] ✓
  ∂c/∂h[3] = dx*(1-h[2]) ✓

import time
import math
import numpy as np
from scipy.optimize import linprog

EPS_IMP = 5e-10
EPS_SUM = 1e-12


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    if vals.size == 0 or not np.all(np.isfinite(vals)):
        return float("inf")
    return float(np.max(vals))


def corr_values(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return np.zeros(0)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def force_sum(h, S, tol=1e-13):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h
    for _ in range(n + 50):
        r = float(S) - float(np.sum(h))
        if abs(r) <= tol:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(r, 1.0 - h[idx])
            if d <= 0:
                break
            h[idx] += d
        else:
            idx = int(np.argmin(h))
            d = min(-r, h[idx])
            if d <= 0:
                break
            h[idx] -= d
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(500):
        r = float(S) - float(np.sum(h))
        if abs(r) <= 5e-15:
            break
        if r > 0:
            idxs = np.where(h < 1.0 - 1e-14)[0]
            if idxs.size == 0:
                break
            h[int(idxs[0])] += min(r, 1.0 - h[int(idxs[0])])
        else:
            idxs = np.where(h > 1e-14)[0]
            if idxs.size == 0:
                break
            h[int(idxs[0])] -= min(-r, h[int(idxs[0])])
    return h


def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return force_sum(np.array([0.5, 0.5]), 1.0)
    if not np.all(np.isfinite(h)):
        return force_sum(np.full(n, 0.5), 0.5 * n)
    np.clip(h, 0.0, 1.0, out=h)
    return force_sum(h, 0.5 * n)


def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return h
    return ensure_feasible(0.5 * (h + h[::-1]))


def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(max(2, n_new))
    if n_old == 0:
        return ensure_feasible(np.full(n_new, 0.5))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)


def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for m in range(L):
        k = m - n + 1
        if k >= 0:
            if n - k > 0:
                G[m, k:] += dx * (1.0 - h[:n - k])
                G[m, :n - k] -= dx * h[k:]
        else:
            kk = -k
            if n - kk > 0:
                G[m, :n - kk] += dx * (1.0 - h[kk:])
                G[m, kk:] -= dx * h[:n - kk]
    return G


def lp_minimax_step(h0, radius, margin=1e-10, max_radius_retry=3):
    n = len(h0)
    if n < 4:
        return None
    L = 2 * n - 1
    S = 0.5 * n
    c0 = corr_values(h0)
    G = compute_grad_matrix(h0)
    b_val = G @ h0 - c0
    b_ub = b_val + margin
    A_ub = np.hstack([G, -np.ones((L, 1))])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    lb = np.minimum(lb, ub)
    bounds_h = list(zip(lb, ub))
    bounds_h.append((0.0, 1.0))
    for attempt in range(max_radius_retry + 1):
        r_cur = radius * (0.6 ** attempt) if attempt > 0 else radius
        lb2 = np.maximum(0.0, h0 - r_cur)
        ub2 = np.minimum(1.0, h0 + r_cur)
        lb2 = np.minimum(lb2, ub2)
        bounds2 = list(zip(lb2, ub2))
        bounds2.append((0.0, 1.0))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds2, method='highs',
                          options={'time_limit': 30.0, 'presolve': True})
            if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):
                return res.x[:n]
        except Exception:
            pass
    return None


def line_search(h0, h_lp, c0, alphas=None):
    if alphas is None:
        alphas = [1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.02]
    h_lp = ensure_feasible(h_lp)
    delta = h_lp - h0
    best_h = None
    best_c = c0
    for a in alphas:
        h_try = h0 + a * delta
        h_try = ensure_feasible(h_try)
        c_try = exact_c5(h_try)
        if c_try < best_c - EPS_IMP:
            best_c = c_try
            best_h = h_try.copy()
            if a == 1.0:
                break
    return best_h, best_c


def local_lp_polish(h, c, stop_time, max_iter=150, r0=None, r_max=0.06, r_min=1e-5):
    n = len(h)
    h = ensure_feasible(h)
    c = exact_c5(h)
    if r0 is None:
        r0 = max(0.008, min(0.03, 12.0 / math.sqrt(max(n, 16))))
    r = r0
    fails = 0
    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break
        h_lp = lp_minimax_step(h, r)
        if h_lp is None:
            fails += 1
            r = max(r_min, r * 0.5)
            if fails > 8:
                break
            continue
        h_new, c_new = line_search(h, h_lp, c)
        if h_new is not None:
            h = h_new
            c = c_new
            fails = 0
            if r < r_max:
                r = min(r_max, r * 1.3)
        else:
            fails += 1
            r = max(r_min, r * 0.45)
            if fails > 10 or r <= r_min * 1.5:
                break
    return h, c


def random_candidate(base, n, mode, rng):
    n = int(max(4, n))
    if base is not None:
        b = resample_h(base, n)
    else:
        b = np.full(n, 0.5)
    if mode == 0:
        scales = [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]
        sc = scales[int(rng.integers(0, len(scales)))]

    elif mode == 1:
        x = (np.arange(n) + 0.5) * (2.0 / n)
        k1 = int(rng.integers(1, 7))
        k2 = int(rng.integers(1, 11))
        a1 = float(rng.uniform(0.2, 0.45))
        a2 = float(rng.uniform(0.04, 0.2))
        p1 = float(rng.uniform(0, 2 * math.pi))
        p2 = float(rng.uniform(0, 2 * math.pi))
        h = 0.5 + a1 * np.sin(math.pi * k1 * x + p1) + a2 * np.sin(math.pi * k2 * x + p2)
        h = 0.5 * (h + b)
    elif mode == 2:
        nb = int(rng.integers(4, 12))
        nb = min(nb, n)
        cuts = np.sort(np.concatenate([[0.0], rng.uniform(0, 2, nb - 1), [2.0]]))
        vals = rng.uniform(0.1, 0.9, nb)
        x = (np.arange(n) + 0.5) * (2.0 / n)

        h = np.where(h > thr, hi, lo)
    else:

def get_parent():
    try:
        ph = globals().get("initial_h_values")
    except Exception:
        ph = None
    if ph is None:
        return None
    arr = np.asarray(ph, dtype=float).reshape(-1)
    if arr.size < 4 or not np.all(np.isfinite(arr)):
        return None
    return ensure_feasible(arr)


def try_update(h, c, n, best):
    if h is None or n <= 0 or not np.isfinite(c):
        return best
    bh, bc, bn = best
    if bh is None:
        return (h, c, n)
    if c < bc - EPS_IMP:
        return (h, c, n)
    if abs(c - bc) <= 1e-8 and n < bn:
        return (h, c, n)
    return best
