For the "coordinated move" step, here's the idea:
- At the current point h, the top-T lags have values c_1, ..., c_T.
- We want to find a direction d (with sum(d)=0, |d_i| small) that simultaneously reduces all c_k for k in top-T.
- The linearized condition is: grad_k · d < 0 for all k in top-T.
- To find such a direction, we can solve: minimize max_k (grad_k · d) subject to ||d||_2 <= 1, sum(d) = 0.
- This is a convex QP (actually a SOCP) that can be solved efficiently.

But for n=45, this is simple enough to just do by projection.

d = -sum_{k in top-T} w_k * grad_k

where w_k are the softmax weights. This is exactly what my gradient_top_T function computes! So my approach already does this.

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_max(h):
    """Return a subgradient of max_k corr[k]."""
    n = len(h)
    corr = compute_corr(h)
    k = np.argmax(corr)
    return _grad_lag(h, k)

def _grad_lag(h, k):
    """Gradient of corr[k] with respect to h."""
    n = len(h)
    offset = n - 1
    p = k - offset
    g = np.zeros(n)
    # First term: (1-h[i-p]) for i where 0<=i-p<n
    if p >= 0:
        g[p:n] += (1.0 - h[0:n-p])
    else:
        g[0:n+p] += (1.0 - h[-p:n])
    # Second term: -h[i+p] for i where 0<=i+p<n
    if p >= 0:
        g[0:n-p] -= h[p:n]
    else:
        g[-p:n] -= h[0:n+p]
    return g

def topT_subgradient(h, T, theta):
    """Smoothed subgradient using top-T softmax."""
    n = len(h)
    corr = compute_corr(h)
    idx = np.argsort(corr)[-T:]
    M = corr[idx].max()
    exps = np.exp((corr[idx] - M) / theta)
    w = exps / np.sum(exps)
    g = np.zeros(n)
    for ki, wi in zip(idx, w):
        if wi < 1e-12:
            continue
        g += wi * _grad_lag(h, ki)
    return g

def pair_optimize(h, s, t, target, max_iter=50):
    """Optimize the pair (s,t) by golden section."""
    n = len(h)
    c0 = h[s]
    ct = h[t]
    if abs(c0 - ct) < 1e-15:
        return h, evaluate_C(h)
    
    # We can move mass from s to t (or vice versa)
    # h[s] -> h[s] - tau, h[t] -> h[t] + tau
    # Feasible range: tau in [max(0, c0-1), min(c0, 1-ct)] if moving from s to t
    # Or more generally, tau in [-ct, 1-c0] (tau>0 means s loses, t gains)
    lo = -ct
    hi = 1.0 - c0
    
    if hi - lo < 1e-14:
        return h, evaluate_C(h)
    
    c_cur = evaluate_C(h)
    
    def eval_tau(tau):
        h[s] += -tau
        h[t] += tau
        c = evaluate_C(h)
        h[s] -= -tau  # restore
        h[t] -= tau
        return c
    
    # Golden section
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - invphi * (b - a)
    c2 = a + invphi * (b - a)
    f1 = eval_tau(c1)
    f2 = eval_tau(c2)
    best_val = min(f1, f2, c_cur)
    best_tau = 0.0
    if f1 < best_val:
        best_val, best_tau = f1, c1
    if f2 < best_val:
        best_val, best_tau = f2, c2
    
    for _ in range(max_iter):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - invphi * (b - a)
            f1 = eval_tau(c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + invphi * (b - a)
            f2 = eval_tau(c2)
        if f1 < best_val:
            best_val, best_tau = f1, c1
        if f2 < best_val:
            best_val, best_tau = f2, c2
    
    # Also check endpoints
    f_lo = eval_tau(lo)
    f_hi = eval_tau(hi)
    if f_lo < best_val:
        best_val, best_tau = f_lo, lo
    if f_hi < best_val:
        best_val, best_tau = f_hi, hi
    
    if best_tau != 0.0 and best_val < c_cur - 1e-14:
        h[s] -= best_tau
        h[t] += best_tau
    
    return h, evaluate_C(h)
