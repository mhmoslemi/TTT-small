                if new_c5 <= cur_c5 or (new_c5 - cur_c5) / T < -np.log(rng.random()):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best = cur.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    cur[i1] = 1.0; cur[i2] = 1.0
                    cur[j1] = 0.0; cur[j2] = 0.0
                    stall += 1
            else:
                # 1-swap
                i = i_pos[rng.integers(len(i_pos))]
                j = j_pos[rng.integers(len(j_pos))]
                cur[i] = 0.0; cur[j] = 1.0
                new_c5 = evaluate_c5(cur, n)
                if new_c5 <= cur_c5 or (new_c5 - cur_c5) / T < -np.log(rng.random()):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best = cur.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    cur[i] = 1.0; cur[j] = 0.0
                    stall += 1
    
    return best, best_c5

def lse_c5(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def compute_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
        else:
            if -s < n:
                grad[:n + s] += wk * (1.0 - h[-s:])
                grad[-s:] -= wk * h[:n + s]
    return grad * dx

def pg_optimize(h, n, lam, max_iter=500, deadline=None):
    h = h.copy()
    f = lse_c5(h, lam, n)
    alpha = 1.0 / n
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = compute_grad(h, lam, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        found = False
        a = alpha
        for _ in range(40):
            h_new = project_feasible(h - a * g, n)
            f_new = lse_c5(h_new, lam, n)
            if f_new < f - 1e-15:
                found = True
                break
            a *= 0.5
        if not found:
            stall += 1
            alpha = max(a, 0.001 / n)
            if stall > 50:
                break
            continue
        stall = 0
        h = h_new
        f = f_new
        alpha = min(a * 1.5, 2.0)
    return h

def resample_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)
