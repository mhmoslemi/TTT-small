def coord_opt_i(h, i, overlaps, dx):
    """Find optimal value of h[i] given all other coordinates fixed."""
    n = len(h)
    # Compute c_k for all k
    k_vals = np.arange(-(n-1), n)  # -(n-1), ..., n-1
    c = np.zeros(2*n-1)
    for idx, k in enumerate(k_vals):
        ik = i + k
        ik2 = i - k
        coeff = 0.0
        if 0 <= ik < n:
            coeff += (1.0 - h[ik])
        if 0 <= ik2 < n:
            coeff += (-h[ik2])
        c[idx] = dx * coeff
    
    # a_k = O(k) - c_k * h[i]
    hi = h[i]
    a = overlaps - c * hi
    
    # Find t in [0,1] minimizing max_k (a_k + c_k * t)
    # Critical points: intersections of pairs of lines
    best_t = hi  # default: keep current
    best_val = np.max(a + c * hi)
    
    # Check boundaries
    val_0 = np.max(a)
    if val_0 < best_val:
        best_val = val_0
        best_t = 0.0
    val_1 = np.max(a + c)
    if val_1 < best_val:
        best_val = val_1
        best_t = 1.0
    
    # Check intersection points

def evaluate_C(h):

def compute_c_for_i(h, i, dx):
    """Compute c_k(i) for all k, the coefficient of h[i] in each O(k)."""
    n = len(h)
    c = np.zeros(2*n-1)
    # k ranges from -(n-1) to n-1, indexed by 0 to 2n-2
    for idx in range(2*n-1):
        k = idx - (n-1)
        coeff = 0.0
        ik = i + k
        if 0 <= ik < n:
            coeff += (1.0 - h[ik])
        ik2 = i - k
        if 0 <= ik2 < n:
            coeff += (-h[ik2])
        c[idx] = dx * coeff
    return c

    hi = h[i]

            best_h = h.copy()

            da = a[np.newaxis, :] - a[:, np.newaxis]  # da[p,q] = a[p] - a[q]
            dc = c[np.newaxis, :] - c[:, np.newaxis]  # dc[p,q] = c[p] - c[q]
            valid = np.abs(dc) > 1e-14
            with np.errstate(divide='ignore', invalid='ignore'):
                t_all = -da / dc  # t_all[p,q] = (a[q]-a[p])/(c[p]-c[q])
            t_all[~valid] = -1.0
            valid_t = valid & (t_all > 0) & (t_all < 1)
            
            if np.any(valid_t):
                tv = t_all[valid_t]
                # Evaluate max at all valid t values
                vals = a[np.newaxis, :] + c[np.newaxis, :] * tv[:, np.newaxis]
                max_vals = np.max(vals, axis=1)
                min_idx = int(np.argmin(max_vals))
                if max_vals[min_idx] < best_val - 1e-15:
                    best_val = float(max_vals[min_idx])
                    best_t = float(tv[min_idx])
            
            delta = best_t - h[i]
            if abs(delta) > 1e-15:
                overlaps += c * delta
                h[i] = best_t
                max_change = max(max_change, abs(delta))
        
        c_cur = evaluate_C(h)
        if c_cur < best_c - 1e-14:
            best_c = c_cur
            best_h = h.copy()
        
        if max_change < 1e-13:
            break
    
    return best_h, best_c

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def make_random_start(n, rng):
    h = np.full(n, 0.5)
    freqs = rng.integers(1, 10, size=4)
    phases = rng.random(4) * 2 * np.pi
    amps = rng.random(4) * 0.25 + 0.05
    x = np.linspace(0, 2, n, endpoint=False)
    for f, p, a in zip(freqs, phases, amps):
        h += a * np.sin(2.0 * np.pi * f * x / 2.0 + p)
    h = np.clip(h, 0.0, 1.0)
    return project_h(h, n / 2.0)

def make_block_start(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blk = max(1, int(rng.integers(1, max(2, n // 4))))
        val = rng.random()
        end = min(pos + blk, n)
        h[pos:end] = val
        pos = end
    return project_h(h, n / 2.0)

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_h(h_dst, n_dst / 2.0)

def coordinate_descent(h, max_sweeps=300, time_limit=None, t_start=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(np.float64)
    overlaps = compute_overlaps(h)
    best_c = evaluate_C(h)
    best_h = h.copy()
    k_vals = np.arange(-(n - 1), n)
    two_n_1 = 2 * n - 1

    for sweep in range(max_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        max_change = 0.0
        for i in range(n):
            c = np.zeros(two_n_1)
            ik = i + k_vals
            valid1 = (ik >= 0) & (ik < n)
            c[valid1] += (1.0 - h[ik[valid1]])
            ik2 = i - k_vals
            valid2 = (ik2 >= 0) & (ik2 < n)
            c[valid2] += (-h[ik2[valid2]])
            c *= dx
            hi = h[i]
            a = overlaps - c * hi
            v0 = float(np.max(a))
            v1 = float(np.max(a + c))
            if v0 <= v1:
                best_t, best_val = 0.0, v0
            else:
                best_t, best_val = 1.0, v1
            da = a[np.newaxis, :] - a[:, np.newaxis]
            dc = c[np.newaxis, :] - c[:, np.newaxis]
            valid = np.abs(dc) > 1e-14
            with np.errstate(divide='ignore', invalid='ignore'):
                t_all = -da / dc
            t_all[~valid] = -1.0
            valid_t = valid & (t_all > 0.0) & (t_all < 1.0)
            if np.any(valid_t):

                vals = a[np.newaxis, :] + c[np.newaxis, :] * tv[:, np.newaxis]
                max_vals = np.max(vals, axis=1)
                min_idx = int(np.argmin(max_vals))

                    best_val = float(max_vals[min_idx])
                    best_t = float(tv[min_idx])
            delta = best_t - h[i]

                h[i] = best_t

            best_h = h.copy()
        if max_change < 1e-13:

def pair_polish_fast(h, rng, n_pairs, max_iter=50):
    n = len(h)
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_pairs):
        valid_s = np.where(h_cur > 1e-8)[0]
        valid_t = np.where(h_cur < 1.0 - 1e-8)[0]
        if len(valid_s) == 0 or len(valid_t) == 0:
            break
        s = valid_s[rng.integers(len(valid_s))]
        t = valid_t[rng.integers(len(valid_t))]
        if s == t:
            continue
        T = min(h_cur[s], 1.0 - h_cur[t])
        if T < 1e-14:
            continue
        a, b = 0.0, T
        c1 = b - invphi * (b - a)
        c2 = a + invphi * (b - a)
        ht1 = h_cur.copy(); ht1[s] -= c1; ht1[t] += c1
        ht2 = h_cur.copy(); ht2[s] -= c2; ht2[t] += c2
        f1, f2 = evaluate_C(ht1), evaluate_C(ht2)
        best_val = min(f1, f2, c_cur)
        best_tau = 0.0
        if f1 < best_val:
            best_val, best_tau = f1, c1
        if f2 < best_val:
            best_val, best_tau = f2, c2
        for _ in range(max_iter):
            if b - a < 1e-13 * (T + 1.0):
                break
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1
                c1 = b - invphi * (b - a)
                ht1 = h_cur.copy(); ht1[s] -= c1; ht1[t] += c1
                f1 = evaluate_C(ht1)
            else:
                a = c1; c1 = c2; f1 = f2
                c2 = a + invphi * (b - a)

                f2 = evaluate_C(ht2)
            if f1 < best_val:

            h_cur[t] += best_tau
            c_cur = best_val
    return h_cur, c_cur
