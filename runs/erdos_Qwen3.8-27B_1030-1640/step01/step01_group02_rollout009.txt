def swap_delta(b, p, q, n):

    delta = np.zeros(2*n-1)
    ps = p + shifts
    m1 = (ps >= 0) & (ps < n)

    m3 = (ps2 >= 0) & (ps2 < n)
    delta[m3] += b[ps2[m3]]
    qs2 = q - shifts
    m4 = (qs2 >= 0) & (qs2 < n)
    delta[m4] -= b[qs2[m4]]

def structured_binary(n, rng):
    """Create a structured binary sequence based on a threshold."""
    x = np.arange(n)
    # Use a quasi-random sequence
    alpha = rng.choice([0.618033988749895, 0.7071067811865476, 0.5773502691896258, 0.7320508075688772])
    phase = rng.random()
    vals = (alpha * x + phase) % 1.0
    # Select top n/2 values
    threshold = np.sort(vals)[n//2]
    b = (vals >= threshold).astype(np.int8)
    # Ensure exactly n/2 ones
    if b.sum() != n//2:

        zeros = np.where(b == 0)[0]

            b[to_remove] = 0
        else:
            deficit = n//2 - b.sum()
            to_add = rng.choice(zeros, size=deficit, replace=False)
            b[to_add] = 1

def lse_objective(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def lse_gradient(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n-s])
                grad[:n-s] -= wk * h[s:]
        else:
            if -s < n:
                grad[:n+s] += wk * (1.0 - h[-s:])
                grad[-s:] -= wk * h[:n+s]
    return grad * dx

        f = f_new
