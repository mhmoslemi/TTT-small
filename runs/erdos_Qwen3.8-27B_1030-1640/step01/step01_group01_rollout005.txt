def optimize_1d(A, B, t_min, t_max, iters=50):
    lo, hi = t_min, t_max
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        vals = A + mid * B
        k_star = np.argmax(vals)
        if B[k_star] > 1e-15:
            hi = mid
        elif B[k_star] < -1e-15:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

        elif bk < -1e-14:
            lo = mid
        else:

    for tc in candidates:

        if v < best_val:
            best_val = v
            best_t = tc
    return best_t
