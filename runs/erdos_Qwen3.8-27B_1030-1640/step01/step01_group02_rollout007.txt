import numpy as np
import time

def task_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum to be exact
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add mass to some elements
            room = 1.0 - h
            order = np.argsort(room)[::-1]
            for idx in order:
                add = min(diff, room[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            need = -diff
            order = np.argsort(h)
            for idx in order:
                sub = min(need, h[idx])
                h[idx] -= sub
                need -= sub
                if abs(need) < 1e-12:
                    break
    return h

def compute_all_coeffs(h, n):
    """Compute u_k(i) and v_k(j) for pair transfer.
    u_k(i) = coefficient of h[i] in C_k = (1-h[i-k]) - h[i+k] (for valid indices)
    Actually let me be more careful.
    
    C_k = sum_j h[j] * (1-h[j-k]) for valid j and j-k in [0,n-1].
    
    The coefficient of h[i] in C_k:
    - From term j=i: (1-h[i-k]) if i-k is in [0,n-1]
    - From term j where j-k=i, i.e., j=i+k: -h[i+k] if i+k is in [0,n-1]
    
    So coeff_i(k) = (1-h[i-k]) * [i-k in range] - h[i+k] * [i+k in range]
    """
    # For efficiency, precompute for all i and k
    # corr has indices 0..2n-2, corresponding to shift s = k-(n-1)
    # C_s = sum_j h[j] * (1-h[j-s]) for valid j, j-s
    # coeff of h[i] in C_s = (1-h[i-s]) * [i-s in [0,n-1]] - h[i+s] * [i+s in [0,n-1]]
    
    # Let's index by shift s from -(n-1) to (n-1)
    # corr[s + n - 1] = C_s
    coeff = np.zeros((n, 2*n-1))
    for i in range(n):
        for s in range(-(n-1), n):
            c = 0.0
            if 0 <= i - s < n:
                c += (1.0 - h[i - s])
            if 0 <= i + s < n:
                c -= h[i + s]
            coeff[i, s + n - 1] = c
    return coeff

def pair_transfer_step(h, n, max_sweeps=50):
    """Perform pair transfer coordinate descent."""
    h = h.copy()
    dx = 2.0 / n
    improved = True
    total_improvements = 0
    for sweep in range(max_sweeps):
        best_improve = 0
        best_pair = None
        best_t = 0
        best_new_c5 = task_exact(h)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        C = corr  # length 2n-1
        
        coeff = compute_all_coeffs(h, n)
        # coeff[i, k] = coefficient of h[i] in C_k (where k is 0..2n-2, shift = k-(n-1))
        
        # For pair (i, j): transfer t from i to j
        # C_k_new = C_k + t * (coeff[j, k] - coeff[i, k])
        # Minimize max_k C_k_new over t in [0, min(h[i], 1-h[j])]
        
        for i in range(n):
            if h[i] < 1e-10:
                continue
            for j in range(n):
                if i == j:
                    continue
                if h[j] > 1.0 - 1e-10:
                    continue
                T_max = min(h[i], 1.0 - h[j])
                if T_max < 1e-10:
                    continue
                d = coeff[j] - coeff[i]  # d[k] = change in C_k per unit t
                # Minimize f(t) = max_k (C[k] + d[k]*t) over t in [0, T_max]
                # f is convex (max of linear)
                # Ternary search
                lo_t, hi_t = 0.0, T_max
                for _ in range(30):
                    m1 = lo_t + (hi_t - lo_t) / 3.0
                    m2 = hi_t - (hi_t - lo_t) / 3.0
                    f1 = np.max(C + d * m1)
                    f2 = np.max(C + d * m2)
                    if f1 < f2:
                        hi_t = m2
                    else:
                        lo_t = m1
                t_opt = (lo_t + hi_t) / 2.0
                f_opt = np.max(C + d * t_opt)
                if f_opt < best_new_c5 - 1e-12:
                    best_improve = best_new_c5 - f_opt
                    best_pair = (i, j)
                    best_t = t_opt
                    best_new_c5 = f_opt
        
        if best_pair is not None:
            i, j = best_pair
            h[i] -= best_t
            h[j] += best_t
            h = np.clip(h, 0.0, 1.0)
            total_improvements += 1
            if best_improve < 1e-10:
                break
        else:
            break
    
    return h, total_improvements

def lse_pg_optimize(h, n, lam, max_iter=500):
    """Log-sum-exp projected gradient optimization."""
    h = h.copy()
    dx = 2.0 / n
    f = lse_value(h, lam, n)
    alpha = 1.0
    stall = 0
    for it in range(max_iter):
        g = lse_grad(h, lam, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        found = False
        a = alpha
        for _ in range(30):
            h_new = project_feasible(h - a * g, n)
            f_new = lse_value(h_new, lam, n)
            if f_new < f - 1e-15:
                found = True
                break
            a *= 0.5
        if not found:
            stall += 1
            alpha = max(a, 0.001)
            if stall > 20:
                break
            continue
        stall = 0
        h = h_new
        f = f_new
        alpha = min(a * 1.5, 10.0)
    return h

def lse_value(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def lse_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
        else:
            if -s < n:
                grad[:n + s] += wk * (1.0 - h[-s:])
                grad[-s:] -= wk * h[:n + s]
    return grad * dx

def resample_to(h, n_old, n_new):

    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

```python
def compute_all_coeffs(h, n):
    # coeff[i, k] where k = s + (n-1), s = -(n-1)..(n-1)
    # coeff[i, k] = (1-h[i-s]) if 0<=i-s<n else 0, minus h[i+s] if 0<=i+s<n else 0
    coeff = np.zeros((n, 2*n-1))
    for s in range(-(n-1), n):
        k = s + n - 1
        # Term 1: (1-h[i-s]) for valid i-s
        if s >= 0:
            # i-s >= 0 means i >= s
            coeff[s:, k] += (1.0 - h[:n-s])
        else:
            # i-s = i+|s| >= 0 always (since i>=0, |s|>0), and i-s < n means i < n-s
            coeff[:n+s, k] += (1.0 - h[-s:])
        # Term 2: -h[i+s] for valid i+s
        if s >= 0:
            # i+s < n means i < n-s
            coeff[:n-s, k] -= h[s:]
        else:
            # i+s = i-|s| >= 0 means i >= |s|
            coeff[-s:, k] -= h[:-s]
    return coeff
```

This is O(n²) but with vectorized numpy operations inside, so it's fast.

- h[i] decreases by t
- h[j] increases by t
- Sum is preserved
- Bounds: need 0 ≤ h[i]-t and 0 ≤ h[j]+t ≤ 1, so t ≤ min(h[i], 1-h[j])

def _lse_value(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    logsum = ms + np.log(np.sum(np.exp(scaled - ms)))
    return m + logsum / lam

def _lse_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
        else:
            if -s < n:
                grad[:n + s] += wk * (1.0 - h[-s:])
                grad[-s:] -= wk * h[:n + s]
    return grad * dx

def make_binary_structured(n, rng):
    """Create a balanced binary sequence using structured method."""
    n_ones = n // 2
    # Use a Beatty sequence approach
    alpha = 0.618033988749895  # golden ratio conjugate
    phase = rng.random()
    x = (np.arange(n) * alpha + phase) % 1.0
    # Select top n_ones values
    threshold = np.sort(x)[-n_ones]
    b = (x >= threshold).astype(float)
    # Ensure exactly n_ones
    if b.sum() != n_ones:
        # Fix by threshold
        sorted_idx = np.argsort(x)[::-1]
        b = np.zeros(n)
        b[sorted_idx[:n_ones]] = 1.0
    return b
