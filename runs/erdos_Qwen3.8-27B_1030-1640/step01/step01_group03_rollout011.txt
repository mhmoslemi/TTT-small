def solve_1d_convex(A, B, lam):
    """Find x in [0,1] minimizing max_m (A[m]*x + B[m]) + lam*x"""
    # Equivalent to: max_m ((A[m]+lam)*x + B[m])
    A_adj = A + lam
    M = len(A)
    # Evaluate at x=0 and x=1
    f0 = np.max(B)
    f1 = np.max(A_adj + B)
    best_x = 0.0 if f0 < f1 else 1.0
    best_f = min(f0, f1)
    
    # Find all intersection points in (0,1)
    # x_ij = (B[j]-B[i]) / (A_adj[i]-A_adj[j])
    # Only valid when A_adj[i] != A_adj[j] and 0 < x_ij < 1
    
    # Vectorized computation
    diff_B = B[:, None] - B[None, :]  # [i,j] = B[i] - B[j]
    diff_A = A_adj[:, None] - A_adj[None, :]  # [i,j] = A_adj[i] - A_adj[j]
    # x = (B[j]-B[i]) / (A_adj[i]-A_adj[j]) = -diff_B[i,j] / diff_A[i,j]
    # Wait: x_ij where line i = line j: A_adj[i]*x + B[i] = A_adj[j]*x + B[j]
    # => (A_adj[i] - A_adj[j])*x = B[j] - B[i]
    # => x = (B[j] - B[i]) / (A_adj[i] - A_adj[j])
    
    with np.errstate(divide='ignore', invalid='ignore'):
        x_pts = (B[None, :] - B[:, None]) / diff_A  # [i,j] -> x where line i meets line j
    
    # Filter: 0 < x < 1, and valid (no NaN/inf)
    mask = (x_pts > 0) & (x_pts < 1) & np.isfinite(x_pts)
    valid_x = np.unique(x_pts[mask])
    
    # Evaluate f at each valid x
    if len(valid_x) > 0:
        # f(x) = max_m (A_adj[m]*x + B[m])
        vals = np.max(A_adj[None, :] * valid_x[:, None] + B[None, :], axis=1)
        min_idx = np.argmin(vals)
        if vals[min_idx] < best_f:
            best_f = vals[min_idx]
            best_x = valid_x[min_idx]
    
    return best_x

    M = 2 * n - 1
    offset = n - 1

C(1-h) = max_k ∫(1-h)(x) * (1-(1-h)(x+k)) dx = max_k ∫(1-h(x)) * h(x+k) dx = C(h)
So yes, C(1-h) = C(h)! This is a symmetry.

What about h(2-x)? (reflection about x=1)
C(h_reflected) = max_k ∫ h(2-x) * (1-h(2-x+k)) dx = max_k ∫ h(2-u) * (1-h(2-u+k)) du (substituting u=2-x)
= max_k ∫ h(u) * (1-h(u-k)) du (substituting u=2-x, flipping the integral)
= max_k ∫ h(u) * (1-h(u-k)) du

This is the same as the original but with lag -k instead of +k. Since we take max over all k (both positive and negative), C(h_reflected) = C(h).

So both symmetries preserve the objective. I can exploit this by symmetrizing: h_sym = 0.5*(h + h_reflected) or h_sym = 0.5*(h + 1-h)... but the latter doesn't preserve the sum constraint.

import numpy as np
import time
from scipy.optimize import minimize_scalar

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def compute_lag_terms(h, k):
    n = len(h)
    M = 2 * n - 1
    offset = n - 1
    c_full = np.correlate(h, 1.0 - h, mode="full")
    m_vals = np.arange(M)
    L = m_vals - offset
    i1 = k - L
    i2 = k + L
    valid1 = (i1 >= 0) & (i1 < n)
    valid2 = (i2 >= 0) & (i2 < n)
    h1 = np.zeros(M)
    h2 = np.zeros(M)
    h1[valid1] = h[i1[valid1]]
    h2[valid2] = h[i2[valid2]]
    A = (1.0 - h1) * valid1.astype(float) - h2 * valid2.astype(float)
    B = c_full.copy()
    B -= h[k] * (1.0 - h1) * valid1.astype(float)
    B -= h2 * (1.0 - h[k]) * valid2.astype(float)
    B += h2 * valid2.astype(float)
    C0 = c_full[offset] - h[k] * (1.0 - h[k])
    return A, B, C0, offset

def minimize_coord(h, k, lam):
    A, B, C0, offset = compute_lag_terms(h, k)
    def f(x):
        vals = A * x + B
        vals[offset] = C0 + x * (1.0 - x)
        return np.max(vals) + lam * x
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def cd_with_lambda(h, lam, n_sweeps=30, order=None):
    n = len(h)
    if order is None:
        order = np.arange(n)
    for _ in range(n_sweeps):
        for k in order:
            h[k] = minimize_coord(h, k, lam)
    return h

def lagrangian_cd_bisect(h_init, n_sweeps=25, max_bisections=25, lam_range=(-2.0, 2.0), tol=1e-5):
    n = len(h_init)
    target = n / 2.0
    lam_lo, lam_hi = lam_range
    h_curr = h_init.copy()
    for _ in range(max_bisections):
        lam = (lam_lo + lam_hi) / 2.0
        h_curr = cd_with_lambda(h_curr, lam, n_sweeps=n_sweeps)
        s = np.sum(h_curr) - target
        if abs(s) < tol:
            break
        if s > 0:
            lam_lo = lam
        else:
            lam_hi = lam
    return h_curr, (lam_lo + lam_hi) / 2.0

def compute_soft_obj_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    G = corr * dx
    Gmax = np.max(G)
    if not np.isfinite(Gmax):
        return np.inf, np.zeros(n)
    exp_G = np.exp((G - Gmax) / theta)
    w = exp_G / np.sum(exp_G)
    offset = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - offset
        if k >= 0:
            j_lo1, j_hi1 = k, n
        else:
            j_lo1, j_hi1 = 0, n + k
        if j_lo1 < j_hi1:
            idx = np.arange(j_lo1, j_hi1) - k
            grad[j_lo1:j_hi1] += wm * (1.0 - h[idx])
        if k >= 0:
            j_lo2, j_hi2 = 0, n - k
        else:
            j_lo2, j_hi2 = -k, n
        if j_lo2 < j_hi2:
            idx2 = np.arange(j_lo2, j_hi2) + k
            grad[j_lo2:j_hi2] += wm * (-h[idx2])
    return Gmax + theta * np.log(np.sum(exp_G)), grad

def pair_polish(h, rng, n_pairs, max_iter=60, theta=3e-4):
    n = len(h)
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    for it in range(n_pairs):
        obj, grad = compute_soft_obj_grad(h_cur, theta)
        if not np.isfinite(obj):
            break
        valid_s = np.where(h_cur > 1e-8)[0]
        valid_t = np.where(h_cur < 1.0 - 1e-8)[0]
        if len(valid_s) == 0 or len(valid_t) == 0:
            break
        best_impr = 0.0
        best_new = None
        n_try = min(10, max(1, len(valid_s) * len(valid_t)))
        for _ in range(n_try):
            r = rng.random()
            if r < 0.4:
                s = valid_s[rng.integers(len(valid_s))]
                t = valid_t[rng.integers(len(valid_t))]
            elif r < 0.8:
                s = valid_s[np.argmax(grad[valid_s])]
                t = valid_t[np.argmin(grad[valid_t])]
            else:
                s = valid_s[np.argmin(grad[valid_s])]
                t = valid_t[np.argmax(grad[valid_t])]
            if s == t:
                continue
            hn, cn = golden_pair(h_cur, s, t, max_iter)
            if cn < c_cur - 1e-13:

                    best_impr = c_cur - cn
                    best_new = (hn, cn)
        if best_new is not None:
            h_cur, c_cur = best_new
    return h_cur, c_cur

def golden_pair(h, s, t, max_iter=80):
    T = min(h[s], 1.0 - h[t])
    if T < 1e-14:
        return h, evaluate_C(h)
    c_cur = evaluate_C(h)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T
    def ev(tau):
        ht = h.copy()
        ht[s] -= tau
        ht[t] += tau
        return evaluate_C(ht)
    c1 = b - invphi * (b - a)
    c2 = a + invphi * (b - a)
    f1 = ev(c1)
    f2 = ev(c2)
    best_val = min(f1, f2, c_cur)
    best_tau = 0.0
    if f1 < best_val:
        best_val, best_tau = f1, c1
    if f2 < best_val:
        best_val, best_tau = f2, c2
    for _ in range(max_iter):
        if b - a < 1e-13 * (T + 1.0):
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - invphi * (b - a); f1 = ev(c1)
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + invphi * (b - a); f2 = ev(c2)
        if f1 < best_val:
            best_val, best_tau = f1, c1
        if f2 < best_val:
            best_val, best_tau = f2, c2
    f_end = ev(T)
    if f_end < best_val:
        best_val, best_tau = f_end, T
    if best_tau > 1e-14 and best_val < c_cur - 1e-13:
        hn = h.copy()
        hn[s] -= best_tau
        hn[t] += best_tau
        return hn, best_val
    return h, c_cur

def smooth_descent(h, theta_schedule, max_iters=200):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    for theta in theta_schedule:
        for it in range(max_iters):
            obj, grad = compute_soft_obj_grad(h_cur, theta)
            if not np.isfinite(obj):
                break
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-14:
                break
            alpha = 0.1 / (gnorm + 1e-10)
            accepted = False
            for _ in range(40):
                y = h_cur - alpha * grad
                y_proj = project_h(y, target)
                c_t = evaluate_C(y_proj)
                if c_t < best_c + theta * 0.5:
                    h_cur = y_proj
                    if c_t < best_c:
                        best_c = c_t
                        best_h = h_cur.copy()
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-15:
                    break
            if not accepted:
                break
    return best_h, best_c

def make_random_start(n, rng):
    h = np.full(n, 0.5)
