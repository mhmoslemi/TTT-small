import time

def project_box_sum(y, n):
    """Euclidean projection onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    y = np.asarray(y, dtype=float).copy()
    if y.size != n:
        y = np.resize(y, n)
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * n
    
    # Bisection on the threshold
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    
    # Fix to exact sum
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            free = np.where(x < 1.0)[0]
            if len(free) > 0:
                x[free] += diff / len(free)
        else:
            free = np.where(x > 0.0)[0]
            if len(free) > 0:
                x[free] += diff / len(free)
        np.clip(x, 0.0, 1.0, out=x)
    
    return x

def overlap_components(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_value(h):
    return float(np.max(overlap_components(h)))

    n = len(h)
    dx = 2.0 / n
    maxc = np.max(c)

    ez = np.exp(z)
    ss = ez.sum()
    F = maxc + tau * (np.log(ss) - np.log(len(c)))
    w = ez / ss

        J[m, i] -= dx * (1.0 - h[j])

import time
import math
import numpy as np
from scipy import optimize


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    if y.size != n:
        new = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            new[:m] = y[:m]
        y = new
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            free = np.where(x < 1.0)[0]
        else:
            free = np.where(x > 0.0)[0]
        if len(free) > 0:
            x[free] += diff / len(free)
    np.clip(x, 0.0, 1.0, out=x)
    return x


def overlap_components(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_value(h):
    return float(np.max(overlap_components(h)))


def lse_FG(h, tau):
    c = overlap_components(h)
    n = len(h)
    dx = 2.0 / n
    maxc = c.max()
    z = (c - maxc) / tau
    ez = np.exp(z)
    ss = ez.sum()
    F = maxc + tau * (np.log(ss) - np.log(len(c)))
    w = ez / ss
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-11:
            continue
        s = n - 1 - m
        i0 = max(0, -s)
        i1 = min(n - 1, n - 1 - s)
        if i0 > i1:
            continue
        i = np.arange(i0, i1 + 1)
        j = i + s
        grad[i] += wm * dx * (1.0 - h[j])
        grad[j] -= wm * dx * h[i]
    return F, grad


def fista_lse(h0, tau, max_iter=20000, deadline=None):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    L = 4.0 / n + 2.0 / max(tau, 1e-4) + 1.0
    step = 1.0 / L
    F_prev = lse_FG(h, tau)[0]
    best_h = h.copy()
    best_F = F_prev
    for k in range(max_iter):
        if deadline is not None and (deadline - time.monotonic()) < 3.0:
            break
        _, g = lse_FG(y, tau)
        z = y - step * g
        x_new = project_box_sum(z, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - h)
        h = x_new
        t = t_new
        if (k + 1) % 1000 == 0:
            F_curr = lse_FG(h, tau)[0]
            if F_curr < best_F:
                best_F = F_curr
                best_h = h.copy()
            if abs(F_curr - F_prev) < 1e-12:
                break
            F_prev = F_curr
    F_final = lse_FG(best_h, tau)[0]
    if F_final < best_F:
        best_h = best_h.copy()
    return best_h


def slqp_lse(h0, tau, maxiter=200, ftol=1e-10, deadline=None):
    n = len(h0)
    h0 = h0.copy()

    def obj(x):
        return lse_FG(x, tau)[0]

    def grad(x):
        return lse_FG(x, tau)[1]

    def eq_fun(x):
        return x.sum() - 0.5 * n

    def eq_jac(x):
        return np.ones((1, n))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    if deadline is not None and (deadline - time.monotonic()) < 5.0:
        return h0
    try:
        res = optimize.minimize(
            obj, h0, jac=grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0}
        )
        if hasattr(res, 'x') and res.x is not None:
            x = np.asarray(res.x, dtype=float)
            if x.size == n and np.all(np.isfinite(x)):
                return project_box_sum(x, n)
    except Exception:
        pass
    return h0


def epigraph_solve_subset(h0, s_init, active, maxiter=80, ftol=1e-10):
    n = len(h0)
    h0 = h0.copy()
    cmax = exact_value(h0)
    s_lo = max(cmax - 0.02, 0.0)
    s_hi = s_init + 0.05

    def obj(x):
        return x[-1]

    def grad(x):
        g = np.zeros(x.size)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - 0.5 * n

    def eq_jac(x):
        J = np.zeros((1, x.size))
        J[0, :-1] = 1.0
        return J

    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        c = overlap_components(h)
        return x[-1] - c[active]

    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        dx = 2.0 / n
        k = len(active)
        J = np.zeros((k, x.size))
        for idx, m in enumerate(active):
            s = n - 1 - m
            i0 = max(0, -s)
            i1 = min(n - 1, n - 1 - s)
            if i0 <= i1:
                i = np.arange(i0, i1 + 1)
                j = i + s
                J[idx, i] -= dx * (1.0 - h[j])
                J[idx, j] += dx * h[i]
            J[idx, -1] = 1.0
        return J

    x0 = np.empty(n + 1)
    x0[:-1] = h0
    x0[-1] = s_init
    bounds = [(0.0, 1.0)] * n + [(s_lo, s_hi)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = optimize.minimize(
            obj, x0, jac=grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0}
        )
        if hasattr(res, 'x') and res.x is not None:
            x = np.asarray(res.x, dtype=float)
            if x.size == n + 1 and np.all(np.isfinite(x)):
                return project_box_sum(x[:-1], n)
    except Exception:
        pass
    return h0


def epigraph_active_set(h0, s_init, max_rounds=30, maxiter_per=60, ftol=1e-11, deadline=None, max_active=25):
    n = len(h0)
    h = h0.copy()
    cur_val = exact_value(h)
    for rnd in range(max_rounds):
        if deadline is not None and (deadline - time.monotonic()) < 8.0:
            break
        c = overlap_components(h)
        t = c.max()
        tol = max(0.002, 0.05 * (t - s_init + 0.001))
        active = np.where(c >= t - tol)[0]
        if len(active) > max_active:
            active = np.argsort(c)[::-1][:max_active]
        h_new = epigraph_solve_subset(h, s_init, active, maxiter_per, ftol)
        if h_new is None:
            break
        val_new = exact_value(h_new)
        if val_new >= cur_val - 1e-11:
            break
        h = h_new
        cur_val = val_new
    return h


def resample(h, n_new):
    n = len(h)
    if n <= 0:
        return None
    if n_new == n:
        return project_box_sum(h, n)
    if n_new < 4:
        return None
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)


def consider_candidate(best, h):

    h = np.asarray(h, dtype=float).reshape(-1)
    if not np.all(np.isfinite(h)):

    h = project_box_sum(h, len(h))
    v = exact_value(h)
    if not math.isfinite(v):

    cur = best['val']
    if v < cur - 1e-11:

        best['val'] = v
        best['n'] = len(h)
    elif abs(v - cur) <= 1e-11 and len(h) < best['n']:
        best['h'] = h.copy()
        best['val'] = v
        best['n'] = len(h)
    return best

            if best['val'] < 0.42 and deadline - time.monotonic() > 35.0:
                h_ref, v_ref = local_refine(hr, deadline, budget_frac=0.5)
                consider_candidate(best, h_ref)

    # Phase 4: Final epigraph polish
    if best['h'] is not None and deadline - time.monotonic() > 15.0:
        for margin in [0.0003, 0.0001, 0.00005, 0.00002]:
            if deadline - time.monotonic() < 12.0:
                break
            h_new = epigraph_active_set(
                best['h'], best['val'] + margin,
                max_rounds=15, maxiter_per=80,
                ftol=1e-12, deadline=deadline, max_active=20
            )
            consider_candidate(best, h_new)

    # Final clean
    final_h = best['h']
    if final_h is None:
        final_h = project_box_sum(np.full(45, 0.5), 45)
    final_h = project_box_sum(final_h, len(final_h))
    final_val = exact_value(final_h)
    if not math.isfinite(final_val):
        final_h = project_box_sum(np.full(45, 0.5), 45)
        final_val = exact_value(final_h)
    return final_h, final_val, len(final_h)
