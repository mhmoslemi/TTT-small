        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h0 = None

    if h0 is not None:
        n = len(h0)
        target = n / 2.0
        h0 = project_h(h0, target)
    else:
        n = 45
        target = n / 2.0
        h0 = np.full(n, 0.5)
        x = np.linspace(0, 2, n, endpoint=False)
        for f in range(1, 8):
            h0 += 0.08 * np.sin(np.pi * f * x + rng.random() * 2 * np.pi)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_h(h0, target)

    best_h = h0.copy()
    best_c = evaluate_C(best_h)

    # Phase 1: Top-K descent
    if remaining() > 60:
        for K in [5, 8, 5, 3, 5]:
            if remaining() < 20:
                break
            h_new, c_new = topk_descent(best_h, K, max_iters=600, target=target)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 2: Block polish width 2
    if remaining() > 50:
        h_new, c_new = block_polish(best_h, rng, n_trials=2000, w=2, target=target)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 3: Gradient-guided block moves
    if remaining() > 50:
        h_new, c_new = gradient_block_optimize(best_h, rng, K=5, w=2, n_trials=1500, target=target)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if remaining() > 30:
            h_new, c_new = gradient_block_optimize(best_h, rng, K=3, w=3, n_trials=800, target=target)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 4: Another top-K round
    if remaining() > 40:
        for K in [3, 5, 8]:
            if remaining() < 15:
                break
            h_new, c_new = topk_descent(best_h, K, max_iters=400, target=target)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 5: Block polish width 3
    if remaining() > 40:
        h_new, c_new = block_polish(best_h, rng, n_trials=1500, w=3, target=target)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 6: Parametric sub-optimization
    if remaining() > 60:
        h_new, c_new = parametric_search(best_h, rng, n_basis=6, max_iters=200, target=target)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            # Refine with top-K
            if remaining() > 30:
                h_new2, c_new2 = topk_descent(best_h, 5, max_iters=300, target=target)
                if c_new2 < best_c:
                    best_c = c_new2
                    best_h = h_new2.copy()

    # Phase 7: Perturbation restart
    if remaining() > 80:
        for attempt in range(3):
            if remaining() < 30:
                break
            h_pert = best_h + rng.normal(0, 0.02, len(best_h))
            h_pert = project_h(h_pert, target)
            c_pert = evaluate_C(h_pert)
            if c_pert < best_c + 0.005:
                h_new, c_new = topk_descent(h_pert, 5, max_iters=300, target=target)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                if remaining() > 20:
                    h_new, c_new = block_polish(best_h, rng, n_trials=500, w=2, target=target)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()

    # Phase 8: Try smaller n
    for n_try in [36, 30, 42]:
        if remaining() < 50:
            break
        h_try = np.interp(
            np.linspace(0, 2, n_try, endpoint=False),
            np.linspace(0, 2, len(best_h), endpoint=False),
            best_h
        )
        h_try = project_h(h_try, n_try / 2.0)
        c_try = evaluate_C(h_try)
        t1 = n_try / 2.0
        if remaining() > 20:
            h_new, c_new = topk_descent(h_try, 5, max_iters=300, target=t1)
            if c_new < c_try:
                c_try = c_new
                h_try = h_new
        if remaining() > 20:
            h_new, c_new = block_polish(h_try, rng, n_trials=800, w=2, target=t1)
            if c_new < c_try:
                c_try = c_new
                h_try = h_new
        if remaining() > 15:
            h_new, c_new = topk_descent(h_try, 3, max_iters=200, target=t1)
            if c_new < c_try:
                c_try = c_new
                h_try = h_new
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 9: Final intensive polish
    if remaining() > 20:
        n_final = len(best_h)
        t_final = n_final / 2.0
        h_new, c_new = topk_descent(best_h, 3, max_iters=300, target=t_final)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_new, c_new = block_polish(best_h, rng, n_trials=1000, w=2, target=t_final)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Final feasibility check
    nf = len(best_h)
    best_h = project_h(best_h, nf / 2.0)
    best_c = evaluate_C(best_h)

    return best_h, best_c, nf
