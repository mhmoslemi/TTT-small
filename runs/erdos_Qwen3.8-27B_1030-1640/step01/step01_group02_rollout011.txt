import time

def project_feasible(x, n):
    """Project x onto {0<=y<=1, sum(y)=n/2}"""
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h, n):
    """Compute exact C5 value."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_correlations(h, n):
    """Get all correlation values (not scaled by dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def lse_objective(h, lam, n):
    """Log-sum-exp approximation of max of correlations."""
    dx = 2.0 / n
    corr = get_correlations(h, n) * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def nesterov_optimize(h, n, lam, max_iter=2000, t_limit=None, t0=None):
    """Nesterov accelerated projected gradient method."""
    h = h.copy()
    y = h.copy()
    t = 1.0
    f_best = lse_objective(h, lam, n)
    h_best = h.copy()
    alpha = 1.0 / np.sqrt(lam)
    
    for it in range(max_iter):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        
        grad = compute_gradient(y, lam, n)
        # Gradient step
        y_new = y - alpha * grad
        # Project
        h_new = project_feasible(y_new, n)
        # Evaluate
        f_new = lse_objective(h_new, lam, n)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        h = h_new
        t = t_new
        alpha = 1.0 / (lam * t_new)
        
        if f_new < f_best:
            f_best = f_new
            h_best = h_new.copy()
    
    return h_best

                    h[j] -= d

                    h[j] -= best_d

import numpy as np
import time

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_objective(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def compute_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        s = k_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[: n - s])
                grad[: n - s] -= wk * h[s:]
        else:
            t = -s
            if t < n:
                grad[: n - t] += wk * (1.0 - h[t:])
                grad[t:] -= wk * h[: n - t]
    return grad * dx

def nesterov_opt(h, n, lam, max_iter=2000, t_limit=None, t0=None):
    h = h.copy().astype(float)
    y = h.copy()
    t_val = 1.0
    f_best = lse_objective(h, lam, n)
    h_best = h.copy()
    alpha0 = 2.0 / (lam * (2.0 / n))
    alpha0 = min(alpha0, 1.0)
    for it in range(max_iter):
        if t_limit is not None:
            if time.time() - t0 > t_limit:
                break
        grad = compute_grad(y, lam, n)
        y_new = y - alpha0 / t_val * grad
        h_new = project_feasible(y_new, n)
        f_new = lse_objective(h_new, lam, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        h = h_new
        t_val = t_new
        if f_new < f_best:
            f_best = f_new
            h_best = h_new.copy()
    return h_best

def exact_pair_descent(h, n, max_sweeps=60, t_limit=None, t0=None):
    h = h.copy().astype(float)
    best_c5 = evaluate_c5(h, n)
    for sweep in range(max_sweeps):
        if t_limit is not None:
            if time.time() - t0 > t_limit:
                break
        improved = False
        rng_sweep = np.random.default_rng(sweep * 7919 + 13)
        order = rng_sweep.permutation(n)
        for idx in range(n):
            i = order[idx]
            diffs = np.abs(h[i] - h).copy()
            diffs[i] = 0.0
            top_k = min(8, n - 1)
            top_j = np.argsort(diffs)[::-1][:top_k]
            found = False
            for j in top_j:
                j = int(j)
                if j == i or diffs[j] < 1e-6:
                    continue
                d_min = max(-h[i], h[j] - 1.0) + 1e-12
                d_max = min(1.0 - h[i], h[j]) - 1e-12
                if d_max <= d_min:
                    continue
                best_d = 0.0
                best_v = best_c5
                n_evals = 7
                for d in np.linspace(d_min, d_max, n_evals):
                    h[i] += d
                    h[j] -= d
                    v = evaluate_c5(h, n)
                    if v < best_v - 1e-12:
                        best_v = v
                        best_d = d
                    h[i] -= d
                    h[j] += d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    best_c5 = best_v
                    improved = True
                    found = True
                    break
            if found:
                continue
        if not improved:
            break
    return h, best_c5

def resample_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def subgradient_step(h, n, alpha, t_limit=None, t0=None):
    """One step of projected subgradient on exact max."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in active:
        s = k_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += (1.0 - h[: n - s])
                grad[: n - s] -= h[s:]
        else:
            t = -s
            if t < n:
                grad[: n - t] += (1.0 - h[t:])
                grad[t:] -= h[: n - t]
    grad *= dx
    h_new = project_feasible(h - alpha * grad, n)
    return h_new

def subgradient_opt(h, n, t_limit=None, t0=None):
    """Project

    h = h.copy().astype(float)
    best_c5 = evaluate_c5(h, n)
    h_best = h.copy()
    alpha = 0.1
    for it in range(3000):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        h_new = subgradient_step(h, n, alpha / (1.0 + 0.01 * it))
        c5 = evaluate_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h_new.copy()
        h = h_new
    return h_best, best_c5
