def pair_anneal(h, n, rng, t_start, time_limit, initial_temp=0.01):
    """Simulated annealing with pair moves"""
    h = h.copy()
    current_c5 = evaluate_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    temp = initial_temp
    n_iter = 0
    no_improve = 0
    
    while time.time() - t_start < time_limit and n_iter < 50000:
        n_iter += 1
        # Try a pair move
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        # Direction: move from larger to smaller
        if h[i] < h[j]:
            i, j = j, i
        
        max_d = min(1.0 - h[i], h[j])
        if max_d < 1e-8:
            continue
        
        # Random d in (0, max_d)
        d = rng.random() * max_d
        if d < 1e-8:
            continue
        
        h[i] += d
        h[j] -= d
        new_c5 = evaluate_c5(h, n)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] -= d
            h[j] += d
            no_improve += 1
        
        # Cool down
        if n_iter % 500 == 0:
            temp *= 0.7
        
        if no_improve > 5000:
            break
    
    return best_h, best_c5

  For i+k = b, i = b-k: was h[b-k]*1 = h[b-k]. Now h'[b-k]*0 = 0.

So Δ_k = -(1-h[a+k]) + h[a-k] + (1-h[b+k]) - h[b-k]
         = (h[a-k] - h[a+k]) + (h[b+k] - h[b-k])

(All terms with invalid indices are 0.)

This is a neat formula! It means for each pair (a,b), I can compute the change in ALL corr_k values in O(n) time (iterating over k), which is much faster than recomputing the full correlation.

But actually, for n=45, even recomputing the full correlation is fast enough. Let me not over-optimize and just use the direct evaluation.

import numpy as np

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def smooth_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n-s])
                grad[:n-s] -= wk * h[s:]
        else:
            if -s < n:
                grad[:n+s] += wk * (1.0 - h[-s:])
                grad[-s:] -= wk * h[:n+s]
    return grad * dx

def resample_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def pair_hill_climb(h, n, rng, t_start, time_limit):
    """Aggressive pair-swap hill climbing"""
    h = h.copy()
    c5 = evaluate_c5(h, n)
    no_improve = 0
    
    while time.time() - t_start < time_limit and no_improve < 2000:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if h[i] < h[j]:
            i, j = j, i
        max_d = min(1.0 - h[i], h[j])
        if max_d < 1e-8:
            no_improve += 1
            continue
        
        best_d = 0
        best_c5_new = c5
        # Try multiple fractions
        for frac in (0.1, 0.25, 0.5, 0.75, 1.0):
            d = frac * max_d
            h[i] += d
            h[j] -= d
            new_c5 = evaluate_c5(h, n)
            if new_c5 < best_c5_new:
                best_c5_new = new_c5
                best_d = d
            h[i] -= d
            h[j] += d
        
        if best_d > 0:
            h[i] += best_d
            h[j] -= best_d
            c5 = best_c5_new
            no_improve = 0
        else:
            no_improve += 1
    
    return h, c5

def binary_local_search(h_binary, n, rng, t_start, time_limit):
    """Local search on binary vectors"""
    h = h_binary.copy()
    c5 = evaluate_c5(h, n)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    no_improve = 0
    
    while time.time() - t_start < time_limit and no_improve < 3000:
        # Try random swap
        ia = int(rng.integers(len(ones)))
        ib = int(rng.integers(len(zeros)))
        a = ones[ia]
        b = zeros[ib]
        
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = evaluate_c5(h, n)
        
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
            # Update ones and zeros
            ones[ia] = b
            zeros[ib] = a
            no_improve = 0
        else:
            h[a] = 1.0
            h[b] = 0.0
            no_improve += 1
    
    return h, c5
