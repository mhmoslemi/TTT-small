    n = len(h)

    c_cur = evaluate_C(h_cur)
    
    for sweep in range(max_sweeps):
        # Compute soft-max weights for gradient
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        G = corr * (2.0/n)
        Gmax = np.max(G)
        exp_G = np.exp((G - Gmax) / theta)
        w = exp_G / np.sum(exp_G)
        
        # Compute gradient
        grad = np.zeros(n)
        offset = n - 1

            wm = w[m]
            if wm < 1e-15: continue

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def compute_soft_weights(h, theta):
    """Compute soft-max weights for the lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    G = corr * dx
    Gmax = np.max(G)
    if theta < 1e-15:
        mask = G >= Gmax - 1e-12
        w = np.zeros(2*n-1)
        w[mask] = 1.0 / max(1, np.sum(mask))
        return w, G
    exp_G = np.exp((G - Gmax) / theta)
    w = exp_G / np.sum(exp_G)
    return w, G

def compute_gradient(h, w, dx):
    """Compute gradient using given soft-max weights."""
    n = len(h)
    grad = np.zeros(n)
    offset = n - 1
    for m in range(2*n-1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - offset
        if k >= 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:n-k])
        else:
            l = -k
            grad[l:] += wm * (1.0 - h[:-l])
            grad[:n-l] += wm * (-h[l:])
    return grad

def optimal_pair_tau(h, s, t):
    """Analytically find optimal tau for h[s]-=tau, h[t]+=tau."""
    n = len(h)
    dx = 2.0 / n
    T = min(h[s], 1.0 - h[t])
    if T < 1e-14:
        return 0.0, np.inf
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    A = corr * dx
    NL = 2 * n - 1
    offset = n - 1
    
    # Compute deltas: dA_k/dtau for the pair move
    deltas = np.zeros(NL)
    ks = np.arange(-(n-1), n)
    for idx in range(NL):
        k = int(ks[idx])
        gs = 0.0
        gt = 0.0
        si = s + k
        if 0 <= si < n:
            gs += dx * (1.0 - h[si])
        si2 = s - k
        if 0 <= si2 < n:
            gs -= dx * h[si2]
        ti = t + k
        if 0 <= ti < n:
            gt += dx * (1.0 - h[ti])
        ti2 = t - k
        if 0 <= ti2 < n:
            gt -= dx * h[ti2]
        deltas[idx] = gt - gs
    
    # Minimize max(A + tau*deltas) over tau in [0, T]
    # Find all pairwise intersection points
    num = A[None, :] - A[:, None]  # (NL,NL), [i,j] = A[j]-A[i]
    den = deltas[:, None] - deltas[None, :]  # (NL,NL), [i,j] = deltas[i]-deltas[j]
    
    valid = np.abs(den) > 1e-15
    with np.errstate(divide='ignore', invalid='ignore'):
        taus = np.where(valid, num / den, np.nan)
    
    valid = valid & (taus >= -1e-12) & (taus <= T + 1e-12) & np.isfinite(taus)
    taus_valid = np.clip(taus[valid], 0.0, T)
    
    all_candidates = np.concatenate(([0.0, T], taus_valid))
    
    # Evaluate max at all candidates (vectorized)
    vals = np.max(A[None, :] + all_candidates[:, None] * deltas[None, :], axis=1)
    idx = int(np.argmin(vals))
    return float(all_candidates[idx]), float(vals[idx])

def pair_sweep(h, theta, n_sweeps, top_k, rng):
    """Main optimization loop."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c_cur = evaluate_C(h_cur)
    
    for sweep in range(n_sweeps):
        # Compute weights and gradient
        w, G = compute_soft_weights(h_cur, theta)
        grad = compute_gradient(h_cur, w, dx)
        
        # Select candidate pairs
        s_mask = h_cur > 1e-8
        t_mask = h_cur < 1.0 - 1e-8
        s_cands = np.where(s_mask)[0]
        t_cands = np.where(t_mask)[0]
        
        if len(s_cands) == 0 or len(t_cands) == 0:
            break
        
        # Top-k selection
        ks = min(top_k, len(s_cands))
        kt = min(top_k, len(t_cands))
        top_s = s_cands[np.argsort(grad[s_cands])[-ks:]]
        top_t = t_cands[np.argsort(grad[t_cands])[:kt]]
        
        best_c = c_cur
        best_move = None
        
        for s in top_s:
            for t in top_t:
                if s == t:
                    continue
                tau, c = optimal_pair_tau(h_cur, s, t)
                if c < best_c - 1e-15:
                    best_c = c
                    best_move = (s, t, tau)
        
        # If no improvement from top-k, try some random pairs
        if best_move is None:
            for _ in range(min(30, len(s_cands) * len(t_cands))):

            h_cur[s] -= tau
            h_cur[t] += tau
            c_cur = best_c
