def evaluate_C(h):

def compute_lag_gradient(h, k, n):
    """
    Compute gradient of C_k with respect to h.
    k is the lag: k = m - (n-1) where m is the index in the full correlation.
    k ranges from -(n-1) to (n-1).
    Returns grad[j] for j = 0, ..., n-1.
    """
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        # C_k = sum_{t=0}^{n-1-k} h[t+k]*(1-h[t])
        # d/dh[j]: (1-h[j-k]) if j>=k, plus -h[j+k] if j<n-k
        if n - 1 - k >= 0:
            # j in [k, n-1]: (1-h[j-k])
            grad[k:] += (1.0 - h[:n-k])
            # j in [0, n-1-k]: -h[j+k]
            grad[:n-k] += (-h[k:])
    else:
        m = -k  # m > 0
        # C_{-m} = sum_{t=m}^{n-1} h[t-m]*(1-h[t])
        # d/dh[j]: (1-h[j+m]) if j < n-m, plus -h[j-m] if j >= m
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += (-h[:n-m])
    return grad

def subgradient_step(h, target, step_size, active_tol=0.005):
    """
    One subgradient step with active set identification.
    Returns new h and the C value.
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    Cmax = np.max(corr)
    
    # Active set: lags within active_tol of max
    active = np.where(corr >= Cmax - active_tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    # Compute average gradient of active lags
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        grad += compute_lag_gradient(h, k, n)
    grad /= len(active)
    
    # Step
    y = h - step_size * grad
    h_new = project_h(y, target)
    return h_new, Cmax

                        step_size_decay=0.999, momentum=0.9, active_tol=0.005):

    Subgradient descent with momentum and Polyak averaging.
    """
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    step_size = step_size_init
    
    # Polyak averaging
    h_avg = h_cur.copy()
    
    for it in range(n_steps):
        h_new, Cmax = subgradient_step(h_cur, target, step_size, active_tol)
        
        # Momentum
        v = momentum * v + (h_new - h_cur)
        h_mom = h_cur + v
        h_mom = project_h(h_mom, target)
        
        c_check = evaluate_C(h_mom)
        
        if c_check < best_c:
            best_c = c_check
            best_h = h_mom.copy()
        
        # Update
        h_cur = h_mom
        h_avg = h_avg + (h_cur - h_avg) / (it + 1)
        
        # Also check averaged point
        c_avg = evaluate_C(project_h(h_avg, target))
        if c_avg < best_c:
            best_c = c_avg
            best_h = project_h(h_avg, target).copy()
        
        step_size *= step_size_decay
    
    return best_h, best_c

def project_h(y, target):

        s = np.sum(np.clip(y + mid, 0.0, 1.0))

        else:
            hi = mid

def compute_lag_gradient(h, k, n):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += (-h[k:])
    else:
        m = -k
        grad[:n-m] += (1.0 - h[m:])

def compute_active_gradient(h, top_k=8, tol=0.003):
    """Compute weighted gradient of the top lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    Cmax = np.max(corr)
    
    # Find active lags
    active_mask = corr >= Cmax - tol
    active_idx = np.where(active_mask)[0]
    
    # If too many active lags, take top top_k
    if len(active_idx) > top_k:
        top_active = corr[active_idx]
        sorted_idx = np.argsort(top_active)[::-1][:top_k]
        active_idx = active_idx[sorted_idx]
        weights = top_active[sorted_idx]
        # Normalize weights
        weights = weights / weights.sum()
    else:
        # Use exponential weighting based on proximity to max
        diffs = Cmax - corr[active_idx]
        weights = np.exp(-diffs / (tol * 0.5))
        weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for i, m in enumerate(active_idx):
        k = m - (n - 1)
        grad += weights[i] * compute_lag_gradient(h, k, n)
    
    return grad, Cmax, len(active_idx)

def subgrad_momentum(h, target, n_steps, step_init, decay, momentum, tol, top_k):

    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    step = step_init
    h_avg = h_cur.copy()

def block_descent(h, target, block_size=4, n_sweeps=10):
    """Block coordinate descent: optimize each block of size block_size."""
    n = len(h)
    from scipy.optimize import minimize as sp_min
    
    h_cur = h.copy()
    best_c = evaluate_C(h_cur)
    
    for sweep in range(n_sweeps):
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            end = start + block_size
            if end > n:
                end = n
            bs = end - start
            
            # Fixed parts
            h_fixed = h_cur.copy()
            fixed_sum = h_fixed.sum() - h_fixed[start:end].sum()
            block_target = target - fixed_sum
            
            def block_obj(x):
                trial = h_fixed.copy()
                trial[start:end] = x
                return evaluate_C(trial)
            
            # Constraint: sum of block = block_target, 0 <= x <= 1
            x0 = h_cur[start:end].copy()
            
            # Try to minimize using SLSQP
            try:
                cons = [{'type': 'eq', 'fun': lambda x: x.sum() - block_target,
                         'jac': lambda x: np.ones(bs)}]
                bounds = [(0, 1)] * bs
                res = sp_min(block_obj, x0, method='SLSQP', bounds=bounds,
                            constraints=cons, options={'maxiter': 50, 'ftol': 1e-10})
                if res.success and block_obj(res.x) < best_c - 1e-12:
                    h_cur[start:end] = res.x
                    best_c = block_obj(res.x)
                    improved = True
            except:
                pass
        
        if not improved:
            break
    
    return h_cur, best_c

def pair_polish(h, rng, n_pairs, max_iter=70, target=None):
    """Golden section search on pairs of coordinates."""
    n = len(h)
    if target is None:
        target = n / 2.0
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    
    for it in range(n_pairs):
        # Choose a pair
        s = rng.integers(n)
        t = rng.integers(n)
        if s == t:
            continue
        
        T = min(h_cur[s], 1.0 - h_cur[t])
        if T < 1e-10:
            continue
        
        a, b = 0.0, T
        c1 = b - invphi * (b - a)
        c2 = a + invphi * (b - a)
        
        def ev(tau):
            ht = h_cur.copy()
            ht[s] -= tau
            ht[t] += tau
            return evaluate_C(ht)
        
        f1 = ev(c1)
        f2 = ev(c2)
        best_val = min(f1, f2, c_cur)
        best_tau = 0.0
        if f1 < best_val:
            best_val, best_tau = f1, c1
        if f2 < best_val:
            best_val, best_tau = f2, c2
        
        for _ in range(max_iter):
            if b - a < 1e-12 * (T + 1.0):
                break
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1
                c1 = b - invphi * (b - a); f1 = ev(c1)
            else:
                a = c1; c1 = c2; f1 = f2
                c2 = a + invphi * (b - a); f2 = ev(c2)
            if f1 < best_val:

            if f2 < best_val:
                best_val, best_tau = f2, c2

            h_cur[s] -= best_tau
            h_cur[t] += best_tau
            c_cur = best_val
    
    return h_cur, c_cur
