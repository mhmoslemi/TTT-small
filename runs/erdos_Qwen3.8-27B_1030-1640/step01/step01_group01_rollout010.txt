import time
import math
import numpy as np
from scipy.optimize import linprog

def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0:
        return float('inf')
    return float(np.max(vals))

def c5_vec(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _waterfill(v, S):
    v = np.asarray(v, dtype=float).reshape(-1)
    n = v.size
    if n == 0:
        return v
    lo, hi = float(v.min()) - 1.5, float(v.max()) + 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) < S:
            hi = mid
        else:
            lo = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h
    S = 0.5 * float(n)
    if not np.all(np.isfinite(h)):
        return np.full(n, 0.5)
    h = _waterfill(h, S)
    for _ in range(n + 20):
        r = S - float(np.sum(h))
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(r, 1.0 - h[idx])
            if d > 0: h[idx] += d
        else:
            idx = int(np.argmin(h))
            d = min(-r, h[idx])
            if d > 0: h[idx] -= d
    np.clip(h, 0.0, 1.0, out=h)
    return h

def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    return ensure_feasible(0.5 * (h + h[::-1]))

def grad_matrix(h):
    """Returns G (L x n) gradient matrix for all lags, including dx."""
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    for m in range(L):
        shift = n - 1 - m
        start = max(0, -shift)
        end = min(n - 1, n - 1 - shift)
        if start > end:
            continue
        i = np.arange(start, end + 1)
        j = i + shift
        if shift == 0:
            G[m, i] = dx * (1.0 - 2.0 * h[i])
        else:
            np.add.at(G[m], i, dx * (1.0 - h[j]))
            np.add.at(G[m], j, -dx * h[i])
    return G

def resample(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(max(2, n_new))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

def lp_step(h0, radius, margin=1e-10):
    """Solve linearized minimax LP. Returns h_lp or None."""
    h0 = np.asarray(h0, dtype=float).reshape(-1)
    n = h0.size
    L = 2 * n - 1
    S = 0.5 * float(n)
    
    G = grad_matrix(h0)
    c0 = c5_vec(h0)
    cmax = float(np.max(c0))
    
    # Build LP
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c0 - margin  # G[m]·h - t <= G[m]·h0 - c0[m] - margin
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = []
    for i in range(n):
        lb = max(0.0, h0[i] - radius)
        ub = min(1.0, h0[i] + radius)
        if lb > ub:
            lb = ub = h0[i]
        bounds.append((lb, ub))
    bounds.append((0.0, 2.0))  # t
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n].copy()
        elif res.x is not None and np.all(np.isfinite(res.x)):
            # Even if not "success", use the solution if it looks reasonable
            return res.x[:n].copy()
    except Exception:
        pass
    return None

import time
import math
import numpy as np
from scipy.optimize import linprog


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0 or not np.all(np.isfinite(vals)):
        return float('inf')
    return float(np.max(vals))


def c5_full_vec(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _waterfill(v, S):
    v = np.asarray(v, dtype=float).reshape(-1)
    n = v.size
    if n == 0:
        return v
    lo, hi = float(v.min()) - 1.5, float(v.max()) + 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) < S:
            hi = mid
        else:
            lo = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h
    S = 0.5 * float(n)
    if not np.all(np.isfinite(h)):
        return np.full(n, 0.5)
    h = _waterfill(h, S)
    for _ in range(n + 30):
        r = S - float(np.sum(h))
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(r, 1.0 - h[idx])
            if d > 0:
                h[idx] += d
            else:
                break
        else:
            idx = int(np.argmin(h))
            d = min(-r, h[idx])
            if d > 0:
                h[idx] -= d
            else:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h


def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    return ensure_feasible(0.5 * (h + h[::-1]))


def grad_matrix(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    for m in range(L):
        shift = n - 1 - m
        start = max(0, -shift)
        end = min(n - 1, n - 1 - shift)
        if start > end:
            continue
        i = np.arange(start, end + 1)
        j = i + shift
        if shift == 0:
            G[m, i] = dx * (1.0 - 2.0 * h[i])
        else:
            np.add.at(G[m], i, dx * (1.0 - h[j]))
            np.add.at(G[m], j, -dx * h[i])
    return G


def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(max(2, n_new))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return ensure_feasible(y)


def lp_step(h0, radius, margin=1e-10):
    h0 = np.asarray(h0, dtype=float).reshape(-1)
    n = h0.size
    if n < 2:
        return None
    L = 2 * n - 1
    S = 0.5 * float(n)
    G = grad_matrix(h0)
    c0 = c5_full_vec(h0)
    if not np.all(np.isfinite(G)) or not np.all(np.isfinite(c0)):
        return None
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c0 - margin
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = []
    for i in range(n):
        lb = max(0.0, h0[i] - radius)
        ub = min(1.0, h0[i] + radius)
        if lb > ub:
            lb = ub = h0[i]
        bounds.append((lb, ub))
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n].copy()
    except Exception:
        pass
    return None
