            b[best_j] = 1.0

def pair_move_line_search(h, n, i, j):
    """Find best δ for moving mass from j to i."""
    # δ ∈ [δ_min, δ_max]
    # h[i] + δ ∈ [0, 1] => δ ∈ [-h[i], 1-h[i]]
    # h[j] - δ ∈ [0, 1] => δ ∈ [h[j]-1, h[j]]
    δ_min = max(-h[i], h[j] - 1)
    δ_max = min(1 - h[i], h[j])
    if δ_max <= δ_min:
        return h, evaluate_c5(h, n)
    
    # Evaluate at a grid of points
    n_grid = 20
    δ_vals = np.linspace(δ_min, δ_max, n_grid)
    best_c5 = np.inf
    best_δ = 0
    for δ in δ_vals:
        h[i] += δ
        h[j] -= δ
        c = evaluate_c5(h, n)
        if c < best_c5:
            best_c5 = c
            best_δ = δ
        h[i] -= δ
        h[j] += δ
    # Refine around best
    if best_δ > 0:
        range_ = (δ_max - δ_min) / n_grid
        δ_fine = np.linspace(best_δ - range_, best_δ + range_, 10)
        for δ in δ_fine:
            if δ < δ_min or δ > δ_max:
                continue
            h[i] += δ
            h[j] -= δ
            c = evaluate_c5(h, n)
            if c < best_c5:
                best_c5 = c
                best_δ = δ
            h[i] -= δ
            h[j] += δ
    h[i] += best_δ
    h[j] -= best_δ
    return h, best_c5

import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_steepest(b, n, max_iter=2000, timeout=20.0):
    b = b.astype(float).copy()
    c5 = evaluate_c5(b, n)
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > timeout:
            break
        improved = False
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        best_c5 = c5
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = evaluate_c5(b, n)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_i, best_j = i, j
                b[i] = 1.0
                b[j] = 0.0
        if best_i >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            c5 = best_c5
            improved = True
        if not improved:
            break
    return b, c5

def binary_anneal(b_init, n, n_iter=20000, T=0.005, timeout=20.0, rng=None):
    b = b_init.astype(float).copy()
    c5 = evaluate_c5(b, n)
    best_b = b.copy()
    best_c5 = c5
    t0 = time.time()
    for _ in range(n_iter):
        if time.time() - t0 > timeout:
            break
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        c5_new = evaluate_c5(b, n)
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / T):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c5

def continuous_refine(h, n, n_iter=500, timeout=15.0, rng=None):
    """Pair-move line search on continuous h."""
    h = h.copy().astype(float)
    c5 = evaluate_c5(h, n)
    t0 = time.time()
    alpha = 1.0
    for _ in range(n_iter):
        if time.time() - t0 > timeout:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Line search: move mass from j to i
        d_min = max(-h[i], h[j] - 1.0)
        d_max = min(1.0 - h[i], h[j])
        if d_max - d_min < 1e-12:
            continue
        # Try a few values of δ
        best_c = c5
        best_d = 0.0
        for d in np.linspace(d_min, d_max, 15):

            h[j] -= d
            c = evaluate_c5(h, n)

def subgradient_refine(h, n, n_iter=2000, timeout=15.0):
    """Projected subgradient method targeting the bottleneck."""
    h = h.copy().astype(float)
    c5 = evaluate_c5(h, n)
    dx = 2.0 / n
    t0 = time.time()
    step = 0.05
    for _ in range(n_iter):
        if time.time() - t0 > timeout:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_max = np.argmax(corr)
        s = k_max - (n - 1)
        grad = np.zeros(n)
        if s >= 0 and s < n:
            m = n - s
            grad[:m] += (1.0 - h[s:])
            grad[s:] -= h[:m]
        elif -s < n and s < 0:
            t = -s
            m = n - t
            grad[t:] += (1.0 - h[:m])
            grad[:m] -= h[t:]
        h_new = project_feasible(h - step * grad, n)
        c5_new = evaluate_c5(h_new, n)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.7
    return h, c5

def resample(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def lse_soft_max(corr, lam):
    m = np.max(corr)
    if lam * (m - m) > 500:  # shouldn't happen
        return m
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def smooth_pg(h, n, lam, max_iter=500, timeout=10.0):
    """Projected gradient with log-sum-exp smoothing."""
    h = h.copy().astype(float)
    dx = 2.0 / n
    f = lse_soft_max(np.correlate(h, 1.0 - h, mode="full") * dx, lam)
    alpha = 1.0
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > timeout:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        scaled = lam * (corr - m)
        ms = np.max(scaled)
        if ms > 300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            ev = np.exp(scaled - ms)
            w = ev / ev.sum()
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            if s >= 0 and s < n:
                m_ = n - s
                grad[:m_] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:m_]
            elif -s < n and s < 0:
                t = -s
                m_ = n - t
                grad[t:] += wk * (1.0 - h[:m_])
                grad[:m_] -= wk * h[t:]
        grad *= dx
        # Projected gradient step
        h_new = project_feasible(h - alpha * grad, n)
        f_new = lse_soft_max(np.correlate(h_new, 1.0 - h_new, mode="full") * dx, lam)
        if f_new < f:
            h = h_new
            f = f_new
            alpha = min(alpha * 1.2, 5.0)
        else:
            alpha *= 0.5
    return h
