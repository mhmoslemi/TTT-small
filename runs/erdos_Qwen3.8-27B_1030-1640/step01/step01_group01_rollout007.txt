def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

def adjust_sum_exact(h, S):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    if n == 0:
        return h
    for _ in range(200):
        r = S - float(np.sum(h))
        if abs(r) <= 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if h[idx] >= 1.0:
                break
            delta = min(r, 1.0 - h[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmin(h))
            if h[idx] <= 0.0:
                break
            delta = min(-r, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    np.clip(h, 0.0, 1.0, out=h)
    # Final correction
    for _ in range(5000):
        r = S - float(np.sum(h))
        if abs(r) <= 1e-15:
            break
        if r > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            idxs = np.where(mask)[0]
            idx = int(idxs[0])
            h[idx] = min(1.0, h[idx] + r)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            idxs = np.where(mask)[0]
            idx = int(idxs[0])
            h[idx] = max(0.0, h[idx] + r)
    return h

def project_box_sum(v):
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * float(n)
    if not np.all(np.isfinite(v)):
        return adjust_sum_exact(np.full(n, 0.5), S)
    
    def sum_at(lam):
        return float(np.sum(np.clip(v - lam, 0.0, 1.0)))
    
    lo, hi = -1.0, 2.0
    for _ in range(20):
        if sum_at(lo) < S:
            lo -= 1.0
    for _ in range(20):
        if sum_at(hi) > S:
            hi += 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if sum_at(mid) < S:
            hi = mid
        else:
            lo = mid
    p = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    return adjust_sum_exact(p, S)

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return np.array([0.5, 0.5])
    S = 0.5 * float(n)
    if not np.all(np.isfinite(h)):
        return adjust_sum_exact(np.full(n, 0.5), S)
    np.clip(h, 0.0, 1.0, out=h)
    if abs(float(np.sum(h)) - S) <= 1e-10:
        return adjust_sum_exact(h, S)
    return project_box_sum(h)

def symmetrize_h(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h
    return adjust_sum_exact(0.5*(h + h[::-1]), 0.5*float(n))

def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = max(2, int(n_new))
    if n_old == 0:
        return adjust_sum_exact(np.full(n_new, 0.5), 0.5*float(n_new))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0/float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0/float(n_new))
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

def compute_overlaps(h):
    """Compute all 2n-1 overlap values (correlation with 1-h, times dx)."""
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def gradient_at_shift(h, shift):
    """
    Compute gradient of the overlap at a specific shift.
    shift is the actual shift (can be negative or positive).
    The overlap at shift s is:
    c(s) = sum_{i: 0<=i<n, 0<=i+s<n} h[i]*(1-h[i+s]) * dx
    
    Gradient: dc/dh[j] = dx * [(1-h[j+s])*1_{valid pair (j, j+s)} - h[j-s]*1_{valid pair (j-s, j)}]
    """
    n = len(h)
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    s = int(shift)
    
    # Valid pairs: i where 0<=i<n and 0<=i+s<n
    i_min = max(0, -s)
    i_max = min(n-1, n-1-s)
    if i_min > i_max:
        return grad
    
    idx1 = np.arange(i_min, i_max+1)
    idx2 = idx1 + s
    
    # d/dh[i] gets (1-h[i+s])*dx for each valid pair (i, i+s)
    # d/dh[i+s] gets -h[i]*dx
    grad[idx1] += (1.0 - h[idx2]) * dx
    grad[idx2] += (-h[idx1]) * dx
    
    return grad

def all_gradients(h):
    """Compute gradients for all shifts. Returns array of shape (2n-1, n)."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        shift = idx - (n-1)
        grads[idx] = gradient_at_shift(h, shift)
    return grads

def nesterov_subgradient(h0, max_iter, t_end, rng):
    """
    Nesterov-accelerated projected subgradient method.
    """
    n = len(h0)
    h = ensure_feasible(h0).copy()
    best_h = h.copy()
    best_c = exact_c5(h)
    
    x_prev = h.copy()
    t_momentum = 0.0
    
    # Initial step size
    c_vals = compute_overlaps(h)
    t_current = float(np.max(c_vals))
    step = (t_current - best_c + 0.01) / (1e-6 + 1.0)  # initial
    
    for it in range(max_iter):
        if time.monotonic() > t_end:
            break
        
        # Compute overlaps
        c_vals = compute_overlaps(h)
        t_current = float(np.max(c_vals))
        
        # Update best
        if t_current < best_c:
            best_c = t_current
            best_h = h.copy()
        
        # Choose which gradient to use
        # Mix: mostly argmax, sometimes top-3 average, sometimes random
        r = rng.random()
        if r < 0.6:
            k_star = int(np.argmax(c_vals))
            g = gradient_at_shift(h, k_star - (n-1))
        elif r < 0.85:
            # Top 3 weighted
            top_idx = np.argsort(c_vals)[-3:][::-1]
            weights = c_vals[top_idx]
            if weights.sum() > 0:
                weights = weights / weights.sum()
            else:
                weights = np.ones(3) / 3
            g = np.zeros(n)
            for wi, ki in zip(weights, top_idx):
                g += wi * gradient_at_shift(h, int(ki) - (n-1))
        else:
            # Random
            k_rand = int(rng.integers(0, 2*n-1))
            g = gradient_at_shift(h, k_rand - (n-1))
        
        # Polyak step size
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq > 1e-15:
            alpha = max(1e-8, (t_current - best_c + 0.005) / g_norm_sq)
        else:
            alpha = 0.01
        
        # Cap step size
        alpha = min(alpha, 0.5)
        
        # Nesterov step
        if t_momentum > 0:
            y = h + t_momentum * (h - x_prev)
        else:
            y = h
        
        x_prev = h.copy()
        h_new = project_box_sum(y - alpha * g)
        
        # Accept or reject (simple: always accept for subgradient)
        h = h_new
        
        # Update momentum
        t_momentum = (1.0 + t_momentum) / (2.0 + t_momentum) * 0.5
        
    return best_h, best_c

def linearized_lp_step(h, delta_max=1.0):
    """
    Solve the linearized LP:
    minimize t
    s.t. f_k(h) + g_k · (h' - h) <= t for all k
         sum(h') = n/2
         0 <= h' <= 1
    
    Returns (h_new, t_LP) or (None, None) if infeasible.
    """
    n = len(h)
    c_vals = compute_overlaps(h)  # shape (2n-1,)
    L = 2*n - 1
    S = 0.5 * float(n)
    
    # Compute all gradients
    G = all_gradients(h)  # shape (L, n)
    
    # LP: min t
    # s.t. G @ h' - t <= G @ h - c_vals  (for each row k)
    #      sum(h') = S
    #      0 <= h' <= 1
    #      -inf <= t <= inf (or some bound)
    
    # Variable vector: [h'(0), ..., h'(n-1), t]  → n+1 variables
    # Objective: minimize t → c_obj = [0,...,0, 1]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G @ h' - t <= G @ h - c_vals
    # A_ub has L rows
    # Each row: [g_k[0], ..., g_k[n-1], -1] @ [h', t] <= g_k·h - c_k
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - c_vals
    
    # Equality: sum(h') = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: 0 <= h'[i] <= 1, -inf <= t <= inf
    bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
        if res.success and res.x is not None:
            h_new = ensure_feasible(res.x[:n])
            t_lp = float(res.x[-1])
            return h_new, t_lp
    except Exception:
        pass
    
    return None, None

def lp_line_search(h, h_target, max_alphas=10):
    """
    Backtracking line search from h toward h_target.
    Returns best point found.
    """
    c_current = exact_c5(h)
    best_point = h.copy()
    best_val = c_current
    
    for i in range(max_alphas):
        alpha = 2.0 ** (-i)  # 1, 0.5, 0.25, ...
        h_trial = h + alpha * (h_target - h)
        h_trial = ensure_feasible(h_trial)
        c_trial = exact_c5(h_trial)
        if c_trial < best_val - 1e-10:
            best_val = c_trial
            best_point = h_trial.copy()
        if alpha < 0.01:
            break
    
    return best_point, best_val

def polish_slsqp(h0, maxiter=80, ftol=1e-12):
    """
    Polish using SLSQP with epigraph formulation.
    We use the constraint t >= c_k(h) for all k,
    but to make it differentiable, we use a smooth approximation
    or just rely on SLSQP's finite-difference gradients.
    """
    n = len(h0)
    h0 = ensure_feasible(h0)
    c0 = exact_c5(h0)
    
    def obj_full(x):
        t = float(x[-1])
        return t
    
    def obj_jac_full(x):
        j = np.zeros(x.size)
        j[-1] = 1.0
        return j
    
    def constr_eq(x):
        return float(np.sum(x[:n]) - 0.5 * n)
    
    def constr_eq_jac(x):
        j = np.zeros(x.size)
        j[:n] = 1.0
        return j
    
    # Inequality: t - max_k c_k(h) >= 0
    # Use smooth max approximation with small tau
    tau = 0.001
    
    def ineq_soft(x):
        h = x[:n]
        c = compute_overlaps(h)
        z = c / tau
        zmax = float(np.max(z))
        s = float(np.sum(np.exp(z - zmax)))
        if s < 1e-300:
            smax = zmax
        else:
            smax = zmax + math.log(s)
        return float(x[-1]) - tau * smax
    
    def ineq_soft_jac(x):
        h = x[:n]
        c = compute_overlaps(h)
        z = c / tau
        zmax = float(np.max(z))
        w = np.exp(z - zmax)
        s = float(np.sum(w))
        if s > 1e-300:
            w = w / s
        else:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        
        # d(soft_max)/dh[i] = sum_k w_k * dc_k/dh[i]
        # dc_k/dh[i] for the k-th correlation value
        # k-th value corresponds to shift = k - (n-1)
        jac_h = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-12:
                continue
            shift = k - (n-1)
            g = gradient_at_shift(h, shift)
            jac_h += w[k] * g
        j = np.zeros(x.size)
        j[:n] = -tau * jac_h
        j[-1] = 1.0
        return j
    
    # Initial t
    t0 = c0 + 0.001
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.05)]
    
    cons = [
        {"type": "eq", "fun": constr_eq, "jac": constr_eq_jac},
        {"type": "ineq", "fun": ineq_soft, "jac": ineq_soft_jac},
    ]
    
    try:
        res = minimize(
            obj_full, x0, jac=obj_jac_full,
            method="SLSQP", bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": ftol, "iprint": 0}
        )
        if res.success or (res.x is not None and np.all(np.isfinite(res.x))):
            return ensure_feasible(res.x[:n])
    except Exception:
        pass
    return None

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    if vals.size == 0 or not np.all(np.isfinite(vals)):
        return float("inf")
    return float(np.max(vals))


def adjust_sum_exact(h, S):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    if n == 0:
        return h
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(300):
        r = S - float(np.sum(h))
        if abs(r) <= 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if h[idx] >= 1.0:
                break
            h[idx] = min(1.0, h[idx] + r)
        else:
            idx = int(np.argmin(h))
            if h[idx] <= 0.0:
                break
            h[idx] = max(0.0, h[idx] + r)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(10000):
        r = S - float(np.sum(h))
        if abs(r) <= 1e-15:
            break
        if r > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            idx = int(np.where(mask)[0][0])
            h[idx] = min(1.0, h[idx] + r)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            idx = int(np.where(mask)[0][0])
            h[idx] = max(0.0, h[idx] + r)
    return h


def project_box_sum(v):
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * float(n)
    if not np.all(np.isfinite(v)):
        return adjust_sum_exact(np.full(n, 0.5), S)

    def sum_at(lam):
        return float(np.sum(np.clip(v - lam, 0.0, 1.0)))

    lo, hi = -2.0, 3.0
    for _ in range(20):
        if sum_at(lo) < S:
            lo -= 1.0
    for _ in range(20):
        if sum_at(hi) > S:
            hi += 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if sum_at(mid) < S:
            hi = mid
        else:
            lo = mid
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum_exact(p, S)


def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return np.array([0.5, 0.5])
    S = 0.5 * float(n)
    if not np.all(np.isfinite(h)):
        return adjust_sum_exact(np.full(n, 0.5), S)
    np.clip(h, 0.0, 1.0, out=h)
    if abs(float(np.sum(h)) - S) <= 1e-10:
        return adjust_sum_exact(h, S)
    return project_box_sum(h)


def symmetrize_h(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h
    return adjust_sum_exact(0.5 * (h + h[::-1]), 0.5 * float(n))


def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = max(2, int(n_new))
    if n_old == 0:
        return adjust_sum_exact(np.full(n_new, 0.5), 0.5 * float(n_new))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return ensure_feasible(y)


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def gradient_at_shift(h, s):
    n = len(h)
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n - 1, n - 1 - s)
    if i_min > i_max:
        return grad
    idx1 = np.arange(i_min, i_max + 1)
    idx2 = idx1 + s
    grad[idx1] += (1.0 - h[idx2]) * dx
    grad[idx2] += (-h[idx1]) * dx
    return grad


def nesterov_subgradient(h0, t_end, rng, max_iter=200000):
    n = len(h0)
    h = ensure_feasible(h0).copy()
    best_h = h.copy()
    best_c = exact_c5(h)
    x_prev = h.copy()
    mom = 0.0

    for it in range(max_iter):
        if time.monotonic() > t_end:
            break
        c_vals = compute_overlaps(h)
        t_cur = float(np.max(c_vals))
        if t_cur < best_c:
            best_c = t_cur
            best_h = h.copy()

        r = rng.random()
        if r < 0.55:
            k_star = int(np.argmax(c_vals))
            g = gradient_at_shift(h, k_star - (n - 1))
        elif r < 0.8:
            top_idx = np.argsort(c_vals)[-4:][::-1]
            w = c_vals[top_idx]
            ws = float(w.sum())
            if ws > 1e-15:
                w = w / ws
            else:
                w = np.ones(len(top_idx)) / len(top_idx)
            g = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                if wi > 1e-12:
                    g += float(wi) * gradient_at_shift(h, int(ki) - (n - 1))
        else:
            k_r = int(rng.integers(0, 2 * n - 1))
            g = gradient_at_shift(h, k_r - (n - 1))

        g_sq = float(np.dot(g, g))
        if g_sq > 1e-15:
            alpha = max(1e-9, (t_cur - best_c + 0.008) / g_sq)
        else:
            alpha = 0.005
        alpha = min(alpha, 0.4)

        if mom > 0:
            y = h + mom * (h - x_prev)
        else:
            y = h
        x_prev = h.copy()
        h_new = project_box_sum(y - alpha * g)
        h = h_new
        mom = (1.0 + mom) / (2.0 + mom) * 0.6

    return best_h, best_c


def lp_linearize_step(h):
    n = len(h)
    c_vals = compute_overlaps(h)
    L = 2 * n - 1
    S = 0.5 * float(n)

    G = np.zeros((L, n))
    dx = 2.0 / float(n)
    for j in range(L):
        s = j - (n - 1)
        i_min = max(0, -s)
        i_max = min(n - 1, n - 1 - s)
        if i_min <= i_max:
            idx1 = np.arange(i_min, i_max + 1)
            idx2 = idx1 + s
            G[j, idx1] += (1.0 - h[idx2]) * dx
            G[j, idx2] += (-h[idx1]) * dx

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - c_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
        if res.success and res.x is not None:
            h_new = ensure_feasible(res.x[:n])
            return h_new, float(res.x[-1])
    except Exception:
        pass
    return None, None


def lp_phase(h_init, t_end, rng, max_steps=500):
    n = len(h_init)
    h_cur = h_init.copy()
    c_cur = exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    stall = 0

    for it in range(max_steps):
        if time.monotonic() > t_end:
            break
        h_lp, t_lp = lp_linearize_step(h_cur)
        if h_lp is None:
            break
        step_vec = h_lp - h_cur
        step_norm = float(np.linalg.norm(step_vec))
        if step_norm < 1e-10:
            stall += 1
            if stall > 5:
                break
            h_pert = h_cur + 0.005 * rng.standard_normal(n)
            h_pert = ensure_feasible(h_pert)
            c_pert = exact_c5(h_pert)
            if c_pert < c_cur:
                h_cur = h_pert
                c_cur = c_pert
            continue

        best_alpha = 0.0
        best_trial = None
        best_trial_c = c_cur
        for ai in range(12):
            alpha = 2.0 ** (-ai)
            h_tr = h_cur + alpha * step_vec
            h_tr = ensure_feasible(h_tr)
            c_tr = exact_c5(h_tr)
            if c_tr < best_trial_c - 1e-12:
                best_trial_c = c_tr
                best_trial = h_tr
                best_alpha = alpha
            if alpha < 0.005:
                break

        if best_trial is not None:
            h_cur = best_trial
            c_cur = best_trial_c
            stall = 0
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            stall += 1
            if stall > 8:
                h_pert = h_cur + 0.01 * rng.standard_normal(n)
                h_pert = ensure_feasible(h_pert)
                c_pert = exact_c5(h_pert)
                if c_pert < c_cur:
                    h_cur = h_pert
                    c_cur = c_pert
                stall = 0
            if stall > 15:
                break

    return best_h, best_c


def polish_slsqp(h0, maxiter=120, ftol=1e-13):
    n = len(h0)
    h0 = ensure_feasible(h0)
    c0 = exact_c5(h0)
    tau = 0.003

    def obj_f(x):
        return float(x[-1])

    def obj_jac(x):
        j = np.zeros(x.size)
        j[-1] = 1.0
        return j

    def con_eq(x):
        return float(np.sum(x[:n]) - 0.5 * n)

    def con_eq_jac(x):
        j = np.zeros(x.size)
        j[:n] = 1.0
        return j

    def con_ineq(x):
        h_part = x[:n]
        c = compute_overlaps(h_part)
        z = c / tau
        zmax = float(np.max(z))
        s = float(np.sum(np.exp(z - zmax)))
        if s < 1e-300:
            smax = zmax
        else:
            smax = zmax + math.log(s)
        return float(x[-1]) - tau * smax

    def con_ineq_jac(x):
        h_part = x[:n]
        c = compute_overlaps(h_part)
        z = c / tau
        zmax = float(np.max(z))
        w = np.exp(z - zmax)
        s = float(np.sum(w))
        if s > 1e-300:
            w = w / s
        else:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        jac_h = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-12:
                continue
            shift = k - (n - 1)
            g = gradient_at_shift(h_part, shift)
            jac_h += w[k] * g
        j = np.zeros(x.size)
        j[:n] = -tau * jac_h
        j[-1] = 1.0
        return j

    t0 = c0 + 0.002
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.03)]
    cons = [
        {"type": "eq", "fun": con_eq, "jac": con_eq_jac},
        {"type": "ineq", "fun": con_ineq, "jac": con_ineq_jac},
    ]
    try:
        res = minimize(obj_f, x0, jac=obj_jac, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": ftol, "iprint": 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return ensure_feasible(res.x[:n])
    except Exception:
        pass
    return None


def polish_rescale(h0, n_new, maxiter=80):
    h_r = resample_h(h0, n_new)
    h_res = polish_slsqp(h_r, maxiter=maxiter)
    if h_res is not None:
        return h_res
    return h_r


def run(seed=42, budget_s=1100, **kwargs):
    seed = int(seed)
    try:
        budget = max(2.0, float(budget_s))
    except Exception:
        budget = 1100.0
    t0 = time.monotonic()
    t_end = t0 + budget
    reserve = 3.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 10 ** 9

    ph = globals().get("initial_h_values")
    if ph is not None:
        ph = np.asarray(ph, dtype=float).reshape(-1)
        if ph.size >= 2 and np.all(np.isfinite(ph)):
            best_h = ensure_feasible(ph)
            best_c = exact_c5(best_h)
            best_n = best_h.size

    if best_h is None:
        best_h = adjust_sum_exact(np.full(64, 0.5), 32.0)
        best_c = exact_c5(best_h)
        best_n = 64

    hs = symmetrize_h(best_h)
    cs = exact_c5(hs)
    if cs < best_c:
        best_h, best_c = hs, cs

    # Phase 1: Nesterov subgradient
    t1 = t0 + min(120.0, budget * 0.12)
    best_h, best_c = nesterov_subgradient(best_h, t1, rng, max_iter=100000)
    best_n = best_h.size
    hs = symmetrize_h(best_h)
    cs = exact_c5(hs)
    if cs < best_c:
        best_h, best_c = hs, cs

    # Phase 2: LP linearization
    t2 = t0 + min(550.0, budget * 0.55)
    best_h, best_c = lp_phase(best_h, t2, rng, max_steps=300)
    best_n = best_h.size
    hs = symmetrize_h(best_h)
    cs = exact_c5(hs)
    if cs < best_c:
        best_h, best_c = hs, cs

    # Phase 3: Try nearby dimensions
    t3 = t0 + min(680.0, budget * 0.65)
    n_base = best_n
    for dn in (-2, -1, 1, 2, -3, 3):
        if time.monotonic() > t3:
            break
        nn = n_base + dn
        if nn < 32 or nn > 128:
            continue
        h_r = resample_h(best_h, nn)
        c_r = exact_c5(h_r)
        if c_r < best_c:
            best_h, best_c, best_n = h_r, c_r, nn
        # Quick subgradient from resampled
        h_sg, c_sg = nesterov_subgradient(h_r, min(t3, time.monotonic() + 8), rng, max_iter=5000)
        if c_sg < best_c:
            best_h, best_c, best_n = h_sg, c_sg, nn
        hs2 = symmetrize_h(best_h)
        cs2 = exact_c5(hs2)
        if cs2 < best_c:
            best_h, best_c = hs2, cs2

    # Phase 4: SLSQP polish (multiple rounds)
    t4 = t_end - reserve
    for attempt in range(12):
        if time.monotonic() > t4:
            break
        h_pol = polish_slsqp(best_h, maxiter=150, ftol=1e-13)
        if h_pol is not None:
            c_pol = exact_c5(h_pol)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol.copy()
        hs3 = symmetrize_h(best_h)
        h_pol2 = polish_slsqp(hs3, maxiter=100, ftol=1e-13)
        if h_pol2 is not None:
            c_pol2 = exact_c5(h_pol2)
            if c_pol2 < best_c:
                best_c = c_pol2
                best_h = h_pol2.copy()
        # Small perturbation for diversity
        h_pert = best_h + 0.003 * rng.standard_normal(best_h.size)
        h_pert = ensure_feasible(h_pert)
        c_pert = exact_c5(h_pert)
        if c_pert < best_c:
            best_h, best_c = h_pert, c_pert

    # Final: one more LP round if time allows
    if time.monotonic() < t_end - reserve - 30:
        t_lp2 = t_end - reserve - 5
        h_lp, c_lp = lp_phase(best_h, t_lp2, rng, max_steps=50)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Final symmetrization check
    hs_f = symmetrize_h(best_h)
    cs_f = exact_c5(hs_f)
    if cs_f < best_c:
        best_h, best_c = hs_f, cs_f

    best_h = ensure_feasible(best_h)
    best_c = exact_c5(best_h)
    return best_h, float(best_c), int(best_h.size)
