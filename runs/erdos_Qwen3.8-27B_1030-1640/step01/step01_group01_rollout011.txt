```python
def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.size == 0 or not np.all(np.isfinite(vals)):
        return float('inf')
    return float(np.max(vals))

def make_c5_vector(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def force_sum(h, S, tol=1e-12, max_iter=10000):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    for _ in range(max_iter):
        r = S - float(np.sum(h))
        if abs(r) <= tol:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            if slack[idx] <= 1e-15:
                break
            delta = min(r, slack[idx])
            h[idx] += delta
        else:
            idx = int(np.argmin(h))
            if h[idx] <= 1e-15:
                break
            delta = min(-r, h[idx])
            h[idx] -= delta
    np.clip(h, 0.0, 1.0, out=h)
    return h

def project_box_sum(v):
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * float(n)
    if not np.all(np.isfinite(v)):
        return force_sum(np.full(n, 0.5), S)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s < S:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return force_sum(p, S)

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return np.array([0.5, 0.5])
    S = 0.5 * float(n)
    if not np.all(np.isfinite(h)):
        return force_sum(np.full(n, 0.5), S)
    np.clip(h, 0.0, 1.0, out=h)
    return force_sum(h, S)

    h = np.asarray(h, dtype=float).reshape(-1).copy()
    h = 0.5 * (h + h[::-1])
    return force_sum(h, 0.5 * float(h.size))

    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = max(2, int(n_new))
    if n_old == 0:
        return force_sum(np.full(n_new, 0.5), 0.5 * n_new)
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)

def compute_gradients(h):
    """Compute gradient matrix G (L x n) and c vector (L,) for all correlation lags."""
    n = h.size
    dx = 2.0 / float(n)
    L = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    for m in range(L):
        shift = m - (n - 1)
        if shift >= 0:
            # c[m] = sum_{i=0}^{n-1-shift} h[i+shift]*(1-h[i])
            for i in range(n - shift):
                # d/dh[j] where j = i+shift: contributes (1-h[i])
                G[m, i + shift] += dx * (1.0 - h[i])
                # d/dh[j] where j = i: contributes -h[i+shift]
                G[m, i] -= dx * h[i + shift]
        else:
            s = -shift
            # c[m] = sum_{i=s}^{n-1} h[i-s]*(1-h[i])
            for i in range(s, n):
                j = i - s
                # d/dh[j]: contributes (1-h[i])
                G[m, j] += dx * (1.0 - h[i])
                # d/dh[i]: contributes -h[j]
                G[m, i] -= dx * h[j]
    return G, c
```

```python
def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n < 2:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def force_sum(h, S, tol=1e-12):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(h.size + 10):
        r = S - h.sum()
        if abs(r) <= tol:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            d = min(r, 1.0 - h[idx])
            if d <= 0: break
            h[idx] += d
        else:
            idx = int(np.argmin(h))
            d = min(-r, h[idx])
            if d <= 0: break
            h[idx] -= d
    np.clip(h, 0, 1, out=h)
    return h

def project(v):

    n = v.size
    S = 0.5 * n
    if not np.all(np.isfinite(v)):
        return force_sum(np.full(n, 0.5), S)
    lo, hi = v.min() - 2, v.max() + 2

            lo = mid
        else:
            hi = mid

def grad_matrix(h):

    dx = 2.0 / n
    L = 2*n - 1
    c = c5_vector(h)

        if s >= 0:
            ns = n - s
            G[m, s:] += dx * (1.0 - h[:ns])
            G[m, :ns] -= dx * h[s:]

            s2 = -s
            ns = n - s2
            G[m, :ns] += dx * (1.0 - h[s2:])
            G[m, s2:] -= dx * h[:ns]
    return G, c
```

```python
def lp_step(h0, radius):
    n = h0.size
    G, c = grad_matrix(h0)
    L = 2*n - 1
    # Variables: [h (n), t (1)]
    # Minimize t
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    # A_ub @ x <= b_ub: G@h - t <= G@h0 - c
    A_ub = np.zeros((L, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - c
    # A_eq @ x = b_eq: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    # Bounds
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub))
    bounds.append((-1.0, 2.0))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n], res.x[-1]
    return None, None
```

            no_improve = 0
        
        # If stuck, try random perturbation
        if no_improve >= 3 and time.monotonic() < deadline - 80:
            for _ in range(3):
                if time.monotonic() > deadline - 70: break
                scale = 0.01 + 0.04*rng.random()
                hr = feas(best_h + rng.normal(0, scale, n_best))
                cr = exact_c5(hr)
                if cr < best_c + 0.02:
                    hrp, crp = local_polish(hr, deadline - 70,
                                            r_init=0.02, r_max=0.1, r_min=0.001, max_iter=100)
                    if crp < best_c - 1e-10:
                        best_h, best_c, n_best = hrp, crp, hrp.size
                        no_improve = 0
                        break
        
        # Try different dimension
        if no_improve >= 5 and time.monotonic() < deadline - 100:
