def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_weighted_grad(h, weights):
    """Compute sum_k weights[k] * grad C_k."""
    n = len(h)
    offset = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if abs(wm) < 1e-16:
            continue
        k = m - offset
        if k >= 0:
            j_lo, j_hi = 0, n - k
        else:
            j_lo, j_hi = 0, n + k
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi)
            if k >= 0:
                grad[idx] += wm * (1.0 - h[idx + k])
                grad[idx + k] += wm * (-h[idx])
            else:
                idx2 = idx - k  # = idx + |k|
                grad[idx2] += wm * (1.0 - h[idx])
                grad[idx] += wm * (-h[idx2])
    return grad

def adam_smooth_descent(h, theta_schedule, max_iters_per_theta=500, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for the log-sum-exp objective."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    t_step = 0
    
    for theta in theta_schedule:
        for it in range(max_iters_per_theta):
            t_step += 1
            G = compute_all_overlap(h_cur)
            Gmax = np.max(G)
            if not np.isfinite(Gmax):
                break
            # Log-sum-exp gradient
            exp_G = np.exp((G - Gmax) / theta)
            w = exp_G / np.sum(exp_G)
            grad = compute_weighted_grad(h_cur, w)
            
            # Adam update
            m_v = beta1 * m_v + (1 - beta1) * grad
            v_v = beta2 * v_v + (1 - beta2) * grad**2
            m_hat = m_v / (1 - beta1**t_step)
            v_hat = v_v / (1 - beta2**t_step)
            
            step = -lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project
            y = h_cur + step
            y_proj = project_h(y, target)
            c_new = evaluate_C(y_proj)
            
            if c_new < best_c:
                best_c = c_new
                best_h = y_proj.copy()
            h_cur = y_proj
    
    return best_h, best_c

def compute_grad_single_lag(h, m):
    """Compute gradient of C_m (m is index in full correlation output)."""
    n = len(h)
    offset = n - 1
    k = m - offset
    grad = np.zeros(n)
    if k >= 0:
        for j in range(n - k):
            grad[j] += (1.0 - h[j + k])
            grad[j + k] += (-h[j])
    else:
        kk = -k
        for j in range(n - kk):
            grad[j + kk] += (1.0 - h[j])
            grad[j] += (-h[j + kk])
    return grad

def golden_pair(h, s, t, max_iter=80):
    """Golden section search on pair (s, t)."""
    T = min(h[s], 1.0 - h[t])
    if T < 1e-14:
        return h, evaluate_C(h)
    c_cur = evaluate_C(h)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T
    
    def ev(tau):
        ht = h.copy()
        ht[s] -= tau
        ht[t] += tau
        return evaluate_C(ht)
    
    c1 = b - invphi * (b - a)
    c2 = a + invphi * (b - a)
    f1, f2 = ev(c1), ev(c2)
    best_val, best_tau = min(f1, f2, c_cur), 0.0
    if f1 < best_val: best_val, best_tau = f1, c1
    if f2 < best_val: best_val, best_tau = f2, c2
    
    for _ in range(max_iter):
        if b - a < 1e-13 * (T + 1.0): break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - invphi * (b - a); f1 = ev(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + invphi * (b - a); f2 = ev(c2)
        if f1 < best_val: best_val, best_tau = f1, c1
        if f2 < best_val: best_val, best_tau = f2, c2
    
    f_end = ev(T)
    if f_end < best_val: best_val, best_tau = f_end, T
    
    if best_tau > 1e-14 and best_val < c_cur - 1e-13:
        hn = h.copy()
        hn[s] -= best_tau
        hn[t] += best_tau
        return hn, best_val
    return h, c_cur

def pair_polish(h, rng, n_pairs, max_iter=60):
    """Polish using pair moves."""
    n = len(h)
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    
    for it in range(n_pairs):
        # Compute gradient for guidance
        G = compute_all_overlap(h_cur)
        Gmax = np.max(G)
        exp_G = np.exp((G - Gmax) / 1e-3)
        w = exp_G / np.sum(exp_G)
        grad = compute_weighted_grad(h_cur, w)
        
        valid_s = np.where(h_cur > 1e-8)[0]
        valid_t = np.where(h_cur < 1.0 - 1e-8)[0]
        if len(valid_s) == 0 or len(valid_t) == 0:
            break
        
        # Try several candidate pairs
        best_impr = 0.0
        best_new = None
        n_try = 8
        for _ in range(n_try):
            r = rng.random()
            if r < 0.4:
                s = valid_s[rng.integers(len(valid_s))]
                t = valid_t[rng.integers(len(valid_t))]
            elif r < 0.7:
                s = valid_s[np.argmax(grad[valid_s])]
                t = valid_t[np.argmin(grad[valid_t])]
            else:
                # Anti-correlated: pick s where h is high and grad is low, t where h is low and grad is high
                score_s = h_cur * (1 - grad)  # want high h, low grad (decrease h)
                score_t = (1 - h_cur) * grad  # want low h, high grad (increase h)
                s = valid_s[np.argmax(score_s[valid_s])]
                t = valid_t[np.argmin(score_t[valid_t])]
            
            if s == t: continue
            hn, cn = golden_pair(h_cur, s, t, max_iter)
            if cn < c_cur - 1e-13:
                if c_cur - cn > best_impr:
                    best_impr = c_cur - cn
                    best_new = (hn, cn)
        
        if best_new is not None:
            h_cur, c_cur = best_new
    
    return h_cur, c_cur

def simulated_annealing(h, rng, n_iterations=5000, T_start=0.01, T_end=0.0001):
    """Simulated annealing with pair moves."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    c_cur = evaluate_C(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    
    for it in range(n_iterations):
        T = T_start * (T_end / T_start) ** (it / max(n_iterations - 1, 1))
        
        # Random pair move
        s = rng.integers(n)
        t = rng.integers(n)
        if s == t: continue
        
        # Determine direction: if h[s] > h[t], move from s to t; else from t to s
        if h_cur[s] >= h_cur[t]:
            donor, receiver = s, t
        else:
            donor, receiver = t, s
        
        max_move = min(h_cur[donor], 1.0 - h_cur[receiver])
        if max_move < 1e-10: continue
        
        # Random move size
        tau = rng.random() * max_move * 0.5
        
        h_trial = h_cur.copy()
        h_trial[donor] -= tau
        h_trial[receiver] += tau
        
        c_trial = evaluate_C(h_trial)
        delta = c_trial - c_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h_cur = h_trial
            c_cur = c_trial
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
    
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import linprog

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_grad_lag(h, m):
    n = len(h)
    offset = n - 1
    k = m - offset
    grad = np.zeros(n)
    if k >= 0:
        for j in range(n - k):
            grad[j] += (1.0 - h[j + k])
            grad[j + k] += (-h[j])
    else:
        kk = -k
        for j in range(n - kk):
            grad[j + kk] += (1.0 - h[j])
            grad[j] += (-h[j + kk])
    return grad

def compute_weighted_grad(h, weights):
    n = len(h)
    offset = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if abs(wm) < 1e-16:
            continue
        k = m - offset
        if k >= 0:
            for j in range(n - k):
                grad[j] += wm * (1.0 - h[j + k])
                grad[j + k] += wm * (-h[j])
        else:
            kk = -k
            for j in range(n - kk):
                grad[j + kk] += wm * (1.0 - h[j])
                grad[j] += wm * (-h[j + kk])
    return grad

def adam_descent(h_init, thetas, max_it=400, lr=0.05, b1=0.85, b2=0.999, eps=1e-8):
    n = len(h_init)
    target = n / 2.0
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_c = evaluate_C(h_cur)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    step_count = 0
    for theta in thetas:
        for it in range(max_it):
            step_count += 1
            G = compute_all_overlap(h_cur)
            Gmax = G.max()
            if not np.isfinite(Gmax):
                break
            exp_G = np.exp((G - Gmax) / theta)
            s_exp = exp_G.sum()
            if s_exp < 1e-300:
                break
            w = exp_G / s_exp
            grad = compute_weighted_grad(h_cur, w)
            m_v = b1 * m_v + (1.0 - b1) * grad
            v_v = b2 * v_v + (1.0 - b2) * (grad * grad)
            m_hat = m_v / (1.0 - b1 ** step_count)
            v_hat = v_v / (1.0 - b2 ** step_count)
            update = -lr * m_hat / (np.sqrt(v_hat) + eps)
            y = h_cur + update
            y_p = project_h(y, target)
            c_new = evaluate_C(y_p)
            if c_new < best_c:
                best_c = c_new
                best_h = y_p.copy()
            h_cur = y_p
    return best_h, best_c

def linprog_minimax_step(h, n_active=15):
    n = len(h)
    target = n / 2.0
    G = compute_all_overlap(h)
    G_max = G.max()
    margin = 0.015 * G_max
    active = np.where(G >= G_max - margin)[0]
    if len(active) > n_active:
        order = np.argsort(G[active])[-n_active:]
        active = active[order]
    if len(active) < 2:
        return h, evaluate_C(h)
    cvec = np.zeros(n + 1)
    cvec[-1] = 1.0
    A_rows = []
    b_vals = []
    for m_idx in active:
        gk = compute_grad_lag(h, m_idx)
        row = np.zeros(n + 1)
        row[:n] = gk
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(np.dot(gk, h) - G[m_idx])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, evaluate_C(h)
    if not res.success:
        return h, evaluate_C(h)
    h_lp = res.x[:n]
    c_cur = evaluate_C(h)
    best_h = h.copy()
    best_c = c_cur
    for alpha in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1]:
        h_t = h + alpha * (h_lp - h)
        h_t = project_h(h_t, target)
        c_t = evaluate_C(h_t)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
    return best_h, best_c

def golden_pair(h, s, t, max_iter=70):
    T = min(h[s], 1.0 - h[t])
    if T < 1e-14:
        return h, evaluate_C(h)
    c_cur = evaluate_C(h)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T

    def ev(tau):
        ht = h.copy()
        ht[s] -= tau
        ht[t] += tau
        return evaluate_C(ht)

    c1 = b - invphi * (b - a)
    c2 = a + invphi * (b - a)
    f1, f2 = ev(c1), ev(c2)
    best_val = min(f1, f2, c_cur)
    best_tau = 0.0
    if f1 < best_val:
        best_val, best_tau = f1, c1
    if f2 < best_val:
        best_val, best_tau = f2, c2
    for _ in range(max_iter):
        if b - a < 1e-13 * (T + 1.0):
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - invphi * (b - a)
            f1 = ev(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + invphi * (b - a)
            f2 = ev(c2)
        if f1 < best_val:
            best_val, best_tau = f1, c1
        if f2 < best_val:
            best_val, best_tau = f2, c2
    f_end = ev(T)
    if f_end < best_val:
        best_val, best_tau = f_end, T
    if best_tau > 1e-14 and best_val < c_cur - 1e-14:
        hn = h.copy()
        hn[s] -= best_tau
        hn[t] += best_tau
        return hn, best_val
    return h, c_cur

def pair_polish(h, rng, n_pairs, max_iter=60):
    n = len(h)
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    for it in range(n_pairs):
        G = compute_all_overlap(h_cur)
        Gmax = G.max()
        exp_G = np.exp((G - Gmax) / (0.3 * Gmax + 1e-8))
        w = exp_G / exp_G.sum()
        grad = compute_weighted_grad(h_cur, w)
        valid_s = np.where(h_cur > 1e-8)[0]
        valid_t = np.where(h_cur < 1.0 - 1e-8)[0]
        if len(valid_s) == 0 or len(valid_t) == 0:
            break
        best_impr = 0.0
        best_new = None
        n_try = 8
        for _ in range(n_try):
            r = rng.random()
            if r < 0.35:
                s = valid_s[rng.integers(len(valid_s))]
                t = valid_t[rng.integers(len(valid_t))]
            elif r < 0.65:
                s = valid_s[np.argmax(grad[valid_s])]
                t = valid_t[np.argmin(grad[valid_t])]
            else:
                score_s = h_cur - 0.5 * grad
                score_t = (1.0 - h_cur) + 0.5 * grad
                s = valid_s[np.argmax(score_s[valid_s])]
                t = valid_t[np.argmin(score_t[valid_t])]
            if s == t:
                continue
            hn, cn = golden_pair(h_cur, s, t, max_iter)
            if cn < c_cur - 1e-14:
                if c_cur - cn > best_impr:
                    best_impr = c_cur - cn
                    best_new = (hn, cn)
        if best_new is not None:
            h_cur, c_cur = best_new
    return h_cur, c_cur

def simulated_anneal(h, rng, n_iter=4000, T_start=0.008, T_end=0.00005):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    c_cur = evaluate_C(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    ratio = T_end / T_start
    for it in range(n_iter):
        T = T_start * (ratio ** (it / max(n_iter - 1, 1)))
        s = rng.integers(n)
        t = rng.integers(n)
        if s == t:
            continue
        if h_cur[s] >= h_cur[t]:
            d, r = s, t
        else:
            d, r = t, s
        max_mv = min(h_cur[d], 1.0 - h_cur[r])
        if max_mv < 1e-10:
            continue
        tau = rng.random() * max_mv * 0.4
        h_tr = h_cur.copy()
        h_tr[d] -= tau
        h_tr[r] += tau
        c_tr = evaluate_C(h_tr)
        delta = c_tr - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h_cur = h_tr
            c_cur = c_tr
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
    return best_h, best_c

def block_move_polish(h, rng, n_moves=500, block_size=3):
    n = len(h)
    target = n / 2.0
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    for it in range(n_moves):
        if n < 6:
            break
        G = compute_all_overlap(h_cur)
        Gmax = G.max()
        exp_G = np.exp((G - Gmax) / (0.5 * Gmax + 1e-8))
        w = exp_G / exp_G.sum()
        grad = compute_weighted_grad(h_cur, w)
        bs = min(block_size, n // 3)
        donors = np.where(h_cur > 0.05)[0]
        receivers = np.where(h_cur < 0.95)[0]
        if len(donors) < bs or len(receivers) < bs:
            break
        d_idx = donors[rng.permutation(len(donors))[:bs]]
        r_idx = receivers[rng.permutation(len(receivers))[:bs]]
        d_idx = d_idx[np.argsort(grad[d_idx] * (1.0 - h_cur[d_idx]))[-bs:]]
        r_idx = r_idx[np.argsort(grad[r_idx] * (1.0 - h_cur[r_idx]))[:bs]]
        total_don = sum(h_cur[i] for i in d_idx)
        total_rcv = sum(1.0 - h_cur[i] for i in r_idx)
        max_tau = min(total_don, total_rcv) / bs
        if max_tau < 1e-10:
            continue
        best_tau = 0.0
        best_c = c_cur
        for frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
            tau = max_tau * frac
            h_t = h_cur.copy()
            for i in d_idx:
                h_t[i] -= tau
            for i in r_idx:
                h_t[i] += tau
            if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
                continue
            c_t = evaluate_C(h_t)
            if c_t < best_c:
                best_c = c_t
                best_tau = tau
        if best_tau > 1e-10:
            for i in d_idx:
                h_cur[i] -= best_tau
            for i in r_idx:
                h_cur[i] += best_tau
            c_cur = best_c
    return h_cur, c_cur
