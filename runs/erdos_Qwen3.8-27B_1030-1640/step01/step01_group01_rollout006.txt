def project_box_sum(v, S):
    """Project v onto {x : 0≤x≤1, sum(x)=S}."""
    v = np.asarray(v, dtype=float).reshape(-1).copy()
    n = v.size

    lo, hi = -1.0, 1.0

        else:
            hi = mid

    if abs(diff) > 1e-12:

def pair_optimize(h, i, j, n_trials=50):
    """Optimize h[i] and h[j] together."""
    n = h.size
    dx = 2.0 / n
    mid = n - 1
    
    hi_save = h[i]
    hj_save = h[j]
    
    # Get base with both at 0
    h[i] = 0.0
    h[j] = 0.0
    c_base = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Get slopes for h[i] (with h[j]=0)
    h[i] = 1.0
    c_hi = np.correlate(h, 1.0 - h, mode="full") * dx
    h[i] = 0.0
    
    # Get slopes for h[j] (with h[i]=0)
    h[j] = 1.0
    c_hj = np.correlate(h, 1.0 - h, mode="full") * dx
    h[j] = 0.0
    
    # Now c[h_i, h_j] ≈ c_base + h_i * (c_hi - c_base) + h_j * (c_hj - c_base)
    # (approximately linear, ignoring the quadratic terms at mid)
    slope_i = c_hi - c_base
    slope_j = c_hj - c_base
    
    # Search on grid
    t_i = np.linspace(0, 1, n_trials)
    t_j = np.linspace(0, 1, n_trials)
    
    best_val = float('inf')
    best_bi = hi_save
    best_bj = hj_save
    
    for a, bi in enumerate(t_i):
        for b, bj in enumerate(t_j):
            c = c_base + bi * slope_i + bj * slope_j
            c[mid] = c_base[mid] + bi*(1-bi)*dx + bj*(1-bj)*dx
            val = c.max()
            if val < best_val:
                best_val = val
                best_bi = bi
                best_bj = bj
    
    h[i] = best_bi
    h[j] = best_bj
    return h

            best_h, best_c, best_n = h_opt, c_opt, h_opt.size
        # Symmetrize and re-optimize
        hs = symmetrize(best_h)
        cs = exact_c5(hs)
        if cs < best_c:
            best_h = hs
            best_c = cs
        if time_left() > 60:
            h_opt2, c_opt2 = optimize_cd(best_h, min(time_left() - 60, 80), n_trials=200)
            if c_opt2 < best_c:
                best_h, best_c, best_n = h_opt2, c_opt2, h_opt2.size

    # Phase 2: Try different dimensions
    dims_to_try = [56, 60, 62, 64, 66, 68, 72]
    for n_dim in dims_to_try:
        if time_left() < 80:
            break
        if n_dim == best_n:
            continue
        h_rep = resample_h(best_h, n_dim)
        alloc = min(time_left() - 80, 50)
        if alloc < 15:
            break
        h_opt, c_opt = optimize_cd(h_rep, alloc, n_trials=130)
        if c_opt < best_c - 1e-8:
            best_h, best_c, best_n = h_opt, c_opt, h_opt.size

    # Phase 3: Perturbations and restarts
    attempt = 0
    while time_left() > 50 and attempt < 80:
        # Small perturbation of best
        if rng.random() < 0.6:
            scale = 0.008 + 0.005 * (attempt % 5)
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = project_box_sum(h_pert, 0.5 * best_n)
        elif rng.random() < 0.8:
            # Sine candidate
            n_r = int(rng.choice([best_n, best_n - 2, best_n + 2]))
            n_r = max(48, min(80, n_r))
            freq = float(rng.integers(1, 5))
            phase = float(rng.uniform(0, 2 * math.pi))
            amp = float(rng.uniform(0.2, 0.4))
            h_pert = make_sine_candidate(n_r, freq, phase, amp)
        else:
            # Block candidate
            n_r = int(rng.choice([best_n, 60, 64]))
            nb = int(rng.integers(4, 10))
            h_pert = make_block_candidate(n_r, nb, rng)

        alloc = min(time_left() - 50, 35)
        if alloc < 10:
            break
        h_opt, c_opt = optimize_cd(h_pert, alloc, n_trials=130, symm_every=2)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, h_opt.size
        attempt += 1

    # Phase 4: Final high-precision refinement
    if time_left() > 30:
        h = best_h.copy()
        n = h.size
        S = 0.5 * n
        for trial_count in [200, 400, 800]:
            if time_left() < 35:
                break
            for _ in range(8):
                if time_left() < 35:
                    break
                order = rng.permutation(n)
                for i in order:
                    if time_left() < 35:
                        break
                    coord_optimize(h, i, n_trials=trial_count)
                h = project_box_sum(h, S)
                c = exact_c5(h)
                if c < best_c:
                    best_h = h.copy()
                    best_c = c
                    best_n = n

    # Final checks
    if best_h is not None:
        hs = symmetrize(best_h)
        cs = exact_c5(hs)
        if cs < best_c:
            best_h = hs
            best_c = cs
        hs2 = smooth_h(best_h, 3)
        cs2 = exact_c5(hs2)
        if cs2 < best_c:
            best_h = hs2
            best_c = cs2

    h_final = project_box_sum(best_h, 0.5 * best_h.size)
    c_final = exact_c5(h_final)
    if not np.isfinite(c_final):
        h_final = project_box_sum(np.full(64, 0.5), 32.0)
        c_final = exact_c5(h_final)

    return h_final, float(c_final), int(h_final.size)
