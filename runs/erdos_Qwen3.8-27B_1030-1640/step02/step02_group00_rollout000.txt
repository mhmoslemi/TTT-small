def project_feasible(y, target_sum):
    """Project y onto {x: 0<=x<=1, sum(x)=target_sum}"""
    y = np.asarray(y, dtype=float).copy()
    n = y.size
    # Bisection on t: sum(clip(y-t, 0, 1)) = target_sum
    lo = y.min() - 1.0

        if s > target_sum:

        else:

    result = np.clip(y - mid, 0.0, 1.0)
    # Fix residual

        if np.any(interior):
            adj = residual / np.sum(interior)
            result[interior] += adj
            np.clip(result, 0.0, 1.0, out=result)
    return result

```python
def lse_gradient(h, beta):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    # Softmax weights (shifted to avoid overflow)
    exponents = beta * (c - c_max)
    # Clip for numerical stability
    exponents = np.clip(exponents, -500, 500)
    weights = np.exp(exponents)
    weights /= weights.sum()
    
    # Gradient of each c_m with respect to h
    # c_m = dx * sum_i h[i] * (1-h[i+m-n+1]) for valid i
    # d(c_m)/d(h[j]) = dx * [sum of (1-h[i+m-n+1]) where i=j (valid) 
    #                        - sum of h[i] where i+m-n+1=j (valid)]
    
    grad = np.zeros(n)
    for m in range(L):
        shift = m - n + 1
        if shift >= 0:
            # c[m] = dx * sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
            # d/dh[j]: 
            #   h[j] as first factor: j = i+shift => i = j-shift, valid if 0<=j-shift<=n-1-shift => shift<=j<=n-1
            #     contribution: dx * (1-h[j-shift])
            #   h[j] as second factor (in 1-h[i]): i=j, valid if 0<=j<=n-1-shift => j<=n-1-shift
            #     contribution: -dx * h[j+shift]
            for j in range(shift, n):
                if j <= n - 1:
                    grad[j] += weights[m] * dx * (1.0 - h[j - shift])
            for j in range(0, n - shift):
                grad[j] -= weights[m] * dx * h[j + shift]
        else:
            s = -shift  # s > 0
            # c[m] = dx * sum_{i=s}^{n-1} h[i] * (1-h[i-s])
            # d/dh[j]:
            #   h[j] as first factor: i=j, valid if s<=j<=n-1
            #     contribution: dx * (1-h[j-s])
            #   h[j] as second factor: i-s=j => i=j+s, valid if 0<=j+s<=n-1 and j+s>=s => 0<=j<=n-1-s
            #     contribution: -dx * h[j+s]
            for j in range(s, n):
                grad[j] += weights[m] * dx * (1.0 - h[j - s])
            for j in range(0, n - s):
                grad[j] -= weights[m] * dx * h[j + s]
    
    return grad
```

This is O(n²) per computation due to the loops. For n=64, that's 4096 × 127 ≈ 500K operations. Fine.

import time

def evaluate_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

def project_feasible(y, target_sum):
    y = np.asarray(y, dtype=float).copy()
    n = y.size
    if n == 0:
        return y
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - mid, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-13:
        interior = (result > 1e-12) & (result < 1.0 - 1e-12)
        count = np.sum(interior)
        if count > 0:
            adj = residual / count
            result[interior] += adj
            np.clip(result, 0.0, 1.0, out=result)
    return result

def pair_coord_descent(h, t_end):
    n = len(h)
    S = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    
    best_c = evaluate_c5(h)
    
    for sweep in range(200):
        if time.monotonic() > t_end:
            break
        max_change = 0.0
        # Alternate direction
        if sweep % 2 == 0:
            order = list(range(0, n-1, 2))  # pairs (0,1), (2,3), ...
        else:
            order = list(range(1, n-1, 2))  # pairs (1,2), (3,4), ...
        
        for j in order:
            i = j + 1
            if i >= n:
                continue
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-15:
                continue
            
            old_j = h[j]
            
            # Evaluate at endpoints
            h[j] = t_lo
            h[i] = s - t_lo
            c_lo = np.correlate(h, 1.0 - h, mode="full") * dx
            
            h[j] = t_hi
            h[i] = s - t_hi
            c_hi = np.correlate(h, 1.0 - h, mode="full") * dx
            
            a = c_hi - c_lo
            b = c_lo
            rt = t_hi - t_lo
            
            # Find min_u max(b + a*u) for u in [0,1]
            best_u = (old_j - t_lo) / rt
            best_val = float(np.max(b + a * best_u))
            
            # Try crossing points of top contributors
            top_idx = np.argsort(b)[::-1][:5]
            top_idx2 = np.argsort(b + a)[::-1][:5]
            for i1 in top_idx:
                for i2 in top_idx2:
                    if i1 != i2 and abs(a[i1] - a[i2]) > 1e-14:
                        u_c = (b[i2] - b[i1]) / (a[i1] - a[i2])
                        if 0.0 <= u_c <= 1.0:
                            val = float(np.max(b + a * u_c))
                            if val < best_val - 1e-15:
                                best_val = val
                                best_u = u_c
            
            # Also try a few fixed points
            for u_try in [0.0, 0.25, 0.5, 0.75, 1.0]:
                val = float(np.max(b + a * u_try))
                if val < best_val - 1e-15:
                    best_val = val
                    best_u = u_try
            
            # Refine
            for _ in range(5):
                improved = False
                for du in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                    u2 = best_u + du
                    if 0.0 <= u2 <= 1.0:
                        val = float(np.max(b + a * u2))
                        if val < best_val - 1e-14:
                            best_val = val
                            best_u = u2
                            improved = True
                if not improved:
                    break
            
            new_j = t_lo + best_u * rt
            max_change = max(max_change, abs(new_j - old_j))
            h[j] = new_j
            h[i] = s - new_j
        
        # Check improvement
        c_now = evaluate_c5(h)
        if c_now < best_c - 1e-12:
            best_c = c_now
        if max_change < 1e-10:
            break
    
    h = project_feasible(h, S)
    return h

def lse_gradient_descent(h, t_end, max_iter=500):
    n = len(h)
    S = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h.copy(), S)
    c_cur = evaluate_c5(h)
    
    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    beta_idx = 0
    
    for it in range(max_iter):
        if time.monotonic() > t_end:
            break
        
        beta = betas[min(beta_idx, len(betas)-1)]
        c_vec = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = float(np.max(c_vec))
        
        # Softmax weights
        exponents = beta * (c_vec - c_max)
        exponents = np.clip(exponents, -300, 300)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c_vec)
            w[np.argmax(c_vec)] = 1.0
            w_sum = 1.0
        w /= w_sum
        
        # Compute gradient (vectorized)
        L = 2 * n - 1
        grad = np.zeros(n)
        for m in range(L):
            shift = m - n + 1
            if shift >= 0:
                grad[shift:] += w[m] * dx * (1.0 - h[:n-shift])
                grad[:n-shift] -= w[m] * dx * h[shift:]
            else:
                s = -shift
                grad[s:] += w[m] * dx * (1.0 - h[:n-s])
                grad[:n-s] -= w[m] * dx * h[s:]
        
        # Project gradient to maintain sum (subtract mean)
        grad -= np.mean(grad)
        
        # Line search: try several step sizes
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            beta_idx = min(beta_idx + 1, len(betas) - 1)
            continue
        
        grad_normed = grad / g_norm
        best_step = 0.0
        best_c = c_cur
        
        # Try geometric steps
        for lr in [0.05, 0.1, 0.2, 0.05, 0.02, 0.01]:
            h_try = project_feasible(h - lr * grad_normed, S)
            c_try = evaluate_c5(h_try)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_step = lr
        
        if best_step > 0:
            h = project_feasible(h - best_step * grad_normed, S)
            c_cur = best_c
            # Reduce beta constraint focus
            beta_idx = max(0, beta_idx - 1)
        else:
            beta_idx = min(beta_idx + 1, len(betas) - 1)
        
        if c_cur < 0.3808:
            break
    
    h = project_feasible(h, S)
    return h
