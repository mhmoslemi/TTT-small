import time

def project_box_sum(y, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    # Bisection on threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1 - x)) if diff > 0 else int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

def exact_value(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return math.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))

def corr_components(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def grad_matrix(h):
    """Compute gradient matrix G where G[m,j] = dc_m/dh_j"""
    h = np.asarray(h, float).ravel()
    n = h.size
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n - 1), n):
        m = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo >= i_hi:
            continue
        i_idx = np.arange(i_lo, i_hi)
        j_idx = i_idx + s
        G[m, i_idx] += dx * (1.0 - h[j_idx])
        G[m, j_idx] -= dx * h[i_idx]
    return G

def logsumexp_objective(h, beta):
    """Smooth surrogate for max(c)"""
    c = corr_components(h)
    c_max = c.max()
    # Numerically stable log-sum-exp
    c_shifted = beta * (c - c_max)
    result = c_max + np.log(np.sum(np.exp(c_shifted))) / beta
    return result

def logsumexp_gradient(h, beta):
    """Gradient of log-sum-exp objective"""
    c = corr_components(h)
    c_max = c.max()
    c_shifted = beta * (c - c_max)
    weights = np.exp(c_shifted) / np.sum(np.exp(c_shifted))
    G = grad_matrix(h)
    return G.T @ weights, c

def pnorm_objective(h, p):
    c = corr_components(h)
    return (np.sum(np.abs(c)**p))**(1.0/p)

def pnorm_gradient(h, p):
    c = corr_components(h)
    G = grad_matrix(h)
    pw = np.abs(c)**(p-1) * np.sign(c)
    pw = pw / np.sum(np.abs(c)**p) * np.abs(c)**(2-p)  # Hmm, this is getting complicated
    # Actually d/dc (sum|c|^p)^(1/p) = (1/p) * (sum|c|^p)^(1/p - 1) * p*|c|^(p-1)*sign(c)
    # = (sum|c|^p)^(1/p - 1) * |c|^(p-1)*sign(c)
    # Let S = sum|c|^p, f = S^(1/p)
    # df/dc_i = (1/p) * S^(1/p-1) * p * |c_i|^(p-1) * sign(c_i)
    #          = S^(1/p-1) * |c_i|^(p-1) * sign(c_i)
    S = np.sum(np.abs(c)**p)
    factor = S**(1.0/p - 1.0)
    dc = factor * np.abs(c)**(p-1) * np.sign(c)
    return G.T @ dc, c

def smooth_slsqp(h, beta, maxiter=200, tol=1e-10):
    """Optimize log-sum-exp surrogate using SLSQP"""
    n = h.size
    h0 = project_box_sum(h, n)
    
    def obj(x):
        return logsumexp_objective(x, beta)
    
    def obj_grad(x):
        return logsumexp_gradient(x, beta)[0]
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5 * n,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    res = optimize.minimize(obj, h0, jac=obj_grad, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': maxiter, 'ftol': tol, 'disp': False})
    if res.success and np.all(np.isfinite(res.x)):
        return project_box_sum(res.x, n)
    return h0

def exact_polish(h, max_iters=1000, tol=1e-12):
    """Polish exact objective using projected steepest descent on active lags"""
    n = h.size
    h = project_box_sum(h, n)
    F = exact_value(h)
    best_h = h.copy()
    best_F = F
    
    for it in range(max_iters):
        c = corr_components(h)
        # Find active lags (within tolerance of max)
        threshold = F - 1e-10
        active = c >= threshold
        if not np.any(active):
            active = c >= F - 1e-8
        
        G = grad_matrix(h)
        # Average gradient over active lags
        G_active = G[active, :]
        d = -G_active.mean(axis=0)
        d -= d.mean()  # Maintain sum constraint
        
        # Line search
        pos = d > 1e-12
        neg = d < -1e-12
        alpha_max = 5.0
        if np.any(pos):
            alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
        alpha_max = max(alpha_max, 1e-10)
        
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            a = frac * alpha_max
            h_new = project_box_sum(h + a * d, n)
            F_new = exact_value(h_new)
            if F_new < F - tol:
                h = h_new
                F = F_new
                improved = True
                break
        
        if not improved:

                break
            # Perturb
            break
        
        if F < best_F:
            best_F = F
            best_h = h.copy()
    
    return best_h, best_F

import time
import math
import numpy as np
from scipy import optimize

_in_n = 45

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1 - x)) if diff > 0 else int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return math.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))

    h = np.asarray(h, float).ravel()
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def grad_matrix(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return np.empty((0, 0))
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n - 1), n):
        m = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo >= i_hi:
            continue
        i_idx = np.arange(i_lo, i_hi)
        j_idx = i_idx + s
        G[m, i_idx] -= dx * h[j_idx]
        G[m, j_idx] += dx * (1.0 - h[i_idx])
    return G

def lse_value(h, beta):
    c = corr_components(h)
    c_max = c.max()
    shifted = beta * (c - c_max)
    return c_max + np.log(np.sum(np.exp(shifted))) / beta

def lse_grad(h, beta):
    c = corr_components(h)
    c_max = c.max()
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    G = grad_matrix(h)
    return G.T @ weights

    return np.sum(x) - 0.5 * _in_n

def _con_jac(x):
    return np.ones(_in_n)

def slsqp_smooth(h, beta, maxiter=300):
    global _in_n
    n = h.size
    _in_n = n
    h0 = project_box_sum(h, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:

                                method='SLSQP', bounds=bounds,
                                constraints=constraints,

        if np.all(np.isfinite(res.x)):

        alpha_max = 1.0
        if np.any(pos):

        if np.any(neg):

    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def lp_direction(h, k_active, bound_scale=0.05):
    n = h.size
    c = corr_components(h)
    order = np.argsort(c)[::-1]
    active = order[:k_active]
    G = grad_matrix(h)
    A = G[active, :]
    c_act = c[active]
    c_max = c.max()
    bounds_vec = bound_scale * np.maximum(0.01, np.minimum(h, 1.0 - h))
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((len(active), 1))])
    b_ub = (c_max - c_act) * 0.5
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bnds = [(max(-bounds_vec[i], -0.5), min(bounds_vec[i], 0.5)) for i in range(n)]
    bnds.append((-2.0, 2.0))
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                               bounds=bnds, method='highs')
        if res.success and res.x[-1] < -1e-10:
            d = res.x[:n].copy()
            d -= d.mean()
            return d
    except Exception:
        pass
    return None

def lp_polish(h, max_iters=500, need_s=10.0, deadline=None):
    n = h.size
    h = project_box_sum(h, n)
    F = exact_value(h)
    best_h = h.copy()
    best_F = F
    k_schedule = [1, 2, 3, 4, 6, 8]
    stall = 0
    for it in range(max_iters):
        if deadline is not None and (deadline - time.monotonic()) < need_s:
            break
        improved = False
        for k in k_schedule:
            d = lp_direction(h, k, bound_scale=0.03)
            if d is None:
                continue
            d_abs = np.abs(d).max()
            if d_abs < 1e-12:
                continue
            d = d / d_abs
            pos = d > 1e-12
            neg = d < -1e-12
            alpha_max = 1.0
            if np.any(pos):
                alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
            if np.any(neg):
                alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
            alpha_max = max(alpha_max, 1e-10)
            for frac in [1.0, 0.5, 0.25, 0.125]:
                a = frac * alpha_max
                h_new = project_box_sum(h + a * d, n)
                F_new = exact_value(h_new)
                if F_new < F - 1e-13:
                    h = h_new
                    F = F_new
                    improved = True
                    break
            if improved:
                if F < best_F:
                    best_F = F
                    best_h = h.copy()
                break
        if not improved:
            stall += 1
            if stall >= 5:
                break
    return best_h, best_F

def de_objective(x, n):
    h = project_box_sum(x, n)
    return exact_value(h)

def run(seed=42, budget_s=1100, **kwargs):
    global _in_n
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(1000 * (int(seed) if seed is not None else 0) + 17)

    best_h = None
    best_val = math.inf
    best_n = 0

    def consider(h_cand):
        nonlocal best_h, best_val, best_n
        if h_cand is None:
            return
        h_cand = np.asarray(h_cand, dtype=float).ravel()
        if h_cand.size <= 0:
            return
        n_c = h_cand.size
        h_cand = project_box_sum(h_cand, n_c)
        v = exact_value(h_cand)
        if not math.isfinite(v):
            return
        if v < best_val - 1e-12 or (abs(v - best_val) <= 1e-12 and n_c < best_n):
            best_h = h_cand.copy()
            best_val = float(v)
            best_n = n_c

    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).ravel()
        if hinit.size > 0 and np.all(np.isfinite(hinit)):
            hinit = project_box_sum(hinit, hinit.size)
            consider(hinit)

    if best_h is None:
        consider(np.full(45, 0.5))

    n_base = best_h.size

    # Phase 1: Log-sum-exp SLSQP with annealing on base resolution
    betas = [5, 10, 20, 40, 80, 160, 320, 640]
    for beta in betas:
        if deadline - time.monotonic() < 80:
            break
        h_s = slsqp_smooth(best_h.copy(), beta, maxiter=400)
        consider(h_s)

    # Phase 2: Exact subgradient polish
    if deadline - time.monotonic() > 60:
        h_p, v_p = exact_polish(best_h.copy(), max_iters=3000)
        consider(h_p)

    # Phase 3: LP-based direction polish
    if deadline - time.monotonic() > 80:
        h_l, v_l = lp_polish(best_h.copy(), max_iters=500, deadline=deadline, need_s=40)
        consider(h_l)

    # Phase 4: Try different resolutions
    n_tries = [30, 35, 40, 45, 50, 55, 60, 65, 70, 25, 20]
    for n_try in n_tries:
        if deadline - time.monotonic() < 60:
            break
        h_r = resample_to_n(best_h, n_try)
        if h_r is None:
            continue
        consider(h_r)
        v_r = exact_value(h_r)
        if v_r < best_val + 0.005 and deadline - time.monotonic() > 80:
            for beta in [20, 50, 100, 200, 400]:
                if deadline - time.monotonic() < 40:
                    break
                h_s = slsqp_smooth(h_r.copy(), beta, maxiter=300)
                consider(h_s)
            if deadline - time.monotonic() > 40:
                h_p2, v_p2 = exact_polish(best_h.copy(), max_iters=1500)
                consider(h_p2)

    # Phase 5: More SLSQP annealing from current best
    if best_h is not None and deadline - time.monotonic() > 100:
        n_cur = best_h.size
        for beta in [100, 200, 400, 800, 1600]:
            if deadline - time.monotonic() < 50:
                break
            h_s = slsqp_smooth(best_h.copy(), beta, maxiter=500)
            consider(h_s)

    # Phase 6: Differential evolution for global exploration
    if deadline - time.monotonic() > 200:
        n_de = min(n_base, 40)
        de_bounds = [(0.0, 1.0)] * n_de
        try:
            res_de = optimize.differential_evolution(
                de_objective, de_bounds, args=(n_de,),
                seed=int(seed) + 1, maxiter=80, popsize=25,
                tol=1e-8, mutation=(0.4, 0.9), recombination=0.8,
                updating='deferred', polish=False, init='latinhypercube')
            if np.all(np.isfinite(res_de.x)):
                h_de = project_box_sum(res_de.x, n_de)
                consider(h_de)
                if deadline - time.monotonic() > 60:
                    h_de_s = slsqp_smooth(h_de, 100, maxiter=300)
                    consider(h_de_s)
                    if deadline - time.monotonic() > 30:
                        h_de_p, _ = exact_polish(h_de_s, max_iters=1000)
                        consider(h_de_p)
        except Exception:
            pass

    # Phase 7: Multi-start with perturbations
    if best_h is not None and deadline - time.monotonic() > 150:
        n_cur = best_h.size
        for attempt in range(6):
            if deadline - time.monotonic() < 50:
                break
            sig = 0.02 * (attempt + 1)
            h_pert = project_box_sum(best_h + sig * rng.standard_normal(n_cur), n_cur)
            consider(h_pert)
            v_pert = exact_value(h_pert)
            if v_pert < best_val + 0.01 and deadline - time.monotonic() > 60:
                for beta in [50, 100, 200]:
                    if deadline - time.monotonic() < 30:
                        break
                    h_s = slsqp_smooth(h_pert, beta, maxiter=300)
                    consider(h_s)

    # Phase 8: LP polish on best
    if best_h is not None and deadline - time.monotonic() > 60:
        h_l2, v_l2 = lp_polish(best_h.copy(), max_iters=500, deadline=deadline, need_s=30)
        consider(h_l2)

    # Phase 9: Final exact polish
    if best_h is not None and deadline - time.monotonic() > 20:
        h_fin, v_fin = exact_polish(best_h.copy(), max_iters=5000)
        consider(h_fin)

    # Phase 10: One more SLSQP at very high beta
    if best_h is not None and deadline - time.monotonic() > 30:
        h_final_s = slsqp_smooth(best_h.copy(), 2000, maxiter=500)
        consider(h_final_s)
        if deadline - time.monotonic() > 15:
            h_final_p, _ = exact_polish(h_final_s, max_iters=2000)
            consider(h_final_p)

    if best_h is None:
        best_h = project_box_sum(np.full(45, 0.5), 45)
        best_val = exact_value(best_h)
        best_n = 45

    final_h = project_box_sum(best_h, best_n)
    final_val = exact_value(final_h)
    if not math.isfinite(final_val) or final_h.size == 0:
        final_h = project_box_sum(np.full(45, 0.5), 45)
        final_val = exact_value(final_h)
    return final_h, float(final_val), int(final_h.size)
