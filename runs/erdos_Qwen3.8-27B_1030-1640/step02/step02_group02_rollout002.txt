def evaluate(h):
    """Compute C5 value for given h vector."""
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return math.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_box_sum(y, n):
    """Project y onto {x: 0≤x_i≤1, sum(x)=n/2}."""
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    
    # Bisection on the water-filling threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    # Final exact fix
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

    return x[n]  # minimize t

def slsqp_objective_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_constraint_inf(x, n):
    """Inequality constraints: t - R[p]*dx >= 0 for all p."""
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return t - R * dx

    h = x[:n]
    dx = 2.0 / n
    L = 2 * n - 1

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

def lse_objective(x, n, beta):
    h = x[:n]
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full") * dx
    M = np.max(R)
    if beta * (R - M).max() < -700:  # underflow protection
        return M
    shifted = beta * (R - M)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return M + np.log(np.sum(np.exp(shifted))) / beta

def lse_jac(x, n, beta):
    h = x[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    R = np.correlate(h, 1.0 - h, mode="full") * dx
    M = np.max(R)
    shifted = beta * (R - M)
    # Softmax weights (numerically stable)
    shifted_max = np.max(shifted)
    w = np.exp(shifted - shifted_max)
    w /= np.sum(w)
    # Gradient: sum_p w_p * dR[p]/dh_j * dx
    G = np.zeros(n)
    for s in range(-(n-1), n):
        p = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        i_idx = np.arange(i_lo, i_hi)
        j_idx = i_idx + s
        G[i_idx] += -dx * w[p] * h[j_idx]
        G[j_idx] += dx * w[p] * (1.0 - h[i_idx])
    return G
