    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(y-mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0, 1)
    d = target - x.sum()
    if abs(d) > 1e-14:
        if d > 0:
            m = x < 1
            if m.any(): x[m] += d/m.sum()
        else:
            m = x > 0
            if m.any(): x[m] += d/m.sum()
        np.clip(x, 0, 1, out=x)
    return x

def _score(h):
    n = len(h)
    if n == 0: return math.inf
    dx = 2.0/n
    return float(np.max(np.correlate(h, 1.0-h, 'full')*dx))

def _cvals(h):
    n = len(h)
    dx = 2.0/n
    return np.correlate(h, 1.0-h, 'full')*dx

def _min_1d_convex(f, lo, hi, n_iter=80):
    """Minimize a convex function on [lo,hi] using golden section."""
    gr = (math.sqrt(5)-1)/2
    a, b = lo, hi
    c = b - gr*(b-a)
    d = a + gr*(b-a)
    fc, fd = f(c), f(d)
    for _ in range(n_iter):
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr*(b-a)
            fc = f(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr*(b-a)
            fd = f(d)
    return 0.5*(a+b)

def _coord_opt(h, n_passes=20, rng=None):
    """Coordinate descent with exact 1D minimization."""
    h = h.copy()
    n = len(h)
    dx = 2.0/n
    L = 2*n-1
    s_arr = np.arange(L) - (n-1)
    
    for p in range(n_passes):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        improved = False
        for i in order:
            # Compute a_k, b_k for coordinate i
            ip = i + s_arr
            im = i - s_arr
            a = np.zeros(L)
            mp = (ip>=0) & (ip<n)
            mm = (im>=0) & (im<n)
            a[mp] = dx*(1.0 - h[ip[mp]])
            a[mm] -= dx*h[im[mm]]
            
            c = _cvals(h)
            b = c - a*h[i]
            
            # Minimize max_k(a_k * t + b_k) for t in [0,1]
            # Use subgradient bisection
            lo_t, hi_t = 0.0, 1.0
            for _ in range(60):
                mid = 0.5*(lo_t+hi_t)
                vals = a*mid + b
                fmid = vals.max()
                active = vals >= fmid - 1e-12
                if not active.any():
                    break
                ms, xs = a[active].min(), a[active].max()
                if ms <= 0 <= xs:
                    break
                elif xs < 0:
                    lo_t = mid
                else:
                    hi_t = mid
            t_new = 0.5*(lo_t+hi_t)
            
            if abs(t_new - h[i]) > 1e-12:
                h[i] = t_new
                h = _proj(h, n)
                improved = True
        
        if not improved:
            break
    
    return h

def _pair_opt(h, n_iters=500, rng=None):
    """Pair moves with golden section 1D minimization."""
    h = h.copy()
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(0)
    
    for it in range(n_iters):
        i, j = rng.choice(n, 2, replace=False)
        lo = max(-h[i], h[j]-1.0)
        hi = min(1.0-h[i], h[j])
        if lo >= hi - 1e-14:
            continue
        
        def f(delta):
            h2 = h.copy()
            h2[i] += delta
            h2[j] -= delta
            return float(np.max(np.correlate(h2, 1.0-h2, 'full')*2.0/n))
        
        delta = _min_1d_convex(f, lo, hi, n_iter=50)
        if abs(delta) > 1e-14:
            h[i] += delta
            h[j] -= delta
    
    return h
