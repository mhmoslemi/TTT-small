```python
        a = np.zeros(n)
        b = 0.0
        i0 = max(0, -s)
        i1 = min(n-1, n-1-s)
        for i in range(i0, i1+1):
            j = i + s
            a[i] += (1.0 - h[j])
            a[j] += (-h[i])
            b += h[i] * h[j]
        
        # Constraint: dx * (a^T h + b) <= t
        # => dx * a^T h - t <= -dx * b
        row = np.concatenate([dx * a, [-1.0]])
        A_rows.append(row)
        b_vec.append(-dx * b)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vec)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    bounds = [(0, 1)] * n + [(-1.0, 1.0)]  # t can be negative but won't be at optimum
    
    res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs')
    
    if res.success:
        h_new = np.clip(res.x[:n], 0, 1)
        return project_box_sum(h_new, n)
    return None
```

```python
import time
import math
import numpy as np
from scipy import optimize

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    if y.size != n:
        z = np.full(n, 0.5, dtype=float)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * float(n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        diff = target - float(np.sum(x))
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            avail = 1.0 - x
            tot = float(np.sum(avail))
            if tot <= 0:
                break
            x += diff * (avail / tot)
        else:
            avail = x.copy()
            tot = float(np.sum(avail))
            if tot <= 0:
                break
            x += diff * (avail / tot)
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + diff)
    return x

def score(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return math.inf
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def overlap_vec(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, 'full') * dx

def get_shift_pairs(n):
    L = 2 * n - 1
    pairs = []
    for k_idx in range(L):
        s = k_idx - (n - 1)
        i0 = max(0, -s)
        i1 = min(n - 1, n - 1 - s)
        pairs.append([(i, i + s) for i in range(i0, i1 + 1)])
    return pairs

def dca_lp_step(h, n, pairs, active_thresh=0.0):
    dx = 2.0 / float(n)
    L = 2 * n - 1
    c_current = overlap_vec(h)
    c_max = float(np.max(c_current))
    if active_thresh > 0:
        active_mask = c_current >= (c_max - active_thresh)
    else:
        active_mask = np.ones(L, dtype=bool)
    n_active = int(np.sum(active_mask))
    if n_active == 0:
        return None
    A_rows = []
    b_vec = []
    for k_idx in range(L):
        if not active_mask[k_idx]:
            continue
        a = np.zeros(n)
        b = 0.0
        for i, j in pairs[k_idx]:
            a[i] += (1.0 - h[j])
            a[j] += (-h[i])
            b += h[i] * h[j]
        row = np.empty(n + 1)
        row[:n] = dx * a
        row[n] = -1.0
        A_rows.append(row)
        b_vec.append(-dx * b)
    if len(A_rows) == 0:
        return None
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vec)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq_arr = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-0.5, 1.0)]
    try:
        res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                               b_eq=b_eq_arr, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            return project_box_sum(h_new, n)
    except Exception:
        pass
    return None

def dca_line_search(h, h_lp, n, max_alpha=1.0):
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_v = score(h)
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
        if alpha > max_alpha:
            continue
        h_try = h + alpha * (h_lp - h)
        np.clip(h_try, 0.0, 1.0, out=h_try)
        h_try = project_box_sum(h_try, n)
        v = score(h_try)
        if v < best_v - 1e-12:
            best_h = h_try
            best_v = v
    return best_h, best_v

def fast_coord_descent(h, n_passes=10, deadline=None, need_s=20.0):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = len(h)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    pairs = get_shift_pairs(n)
    best_h = h.copy()
    best_val = score(h)
    for p in range(n_passes):
        if deadline is not None and (deadline - time.monotonic()) < need_s:
            break
        order = np.arange(n)
        for i in order:
            a_k = np.zeros(L)
            b_k = np.zeros(L)
            for k_idx in range(L):
                s = k_idx - (n - 1)
                coeff = 0.0
                const = 0.0
                for j_i, j_j in pairs[k_idx]:
                    if j_i == i:
                        coeff += (1.0 - h[j_j])
                        const += 0.0
                    elif j_j == i:
                        coeff += (-h[j_i])
                        const += h[j_i]
                    else:
                        const += h[j_i] * (1.0 - h[j_j])
                # c_k = dx * (coeff * t + const) where t = h[i]
                # Wait: c_k = dx * sum h[j]*(1-h[j-s])
                # Terms with h[i]: if i appears as j: h[i]*(1-h[i-s])
                #                  if i appears as j-s: h[j]*(1-h[i]) where j = i+s
                a_k[k_idx] = dx * coeff
                b_k[k_idx] = dx * const
            # Minimize max_k (a_k[k]*t + b_k[k]) over t in [0,1]
            # Use golden section search
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            lo_t = 0.0
            hi_t = 1.0
            for _ in range(35):
                t1 = lo_t + gr * (hi_t - lo_t)
                t2 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(a_k * t1 + b_k))
                f2 = float(np.max(a_k * t2 + b_k))
                if f1 < f2:
                    hi_t = t2
                else:
                    lo_t = t1
            h[i] = 0.5 * (lo_t + hi_t)
        h = project_box_sum(h, n)
        v = score(h)
        if v < best_val - 1e-12:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def pair_move_cd(h, n_pairs=200, deadline=None, need_s=15.0):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = len(h)
    rng = np.random.default_rng(12345)
    best_h = h.copy()
    best_val = score(h)
    for _ in range(n_pairs):
        if deadline is not None and (deadline - time.monotonic()) < need_s:
            break
        a, b = rng.integers(0, n, size=2)
        if a == b:
            continue
        d_min = max(-h[a], -(1.0 - h[b]))
        d_max = min(1.0 - h[a], h[b])
        if d_min >= d_max:
            continue
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        lo_d, hi_d = d_min, d_max
        for _ in range(25):
            t1 = lo_d + gr * (hi_d - lo_d)
            t2 = hi_d - gr * (hi_d - lo_d)
            h_test = h.copy()
            h_test[a] += t1
            h_test[b] -= t1
            f1 = score(h_test)
            h_test[a] += t2
            h_test[b] -= t2
            f2 = score(h_test)
            if f1 < f2:
                hi_d = t2
            else:
                lo_d = t1
        delta = 0.5 * (lo_d + hi_d)
        h[a] += delta
        h[b] -= delta
        np.clip(h, 0.0, 1.0, out=h)
        h = project_box_sum(h, n)
        v = score(h)
        if v < best_val - 1e-12:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def resample_h(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = len(h)
    n_new = int(n_new)
    if n <= 0 or n_new <= 0 or n_new == n:
        return project_box_sum(h, n) if n > 0 else np.empty(0)
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def make_fallback(n=45):
    h = np.full(n, 0.5, dtype=float)
    rng = np.random.default_rng(99)
    k = n // 2
    idx = rng.choice(n, k, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[rng.choice(n)] = 0.5
    else:
        pass
    # Fix sum
    rem = n // 2 - int(np.sum(h > 0.5))
    fracs = np.where(h == 0.5)[0]
    for i in range(rem):
        if i < len(fracs):
            h[fracs[i]] = 1.0
    return project_box_sum(h, n)

def sym_h(h):
    h = np.asarray(h, dtype=float).reshape(-1)

def subgradient_step(h, n, step_size, n_iter=200, deadline=None, need_s=20.0):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = len(h)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_val = score(h)
    pairs = get_shift_pairs(n)
    alpha = step_size
    for _ in range(n_iter):
        if deadline is not None and (deadline - time.monotonic()) < need_s:
            break
        c = overlap_vec(h)
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        g = np.zeros(n)
        for i, j in pairs[k_star]:
            g[i] += dx * (1.0 - h[j])
            g[j] += -dx * h[i]
        y = h - alpha * g
        y = project_box_sum(y, n)
        v = score(y)
        if v < best_val - 1e-12:
            best_val = v
            best_h = y.copy()
            h = y
            alpha = min(5.0, alpha * 1.03)
        else:
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_val

def nelder_mead_1d(h, n, idx, n_iter=30):

        const = 0.0
        for j_i, j_j in pairs[k_idx]:
            if j_i == idx:
                coeff += (1.0 - h[j_j])
            elif j_j == idx:
                coeff += (-h[j_i])
                const += h[j_i]
            else:
                const += h[j_i] * (1.0 - h[j_j])
        a_k[k_idx] = dx * coeff
        b_k[k_idx] = dx * const
    lo_t = 0.0
    hi_t = 1.0
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_iter):
        t1 = lo_t + gr * (hi_t - lo_t)
        t2 = hi_t - gr * (hi_t - lo_t)
        f1 = float(np.max(a_k * t1 + b_k))
        f2 = float(np.max(a_k * t2 + b_k))
        if f1 < f2:
            hi_t = t2
        else:
            lo_t = t1
    return 0.5 * (lo_t + hi_t)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(int(seed) * 1000 + 7)

    best_h = None
    best_val = math.inf
    best_n = 0

    def consider(h):
        nonlocal best_h, best_val, best_n
        if h is None or len(h) == 0:
            return
        h = project_box_sum(h, len(h))
        if not np.all(np.isfinite(h)):
            return
        if np.min(h) < -1e-9 or np.max(h) > 1.0 + 1e-9:
            return
        v = score(h)
        if not math.isfinite(v):
            return
        if v < best_val - 1e-12 or (abs(v - best_val) <= 1e-12 and len(h) < best_n):
            best_h = h.copy()
            best_val = float(v)
            best_n = len(h)

    # Get initial
    init = globals().get('initial_h_values', None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0 and np.all(np.isfinite(hinit)):
            hinit = project_box_sum(hinit, len(hinit))
            consider(hinit)
            consider(sym_h(hinit))

    if best_h is None:
        consider(make_fallback(45))

    if best_h is None:
        h0 = np.full(45, 0.5, dtype=float)
        consider(h0)

    n_cur = best_n
    pairs_cur = get_shift_pairs(n_cur)

    # Phase 1: DCA with LP (main driver)
    max_dca_iters = 80
    for it in range(max_dca_iters):
        if deadline - time.monotonic() < 80:
            break
        # Adaptive active threshold: start tight, loosen if stuck
        thresh = 0.0 if it < 20 else (0.005 + 0.001 * min(it - 20, 40))
        h_lp = dca_lp_step(best_h, n_cur, pairs_cur, active_thresh=thresh)
        if h_lp is not None:
            v_lp = score(h_lp)
            if v_lp < best_val - 1e-10:
                best_h = h_lp
                best_val = v_lp
            else:
                # Try line search
                h_ls, v_ls = dca_line_search(best_h, h_lp, n_cur)
                if v_ls < best_val - 1e-10:
                    best_h = h_ls
                    best_val = v_ls
        # Periodic coordinate descent polish
        if it % 5 == 4:
            h_cd, v_cd = fast_coord_descent(best_h, n_passes=3, deadline=deadline, need_s=40)
            if v_cd < best_val - 1e-10:
                best_h = h_cd
                best_val = v_cd

    # Phase 2: Aggressive coordinate descent
    if deadline - time.monotonic() > 60:
        h_cd, v_cd = fast_coord_descent(best_h, n_passes=25, deadline=deadline, need_s=30)
        if v_cd < best_val - 1e-12:
            best_h = h_cd
            best_val = v_cd

    # Phase 3: Pair moves
    if deadline - time.monotonic() > 40:
        h_pm, v_pm = pair_move_cd(best_h, n_pairs=300, deadline=deadline, need_s=20)
        if v_pm < best_val - 1e-12:
            best_h = h_pm
            best_val = v_pm

    # Phase 4: Subgradient for different basin escape
    if deadline - time.monotonic() > 50:
        h_sg, v_sg = subgradient_step(best_h, n_cur, step_size=0.5, n_iter=200,
                                      deadline=deadline, need_s=30)
        if v_sg < best_val - 1e-12:
            best_h = h_sg
            best_val = v_sg
        # Coordinate descent after subgradient
        h_cd2, v_cd2 = fast_coord_descent(h_sg, n_passes=10, deadline=deadline, need_s=20)
        if v_cd2 < best_val - 1e-12:
            best_h = h_cd2
            best_val = v_cd2

    # Phase 5: Try different n values
    if best_h is not None and deadline - time.monotonic() > 60:
        cur_n = best_n
        for d in [1, -1, 2, -2, 3, -3, 4, -4, 6, -6]:
            if deadline - time.monotonic() < 45:
                break
            nn = cur_n + d
            if nn < 30 or nn > 80:
                continue
            hr = resample_h(best_h, nn)
            hr = project_box_sum(hr, nn)
            vh = score(hr)
            if vh < best_val + 0.005 and deadline - time.monotonic() > 35:
                # Refine at new n
                pairs_nn = get_shift_pairs(nn)
                for _ in range(15):
                    if deadline - time.monotonic() < 30:
                        break
                    h_lp = dca_lp_step(hr, nn, pairs_nn, active_thresh=0.002)
                    if h_lp is not None:
                        v_lp = score(h_lp)
                        if v_lp < vh - 1e-10:
                            hr = h_lp
                            vh = v_lp
                    if _ % 5 == 4:
                        hr2, vh2 = fast_coord_descent(hr, n_passes=3, deadline=deadline, need_s=25)
                        if vh2 < vh:
                            hr = hr2
                            vh = vh2
                consider(hr)

    # Phase 6: Random perturbations + refine
    if best_h is not None and deadline - time.monotonic() > 50:
        for sig in [0.02, 0.05, 0.1]:
            if deadline - time.monotonic() < 30:
                break
            hp = best_h.copy() + sig * rng.standard_normal(len(best_h))
            hp = project_box_sum(hp, len(best_h))
            vh = score(hp)
            if vh < best_val + 0.01 and deadline - time.monotonic() > 25:
                pairs_h = get_shift_pairs(len(best_h))
                h_lp = dca_lp_step(hp, len(best_h), pairs_h, active_thresh=0.001)
                if h_lp is not None:
                    consider(h_lp)
                h_cd3, v_cd3 = fast_coord_descent(hp, n_passes=8, deadline=deadline, need_s=15)
                consider(h_cd3)

    # Phase 7: Final polish - more DCA + coord descent
    if best_h is not None and deadline - time.monotonic() > 25:
        n_f = best_n
        pairs_f = get_shift_pairs(n_f)
        h_f = best_h.copy()
        v_f = best_val
        for it in range(40):
            if deadline - time.monotonic() < 15:
                break
            h_lp = dca_lp_step(h_f, n_f, pairs_f, active_thresh=0.0005)
            if h_lp is not None:
                v_lp = score(h_lp)
                if v_lp < v_f - 1e-11:
                    h_f = h_lp
                    v_f = v_lp
            if it % 5 == 4:
                h_f2, v_f2 = fast_coord_descent(h_f, n_passes=5, deadline=deadline, need_s=10)
                if v_f2 < v_f - 1e-11:
                    h_f = h_f2
                    v_f = v_f2
        consider(h_f)

    # Phase 8: Very fine coordinate descent
    if best_h is not None and deadline - time.monotonic() > 15:
        h_fin, v_fin = fast_coord_descent(best_h, n_passes=30, deadline=deadline, need_s=5)
        consider(h_fin)
        if deadline - time.monotonic() > 8:
            h_fin2, v_fin2 = pair_move_cd(h_fin, n_pairs=200, deadline=deadline, need_s=3)
            consider(h_fin2)

    if best_h is None:
        best_h = project_box_sum(np.full(45, 0.5, dtype=float), 45)
        best_val = score(best_h)
        best_n = 45

    return best_h, float(best_val), int(best_n)
```
