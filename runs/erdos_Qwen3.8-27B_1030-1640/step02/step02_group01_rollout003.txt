            break

        # Weight active constraints
        active_thresh = C - 0.001
        active = np.where(c >= active_thresh)[0]
        if len(active) == 0:

    h = clean_feasible(h, n)
    L = 2 * n - 1
    dx = 2.0 / n

    # infl[i][m] = (1 - h[i - s]) * dx if 0 <= i-s < n else 0
    # where s = m - (n-1)
    infl = np.zeros((n, L))
    for i in range(n):
        for m in range(L):
            s = m - (n - 1)
            j = i - s
            if 0 <= j < n:
                infl[i, m] = (1.0 - h[j]) * dx
    
    best_C = float(np.max(overlap_components(h)))
    delta_grid = np.linspace(-0.1, 0.1, 21)
    
    for p in range(max_passes):
        if not enough_time(deadline, need):
            break
        improved = False
        C_vec = overlap_components(h)
        for i in range(n):
            for j in range(i+1, n):
                d = infl[j] - infl[i]
                # Try different delta values
                best_delta = 0.0
                best_f = float(np.max(C_vec))
                for delta in delta_grid:
                    # Check bounds
                    if h[i] - delta < -1e-10 or h[i] - delta > 1 + 1e-10:

                        best_delta = delta
                if best_delta != 0.0:
                    h[i] -= best_delta
                    h[j] += best_delta

            if tot <= 0:
                break
            x += diff / tot * avail
        else:
            tot = float(np.sum(x))
            if tot <= 0:
                break
            x += diff / tot * x
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))
    return x


def exact_value(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return math.inf
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))


def overlap_components(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return np.empty(0, dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def overlap_grad_matrix(h):
    n = int(h.size)
    if n <= 0:
        return np.empty((0, 0), dtype=float)
    L = 2 * n - 1
    dx = 2.0 / float(n)
    G = np.zeros((L, n), dtype=float)
    for m in range(L):
        s = m - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, s + n)
        if lo1 < hi1:
            G[m, lo1:hi1] += dx * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            G[m, lo2:hi2] -= dx * h[lo2 + s:hi2 + s]
    return G


def sym_candidate(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return h
    return project_box_sum(0.5 * (h + h[::-1]), n)


def resample_candidate(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    n_new = int(n_new)
    if n <= 0 or n_new <= 0:
        return None
    if n_new == n:
        return clean_feasible(h, n)
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)


# ===== Cutting Plane =====

def solve_lp_cuts(h_pool, G_pool, c_pool, n):
    n_cuts = len(h_pool)
    L = 2 * n - 1
    A_ub = np.zeros((n_cuts * L, n + 1), dtype=float)
    b_ub = np.zeros(n_cuts * L, dtype=float)
    for k in range(n_cuts):
        Gk = G_pool[k]
        ck = c_pool[k]
        hk = h_pool[k]
        A_ub[k * L:(k + 1) * L, :n] = Gk
        A_ub[k * L:(k + 1) * L, -1] = -1.0
        b_ub[k * L:(k + 1) * L] = ck - Gk.dot(hk)
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * float(n)], dtype=float)
    try:
        res = optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
            A_eq=A_eq, b_eq=b_eq, method='highs',
            options={"presolve": True, "time_limit": 30}
        )
        if res.success:
            return np.clip(res.x[:n], 0.0, 1.0), float(res.x[-1])
    except Exception:
        pass
    return None, None


def cutting_plane_optimize(h0, max_iters=100, deadline=None, need=20.0):
    n = int(np.asarray(h0).size)
    h = clean_feasible(h0, n)
    if n <= 0:
        return h, math.inf
    c = overlap_components(h)
    C = float(np.max(c))
    G = overlap_grad_matrix(h)
    h_pool = [h.copy()]
    G_pool = [G.copy()]
    c_pool = [c.copy()]
    max_pool = 40
    for it in range(max_iters):
        if deadline is not None and not enough_time(deadline, need):
            break
        h_new, s_new = solve_lp_cuts(h_pool, G_pool, c_pool, n)
        if h_new is None:
            break
        h_new = project_box_sum(np.clip(h_new, 0.0, 1.0), n)
        c_new = overlap_components(h_new)
        C_new = float(np.max(c_new))
        if abs(C_new - C) < 1e-12 and float(np.max(np.abs(h_new - h))) < 1e-10:
            h = h_new
            C = C_new
            break
        G_new = overlap_grad_matrix(h_new)
        if len(h_pool) < max_pool:
            h_pool.append(h_new.copy())
            G_pool.append(G_new.copy())
            c_pool.append(c_new.copy())
        else:
            # Replace the cut with the largest margin (least active)
            margins = []
            for k in range(len(h_pool)):
                margins.append(float(np.max(c_pool[k] - G_pool[k].dot(h))))
            idx = int(np.argmax(margins))
            h_pool[idx] = h_new.copy()
            G_pool[idx] = G_new.copy()
            c_pool[idx] = c_new.copy()
        h = h_new
        C = C_new
    return h, C


# ===== Pairwise Transfer =====

def find_optimal_delta(C_vec, d, delta_lo, delta_hi):
    if delta_lo >= delta_hi:
        return 0.0
    lo, hi = delta_lo, delta_hi
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C_vec + mid * d
        max_val = float(np.max(vals))
        active = np.where(np.abs(vals - max_val) < 1e-9)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(vals))])
        min_sg = float(d[active].min())
        max_sg = float(d[active].max())
        if min_sg <= 1e-11 and max_sg >= -1e-11:
            break
        if min_sg > 0:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)


def pairwise_transfer(h, max_passes=300, deadline=None, need=10.0):
    n = int(h.size)
    h = h.copy()
    L = 2 * n - 1

    C_vec = overlap_components(h)
    best_C = float(np.max(C_vec))
    step_cap = 0.04

    for p in range(max_passes):
        if deadline is not None and not enough_time(deadline, need):
            break
        G = overlap_grad_matrix(h)
        C_vec = overlap_components(h)
        current_max = float(np.max(C_vec))
        improved = False

        for i in range(n):
            for j in range(i + 1, n):
                d = G[:, j] - G[:, i]
                # Bounds on delta
                d_lo = max(-(1.0 - h[i]), -h[j], -step_cap)
                d_hi = min(h[i], 1.0 - h[j], step_cap)
                if d_lo >= d_hi:
                    continue
                delta = find_optimal_delta(C_vec, d, d_lo, d_hi)
                if abs(delta) < 1e-10:
                    continue
                # Apply and check
                old_h = h.copy()
                h[i] -= delta
                h[j] += delta
                np.clip(h, 0.0, 1.0, out=h)
                h = project_box_sum(h, n)
                new_val = exact_value(h)
                if new_val < current_max - 1e-13:
                    C_vec = overlap_components(h)
                    current_max = new_val
                    if current_max < best_C:
                        best_C = current_max
                    improved = True
                    # Update G row for changed entries (approximate - just recompute)
                    G = overlap_grad_matrix(h)
                else:
                    h = old_h

        if not improved:
            break

    return h, exact_value(h)


# ===== LSE-based smoothing for final polish =====

def lse_FG(x, tau):
    n = int(x.size)
    if n <= 0:
        return 0.0, np.empty(0, dtype=float)
    c = overlap_components(x)
    if c.size == 0:
        return 0.0, np.zeros(n, dtype=float)
    maxc = float(np.max(c))
    z = (c - maxc) / tau
    z = z - z.max()
    ez = np.exp(z)
    ss = float(np.sum(ez))
    if ss <= 0 or not np.isfinite(ss):
        F = maxc
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        F = maxc + tau * (math.log(ss) - math.log(float(c.size)))
        w = ez / ss
    G = overlap_grad_matrix(x)
    grad = G.T.dot(w)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=float)
    return float(F), np.asarray(grad, dtype=float)


def gradient_polish(h, tau=0.005, max_iter=200, deadline=None, need=5.0):
    n = int(h.size)
    h = clean_feasible(h, n)
    if n <= 0:
        return h
    F, g = lse_FG(h, tau)
    alpha = min(2.0, 2.0 / (float(np.linalg.norm(g, ord=np.inf)) + 1e-8))
    for it in range(max_iter):
        if deadline is not None and not enough_time(deadline, need):
            break
        h_new = project_box_sum(h - alpha * g, n)
        F_new, g_new = lse_FG(h_new, tau)
        if np.isfinite(F_new) and F_new < F - 1e-12:
            h = h_new
            F = F_new
            g = g_new
            alpha = min(3.0, alpha * 1.05)
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                break
    return h


# ===== Epigraph SLSQP =====

def _epi_eq_fun(x):
    return float(np.sum(x[:-1]) - 0.5 * float(x.size - 1))


def _epi_eq_jac(x):
    J = np.zeros((1, x.size))
    J[0, :-1] = 1.0
    return J


def _epi_ineq_fun(x):
    h = np.clip(x[:-1], 0.0, 1.0)
    c = overlap_components(h)
    return float(x[-1]) - c


def _epi_ineq_jac(x):
    h = np.clip(x[:-1], 0.0, 1.0)
    G = overlap_grad_matrix(h)
    L = G.shape[0]
    J = np.zeros((L, x.size))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J


def run_epigraph_slqp(h0, s_start, s_ub, maxiter=80, ftol=1e-11):
    h0 = clean_feasible(h0, int(h0.size))
    n = int(h0.size)
    if n <= 0:
        return None
    cmax = exact_value(h0)
    s_start = float(max(s_start, cmax + 1e-8))
    s_ub = float(max(s_ub, s_start + 1e-5, 0.95))
    x0 = np.empty(n + 1, dtype=float)
    x0[:-1] = h0
    x0[-1] = s_start
    bounds = [(0.0, 1.0)] * n + [(s_start, s_ub)]
    cons = [
        {"type": "eq", "fun": _epi_eq_fun, "jac": _epi_eq_jac},
        {"type": "ineq", "fun": _epi_ineq_fun, "jac": _epi_ineq_jac},
    ]

    def obj(x):
        return float(x[-1])

    def grad(x):
        g = np.zeros(x.size)
        g[-1] = 1.0
        return g

    try:
        res = optimize.minimize(
            obj, x0, jac=grad, bounds=bounds, constraints=cons,
            method="SLSQP", options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": 0}
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=float).reshape(-1)
            if x.size == n + 1 and np.all(np.isfinite(x)):
                return clean_feasible(x[:-1], n)
    except Exception:
        pass
    return None


# ===== Main run =====

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)

    best = {"h": None, "val": math.inf, "n": 0}

    def consider(h):
        if h is None or h.size <= 0:
            return
        h = clean_feasible(h, int(h.size))
        v = exact_value(h)
        if not math.isfinite(v):
            return
        if v < best["val"] - 1e-12 or (abs(v - best["val"]) <= 1e-12 and h.size < best["n"]):
            best["h"] = h.copy()
            best["val"] = float(v)
            best["n"] = int(h.size)

    # Get initial
    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0:
            hinit = clean_feasible(hinit, int(hinit.size))
            consider(hinit)
            consider(sym_candidate(hinit))

    if best["h"] is None:
        h_fb = np.full(44, 0.5, dtype=float)
        consider(h_fb)

    if best["h"] is None:
        h_out = np.full(44, 0.5, dtype=float)
        return h_out, 0.5, 44

    # Phase 1: Cutting plane
    if enough_time(deadline, 150):
        hp, vp = cutting_plane_optimize(best["h"], max_iters=120, deadline=deadline, need=40)
        consider(hp)

    # Phase 2: Pairwise transfer
    if enough_time(deadline, 120):
        hp2, vp2 = pairwise_transfer(best["h"], max_passes=200, deadline=deadline, need=30)
        consider(hp2)

    # Phase 3: Gradient polish with small tau
    if enough_time(deadline, 60):
        for tau in [0.005, 0.002, 0.001]:
            if not enough_time(deadline, 30):
                break
            hg = gradient_polish(best["h"], tau=tau, max_iter=300, deadline=deadline, need=15)
            consider(hg)

    # Phase 4: More cutting plane from improved point
    if enough_time(deadline, 100):
        hp3, vp3 = cutting_plane_optimize(best["h"], max_iters=80, deadline=deadline, need=35)
        consider(hp3)

    # Phase 5: Pairwise transfer again
    if enough_time(deadline, 100):
        hp4, vp4 = pairwise_transfer(best["h"], max_passes=200, deadline=deadline, need=20)
        consider(hp4)

    # Phase 6: Epigraph SLSQP refinement
    if enough_time(deadline, 80):
        margins = [0.001, 0.0005, 0.0002, 0.0001]
        for m in margins:
            if not enough_time(deadline, 30):
                break
            bv = best["val"]
            he = run_epigraph_slqp(best["h"], bv + m, bv + 0.05, maxiter=100, ftol=1e-11)
            consider(he)
            # Follow up with pairwise
            if enough_time(deadline, 20):
                hp5, _ = pairwise_transfer(best["h"], max_passes=100, deadline=deadline, need=10)
                consider(hp5)

    # Phase 7: Try nearby n values
    if enough_time(deadline, 60) and best["h"] is not None:
        cur_n = best["n"]
        for dn in [2, -2, 4, -4]:
            if not enough_time(deadline, 30):
                break
            nn = cur_n + dn
            if nn < 30 or nn > 80 or nn % 2 != 0:
                continue
            hr = resample_candidate(best["h"], nn)
            if hr is not None:
                consider(hr)
                # Quick optimize
                if enough_time(deadline, 25):
                    hpc, _ = cutting_plane_optimize(hr, max_iters=40, deadline=deadline, need=15)
                    consider(hpc)
                if enough_time(deadline, 20):
                    hpp, _ = pairwise_transfer(best["h"] if best["n"] == nn else hr,
                                                max_passes=80, deadline=deadline, need=10)
                    if best["n"] != nn:
                        hpp = clean_feasible(hpp, nn)
                    consider(hpp)

    # Phase 8: Final gradient polish
    if enough_time(deadline, 30):
        for tau in [0.001, 0.0005, 0.0003]:
            if not enough_time(deadline, 15):
                break
            hg = gradient_polish(best["h"], tau=tau, max_iter=200, deadline=deadline, need=10)
            consider(hg)

    # Phase 9: Final pairwise squeeze
    if enough_time(deadline, 20):
        hp_final, _ = pairwise_transfer(best["h"], max_passes=100, deadline=deadline, need=5)
        consider(hp_final)

    # Phase 10: Final SLSQP
    if enough_time(deadline, 25):
        bv = best["val"]
        he = run_epigraph_slqp(best["h"], bv + 0.0001, bv + 0.03, maxiter=120, ftol=1e-12)
        consider(he)

    final_h = best["h"]
    if final_h is None or final_h.size <= 0:
        final_h = np.full(44, 0.5, dtype=float)
    final_h = clean_feasible(final_h, int(final_h.size))
    final_val = exact_value(final_h)
    if not math.isfinite(final_val):
        final_h = np.full(44, 0.5, dtype=float)
        final_val = 0.5
    return final_h, float(final_val), int(final_h.size)
