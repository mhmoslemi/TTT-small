def grad_shift(h, s):
    n = h.size
    dx = 2.0 / n
    g = np.zeros(n)

        g[:] = dx * (1.0 - 2.0 * h)

    else:
        t = -s
        g[t:] += dx * (1.0 - h[:n - t])
        g[:n - t] -= dx * h[t:]
    return g
