def project_box_sum(y, n):
    """Project y to [0,1]^n with sum = n/2."""
    y = np.asarray(y, dtype=float).reshape(-1).copy()

        m = min(y.size, n)
        if m > 0:

    target = 0.5 * n

    lo = y.min() - 1.0
    hi = y.max() + 1.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0, 1)
    # Fix sum

            break

            mask = x < 1.0
            if not mask.any():

        else:

            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = np.argmax(x)
            x[idx] = max(0.0, x[idx] + diff)
