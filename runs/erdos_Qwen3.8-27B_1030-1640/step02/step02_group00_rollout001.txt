def compute_corr_grad_row(h, m_idx):
    """Compute gradient of correlation value at index m_idx (0-indexed in np.correlate full mode)."""

    k = m_idx - n + 1  # lag

    """Optimize the transfer between positions i and j using ternary search."""
    n = len(h)
    c = h[i] + h[j]  # sum to preserve
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if lo >= hi:
        return h, exact_c5(h)
    
    def eval_at(val):
        h_trial = h.copy()
        h_trial[i] = val
        h_trial[j] = c - val
        # No need to project since we're on the line
        return exact_c5(h_trial)
    
    # Ternary search
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = eval_at(m1)
        f2 = eval_at(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    h_new = h.copy()
    h_new[i] = val_opt
    h_new[j] = c - val_opt
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # Ensure exact sum
    h_new = project_feasible(h_new, 0.5 * n)
    return h_new, exact_c5(h_new)

```python
def smooth_grad(h, beta):
    """Compute smoothed log-sum-exp gradient projected to sum-zero subspace."""
    n = len(h)
    cv = corr_values(h)
    L = len(cv)
    c_max = float(np.max(cv))
    exponents = beta * (cv - c_max)
    exponents = np.clip(exponents, -50, 50)
    w = np.exp(exponents)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros(L)
        w[int(np.argmax(cv))] = 1.0
        w_sum = 1.0
    w = w / w_sum
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for m in range(L):
        if w[m] < 1e-12:
            continue
        wk = w[m] * dx
        k = m - (n - 1)  # lag
        if k >= 0:
            # C_k = dx * sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
            # d/dh[p] = dx*(1-h[p-k]) if k<=p<n, plus -dx*h[p+k] if p<=n-1-k
            p_range = range(k, n)  # where h[p] appears as h[i+k] with i=p-k
            for p in p_range:
                grad[p] += wk * (1.0 - h[p - k])
            p_range2 = range(0, n - k)  # where h[p] appears in (1-h[i]) with i=p
            for p in p_range2:
                grad[p] -= wk * h[p + k]
        else:
            s = -k  # positive
            # C_{-s} = dx * sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
            # d/dh[p] = dx*(1-h[p+s]) if p<=n-1-s, plus -dx*h[p-s] if s<=p<n
            for p in range(0, n - s):
                grad[p] += wk * (1.0 - h[p + s])
            for p in range(s, n):
                grad[p] -= wk * h[p - s]
    
    grad -= np.mean(grad)
    return grad, c_max, cv
```
