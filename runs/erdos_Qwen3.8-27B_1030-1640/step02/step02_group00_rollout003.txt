def project_feasible(h, n):

def c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c5(h):
    return float(np.max(c5_vec(h)))


def grad_row(h, m, dx):
    n = len(h)
    g = np.zeros(n)
    k = m - n + 1
    if k >= 0:
        if n - k > 0:
            g[:n - k] += dx * (1.0 - h[k:])
            g[k:] += -dx * h[:n - k]
    else:
        s = -k
        if n - s > 0:
            g[s:] += dx * (1.0 - h[:n - s])
            g[:n - s] += -dx * h[s:]
    g -= g.mean()
    return g


def lse_grad(h, beta):

    dx = 2.0 / n
    P = c5_vec(h)
    C = P.max()
    shifted = beta * (P - C)
    shifted = np.clip(shifted, -500, 500)

    for m in range(len(P)):

            continue
        grad += w[m] * grad_row(h, m, dx)
    return grad

def pair_descent(h, c_cur, max_rounds, rng, stop_time):

            if c_new < c - EPS:
                h = h_new
                c = c_new
                improved = True
                break
        if not improved:
            break
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1100.0
    t_end = t0 + budget
    deadline = t_end - 25.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 10**9

    # Get parent
    parent = None
    try:
        ph = globals().get("initial_h_values")
        if ph is not None:
            arr = np.asarray(ph, dtype=float).reshape(-1)
            if arr.size >= 4 and np.all(np.isfinite(arr)):
                parent = project_feasible(arr, arr.size)
    except Exception:
        pass

    if parent is not None:
        c_p = exact_c5(parent)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, parent, c_p, len(parent))
        h_sym = symmetrize(parent)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_sym, exact_c5(h_sym), len(h_sym))
        h_comp = project_feasible(1.0 - parent, len(parent))
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_comp, exact_c5(h_comp), len(h_comp))
    else:
        n_fb = 64
        h_fb = np.zeros(n_fb)
        h_fb[16:48] = 1.0
        h_fb = project_feasible(h_fb, n_fb)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_fb, exact_c5(h_fb), n_fb)

    n0 = best_n

    # Phase 1: LP polish at parent dimension
    if time.monotonic() < deadline - 200:
        h1, c1 = polish_lp(best_h, exact_c5(best_h), deadline - 200, max_iter=100, r0=0.015, r_max=0.04)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h1, c1, len(h1))
        h_sym = symmetrize(h1)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_sym, exact_c5(h_sym), len(h_sym))
        h_comp = project_feasible(1.0 - h1, len(h1))
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_comp, exact_c5(h_comp), len(h_comp))

    # Phase 2: LSE polish
    if time.monotonic() < deadline - 150:
        h2, c2 = polish_lse(best_h, exact_c5(best_h), deadline - 150,
                            betas=[1000, 3000, 10000, 30000], max_iter_per_beta=15, scale=0.05)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h2, c2, len(h2))
        h_sym = symmetrize(h2)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_sym, exact_c5(h_sym), len(h_sym))

    # Phase 3: Active constraint polish
    if time.monotonic() < deadline - 120:
        h3, c3 = active_constraint_polish(best_h, exact_c5(best_h), deadline - 120, tol=0.003, max_iter=30)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h3, c3, len(h3))

    # Phase 4: Pair descent
    if time.monotonic() < deadline - 100:
        h4, c4 = pair_descent(best_h, exact_c5(best_h), 120, rng, deadline - 100)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h4, c4, len(h4))

    # Phase 5: Multi-dimension exploration
    dims = sorted(set([n0 + d for d in [-8, -6, -4, -3, -2, -1, 0, 1, 2, 3, 4, 6, 8, 12, 16, 24]
                       if 24 <= n0 + d <= 110]))
    for nid in dims:
        if time.monotonic() > deadline - 80:
            break
        h_rep = resample_h(best_h, nid)
        c_rep = exact_c5(h_rep)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_rep, c_rep, nid)
        if nid <= 90 and time.monotonic() < deadline - 80:
            h_pol, c_pol = polish_lp(h_rep, c_rep, deadline - 80, max_iter=40, r0=0.01, r_max=0.03)
            best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_pol, c_pol, nid)
            h_sym = symmetrize(h_pol)
            best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_sym, exact_c5(h_sym), nid)

    # Phase 6: Coarse-to-fine
    if time.monotonic() < deadline - 60:
        for nc in [24, 32, 40, 48, 56]:
            if time.monotonic() > deadline - 60:
                break
            h_co = resample_h(best_h, nc)
            c_co = exact_c5(h_co)
            if time.monotonic() > deadline - 40:
                continue
            h_pol, c_pol = polish_lp(h_co, c_co, deadline - 40, max_iter=30, r0=0.02, r_max=0.06)
            h_fine = resample_h(h_pol, n0)
            c_fine = exact_c5(h_fine)
            best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_fine, c_fine, n0)

    # Phase 7: Random restarts with polish
    if time.monotonic() < deadline - 50:
        dim_pool = [n0 + d for d in [-2, -1, 0, 1, 2, 4] if 24 <= n0 + d <= 90]
        if not dim_pool:
            dim_pool = [n0]
        attempts = 0
        while time.monotonic() < deadline - 50 and attempts < 40:
            attempts += 1
            ntry = dim_pool[int(rng.integers(0, len(dim_pool)))]
            if rng.random() < 0.7:
                base = resample_h(best_h, ntry)
            else:
                base = np.full(ntry, 0.5)
            mode = int(rng.integers(4))
            if mode == 0:
                h_r = base + rng.normal(0, 0.03, ntry)
            elif mode == 1:
                x = (np.arange(ntry) + 0.5) * (2.0 / ntry)
                k1 = int(rng.integers(1, 8))
                a1 = float(rng.uniform(0.15, 0.4))
                p1 = float(rng.uniform(0, 2 * math.pi))
                h_r = 0.5 + a1 * np.sin(math.pi * k1 * x + p1)
                h_r = 0.5 * h_r + 0.5 * base
            elif mode == 2:
                h_r = base + rng.normal(0, 0.015, ntry)
                thr = float(np.median(h_r))
                hi = float(rng.uniform(0.55, 0.85))
                lo = float(rng.uniform(0.15, 0.45))
                h_r = np.where(h_r > thr, hi, lo)
            else:
                h_r = base.copy()
                idx = rng.integers(0, ntry, 20)
                h_r[idx] = rng.uniform(0.0, 1.0, 20)
            h_r = project_feasible(h_r, ntry)
            c_r = exact_c5(h_r)
            best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_r, c_r, ntry)
            if c_r < best_c + 0.008 and time.monotonic() < deadline - 50:
                h_pol, c_pol = polish_lse(h_r, c_r, deadline - 50, betas=[2000, 10000],
                                          max_iter_per_beta=8, scale=0.04)
                best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_pol, c_pol, ntry)

    # Phase 8: Final combined polish
    if best_h is not None and time.monotonic() < deadline - 15:
        h_f = best_h.copy()
        c_f = exact_c5(h_f)
        if time.monotonic() < deadline - 15:
            h_f, c_f = polish_lp(h_f, c_f, deadline - 10, max_iter=50, r0=0.008, r_max=0.025)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_f, c_f, len(h_f))
        if time.monotonic() < deadline - 5:
            h_f, c_f = polish_lse(h_f, c_f, deadline - 5, betas=[10000, 30000],
                                  max_iter_per_beta=10, scale=0.02)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_f, c_f, len(h_f))
        h_sym = symmetrize(best_h)
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_sym, exact_c5(h_sym), len(h_sym))
        h_comp = project_feasible(1.0 - best_h, len(best_h))
        best_h, best_c, best_n = compare_update(best_h, best_c, best_n, h_comp, exact_c5(h_comp), len(h_comp))

    # Final verification
    if best_h is None:
        best_h = project_feasible(np.full(64, 0.5), 64)
    h_ret = project_feasible(best_h, len(best_h))
    c_ret = exact_c5(h_ret)
    if not np.isfinite(c_ret):
        h_ret = project_feasible(np.full(64, 0.5), 64)
        c_ret = exact_c5(h_ret)
    return h_ret, float(c_ret), int(h_ret.size)
