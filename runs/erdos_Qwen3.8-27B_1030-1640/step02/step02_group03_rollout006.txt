        return F, g

    def eq_fun(x):
        return float(np.sum(x) - 0.5 * len(x))

    def eq_jac(x):
        return np.ones((1, len(x)))

    try:
        res = optimize.minimize(lambda x: obj_grad(np.clip(x, 0, 1))[0],
                                h.copy(),
                                jac=lambda x: obj_grad(np.clip(x, 0, 1))[1],
                                bounds=[(0.0, 1.0)] * n,
                                constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                                method='SLSQP',
                                options={'maxiter': int(maxiter), 'ftol': 1e-9})
        if res is not None and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return None


def subgrad_descent(h0, n_iter=100, deadline=None, need=20.0):
    h = h0.copy()
    n = len(h)
    best_val = exact_value(h)
    best_h = h.copy()
    alpha = 4.0 / n
    for _ in range(n_iter):
        if deadline is not None and (deadline - time.monotonic()) < need:
            break
        c = overlap_vec(h)
        k_star = int(np.argmax(c))
        g = grad_overlap(h, k_star)
        y = h - alpha * g
        y = project_box_sum(y, n)
        v = exact_value(y)
        if v < best_val - 1e-12:
            best_val = v
            best_h = y.copy()
            h = y
            alpha = min(8.0, alpha * 1.03)
        else:
            alpha *= 0.75
            if alpha < 1e-10:
                break
    return best_h


def lse_pipeline(h0, deadline, need=30.0):
    h = h0.copy()
    n = len(h)
    val = exact_value(h)
    base_tau = max(0.004, 0.15 * val)
    taus = [base_tau, base_tau * 0.5, base_tau * 0.2, base_tau * 0.08, base_tau * 0.03]
    for i, tau in enumerate(taus):
        if deadline - time.monotonic() < need + 5 * (len(taus) - i):
            break
        h2 = lse_slqp(h, tau, maxiter=50)
        if h2 is not None:
            v2 = exact_value(h2)
            if v2 < val - 1e-11:
                h, val = h2, v2
            hs = project_box_sum(0.5 * (h2 + h2[::-1]), n)
            vs = exact_value(hs)
            if vs < val - 1e-11:
                h, val = hs, vs
    return h, val


def epigraph_solve(h0, margin, top_k=14, maxiter=60):
    h = np.asarray(h0, dtype=float).reshape(-1).copy()
    n = len(h)
    L = 2 * n - 1
    c = overlap_vec(h)
    s_curr = float(np.max(c))
    order = np.argsort(c)[::-1][:top_k]

    def epi_obj(x):
        return float(x[-1])

    def epi_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def epi_eq(x):
        return float(np.sum(x[:-1]) - 0.5 * n)

    def epi_eq_j(x):
        J = np.zeros((1, n + 1))
        J[0, :-1] = 1.0
        return J

    def epi_ineq(x):
        h_c = np.clip(x[:-1], 0.0, 1.0)
        cc = overlap_vec(h_c)
        return float(x[-1]) - cc[order]

    def epi_ineq_j(x):
        h_c = np.clip(x[:-1], 0.0, 1.0)
        J = np.zeros((len(order), n + 1))
        for ii, k_idx in enumerate(order):
            J[ii, :-1] = -grad_overlap(h_c, int(k_idx))
            J[ii, -1] = 1.0
        return J

    x0 = np.empty(n + 1)
    x0[:-1] = h
    x0[-1] = s_curr + margin
    try:
        res = optimize.minimize(epi_obj, x0, jac=epi_grad,
                                bounds=[(0.0, 1.0)] * n + [(0.0, s_curr + 0.2)],
                                constraints=[
                                    {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_j},
                                    {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_j},
                                ],
                                method='SLSQP',
                                options={'maxiter': int(maxiter), 'ftol': 1e-11})
        if res is not None and res.x is not None:
            return project_box_sum(res.x[:-1], n)
    except Exception:
        pass
    return None


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = len(h)
    n_new = int(n_new)
    if n <= 0 or n_new <= 0 or n_new == n:
        return project_box_sum(h, n)
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)


def gen_balanced_binary(n, pattern='block'):
    """Generate a balanced binary pattern with n/2 ones."""
    n = int(n)
    if n % 2 != 0:
        n -= 1
    h = np.zeros(n)
    if pattern == 'block':
        # Alternate blocks: 11001100...
        block_len = 2
        for i in range(n):
            phase = (i // block_len) % 2
            h[i] = 1.0 if phase == 0 else 0.0
    elif pattern == 'sturm':
        # Sturmian word with slope 1/2
        for i in range(n):
            h[i] = 1.0 if (2 * i + 1) % 4 < 2 else 0.0
    else:
        # Alternating
        h = np.tile([1.0, 0.0], n // 2)
    # Ensure exactly n/2 ones
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        idx_ones = np.where(h == 1.0)[0]
        idx_zeros = np.where(h == 0.0)[0]
        extra = ones - target
        for k in range(extra):
            h[idx_ones[k]] = 0.0
    elif ones < target:
        idx_zeros = np.where(h == 0.0)[0]
        deficit = target - ones
        for k in range(deficit):
            h[idx_zeros[k]] = 1.0
    return h


def gen_smooth_h(n, freq=1.0, amp=0.3, seed=0):
    """Generate a smooth pattern."""
    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h_raw = np.ones(n) * 0.5
    for m in range(1, 5):
        a = rng.standard_normal() * amp / m
        h_raw += a * np.sin(np.pi * m * x)
        h_raw += a * 0.5 * np.cos(np.pi * m * x)
    return project_box_sum(h_raw, n)


def sym_h(h):
    return project_box_sum(0.5 * (h + h[::-1]), len(h))


def refine_one(h0, deadline, need=40):
    """Full refinement pipeline for one candidate."""
    h = project_box_sum(h0.copy(), len(h0))
    n = len(h)
    val = exact_value(h)

    # Phase 1: subgradient
    if deadline - time.monotonic() > need + 30:
        h = subgrad_descent(h, n_iter=80, deadline=deadline, need=need + 20)
        val = exact_value(h)

    # Phase 2: coordinate descent (main workhorse)
    if deadline - time.monotonic() > need + 20:
        h = coord_desc_exact(h, n_passes=150)
        val = exact_value(h)

    # Phase 3: LSE smoothing
    if deadline - time.monotonic() > need + 15:
        h2, v2 = lse_pipeline(h, deadline, need=need)
        if v2 < val - 1e-11:
            h, val = h2, v2

    # Phase 4: more coordinate descent
    if deadline - time.monotonic() > need + 10:
        h = coord_desc_exact(h, n_passes=200, rng=np.random.default_rng(42))
        val = exact_value(h)

    # Phase 5: epigraph
    if deadline - time.monotonic() > need + 8:
        for m in [0.002, 0.0005, 0.0001]:
            if deadline - time.monotonic() < need + 5:
                break
            he = epigraph_solve(h, m, top_k=14, maxiter=50)
            if he is not None:
                ve = exact_value(he)
                if ve < val - 1e-11:
                    h, val = he, ve

    # Phase 6: final coordinate descent polish
    if deadline - time.monotonic() > need + 3:
        h = coord_desc_exact(h, n_passes=300)
        val = exact_value(h)

    # Phase 7: try symmetry
    hs = sym_h(h)
    vs = exact_value(hs)
    if vs < val - 1e-11:
        h, val = hs, vs
        h = coord_desc_exact(h, n_passes=200)
        val = exact_value(h)

    return h, val


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(int(seed) * 1000 + 7)

    best_h = None
    best_val = math.inf
    best_n = 0

    def consider(h):
        nonlocal best_h, best_val, best_n
        if h is None or len(h) == 0:
            return
        h = project_box_sum(h, len(h))
        v = exact_value(h)
        if not math.isfinite(v):
            return
        if v < best_val - 1e-12 or (abs(v - best_val) <= 1e-12 and len(h) < best_n):
            best_h = h.copy()
            best_val = float(v)
            best_n = len(h)

    # Generate diverse starting points
    candidates = []

    # From initial if available
    init = globals().get('initial_h_values', None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0 and np.any(np.isfinite(hinit)):
            hinit = project_box_sum(hinit, len(hinit))
            candidates.append(hinit)
            candidates.append(sym_h(hinit))

    # Binary patterns
    for n in [44, 46, 48]:
        for pat in ['block', 'sturm']:
            candidates.append(gen_balanced_binary(n, pat))

    # Smooth patterns
    for n in [44, 46, 48]:
        candidates.append(gen_smooth_h(n, seed=int(seed) + n))

    # Half-ones half-zeros for the initial n
    if init is not None:
        n0 = len(init)
        candidates.append(gen_balanced_binary(n0, 'block'))

    # Constant (fallback)
    candidates.append(np.full(45, 0.5))

    # Evaluate all candidates quickly
    for h in candidates:
        if h is not None and len(h) > 0:
            consider(h)

    if best_h is None:
        best_h = np.full(45, 0.5)
        best_val = exact_value(best_h)
        best_n = 45

    # Refine top candidates (by initial value)
    scored = []
    seen = set()
    for h in candidates:
        if h is None or len(h) == 0:
            continue
        key = tuple(np.round(h, 6))
        if key in seen:
            continue
        seen.add(key)
        v = exact_value(h)
        scored.append((v, h))
    scored.sort(key=lambda x: x[0])

    # Refine the best few
    n_refine = 0
    for v, h in scored[:5]:
        if deadline - time.monotonic() < 80:
            break
        # Quick refinement first
        if deadline - time.monotonic() < 60:
            hk = coord_desc_exact(h, n_passes=50)
            consider(hk)
        else:
            hr, vr = refine_one(h, deadline, need=70)
            consider(hr)
        n_refine += 1

    # Resample best solution to nearby n values
    if best_h is not None and deadline - time.monotonic() > 80:
        cur_n = best_n
        for d in [-2, -1, 1, 2, -3, 3, -4, 4, 5, -5]:
            if deadline - time.monotonic() < 70:
                break
            nn = cur_n + d
            if nn < 32 or nn > 64:
                continue
            hr = resample_h(best_h, nn)
            consider(hr)
            if deadline - time.monotonic() > 60:
                hr2 = coord_desc_exact(hr, n_passes=100)
                consider(hr2)

    # Try more aggressive coordinate descent on best
    if best_h is not None and deadline - time.monotonic() > 40:
        hfinal = coord_desc_exact(best_h, n_passes=500, rng=np.random.default_rng(99))
        consider(hfinal)

    # Final epigraph attempt
    if best_h is not None and deadline - time.monotonic() > 25:
        he = epigraph_solve(best_h, 0.0001, top_k=16, maxiter=80)
        if he is not None:
            consider(he)
            if deadline - time.monotonic() > 15:
                he2 = coord_desc_exact(he, n_passes=200)
                consider(he2)

    # Very final polish
    if best_h is not None and deadline - time.monotonic() > 15:
        hpol = coord_desc_exact(best_h, n_passes=500)
        consider(hpol)

    if best_h is None:
        best_h = np.full(45, 0.5)
        best_val = exact_value(best_h)
        best_n = 45

    return best_h, float(best_val), int(best_n)
