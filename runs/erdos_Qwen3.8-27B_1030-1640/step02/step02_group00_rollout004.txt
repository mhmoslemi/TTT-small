    n_old = len(h_old)

    n_old = len(h_old)
    if n_old == n_new:
        return ensure_feasible(h_old.copy())
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return ensure_feasible(y)


def line_search(h0, d, c0, deadline):
    n = len(h0)
    d = d.copy()
    d -= np.mean(d)
    dm = np.max(np.abs(d))
    if dm < 1e-14:
        return None, c0
    d /= dm
    pos = d > 1e-10
    neg = d < -1e-10
    tl, tu = 0.0, 0.0
    if np.any(pos):
        tl = max(tl, float(np.max(-h0[pos] / d[pos])))
        tu = min(tu, float(np.min((1.0 - h0[pos]) / d[pos])))
    if np.any(neg):
        tl = max(tl, float(np.max((1.0 - h0[neg]) / d[neg])))
        tu = min(tu, float(np.min(-h0[neg] / d[neg])))
    if tu - tl < 1e-10:
        return None, c0
    best_t, best_c = 0.0, c0
    ts_coarse = np.array([tl * 0.8, tu * 0.5, -0.15, -0.05, 0.0, 0.05, 0.15, -tu * 0.5, tl * 0.5])
    ts_coarse = np.clip(ts_coarse, tl, tu)
    ts_coarse = np.sort(np.unique(ts_coarse))
    for t in ts_coarse:
        if time.monotonic() > deadline:
            return None, c0
        h_try = h0 + t * d
        ct = exact_c5(h_try)
        if ct < best_c:
            best_c, best_t = ct, t
    if best_t == 0.0:
        return None, c0
    idx = int(np.argmin(np.abs(ts_coarse - best_t)))
    lo_idx = max(0, idx - 1)
    hi_idx = min(len(ts_coarse) - 1, idx + 1)
    lo_t, hi_t = ts_coarse[lo_idx], ts_coarse[hi_idx]
    for _ in range(5):
        if time.monotonic() > deadline:
            break
        ts_r = np.linspace(lo_t, hi_t, 9)
        local_best_t, local_best_c = best_t, best_c
        for t in ts_r:
            h_try = h0 + t * d
            ct = exact_c5(h_try)
            if ct < local_best_c:
                local_best_c, local_best_t = ct, t
        if local_best_c < best_c:
            best_c, best_t = local_best_c, local_best_t
        idx_r = int(np.argmin(np.abs(ts_r - best_t)))
        lo_t = ts_r[max(0, idx_r - 1)]
        hi_t = ts_r[min(len(ts_r) - 1, idx_r + 1)]
    h_new = h0 + best_t * d
    h_new = ensure_feasible(h_new)
    c_new = exact_c5(h_new)
    if c_new < c0 - 5e-11:
        return h_new, c_new
    return None, c0


def active_lag_gradient(h, m):
    n = len(h)
    k = m - n + 1
    g = np.zeros(n)
    if k > 0 and k < n:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    elif k < 0:
        s = -k
        if s < n:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    g -= np.mean(g)
    return g


def generate_directions(h, rng, n_active=8, n_pairs_per_lag=6):
    n = len(h)
    directions = []
    c_vec = corr_values(h)
    active_lags = np.argsort(c_vec)[-n_active:]
    for m in active_lags:
        k = m - n + 1
        # Pair directions
        if k > 0 and k < n:
            n_valid = n - k
            step = max(1, n_valid // n_pairs_per_lag)
            for i in range(0, n_valid, step):
                d = np.zeros(n)
                d[i] = -1.0
                d[i + k] = 1.0
                directions.append(d)
        elif k < 0:
            s = -k
            if s < n:
                n_valid = n - s
                step = max(1, n_valid // n_pairs_per_lag)
                for i in range(s, n, step):
                    d = np.zeros(n)
                    d[i] = -1.0
                    d[i - s] = 1.0
                    directions.append(d)
        # Gradient direction for this lag
        g = active_lag_gradient(h, m)
        if np.max(np.abs(g)) > 1e-10:
            directions.append(g)
    # Binary push
    d_bin = np.sign(h - 0.5)
    d_bin[np.abs(h - 0.5) < 0.02] = 0.0
    d_bin -= np.mean(d_bin)
    if np.max(np.abs(d_bin)) > 1e-8:
        directions.append(d_bin)
    # Smooth sine directions
    x = (np.arange(n) + 0.5) * (2.0 / n)
    for freq in [1, 2, 3, 5]:
        d_s = np.sin(np.pi * freq * x + rng.uniform(0, 2 * math.pi))
        d_s -= np.mean(d_s)
        directions.append(d_s)
    # Random
    for _ in range(3):
        d_r = rng.normal(0, 1, n)
        d_r -= np.mean(d_r)
        directions.append(d_r)
    return directions


def local_optimize(h, c, deadline, max_rounds=50, n_active=8):
    n = len(h)
    rng_local = np.random.default_rng(12345)
    stall = 0
    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if stall > 12:
            break
        dirs = generate_directions(h, rng_local, n_active=n_active, n_pairs_per_lag=6)
        improved = False
        for d in dirs:
            if time.monotonic() > deadline:
                break
            h_new, c_new = line_search(h, d, c, deadline)
            if h_new is not None:
                h, c = h_new, c_new
                improved = True
        if not improved:
            stall += 1
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    seed = int(seed)
    budget = max(10.0, float(budget_s))
    t0 = time.monotonic()
    deadline = t0 + budget - 40.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 10**9

    parent = None
    try:
        ph = globals().get("initial_h_values")
        if ph is not None:
            arr = np.asarray(ph, dtype=float).ravel()
            if arr.size >= 4 and np.all(np.isfinite(arr)):
                parent = ensure_feasible(arr)
    except Exception:
        pass

    if parent is not None:
        cands = [(parent.copy(), exact_c5(parent), len(parent))]
        h_sym = ensure_feasible(0.5 * (parent + parent[::-1]))
        cands.append((h_sym, exact_c5(h_sym), len(h_sym)))
        h_comp = ensure_feasible(1.0 - parent)
        cands.append((h_comp, exact_c5(h_comp), len(h_comp)))
        for s in [-2, -1, 1, 2]:
            h_sh = ensure_feasible(np.roll(parent, s))
            cands.append((h_sh, exact_c5(h_sh), len(h_sh)))
        # Jittered variants
        for _j in range(3):
            h_jit = ensure_feasible(parent + rng.normal(0, 0.003, len(parent)))
            cands.append((h_jit, exact_c5(h_jit), len(parent)))
    else:
        cands = [(ensure_feasible(np.full(64, 0.5)), 0.5, 64)]

    for h, c, n in cands:
        if c < best_c - 1e-10 or (abs(c - best_c) <= 2e-8 and n < best_n):
            best_h, best_c, best_n = h, c, n

    n_base = best_n

    # Phase 1: Intensive local optimization at base dimension
    phase1_deadline = deadline - 250.0
    h_work = best_h.copy()
    c_work = best_c
    h_work, c_work = local_optimize(h_work, c_work, phase1_deadline, max_rounds=200, n_active=8)
    if c_work < best_c - 1e-10:
        best_h, best_c, best_n = h_work.copy(), c_work, n_base

    # Try complement and sym of improved
    h_sym2 = ensure_feasible(0.5 * (h_work + h_work[::-1]))
    c_sym2 = exact_c5(h_sym2)
    if c_sym2 < best_c - 1e-10:
        best_h, best_c, best_n = h_sym2, c_sym2, n_base
    h_comp2 = ensure_feasible(1.0 - h_work)
    c_comp2 = exact_c5(h_comp2)
    if c_comp2 < best_c - 1e-10:
        best_h, best_c, best_n = h_comp2, c_comp2, n_base

    # Phase 2: Dimension sweep with local optimization
    phase2_deadline = deadline - 100.0
    dims = sorted(set([max(24, n_base + d) for d in list(range(-6, 7)) + [8, 12, 16, -8, -12] if 24 <= n_base + d <= 88]))
    for nid in dims:
        if time.monotonic() > phase2_deadline:
            break
        h_rep = resample_vec(best_h, nid)
        c_rep = exact_c5(h_rep)
        if c_rep < best_c + 0.005:
            time_per_dim = max(5.0, (phase2_deadline - time.monotonic()) / max(1, len(dims)))
            h_rep, c_rep = local_optimize(h_rep, c_rep, time.monotonic() + time_per_dim, max_rounds=40, n_active=6)
        if c_rep < best_c - 1e-10 or (abs(c_rep - best_c) <= 3e-8 and nid < best_n):
            best_h, best_c, best_n = h_rep.copy(), c_rep, nid

    # Phase 3: Focused final polish at best dimension
    if time.monotonic() < deadline - 15:
        h_f = best_h.copy()
        c_f = best_c
        n_f = len(h_f)
        # Use more active lags and finer pair sampling
        h_f, c_f = local_optimize(h_f, c_f, deadline - 15, max_rounds=100, n_active=10)
        if c_f < best_c - 1e-10:
            best_h, best_c, best_n = h_f.copy(), c_f, n_f

    # Phase 4: Try to find smaller dimension with same quality
    if best_n > 32 and time.monotonic() < deadline - 10:
        for nid in sorted(set([best_n - 2, best_n - 4, best_n - 6, 48, 44, 40, 36, 32])):
            if nid < 20:
                continue
            if time.monotonic() > deadline - 10:
                break
            h_sm = resample_vec(best_h, nid)
            c_sm = exact_c5(h_sm)
            if c_sm < best_c + 2e-5:
                h_sm, c_sm = local_optimize(h_sm, c_sm, min(deadline - 10, time.monotonic() + 20), max_rounds=30, n_active=6)
            if c_sm < best_c - 1e-10 or (abs(c_sm - best_c) <= 3e-8 and nid < best_n):
                best_h, best_c, best_n = h_sm.copy(), c_sm, nid

    # Final verification
    if best_h is None:
        best_h = ensure_feasible(np.full(64, 0.5))
        best_c = exact_c5(best_h)
        best_n = 64

    h_ret = best_h.copy()
    c_ret = exact_c5(h_ret)
    if not np.isfinite(c_ret) or c_ret > best_c + 1e-5:
        h_ret = ensure_feasible(best_h)
        c_ret = exact_c5(h_ret)
        if not np.isfinite(c_ret):
            h_ret = ensure_feasible(np.full(64, 0.5))
            c_ret = exact_c5(h_ret)
            best_n = 64

    return h_ret, float(c_ret), int(h_ret.size)
