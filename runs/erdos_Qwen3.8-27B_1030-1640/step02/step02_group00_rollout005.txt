```python
def min_convex_pl_1d(a, b, t_lo, t_hi):
    """Minimize max_k(a_k*t + b_k) over [t_lo, t_hi]. a, b are arrays."""
    # The minimum is at an endpoint or at a point where two functions
    # are simultaneously maximum with slopes straddling 0.
    
    # Evaluate at endpoints
    candidates = [t_lo, t_hi]
    
    # For each pair (k, l) with a[k] < a[l], find crossing t
    # t_cross = (b[k] - b[l]) / (a[l] - a[k])
    # If t_cross is in [t_lo, t_hi], add to candidates
    # But there are O(n^2) such crossings, too many
    
    # Better: use the "cutting plane" / separation approach
    # Start with a bracket, shrink it
    
    lo, hi = t_lo, t_hi
    for _ in range(60):
        mid = (lo + hi) / 2
        # Compute which indices are active at mid
        vals = a * mid + b
        active = np.where(vals >= vals.max() - 1e-12)[0]
        min_slope = a[active].min()
        max_slope = a[active].max()
        
        if min_slope > 1e-14:
            # Function is decreasing at mid, so minimum is to the left
            hi = mid
        elif max_slope < -1e-14:
            # Function is increasing at mid, so minimum is to the right
            lo = mid
        else:
            # 0 is in the subgradient, so mid is optimal
            return mid, vals.max()
    
    t_opt = (lo + hi) / 2
    return t_opt, np.max(a * t_opt + b)
```

This is a bisection on the subgradient, which converges in O(log(precision/interval)) iterations. For 60 iterations, we get precision of about 1e-18.

    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def corr_values(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def ensure_feasible(h):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    n = h.size
    if n == 0:
        return h
    target = 0.5 * n
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(1000):
        s = np.sum(h) - target
        if abs(s) < 1e-14:
            break
        if s > 0:
            mask = h > 1e-15
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] -= s / cnt
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] -= s / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def pair_slope(h, i, j, n, dx):
    """
    Compute the slopes of all correlation values when h[i] += t, h[j] -= t.
    slope_k = dx * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
    where h[x] = 0 for x outside [0, n-1].
    """
    def hval(idx):
        if 0 <= idx < n:
            return h[idx]
        return 0.0
    
    L = 2 * n - 1
    c0 = corr_values(h)
    slopes = np.zeros(L)
    for k in range(-n+1, n):
        # c_k corresponds to m = k + n - 1 in the full output
        s = dx * ((1.0 - hval(i+k)) - hval(i-k) - (1.0 - hval(j+k)) + hval(j-k))
        slopes[k + n - 1] = s
    return c0, slopes

def min_convex_pl(c0, slopes, t_lo, t_hi):
    """
    Minimize max_k(c0[k] + slopes[k] * t) for t in [t_lo, t_hi].
    Uses bisection on the subgradient.
    """
    if t_lo >= t_hi:
        t_opt = t_lo
        return t_opt, float(np.max(c0 + slopes * t_opt))
    
    lo, hi = t_lo, t_hi
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = c0 + slopes * mid
        v_max = np.max(vals)
        active = vals >= v_max - 1e-12
        if not active.any():
            active = vals >= v_max - 1e-8
        min_s = np.min(slopes[active])
        max_s = np.max(slopes[active])
        if min_s > 1e-14:
            hi = mid
        elif max_s < -1e-14:
            lo = mid
        else:
            break
    t_opt = (lo + hi) / 2.0
    return t_opt, float(np.max(c0 + slopes * t_opt))

def try_pair_direction(h, i, j, dx, n):
    """
    Try direction: h[i] += t, h[j] -= t.
    Returns (h_new, c_new) if improved, else (None, None).
    """
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return None, None
    
    c0, slopes = pair_slope(h, i, j, n, dx)
    c_base = float(np.max(c0))
    
    t_opt, c_opt = min_convex_pl(c0, slopes, t_lo, t_hi)
    
    if c_opt < c_base - 1e-12:
        h_new = h.copy()
        h_new[i] += t_opt
        h_new[j] -= t_opt
        return h_new, c_opt
    return None, None
