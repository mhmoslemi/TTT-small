import time
import math
import numpy as np
from scipy import optimize

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1).copy()
    n = int(n)
    if y.size != n:
        y = np.full(n, 0.5)
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * float(n)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + diff)
    return x

def exact_value(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = len(h)
    if n <= 0:
        return math.inf
    if not np.all(np.isfinite(h)):
        return math.inf
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, 'full') * dx))

def overlap_vec(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, 'full') * dx

def compute_coeff(h, i):
    n = len(h)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    a = np.zeros(L)
    for k_idx in range(L):
        s = k_idx - (n - 1)
        coeff = 0.0
        if 0 <= i - s < n:
            coeff += dx * (1.0 - h[i - s])
        if 0 <= i + s < n:
            coeff -= dx * h[i + s]
        a[k_idx] = coeff
    return a

def golden_min_pl(a, b, lo=0.0, hi=1.0, n_iter=45):
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_iter):
        t1 = lo + gr * (hi - lo)
        t2 = hi - gr * (hi - lo)
        if np.max(a * t1 + b) < np.max(a * t2 + b):
            hi = t2
        else:
            lo = t1
    return 0.5 * (lo + hi)

def coord_descent_fast(h, n_passes=200, rng=None, deadline=None, need=5.0):
    h = np.array(h, dtype=float)
    n = len(h)
    L = 2 * n - 1
    c = overlap_vec(h)
    best_val = float(np.max(c))
    best_h = h.copy()
    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline - need:
            break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        any_change = False
        for i in order:
            a = compute_coeff(h, i)
            b = c - a * h[i]
            new_val = float(golden_min_pl(a, b))
            if abs(new_val - h[i]) > 1e-13:
                delta = new_val - h[i]
                h[i] = new_val
                c += delta * a
                any_change = True
        if not any_change:
            break
        h = project_box_sum(h, n)
        c = overlap_vec(h)
        v = float(np.max(c))
        if v < best_val - 1e-14:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def pair_moves(h, n_rounds=50, rng=None, deadline=None, need=5.0):
    h = np.array(h, dtype=float)
    n = len(h)
    c = overlap_vec(h)
    best_val = float(np.max(c))
    best_h = h.copy()
    for r in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline - need:
            break
        n_pairs = max(10, n)
        for _ in range(n_pairs):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            a_i = compute_coeff(h, i)
            a_j = compute_coeff(h, j)
            d = a_i - a_j
            if np.max(np.abs(d)) < 1e-15:
                continue
            b = c - d * 0.0
            lo_t = max(-h[i], h[j] - 1.0)
            hi_t = min(1.0 - h[i], h[j])
            if lo_t >= hi_t:
                continue
            t_best = float(golden_min_pl(d, b, lo=lo_t, hi=hi_t, n_iter=40))
            new_i = h[i] + t_best
            new_j = h[j] - t_best
            new_c = c + t_best * d
            v_new = float(np.max(new_c))
            if v_new < best_val - 1e-13:
                h[i] = new_i
                h[j] = new_j
                c = new_c
                best_val = v_new
                best_h = h.copy()
        h = project_box_sum(h, n)
        c = overlap_vec(h)
        v = float(np.max(c))
        if v < best_val:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def subgradient_topk(h, top_k=5, n_iters=500, alpha_init=0.02, deadline=None, need=5.0):
    h = np.array(h, dtype=float)
    n = len(h)
    c = overlap_vec(h)
    best_val = float(np.max(c))
    best_h = h.copy()
    alpha = alpha_init
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline - need:
            break
        mc = float(np.max(c))
        idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(len(idx))
        for ii, k_idx in enumerate(idx):
            weights[ii] = math.exp((c[k_idx] - mc) / 0.01)
        weights /= weights.sum()
        g = np.zeros(n)
        for ii, k_idx in enumerate(idx):
            if weights[ii] < 1e-10:
                continue
            s = k_idx - (n - 1)
            for j in range(n):
                term = 0.0
                if 0 <= j - s < n:
                    term += (1.0 - h[j - s])
                if 0 <= j + s < n:
                    term -= h[j + s]
                g[j] += weights[ii] * (2.0 / n) * term
        y = h - alpha * g
        y = project_box_sum(y, n)
        c = overlap_vec(y)
        v = float(np.max(c))
        if v < best_val - 1e-14:
            best_val = v
            best_h = y.copy()
            h = y
            alpha = min(1.0, alpha * 1.1)
        else:
            alpha *= 0.85
            if alpha < 1e-8:
                break
    return best_h, best_val

def lse_objective(x, tau):
    h = np.clip(x, 0.0, 1.0)
    c = overlap_vec(h)
    mc = float(np.max(c))
    z = (c - mc) / tau
    ez = np.exp(z - z.max())
    return mc + tau * math.log(ez.sum() / len(c))

def lse_objective_grad(x, tau):
    h = np.clip(x, 0.0, 1.0)
    c = overlap_vec(h)
    mc = float(np.max(c))
    z = (c - mc) / tau
    ez = np.exp(z - z.max())
    w = ez / ez.sum()
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k_idx in range(len(c)):
        if w[k_idx] < 1e-12:
            continue
        s = k_idx - (n - 1)
        wj = w[k_idx]
        for j in range(n):
            term = 0.0
            if 0 <= j - s < n:
                term += (1.0 - h[j - s])
            if 0 <= j + s < n:
                term -= h[j + s]
            g[j] += wj * dx * term
    return lse_objective(x, tau), g

def _lse_eq_fun(x):
    return float(np.sum(x) - 0.5 * len(x))

def _lse_eq_jac(x):
    return np.ones((1, len(x)))

def lse_slsqp(h0, tau, maxiter=40, ftol=1e-9):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _lse_eq_fun, 'jac': _lse_eq_jac}]
    try:
        res = optimize.minimize(lse_objective_grad, h0.copy(), jac=lambda x: lse_objective_grad(x, tau)[1],
                                bounds=bounds, constraints=cons, method='SLSQP',
                                options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_box_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(int(seed) * 1009 + 13)

    best_h = None
    best_val = math.inf
    best_n = 0

    def consider(cand_h):
        nonlocal best_h, best_val, best_n
        if cand_h is None or len(cand_h) == 0:
            return
        cand_h = project_box_sum(cand_h, len(cand_h))
        v = exact_value(cand_h)
        if not math.isfinite(v):
            return
        if v < best_val - 1e-13 or (abs(v - best_val) <= 1e-13 and len(cand_h) < best_n):
            best_h = cand_h.copy()
            best_val = float(v)
            best_n = len(cand_h)

    init = globals().get('initial_h_values', None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0 and np.all(np.isfinite(hinit)):
            hinit = project_box_sum(hinit, len(hinit))
            consider(hinit)
            consider(project_box_sum(hinit[::-1].copy(), len(hinit)))
            consider(project_box_sum(1.0 - hinit, len(hinit)))

    if best_h is None:
        consider(make_fallback(46))

    # Phase 1: Intensive refinement on initial
    if time.monotonic() < deadline - 60:
        rh, rv = refine_single(best_h, deadline, need=55)
        consider(rh)

    # Phase 2: Try different n values by resampling
    if best_h is not None and time.monotonic() < deadline - 80:
        cur_n = best_n
        n_candidates = sorted(set([cur_n + d for d in [-3, -2, -1, 1, 2, 3, 5, 7, 10, 15, 20, -5]
                                   if cur_n + d >= 30 and cur_n + d <= 100]))
        for nn in n_candidates:
            if time.monotonic() > deadline - 40:
                break
            hr = resample_h(best_h, nn)
            vr = exact_value(hr)
            consider(hr)
            if vr < best_val + 0.005 and time.monotonic() < deadline - 35:
                rh2, rv2 = refine_single(hr, deadline, need=30)
                consider(rh2)

    # Phase 3: Random perturbations of best
    if best_h is not None and time.monotonic() < deadline - 50:
        for trial in range(8):
            if time.monotonic() > deadline - 30:
                break
            sig = float(rng.choice(np.array([0.02, 0.04, 0.08, 0.12])))
            hp = best_h.copy() + sig * rng.standard_normal(len(best_h))
            hp = project_box_sum(hp, len(best_h))
            consider(hp)
            vh = exact_value(hp)
            if vh < best_val + 0.02 and time.monotonic() < deadline - 25:
                rh3, rv3 = refine_single(hp, deadline, need=20)
                consider(rh3)

    # Phase 4: Structured patterns
    if best_h is not None and time.monotonic() < deadline - 30:
        n = best_n
        # Try periodic-like patterns
        for period in [3, 4, 5, 6, 7]:
            if time.monotonic() > deadline - 15:
                break
            pattern = np.zeros(n)
            for i in range(n):
                pattern[i] = 1.0 if (i % period) < (period // 2) else 0.0
            hp = project_box_sum(pattern, n)
            consider(hp)
            # Blend with best
            for alpha in [0.3, 0.5, 0.7]:
                blended = alpha * best_h + (1 - alpha) * pattern
                consider(project_box_sum(blended, n))

    # Phase 5: Final intensive polish
    if best_h is not None and time.monotonic() < deadline - 15:
        rng2 = np.random.default_rng(999)
        h_polish = best_h.copy()
        # Multi-round coord + pair
        for round_i in range(3):
            if time.monotonic() > deadline - 5:
                break
            h_polish, _ = coord_descent_fast(h_polish, n_passes=200, rng=rng2, deadline=deadline, need=5)
            if time.monotonic() > deadline - 5:
                break
            h_polish, _ = pair_moves(h_polish, n_rounds=50, rng=rng2, deadline=deadline, need=5)
        consider(h_polish)

        # Final LSE with very small tau
        if time.monotonic() < deadline - 3:
            for tau in [0.001, 0.0003, 0.0001]:
                if time.monotonic() > deadline - 2:
                    break
                h2 = lse_slsqp(h_polish, tau, maxiter=60, ftol=1e-11)
                if h2 is not None:
                    consider(h2)

    # Phase 6: One more coord descent sweep on absolute best
    if best_h is not None and time.monotonic() < deadline - 2:
        h_final, _ = coord_descent_fast(best_h, n_passes=100, rng=rng, deadline=deadline, need=1)
        consider(h_final)

    if best_h is None:
        best_h = make_fallback(46)
        best_val = exact_value(best_h)
        best_n = 46

    # Final sanity check
    best_h = project_box_sum(best_h, best_n)
    best_val = exact_value(best_h)
    if not math.isfinite(best_val):
        best_h = make_fallback(46)
        best_val = exact_value(best_h)
        best_n = 46

    return best_h, float(best_val), int(best_n)
