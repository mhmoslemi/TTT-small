def fix_feasible(h, n=None):

    hi = 1.0 + h.max()

        else:

    if n_old == n_new:
        return project(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project(y)

def feas_interval(h, d):
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo, t_hi = -1e10, 1e10
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])) + 1e-12)
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])) - 1e-12)
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])) + 1e-12)
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])) - 1e-12)
    return t_lo, t_hi

def line_search(h0, d, c0, n_coarse=12, n_refine=10, n_rounds=3):
    dmax = float(np.max(np.abs(d)))
    if dmax < 1e-14:
        return None, c0
    d = d / dmax
    t_lo, t_hi = feas_interval(h0, d)
    if t_hi - t_lo < 1e-12:
        return None, c0
    t_lo = min(t_lo, -1e-13)
    t_hi = max(t_hi, 1e-13)
    best_t = 0.0
    best_c = c0
    ts = np.linspace(t_lo, t_hi, n_coarse)
    for t in ts:
        ht = h0 + t * d
        if ht.min() < -1e-8 or ht.max() > 1.0 + 1e-8:
            continue
        ht = project(ht)
        ct = c5_val(ht)
        if ct < best_c - 1e-12:
            best_c = ct
            best_t = t
    for _ in range(n_rounds):
        span = (t_hi - t_lo) / n_coarse
        lo = max(t_lo, best_t - 1.5 * span)
        hi = min(t_hi, best_t + 1.5 * span)
        if hi - lo < 1e-14:
            break
        ts = np.linspace(lo, hi, n_refine)
        for t in ts:
            ht = h0 + t * d
            if ht.min() < -1e-8 or ht.max() > 1.0 + 1e-8:
                continue
            ht = project(ht)
            ct = c5_val(ht)
            if ct < best_c - 1e-12:
                best_c = ct
                best_t = t
    if best_t == 0.0:
        return None, c0
    return project(h0 + best_t * d), best_c

def grad_direction(h, k, n):
    d = np.zeros(n)
    if k >= 0:
        lim = n - 1 - k
        for j in range(lim + 1):
            d[j] -= (1.0 - h[j + k])
        for j in range(k, n):
            d[j] -= h[j - k] * (-1.0)
            d[j] += h[j - k]
    else:
        s = -k
        lim = n - 1 - s
        for j in range(s, n):
            d[j] -= h[j - s]
        for j in range(lim + 1):
            d[j] -= (1.0 - h[j + s])
        for j in range(s, n):
            d[j] += (1.0 - h[j - s])
    d -= np.mean(d)
    dmax = np.max(np.abs(d))
    if dmax < 1e-14:
        return None
    return d / dmax

def make_gdir(h, k, n):
    d = np.zeros(n)
    if k >= 0:
        imax = n - 1 - k
        d[:imax+1] -= (1.0 - h[k:k+imax+1])
        d[k:] += h[:imax+1]
    else:
        s = -k
        imax = n - 1 - s
        d[s:] -= h[:imax+1]
        d[:imax+1] -= (1.0 - h[s:s+imax+1])
    d -= np.mean(d)
    dm = np.max(np.abs(d))
    if dm < 1e-14:
        return None
    return d / dm

def pair_dirs_for_lag(h, k, n, top_pairs=4):
    dirs = []
    if k >= 0:
        imax = n - 1 - k
        if imax < 0:
            return dirs
        contribs = h[:imax+1] * (1.0 - h[k:k+imax+1])
        top_idx = np.argsort(contribs)[-top_pairs:][::-1]
        for i in top_idx:
            d = np.zeros(n)
            d[i] = -1.0
            d[i + k] = 1.0
            dirs.append(d)
    else:
        s = -k
        imax = n - 1 - s
        if imax < 0:
            return dirs
        contribs = h[s:s+imax+1] * (1.0 - h[:imax+1])
        top_idx = np.argsort(contribs)[-top_pairs:][::-1]
        for i in top_idx:
            d = np.zeros(n)
            d[i + s] = -1.0
            d[i] = 1.0
            dirs.append(d)
    return dirs

def optimize_main(h, deadline, top_lags=10, max_pairs=5, max_rounds=500):
    n = len(h)
    c = c5_val(h)
    round_num = 0
    while round_num < max_rounds and time.monotonic() < deadline:
        round_num += 1
        cv = corr_vec(h)
        C = float(np.max(cv))
        top_m = np.argsort(cv)[-top_lags:][::-1]
        improved = False
        for m in top_m:
            if time.monotonic() > deadline:
                break
            k = m - (n - 1)
            dirs = pair_dirs_for_lag(h, k, n, top_pairs=max_pairs)
            gd = make_gdir(h, k, n)
            if gd is not None:
                dirs.append(gd)
            for d in dirs:
                if time.monotonic() > deadline:
                    break
                h_new, c_new = line_search(h, d, C, n_coarse=10, n_refine=8, n_rounds=2)
                if h_new is not None and c_new < C - 5e-10:
                    h = h_new
                    C = c_new
                    improved = True
                    c = c_new
        if not improved:
            break
    return h, c5_val(h)

def structured_dirs(h, rng, n_count=6):
    n = len(h)
    x = np.arange(n, dtype=float)
    dirs = []
    for _ in range(n_count):
        k1 = int(rng.integers(1, 8))
        k2 = int(rng.integers(1, 12))
        p1 = float(rng.uniform(0, 2 * math.pi))
        p2 = float(rng.uniform(0, 2 * math.pi))
        a1 = float(rng.uniform(0.5, 1.0))
        a2 = float(rng.uniform(0.2, 0.6))
        d = a1 * np.sin(2.0 * math.pi * k1 * x / n + p1)
        d += a2 * np.sin(2.0 * math.pi * k2 * x / n + p2)
        d -= np.mean(d)
        dm = np.max(np.abs(d))
        if dm > 1e-10:
            dirs.append(d / dm)
    # Step direction
    half = n // 2
    d = np.zeros(n)
    d[:half] = 1.0
    d[half:] = -1.0
    d -= np.mean(d)
    dm = np.max(np.abs(d))
    if dm > 1e-10:
        dirs.append(d / dm)
    # Reverse gradient
    d = np.sort(h) - h
    d -= np.mean(d)
    dm = np.max(np.abs(d))
    if dm > 1e-10:
        dirs.append(d / dm)
    return dirs

def plane_search2d(h0, d1, d2, c0, n_grid=7, deadline=None):
    n = len(h0)
    d1n = np.max(np.abs(d1))
    d2n = np.max(np.abs(d2))
    if d1n < 1e-14 or d2n < 1e-14:
        return None, c0
    d1 = d1 / d1n
    d2 = d2 / d2n
    # Find feasible bounds for t1 and t2
    # For each component i: 0 <= h0[i] + t1*d1[i] + t2*d2[i] <= 1
    # We'll use a simple approach: vary t1, then for each t1 find valid t2 range
    best_h = None
    best_c = c0
    # Coarse: try t1 in [-1, 1], for each try t2 in [-1, 1]
    t1_vals = np.linspace(-0.5, 0.5, n_grid)
    t2_vals = np.linspace(-0.5, 0.5, n_grid)
    for t1 in t1_vals:
        if deadline is not None and time.monotonic() > deadline:
            break
        h_mid = h0 + t1 * d1
        if h_mid.min() < -0.5 or h_mid.max() > 1.5:
            continue
        # Find t2 range
        rem = h_mid
        pos2 = d2 > 1e-14
        neg2 = d2 < -1e-14
        t2_lo, t2_hi = -0.5, 0.5
        if np.any(pos2):
            t2_lo = max(t2_lo, float(np.max(-rem[pos2] / d2[pos2])) + 1e-12)
            t2_hi = min(t2_hi, float(np.min((1.0 - rem[pos2]) / d2[pos2])) - 1e-12)
        if np.any(neg2):
            t2_lo = max(t2_lo, float(np.max((1.0 - rem[neg2]) / d2[neg2])) + 1e-12)
            t2_hi = min(t2_hi, float(np.min(-rem[neg2] / d2[neg2])) - 1e-12)
        if t2_hi - t2_lo < 1e-10:
            continue
        t2_lo = max(t2_lo, -1.0)
        t2_hi = min(t2_hi, 1.0)
        ts2 = np.linspace(t2_lo, t2_hi, n_grid)
        for t2 in ts2:
            ht = h0 + t1 * d1 + t2 * d2
            if ht.min() < -1e-8 or ht.max() > 1.0 + 1e-8:
                continue
            ht = project(ht)
            ct = c5_val(ht)
            if ct < best_c - 5e-10:
                best_c = ct
                best_h = ht.copy()
    if best_h is not None:
        return project(best_h), best_c
    return None, c0

def get_parent():
    try:
        ph = globals().get("initial_h_values")
    except Exception:
        ph = None
    if ph is None:
        return None
    arr = np.asarray(ph, dtype=float).ravel()
    if arr.size < 4 or not np.all(np.isfinite(arr)):
        return None
    return project(arr)

def try_update(best_h, best_c, best_n, h, c, n):
    if h is None or n <= 0 or not np.isfinite(c):
        return best_h, best_c, best_n
    if best_h is None:
        return h, c, n
    if c < best_c - 5e-10:
        return h, c, n
    if abs(c - best_c) <= 1e-8 and n < best_n:
        return h, c, n
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    seed = int(seed)
    try:
        budget = max(30.0, float(budget_s))
    except Exception:
        budget = 1100.0
    t0 = time.monotonic()
    deadline = t0 + budget - 40.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 10**9

    parent = get_parent()
    if parent is not None:
        cp = c5_val(parent)
        best_h, best_c, best_n = try_update(best_h, best_c, best_n, parent, cp, len(parent))
        # Variants
        h_derivs = [parent, parent[::-1], 1.0 - parent]
        for hd in h_derivs:
            hd = project(hd)
            chd = c5_val(hd)
            best_h, best_c, best_n = try_update(best_h, best_c, best_n, hd, chd, len(hd))
        # Symmetrize
        h_sym = project(0.5 * (parent + parent[::-1]))
        csym = c5_val(h_sym)
        best_h, best_c, best_n = try_update(best_h, best_c, best_n, h_sym, csym, len(h_sym))
    else:
        best_h = project(np.full(64, 0.5))
        best_c = c5_val(best_h)
        best_n = 64

    n_base = len(best_h)

    # Phase 1: Main optimization at base dimension
    if time.monotonic() < deadline - 200:
        h1, c1 = optimize_main(best_h, deadline - 200, top_lags=8, max_pairs=5, max_rounds=1000)
        best_h, best_c, best_n = try_update(best_h, best_c, best_n, h1, c1, len(h1))

    # Phase 2: Second pass with more lags
    if time.monotonic() < deadline - 150:
        h2, c2 = optimize_main(best_h, deadline - 150, top_lags=12, max_pairs=6, max_rounds=500)
        best_h, best_c, best_n = try_update(best_h, best_c, best_n, h2, c2, len(h2))

    # Phase 3: Structured directions
    if time.monotonic() < deadline - 100:
        for sp in range(30):
            if time.monotonic() > deadline - 100:
                break
            dirs = structured_dirs(best_h, rng, n_count=8)
            improved = False
            for d in dirs:
                h_new, c_new = line_search(best_h, d, best_c, n_coarse=12, n_refine=10, n_rounds=3)
                if h_new is not None and c_new < best_c - 5e-10:
                    best_h = h_new
                    best_c = c_new
                    improved = True
            if not improved and sp > 10:
                break

    # Phase 4: 2D plane search
    if time.monotonic() < deadline - 80 and len(best_h) <= 80:
        n = len(best_h)
        cv = corr_vec(best_h)
        top_m = np.argsort(cv)[-2:][::-1]
        dir_pool = []
        for m in top_m:
            k = m - (n - 1)
            gd = make_gdir(best_h, k, n)
            if gd is not None:
                dir_pool.append(gd)
        if len(dir_pool) >= 2:
            h_pl, c_pl = plane_search2d(best_h, dir_pool[0], dir_pool[1], best_c, n_grid=7, deadline=deadline - 80)
            best_h, best_c, best_n = try_update(best_h, best_c, best_n, h_pl, c_pl, len(best_h))
        # Try pair directions in plane
        for m in top_m[:1]:
            k = m - (n - 1)
            pd = pair_dirs_for_lag(best_h, k, n, top_pairs=2)
            for pd_i in pd[:1]:
                if len(dir_pool) > 0:
                    h_pl2, c_pl2 = plane_search2d(best_h, pd_i, dir_pool[0], best_c, n_grid=6, deadline=deadline - 80)
                    best_h, best_c, best_n = try_update(best_h, best_c, best_n, h_pl2, c_pl2, len(best_h))

    # Phase 5: Dimension exploration
    dims_to_try = sorted(set([n_base + d for d in range(-8, 9) if 16 <= n_base + d <= 96]))
    for nid in dims_to_try:
        if time.monotonic() > deadline - 60:
            break
        h_rep = resample(best_h, nid)
        c_rep = c5_val(h_rep)
        best_h, best_c, best_n = try_update(best_h, best_c, best_n, h_rep, c_rep, nid)
        if c_rep < best_c + 0.003 and nid <= 80:
            h_pol, c_pol = optimize_main(h_rep, deadline - 60, top_lags=8, max_pairs=4, max_rounds=200)
            best_h, best_c, best_n = try_update(best_h, best_c, best_n, h_pol, c_pol, len(h_pol))

    # Phase 6: Final tight optimization
    if time.monotonic() < deadline - 10:
        h_fin, c_fin = optimize_main(best_h, deadline - 10, top_lags=12, max_pairs=8, max_rounds=500)
        best_h, best_c, best_n = try_update(best_h, best_c, best_n, h_fin, c_fin, len(h_fin))

    # Phase 7: Final polish with structured + pair
    if time.monotonic() < t0 + budget - 15:
        for _ in range(10):
            if time.monotonic() > t0 + budget - 15:
                break
            h_new, c_new = optimize_main(best_h, t0 + budget - 15, top_lags=15, max_pairs=10, max_rounds=200)
            if c_new < best_c - 5e-10:
                best_h = h_new
                best_c = c_new
            else:
                break

    # Final validation
    if best_h is None:
        best_h = project(np.full(64, 0.5))
    h_ret = project(best_h)
    c_ret = c5_val(h_ret)
    if not np.isfinite(c_ret):
        h_ret = project(np.full(64, 0.5))
        c_ret = c5_val(h_ret)
    return h_ret, float(c_ret), int(h_ret.size)
