def project_box_sum(y, n):
    """Project y onto {h in [0,1]^n : sum(h) = n/2}"""
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    if y.size != n:
        z = np.full(n, 0.5, dtype=float)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = np.array(y, dtype=float, copy=True)
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0.0, 1.0, out=y)
    target = 0.5 * float(n)
    # Bisection for the threshold

    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid

    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

def overlap_components(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return np.empty(0, dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def overlap_grad_for_lag(h, k_idx):
    """Gradient of c[k_idx] with respect to h.
    c[k_idx] = dx * sum_n h[n+k] * (1-h[n]) where k is the actual lag.
    
    In the full correlation output of length 2n-1, index k_idx corresponds to
    shift s = k_idx - (n-1).
    c = dx * sum_{n: valid} h[n+s] * (1-h[n])
    
    ∂c/∂h[j] = dx * [(1-h[j-s]) * [0<=j-s<n] - h[j+s] * [0<=j+s<n]]
    """
    n = int(h.size)
    dx = 2.0 / float(n)
    s = k_idx - (n - 1)
    g = np.zeros(n, dtype=float)
    # First term: dx * (1-h[j-s]) for valid j-s
    if s >= 0:
        # j ranges from s to n-1
        g[s:] += dx * (1.0 - h[:n-s])
    else:
        # j ranges from 0 to n-1+s
        t = -s
        g[:n-t] += dx * (1.0 - h[t:])
    # Second term: -dx * h[j+s] for valid j+s
    if s >= 0:
        # j ranges from 0 to n-1-s
        g[:n-s] -= dx * h[s:]
    else:
        # j ranges from -s to n-1
        t = -s
        g[t:] -= dx * h[:n-t]
    return g

def max_subgradient(h, epsilon=0.0):
    """Compute subgradient of max(c) at h.
    Returns the average gradient over active set."""
    c = overlap_components(h)
    C = float(np.max(c))
    active = np.where(c >= C - epsilon)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    g = np.zeros(h.size, dtype=float)
    for k_idx in active:
        g += overlap_grad_for_lag(h, int(k_idx))
    g /= float(len(active))
    return g, C, active

def lse_value_grad(h, tau):
    """Log-sum-exp value and gradient."""
    c = overlap_components(h)
    C = float(np.max(c))
    z = (c - C) / tau
    z = z - z.max()
    ez = np.exp(z)
    ss = float(np.sum(ez))
    if ss <= 0 or not np.isfinite(ss):
        F = C
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        F = C + tau * (math.log(ss) - math.log(float(c.size)))
        w = ez / ss
    # Weighted gradient
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    for k_idx in range(len(c)):
        if abs(w[k_idx]) > 1e-15:
            g += w[k_idx] * overlap_grad_for_lag(h, k_idx)
    return F, g

def resample_candidate(h, n_new):

def max_subgrad_nesterov(h0, max_iter=300, deadline=None, need=5.0, 
                          use_lse=False, tau=0.01):
    """Nesterov-accelerated projected gradient on the max (or LSE) objective."""
    n = int(h0.size)
    h = clean_feasible(h0, n)
    if n <= 0:
        return h
    
    if use_lse:
        F_y, g_y = lse_value_grad(h, tau)
    else:
        g_y, C, _ = max_subgradient(h, epsilon=0.002)
        F_y = C
    
    x = h.copy()
    y = h.copy()
    t_coeff = 1.0
    alpha = 0.5
    best_h = h.copy()
    best_val = F_y
    
    for it in range(max_iter):
        if deadline is not None and not enough_time(deadline, need):
            break
        # Gradient step
        x_new = project_box_sum(y - alpha * g_y, n)
        
        if use_lse:
            F_x, g_new = lse_value_grad(x_new, tau)
        else:
            g_new, C_x, _ = max_subgradient(x_new, epsilon=0.002)
            F_x = C_x
        
        # Check improvement
        if F_x < F_y - 1e-12:
            # Accept step
            t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_coeff * t_coeff))
            mu = (t_coeff - 1.0) / t_new
            y = x_new + mu * (x_new - x)
            y = project_box_sum(y, n)
            x = x_new
            t_coeff = t_new
            F_y, g_y = (F_x, g_new) if use_lse else (C_x, g_new)
            
            # Track best
            true_val = exact_value(x)
            if true_val < best_val:
                best_val = true_val
                best_h = x.copy()
            
            alpha = min(2.0, alpha * 1.05)
        else:
            alpha *= 0.7
    
    return best_h

def pair_swap_optimize(h0, max_iters=500, deadline=None, need=5.0):
    """Local optimization via pairwise coordinate swaps.
    At each step, pick two indices i,j and optimize h[i],h[j] 
    (keeping h[i]+h[j] fixed) to minimize the max objective."""
    n = int(h0.size)
    h = clean_feasible(h0, n).copy()
    if n <= 0:
        return h
    
    best_val = exact_value(h)
    best_h = h.copy()
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if deadline is not None and not enough_time(deadline, need):
            break
        
        # Pick two random indices
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            j = (i + 1) % n
        
        s = h[i] + h[j]  # fixed sum
        # Optimize h[i] in [max(0, s-1), min(1, s)]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-12:
            continue
        
        # Golden section search on h[i] (h[j] = s - h[i])
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        best_c = hi
        best_f = exact_value(np.where(h == h[i], 0, h))  # placeholder
        # Actually let's just do a small grid
        n_eval = 16
        candidates = np.linspace(lo + 1e-6, hi - 1e-6, n_eval)
        best_ci = h[i]
        best_fi = exact_value(h)
        
        for ci in candidates:
            cj = s - ci
            if ci < 0 or ci > 1 or cj < 0 or cj > 1:
                continue
            h[i] = ci
            h[j] = cj
            f = exact_value(h)
            if f < best_fi:
                best_fi = f
                best_ci = ci
                best_cj = cj
        
        # Restore best
        if best_fi < best_f - 1e-12:
            h[i] = best_ci
            h[j] = best_cj
            best_f = best_fi
            if best_f < best_val:
                best_val = best_f
                best_h = h.copy()
        else:
            h[i] = best_ci  # restore (or original if no improvement)
            h[j] = s - best_ci
    
    return best_h

    n_new = int(n_new)
    if n <= 0 or n_new <= 0 or n_new == n:
        return clean_feasible(h, n) if n > 0 else None
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)


def sym_candidate(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return h
    return project_box_sum(0.5 * (h + h[::-1]), n)


def nesterov_subgrad(h0, max_iter, deadline, need, tau=None):
    n = int(h0.size)
    h = clean_feasible(h0, n)
    if n <= 0:
        return h
    use_lse = tau is not None
    if use_lse:
        F_y, g_y = lse_val_grad(h, tau)
    else:
        g_y, C, _ = max_subgrad(h, 0.003)
        F_y = C
    x = h.copy()
    y = h.copy()
    t_coeff = 1.0
    alpha = 0.3
    best_h = h.copy()
    best_val = exact_value(h)
    stall = 0
    for it in range(max_iter):
        if deadline is not None and not enough_time(deadline, need):
            break
        x_new = project_box_sum(y - alpha * g_y, n)
        if use_lse:
            F_x, g_new = lse_val_grad(x_new, tau)
        else:
            g_new, C_x, _ = max_subgrad(x_new, 0.003)
            F_x = C_x
        true_val = exact_value(x_new)
        if true_val < best_val - 1e-12:
            best_val = true_val
            best_h = x_new.copy()
        if F_x < F_y - 1e-13:
            t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_coeff * t_coeff))
            mu = (t_coeff - 1.0) / t_new
            y = project_box_sum(x_new + mu * (x_new - x), n)
            x = x_new
            t_coeff = t_new
            F_y, g_y = F_x, g_new
            alpha = min(1.5, alpha * 1.03)
            stall = 0
        else:
            alpha *= 0.6
            stall += 1
            if stall > 60:
                break
    return best_h


def pair_swap(h0, max_iters, deadline, need):
    n = int(h0.size)
    h = clean_feasible(h0, n).copy()
    if n <= 0:
        return h
    rng = np.random.default_rng(777)
    best_val = exact_value(h)
    best_h = h.copy()
    for it in range(max_iters):
        if deadline is not None and not enough_time(deadline, need):
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            j = (i + 1) % n
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0) + 1e-8
        hi = min(1.0, s) - 1e-8
        if hi - lo < 1e-7:
            continue
        cand = np.linspace(lo, hi, 12)
        orig_i, orig_j = h[i], h[j]
        bv = exact_value(h)
        bi, bj = orig_i, orig_j
        for ci in cand:
            cj = s - ci
            if ci < 0 or ci > 1 or cj < 0 or cj > 1:
                continue
            h[i] = ci
            h[j] = cj
            v = exact_value(h)
            if v < bv:
                bv, bi, bj = v, ci, cj
        h[i], h[j] = bi, bj
        if bv < best_val - 1e-13:
            best_val = bv
            best_h = h.copy()
    return best_h


def full_local_search(h0, deadline, need=12):
    n = int(h0.size)
    h = clean_feasible(h0, n)
    if n <= 0:
        return h, math.inf
    best = {"h": h.copy(), "val": exact_value(h), "n": n}
    consider_candidate(best, sym_candidate(h))
    work = best["h"].copy()
    # Subgradient Nesterov
    if enough_time(deadline, need + 8):
        hp = nesterov_subgrad(work, 300, deadline, need)
        consider_candidate(best, hp)
        work = best["h"].copy()
    # LSE descent with decreasing tau
    for tau in [0.03, 0.015, 0.008, 0.004]:
        if not enough_time(deadline, need + 5):
            break
        hp = nesterov_subgrad(work, 250, deadline, need, tau=tau)
        consider_candidate(best, hp)
        work = best["h"].copy()
    # Pair swap
    if enough_time(deadline, need + 5):
        hp = pair_swap(work, 400, deadline, need)
        consider_candidate(best, hp)
        work = best["h"].copy()
    # Another subgradient pass
    if enough_time(deadline, need + 5):
        hp = nesterov_subgrad(work, 200, deadline, need)
        consider_candidate(best, hp)
    # Final pair swap
    if enough_time(deadline, need + 3):
        hp = pair_swap(best["h"], 300, deadline, need)
        consider_candidate(best, hp)
    return best["h"], float(best["val"])


def random_profile(n, rng, mode=0):
    n = int(n)
    rmode = int(mode) % 4
    if rmode == 0:
        h = 0.5 + 0.3 * rng.standard_normal(n)
    elif rmode == 1:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif rmode == 2:
        t = np.linspace(0, 2 * math.pi, n, endpoint=False)
        h = 0.5 + 0.25 * np.sin(3 * t) + 0.05 * rng.standard_normal(n)
    else:
        nb = int(rng.integers(3, 8))
        edges = np.sort(rng.choice(np.arange(1, n), nb - 1, replace=False))
        h = np.zeros(n)
        for b in range(nb):
            st = 0 if b == 0 else edges[b - 1]
            en = n if b == nb - 1 else edges[b]
            h[st:en] = rng.uniform(0.1, 0.9)
    return project_box_sum(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(1000 * (int(seed) if seed is not None else 0) + 17)
    best = {"h": None, "val": math.inf, "n": 0}

    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0:
            hinit = clean_feasible(hinit, int(hinit.size))
            consider_candidate(best, hinit)
            consider_candidate(best, sym_candidate(hinit))

    if best["h"] is None:
        h_init = np.full(44, 0.5)
        h_init[::2] = 0.15
        h_init[1::2] = 0.85
        h_init = project_box_sum(h_init, 44)
        consider_candidate(best, h_init)

    if best["h"] is None:
        h_out = np.full(44, 0.5)
        return h_out, exact_value(h_out), 44

    # Phase 1: Aggressive local search on initial
    if enough_time(deadline, 250):
        full_local_search(best, deadline, need=12)

    # Phase 2: Explore nearby n values
    if enough_time(deadline, 200):
        cur_n = int(best["n"])
        for d in [2, -2, 4, -4, 6, 8, -6, 10, -8]:
            if not enough_time(deadline, 100):
                break
            nn = cur_n + d
            if nn < 22 or nn > 90:
                continue
            hr = resample_candidate(best["h"], nn)
            if hr is not None:
                consider_candidate(best, hr)
                rv = exact_value(hr)
                if math.isfinite(rv) and rv < float(best["val"]) + 0.008:
                    if enough_time(deadline, 90):
                        full_local_search(best, deadline, need=10)

    # Phase 3: Random restarts
    if enough_time(deadline, 100):
        for attempt in range(4):
            if not enough_time(deadline, 70):
                break
            n_try = int(best["n"])
            h_rand = random_profile(n_try, rng, attempt)
            consider_candidate(best, h_rand)
            rv = exact_value(h_rand)
            if math.isfinite(rv) and rv < float(best["val"]) + 0.02:
                if enough_time(deadline, 60):
                    full_local_search(best, deadline, need=10)

    # Phase 4: Final intensive polish
    if enough_time(deadline, 50):
        hp = nesterov_subgrad(best["h"], 300, deadline, 20)
        consider_candidate(best, hp)
    if enough_time(deadline, 30):
        hp = pair_swap(best["h"], 600, deadline, 15)
        consider_candidate(best, hp)
    if enough_time(deadline, 20):
        hp = nesterov_subgrad(best["h"], 200, deadline, 10, tau=0.003)
        consider_candidate(best, hp)
    if enough_time(deadline, 15):
        hp = pair_swap(best["h"], 400, deadline, 10)
        consider_candidate(best, hp)

    final_h = best["h"]
    final_h = clean_feasible(final_h, int(final_h.size))
    final_val = exact_value(final_h)
    if not math.isfinite(final_val):
        final_h = np.full(44, 0.5)
        final_val = exact_value(final_h)
    return final_h, float(final_val), int(final_h.size)
