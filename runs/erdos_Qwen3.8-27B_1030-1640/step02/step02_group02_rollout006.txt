```python
    h = x[:n]
    t = x[n]
```

```python
import time
import math
import numpy as np
from scipy import optimize

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1 - x)) if diff > 0 else int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

def exact_value(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return math.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))

def corr_components(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return np.empty(0)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def grad_matrix(h):
    n = h.size
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n - 1), n):
        m = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo >= i_hi:
            continue
        i_idx = np.arange(i_lo, i_hi)
        G[m, i_idx] += dx * (1.0 - h[i_idx + s])
        G[m, i_idx + s] -= dx * h[i_idx]
    return G

def resample(h, n_new):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0 or n_new <= 0:
        return None
    if n_new == n:
        return project_box_sum(h, n)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def barrier_solve(h0, n, mu_schedule=None, deadline=None, need=20):
    """Solve using log-barrier method with decreasing mu."""
    h = project_box_sum(h0, n)
    F = exact_value(h)
    best_h = h.copy()
    best_F = F
    
    if mu_schedule is None:
        mu_schedule = [50, 25, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    for mu in mu_schedule:
        if deadline is not None and (deadline - time.monotonic()) < need:
            break
        
        # Interior point init
        h_int = np.clip(h, 0.02, 0.98)
        h_int = project_box_sum(h_int, n)
        c = corr_components(h_int)
        t_init = float(np.max(c)) + 0.1
        
        def obj_and_grad(x):
            h_x = x[:n]
            t_x = x[n]
            margin = t_x - corr_components(h_x)
            if np.any(margin <= 0) or np.any(h_x <= 0) or np.any(h_x >= 1):
                return 1e10, np.zeros(n + 1)
            barrier = -np.sum(np.log(margin)) - np.sum(np.log(h_x)) - np.sum(np.log(1 - h_x))
            obj = t_x - barrier / mu
            
            G = grad_matrix(h_x)
            grad_h = (G.T @ (1.0 / margin)) / mu + (-1.0/h_x + 1.0/(1.0-h_x)) / mu
            grad_t = 1.0 - np.sum(1.0 / margin) / mu
            grad = np.concatenate([grad_h, [grad_t]])
            return obj, grad
        
        # Use SLSQP
        x0 = np.concatenate([h_int, [t_init]])
        bounds = [(1e-8, 1-1e-8)] * n + [(float(np.max(corr_components(h_int))) - 1e-8, 10.0)]
        cons = [{'type': 'eq', 'fun': lambda x: sum(x[:n]) - 0.5*n}]
        
        try:
            res = optimize.minimize(obj_and_grad, x0, jac=True, method='SLSQP',
                                    bounds=bounds, constraints=cons,
                                    options={'maxiter': 200, 'ftol': 1e-12})
            if res.success or res.fun < 1e10:
                h_new = project_box_sum(res.x[:n], n)
                F_new = exact_value(h_new)
                if F_new < best_F:
                    best_F = F_new
                    best_h = h_new.copy()
                    h = h_new
        except Exception:
            pass
    
    return best_h, best_F
```

import time
import math
import numpy as np
from scipy import optimize

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1 - x)) if diff > 0 else int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

def exact_value(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return math.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))

def corr_components(h):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0:
        return np.empty(0)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def grad_matrix(h):
    n = h.size
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n - 1), n):
        m = s + n - 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo >= i_hi:
            continue
        i_idx = np.arange(i_lo, i_hi)
        G[m, i_idx] += dx * (1.0 - h[i_idx + s])
        G[m, i_idx + s] -= dx * h[i_idx]
    return G

def resample(h, n_new):
    h = np.asarray(h, float).ravel()
    n = h.size
    if n == 0 or n_new <= 0:
        return None
    if n_new == n:
        return project_box_sum(h, n)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def lse_objective_val(h, T):
    c = corr_components(h)
    if c.size == 0:
        return 0.0
    c_max = np.max(c)
    if T < 1e-10:
        return float(c_max)
    return float(c_max + np.log(np.sum(np.exp(T * (c - c_max)))) / T)

def lse_grad(h, T):
    n = h.size
    c = corr_components(h)
    c_max = np.max(c)
    if T < 1e-10:
        m = int(np.argmax(c))
        return grad_matrix(h)[m]
    w = np.exp(T * (c - c_max))
    w /= w.sum()
    G = grad_matrix(h)
    return w @ G

def lse_wrap_obj(x, T, n, penalty):
    h_x = np.clip(x, 1e-15, 1.0 - 1e-15)
    val = lse_objective_val(h_x, T)
    val += penalty * (np.sum(h_x) - 0.5 * n) ** 2
    return val

def lse_wrap_grad(x, T, n, penalty):
    h_x = np.clip(x, 1e-15, 1.0 - 1e-15)
    g = lse_grad(h_x, T)
    g += 2.0 * penalty * (np.sum(h_x) - 0.5 * n)
    return g

def lse_optimize(h0, n, T_schedule, deadline, need=20):

    best_h = h.copy()
    best_F = exact_value(h)
    penalty = 1e6

            break
        try:
            res = optimize.minimize(lse_wrap_obj, h, args=(T, n, penalty),
                                    jac=lse_wrap_grad, method='L-BFGS-B',
                                    bounds=[(0.0, 1.0)] * n,
                                    options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_box_sum(res.x, n)
            F_new = exact_value(h_new)
            if F_new < best_F:
                best_F = F_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    
    return best_h, best_F
