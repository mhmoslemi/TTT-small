def project_box_sum(y, n):
    """Project y onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    y = np.asarray(y, dtype=float).reshape(-1)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.astype(float).copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    
    # Bisection on threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    
    # Fix any remaining discrepancy
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)

    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).ravel()
    if n <= 0:
        return np.empty(0)
    if y.size != n:
        z = np.full(n, 0.5)
        m = min(y.size, n)
        if m > 0:
            z[:m] = y[:m]
        y = z
    y = y.copy()
    y[~np.isfinite(y)] = 0.5
    np.clip(y, 0, 1, out=y)
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            avail = 1.0 - x
            tot = avail.sum()
            if tot > 0:
                x += diff * avail / tot
        else:
            tot = x.sum()
            if tot > 0:
                x += diff * x / tot
        np.clip(x, 0, 1, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1 - x)) if diff > 0 else int(np.argmax(x))
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

            G[m, s:] -= dx * h[:n-s]

            G[m, :n-sp] -= dx * h[sp:]
