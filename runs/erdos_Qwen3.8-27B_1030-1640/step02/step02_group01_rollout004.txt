        z = z - z.max()
        w = np.exp(z)

            avail = 1.0 - x
            tot = float(avail.sum())
            if tot < 1e-15:
                break
            x += diff * avail / tot
        else:
            tot = float(x.sum())
            if tot < 1e-15:
                break
            x += diff * x / tot
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - x)) if diff > 0 else int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))
    return x

def exact_value(h):
    n = h.size
    if n <= 0:
        return math.inf
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def get_components(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_component(h, m):
    n = h.size
    dx = 2.0 / float(n)
    lag = m - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] += -h[:n-lag]
    else:
        t = -lag
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] += -h[t:]
    return dx * g

def weighted_subgradient(h, tau):
    n = h.size
    L = 2 * n - 1
    c = get_components(h)
    C = float(np.max(c))
    z = (c - C) / tau
    z = z - z.max()
    ez = np.exp(z)
    w = ez / ez.sum()
    g = np.zeros(n)
    for m in range(L):
        if w[m] > 1e-8:
            g += w[m] * grad_component(h, m)
    return C, g, c

def nesterov_smooth(h0, tau0, tau_end, n_iter=600, deadline=None, need=5.0):
    n = h0.size
    x = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_C = exact_value(h0)
    C_y, g_y, _ = weighted_subgradient(y, tau0)
    lr = 2.0 / (1.0 + float(np.linalg.norm(g_y)))
    for it in range(n_iter):
        if deadline is not None and (time.monotonic() - float(deadline)) > -need:
            break
        tau = max(tau_end, tau0 * (tau_end / tau0) ** (it / max(1, n_iter)))
        x_new = project_box_sum(y - lr * g_y, n)
        F_x, _, _ = weighted_subgradient(x_new, tau)
        if not np.isfinite(F_x):
            break
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k))
        mu = (t_k - 1.0) / t_new
        y = project_box_sum(x_new + mu * (x_new - x), n)
        C_y, g_y, _ = weighted_subgradient(y, tau)
        gn = float(np.linalg.norm(g_y)) + 1e-10
        lr = min(3.0, lr * 1.01, 2.0 / gn)
        x = x_new
        t_k = t_new
        C_true = exact_value(x)
        if C_true < best_C:
            best_C = C_true
            best_h = x.copy()
    return best_h, best_C

def lse_objective(x, tau):
    h = np.clip(x, 0.0, 1.0)
    c = get_components(h)
    maxc = float(np.max(c))
    z = (c - maxc) / tau
    z = z - z.max()
    ez = np.exp(z)
    ss = float(ez.sum())
    if ss <= 0 or not np.isfinite(ss):
        return maxc
    return maxc + tau * (math.log(ss) - math.log(c.size))

def lse_grad(x, tau):
    h = np.clip(x, 0.0, 1.0)
    n = h.size
    L = 2 * n - 1
    c = get_components(h)
    maxc = float(np.max(c))
    z = (c - maxc) / tau
    z = z - z.max()
    ez = np.exp(z)
    ss = float(ez.sum())
    if ss <= 0 or not np.isfinite(ss):
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ez / ss
    g = np.zeros(n)
    for m in range(L):
        if w[m] > 1e-10:
            g += w[m] * grad_component(h, m)
    return g

def slqp_smooth(h0, tau, maxiter=60, deadline=None, need=5.0):
    n = h0.size
    if n <= 0:
        return h0

    def obj_fn(x):
        return lse_objective(x, tau)

    def grad_fn(x):
        return lse_grad(x, tau)

    def eq_fun(x):
        return float(np.sum(x) - 0.5 * float(x.size))

    def eq_jac(x):
        return np.ones((1, x.size))

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
    try:
        res = optimize.minimize(
            obj_fn, h0.copy(), jac=grad_fn,
            bounds=bounds, constraints=cons, method="SLSQP",
            options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            return clean_feasible(np.asarray(res.x, dtype=float).reshape(-1), n)
    except Exception:
        pass
    return h0

def clean_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    if h.size != n:
        h = np.resize(h, n)
    h[~np.isfinite(h)] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    if abs(0.5 * float(n) - float(np.sum(h))) > 1e-10:
        h = project_box_sum(h, n)
    return h

def epigraph_obj(x, reg=0.0, ref=None):
    val = float(x[-1])
    if reg > 0.0 and ref is not None:
        d = x[:-1] - ref
        val += reg * float(np.dot(d, d))
    return val

def epigraph_grad(x, reg=0.0, ref=None):
    g = np.zeros(x.size)
    g[-1] = 1.0
    if reg > 0.0 and ref is not None:
        g[:-1] = 2.0 * reg * (x[:-1] - ref)
    return g

def active_ineq_fun(x, active_idx, n):
    h = np.clip(x[:-1], 0.0, 1.0)
    c = get_components(h)
    return float(x[-1]) - c[active_idx]

def active_ineq_jac(x, active_idx, n):
    h = np.clip(x[:-1], 0.0, 1.0)
    L = len(active_idx)
    J = np.zeros((L, x.size))
    for ii, m in enumerate(active_idx):
        J[ii, :-1] = -grad_component(h, m)
        J[ii, -1] = 1.0
    return J

def epigraph_slqp(h0, s_start, s_ub, active_idx, maxiter=50, reg=1e-8, deadline=None, need=5.0):
    n = h0.size
    cmax = exact_value(h0)
    s_start = float(max(s_start, cmax + 1e-10))
    s_ub = float(max(s_ub, s_start + 1e-6, 0.9))
    x0 = np.empty(n + 1)
    x0[:-1] = h0
    x0[-1] = s_start
    bounds = [(0.0, 1.0)] * n + [(0.0, s_ub)]

    def eq_fun(x):
        return float(np.sum(x[:-1]) - 0.5 * float(x.size - 1))

    def eq_jac(x):
        J = np.zeros((1, x.size))
        J[0, :-1] = 1.0
        return J

    cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
    if active_idx is not None and len(active_idx) > 0:
        ai = list(active_idx)
        cons.append({"type": "ineq", "fun": lambda x: active_ineq_fun(x, ai, n),
                     "jac": lambda x: active_ineq_jac(x, ai, n)})
    try:
        res = optimize.minimize(
            epigraph_obj, x0, jac=epigraph_grad,
            bounds=bounds, constraints=cons, method="SLSQP",
            options={"maxiter": maxiter, "ftol": 1e-11, "disp": 0}
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            xr = np.asarray(res.x, dtype=float).reshape(-1)
            if xr.size == n + 1:
                return clean_feasible(xr[:-1], n)
    except Exception:
        pass
    return h0

def paired_cd_step(h, i, j, n_vals=60):
    n = h.size
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-12:
        return h, False
    old_i, old_j = h[i], h[j]
    best_t = old_i
    best_val = exact_value(h)
    tvals = np.linspace(t_lo, t_hi, n_vals)
    for t in tvals:
        h[i] = t
        h[j] = s - t
        val = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)
        if val < best_val:
            best_val = val
            best_t = t
    h[i] = best_t
    h[j] = s - best_t
    improved = best_t != old_i
    return h, improved

def paired_cd(h, n_rounds=200, rng=None, deadline=None, need=5.0):
    n = h.size
    if rng is None:
        rng = np.random.default_rng()
    h = h.copy()
    best_C = exact_value(h)
    best_h = h.copy()
    for r in range(n_rounds):
        if deadline is not None and (time.monotonic() - float(deadline)) > -need:
            break
        improved_any = False
        for _ in range(2 * n):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                j = (j + 1) % n
            h, imp = paired_cd_step(h, i, j, n_vals=40)
            if imp:
                improved_any = True
        C = exact_value(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
        if not improved_any:
            break
    return best_h, best_C

def resample(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n_new == n:
        return clean_feasible(h, n)
    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def full_local_optimize(h, deadline, need=8.0):
    n = h.size
    best_h = h.copy()
    best_C = exact_value(h)

    def try_update(hc):
        nonlocal best_h, best_C
        if hc is None:
            return
        hc = clean_feasible(hc, n)
        C = exact_value(hc)
        if C < best_C:
            best_C = C
            best_h = hc.copy()

    try_update(h)

    taus = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for tau in taus:
        if deadline is not None and (time.monotonic() - float(deadline)) < need + 5.0:
            break
        hn, _ = nesterov_smooth(best_h, tau, max(0.0003, tau * 0.2), n_iter=300,
                                deadline=deadline, need=need)
        try_update(hn)
        if deadline is not None and (time.monotonic() - float(deadline)) < need + 5.0:
            break
        hs = slqp_smooth(best_h, tau, maxiter=50, deadline=deadline, need=need)
        try_update(hs)

    c = get_components(best_h)
    C = float(np.max(c))
    thresh = C - 0.005
    active = np.where(c > thresh)[0].tolist()
    if len(active) == 0:
        active = [int(np.argmax(c))]

    for margin in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
        if deadline is not None and (time.monotonic() - float(deadline)) < need + 5.0:
            break
        he = epigraph_slqp(best_h, best_C + margin, best_C + 0.05,
                           active, maxiter=40, reg=1e-9, deadline=deadline, need=need)
        try_update(he)
        c = get_components(best_h)
        thresh = float(np.max(c)) - 0.003
        active = np.where(c > thresh)[0].tolist()
        if len(active) == 0:
            active = [int(np.argmax(c))]

    if deadline is not None and (time.monotonic() - float(deadline)) > need + 10.0:
        hp, _ = paired_cd(best_h, n_rounds=100, deadline=deadline, need=need)
        try_update(hp)

    return best_h, best_C

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(1000 * (int(seed) if seed is not None else 0) + 17)

    best_h = None
    best_C = math.inf
    best_n = 0

    def consider(hc):
        nonlocal best_h, best_C, best_n
        if hc is None or hc.size <= 0:
            return
        n = hc.size
        hc = clean_feasible(hc, n)
        C = exact_value(hc)
        if not np.isfinite(C):
            return
        if C < best_C - 1e-12 or (abs(C - best_C) <= 1e-12 and n < best_n):
            best_C = C
            best_h = hc.copy()
            best_n = n

    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0 and np.any(np.isfinite(hinit)):
            hinit = clean_feasible(hinit, hinit.size)
            consider(hinit)

    if best_h is None:
        n0 = 44
        h0 = np.full(n0, 0.5)
        h0[::2] += 0.1
        h0[1::2] -= 0.1
        h0 = project_box_sum(h0, n0)
        consider(h0)

    if best_h is None:
        h_fb = np.full(44, 0.5)
        consider(h_fb)
        return h_fb, exact_value(h_fb), 44

    n_cur = best_n
    consider(best_h)

    if (deadline - time.monotonic()) > 120.0:
        hopt, _ = full_local_optimize(best_h, deadline, need=10.0)
        consider(hopt)

    if best_h is not None and (deadline - time.monotonic()) > 80.0:
        deltas = [1, -1, 2, -2, 3, -3, 4, -4, 6, -6, 8, -8]
        tried = 0
        for d in deltas:
            if tried >= 6 or (deadline - time.monotonic()) < 60.0:
                break
            nn = best_n + d
            if nn < 24 or nn > 100:
                continue
            hr = resample(best_h, nn)
            if hr is None:
                continue
            consider(hr)
            tried += 1
            if np.isfinite(exact_value(hr)) and exact_value(hr) < best_C + 0.02:
                if (deadline - time.monotonic()) > 50.0:
                    hopt, _ = full_local_optimize(hr, deadline, need=8.0)
                    consider(hopt)

    if best_h is not None and (deadline - time.monotonic()) > 60.0:
        for attempt in range(4):
            if (deadline - time.monotonic()) < 40.0:
                break
            nbase = best_n
            base = best_h
            sig = float(rng.choice(np.array([0.005, 0.01, 0.02, 0.04], dtype=float)))
            hs = project_box_sum(base + sig * rng.standard_normal(nbase), nbase)
            consider(hs)
            if exact_value(hs) < best_C + 0.015:
                if (deadline - time.monotonic()) > 40.0:
                    hopt, _ = full_local_optimize(hs, deadline, need=8.0)
                    consider(hopt)

    if best_h is not None and (deadline - time.monotonic()) > 30.0:
        c = get_components(best_h)
        C = float(np.max(c))
        thresh = C - 0.003
        active = np.where(c > thresh)[0].tolist()
        if len(active) == 0:
            active = [int(np.argmax(c))]
        for margin in [0.0005, 0.0002, 0.0001, 0.00005]:
            if (deadline - time.monotonic()) < 10.0:
                break
            he = epigraph_slqp(best_h, best_C + margin, best_C + 0.03,
                               active, maxiter=50, reg=5e-10,
                               deadline=deadline, need=8.0)
            consider(he)
            if he is not None and exact_value(he) < best_C:
                c = get_components(best_h)
                thresh = float(np.max(c)) - 0.002
                active = np.where(c > thresh)[0].tolist()
                if len(active) == 0:
                    active = [int(np.argmax(c))]

    if best_h is not None and (deadline - time.monotonic()) > 15.0:
        hp, _ = paired_cd(best_h, n_rounds=150, rng=rng, deadline=deadline, need=5.0)
        consider(hp)

    final_h = best_h if best_h is not None else np.full(44, 0.5)
    final_h = clean_feasible(final_h, final_h.size)
    final_val = exact_value(final_h)
    if not np.isfinite(final_val):
        final_h = np.full(44, 0.5)
        final_val = exact_value(final_h)
    return final_h, float(final_val), int(final_h.size)
