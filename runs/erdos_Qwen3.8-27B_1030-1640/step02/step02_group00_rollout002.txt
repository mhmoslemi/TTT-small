    t_hi += 1e-14

    """Score ordered pairs (a,b) for direction e_b - e_a"""
    n = len(h)
    C = float(np.max(c_vals))

    w = np.exp(beta * (c_vals - C))  # w[0]=1 for max row, others < 1
    # For simplicity, use top rows
    top_rows = np.argsort(c_vals)[-top_k:]
    row_w = {int(m): float(w[m]) for m in top_rows}
    
    scores = np.zeros((n, n))
    for m in top_rows:
        k = int(m) - n + 1
        wm = row_w[int(m)]
        if k >= 0:
            # c[m] = sum_i h[i]*(1-h[i+k]) * dx for i=0..n-1-k
            for i in range(n - k):
                # Direction: decrease h[i], increase h[i+k] to reduce h[i]*(1-h[i+k])
                # d = e_{i+k} - e_i
                scores[i+k, i] += wm * h[i] * (1.0 - h[i+k])

                # d = e_i - e_{i+k}

    return scores

def corr_values(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return np.zeros(0)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    cv = corr_values(h)
    if cv.size == 0 or not np.all(np.isfinite(cv)):
        return float("inf")
    return float(np.max(cv))


def ensure_feasible(h):

def resample_h(h_old, n_new):
    h = np.asarray(h_old, dtype=float).reshape(-1)
    n_old = h.size
    n_new = int(max(4, n_new))
    if n_old == 0:
        return ensure_feasible(np.full(n_new, 0.5))
    if n_old == n_new:
        return ensure_feasible(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return ensure_feasible(y)


def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for m in range(L):
        lag = m - n + 1
        if lag >= 0:
            # c = dx * sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # d/dh[j]: if lag<=j<n: +dx*(1-h[j-lag]); if 0<=j<=n-1-lag: -dx*h[j+lag]
            G[m, lag:n] += dx * (1.0 - h[0:n - lag])
            G[m, 0:n - lag] -= dx * h[lag:n]
        else:
            s = -lag
            # c = dx * sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[j]: if 0<=j<=n-1-s: +dx*(1-h[j+s]); if s<=j<n: -dx*h[j-s]
            G[m, 0:n - s] += dx * (1.0 - h[s:n])
            G[m, s:n] -= dx * h[0:n - s]
    return G


def lp_direction(h0, radius=0.05, time_limit=15.0):
    n = len(h0)
    if n < 4:
        return None
    L = 2 * n - 1
    S = 0.5 * n
    c0 = corr_values(h0)
    G = build_grad_matrix(h0)
    b_val = G @ h0 - c0
    margin = 1e-10
    A_ub = np.hstack([G, -np.ones((L, 1))])
    b_ub = b_val + margin
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    lb = np.minimum(lb, ub)
    bounds = list(zip(lb, ub)) + [(0.0, 1.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None


def feasible_interval(h0, d):
    t_lo = 0.0
    t_hi = float("inf")
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h0[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(-h0[neg] / d[neg])))
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h0[pos] / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h0[neg]) / d[neg])))
    if not np.isfinite(t_hi):
        t_hi = 1.0
    t_lo -= 1e-13
    t_hi += 1e-13
    if t_hi <= t_lo:
        t_lo = -1e-13
        t_hi = 1e-13
    return t_lo, t_hi


def line_search_exact(h0, d, base_c, t_lo, t_hi, n_grid=18, n_refine=5):

        return h0.copy(), base_c
    best_c = base_c
    best_t = 0.0
    best_h = h0.copy()
    ts = np.linspace(t_lo, t_hi, n_grid)

        h_try = h0 + t * d
        c_try = exact_c5(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_t = t
            best_h = h_try.copy()
    if best_t == 0.0:
        return best_h, best_c
    idx_best = int(np.argmin([exact_c5(h0 + t * d) for t in ts]))
    if idx_best == 0:
        lo, hi = t_lo, ts[min(1, n_grid - 1)]
    elif idx_best == n_grid - 1:
        lo, hi = ts[max(0, n_grid - 2)], t_hi
    else:

                best_t = t

            lo = ts_r[0]
            hi = ts_r[1]

            hi = ts_r[10]

        c_try = exact_c5(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_h = h_try.copy()
    return best_h, best_c


def symmetrize_v(h):

def try_update(best, h, c, n):

    bh, bc, bn = best
    if bh is None:

    if c < bc - 1e-10:
        return (h, c, n)

        return (h, c, n)
    return best


def lp_1d_polish(h_init, c_init, deadline, max_iter=100, r_init=0.04, r_max=0.07, r_min=0.003):

    r = r_init
    fails = 0

            r = max(r_min, r * 0.6)

            continue
        delta = h_lp - h
        dmax = float(np.max(np.abs(delta)))
        if dmax < 1e-14:

            r = max(r_min, r * 0.6)
            continue

        d -= np.mean(d)
        t_lo, t_hi = feasible_interval(h, d)
        h_new, c_new = line_search_exact(h, d, c, t_lo, t_hi, n_grid=18, n_refine=5)
        if c_new < c - 1e-11:
            h = h_new
            c = c_new
            fails = 0
            if r < r_max:

        else:
            fails += 1
            r = max(r_min, r * 0.6)
            if fails > 12 or r <= r_min * 1.5:
                break
    return h, c


def pair_search_loop(h_init, c_init, deadline, max_rounds=300):
    h = np.asarray(h_init, dtype=float).copy()
    c = c_init
    n = len(h)
    stall = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c_vals = corr_values(h)
        C = float(np.max(c_vals))
        top_rows = np.argsort(c_vals)[-10:]
        scores = np.zeros((n, n))
        for m in top_rows:
            lag = int(m) - n + 1
            weight = math.exp(1500.0 * (c_vals[m] - C))
            if lag >= 0:
                for i in range(n - lag):
                    scores[i + lag, i] += weight * h[i + lag] * (1.0 - h[i])
            else:
                s = -lag
                for i in range(n - s):
                    scores[i, i + s] += weight * h[i] * (1.0 - h[i + s])
        top_pairs = np.argsort(scores.ravel())[-30:]
        improved = False
        for fi in top_pairs[::-1]:
            b = int(fi // n)
            a = int(fi % n)
            if a == b:
                continue
            if scores[b, a] < 1e-12:
                continue
            d = np.zeros(n)
            d[b] = 1.0
            d[a] = -1.0
            t_lo, t_hi = feasible_interval(h, d)
            h_new, c_new = line_search_exact(h, d, c, t_lo, t_hi, n_grid=12, n_refine=3)
            if c_new < c - 1e-11:
                h = h_new
                c = c_new
                improved = True
                best_update = (h.copy(), c, n)
                break
        if not improved:
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0
    return h, c


def structured_directions(h, rng, count=20):
    n = len(h)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    dirs = []
    for k in range(1, 8):
        for p in [0.0, math.pi / 3, 2.0 * math.pi / 3, math.pi]:
            d = np.sin(math.pi * k * x + p)
            d -= np.mean(d)
            dm = float(np.max(np.abs(d)))
            if dm > 1e-10:
                dirs.append(d / dm)
    for _ in range(count):
        d = rng.normal(0, 1, n)
        d -= np.mean(d)
        dm = float(np.max(np.abs(d)))
        if dm > 1e-10:
            dirs.append(d / dm)
    med = float(np.median(h))
    d_med = np.where(h > med, 1.0, np.where(h < med, -1.0, 0.0))
    d_med -= np.mean(d_med)
    dm = float(np.max(np.abs(d_med)))
    if dm > 1e-10:
        dirs.append(d_med / dm)
    return dirs

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 10 ** 9

    parent = None
    try:
        ph = globals().get("initial_h_values")
    except Exception:
        ph = None
    if ph is not None:
        arr = np.asarray(ph, dtype=float).reshape(-1)
        if arr.size >= 4 and np.all(np.isfinite(arr)):
            parent = ensure_feasible(arr)

    if parent is not None:
        c_p = exact_c5(parent)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), parent, c_p, len(parent))
        h_sym = symmetrize_v(parent)
        c_sym = exact_c5(h_sym)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_sym, c_sym, len(h_sym))
        h_neg = complement_v(parent)
        c_neg = exact_c5(h_neg)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_neg, c_neg, len(h_neg))
    else:
        best_h = ensure_feasible(np.full(64, 0.5))
        best_c = exact_c5(best_h)
        best_n = 64

    n_base = best_n

    # Phase 1: Main LP+1D polish at base dimension
    h_cur = best_h.copy()
    c_cur = best_c
    h_cur, c_cur = lp_1d_polish(h_cur, c_cur, deadline - 300, max_iter=150, r_init=0.04, r_max=0.07)
    best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_cur, c_cur, len(h_cur))

    # Symmetrize and complement check
    h_s = symmetrize_v(h_cur)
    c_s = exact_c5(h_s)
    best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_s, c_s, len(h_s))
    h_n = complement_v(h_cur)
    c_n = exact_c5(h_n)
    best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_n, c_n, len(h_n))

    # Phase 2: Pair direction search
    if time.monotonic() < deadline - 200:
        h_pair, c_pair = pair_search_loop(best_h, best_c, deadline - 200, max_rounds=200)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_pair, c_pair, len(h_pair))
        h_ps = symmetrize_v(h_pair)
        c_ps = exact_c5(h_ps)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_ps, c_ps, len(h_ps))

    # Phase 3: Structured global directions
    if time.monotonic() < deadline - 150:
        h_str = best_h.copy()
        c_str = best_c
        dirs = structured_directions(h_str, rng, count=16)
        for d in dirs:
            if time.monotonic() > deadline - 150:
                break
            t_lo, t_hi = feasible_interval(h_str, d)
            if t_hi - t_lo < 1e-14:
                continue
            h_new, c_new = line_search_exact(h_str, d, c_str, t_lo, t_hi, n_grid=16, n_refine=4)
            if c_new < c_str - 1e-11:
                h_str = h_new
                c_str = c_new
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_str, c_str, len(h_str))

    # Phase 4: Dimension scanning with polish
    if time.monotonic() < deadline - 120:
        dim_candidates = sorted(set([
            n_base - 4, n_base - 3, n_base - 2, n_base - 1,
            n_base + 1, n_base + 2, n_base + 3, n_base + 4,
            32, 36, 40, 44, 48, 52, 56, 60, 64, 72, 80
        ]))
        for nid in dim_candidates:
            if time.monotonic() > deadline - 80:
                break
            if nid < 20 or nid > 100:
                continue
            h_rep = resample_h(best_h, nid)
            c_rep = exact_c5(h_rep)
            best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_rep, c_rep, nid)
            if nid <= 80 and c_rep < best_c + 0.003:
                h_pol, c_pol = lp_1d_polish(h_rep, c_rep, deadline - 80, max_iter=25, r_init=0.03, r_max=0.05)
                best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_pol, c_pol, len(h_pol))
                h_ps2 = symmetrize_v(h_pol)
                c_ps2 = exact_c5(h_ps2)
                best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_ps2, c_ps2, len(h_ps2))

    # Phase 5: More LP iterations with different radii
    if time.monotonic() < deadline - 60:
        for r_try in [0.02, 0.035, 0.05, 0.015, 0.06]:
            if time.monotonic() > deadline - 40:
                break
            h_try = best_h.copy()
            c_try = best_c
            h_try, c_try = lp_1d_polish(h_try, c_try, deadline - 40, max_iter=40, r_init=r_try, r_max=r_try * 1.5, r_min=0.002)
            best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_try, c_try, len(h_try))

    # Phase 6: Final pair + LP combo
    if time.monotonic() < deadline - 30:
        h_fin = best_h.copy()
        c_fin = best_c
        h_fin, c_fin = pair_search_loop(h_fin, c_fin, deadline - 30, max_rounds=80)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_fin, c_fin, len(h_fin))
        h_fin, c_fin = lp_1d_polish(h_fin, c_fin, deadline - 10, max_iter=30, r_init=0.015, r_max=0.04)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_fin, c_fin, len(h_fin))
        h_fs = symmetrize_v(h_fin)
        c_fs = exact_c5(h_fs)
        best_h, best_c, best_n = try_update((best_h, best_c, best_n), h_fs, c_fs, len(h_fs))

    # Final return
    if best_h is None:
        best_h = ensure_feasible(np.full(64, 0.5))
        best_c = exact_c5(best_h)
        best_n = 64
    h_ret = ensure_feasible(best_h)
    c_ret = exact_c5(h_ret)
    if not np.isfinite(c_ret) or c_ret > 0.5:
        h_ret = ensure_feasible(best_h)
        c_ret = exact_c5(h_ret)
    return h_ret, float(c_ret), int(h_ret.size)
