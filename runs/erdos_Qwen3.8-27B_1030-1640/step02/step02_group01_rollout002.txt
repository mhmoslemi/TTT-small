def project_box_sum(y, n):

    n = int(n)

        return np.empty(0, dtype=float)

    n = int(h.size)

    """Compute multi-gradient direction focusing on top K active lags."""
    n = int(h.size)

    # Project onto zero-sum subspace
    g = g - float(np.mean(g))

def resample(h, n_new):
    """Resample h to n_new points using linear interpolation."""
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    n_new = int(n_new)
    if n <= 0 or n_new <= 0 or n == n_new:
        return clean_feasible(h, n) if n == n_new else None
    dx_old = 2.0 / float(n)
    dx_new = 2.0 / float(n_new)
    x_old = (np.arange(n, dtype=float) + 0.5) * dx_old
    x_new = (np.arange(n_new, dtype=float) + 0.5) * dx_new
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def clean_feasible(h, n=None):

def coordinate_descent(h0, max_iters=1000, deadline=None, need=3.0):

        j = int(rng.integers(n))

        if val < cur_val - 1e-12:
            h = h_new
            cur_val = val
    return h, cur_val

    return float(np.max(vals))

def overlap_components(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_at_lag(h, p):
    n = int(h.size)
    dx = 2.0 / float(n)
    k = int(p) - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[k:n] += dx * (1.0 - h[0:n-k])
            g[0:n-k] -= dx * h[k:n]
    else:
        s = -k
        if n - s > 0:
            g[0:n-s] += dx * (1.0 - h[s:n])
            g[s:n] -= dx * h[0:n-s]
    return g

def clean_feasible(h, n=None):
    h = np.asarray(h, dtype=float).reshape(-1)
    if n is None:
        n = h.size
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    if h.size != n:
        z = np.full(n, 0.5)
        m = min(h.size, n)
        if m > 0:
            z[:m] = h[:m]
        h = z
    h = np.array(h, dtype=float, copy=True)
    h[~np.isfinite(h)] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    if abs(0.5 * float(n) - float(np.sum(h))) > 1e-10:
        h = project_box_sum(h, n)
    return h

def consider_candidate(best, h):
    h = np.asarray(h, dtype=float)
    if h.size <= 0:
        return best
    h = clean_feasible(h, int(h.size))
    val = exact_value(h)
    if not math.isfinite(val):
        return best
    cur = float(best["val"])
    curn = int(best["n"])
    if val < cur - 1e-10:
        best["h"] = h.copy()
        best["val"] = float(val)
        best["n"] = int(h.size)
    elif abs(val - cur) <= 1e-10 and int(h.size) < curn:
        best["h"] = h.copy()
        best["val"] = float(val)
        best["n"] = int(h.size)
    return best

def symmetrize(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    return project_box_sum(0.5 * (h + h[::-1]), n)

def resample(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    n_new = int(n_new)
    if n <= 0 or n_new <= 0:
        return None
    if n == n_new:
        return clean_feasible(h, n)
    dx_old = 2.0 / float(n)
    dx_new = 2.0 / float(n_new)
    x_old = (np.arange(n, dtype=float) + 0.5) * dx_old
    x_new = (np.arange(n_new, dtype=float) + 0.5) * dx_new
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(y, n_new)

def amgd_step(h, K, delta, max_alpha=3.0):
    n = int(h.size)
    c = overlap_components(h)
    c_max = float(np.max(c))
    threshold = c_max - delta
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > K:
        order = active[np.argsort(-c[active])]
        idx = order[:K]
    else:
        idx = active
    g = np.zeros(n)
    for p in idx:
        g += grad_at_lag(h, int(p))
    g /= float(len(idx))
    g = g - float(np.mean(g))
    g_norm = float(np.linalg.norm(g))
    if g_norm < 1e-14:
        return h, c_max
    g = g / g_norm
    best_h = h
    best_val = c_max
    alpha = 1.0
    for _ in range(25):
        h_new = project_box_sum(h - alpha * g, n)
        val = exact_value(h_new)
        if val < best_val - 1e-13:
            best_h = h_new
            best_val = val
            alpha = min(5.0, alpha * 1.4)
        else:
            alpha *= 0.65
        if alpha < 1e-7:
            break
    return best_h, best_val

def amgd_optimize(h0, deadline, K=12, delta=0.08, max_rounds=300, need=3.0):
    h = clean_feasible(h0, int(np.asarray(h0).size))
    n = int(h.size)
    if n <= 0:
        return h, math.inf
    cur_val = exact_value(h)
    delta_cur = delta
    K_cur = K
    stagnant = 0
    moment = np.zeros(n)
    beta = 0.8
    for r in range(max_rounds):
        if deadline is not None and (time.monotonic() - deadline) < need:
            break
        c = overlap_components(h)
        c_max = float(np.max(c))
        threshold = c_max - delta_cur
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > K_cur:
            order = active[np.argsort(-c[active])]
            idx = order[:K_cur]
        else:
            idx = active
        g = np.zeros(n)
        for p in idx:
            g += grad_at_lag(h, int(p))
        g /= float(len(idx))
        g = g - float(np.mean(g))
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break
        g = g / g_norm
        moment = beta * moment - g
        m_norm = float(np.linalg.norm(moment))
        if m_norm < 1e-14:
            break
        direction = moment / m_norm
        best_h = h
        best_val = cur_val
        alpha = 1.0
        for _ in range(20):
            h_new = project_box_sum(h - alpha * direction, n)
            val = exact_value(h_new)
            if val < best_val - 1e-13:
                best_h = h_new
                best_val = val
                alpha = min(4.0, alpha * 1.3)
            else:
                alpha *= 0.6
            if alpha < 1e-7:
                break
        if best_val < cur_val - 1e-12:
            h = best_h
            cur_val = best_val
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 8 == 0:
                delta_cur = max(0.002, delta_cur * 0.6)
                K_cur = max(1, K_cur // 2)
            if stagnant > 40:
                rng_esc = np.random.default_rng(r * 31 + 7)
                h = clean_feasible(h + 0.02 * rng_esc.standard_normal(n), n)
                cur_val = exact_value(h)
                moment = np.zeros(n)
                stagnant = 0
                delta_cur = max(0.02, delta)
                K_cur = K
    return h, cur_val

def lse_optimize(h0, deadline, tau_init=0.05, max_rounds=200, need=3.0):
    h = clean_feasible(h0, int(np.asarray(h0).size))
    n = int(h.size)
    if n <= 0:
        return h, math.inf
    cur_val = exact_value(h)
    tau = tau_init
    moment = np.zeros(n)
    beta = 0.7
    stagnant = 0
    for r in range(max_rounds):
        if deadline is not None and (time.monotonic() - deadline) < need:
            break
        c = overlap_components(h)
        c_max = float(np.max(c))
        z = (c - c_max) / tau
        z = z - z.max()
        ez = np.exp(z)
        ss = float(np.sum(ez))
        if ss <= 0:
            break
        w = ez / ss
        g = np.zeros(n)
        for p in range(len(c)):
            if w[p] > 1e-8:
                g += w[p] * grad_at_lag(h, p)
        g = g - float(np.mean(g))
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break
        g = g / g_norm
        moment = beta * moment - g
        m_norm = float(np.linalg.norm(moment))
        if m_norm < 1e-14:
            break
        direction = moment / m_norm
        best_h = h
        best_val = cur_val
        alpha = 0.8
        for _ in range(15):
            h_new = project_box_sum(h - alpha * direction, n)
            val = exact_value(h_new)
            if val < best_val - 1e-13:
                best_h = h_new
                best_val = val
                alpha = min(3.0, alpha * 1.2)
            else:
                alpha *= 0.5
            if alpha < 1e-7:
                break
        if best_val < cur_val - 1e-12:
            h = best_h
            cur_val = best_val
            stagnant = 0
            tau = min(0.1, tau * 1.05)
        else:
            stagnant += 1
            tau = max(0.001, tau * 0.7)
            if stagnant > 30:
                moment = np.zeros(n)
                stagnant = 0
    return h, cur_val

def binary_hill_climb(n, max_iters=5000, deadline=None, need=5.0):
    if n < 4 or n % 2 != 0:
        return None, math.inf
    half = n // 2
    rng = np.random.default_rng(99)
    h = np.zeros(n)
    positives = list(range(0, n, 2))
    rng.shuffle(positives)
    h[positives[:half]] = 1.0

    def eval_bin(hb):
        c = np.correlate(hb, 1.0 - hb, mode='full') * (2.0 / n)
        return float(np.max(c))

    cur_val = eval_bin(h)
    no_improve = 0
    for it in range(max_iters):
        if deadline is not None and (time.monotonic() - deadline) < need:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h[i] = 0.0
        h[j] = 1.0
        new_val = eval_bin(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
            if no_improve > 200:
                # Try a double swap
                if len(ones) >= 2 and len(zeros) >= 2:
                    i1 = ones[int(rng.integers(len(ones)))]
                    i2 = ones[int(rng.integers(len(ones)))]
                    j1 = zeros[int(rng.integers(len(zeros)))]
                    j2 = zeros[int(rng.integers(len(zeros)))]
                    if i1 == i2 or j1 == j2:
                        continue
                    h[i1] = 0.0
                    h[j1] = 1.0
                    h[i2] = 0.0
                    h[j2] = 1.0
                    new_val = eval_bin(h)
                    if new_val < cur_val - 1e-12:
                        cur_val = new_val
                        no_improve = 0
                    else:
                        h[i1] = 1.0
                        h[j1] = 0.0
                        h[i2] = 1.0
                        h[j2] = 0.0
                if no_improve > 500:
                    break
    return h, cur_val

def coordinate_descent(h0, max_iters=3000, deadline=None, need=3.0):
    h = clean_feasible(h0, int(np.asarray(h0).size))
    n = int(h.size)
    cur_val = exact_value(h)
    rng = np.random.defaultsgi_rng = np.random.default_rng(200 + int(n))
    for it in range(max_iters):
        if deadline is not None and (time.monotonic() - deadline) < need:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta_try = float(rng.choice(np.array([0.01, 0.005, 0.002, -0.01, -0.005, -0.002])))
        h_new = h.copy()
        h_new[i] += delta_try
        h_new[j] -= delta_try
        h_new = clean_feasible(h_new, n)
        val = exact_value(h_new)
        if val < cur_val - 1e-13:
            h = h_new
            cur_val = val
    return h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(1000 * (int(seed) if seed is not None else 0) + 17)

    best = {"h": None, "val": math.inf, "n": 0}

    init = globals().get("initial_h_values", None)
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        if hinit.size > 0 and np.any(np.isfinite(hinit)):
            hinit = clean_feasible(hinit, int(hinit.size))
            consider_candidate(best, hinit)
            consider_candidate(best, symmetrize(hinit))

    if best["h"] is None:
        n_init = 44
        h_init = np.full(n_init, 0.5)
        h_init += 0.3 * np.sin(2.0 * np.pi * np.arange(n_init) / (n_init / 3.0))
        h_init = project_box_sum(h_init, n_init)
        consider_candidate(best, h_init)

    if best["h"] is None:
        h_out = np.full(44, 0.5)
        return h_out, exact_value(h_out), 44

    # Phase 1: AMGD with momentum on initial
    if deadline - time.monotonic() > 80:
        h_a, _ = amgd_optimize(best["h"], deadline - 5, K=16, delta=0.1, max_rounds=400, need=4.0)
        consider_candidate(best, h_a)

    # Phase 2: LSE-based optimization
    if deadline - time.monotonic() > 60:
        h_l, _ = lse_optimize(best["h"], deadline - 5, tau_init=0.08, max_rounds=150, need=4.0)
        consider_candidate(best, h_l)

    # Phase 3: Try different n values
    cur_n = int(best["n"])
    n_tries = sorted(set([cur_n+2, cur_n-2, cur_n+4, cur_n-4, cur_n+6, 46, 50, 54, 58, 62]))
    n_tries = [nt for nt in n_tries if 20 <= nt <= 120 and nt % 2 == 0 and nt != cur_n]
    for n_try in n_tries[:8]:
        if deadline - time.monotonic() < 25:
            break
        hr = resample(best["h"], n_try)
        if hr is None:
            continue
        hr = clean_feasible(hr, n_try)
        consider_candidate(best, hr)
        if deadline - time.monotonic() > 20:
            h_res, _ = amgd_optimize(hr, deadline - 5, K=12, delta=0.08, max_rounds=100, need=5.0)
            consider_candidate(best, h_res)

    # Phase 4: Binary pattern search + continuous refinement
    for n_bin in [int(best["n"]), int(best["n"])+2, int(best["n"])-2, 46, 50, 54, 58]:
        if deadline - time.monotonic() < 35:
            break
        if n_bin < 20 or n_bin > 120 or n_bin % 2 != 0:
            continue
        hb, vb = binary_hill_climb(n_bin, max_iters=4000, deadline=deadline - 10, need=8.0)
        if hb is not None and math.isfinite(vb):
            h_smooth = project_box_sum(hb + 0.08 * np.random.default_rng(n_bin).standard_normal(n_bin), n_bin)
            consider_candidate(best, h_smooth)
            consider_candidate(best, clean_feasible(hb, n_bin))
            if deadline - time.monotonic() > 20:
                h_am, _ = amgd_optimize(h_smooth, deadline - 5, K=10, delta=0.06, max_rounds=150, need=5.0)
                consider_candidate(best, h_am)
                h_ls, _ = lse_optimize(h_smooth, deadline - 5, tau_init=0.04, max_rounds=80, need=5.0)
                consider_candidate(best, h_ls)

    # Phase 5: Coordinate descent polish
    if deadline - time.monotonic() > 30:
        h_cd, _ = coordinate_descent(best["h"], max_iters=3000, deadline=deadline - 5, need=3.0)
        consider_candidate(best, h_cd)

    # Phase 6: Final aggressive AMGD with narrow active set
    if deadline - time.monotonic() > 20:
        h_fin, _ = amgd_optimize(best["h"], deadline - 2, K=4, delta=0.015, max_rounds=200, need=3.0)
        consider_candidate(best, h_fin)

    # Phase 7: Symmetrize and re-optimize
    hs = symmetrize(best["h"])
    consider_candidate(best, hs)
    if exact_value(hs) < float(best["val"]) + 0.001 and deadline - time.monotonic() > 15:
        h_sym, _ = amgd_optimize(hs, deadline - 2, K=6, delta=0.02, max_rounds=100, need=3.0)
        consider_candidate(best, h_sym)

    final_h = best["h"]
    if final_h is None:
        final_h = np.full(44, 0.5)
    final_h = clean_feasible(final_h, int(np.asarray(final_h).size))
    if final_h.size <= 0:
        final_h = np.full(44, 0.5)
        final_h = clean_feasible(final_h, 44)
    final_val = exact_value(final_h)
    if not math.isfinite(final_val):
        final_h = np.full(44, 0.5)
        final_h = clean_feasible(final_h, 44)
        final_val = exact_value(final_h)
    return final_h, float(final_val), int(final_h.size)
