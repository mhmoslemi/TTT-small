def make_constraint_jac(k, n):

    def jac(z, *args):
        tau = z[0]
        h = z[1:]
        dx = 2.0 / n
        lo = max(0, -k)
        hi = min(n, n - k)
        idx = np.arange(lo, hi)

        grad[1:][jidx] += dx * (-h[idx])
        return grad
    return jac

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(overlaps))


def overlap_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def repair_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_func(z, *args):
    return z[0]


def obj_grad(z, *args):
    grad = np.zeros(len(z))
    grad[0] = 1.0
    return grad


def constr_fun(z, k, n):
    h = z[1:]
    dx = 2.0 / n
    lo = max(0, -k)
    hi = min(n, n - k)
    idx = np.arange(lo, hi)
    jidx = idx + k
    return dx * np.sum(h[idx] * (1.0 - h[jidx])) - z[0]


def constr_jac(z, k, n):
    h = z[1:]
    dx = 2.0 / n
    lo = max(0, -k)
    hi = min(n, n - k)
    idx = np.arange(lo, hi)
    jidx = idx + k
    grad = np.zeros(len(z))
    grad[0] = -1.0
    grad[1:][idx] += dx * (1.0 - h[jidx])
    grad[1:][jidx] += -dx * h[idx]
    return grad


def eq_fun(z, n):
    return np.sum(z[1:]) - n / 2.0


def eq_jac(z, n):
    grad = np.zeros(len(z))
    grad[1:] = 1.0
    return grad

    ov = overlap_vector(h)
    top_idx = np.argsort(ov)[::-1][:num]
    return (top_idx - (n - 1)).tolist()

def binary_start(n, rng):
    p = 0.5 + 0.1 * rng.standard_normal()
    p = np.clip(p, 0.2, 0.8)
    h = (rng.random(n) < p).astype(float)

    period = rng.randint(3, max(4, n // 4))
    pattern_len = min(period, n)
    pattern = (rng.random(pattern_len) < 0.5).astype(float)
    h = np.tile(pattern, n // pattern_len + 1)[:n]
    h += 0.1 * rng.standard_normal(n)
    return repair_feasible(h, n)


def slsqp_solve(h_init, n, active_lags, maxiter=150):
    h_init = repair_feasible(h_init.copy(), n)
    c5_init = evaluate_c5(h_init)
    tau_init = c5_init + 0.005
    z0 = np.concatenate([[tau_init], h_init])

    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n

    constraints = []
    for k in active_lags:
        constraints.append({
            'type': 'ineq',
            'fun': constr_fun,
            'jac': constr_jac,
            'args': (k, n)
        })
    constraints.append({
        'type': 'eq',
        'fun': eq_fun,
        'jac': eq_jac,
        'args': (n, 0)
    })

    try:
        result = minimize(
            obj_func, z0,
            jac=obj_grad,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-9}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):

            c5_sol = evaluate_c5(h_sol)

        pass

def transfer_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    h_new = np.zeros(n_new)

        x = (j + 0.5) * 2.0 / n_new
        pos = x * n_old / 2.0
        i0 = int(np.floor(pos))
        i1 = min(i0 + 1, n_old - 1)
        i0 = min(i0, n_old - 1)
        frac = pos - np.floor(pos)
        h_new[j] = (1.0 - frac) * h_old[i0] + frac * h_old[i1]
    return repair_feasible(h_new, n_new)
