import numpy as np
import time

def compute_c5_values(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h, dx):
    return float(np.max(compute_c5_values(h, dx)))

def project_to_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    s0 = np.sum(np.clip(h, 0.0, 1.0))
    if abs(s0 - target_sum) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = -h.max() - 0.5
    hi = 1.0 - h.min() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def ps_gradient(h, p, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    c_safe = np.maximum(c, 1e-12)
    S = np.sum(c_safe ** p)
    if S < 1e-30:
        return np.zeros(n), 0.0
    scale = S ** (1.0 / p - 1.0)
    w = c_safe ** (p - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
        else:
            mp = -m
            grad[:n - mp] += wk * (1.0 - h[mp:])
            grad[mp:] -= wk * h[:n - mp]
    val = dx * S ** (1.0 / p)
    return dx * scale * grad, val

def subgrad_h(h, k_star, n):
    d = k_star - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        grad[d:] = 1.0 - h[:n - d]
        grad[:n - d] -= h[d:]
    else:
        dn = -d
        grad[:n - dn] += 1.0 - h[dn:]
        grad[dn:] -= h[:n - dn]
    return grad

def optimize_phase(h, dx, target_sum, p, max_time, rng):
    n = len(h)
    h = project_to_feasible(h, target_sum).copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.08
    momentum = np.zeros(n)
    beta = 0.85
    no_improve = 0
    t_start = time.time()
    
    while time.time() - t_start < max_time:
        grad, _ = ps_gradient(h, p, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        gnorm = grad / gn
        momentum = beta * momentum + (1.0 - beta) * gnorm
        h_try = h - lr * momentum
        h_try = project_to_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-10:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 300 == 0:
                lr *= 0.92
            if no_improve > 3000:
                h = best_h + rng.randn(n) * 0.02
                h = project_to_feasible(h, target_sum)
                momentum *= 0.3
                no_improve = 0
                lr = 0.08
    return best_h, best_c5

def subgrad_polish(h, dx, target_sum, max_time):
    n = len(h)
    h = project_to_feasible(h, target_sum).copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.008
    no_improve = 0
    t_start = time.time()
    
    while time.time() - t_start < max_time:
        c = compute_c5_values(h, dx)
        # Use top-3 active constraints
        top_idx = np.argsort(c)[-3:]
        grad = np.zeros(n)
        for ki in top_idx:
            grad += subgrad_h(h, int(ki), n)
        grad /= len(top_idx)
        
        h_try = h - lr * grad
        h_try = project_to_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-11:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
            if no_improve > 2000:
                h = best_h + no_improve * 0.001 * np.random.randn(n) * 0.01
                h = project_to_feasible(h, target_sum)
                no_improve = 0
                lr = 0.008
    return best_h, best_c5

def make_random_initial(n, rng):
    h = np.zeros(n)
    n_ones = int(n // 2)
    remainder = n / 2.0 - n_ones
    if remainder > 0:
        h[rng.randint(n)] = remainder
        idx = rng.choice(n - 1, size=n_ones, replace=False)
        idx = np.sort(np.concatenate([idx, [n - 1 if idx[-1] < n - 1 else n - 2]]))
        h[idx] = 1.0
    else:
        idx = rng.choice(n, size=n_ones, replace=False)
        h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = make_random_initial(63, rng)
    
    n = len(h0)

    target_sum = n / 2.0
    h0 = project_to_feasible(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    
    remaining = budget_s - (time.time() - t0) - 40
    if remaining < 10:
        return best_h, best_c5, n
    
    # Phase 1: coarse power-sum (p=8) - 10%
    t1 = 0.10 * remaining
    h, c5 = optimize_phase(best_h, dx, target_sum, 8, t1, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h
    
    # Phase 2: medium power-sum (p=20) - 30%
    t2 = 0.30 * remaining
    h, c5 = optimize_phase(best_h, dx, target_sum, 20, t2, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h
    
    # Phase 3: fine power-sum (p=50) - 30%
    t3 = 0.30 * remaining
    h, c5 = optimize_phase(best_h, dx, target_sum, 50, t3, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    t4 = 0.15 * remaining
    h, c5 = optimize_phase(best_h, dx, target_sum, 100, t4, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h

        h, c5 = subgrad_polish(best_h, dx, target_sum, time_left)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    
    return best_h, best_c5, n
