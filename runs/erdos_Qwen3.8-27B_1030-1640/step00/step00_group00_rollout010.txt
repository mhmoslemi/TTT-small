def evaluate_C(h):

def project_h(y, target_mass):
    """Project y onto {0 <= z <= 1, sum(z) = target_mass}."""
    n = len(y)
    # Bisection on lambda: z_i = clip(y_i + lambda, 0, 1)
    lo = -max(y.max() + 1, target_mass / n + 1)
    hi = max(1 - y.min() + 1, target_mass / n + 1)
    
    def sum_at(lam):
        return np.sum(np.clip(y + lam, 0, 1))
    
    # Expand bracket
    for _ in range(50):
        if sum_at(lo) >= target_mass:
            break
        lo *= 2
    for _ in range(50):
        if sum_at(hi) <= target_mass:
            break
        hi = 2*hi if hi > 0 else hi/2
    
    # Bisection
    for _ in range(60):
        mid = (lo + hi) / 2
        if sum_at(mid) < target_mass:
            lo = mid
        else:
            hi = mid
    
    return np.clip(y + (lo+hi)/2, 0, 1)

def soft_max_objective_and_grad(h, theta):
    """Compute soft-max smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    # Soft max
    max_corr = np.max(corr)
    exp_corr = np.exp((corr - max_corr) / theta)
    Z = np.sum(exp_corr)
    if Z < 1e-300 or not np.isfinite(Z):
        return np.nan, np.zeros(n)
    weights = exp_corr / Z  # sums to 1
    obj = max_corr + theta * np.log(Z)
    
    # Gradient
    grad = np.zeros(n)
    for p in range(2*n-1):
        w = weights[p]
        if w < 1e-15:
            continue
        ell = p - (n-1)
        lo = max(0, ell)
        hi = min(n, n + ell)
        if lo >= hi:
            continue
        i = np.arange(lo, hi)
        grad[i] += w * (1 - h[i - ell])
        grad[i - ell] -= w * h[i]
    grad *= dx
    return obj, grad

def pair_line_search(h, s, t, n_evals=60):
    """Golden section search for optimal transfer from s to t."""
    n = len(h)
    dx = 2.0 / n
    T = min(h[s], 1 - h[t])
    if T < 1e-12:
        return h, evaluate_C(h)
    
    def line_obj(tau):
        h_new = h.copy()
        h_new[s] = h[s] - tau
        h_new[t] = h[t] + tau
        return evaluate_C(h_new)
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a, b = 0.0, T
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = line_obj(c)
    fd = line_obj(d)
    feval = line_obj(0.0)  # current value
    best_tau = 0.0
    best_val = feval

            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = line_obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = line_obj(d)
        if fc < best_val:

        h2[t] += tau

def soft_max_obj_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    ec = np.exp((corr - m) / theta)
    Z = np.sum(ec)
    if not np.isfinite(Z) or Z < 1e-300:
        return np.nan, np.zeros(n)
    w = ec / Z
    obj = m + theta * np.log(Z)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-15:
            continue
        ell = p - (n - 1)
        lo = max(0, ell)
        hi = min(n, n + ell)
        if lo >= hi:
            continue
        idx = np.arange(lo, hi)
        grad[idx] += wp * (1.0 - h[idx - ell])
        grad[idx - ell] -= wp * h[idx]
    grad *= dx
    return obj, grad

def gradient_descent(h0, thetas, max_steps=100, deadline=None):
    n = len(h0)
    target = n / 2.0
    h = project_h(h0.copy(), target)
    best = h.copy()
    best_c = evaluate_C(best)
    for theta in thetas:
        if deadline is not None and time.time() > deadline:
            break
        for _ in range(max_steps):
            if deadline is not None and time.time() > deadline:
                break
            obj, grad = soft_max_obj_grad(h, theta)
            if not np.isfinite(obj) or np.allclose(grad, 0):
                break
            gmax = np.max(np.abs(grad))
            if gmax < 1e-12:
                break
            alpha = 0.5 / gmax
            for _ in range(15):
                h_try = project_h(h - alpha * grad, target)
                obj_try, _ = soft_max_obj_grad(h_try, theta)

                    h = h_try
                    c_exact = evaluate_C(h)
                    if c_exact < best_c:

                        best_c = c_exact
                    break
                alpha *= 0.5
    return best, best_c
