import numpy as np
import time
from scipy.optimize import minimize
from scipy.ndimage import correlate1d

def evaluate_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    """Project h to [0,1]^N with sum = N/2."""
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    # Bisection on lambda: sum(clip(h + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    # First, clip
    h_clipped = np.clip(h, 0, 1)
    s = h_clipped.sum()
    if abs(s - target) < 1e-12:
        return h_clipped
    # Need to adjust
    if s > target:
        # Need to decrease: h_clipped - lambda for values in (0,1)
        # Binary search lambda
        lo, hi = 0.0, 1.0
        for _ in range(100):
            mid = (lo + hi) / 2
            h_new = np.clip(h_clipped - mid, 0, 1)
            if h_new.sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h_clipped - (lo+hi)/2, 0, 1)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(100):
            mid = (lo + hi) / 2
            h_new = np.clip(h_clipped + mid, 0, 1)
            if h_new.sum() < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h_clipped + (lo+hi)/2, 0, 1)
    # Fix exact sum
    diff = target - h.sum()
    # Distribute the difference among non-extremal elements
    non_ext = (h > 1e-10) & (h < 1 - 1e-10)
    if non_ext.any():
        h[non_ext] += diff / non_ext.sum()
        h = np.clip(h, 0, 1)
    # If still off, adjust the closest to 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
    return np.clip(h, 0, 1)

def binary_from_continuous(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    n = len(h)
    if n % 2 != 0:
        # Can't have exact binary for odd n
        return None
    # Sort by value descending, take top n/2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def random_binary(n, rng):
    """Generate random binary vector with exactly n/2 ones."""
    if n % 2 != 0:
        return None
    positions = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[positions] = 1.0
    return h

def swap_improve(h, max_iters=10000, time_limit=None, start_time=None):
    """Improve a binary or near-binary solution via pair swaps."""
    h = h.copy()
    n = len(h)
    target = n / 2.0
    best_c = evaluate_c5(h)
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    improved = True
    iters = 0
    while improved and iters < max_iters:
        improved = False
        iters += 1
        if time_limit and start_time and (time.perf_counter() - start_time) > time_limit:
            break
        
        best_delta = 1e-12
        best_pair = None
        
        # Try all pairs (one from ones, one from zeros)
        for i in ones:
            for j in zeros:
                # Swap: h[i] -> 0, h[j] -> 1
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c_new = evaluate_c5(h_new)
                delta = best_c - c_new
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c = evaluate_c5(h)
            improved = True
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    
    return h, best_c

def continuous_optimize(h, max_iter=200, time_limit=30):
    """Use SLSQP to optimize the continuous problem."""
    n = len(h)
    target = n / 2.0
    start_time = time.perf_counter()
    
    def objective(h_flat):
        if (time.perf_counter() - start_time) > time_limit:
            return 100.0
        return evaluate_c5(h_flat)
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bounds = [(0, 1)] * n
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-10})
    if result.success or result.fun < evaluate_c5(h):
        h_opt = project_feasible(result.x)
        return h_opt
    return h

def resample(h_parent, n_new):
    """Resample parent to new length using linear interpolation."""
    n_old = len(h_parent)
    # Map new indices to old indices
    # Center of i-th new sample: (i + 0.5) * 2/n_new
    # In old coordinates: (i + 0.5) * 2/n_new / (2/n_old) = (i + 0.5) * n_old / n_new
    src = (np.arange(n_new) + 0.5) * n_old / n_new
    src = np.clip(src, 0, n_old - 1)
    idx = np.floor(src).astype(int)
    frac = src - idx
    idx = np.minimum(idx, n_old - 2)
    h_new = h_parent[idx] * (1 - frac) + h_parent[idx + 1] * frac
    return project_feasible(h_new)

def symmetric_h(h):
    """Create symmetric version of h."""
    h_sym = 0.5 * (h + h[::-1])
    return project_feasible(h_sym)

def make_odd_feasible(h_even, n_odd):
    """Create odd-length feasible from even binary."""
    n_even = len(h_even)
    # Resample to odd length
    return resample(h_even, n_odd)

def pairwise_exchange(h, n_candidates=50, time_limit=5.0):
    """Try pairwise exchanges of mass between positions."""
    h = h.copy()
    n = len(h)
    start_time = time.perf_counter()
    best_c = evaluate_c5(h)
    
    for _ in range(n_candidates):
        if (time.perf_counter() - start_time) > time_limit:
            break
        
        # Pick two random positions
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Try transferring mass from i to j (or vice versa)
        for sign in [1, -1]:
            # h[i] - sign*tau, h[j] + sign*tau
            tau_max = min(h[i] if sign > 0 else 1 - h[i],
                         1 - h[j] if sign > 0 else h[j])
            if tau_max <= 1e-12:
                continue
            
            # Binary search for best tau
            lo, hi = 0.0, tau_max
            best_tau = 0.0
            best_c_local = best_c
            
            for _ in range(20):
                mid = (lo + hi) / 2
                if mid < 1e-14:
                    break
                h_test = h.copy()
                h_test[i] -= sign * mid
                h_test[j] += sign * mid
                h_test = np.clip(h_test, 0, 1)
                c_test = evaluate_c5(h_test)
                if c_test < best_c_local:
                    best_c_local = c_test
                    best_tau = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_tau > 0 and best_c_local < best_c:
                h[i] -= sign * best_tau
                h[j] += sign * best_tau
                h = np.clip(h, 0, 1)
                best_c = best_c_local
    
    return h, best_c

def fix_sum(h):
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    idx = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
    if len(idx) > 0:
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        h = h.copy()
        h[np.argmin(np.abs(h - 0.5))] += diff
    return np.clip(h, 0.0, 1.0)

def project_feasible(h):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    return fix_sum(h)

def resample_linear(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_new = (np.arange(n_new) + 0.5) / n_new
    x_old = (np.arange(n_old) + 0.5) / n_old
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new)

def symmetric_h(h):
    return project_feasible(0.5 * (h + h[::-1]))

def binary_from_rank(h, n):
    if n % 2 != 0:
        return None
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[order[:n // 2]] = 1.0
    return h_bin

def random_binary(n, rng):
    if n % 2 != 0:
        return None
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def pairwise_improve(h, max_rounds=200, time_limit=60.0, rng=None):
    h = h.copy().astype(np.float64)
    n = len(h)
    t_start = time.perf_counter()
    best_c = evaluate_c5(h)
    gr = (np.sqrt(5) - 1) / 2.0
    
    for rnd in range(max_rounds):
        if time.perf_counter() - t_start > time_limit:
            break
        improved = False
        # Build list of active pairs (at least one not at boundary)
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                if (h[i] > 1e-10 or h[j] > 1e-10) and (h[i] < 1 - 1e-10 or h[j] < 1 - 1e-10):
                    pairs.append((i, j))
        if rng is not None and len(pairs) > 200:
            rng.shuffle(pairs)
        
        for i, j in pairs:
            if time.perf_counter() - t_start > time_limit:
                break
            best_c_pair = best_c
            best_tau = 0.0
            best_dir = 0
            for direction in (1, -1):
                if direction == 1:
                    tau_max = min(h[i], 1.0 - h[j])
                else:
                    tau_max = min(h[j], 1.0 - h[i])
                if tau_max < 1e-14:
                    continue
                # Golden section on tau in [0, tau_max]
                a, b = 0.0, tau_max
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                h_c = h.copy()
                h_d = h.copy()
                if direction == 1:
                    h_c[i] -= c_pt; h_c[j] += c_pt
                    h_d[i] -= d_pt; h_d[j] += d_pt
                else:
                    h_c[i] += c_pt; h_c[j] -= c_pt
                    h_d[i] += d_pt; h_d[j] -= d_pt
                fc = evaluate_c5(h_c)
                fd = evaluate_c5(h_d)
                for _ in range(35):
                    if b - a < 1e-12:
                        break
                    if fc < fd:
                        b = d_pt; d_pt = c_pt; fd = fc
                        c_pt = b - gr * (b - a)
                        h_c = h.copy()
                        if direction == 1:
                            h_c[i] -= c_pt; h_c[j] += c_pt
                        else:
                            h_c[i] += c_pt; h_c[j] -= c_pt
                        fc = evaluate_c5(h_c)
                    else:
                        a = c_pt; c_pt = d_pt; fc = fd
                        d_pt = a + gr * (b - a)
                        h_d = h.copy()
                        if direction == 1:
                            h_d[i] -= d_pt; h_d[j] += d_pt
                        else:
                            h_d[i] += d_pt; h_d[j] -= d_pt
                        fd = evaluate_c5(h_d)
                tau_opt = (a + b) / 2.0
                h_test = h.copy()
                if direction == 1:
                    h_test[i] -= tau_opt; h_test[j] += tau_opt
                else:
                    h_test[i] += tau_opt; h_test[j] -= tau_opt
                h_test = np.clip(h_test, 0, 1)
                c_opt = evaluate_c5(h_test)
                if c_opt < best_c_pair - 1e-13:
                    best_c_pair = c_opt
                    best_tau = tau_opt
                    best_dir = direction
            if best_dir == 1:
                h[i] -= best_tau; h[j] += best_tau
            elif best_dir == -1:
                h[i] += best_tau; h[j] -= best_tau
            h = np.clip(h, 0, 1)
            h = fix_sum(h)
            nc = evaluate_c5(h)
            if nc < best_c:
                best_c = nc
                improved = True
        if not improved:
            break
    return h, best_c

def swap_improve_binary(h, max_rounds=50, time_limit=30.0, rng=None):
    h = h.copy().astype(np.float64)
    n = len(h)
    t_start = time.perf_counter()
    best_c = evaluate_c5(h)
    for rnd in range(max_rounds):
        if time.perf_counter() - t_start > time_limit:
            break
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        if rng is not None:
            rng.shuffle(ones)
            rng.shuffle(zeros)
        for i in ones:
            if time.perf_counter() - t_start > time_limit:
                break
            for j in zeros:
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                c2 = evaluate_c5(h2)
                if c2 < best_c - 1e-13:
                    h = h2
                    best_c = c2
                    improved = True
                    ones = list(np.where(h > 0.5)[0])
                    zeros = list(np.where(h < 0.5)[0])
                    break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 5.0
    
    def time_left():
        return max(0.0, deadline - time.perf_counter())
    
    rng = np.random.default_rng(seed)
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Get parent
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64)
        if p.size > 0 and np.all(np.isfinite(p)):
            parent = np.clip(p, 0, 1)
    except (NameError, AttributeError):
        pass
    
    n_parent = len(parent) if parent is not None else 46
    
    # Phase 1: Initialize candidates
    candidates = []  # list of (c, h, n)
    
    if parent is not None:
        c_p = evaluate_c5(parent)
        candidates.append((c_p, parent.copy(), n_parent))
        # Symmetric
        h_sym = symmetric_h(parent)
        candidates.append((evaluate_c5(h_sym), h_sym, n_parent))
    
    # Try resampling to various N
    for n_try in range(max(14, n_parent - 10), min(250, n_parent + 60)):
        if time_left() < 10:
            break
        if parent is not None:
            h_r = resample_linear(parent, n_try)
        else:
            h_r = project_feasible(np.full(n_try, 0.5))
        c_r = evaluate_c5(h_r)
        candidates.append((c_r, h_r, n_try))
        # Binary for even
        if n_try % 2 == 0:
            h_b = binary_from_rank(h_r, n_try)
            if h_b is not None:
                candidates.append((evaluate_c5(h_b), h_b, n_try))
    
    # Random binary starts
    for n_try in [44, 46, 48, 50, 52, 56, 60, 64]:
        if time_left() < 10:
            break
        for _ in range(3):
            h_r = random_binary(n_try, rng)
            if h_r is not None:
                candidates.append((evaluate_c5(h_r), h_r, n_try))
    
    # Sort by objective
    candidates.sort(key=lambda x: x[0])
    
    # Phase 2: Optimize top candidates
    for idx, (c_init, h_init, n_curr) in enumerate(candidates[:15]):
        if time_left() < 20:
            break
        # Allocate time based on rank
        frac = max(0.05, 0.8 / (idx + 1))
        alloc = min(frac * budget_s * 0.7, time_left() - 15)
        if alloc < 3:
            continue
        
        h_work = h_init.copy()
        
        # If nearly binary, do swap improvement first
        if n_curr % 2 == 0:
            h_bin = binary_from_rank(h_work, n_curr)
            if h_bin is not None:
                alloc_swap = min(alloc * 0.3, 20.0)
                h_sw, c_sw = swap_improve_binary(h_bin, max_rounds=30,
                                                 time_limit=alloc_swap, rng=rng)
                if c_sw < best_c:
                    best_c = c_sw
                    best_h = h_sw.copy()
                    best_n = n_curr
                # Continue with continuous from the better of h_work and h_sw
                if c_sw < c_init:
                    h_work = h_sw.copy()
        
        # Pairwise continuous improvement
        alloc_pw = min(alloc * 0.7, time_left() - 15)
        if alloc_pw > 2:
            h_pw, c_pw = pairwise_improve(h_work, max_rounds=100,
                                          time_limit=alloc_pw, rng=rng)
            if c_pw < best_c:
                best_c = c_pw
                best_h = h_pw.copy()
                best_n = n_curr
        
        # Symmetrize and retry
        if time_left() > 10:
            h_s = symmetric_h(h_pw if 'h_pw' in dir() else h_work)
            h_sw2, c_sw2 = pairwise_improve(h_s, max_rounds=50,
                                            time_limit=min(10, time_left() / 2), rng=rng)
            if c_sw2 < best_c:
                best_c = c_sw2
                best_h = h_sw2.copy()
                best_n = n_curr
    
    # Phase 3: Polish best
    if time_left() > 10 and best_h is not None:
        h_pol, c_pol = pairwise_improve(best_h, max_rounds=50,
                                        time_limit=time_left() - 5, rng=rng)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol.copy()
            best_n = best_n  # same n
    
    # Final: try resampling best to nearby N and re-optimize
    if time_left() > 15 and best_h is not None:
        for delta in [-3, -2, -1, 1, 2, 3, 5]:
            if time_left() < 8:
                break
            n_new = best_n + delta
            if n_new < 12:
                continue
            h_r = resample_linear(best_h, n_new)
            h_o, c_o = pairwise_improve(h_r, max_rounds=30,
                                        time_limit=min(8, time_left() / 2 - 1), rng=rng)
            if c_o < best_c:
                best_c = c_o
                best_h = h_o.copy()
                best_n = n_new
    
    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(46, 0.5))
        best_c = evaluate_c5(best_h)
        best_n = 46
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h)
    best_c = evaluate_c5(best_h)
    
    return (best_h, best_c, len(best_h))
