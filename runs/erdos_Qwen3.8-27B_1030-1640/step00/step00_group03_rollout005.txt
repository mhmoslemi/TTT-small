import time

    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0, 1)
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-12:
        mask = (projected > 1e-10) & (projected < 1 - 1e-10)
        if np.sum(mask) > 0:
            projected[mask] += diff / np.sum(mask)
            projected = np.clip(projected, 0, 1)
    return projected

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return logsumexp(lam * corr * dx) / lam

def lse_gradient(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    scaled = lam * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)
    
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    # First term: A[m] = sum_j (1-h[j]) * w[m-j] (valid indices)
    # This is conv(1-h, w)
    A = np.convolve(one_minus_h, w, 'full')
    # A has length n + len(w) - 1 = n + 2n-1 - 1 = 3n-2
    # A[m] for m=0..n-1 gives us the first term
    g += A[0:n]
    
    # Second term: B[m] = sum_k w[k] * h[m+k] (valid)
    # B[m] = sum_j h[j] * w[j-m] (valid)
    # = correlate(w, h) ... let me think

    # I want sum_n h[n] * w[n-m]
    # Let's use: sum_n h[n] * w[n-m] = np.dot(h, w_shifted)
    # where w_shifted[n] = w[n-m]
    # This is the inner product of h with a shifted version of w
    # Alternatively: B[m] = sum_k w[k] * h[m+k]
    # For m=0: sum_k w[k] * h[k] (k from 0 to n-1)
    # For m=1: sum_k w[k] * h[k+1] (k from 0 to n-2)
    # etc.
    # This is like a cross-correlation
    # Let's just compute it directly for small n
    wlen = len(w)
    for m in range(n):
        # k ranges such that m+k < n, i.e., k < n-m
        max_k = min(n - m, wlen)
        if max_k > 0:
            g[m] -= np.dot(w[:max_k], h[m:m+max_k])
    
    g *= dx
    return g

Let me also think about the problem structure more. 

The objective is max_k c_k where c_k = sum_n h[n+k]*(1-h[n]).

Notice that sum_k c_k over all k is related to the total "mismatch". Let's compute:
sum_k c_k = sum_k sum_n h[n+k]*(1-h[n])

For each pair (i,j) with i≠j, this contributes h[i]*(1-h[j]) if j = i-k for some valid k, which is always the case in the cyclic sense. Actually, in the linear (non-cyclic) case with full correlation, each pair (i,j) with i>j contributes h[i]*(1-h[j]) to c_{i-j} and h[j]*(1-h[i]) to c_{j-i}.

So sum_k c_k = sum_{i>j} h[i]*(1-h[j]) + sum_{i<j} h[i]*(1-h[j])
= sum_{i≠j} h[i]*(1-h[j])
= sum_i h[i] * (n-1 - sum_{j≠i} h[j])
= sum_i h[i] * (n-1 - (sum_j h[j] - h[i]))
= sum_i h[i] * (n-1 - n/2 + h[i])
= sum_i h[i] * (n/2 - 1 + h[i])
= (n/2 - 1) * sum_i h[i] + sum_i h[i]^2
= (n/2 - 1) * n/2 + sum_i h[i]^2
= n(n/2 - 1)/2 + sum_i h[i]^2

So sum_k c_k = n(n-2)/4 + sum_i h[i]^2

This is minimized when sum_i h[i]^2 is minimized, which happens when all h[i] = 1/2 (by convexity of x^2). At h[i] = 1/2:
sum_k c_k = n(n-2)/4 + n*(1/4) = n(n-2+1)/4 = n(n-1)/4

And the average c_k = n(n-1)/4 / (2n-1) ≈ n/8

So the minimum possible max_k c_k is at least the average, which is about n/8.
C5 ≥ (n/8) * (2/n) = 1/4 = 0.25

So the theoretical lower bound is 0.25 (achieved when all h[i] = 1/2, but then max_k c_k = average c_k = n(n-1)/(4(2n-1)) * 2/n ≈ 1/4).

c_k = sum_n (1/2)(1-1/2) = (number of valid n) * 1/4

The number of valid n for lag k is n - |k|. So:
c_k = (n-|k|)/4

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = lam * corr * dx
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    w = exp_s / exp_s.sum()
    
    # Gradient via convolution
    ome_h = 1.0 - h
    # First term: conv(1-h, w)
    A = np.convolve(ome_h, w, 'full')[:n]
    # Second term: for each m, sum_k w[k]*h[m+k]
    # This is the "autocorrelation-like" term
    B = np.zeros(n)
    wlen = len(w)
    for m in range(n):
        mk = min(n - m, wlen)
        if mk > 0:
            B[m] = np.dot(w[:mk], h[m:m+mk])
    
    g = (A - B) * dx
    return g

    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def lse_c5(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(lam * corr * dx) / lam)

def lse_gradient(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = lam * corr * dx

    exp_s = np.exp(scaled - m)

    ome_h = 1.0 - h
    A = np.convolve(ome_h, w, 'full')[:n]

        mk = min(n - i, wlen)
        if mk > 0:
            B[i] = np.dot(w[:mk], h[i:i+mk])
    return (A - B) * dx
