import numpy as np
from scipy.optimize import minimize
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(z, target):
    n = len(z)
    # Bisection to find alpha such that sum(clip(z-alpha, 0, 1)) = target
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            rsum = room.sum()
            if rsum > 1e-15:
                h += d * room / rsum
        else:
            hsum = h.sum()
            if hsum > 1e-15:
                h += d * h / hsum
        h = np.clip(h, 0.0, 1.0)
    return h

def _lag_values(h):
    """Compute all values matching np.correlate(h, 1-h, mode='full')."""
    n = len(h)
    q = 1.0 - h
    nv = 2 * n - 1
    vals = np.empty(nv)
    for p in range(nv):
        d = p - (n - 1)
        if d >= 0:
            m = n - d
            vals[p] = np.dot(h[d:], q[:m])
        else:
            m = n + d
            vals[p] = np.dot(h[:m], q[-d:])
    return vals

def _smooth_obj_and_grad(x, args):
    tau, nval = args
    n = len(x)
    dx = 2.0 / n
    vals = _lag_values(x)
    a = vals * dx / tau
    amax = np.max(a)
    ea = np.exp(a - amax)
    Z = ea.sum()
    if Z < 1e-300:
        return float(np.max(vals) * dx), np.zeros(n)
    p = ea / Z
    obj = tau * (amax + np.log(Z))
    # Gradient: sum_l p[l] * dx * d(vals[l])/d(h)
    # For lag d = p_idx - (n-1):
    #   if d >= 0: pairs (i+d, i) for i=0..n-1-d
    #     d(val)/d(h[m]): if m in [d, n-1]: +(1-h[m-d]); if m in [0, n-1-d]: -h[m+d]
    #   if d < 0: pairs (i+d, i) for i=-d..n-1, i.e., (j, j-d) for j=0..n-1+d wait...
    # Let me just compute it directly per lag.
    q = 1.0 - x
    grad = np.zeros(n)
    for p_idx in range(nval):
        w = p[p_idx] * dx
        if w < 1e-18:
            continue
        d = p_idx - (n - 1)
        if d >= 0:
            m = n - d
            # pairs: (i+d, i) for i in 0..m-1
            js = np.arange(d, n)    # j indices
            is_ = np.arange(m)       # i indices
            # d(val)/d(h[j]) = (1-h[i]) = q[i]
            # d(val)/d(h[i]) = -h[j]
            grad[js] += w * q[is_]
            grad[is_] += -w * x[js]
        else:
            m = n + d  # = n - |d|
            # pairs: (i+d, i) for i in -d..n-1
            # Let t = -d > 0
            # i ranges t to n-1, j = i-t ranges 0 to n-1-t = m-1
            t = -d
            js = np.arange(m)        # j = 0..m-1
            is_ = np.arange(t, n)    # i = t..n-1
            grad[js] += w * q[is_]
            grad[is_] += -w * x[js]
    return float(obj), grad

def _obj(x, tau, nval):
    return _smooth_obj_and_grad(x, (tau, nval))[0]

def _grad(x, tau, nval):
    return _smooth_obj_and_grad(x, (tau, nval))[1]

def _eq_con(x, tgt):
    return x.sum() - tgt

def _eq_jac(x, tgt):
    return np.ones_like(x)

def _run_slsqp(h0, n, target, tau):
    nval = 2 * n - 1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (target,)}]
    res = minimize(
        _obj, h0, jac=_grad, args=(tau, nval),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    return res.x

def _tau_schedule():
    return [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]

def _optimize_start(h0, deadline):
    """Run SLSQP with tau schedule from h0."""
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_s = score_exact(h0)
    for tau in _tau_schedule():
        if time.time() > deadline:
            break
        try:
            x_new = _run_slsqp(best_h, n, target, tau)
            cand = project_box_sum(x_new, target)
            s = score_exact(cand)
            if s < best_s:
                best_s = s
                best_h = cand
        except Exception:
            pass
    return best_h, best_s

    target_i = int(round(target))
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    # Fix count
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target_i:
        for idx in ones[target_i:]:
            h[idx] = 0.0
    elif len(ones) < target_i:
        for idx in zeros[:target_i - len(ones)]:
            h[idx] = 1.0
    best_s = score_exact(h)
    best_h = h.copy()
    for it in range(iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # undo
    return best_h, best_s
