        else:
            idx = np.argsort(h)
            for i in idx:
                if h[i] > 0.0:
                    amt = min(-diff, h[i])
                    h[i] -= amt
                    diff += amt
                    if abs(diff) < 1e-14:
                        break
    return h


def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    # Interpolate (treating h as periodic over [0,2))
    h_new = np.interp(x_new % 2.0, np.sort(x_old), h_old[np.argsort(x_old)])
    return repair_feasible(h_new, n_new)


def get_top_lags(h, n, top_k=30):
    dx = 2.0 / n
    vals = np.zeros(2 * n - 1)
    for i, k in enumerate(range(-(n - 1), n)):
        vals[i] = overlap_for_lag(h, k) * dx
    top_indices = np.argsort(vals)[::-1][:top_k]
    lags = list(range(-(n - 1), n))
    return [lags[i] for i in top_indices]


def solve_epigraph(h_init, n, time_limit):
    """Solve the epigraph SLSQP problem with active set refinement."""
    start_time = time.perf_counter()
    h = repair_feasible(h_init.copy(), n)
    
    # Get initial active lags
    active_lags = get_top_lags(h, n, top_k=25)
    
    current_max = evaluate_c5(h)
    x0 = np.zeros(n + 1)
    x0[0] = current_max + 0.05
    x0[1:] = h
    
    bounds = [(-0.01, 2.5)] + [(0.0, 1.0)] * n
    
    best_local_c5 = current_max
    best_local_h = h.copy()
    
    for iteration in range(5):
        elapsed = time.perf_counter() - start_time
        if elapsed > time_limit * 0.85:
            break
        
        constraints = []
        for k in active_lags:
            constraints.append({
                'type': 'ineq',
                'fun': lag_constraint,
                'jac': lag_constraint_grad,
                'args': (n, k)
            })
        
        constraints.append({
            'type': 'eq',
            'fun': equality_constraint,
            'jac': equality_constraint_grad,
            'args': (n,)
        })
        
        try:
            result = minimize(
                epigraph_objective, x0.copy(),
                jac=epigraph_objective_grad,
                bounds=bounds,
                constraints=constraints,
                args=(n,),
                method='SLSQP',
                options={'maxiter': 300, 'ftol': 1e-12}
            )
        except Exception:
            break
        
        if result.x is None:
            break
        
        h_new = repair_feasible(result.x[1:], n)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_local_c5:
            best_local_c5 = c5_new
            best_local_h = h_new.copy()
        
        # Check for violated constraints
        dx = 2.0 / n
        tau_current = best_local_c5
        violators = []
        all_lags = list(range(-(n - 1), n))
        for k in all_lags:
            val = overlap_for_lag(best_local_h, k) * dx
            if val > tau_current + 1e-5:
                violators.append(k)
        
        if len(violators) == 0:
            # Tighten: check with smaller tolerance
            violators_tight = []
            for k in all_lags:
                val = overlap_for_lag(best_local_h, k) * dx
                if val > best_local_c5 + 1e-8:
                    violators_tight.append(k)
            if len(violators_tight) == 0:
                break
            violators = violators_tight
        
        # Add violators
        active_lags = list(set(active_lags + violators))
        # Keep top 50 by current overlap
        vals = np.array([overlap_for_lag(best_local_h, k) * dx for k in active_lags])
        top_idx = np.argsort(vals)[::-1][:50]
        active_lags = [active_lags[i] for i in top_idx]
        
        x0 = np.zeros(n + 1)
        x0[0] = best_local_c5 + 0.01
        x0[1:] = best_local_h
    
    return best_local_h, best_local_c5


def coordinate_descent(h_init, n, time_limit, step_size=0.05):
    """Simple coordinate descent to reduce max overlap."""
    start_time = time.perf_counter()
    h = repair_feasible(h_init.copy(), n)
    current_c5 = evaluate_c5(h)
    dx = 2.0 / n
    
    for step in range(2000):
        elapsed = time.perf_counter() - start_time
        if elapsed > time_limit:
            break
        
        improved = False
        for i in range(n):
            # Try perturbing h[i]
            best_delta = 0.0
            best_val = current_c5 - 1e-15
            
            for delta in [-step_size, -step_size/2, step_size/2, step_size]:
                new_val = h[i] + delta
                if new_val < 0.0 or new_val > 1.0:
                    continue
                h[i] = new_val
                # Adjust sum: redistribute to another element
                sum_diff = -delta
                # Find best redistribution target
                if sum_diff > 0:
                    j = int(np.argmax(1.0 - h))
                else:
                    j = int(np.argmax(h))
                if j == i:
                    h[i] = new_val - delta  # revert
                    continue
                h[j] = np.clip(h[j] + sum_diff, 0.0, 1.0)
                
                c5_try = evaluate_c5(h)
                if c5_try < best_val:
                    best_val = c5_try
                    best_delta = delta
                    
                # Revert
                h[j] = h[j] - sum_diff
                h[i] = h[i] - delta
            
            if best_delta != 0.0:
                h[i] += best_delta
                sum_diff = -best_delta
                if sum_diff > 0:
                    mask = h < 1.0
                    mask[i] = False
                    if np.sum(mask) > 0:
                        h[mask] += sum_diff / np.sum(mask)
                else:
                    mask = h > 0.0
                    mask[i] = False
                    if np.sum(mask) > 0:
                        h[mask] += sum_diff / np.sum(mask)
                h = repair_feasible(h, n)
                current_c5 = evaluate_c5(h)
                improved = True
        
        if not improved:
            break
    
    return h, current_c5


def push_binary(h, n):
    """Push values toward 0 or 1, maintaining feasibility."""
    h = h.copy()
    # Push values with |h - 0.5| > threshold toward nearest boundary
    for threshold in [0.3, 0.25, 0.2, 0.15]:
        for i in range(n):
            if h[i] > 0.5 + threshold:
                # Push toward 1
                old = h[i]
                h[i] = 1.0
                diff = 1.0 - old
                # Remove diff from other elements
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask = mask & (h > 0.0)
                if np.sum(mask) > 0:
                    h[mask] -= diff / np.sum(mask)
                    h[mask] = np.maximum(h[mask], 0.0)
            elif h[i] < 0.5 - threshold:
                # Push toward 0
                old = h[i]
                h[i] = 0.0
                diff = old
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask = mask & (h < 1.0)
                if np.sum(mask) > 0:
                    h[mask] += diff / np.sum(mask)
                    h[mask] = np.minimum(h[mask], 1.0)
        h = repair_feasible(h, n)
    return h


def make_random_feasible(n, rng):
    """Create a random feasible vector."""
    h = rng.random(n)
    # Make it more binary
    h[h > 0.7] = 1.0
    h[h < 0.3] = 0.0
    h = repair_feasible(h, n)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    if budget_s is None:
        budget_s = 1100
    end_time = start_time + budget_s * 0.88
    
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
        h_parent = repair_feasible(h_parent, n_parent)
        c5_parent = evaluate_c5(h_parent)
        best_h = h_parent.copy()
        best_c5 = c5_parent
        best_n = n_parent
    else:
        n_parent = 45
        h_parent = np.full(n_parent, 0.5)
        h_parent = repair_feasible(h_parent, n_parent)
        best_h = h_parent.copy()
        best_c5 = evaluate_c5(h_parent)
        best_n = n_parent
    
    def update_best(h_new, c5_new, n_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = n_new
    
    # Phase 1: Optimize at parent n
    n_candidates = sorted(set([n_parent, 41, 45, 49, 53, 57, 61, 65]))
    
    for n in n_candidates:
        if time.perf_counter() > end_time:
            break
        
        # Get starting vector
        if n == best_n:
            h_start = best_h.copy()
        elif n == n_parent:
            h_start = h_parent.copy()
        else:
            h_start = transfer_to_n(best_h, n)
        
        time_left_total = end_time - time.perf_counter()
        if time_left_total < 10:
            break
        
        # Run SLSQP optimization
        t_lim = min(80.0, time_left_total * 0.5)
        h_opt, c5_opt = solve_epigraph(h_start, n, t_lim)
        update_best(h_opt, c5_opt, n)
        
        # Try push binary then re-optimize
        if time.perf_counter() < end_time:
            h_bin = push_binary(h_opt, n)
            c5_bin = evaluate_c5(h_bin)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin
                best_n = n
            # Re-optimize from binary start
            t_lim2 = min(40.0, (end_time - time.perf_counter()) * 0.4)
            if t_lim2 > 10:
                h_opt2, c5_opt2 = solve_epigraph(h_bin, n, t_lim2)
                update_best(h_opt2, c5_opt2, n)
        
        # Coordinate descent polish
        if time.perf_counter() < end_time:
            t_lim3 = min(30.0, (end_time - time.perf_counter()) * 0.3)
            if t_lim3 > 5:
                h_cd, c5_cd = coordinate_descent(best_h, best_n, t_lim3)
                update_best(h_cd, c5_cd, best_n)
    
    # Phase 2: More aggressive search with different n
    if time.perf_counter() < end_time:
        for n in [50, 55, 60, 65, 70, 75, 80, 85, 90]:
            if time.perf_counter() > end_time:
                break
            if n == best_n or n == n_parent:
                continue
            
            time_left = end_time - time.perf_counter()
            if time_left < 15:
                break
            
            h_start = transfer_to_n(best_h, n)
            t_lim = min(50.0, time_left * 0.6)
            h_opt, c5_opt = solve_epigraph(h_start, n, t_lim)
            update_best(h_opt, c5_opt, n)
            
            # Binary push
            if time.perf_counter() < end_time:
                h_bin = push_binary(h_opt, n)
                c5_bin = evaluate_c5(h_bin)
                if c5_bin < best_c5:
                    best_h = h_bin.copy()
                    best_c5 = c5_bin
                    best_n = n
    
    # Phase 3: Final polishing with coordinate descent
    if time.perf_counter() < end_time:
        time_left = end_time - time.perf_counter()
        if time_left > 5:
            h_cd, c5_cd = coordinate_descent(best_h, best_n, time_left * 0.7)
            update_best(h_cd, c5_cd, best_n)
    
    # Final validation
    best_h = repair_feasible(best_h, best_n)
    final_c5 = evaluate_c5(best_h)
    best_c5 = final_c5
    
    return (best_h, best_c5, best_n)
