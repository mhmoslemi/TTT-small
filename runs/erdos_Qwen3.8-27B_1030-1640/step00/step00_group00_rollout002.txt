def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0

        mid = (lo + hi) / 2.0

            lo = mid
        else:

    """Returns the 2n-1 correlation values (times dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_soft_grad(h, theta):
    """Compute gradient of soft-max surrogate."""
    n = len(h)
    dx = 2.0 / n
    G = compute_correlation(h)  # shape (2n-1,)
    Gmax = np.max(G)
    if not np.isfinite(Gmax):
        return np.inf, np.zeros(n)
    
    exp_G = np.exp((G - Gmax) / theta)
    w = exp_G / np.sum(exp_G)
    soft_obj = Gmax + theta * np.log(np.sum(exp_G))
    
    offset = n - 1
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue

        # d(correlation[m])/dh[j] = (1-h[j-k]) * [0<=j-k<n] - h[j+k] * [0<=j+k<n]
        
        # First term: (1-h[j-k]) for j where 0<=j-k<n
        if k >= 0:
            j_lo, j_hi = k, n
        else:
            j_lo, j_hi = 0, n + k
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi) - k
            grad[j_lo:j_hi] += wm * (1.0 - h[idx])

        h_tmp[s] -= tau
        h_tmp[t] += tau
        return evaluate_C(h_tmp)

    c2 = a + invphi * (b - a)

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y, target):
    n = len(y)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def compute_soft_obj_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    G = corr * dx
    Gmax = np.max(G)
    if not np.isfinite(Gmax):
        return np.inf, np.zeros(n)
    exp_G = np.exp((G - Gmax) / theta)
    w = exp_G / np.sum(exp_G)
    soft_obj = Gmax + theta * np.log(np.sum(exp_G))
    offset = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - offset
        if k >= 0:
            j_lo1, j_hi1 = k, n
        else:
            j_lo1, j_hi1 = 0, n + k
        if j_lo1 < j_hi1:
            idx = np.arange(j_lo1, j_hi1) - k
            grad[j_lo1:j_hi1] += wm * (1.0 - h[idx])
        if k >= 0:
            j_lo2, j_hi2 = 0, n - k
        else:
            j_lo2, j_hi2 = -k, n
        if j_lo2 < j_hi2:
            idx2 = np.arange(j_lo2, j_hi2) + k
            grad[j_lo2:j_hi2] += wm * (-h[idx2])
    return soft_obj, grad

def golden_pair(h, s, t, max_iter=80):
    T = min(h[s], 1.0 - h[t])
    if T < 1e-14:
        return h, evaluate_C(h)
    c_cur = evaluate_C(h)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T
    def ev(tau):
        ht = h.copy()
        ht[s] -= tau
        ht[t] += tau
        return evaluate_C(ht)
    c1 = b - invphi * (b - a)
    c2 = a + invphi * (b - a)
    f1 = ev(c1)
    f2 = ev(c2)
    best_val = min(f1, f2, c_cur)
    best_tau = 0.0
    if f1 < best_val:
        best_val, best_tau = f1, c1
    if f2 < best_val:
        best_val, best_tau = f2, c2
    for _ in range(max_iter):
        if b - a < 1e-13 * (T + 1.0):
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - invphi * (b - a); f1 = ev(c1)
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + invphi * (b - a); f2 = ev(c2)
        if f1 < best_val:
            best_val, best_tau = f1, c1
        if f2 < best_val:
            best_val, best_tau = f2, c2
    f_end = ev(T)
    if f_end < best_val:
        best_val, best_tau = f_end, T
    if best_tau > 1e-14 and best_val < c_cur - 1e-13:
        hn = h.copy()
        hn[s] -= best_tau
        hn[t] += best_tau
        return hn, best_val
    return h, c_cur

def smooth_descent(h, theta_schedule, max_iters=300):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c = evaluate_C(h_cur)
    for theta in theta_schedule:
        for it in range(max_iters):
            obj, grad = compute_soft_obj_grad(h_cur, theta)
            if not np.isfinite(obj):
                break
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-14:
                break
            alpha = 0.1 / (gnorm + 1e-10)
            accepted = False
            for _ in range(40):
                y = h_cur - alpha * grad
                y_proj = project_h(y, target)
                c_t = evaluate_C(y_proj)
                if c_t < best_c + theta * 0.5:
                    h_cur = y_proj
                    if c_t < best_c - 1e-13:
                        best_c = c_t
                        best_h = h_cur.copy()
                    if c_t < best_c:
                        best_c = c_t
                        best_h = h_cur.copy()
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-15:
                    break
            if not accepted:
                break
    return best_h, best_c

def pair_polish(h, rng, n_pairs, max_iter=60, theta=3e-4):
    n = len(h)
    c_cur = evaluate_C(h)
    h_cur = h.copy()
    for it in range(n_pairs):
        obj, grad = compute_soft_obj_grad(h_cur, theta)
        if not np.isfinite(obj):
            break
        valid_s = np.where(h_cur > 1e-8)[0]
        valid_t = np.where(h_cur < 1.0 - 1e-8)[0]
        if len(valid_s) == 0 or len(valid_t) == 0:
            break
        best_impr = 0.0
        best_new = None
        n_try = min(10, len(valid_s) * len(valid_t))
        for _ in range(n_try):
            r = rng.random()
            if r < 0.5:
                s = valid_s[rng.integers(len(valid_s))]
                t = valid_t[rng.integers(len(valid_t))]
            elif r < 0.8:
                s = valid_s[np.argmax(grad[valid_s])]
                t = valid_t[np.argmin(grad[valid_t])]
            else:
                s_candidates = valid_s[:min(10, len(valid_s))]
                t_candidates = valid_t[:min(10, len(valid_t))]
                s = s_candidates[rng.integers(len(s_candidates))]
                t = t_candidates[rng.integers(len(t_candidates))]
            if s == t:
                continue
            hn, cn = golden_pair(h_cur, s, t, max_iter)
            if cn < c_cur - 1e-13:
                if c_cur - cn > best_impr:
                    best_impr = c_cur - cn
                    best_new = (hn, cn)
        if best_new is not None:
            h_cur, c_cur = best_new
    return h_cur, c_cur

    h = np.full(n, 0.5)
    freqs = rng.integers(1, 8, size=3)
