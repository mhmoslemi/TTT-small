def smoothed_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx  # length 2n-1
    
    # Softmax weights
    max_val = np.max(corr)
    exp_vals = np.exp((corr - max_val) / theta)
    w = exp_vals / np.sum(exp_vals)  # length 2n-1, w[i] corresponds to lag i-(n-1)
    
    grad = np.zeros(n)
    for i in range(2*n - 1):
        L = i - (n - 1)  # this is the lag
        wi = w[i] * dx
        if wi < 1e-15:
            continue
        # dG_L/dh[m] = wi * [(1-h[m+L]) if m+L in [0,n-1] - h[m-L] if m-L in [0,n-1]]
        if L >= 0:
            # m+L valid: m in [0, n-1-L]
            # m-L valid: m in [L, n-1]
            m1_start, m1_end = 0, n - L  # m in [0, n-1-L], i.e., indices 0 to n-1-L
            m2_start, m2_end = L, n  # m in [L, n-1], i.e., indices L to n-1

def pair_line_search(h, s, t, max_iters=50):

    T = min(h[s], 1.0 - h[t])

    d = a + gr * (b - a)
    fc = pair_eval(h, s, t, c)
    fd = pair_eval(h, s, t, d)

    h_new[t] += tau_best

    if c_T <= c_best and c_T <= c_0:

import numpy as np

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_h(y):
    n = len(y)
    target = n / 2.0
    ymin = float(np.min(y))
    ymax = float(np.max(y))
    lo = min(ymin, 0.0) - 1.0
    hi = max(ymax, 1.0) + 1.0
    for _ in range(30):
        if np.sum(np.clip(y + lo, 0.0, 1.0)) <= target:
            break
        lo -= 1.0
    for _ in range(30):
        if np.sum(np.clip(y + hi, 0.0, 1.0)) >= target:
            break
        hi += 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def pair_line_search(h, s, t, max_iters=30):
    T = min(h[s], 1.0 - h[t])
    if T < 1e-13:
        return h, evaluate_C(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hc = h.copy(); hc[s] -= c; hc[t] += c
    fc = evaluate_C(hc)
    hd = h.copy(); hd[s] -= d; hd[t] += d

def pair_pass_first(h, current_c, deadline, max_iters=30):
    n = len(h)
    for s in range(n):
        if h[s] < 1e-10:
            continue
        for t in range(n):
            if t == s or h[t] > 1.0 - 1e-10:
                continue
            if time.time() > deadline:
                return h, current_c
            h_new, c_new = pair_line_search(h, s, t, max_iters=max_iters)
            if c_new < current_c - 1e-13:
                return h_new, c_new
    return h, current_c

def pair_pass_rand(h, current_c, deadline, rng, max_iters=30, max_tries=500):

def pair_refine(h, deadline, rng, max_iters=30):

    current_c = evaluate_C(h)
    stuck = 0

def compute_grad_sm(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    exps = np.exp((corr - max_c) / theta)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        L = i - (n - 1)
        wi = w[i] * dx
        if wi < 1e-16:
            continue
        if L >= 0:
            me = n - L
            grad[0:me] += wi * (1.0 - h[L:L + me])
            grad[L:n] -= wi * h[0:me]
        else:
            p = -L
            grad[p:n] += wi * (1.0 - h[0:n - p])
            grad[0:n - p] -= wi * h[p:n]
    return w, grad, corr
