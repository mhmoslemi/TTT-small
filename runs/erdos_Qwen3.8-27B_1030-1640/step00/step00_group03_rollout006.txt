If l < 0: corr = sum_{i=0}^{n-1-|l|} h[i+|l|]*(1-h[i])

- If h[i] is low and h[i+l] is high: increase h[i] or reduce h[i+l]

3. For each conflicting pair (i, i+l*), transfer a small amount from the higher to the lower

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_feasible(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_obj_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    max_g = np.max(g)
    shifted = np.exp(lam * (g - max_g))
    sum_exp = np.sum(shifted)
    weights = shifted / sum_exp
    obj = max_g + np.log(sum_exp) / lam
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        l = idx - (n - 1)
        if l == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif l > 0:
            grad[:n-l] += w * dx * (1.0 - h[l:])
            grad[l:] += w * dx * (-h[:n-l])
        else:
            m = -l
            grad[m:] += w * dx * (1.0 - h[:n-m])
            grad[:n-m] += w * dx * (-h[m:])
    return obj, grad

def _c_fun(h, ts):
    return np.sum(h) - ts

def _c_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, lam, maxiter=200):
    ts = n * 0.5
    bv = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac, 'args': (ts, n)}]
    try:
        res = minimize(lse_obj_grad, h0, args=(lam, n),
                      method='SLSQP', bounds=bv, constraints=cs,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h0.copy()

def subgrad_topk(h, n, step_size, topk=3):
    dx = 2.0 / n
    ts = n * 0.5
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    top_indices = np.argsort(v)[-topk:]
    grad = np.zeros(n)
    for idx in top_indices:
        l = idx - (n - 1)
        if l == 0:
            grad += dx * (1.0 - 2.0 * h) / topk
        elif l > 0:
            grad[:n-l] += dx * (1.0 - h[l:]) / topk
            grad[l:] -= dx * h[:n-l] / topk
        else:
            m = -l
            grad[m:] += dx * (1.0 - h[:n-m]) / topk
            grad[:n-m] -= dx * h[m:] / topk
    h_new = h - step_size * grad
    return project_feasible(h_new, ts)

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)
