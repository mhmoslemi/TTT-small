import numpy as np
import time
import math

def overlap_vector(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')

def c5_value(h):
    return float(np.max(overlap_vector(h)))

def make_feasible(h, n):
    """Repair h to be feasible: 0<=h<=1, sum(h)=n/2"""
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Water-filling to fix sum
    for _ in range(100):
        s = np.sum(h)
        delta = target - s
        if abs(delta) < 1e-12:
            break
        if delta > 0:
            # Need to add mass; find coordinates with headroom
            headroom = 1.0 - h
            mask = headroom > 1e-15
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            add = np.min(delta, np.sum(headroom[idx]))
            # Distribute proportionally to headroom
            frac = headroom[idx] / np.sum(headroom[idx])
            h[idx] += add * frac
        else:
            # Need to remove mass
            mask = h > 1e-15
            if not np.any(mask):
                break
            idx = np.where(mask)[0]
            rem = np.min(-delta, np.sum(h[idx]))
            frac = h[idx] / np.sum(h[idx])
            h[idx] -= rem * frac
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Spread residual
        residual = target - s
        n_pos = np.sum((1.0 - h) > 1e-15) if residual > 0 else np.sum(h > 1e-15)
        if n_pos > 0:
            if residual > 0:

                h[idx] += residual / len(idx)
            else:
                idx = np.where(h > 1e-15)[0]
                h[idx] += residual / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def feasible_interval(h, d):
    """Compute [t_lo, t_hi] such that h + t*d is in [0,1]^n for all t in interval."""
    n = len(h)
    t_lo = 0.0
    t_hi = 0.0
    
    pos = d > 1e-15
    neg = d < -1e-15
    
    if np.any(pos):
        # h_i + t*d_i <= 1 → t <= (1-h_i)/d_i
        ub = (1.0 - h[pos]) / d[pos]
        t_hi = min(t_hi, float(np.min(ub)))
        # h_i + t*d_i >= 0 → t >= -h_i/d_i
        lb = -h[pos] / d[pos]
        t_lo = max(t_lo, float(np.max(lb)))
    
    if np.any(neg):
        # h_i + t*d_i >= 0 → t*d_i >= -h_i → t <= -h_i/d_i (d_i<0, so this is positive)
        ub = -h[neg] / d[neg]  # = h[neg]/|d[neg]|
        t_hi = min(t_hi, float(np.min(ub)))
        # h_i + t*d_i <= 1 → t*d_i <= 1-h_i → t >= (1-h_i)/d_i (d_i<0, so this is negative)
        lb = (1.0 - h[neg]) / d[neg]  # = -(1-h[neg])/|d[neg]|
        t_lo = max(t_lo, float(np.max(lb)))
    
    # Clamp to ensure 0 is in interval (should be, but numerical safety)
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    
    return t_lo, t_hi

def quadratic_coefficients(y0, y1, y2, t1, t2):
    """Given f(0)=y0, f(t1)=y1, f(t2)=y2, return A, B, Q for f(t)=A+B*t+Q*t^2"""
    D1 = y1 - y0
    D2 = y2 - y0
    denom = t1 * t2 * (t1 - t2)
    if abs(denom) < 1e-30:
        return None, None, None
    Q = (D1 * t2 - D2 * t1) / denom
    B = D1 / t1 - Q * t1
    A = y0
    return A, B, Q

def active_gradient(h, cvals, tol_frac=0.005):
    """Compute gradient of the active (max) overlap components, projected to sum-zero."""
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    cmax = np.max(cvals)
    
    # Active set
    threshold = cmax - tol_frac * cmax
    active = cvals >= threshold
    if not np.any(active):
        active = np.zeros(L, dtype=bool)
        active[np.argmax(cvals)] = True
    
    alpha = np.zeros(L)
    alpha[active] = 1.0 / np.sum(active)
    
    # Compute gradient: for each overlap component m,
    # f_m = dx * sum_{i in S_m} h[i] * (1 - h[j(i,m)])
    # where j(i,m) = i + n - 1 - m
    # df_m/dh[k] = dx * (sum_{i: i=k, k in S_m} (1-h[j]) + sum_{j: j=k, i in S_m} (-h[i]))
    
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo > hi:
            continue
        sources = np.arange(lo, hi + 1)
        targets = sources + n - 1 - m
        a = alpha[m]
        # Source contribution: +dx * (1-h[target])
        grad[sources] += dx * a * (1.0 - h[targets])
        # Target contribution: -dx * h[source]
        grad[targets] -= dx * a * h[sources]
    
    # Project to sum-zero
    grad = grad - np.mean(grad)
    return grad

def smooth_gradient(h, cvals, tau):
    """Compute smoothed gradient using softmax weights."""
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    cmax = np.max(cvals)
    
    z = (cvals - cmax) / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    alpha = exp_z / np.sum(exp_z)
    
    grad = np.zeros(n)
    for m in range(L):
        if alpha[m] < 1e-10:
            continue
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo > hi:
            continue
        sources = np.arange(lo, hi + 1)
        targets = sources + n - 1 - m
        a = alpha[m]
        grad[sources] += dx * a * (1.0 - h[targets])
        grad[targets] -= dx * a * h[sources]
    
    grad = grad - np.mean(grad)
    return grad

def random_direction(n, rng):
    """Generate a random sum-zero direction."""
    d = rng.standard_normal(n)
    d -= np.mean(d)
    return d

def pair_direction(i, j, n):
    """Create direction e_j - e_i (transfer from i to j)."""
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d

        hi = min(m, n - 1)
        if lo > hi:

        targets = sources + n - 1 - m
        a = alpha[m]
        grad[sources] += dx * a * (1.0 - h[targets])
        grad[targets] -= dx * a * h[sources]
    grad = grad - np.mean(grad)
    return grad

            n = len(current_h)

        cmax = current_c5
        if cmax < 1e-6:
            break

        d = active_gradient(current_h, current_cvals, tol_frac=0.003)
        if np.max(np.abs(d)) > 1e-12:
            new_h, new_c5 = line_search(current_h, current_cvals, current_c5, d)
            if new_c5 < current_c5 - 1e-13:
                current_h = new_h
                current_c5 = new_c5
                current_cvals = overlap_vector(current_h)
                improved = True

        if time.monotonic() < eff_deadline:
            for tau_frac in [0.008, 0.025, 0.08]:
                if time.monotonic() > eff_deadline:
                    break
                tau = tau_frac * cmax
                d = smooth_gradient(current_h, current_cvals, tau)
                if np.max(np.abs(d)) > 1e-12:
                    new_h, new_c5 = line_search(current_h, current_cvals, current_c5, d)
                    if new_c5 < current_c5 - 1e-13:
                        current_h = new_h
                        current_c5 = new_c5
                        current_cvals = overlap_vector(current_h)
                        improved = True

        if time.monotonic() < eff_deadline and n <= 120:
            grad = active_gradient(current_h, current_cvals, tol_frac=0.01)
            order = np.argsort(-np.abs(grad))
            top_k = min(16, n)
            tried = 0
            found_pair = False
            for ai in range(top_k):
                if found_pair or time.monotonic() > eff_deadline:
                    break
                for bi in range(top_k):
                    if ai == bi or tried >= 256:
                        continue
                    i_idx = int(order[ai])
                    j_idx = int(order[bi])
                    d = pair_direction(i_idx, j_idx, n)
                    new_h, new_c5 = line_search(current_h, current_cvals, current_c5, d)
                    tried += 1
                    if new_c5 < current_c5 - 1e-13:
                        current_h = new_h
                        current_c5 = new_c5
                        current_cvals = overlap_vector(current_h)
                        improved = True
                        found_pair = True
                        break

        if not improved and time.monotonic() < eff_deadline:
            for _ in range(4):
                if time.monotonic() > eff_deadline:
                    break
                d = random_direction(n, rng)
                new_h, new_c5 = line_search(current_h, current_cvals, current_c5, d)
                if new_c5 < current_c5 - 1e-13:
                    current_h = new_h
                    current_c5 = new_c5
                    current_cvals = overlap_vector(current_h)
                    improved = True
                    break

        if not improved and time.monotonic() < eff_deadline:
            n_block = min(n, max(3, n // 4))
            for _ in range(2):
                if time.monotonic() > eff_deadline:
                    break
                block = rng.integers(0, n, size=n_block, dtype=int)
                block = np.unique(block)
                if len(block) < 2:
                    continue
                d = np.zeros(n)
                d[block] = rng.standard_normal(len(block))
                d -= np.mean(d)
                new_h, new_c5 = line_search(current_h, current_cvals, current_c5, d)
                if new_c5 < current_c5 - 1e-13:
                    current_h = new_h
                    current_c5 = new_c5
                    current_cvals = overlap_vector(current_h)
                    improved = True
                    break

        if current_c5 < best_c5 - 1e-13:
            best_h = current_h.copy()
            best_c5 = current_c5
            n = len(best_h)
            stall_count = 0
        elif improved:
            stall_count = 0
        else:
            stall_count += 1

        if stall_count >= max_stall:
            if time.monotonic() > eff_deadline:
                break
            current_h = best_h.copy()
            current_c5 = best_c5
            current_cvals = overlap_vector(current_h)
            n = len(current_h)
            stall_count = 0

            found_escape = False
            for attempt in range(15):
                if time.monotonic() > eff_deadline:
                    break
                d = random_direction(n, rng)
                d = d * 0.5 + 0.5 * active_gradient(current_h, current_cvals, 0.02)
                d -= np.mean(d)
                new_h, new_c5 = line_search(current_h, current_cvals, current_c5, d)
                if new_c5 < current_c5 - 1e-13:
                    current_h = new_h
                    current_c5 = new_c5
                    current_cvals = overlap_vector(current_h)
                    found_escape = True
                    break

            if not found_escape and time.monotonic() < eff_deadline - 60:
                dim_options = [n - 3, n - 2, n - 1, n + 1, n + 2, n + 3, n + 5, n - 5]
                rng.shuffle(dim_options)
                for new_n in dim_options:
                    if new_n < 20 or new_n > 150 or new_n == n:
                        continue
                    if time.monotonic() > eff_deadline - 30:
                        break
                    h_dim = transfer_to_dim(best_h, new_n)
                    c_dim = c5_value(h_dim)
                    cur_dim_h = h_dim.copy()
                    cur_dim_c5 = c_dim
                    cur_dim_cvals = overlap_vector(cur_dim_h)
                    for _ in range(15):
                        if time.monotonic() > eff_deadline - 30:
                            break
                        d = active_gradient(cur_dim_h, cur_dim_cvals, tol_frac=0.005)
                        if np.max(np.abs(d)) > 1e-12:
                            nh, nc = line_search(cur_dim_h, cur_dim_cvals, cur_dim_c5, d)
                            if nc < cur_dim_c5 - 1e-13:
                                cur_dim_h = nh
                                cur_dim_c5 = nc
                                cur_dim_cvals = overlap_vector(cur_dim_h)
                        d2 = random_direction(new_n, rng)
                        nh, nc = line_search(cur_dim_h, cur_dim_cvals, cur_dim_c5, d2)
                        if nc < cur_dim_c5 - 1e-13:
                            cur_dim_h = nh
                            cur_dim_c5 = nc
                            cur_dim_cvals = overlap_vector(cur_dim_h)
                    if cur_dim_c5 < best_c5 - 1e-13:
                        best_h = cur_dim_h
                        best_c5 = cur_dim_c5
                        current_h = cur_dim_h.copy()
                        current_c5 = cur_dim_c5
                        current_cvals = cur_dim_cvals
                        n = new_n
                        break

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = c5_value(best_h)
    final_n = len(best_h)
    return (best_h, final_c5, final_n)
