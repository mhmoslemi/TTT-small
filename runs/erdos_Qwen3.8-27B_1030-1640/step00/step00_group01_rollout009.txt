def project_to_constraint(h, s):

    n = len(h)
    lo = h.min() - s / n - 1
    hi = h.max() - s / n + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        if x.sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max_value(corr, dx, temp):
    vals = corr * dx
    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp)))

def smooth_max_grad(h, n, dx, temp):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    exp_c = np.exp((corr - m) / temp)
    p = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        r = k - (n - 1)
        # (1-h[i+r]) for valid i
        i_start = max(0, -r)
        i_end = min(n, n - r)
        if i_start < i_end:
            grad[i_start:i_end] += p[k] * (1 - h[i_start + r:i_end + r])
        # -h[i-r] for valid i
        i_start2 = max(0, r)
        i_end2 = min(n, n + r)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= p[k] * h[i_start2 - r:i_end2 - r]
    return grad * dx

def projected_gradient_descent(h, n, dx, target_sum, temp=0.01, lr=0.01, max_iter=5000):

    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    for it in range(max_iter):
        val = smooth_max_value(np.correlate(h, 1-h, mode="full"), dx, temp)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        grad = smooth_max_grad(h, n, dx, temp)
        # Gradient step (in the direction that decreases the objective)
        h_new = h - lr * grad
        # Project
        h_new = project_to_constraint(h_new, target_sum)
        
        # Check if we improved
        new_val = smooth_max_value(np.correlate(h_new, 1-h_new, mode="full"), dx, temp)
        if new_val < val:
            h = h_new
        else:
            # Halve learning rate
            lr *= 0.5
            h_new = project_to_constraint(h - lr * grad, target_sum)
            h = h_new
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, s):
    n = len(h)
    lo = h.min() - s / n - 1.0
    hi = h.max() - s / n + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, dx, temp):
    vals = corr * dx
    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp)))

def smooth_max_grad(h, n, dx, temp):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    exp_c = np.exp((corr - m) / temp)
    p = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        r = k - (n - 1)
        i1 = max(0, -r)
        i2 = min(n, n - r)
        if i1 < i2:
            grad[i1:i2] += p[k] * (1.0 - h[i1 + r:i2 + r])
        j1 = max(0, r)
        j2 = min(n, n + r)
        if j1 < j2:
            grad[j1:j2] -= p[k] * h[j1 - r:j2 - r]
    return grad * dx

def pgd_optimize(h, n, dx, target_sum, temps, max_iter=3000, init_lr=0.5):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = init_lr
    for temp in temps:
        stall = 0
        for it in range(max_iter):
            val = smooth_max_val(np.correlate(h, 1 - h, mode="full"), dx, temp)
            if val < best_c5 - 1e-9:
                best_c5 = val
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            grad = smooth_max_grad(h, n, dx, temp)
            improved = False
            for _ in range(25):
                h_new = project_to_constraint(h - lr * grad, target_sum)
                v_new = smooth_max_val(np.correlate(h_new, 1 - h_new, mode="full"), dx, temp)
                if v_new < val:
                    h = h_new
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                lr = max(lr, 1e-6)
            lr = min(lr * 1.05, init_lr)
            if stall > 500:
                break
    return best_h, best_c5

def binary_local_search(h, n, dx, target_sum, rng, max_iter=5000, timeout=60):
    import time
    t0 = time.time()
    h = h.copy()
    c5 = compute_c5(h, n)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_c5 = np.max(corr)
    for it in range(max_iter):
        if time.time() - t0 > timeout:
            break
        if max_c5 <= 0:
            break
        # Find positions with 1 and 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0
        best_i, best_j = -1, -1
        # Try random subset for speed
        n_try = min(len(ones) * len(zeros), 500)
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            corr_new = np.correlate(h, 1 - h, mode="full") * dx
            mc = np.max(corr_new)
            if mc < max_c5 + best_delta - 1e-12:
                best_delta = mc - max_c5
                best_i, best_j = i, j
            h[i] = 1
            h[j] = 0
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1 - h, mode="full") * dx
            max_c5 = np.max(corr)
        else:
            # Try all moves (more exhaustive)
            found = False
            for i in ones:
                if time.time() - t0 > timeout:
                    break
                for j in zeros:
                    h[i] = 0
                    h[j] = 1
                    corr_new = np.correlate(h, 1 - h, mode="full") * dx
                    mc = np.max(corr_new)
                    if mc < max_c5 - 1e-12:
                        best_i, best_j = i, j
                        max_c5 = mc
                        found = True
                        h[i] = 1
                        h[j] = 0
                        break
                    h[i] = 1
                    h[j] = 0
                if found:
                    break
            if found:
                h[best_i] = 0
                h[best_j] = 1
            else:
                # Perturb: random swap to escape local optimum
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = np.correlate(h, 1 - h, mode="full") * dx
                max_c5 = np.max(corr)
    return h, compute_c5(h, n)

def continuous_local_search(h, n, dx, target_sum, rng, max_iter=10000, timeout=60):
    import time
    t0 = time.time()
    h = h.copy()
    c5 = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() - t0 > timeout:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        c5_new = compute_c5(h_new, n)
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
        else:
            # Small random perturbation to avoid stagnation
            idx = rng.randint(n)
            h[idx] += rng.normal(0, 0.01)
            h = project_to_constraint(h, target_sum)
    return h, c5
