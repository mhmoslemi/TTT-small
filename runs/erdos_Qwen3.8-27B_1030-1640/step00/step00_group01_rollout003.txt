∂c[k]/∂h[m]:
- If m = i for some valid i: contributes -h[i+delta] (from the (1-h[i]) term)
- If m = i+delta for some valid i, i.e., i = m-delta: contributes (1-h[m-delta]) (from the h[i+delta] term)

So:
∂c[k]/∂h[m] = (1 - h[m-delta]) * I(m-delta valid) - h[m+delta] * I(m+delta valid)

where "valid" means 0 ≤ index < n.

And the weighted gradient is:
∂f/∂h[m] = sum_k weights[k] * [(1 - h[m-delta_k]) * I(m-delta_k valid) - h[m+delta_k] * I(m+delta_k valid)]

where delta_k = k - (n-1).

Let me code this up efficiently. For each k, delta_k = k - (n-1). For a fixed m:

- Second term: -h[m + delta_k] is valid when 0 ≤ m + delta_k < n, i.e., -delta_k ≤ m < n - delta_k

    return np.sum(h) - _S_global

def _con_jac(h):
    return np.ones(len(h))

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, S):
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = float(np.max(corr))
    shifted = T * (corr - c_max)
    shifted = np.where(shifted < -50, -50.0, shifted)
    return c_max + (1.0 / T) * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad(h, n, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = float(np.max(corr))
    shifted = T * (corr - c_max)
    shifted = np.where(shifted < -50, -50.0, shifted)
    ev = np.exp(shifted)
    w = ev / float(ev.sum())
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-9:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        wk = w[k] * dx
        for i in range(i_lo, i_hi):
            j = i + d
            grad[j] += wk * (1.0 - h[i])
            grad[i] -= wk * h[j]
    return grad

def subgrad_step(h, n, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n, n - d)
    for i in range(i_lo, i_hi):
        j = i + d
        grad[j] += dx * (1.0 - h[i])
        grad[i] -= dx * h[j]
    return h - alpha * grad

def _slsqp_opt(hc, n, S, dx, T, maxiter):
    from scipy.optimize import minimize
    _S_global = S
    def _obj(hx):
        return smooth_max_obj(hx, dx, T)
    def _grad(hx):
        return smooth_max_grad(hx, n, dx, T)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj, hc, jac=_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res

def run(seed=42, budget_s=1100, **kwargs):
    global _S_global
    t0 = _time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)

    best_h = project(np.clip(h_init, 0.0, 1.0), n0 / 2.0)
    best_c5 = eval_c5(best_h)
    best_n = n0

    def time_ok():
        return (_time.time() - t0) < float(budget_s) - 25.0

    def try_optimize(hc, n, S, dx, t_list, maxiter):
        nonlocal best_h, best_c5, best_n
        for T in t_list:
            if not time_ok():
                return
            try:
                res = _slsqp_opt(hc.copy(), n, S, dx, T, maxiter)
                hn = project(res.x, S)
                c5n = eval_c5(hn)
                if c5n < best_c5:
                    best_h = hn
                    best_c5 = c5n
                    best_n = n
            except Exception:
                pass

    # Phase 1: Optimize at initial n with annealing
    try_optimize(best_h.copy(), best_n, best_n / 2.0, 2.0 / best_n,
                 [8.0, 15.0, 30.0, 60.0, 120.0, 250.0, 500.0], 400)

    if not time_ok():
        return best_h, best_c5, len(best_h)

    # Phase 2: Try nearby n values
    n_candidates = list(range(max(4, n0 - 8), n0 + 13))
    np.random.shuffle(n_candidates)
    for n_try in n_candidates:
        if not time_ok():
            break
        S_try = n_try / 2.0
        dx_try = 2.0 / n_try
        old_n = len(best_h)
        old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
        new_x = (np.arange(n_try) + 0.5) * dx_try
        hc = np.interp(new_x, old_x, best_h)
        hc = project(np.clip(hc, 0.0, 1.0), S_try)
        try_optimize(hc, n_try, S_try, dx_try, [20.0, 60.0, 150.0], 300)

    if not time_ok():
        return best_h, best_c5, len(best_h)

    # Phase 3: Random restarts with perturbation
    for _ in range(30):
        if not time_ok():
            break
        n_try = best_n + int(np.random.randint(-12, 13))
        if n_try < 20:
            n_try = 20
        S_try = n_try / 2.0
        dx_try = 2.0 / n_try
        old_n = len(best_h)
        old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
        new_x = (np.arange(n_try) + 0.5) * dx_try
        hc = np.interp(new_x, old_x, best_h)
        sigma = 0.05 + 0.15 * np.random.rand()
        hc = hc + sigma * np.random.randn(n_try)
        hc = project(np.clip(hc, 0.0, 1.0), S_try)
        try_optimize(hc, n_try, S_try, dx_try, [30.0, 80.0], 250)

    if not time_ok():
        return best_h, best_c5, len(best_h)

    # Phase 4: Subgradient polishing
    n = len(best_h)
    S = n / 2.0
    dx = 2.0 / n
    h = best_h.copy()
    alpha = 0.02
    for it in range(5000):
        if not time_ok():
            break
        h = subgrad_step(h, n, dx, alpha)
        h = project(h, S)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        if it % 100 == 0:
            alpha *= 0.95

    # Phase 5: Final SLSQP polish on best solution
    if time_ok():
        n = len(best_h)
        S = n / 2.0
        dx = 2.0 / n
        try_optimize(best_h.copy(), n, S, dx, [300.0, 600.0, 1000.0], 500)

    # Final check
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
