def _fix_sum(y, target, quick):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    res = target - float(np.sum(y))
    if abs(res) <= 1e-12:
        return y
    if (not quick) or abs(res) > 0.5:
        # bisection projection
        x = y
        lo = float(np.min(x)) - 1.0 - 1e-12
        hi = float(np.max(x)) + 1.0 + 1e-12
        for _ in range(60):
           mid=0.5*(lo+hi)
           s=float(np.sum(np.clip(x-mid,0.0,1.0)))
           if s > target: lo=mid
           else: hi=mid
        y=np.clip(x-0.5*(lo+hi),0,1)
        res = target - float(np.sum(y))
    if abs(res) <= 1e-12: return y
    # water fill fix
    if res > 0:
        order = np.argsort(y)
    else:
        order = np.argsort(-y)
    rem = abs(res)
    for idx in order:
        if rem <= 1e-15: break
        if res > 0:
            add = min(1.0 - y[idx], rem)
            y[idx] += add
        else:
            add = min(y[idx], rem)
            y[idx] -= add
        rem -= add
    return y

def _project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)
    lo = min(float(np.min(x)), 0.0) - 1.0 - float(target)
    hi = max(float(np.max(x)), 1.0) + 1.0 + float(target)
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    res = target - float(np.sum(y))
    if abs(res) > 1e-12:
       if res > 0:
          order = np.argsort(y)
       else:

       rem = abs(res)
       for idx in order:

          if res > 0:
             add = min(1.0 - y[idx], rem)
             y[idx] += add
          else:
             add = min(y[idx], rem)
             y[idx] -= add
          rem -= add
    return y

def overlap_grad_matrix(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)

    dx = 2.0 / float(n)
    G = np.zeros((L, n), dtype=float)

        s = n - 1 - m
        i_start = 0 if m < n - 1 else m - (n - 1)
        i_end = m if m < n else n - 1
        if i_start > i_end:
            continue
        i = np.arange(i_start, i_end + 1)
        j = i + s
        G[m, i] += dx * (1.0 - h[j])
        G[m, j] -= dx * h[i]
    return G

        best_h = np.full(47, 0.5, dtype=float)
        best_h = ensure_feasible(best_h, best_h.size)
        best_c = exact_c5(best_h)
        best_n = int(best_h.size)

    out = ensure_feasible(best_h, best_h.size)
    out_c = exact_c5(out)
    return out, float(out_c), int(out.size)
