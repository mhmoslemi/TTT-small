import numpy as np
import time
import scipy.optimize


def overlap_values_np(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def exact_bound(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return 0.0
    vals = overlap_values_np(h)
    if vals.size == 0:
        return 0.0
    v = float(np.max(vals))
    return v if np.isfinite(v) else np.inf


def overlap_values_fast(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    dx = 2.0 / float(n)
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    m = (n - 1 - (j - i)).ravel()
    w = (h[:, None] * (1.0 - h[None, :])).ravel() * dx
    return np.bincount(m, weights=w, minlength=2 * n - 1).astype(float)


def overlap_grad_values(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float), np.zeros((0, 0), dtype=float)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    m = (n - 1 - (j - i)).ravel()
    ir = i.ravel()
    jr = j.ravel()
    G = np.zeros((L, n), dtype=float)
    ci = (1.0 - h[None, :]).ravel() * dx
    cj = -(h[:, None]).ravel() * dx
    G[m, ir] += ci
    G[m, jr] += cj
    w = (h[:, None] * (1.0 - h[None, :])).ravel() * dx
    c = np.bincount(m, weights=w, minlength=L).astype(float)
    return c, G


def enforce_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    target = float(target)
    for _ in range(int(y.size * 3 + 20)):
        r = target - float(np.sum(y))
        if abs(r) < 1e-13:
            break
        if r > 0:
            mask = y < 1.0 - 1e-12
            if not np.any(mask):
                break
            idx = int(np.argmax(np.where(mask, y, -np.inf)))
            inc = min(1.0 - y[idx], r)
            y[idx] += inc
        else:
            mask = y > 1e-12
            if not np.any(mask):
                break
            idx = int(np.argmin(np.where(mask, y, np.inf)))
            dec = min(y[idx], -r)
            y[idx] -= dec
    r = target - float(np.sum(y))
    if abs(r) > 0.0:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if np.any(mask):
            y[mask] += r / float(np.sum(mask))
            y = np.clip(y, 0.0, 1.0)
    return y


def project_bounded_simplex(v, target):
    v = np.asarray(v, dtype=float).ravel()
    target = float(target)
    n = int(v.size)
    if n == 0:
        return v
    lo = float(np.min(v)) - 1.0 - 1e-12
    hi = float(np.max(v)) + 1e-12
    for _ in range(100):
        mu = 0.5 * (lo + hi)
        y = np.clip(v - mu, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = mu
        else:
            hi = mu
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return enforce_sum(y, target)


def quick_feasible(h):
    h = np.asarray(h, dtype=float).ravel().copy()
    if h.size == 0:
        return h
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    target = float(h.size * 0.5)
    r = target - float(np.sum(h))
    if abs(r) <= 1e-9:
        h = enforce_sum(h, target)
    else:
        h = project_bounded_simplex(h, target)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target)


def consider_candidate(h, best_h, best_c):
    if h is None:
        return best_h, best_c, False
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0 or not np.isfinite(h).all():
        return best_h, best_c, False
    h = quick_feasible(h)
    if not np.isfinite(h).all():
        return best_h, best_c, False
    c = exact_bound(h)
    if not np.isfinite(c):
        return best_h, best_c, False
    accepted = False
    if best_h is None:
        best_h = h.copy()
        best_c = float(c)
        accepted = True
    else:
        bh = np.asarray(best_h, dtype=float).ravel()
        if c < float(best_c) - 1e-11:
            best_h = h.copy()
            best_c = float(c)
            accepted = True
        elif abs(c - float(best_c)) <= 1e-11 and h.size < bh.size:
            best_h = h.copy()
            best_c = float(c)
            accepted = True
    return best_h, best_c, accepted


def symmetrize(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return h
    return 0.5 * (h + h[::-1])


def make_uniform(n):
    n = int(max(1, n))
    return np.full(n, 0.5, dtype=float)


def resample_profile(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float).ravel()
    n_old = int(h_old.size)
    n_new = int(n_new)
    if n_old <= 0 or n_new <= 0:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return quick_feasible(y)


def make_random_profile(n, rng, kind):
    n = int(max(1, n))
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if kind == 0:
        y = 0.5 + 0.35 * rng.standard_normal(n)
    elif kind == 1:
        y = rng.random(n)
    elif kind == 2:
        f = int(rng.integers(1, 4))
        p = float(rng.uniform(0.0, 2.0 * np.pi))
        a = float(rng.uniform(0.1, 0.35))
        y = 0.5 + a * np.sin(2.0 * np.pi * f * x + p)
    elif kind == 3:
        y = (rng.random(n) > 0.5).astype(float)
    elif kind == 4:
        y = (rng.random(n) > 0.5).astype(float) + 0.1 * rng.standard_normal(n)
    else:
        y = 0.5 + 0.1 * rng.standard_normal(n)
    return quick_feasible(y)


def lse_objective(x, tau):
    x = np.asarray(x, dtype=float).ravel()
    c = overlap_values_fast(x)
    if c.size == 0:
        return 0.0
    tau = float(tau)
    if tau <= 1e-12:
        return float(np.max(c))
    z = c / tau
    zmax = float(np.max(z))
    return float(tau * (zmax + np.log(np.sum(np.exp(z - zmax)))))


def lse_gradient(x, tau):
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0:
        return x.copy()
    tau = float(tau)
    c, G = overlap_grad_values(x)
    if c.size == 0:
        return x.copy()
    if tau <= 1e-12:
        mc = float(np.max(c))
        act = np.abs(c - mc) <= 1e-12
        if not np.any(act):
            act = c == mc
        w = np.zeros_like(c)
        w[act] = 1.0 / float(np.count_nonzero(act))
    else:
        z = c / tau
        zmax = float(np.max(z))
        w = np.exp(z - zmax)
        s = float(np.sum(w))
        if s <= 0.0:
            w = np.full(c.shape, 1.0 / c.size)
        else:
            w /= s
    return G.T.dot(w)


def sum_eq(x, target):
    return np.array([float(np.sum(x)) - float(target)])


def sum_eq_jac(x, target):
    return np.ones((1, x.size), dtype=float)


def sum_eq_last(x, target):
    x = np.asarray(x, dtype=float).ravel()
    return np.array([float(np.sum(x[:-1])) - float(target)])


def sum_eq_last_jac(x, target):
    x = np.asarray(x, dtype=float).ravel()
    j = np.ones((1, x.size), dtype=float)
    if x.size:
        j[0, -1] = 0.0
    return j


def epi_obj(x):
    x = np.asarray(x, dtype=float).ravel()
    return float(x[-1]) if x.size else 0.0


def epi_grad(x):
    x = np.asarray(x, dtype=float).ravel()
    g = np.zeros_like(x)
    if x.size:
        g[-1] = 1.0
    return g


def epi_ineq(x):
    x = np.asarray(x, dtype=float).ravel()
    if x.size <= 1:
        return np.array([float(x[-1]) if x.size else 0.0])
    h = x[:-1]
    c = overlap_values_fast(h)
    return x[-1] - c


def epi_ineq_jac(x):
    x = np.asarray(x, dtype=float).ravel()
    if x.size <= 1:
        return np.ones((1, max(1, x.size)), dtype=float)
    h = x[:-1]
    _, G = overlap_grad_values(h)
    L = int(G.shape[0])
    J = np.empty((L, x.size), dtype=float)
    J[:, :-1] = G
    J[:, -1] = 1.0
    return J


def solve_lse(h0, tau, maxiter):
    h0 = quick_feasible(h0)
    n = int(h0.size)
    if n == 0:
        return h0
    target = float(n * 0.5)
    bounds = [(0.0, 1.0)] * n
    cons = [{
        "type": "eq",
        "fun": sum_eq,
        "jac": sum_eq_jac,
        "args": (target,)
    }]
    options = {"maxiter": int(max(1, maxiter)), "ftol": 1e-12, "disp": False}
    try:
        res = scipy.optimize.minimize(
            lse_objective,
            h0,
            method="SLSQP",
            bounds=bounds,
            jac=lse_gradient,
            args=(float(tau),),
            constraints=cons,
            options=options
        )
        return np.asarray(res.x, dtype=float).ravel()
    except Exception:
        return h0


def solve_epigraph(h0, c0, slack, maxiter):
    h0 = quick_feasible(h0)
    n = int(h0.size)
    if n == 0:
        return np.array([0.0], dtype=float)
    target = float(n * 0.5)
    cform = overlap_values_fast(h0)
    cfmax = float(np.max(cform)) if cform.size else 0.0
    cmax = max(float(c0), cfmax, 0.0)
    s0 = cmax + max(float(slack), 1e-8) + 1e-9
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = s0
    bounds = [(0.0, 1.0)] * n + [(0.0, s0 + 1e-10)]
    cons = [
        {
            "type": "eq",
            "fun": sum_eq_last,
            "jac": sum_eq_last_jac,
            "args": (target,)
        },
        {
            "type": "ineq",
            "fun": epi_ineq,
            "jac": epi_ineq_jac
        }
    ]
    options = {"maxiter": int(max(1, maxiter)), "ftol": 1e-12, "disp": False}
    try:
        res = scipy.optimize.minimize(
            epi_obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            jac=epi_grad,
            constraints=cons,
            options=options
        )
        return np.asarray(res.x, dtype=float).ravel()
    except Exception:
        return x0


def choose_maxiter(n, left):
    n = int(n)
    left = float(left)
    if left < 120.0:
        return 40
    if left < 300.0:
        return 90
    if left < 600.0:
        return 140
    if n <= 55:
        return 170
    if n <= 80:
        return 120
    return 90
