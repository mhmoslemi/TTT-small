    """Compute C_k for a specific lag k (k ranges from -(n-1) to n-1)."""
    dx = 2.0 / n
    if k >= 0:
        if k >= n:
            return 0.0
        return float(dx * np.sum(h[k:] * (1.0 - h[:n-k])))

        h_new = repair_feasible(res.x[1:], target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_init:
            return h_new, c5_new
    return h, c5_init

def local_search_swaps(h, n, max_iter=5000, rng=None, time_limit=None, t0=None):
    """Local search using pairwise swap moves."""
    if rng is None:
        rng = np.random.default_rng()
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    no_improve = 0
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        
        improved = False
        # Try random swaps
        for _ in range(20):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                h_best = h.copy()
                improved = True
            else:
                h[i], h[j] = h[j], h[i]  # undo
                break  # actually don't break, keep trying
            # don't break, keep trying with the improved h
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
    
    return h_best, best_c5

def local_search_push(h, n, max_iter=5000, rng=None, time_limit=None, t0=None):
    """Local search using push moves (decrease one, increase another)."""
    if rng is None:
        rng = np.random.default_rng()
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    no_improve = 0
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        
        improved = False
        for _ in range(30):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            delta = 0.01 + 0.09 * rng.random()
            direction = 1 if rng.random() < 0.5 else -1
            h_new = h.copy()
            h_new[i] += direction * delta
            h_new[j] -= direction * delta
            if np.all(h_new >= -1e-10) and np.all(h_new <= 1 + 1e-10):
                h_new = np.clip(h_new, 0, 1)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    h_best = h_new.copy()
                    h = h_new
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve > 300:
                break
        else:
            no_improve = 0
    
    return h_best, best_c5

def transfer_h(h_old, n_old, n_new):
    """Transfer a solution from one grid size to another."""
    # Map old cell centers to new grid
    old_centers = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    new_edges_old = np.arange(n_old + 1) * (2.0 / n_old)
    
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_start = j * (2.0 / n_new)
        x_end = (j + 1) * (2.0 / n_new)
        # Find which old cells overlap with [x_start, x_end)
        overlap = 0.0
        total = 0.0
        for i in range(n_old):
            o_start = max(x_start, i * (2.0 / n_old))
            o_end = min(x_end, (i + 1) * (2.0 / n_old))
            if o_end > o_start:
                overlap += h_old[i] * (o_end - o_start)
                total += (o_end - o_start)
        if total > 0:
            h_new[j] = overlap / total
        else:
            h_new[j] = 0.5
    
    return repair_feasible(h_new, n_new / 2.0)
