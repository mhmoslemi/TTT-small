def pair_line_search(h, i, j, max_evals=50):
    """Find best τ in [0, T] for moving mass from i to j."""
    h = h.copy()
    T = min(h[i], 1 - h[j])
    if T < 1e-12:
        return h, evaluate_C(h)
    
    # Evaluate at endpoints
    h0 = h.copy()
    c0 = evaluate_C(h0)
    
    h_T = h.copy()
    h_T[i] -= T
    h_T[j] += T
    cT = evaluate_C(h_T)
    
    best_c = min(c0, cT)
    best_tau = 0 if c0 <= cT else T
    
    # Ternary search
    lo, hi = 0.0, T
    # Also check midpoint
    results = [(0, c0), (T, cT), (T/2, evaluate_C(h at T/2))]
    
    for _ in range(max_evals - 3):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        h1 = h.copy(); h1[i] -= m1; h1[j] += m1
        h2 = h.copy(); h2[i] -= m2; h2[j] += m2
        c1 = evaluate_C(h1)
        c2 = evaluate_C(h2)
        results.append((m1, c1))
        results.append((m2, c2))
        if c1 < c2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-9 * T:
            break
    
    # Find best
    best_tau = min(results, key=lambda x: x[1])[0]
    h_best = h.copy()
    h_best[i] -= best_tau
    h_best[j] += best_tau
    return h_best, evaluate_C(h_best)

```python
def compute_smoothed_gradient(h, theta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / theta)
    w = exp_c / exp_c.sum()
    
    # Gradient: for each output index k, lag L = k - (n-1)
    # ∂c_L/∂h[m] = (1-h[m-L])*I[0≤m-L≤n-1] - h[m+L]*I[0≤m+L≤n-1]
    # ∇S[m] = dx * Σ_k w[k] * ∂c_{k-(n-1)}/∂h[m]
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        # First term: (1-h[m-L]) for m where 0 ≤ m-L ≤ n-1, i.e., L ≤ m ≤ L+n-1, intersected with [0, n-1]
        m_lo = max(0, L)
        m_hi = min(n - 1, L + n - 1)
        if m_lo <= m_hi:
            idx = np.arange(m_lo, m_hi + 1)
            grad[idx] += wk * (1 - h[idx - L])
        # Second term: -h[m+L] for m where 0 ≤ m+L ≤ n-1, i.e., -L ≤ m ≤ n-1-L, intersected with [0, n-1]
        m_lo2 = max(0, -L)
        m_hi2 = min(n - 1, n - 1 - L)
        if m_lo2 <= m_hi2:
            idx2 = np.arange(m_lo2, m_hi2 + 1)
            grad[idx2] -= wk * h[idx2 + L]
    
    return grad
```

import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_h(y, target_mass):
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    # Expand bracket
    for _ in range(10):
        if np.sum(np.clip(y + lo, 0.0, 1.0)) <= target_mass:
            break
        lo -= 1.0
    for _ in range(10):
        if np.sum(np.clip(y + hi, 0.0, 1.0)) >= target_mass:
            break
        hi += 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y + mid, 0.0, 1.0)) > target_mass:
            hi = mid
        else:
            lo = mid
    z = np.clip(y + 0.5*(lo+hi), 0.0, 1.0)
    return z

def smoothed_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_shifted = c - np.max(c)
    exp_c = np.exp(c_shifted / theta)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        # First term: (1-h[m-L]) for m in [max(0,L), min(n-1, L+n-1)]
        m_lo = max(0, L)
        m_hi = min(n-1, L+n-1)
        if m_lo <= m_hi:
            idx = np.arange(m_lo, m_hi+1)
            grad[idx] += wk * (1.0 - h[idx - L])
        # Second term: -h[m+L] for m in [max(0,-L), min(n-1, n-1-L)]
        m_lo2 = max(0, -L)
        m_hi2 = min(n-1, n-1-L)
        if m_lo2 <= m_hi2:
            idx2 = np.arange(m_lo2, m_hi2+1)
            grad[idx2] -= wk * h[idx2 + L]
    return grad

def pair_line_search(h, i, j, max_iter=60):
    n = len(h)
    T = min(h[i], 1.0 - h[j])
    if T < 1e-12:
        return None
    
    def eval_tau(tau):
        h2 = h.copy()
        h2[i] -= tau
        h2[j] += tau
        return evaluate_C(h2)
    
    # Evaluate endpoints and a few points
    best_c = eval_tau(0.0)
    best_tau = 0.0
    c_T = eval_tau(T)
    if c_T < best_c:
        best_c = c_T
        best_tau = T
    
    # Ternary search
    lo, hi = 0.0, T
    for _ in range(max_iter):
        if hi - lo < 1e-10 * (T + 1e-15):
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = eval_tau(m1)
        c2 = eval_tau(m2)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
        if c1 < best_c:
            best_c = c1
            best_tau = m1
        if c2 < best_c:
            best_c = c2
            best_tau = m2
    
    if best_tau <= 1e-12 or best_c >= best_c - 1e-12:
        if best_tau <= 1e-12:

    h_new = h.copy()
    h_new[i] -= best_tau
    h_new[j] += best_tau
    # Fix exact mass

    return h_new, best_tau, best_c
