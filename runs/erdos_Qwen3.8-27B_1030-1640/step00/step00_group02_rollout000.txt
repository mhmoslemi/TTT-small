    if len(cand_ts) == 0:
        return None
    
    # Sort and de-duplicate
    cand_ts = np.sort(cand_ts)
    unique_ts = [cand_ts[0]]
    for t in cand_ts[1:]:
        if t - unique_ts[-1] > 1e-10:
            unique_ts.append(t)
    unique_ts = np.array(unique_ts)
    
    if len(unique_ts) > 200:
        # Too many candidates; subsample
        # Keep endpoints and evenly spaced
        idx = np.unique(np.round(np.linspace(0, len(unique_ts) - 1, 200)).astype(int))
        unique_ts = unique_ts[idx]
    
    # Evaluate surrogate: max_m (A + B*t + Q*t²) for each candidate t
    # Do this in chunks
    best_surrogate = np.inf
    best_t = 0.0
    n_cand = len(unique_ts)
    
    for t in unique_ts:
        vals = A + B * t + Q * t * t
        s = np.max(vals)
        if s * dx < best_surrogate:
            best_surrogate = s * dx
            best_t = t
    
    # Now evaluate top candidates with exact evaluator
    # Sort by surrogate value
    surrogate_vals = np.array([np.max(A + B * t + Q * t * t) * dx for t in unique_ts])
    order_cand = np.argsort(surrogate_vals)
    
    # Evaluate top N candidates exactly
    n_eval = min(n_cand, 40)
    best_new_c5 = current_c5
    best_new_h = None
    
    for idx in order_cand[:n_eval]:
        if deadline is not None and time.monotonic() >= deadline:
            break
        t = unique_ts[idx]
        h_new = h + t * d
        h_new = make_feasible(h_new)
        c5_new = c5_value(h_new)
        if c5_new < best_new_c5 - 1e-15:
            best_new_c5 = c5_new
            best_new_h = h_new.copy()
    
    if best_new_h is not None:
        return best_new_h, best_new_c5
    return None

def transfer_dimension(h_old, n_new):
    """Transfer h from old dimension to new dimension via interpolation."""
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new)
