def _lse_stats(vals, tau):
   vals=np.asarray(vals,float)
   if vals.size==0: return 0.0, np.array([])
   maxv=float(vals.max())
   if tau <= 1e-8 or not np.isfinite(tau):
       tol = max(1e-7, 1e-6*max(1.0,abs(maxv)))
       active = vals >= maxv - tol
       if not active.any():
           active = np.zeros(vals.shape,bool); active[int(np.argmax(vals))]=True
       p=np.zeros_like(vals)
       p[active]=1.0/float(active.sum())
       return maxv,p
   scaled=vals/tau
   max_scaled=float(scaled.max())
   expv=np.exp(scaled-max_scaled)
   Z=float(expv.sum())
   if not np.isfinite(Z) or Z <= 0.0:
       p=np.zeros_like(vals); p[int(np.argmax(scaled))]=1.0; return maxv,p
   p=expv/Z
   F=maxv + tau*math.log(Z)

def interpolate_h(hc, n):
   hc=np.asarray(hc,float)
   if hc.size==0: return np.full(n,0.5)
   if hc.size==n: return hc.copy()
   if hc.size==1: return np.full(n,float(hc[0]))

import time
import math
import warnings
import numpy as np
from scipy.optimize import minimize

_USE_REV_P = False
_USE_DIRECT_GRAD = False
_LAG_SLICES = {}
_BEST = {"h": None, "c": float("inf"), "n": 0}


def evaluate_c5(h):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size < 2:
        return float("inf")
    if not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(vals) * dx)


def repair_sum_inplace(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(target)
    if h.size == 0:
        return h

    for _ in range(50):
        d = target - float(h.sum())
        if abs(d) <= 1e-12:
            break
        if d > 0.0:
            room = 1.0 - h
            pos = room > 1e-14
            if not np.any(pos):
                break
            total_room = float(room[pos].sum())
            if total_room <= 0.0:
                break
            scale = d / total_room
            if scale > 1.0:
                h[pos] += room[pos]
            else:
                h[pos] += room[pos] * scale
        else:
            amt = h
            pos = amt > 1e-14
            if not np.any(pos):
                break
            total_amt = float(amt[pos].sum())
            if total_amt <= 0.0:
                break
            scale = -d / total_amt
            if scale > 1.0:
                h[pos] -= amt[pos]
            else:
                h[pos] -= amt[pos] * scale
        h = np.clip(h, 0.0, 1.0)

    d = target - float(h.sum())
    if abs(d) > 1e-14:
        if d > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] += min(d, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            h[idx] += max(d, -h[idx])
    return np.clip(h, 0.0, 1.0)


def _clip_sum(z, alpha):
    return float(np.clip(z - alpha, 0.0, 1.0).sum())


def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).copy()
    n = z.size
    if n == 0:
        return z
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = float(target)

    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    if (np.min(z) >= -1e-12 and np.max(z) <= 1.0 + 1e-12 and
            abs(_clip_sum(z, 0.0) - target) <= 1e-10):
        z = np.clip(z, 0.0, 1.0)
        return repair_sum_inplace(z, target)

    amp = max(1.0, float(np.max(np.abs(z))))
    low = -amp - 1.0
    high = amp + 1.0

    s_low = _clip_sum(z, low)
    s_high = _clip_sum(z, high)
    for _ in range(20):
        if s_low < target:
            low -= max(1.0, 1.0 - low)
            s_low = _clip_sum(z, low)
        if s_high > target:
            high += max(1.0, 1.0 + high)
            s_high = _clip_sum(z, high)
        if s_low >= target and s_high <= target:
            break

    for _ in range(70):
        mid = 0.5 * (low + high)
        s = _clip_sum(z, mid)
        if s > target:
            low = mid
        else:
            high = mid

    h = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    return repair_sum_inplace(h, target)


def is_feasible(h, tol=1e-7):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    target = 0.5 * float(h.size)
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    return abs(float(h.sum()) - target) <= max(tol, 1e-9)


def update_best(h):
    global _BEST
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * float(h.size)
    if np.min(h) < -1e-7 or np.max(h) > 1.0 + 1e-7:
        h = np.clip(h, 0.0, 1.0)

        h = project_box_sum(h, target)

    c = evaluate_c5(h)
    if not np.isfinite(c):
        return

    if c < old - 1e-12 or (abs(c - old) <= 1e-12 and h.size < int(_BEST["n"])):
        _BEST["h"] = h.copy()
        _BEST["c"] = float(c)
        _BEST["n"] = int(h.size)


def interpolate_h(hc, n):
    hc = np.asarray(hc, dtype=float)
    n = int(n)
    if hc.size == 0:
        return np.full(n, 0.5, dtype=float)
    if hc.size == n:
        return hc.copy()
    if hc.size == 1:
        return np.full(n, float(hc[0]), dtype=float)
    x_old = (np.arange(hc.size, dtype=float) + 0.5) / float(hc.size)
    x_new = (np.arange(n, dtype=float) + 0.5) / float(n)
    return np.interp(x_new, x_old, hc, left=hc[0], right=hc[-1])


def _mse_from_stats(vals, tau):
    vals = np.asarray(vals, dtype=float)
    if vals.size == 0:
        return 0.0, np.array([], dtype=float)
    maxv = float(np.max(vals))

    if tau <= 1e-8 or not np.isfinite(tau):
        tol = max(1e-7, 1e-6 * max(1.0, abs(maxv)))
        active = vals >= (maxv - tol)
        if not np.any(active):
            active = np.zeros(vals.shape, dtype=bool)
            active[int(np.argmax(vals))] = True
        p = np.zeros_like(vals, dtype=float)
        p[active] = 1.0 / float(active.sum())
        return maxv, p

    scaled = vals / tau
    max_scaled = float(np.max(scaled))
    expv = np.exp(scaled - max_scaled)
    Z = float(np.sum(expv))
    if not np.isfinite(Z) or Z <= 0.0:
        p = np.zeros_like(vals, dtype=float)
        p[int(np.argmax(scaled))] = 1.0
        return maxv, p

    p = expv / Z
    F = maxv + tau * math.log(Z)
    return float(F), p


def soft_value_correlate(h, tau):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return 0.0
    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    F, _ = _mse_from_stats(vals, tau)
    return F


def _get_lag_slices(n):
    if n in _LAG_SLICES:
        return _LAG_SLICES[n]
    arr = []
    for r in range(-(n - 1), n):
        if r >= 0:
            m = n - r
            arr.append((r, r, r + m, 0, m))
        else:
            t = -r
            m = n - t
            arr.append((r, 0, m, t, t + m))
    _LAG_SLICES[n] = arr
    return arr


def _direct_lse_val_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    q = 1.0 - h
    dx = 2.0 / float(n)
    slices = _get_lag_slices(n)

    vals = np.empty(len(slices), dtype=float)
    for k, (_, sa, ea, sb, eb) in enumerate(slices):
        vals[k] = dx * float(np.dot(h[sa:ea], q[sb:eb]))

    F, p = _mse_from_stats(vals, tau)
    grad = np.zeros(n, dtype=float)

    for k, (_, sa, ea, sb, eb) in enumerate(slices):
        pk = float(p[k])
        if pk == 0.0:
            continue
        ha = h[sa:ea]
        qb = q[sb:eb]
        grad[sa:ea] += pk * qb
        grad[sb:eb] -= pk * ha

    grad *= dx
    return F, grad


def _conv_lse_val_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    q = 1.0 - h

    vals = np.correlate(h, q, mode="full") * dx
    F, p_raw = _mse_from_stats(vals, tau)
    p = p_raw if not _USE_REV_P else p_raw[::-1]

    off = n - 1
    term1 = np.convolve(p, q, mode="full")[off:off + n]
    term2 = np.convolve(p[::-1], h, mode="full")[off:off + n]
    grad = dx * (term1 - term2)
    return float(F), grad


def soft_value_grad(h, tau):
    h = np.asarray(h, dtype=float)
    if _USE_DIRECT_GRAD:
        return _direct_lse_val_grad(h, tau)
    return _conv_lse_val_grad(h, tau)


def _check_orientation():
    global _USE_REV_P, _USE_DIRECT_GRAD
    for rev in (False, True):
        _USE_REV_P = rev
        ok = True
        for n in (3, 4, 5):
            rng = np.random.default_rng(2024 + n)
            h = 0.5 + 0.35 * rng.random(n)
            h = np.clip(h, 0.02, 0.98)
            tau = 0.05
            _, g = _conv_lse_val_grad(h, tau)
            if not np.all(np.isfinite(g)):
                ok = False
                break
            eps = 1e-5
            for idx in (0, n // 2, n - 1):
                hp = h.copy()
                hm = h.copy()
                hp[idx] = min(1.0 - 1e-4, hp[idx] + eps)
                hm[idx] = max(1e-4, hm[idx] - eps)
                denom = hp[idx] - hm[idx]
                if abs(denom) < 1e-12:
                    continue
                fp = soft_value_correlate(hp, tau)
                fm = soft_value_correlate(hm, tau)
                fd = (fp - fm) / denom
                if not np.isfinite(fd) or abs(fd - g[idx]) > 1e-3:
                    ok = False
                    break
            if not ok:
                break
        if ok:
            return True
    _USE_DIRECT_GRAD = True
    return False


def _lbfgs_obj(x, tau, pen):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100
    val = soft_value_correlate(x, tau)
    s = float(np.sum(x) - 0.5 * float(x.size))
    return float(val + pen * s * s)


def _lbfgs_jac(x, tau, pen):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros_like(x, dtype=float)
    _, g = soft_value_grad(x, tau)
    s = float(np.sum(x) - 0.5 * float(x.size))
    g = g + 2.0 * pen * s * np.ones_like(g)
    return g


def _slsqp_func(x, tau):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100
    return soft_value_correlate(x, tau)


def _slsqp_jac(x, tau):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros_like(x, dtype=float)
    _, g = soft_value_grad(x, tau)
    return g


def _sum_eq(x):
    x = np.asarray(x, dtype=float)
    return float(np.sum(x) - 0.5 * float(x.size))


def _sum_jac(x):
    x = np.asarray(x, dtype=float)
    return np.ones_like(x, dtype=float)


def make_bounds(n):
    return [(0.0, 1.0) for _ in range(int(n))]


def _safe_minimize(fun, x0, jac, args, method, bounds, constraints, options):
    if constraints is None:
        constraints = []
    if options is None:
        options = {}
    x0 = np.asarray(x0, dtype=float)
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            return minimize(
                fun,
                x0,
                jac=jac,
                args=args,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options=options,
            )
    except Exception:
        return None


def pg_polish(h, tau, max_iter=50, deadline=None):

    if h.size == 0:

    target = 0.5 * float(h.size)

    no_accept = 0

        F, g = soft_value_grad(h, tau)
        if not np.isfinite(F) or not np.all(np.isfinite(g)):

        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-10:
            break

        old_alpha = alpha
        accepted = False
        for _ in range(25):
            hn = project_box_sum(h - alpha * g, target)
            Fnew = soft_value_correlate(hn, tau)

                F = Fnew
                accepted = True

            alpha = max(1e-5, old_alpha * 0.2)
            continue

def continuous_to_binary(hc, n, seed=42):
    n = int(n)
    if n < 4 or n % 2 != 0:
        return None
    hc = interpolate_h(hc, n)
    hc = np.nan_to_num(hc, nan=0.5, posinf=1.0, neginf=0.0)
    idx = np.argsort(hc, kind="stable")
    h = np.zeros(n, dtype=np.uint8)
    h[idx[len(idx) // 2:]] = 1
    return h


def _binary_counts_arrays(h):
    n = int(h.size)
    c = np.zeros(2 * n - 1, dtype=np.int64)
    off = n - 1
    ones = np.nonzero(h == 1)[0].astype(np.int64, copy=False)
    zeros = np.nonzero(h == 0)[0].astype(np.int64, copy=False)
    if ones.size > 0 and zeros.size > 0:
        for a in ones:
            idx = off + int(a) - zeros
            c[idx] += 1
    return c, ones, zeros


def _binary_swap_delta(ones, zeros, i, j, n):
    n = int(n)
    off = n - 1
    d = np.zeros(2 * n - 1, dtype=np.int32)

    idx = off + i - zeros
    if idx.size:
        d[idx] -= 1

    mask = zeros != j
    if np.any(mask):
        d[off + j - zeros[mask]] += 1

    d[off + j - i] += 1

    idx = off + ones - j
    if idx.size:
        d[idx] -= 1

    mask = ones != i
    if np.any(mask):
        d[off + ones[mask] - i] += 1

    d[off + i - j] += 1
    return d


def _binary_active_candidate(h, ones, zeros, c, rng):
    n = int(h.size)
    off = n - 1
    maxv = int(c.max())
    idxs = np.where(c == maxv)[0]
    if idxs.size == 0:
        return None

    li = int(idxs[rng.integers(idxs.size)])
    d = li - off

    if d >= 0:
        m = n - d
        b = np.arange(m, dtype=np.int64)
        sel = np.where((h[b + d] == 1) & (h[b] == 0))[0]
        if sel.size == 0:
            return None
        p = int(sel[rng.integers(sel.size)])
        a = int(b[p] + d)
        bb = int(b[p])
    else:
        t = -d
        m = n - t
        b = np.arange(t, n, dtype=np.int64)
        a_arr = b - t
        sel = np.where((h[a_arr] == 1) & (h[b] == 0))[0]
        if sel.size == 0:
            return None
        p = int(sel[rng.integers(sel.size)])
        a = int(a_arr[p])
        bb = int(b[p])

    r = float(rng.random())
    if r < 0.4:
        return a, bb
    if r < 0.7:
        if zeros.size == 0:
            return a, bb
        b2 = int(zeros[rng.integers(zeros.size)])
        if b2 == a:
            return a, bb
        return a, b2
    if ones.size == 0:
        return a, bb
    a2 = int(ones[rng.integers(ones.size)])
    if a2 == bb:
        return a, bb
    return a2, bb


def binary_refine(n, h_init, deadline, seed=42):
    n = int(n)
    if n < 8 or n % 2 != 0 or deadline is None or deadline <= time.time():
        return None, float("inf")

    h = np.zeros(n, dtype=np.uint8)
    if h_init is not None:
        h_init = np.asarray(h_init)
        if h_init.dtype == np.uint8 and h_init.size == n:
            h = h_init.copy()
        else:
            hb = continuous_to_binary(h_init, n, seed)
            if hb is not None:
                h = hb
    else:
        rng0 = np.random.default_rng(seed)
        idx = rng0.permutation(n)
        h[idx[:n // 2]] = 1

    if int(h.sum()) != n // 2:
        vals = np.zeros(n, dtype=float)
        if h_init is not None:
            vals = np.asarray(h_init, dtype=float)
            vals = interpolate_h(vals, n)
        else:
            vals = np.random.default_rng(seed + 12345).random(n)
        idx = np.argsort(vals, kind="stable")
        h = np.zeros(n, dtype=np.uint8)
        h[idx[len(idx) // 2:]] = 1

    c, ones, zeros = _binary_counts_arrays(h)
    curr_max = int(c.max())
    curr_dot = int(np.dot(c, c))
    best_h = h.copy()
    best_max = curr_max

    rng = np.random.default_rng(seed + 1000)
    stall = 0
    batch = 220

    while time.time() < deadline:
        for _ in range(batch):
            if time.time() > deadline:
                break
            if ones.size == 0 or zeros.size == 0:
                break

            cand = None
            if rng.random() < 0.55:
                cand = _binary_active_candidate(h, ones, zeros, c, rng)
            if cand is None:
                ii = int(ones[rng.integers(ones.size)])
                jj = int(zeros[rng.integers(zeros.size)])
                cand = (ii, jj)
            if cand is None:
                continue

            ii, jj = cand
            if ii == jj:
                continue

            d = _binary_swap_delta(ones, zeros, ii, jj, n)
            touched = np.nonzero(d)[0]
            if touched.size == 0:
                continue
            dv = d[touched]
            c[touched] += dv

            new_max = int(c.max())
            new_dot = int(np.dot(c, c))

            accept = False
            if new_max < curr_max:
                accept = True
            elif new_max == curr_max and new_dot < curr_dot:
                accept = True
            elif new_max == curr_max + 1 and rng.random() < 0.02:
                accept = True
            elif new_max < curr_max + 2 and rng.random() < 0.001:
                accept = True

            if accept:
                curr_max = new_max
                curr_dot = new_dot
                h[ii] = 0
                h[jj] = 1
                ones = np.concatenate((ones[ones != ii], np.array([jj], dtype=np.int64)))
                zeros = np.concatenate((zeros[zeros != jj], np.array([ii], dtype=np.int64)))
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                stall = 0
            else:
                c[touched] -= dv
                stall += 1

        if stall > 5000:
            h = best_h.copy()
            c, ones, zeros = _binary_counts_arrays(h)
            curr_max = int(c.max())
            curr_dot = int(np.dot(c, c))
            stall = 0

            for _ in range(25):
                if time.time() > deadline:
                    break
                if ones.size == 0 or zeros.size == 0:
                    break
                ii = int(ones[rng.integers(ones.size)])
                jj = int(zeros[rng.integers(zeros.size)])
                if ii == jj:
                    continue
                d = _binary_swap_delta(ones, zeros, ii, jj, n)
                touched = np.nonzero(d)[0]
                if touched.size == 0:
                    continue
                dv = d[touched]
                c[touched] += dv
                nm = int(c.max())
                if nm <= curr_max + 1:
                    nd = int(np.dot(c, c))
                    h[ii] = 0
                    h[jj] = 1
                    ones = np.concatenate((ones[ones != ii], np.array([jj], dtype=np.int64)))
                    zeros = np.concatenate((zeros[zeros != jj], np.array([ii], dtype=np.int64)))
                    curr_max = nm
                    curr_dot = nd
                    if nm < best_max:
                        best_max = nm
                        best_h = h.copy()
                else:
                    c[touched] -= dv

    score = float(best_max) * (2.0 / float(n))
    return best_h.astype(float, copy=False), score


def run(seed=42, budget_s=1100, **kwargs):
    global _BEST
    _BEST = {"h": None, "c": float("inf"), "n": 0}

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    budget = max(15.0, budget)

    t0 = time.time()
    deadline = t0 + budget - min(10.0, 0.05 * budget)
    if deadline < t0 + 5.0:
        deadline = t0 + budget

    _check_orientation()
    max_dim = 128 if _USE_DIRECT_GRAD else 256

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        if is_feasible(parent, tol=1e-6):
            update_best(parent)
        else:
            update_best(project_box_sum(parent, 0.5 * float(parent.size)))

    if _BEST["h"] is None:
        rng = np.random.default_rng(seed)
        n = 64
        h = np.zeros(n, dtype=float)
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
        update_best(h)

    for nb in (64, 96, 128):
        if nb > max_dim:
            continue
        if time.time() > deadline - 10.0:
            break
        hb = continuous_to_binary(_BEST["h"], nb, seed + nb)
        if hb is None:
            continue
        local_deadline = min(deadline, time.time() + 8.0)
        bh, sc = binary_refine(nb, hb, local_deadline, seed + nb)
        if bh is not None:
            update_best(bh)

    h = improve_continuous(
        _BEST["h"],
        (0.04, 0.02, 0.01, 0.005, 0.002),
        lb_iters=50,
        sl_iters=90,
        deadline=deadline,
        use_lb=True,
        use_pg=True,
    )
    update_best(h)

    for k, sigma in enumerate((0.05, 0.10, 0.15, 0.25)):
        if time.time() > deadline - 25.0:
            break
        n = int(_BEST["n"])
        rngp = np.random.default_rng(seed + 10000 + 7 * k)
        h0 = _BEST["h"] + rngp.normal(0.0, float(sigma), size=n)
        h0 = project_box_sum(h0, 0.5 * float(n))
        h0 = improve_continuous(
            h0,
            (0.03, 0.015, 0.008),
            lb_iters=35,
            sl_iters=60,
            deadline=deadline,
            use_lb=True,
            use_pg=False,
        )
        update_best(h0)

    seen = set()
    if _BEST["n"] > 0:
        seen.add(int(_BEST["n"]))
    base = int(_BEST["n"]) if _BEST["n"] > 0 else 64

    larger_ns = []
    for mult in (2, 3, 4):
        n = base * mult
        if 16 <= n <= max_dim and n not in seen:
            seen.add(n)
            larger_ns.append(n)
    for n in (192, 256):
        if 16 <= n <= max_dim and n not in seen:
            seen.add(n)
            larger_ns.append(n)

    for n in larger_ns:
        if time.time() > deadline - 45.0:
            break
        h0 = interpolate_h(_BEST["h"], n)
        h0 = project_box_sum(h0, 0.5 * float(n))
        update_best(h0)
        h0 = improve_continuous(
            h0,
            (0.03, 0.015, 0.008),
            lb_iters=45,
            sl_iters=70,
            deadline=deadline,
            use_lb=True,
            use_pg=False,
        )
        update_best(h0)

        if n % 2 == 0 and time.time() < deadline - 25.0:
            hb = continuous_to_binary(_BEST["h"], n, seed + 20000 + n)
            if hb is not None:
                bh, sc = binary_refine(n, hb, min(deadline, time.time() + 12.0), seed + 20000 + n)
                if bh is not None:
                    update_best(bh)

    if time.time() < deadline - 5.0:
        h = improve_continuous(
            _BEST["h"],
            (0.002, 0.001),
            lb_iters=0,
            sl_iters=120,
            deadline=deadline,
            use_lb=False,
            use_pg=False,
        )
        update_best(h)

    if _BEST["h"] is None:
        h = np.full(64, 0.5, dtype=float)
        update_best(h)

    h_final = project_box_sum(_BEST["h"], 0.5 * float(_BEST["n"]))
    update_best(h_final)

    return _BEST["h"], float(_BEST["c"]), int(_BEST["n"])
