def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max() + 1.0

        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

    return float(np.max(corr) * dx)

def lse_c5(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def compute_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]

            continue
        s = k - (n - 1)

            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] -= wk * h[s:]
        else:
            grad[:n+s] += wk * (1.0 - h[-s:])
            grad[-s:] -= wk * h[:n+s]
    return grad * dx

import numpy as np
import time

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_c5(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + ms / lam
    return m + (ms + np.log(np.mean(np.exp(scaled - ms)))) / lam

def compute_grad(h, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    scaled = lam * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        ev = np.exp(scaled - ms)
        w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
        else:
            if -s < n:
                grad[:n + s] += wk * (1.0 - h[-s:])
                grad[-s:] -= wk * h[:n + s]
    return grad * dx

def pg_optimize(h, n, lam, max_iter=1000):
    h = h.copy()
    f = lse_c5(h, lam, n)
    alpha = 2.0
    stall = 0
    for it in range(max_iter):
        g = compute_grad(h, lam, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        found = False
        a = alpha
        for _ in range(50):
            h_new = project_feasible(h - a * g, n)
            f_new = lse_c5(h_new, lam, n)
            if f_new < f - 1e-14:
                found = True
                break
            a *= 0.5
        if not found:
            stall += 1
            alpha = max(a, 0.001)
            if stall > 30:
                break
            continue
        stall = 0
        h = h_new
        f = f_new
        alpha = min(a * 1.3, 20.0)
    return h

def resample_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def make_binary_init(n, rng):
    n_ones = n // 2
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n_ones]] = 1.0
    h += rng.normal(0, 0.03, n)
    return project_feasible(h, n)

def make_smooth_init(n, rng):
    base = rng.uniform(0.2, 0.8, n)
    for _ in range(3):
        base = np.convolve(base, np.ones(5)/5.0, mode='same')
    return project_feasible(base, n)
