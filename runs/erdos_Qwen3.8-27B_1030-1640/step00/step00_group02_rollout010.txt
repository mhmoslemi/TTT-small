def adjust_sum_exact(h, S):
    h = np.asarray(h, dtype=float).copy()
    n = h.size

    for _ in range(n + 10):
        cur = float(np.sum(h))
        r = S - cur

        if r > 0:
            slack = 1.0 - h

    s = float(np.sum(np.exp(z - zmax)))
    if s <= 0.0 or not np.isfinite(s):
        return cmx
    return float(tau * (zmax + math.log(s)))

        b_arr = np.asarray(base, dtype=float).reshape(-1)
        if b_arr.size > 0 and np.all(np.isfinite(b_arr)):
            if b_arr.size != n:
                b_arr = resample_h(b_arr, n)
            b = ensure_feasible(b_arr)
    if b is None:
        fb = np.full(n, 0.5, dtype=float)
    else:
        fb = b

    if mode == "noise":
        scales = np.array([0.0015, 0.004, 0.009, 0.018, 0.04, 0.09], dtype=float)
        scale = float(scales[int(rng.integers(0, scales.size))])
        h = fb + rng.normal(0.0, scale, n)
    elif mode == "threshold":
        th = fb + rng.normal(0.0, 0.02, n)
        p = float(rng.uniform(0.35, 0.65))
        hi = float(rng.uniform(0.65, 0.95))
        lo = float(rng.uniform(0.05, 0.35))
        h = np.where(th > p, hi, lo)
    elif mode == "binary":
        h = (rng.random(n) < 0.5).astype(float)
    elif mode == "wave":
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        k1 = int(rng.integers(1, 8))
        k2 = int(rng.integers(1, 12))
        amp1 = float(rng.uniform(0.22, 0.48))
        amp2 = float(rng.uniform(0.05, 0.25))
        ph1 = float(rng.uniform(0.0, 2.0 * math.pi))
        ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
        h = 0.5 + amp1 * np.sin(math.pi * k1 * x + ph1) + amp2 * np.sin(math.pi * k2 * x + ph2)
    elif mode == "block":
        nb = int(rng.integers(4, 13))
        if nb > n:
            nb = n
        cuts = np.empty(nb + 1, dtype=float)
        cuts[0] = 0.0
        cuts[-1] = 2.0
        if nb > 1:
            cuts[1:-1] = np.sort(rng.uniform(0.0, 2.0, nb - 1))
        vals = rng.uniform(0.1, 0.9, nb)
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        idx = np.searchsorted(cuts, x, side="right") - 1
        idx = np.clip(idx, 0, nb - 1)
        h = vals[idx]
    else:
        h = fb.copy()
    return ensure_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    seed = int(seed)
    try:
        budget = max(2.0, float(budget_s))
    except Exception:
        budget = 1100.0
    t0 = time.monotonic()
    t_end = t0 + budget
    reserve = 60.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 10 ** 9

    parent = get_initial_parent()
    if parent is not None:
        best_h, best_c, best_n = try_update(parent, best_h, best_c, best_n)
        best_h, best_c, best_n = try_update(symmetrize_h(parent), best_h, best_c, best_n)

    if best_h is None:
        best_h, best_c, best_n = try_update(np.full(48, 0.5, dtype=float), best_h, best_c, best_n)

    if best_h is None:
        best_h = adjust_sum_exact(np.full(48, 0.5, dtype=float), 24.0)
        best_c = exact_c5(best_h)
        best_n = int(best_h.size)

    tau_list = make_tau_list(best_c)
    for tau in tau_list[:4]:
        if time.monotonic() > t_end - reserve - 90:
            break
        h1 = solve_penalty(best_h, tau, 50.0, maxiter=110)
        best_h, best_c, best_n = try_update(h1, best_h, best_c, best_n)
        hs = symmetrize_h(h1)
        hse, cse, nse = eval_feasible(hs)
        best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

    slacks = make_slacks(best_c)
    if best_n <= 70:
        for sl in slacks[:5]:
            if time.monotonic() > t_end - reserve - 80:
                break
            x0 = np.concatenate([best_h, [best_c + sl]])
            h1 = solve_epigraph(x0, s_bound=best_c + sl + 0.12, maxiter=90)
            if h1 is None:
                continue
            best_h, best_c, best_n = try_update(h1, best_h, best_c, best_n)
            hs = symmetrize_h(h1)
            hse, cse, nse = eval_feasible(hs)
            best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

    for tau in tau_list[4:9]:
        if time.monotonic() > t_end - reserve - 80:
            break
        h1 = solve_penalty(best_h, tau, 150.0, maxiter=110)
        best_h, best_c, best_n = try_update(h1, best_h, best_c, best_n)
        hs = symmetrize_h(h1)
        hse, cse, nse = eval_feasible(hs)
        best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

    if best_n <= 70:
        for sl in slacks:
            if time.monotonic() > t_end - reserve - 80:
                break
            x0 = np.concatenate([best_h, [best_c + sl]])
            h1 = solve_epigraph(x0, s_bound=best_c + sl + 0.08, maxiter=110)
            if h1 is None:
                continue
            best_h, best_c, best_n = try_update(h1, best_h, best_c, best_n)
            hs = symmetrize_h(h1)
            hse, cse, nse = eval_feasible(hs)
            best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

    dims = get_trial_dims(best_n)
    for nnew in dims:
        if time.monotonic() > t_end - reserve - 120:
            break
        hrep = resample_h(best_h, nnew)
        hre, cre, nre = eval_feasible(hrep)
        best_h, best_c, best_n = try_update_values(hre, cre, nre, best_h, best_c, best_n)

        tau = max(0.02, 0.25 * best_c)
        hpen = solve_penalty(hre, tau, 60.0, maxiter=90)
        hpe, cpe, npe = eval_feasible(hpen)
        best_h, best_c, best_n = try_update_values(hpe, cpe, npe, best_h, best_c, best_n)
        hs = symmetrize_h(hpen)
        hse, cse, nse = eval_feasible(hs)
        best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

        if nre <= 70:
            sl = max(0.0008, 0.0025 * cre)
            x0 = np.concatenate([hre, [cre + sl]])
            hpol = solve_epigraph(x0, s_bound=cre + sl + 0.08, maxiter=70)
            if hpol is not None:
                hpole, cpol, npol = eval_feasible(hpol)
                best_h, best_c, best_n = try_update_values(hpole, cpol, npol, best_h, best_c, best_n)
                hps = symmetrize_h(hpol)
                hse2, cse2, nse2 = eval_feasible(hps)
                best_h, best_c, best_n = try_update_values(hse2, cse2, nse2, best_h, best_c, best_n)

    modes = ("noise", "noise", "noise", "wave", "block", "threshold", "binary")
    dim_small = filter_dims(get_trial_dims(best_n), 72)
    if len(dim_small) == 0:
        dim_small = [int(best_n)]
    attempts = 0
    while time.monotonic() < t_end - reserve - 100 and attempts < 110:
        if best_h is not None and rng.random() < 0.72:
            base = best_h
        else:
            base = None
        ntry = int(dim_small[int(rng.integers(0, len(dim_small)))])
        mode = modes[int(rng.integers(0, len(modes)))]
        hr = make_random_candidate(mode, ntry, rng, base)
        hre, cre, nre = eval_feasible(hr)
        best_h, best_c, best_n = try_update_values(hre, cre, nre, best_h, best_c, best_n)

        if cre < best_c + 0.018 and time.monotonic() < t_end - reserve - 180:
            tau = max(0.03, 0.28 * best_c)
            hpen = solve_penalty(hre, tau, 70.0, maxiter=90)
            hpe, cpe, npe = eval_feasible(hpen)
            best_h, best_c, best_n = try_update_values(hpe, cpe, npe, best_h, best_c, best_n)
            hs = symmetrize_h(hpen)
            hse, cse, nse = eval_feasible(hs)
            best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

            if npe <= 66 and time.monotonic() < t_end - reserve - 120:
                sl = max(0.0008, 0.0025 * cpe)
                x0 = np.concatenate([hpe, [cpe + sl]])
                hpol = solve_epigraph(x0, s_bound=cpe + sl + 0.08, maxiter=70)
                if hpol is not None:
                    hpole, cpol, npol = eval_feasible(hpol)
                    best_h, best_c, best_n = try_update_values(hpole, cpol, npol, best_h, best_c, best_n)
                    hps = symmetrize_h(hpol)
                    hse2, cse2, nse2 = eval_feasible(hps)
                    best_h, best_c, best_n = try_update_values(hse2, cse2, nse2, best_h, best_c, best_n)
        attempts += 1

    if best_h is not None:
        hs = symmetrize_h(best_h)
        hse, cse, nse = eval_feasible(hs)
        best_h, best_c, best_n = try_update_values(hse, cse, nse, best_h, best_c, best_n)

        if best_n <= 60 and time.monotonic() < t_end - reserve - 70:
            hpol = best_h
            cpol = exact_c5(hpol)
            sl = max(0.00025, 0.0008 * cpol)
            x0 = np.concatenate([hpol, [cpol + sl]])
            h1 = solve_epigraph(x0, s_bound=cpol + sl + 0.06, maxiter=130)
            if h1 is not None:
                best_h, best_c, best_n = try_update(h1, best_h, best_c, best_n)
                hs2 = symmetrize_h(h1)
                hse2, cse2, nse2 = eval_feasible(hs2)
                best_h, best_c, best_n = try_update_values(hse2, cse2, nse2, best_h, best_c, best_n)

        if best_h is not None:
            hs3 = symmetrize_h(best_h)
            hse3, cse3, nse3 = eval_feasible(hs3)
            best_h, best_c, best_n = try_update_values(hse3, cse3, nse3, best_h, best_c, best_n)

    if best_h is None:
        best_h = adjust_sum_exact(np.full(48, 0.5, dtype=float), 24.0)
    h_ret = ensure_feasible(best_h)
    c_ret = exact_c5(h_ret)
    if not np.isfinite(c_ret):
        h_ret = adjust_sum_exact(np.full(48, 0.5, dtype=float), 24.0)
        c_ret = exact_c5(h_ret)
    return h_ret, float(c_ret), int(h_ret.size)
