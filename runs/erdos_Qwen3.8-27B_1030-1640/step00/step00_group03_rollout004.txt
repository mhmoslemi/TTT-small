def project_to_feasible(v, target_sum):
    """Project v onto {h : 0 <= h[i] <= 1, sum(h) = target_sum}"""
    n = len(v)
    # Bisection on t: find t such that sum(clamp(v - t, 0, 1)) = target_sum
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:

def project_to_feasible(v, target_sum):

        s = np.clip(v - mid, 0.0, 1.0).sum()

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(lam * (corr - m))
    log_sum = np.log(exp_vals.sum()) + m
    obj = log_sum / lam * dx
    
    p = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-18:

        else:
            t = -tau
            grad[t:] += w * (1.0 - h[:n-t])
            grad[:n-t] -= w * h[t:n]
    grad *= dx

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h_old)

import time

def project_to_feasible(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(v - t, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj_and_grad(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(lam * (corr - m))
    log_sum_exp = np.log(exp_vals.sum()) + m
    obj = log_sum_exp / lam * dx
    p = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-18:
            continue
        tau = k - (n - 1)
        if tau >= 0:
            grad[:n-tau] += w * (1.0 - h[tau:n])
            grad[tau:] -= w * h[:n-tau]
        else:
            t = -tau
            grad[t:] += w * (1.0 - h[:n-t])
            grad[:n-t] -= w * h[t:n]
    grad *= dx
    return obj, grad
