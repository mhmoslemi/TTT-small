import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def overlap_lag(h, k, n):
    lo = max(0, -k)
    hi = min(n, n - k)
    if lo >= hi:
        return 0.0
    idx = np.arange(lo, hi)
    return float(np.sum(h[idx] * (1.0 - h[idx + k]))) * (2.0 / n)

def repair_feasible(v, n):
    v = np.clip(v.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

        diff = target - s
        if abs(diff) < 1e-14:
            break

            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            v[mask] += diff / cnt
        else:
            room = v
            mask = room > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            v[mask] += diff / cnt

    return v

def dyadic_align(h, n):
    Q = 2**20
    target_units = n * Q // 2
    r = np.rint(h * Q).astype(np.int64)
    r = np.clip(r, 0, Q)
    cur = int(np.sum(r))
    diff = target_units - cur
    # Sort indices by how "easy" they are to adjust
    # For positive diff (need to increase): prefer elements with room
    # For negative diff (need to decrease): prefer elements with mass
    for _ in range(abs(diff) + n):
        if diff == 0:
            break
        if diff > 0:
            candidates = np.where(r < Q)[0]
            if len(candidates) == 0:
                break
            # Pick element with most room (furthest from 1)
            # to minimize disruption
            best = candidates[np.argmin(r[candidates])]
            step = min(diff, Q - r[best])
            r[best] += step
            cur += step
            diff -= step
        else:
            candidates = np.where(r > 0)[0]
            if len(candidates) == 0:
                break
            best = candidates[np.argmax(r[candidates])]
            step = min(-diff, r[best])
            r[best] -= step
            cur -= step
            diff += step
    if int(np.sum(r)) != target_units:
        # Fallback
        r = np.full(n, target_units // n, dtype=np.int64)
        rem = target_units - int(np.sum(r))
        for i in range(int(rem)):
            if i < n:
                r[i] += 1
    return r.astype(float) / Q

def obj_grad(z, *args):

    g = np.zeros(n + 1)
    g[0] = 1.0

def lag_constraint_fun(z, k, n):
    h = z[1:]
    lo = max(0, -k)
    hi = min(n, n - k)
    if lo >= hi:
        return 1.0  # trivially satisfied
    idx = np.arange(lo, hi)
    c_k = float(np.sum(h[idx] * (1.0 - h[idx + k]))) * (2.0 / n)
    return z[0] - c_k

def lag_constraint_jac(z, k, n):
    h = z[1:]
    dx = 2.0 / n
    lo = max(0, -k)
    hi = min(n, n - k)
    g = np.zeros(n + 1)
    g[0] = 1.0
    if lo >= hi:
        return g
    idx = np.arange(lo, hi)

    if lo2 < hi2:

def solve_local(h_init, n, active_lags, tau_init, max_iter=80, ftol=1e-9):
    """Solve the epigraph SLSQP problem for given active lags."""
    x0 = np.zeros(n + 1)
    x0[0] = tau_init
    x0[1:] = h_init
    
    bounds = [(-0.01, 2.0)] + [(0.0, 1.0)] * n
    
    constraints = []
    for k in active_lags:
        constraints.append({
            'type': 'ineq',
            'fun': lag_constraint_fun,
            'jac': lag_constraint_jac,
            'args': (k, n)
        })
    # Sum constraint (equality)
    constraints.append({
        'type': 'eq',
        'fun': sum_constraint_fun,
        'jac': sum_constraint_jac,
        'args': (n,)
    })
    
    res = minimize(
        obj_func, x0, jac=obj_grad,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': max_iter, 'ftol': ftol}
    )
    
    if res.success or res.fun < tau_init:
        h_sol = repair_feasible(res.x[1:], n)
        return h_sol, res.x[0]
    return None, tau_init

def sum_constraint_fun(z, n):
    return float(np.sum(z[1:])) - n / 2.0

def sum_constraint_jac(z, n):
    g = np.zeros(n + 1)
    g[1:] = 1.0
    return g

def find_active_lags(h, n, k_max=20):
    """Find the k_max lags with largest overlap."""
    lags = list(range(-(n-1), n))
    vals = []
    for k in lags:
        v = overlap_lag(h, k, n)
        vals.append((v, k))
    vals.sort(reverse=True)
    return [v[1] for v in vals[:k_max]]

    # Map each new point to the closest old point
    x_new = (np.arange(n_new) + 0.5) * dx_new
    cells_old = x_new / dx_old
    idx = np.clip(np.floor(cells_old).astype(int), 0, n_old - 1)
    h_new = h_old[idx]
    h_new = repair_feasible(h_new, n_new)
    return h_new

    h_new = h_old[idx].copy()
    return repair_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    end_time = start_time + budget_s * 0.92
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is None or len(h0) == 0:
        n0 = 45
        h0 = np.full(n0, 0.5)
    else:
        n0 = len(h0)

    h0 = repair_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_score = exact_c5(best_h)
    print(f"Initial: N={n0}, C5={best_score:.6f}")

    if time.time() > end_time - 120:
        best_h = dyadic_align(best_h, best_n, rng)
        best_score = exact_c5(best_h)
        return (best_h, best_score, best_n)

    # Phase 1: Smooth max optimization at n0
    for T_val in [30.0, 60.0, 100.0, 150.0]:
        if time.time() > end_time - 60:
            break
        h_sm = solve_smooth(best_h, best_n, T=T_val, max_iter=80)
        score = exact_c5(h_sm)
        if score < best_score - 1e-12:
            best_h = h_sm.copy()
            best_score = score
            print(f"  Smooth T={T_val}: C5={best_score:.6f}")

    # Phase 2: SLSQP epigraph with active set at n0
    for slsqp_iter in range(8):
        if time.time() > end_time - 60:
            break
        tau_init = best_score + 0.005
        active = find_active_lags(best_h, best_n, k_max=18)
        h_sol, tau_sol = solve_slsqp(best_h, best_n, active, tau_init,
                                     max_iter=80, ftol=1e-9)
        if h_sol is not None:
            score = exact_c5(h_sol)
            if score < best_score - 1e-12:
                best_h = h_sol.copy()
                best_score = score
                print(f"  SLSQP epigraph iter={slsqp_iter}: C5={best_score:.6f}")
            # Check for violators and expand active set
            all_lags = list(range(-(best_n - 1), best_n))
            violators = []
            for k in all_lags:
                if k not in active:
                    v = overlap_lag(best_h, k, best_n)
                    if v > best_score + 1e-5:
                        violators.append(k)
            if violators:
                active = list(set(active + violators))[:26]

    # Phase 3: Try different n values
    n_options = []
    for dn in [-8, -4, -2, 2, 4, 8, 12, 16]:
        nt = best_n + dn
        if nt >= 10:
            n_options.append(nt)
    # Deduplicate and sort by closeness to best_n
    n_options = sorted(set(n_options), key=lambda x: abs(x - best_n))

    for n_trial in n_options:
        if time.time() > end_time - 90:
            break
        if n_trial == best_n:
            continue
        h_start = transfer_to_n(best_h, best_n, n_trial)
        # Smooth max at new n
        h_sm = solve_smooth(h_start, n_trial, T=60.0, max_iter=60)
        score = exact_c5(h_sm)
        # SLSQP at new n
        if time.time() < end_time - 60:
            tau_init = exact_c5(h_sm) + 0.005
            active = find_active_lags(h_sm, n_trial, k_max=14)
            h_sol, tau_sol = solve_slsqp(h_sm, n_trial, active, tau_init,
                                         max_iter=50, ftol=1e-8)
            if h_sol is not None:
                score_sol = exact_c5(h_sol)
                if score_sol < score:
                    h_sm = h_sol
                    score = score_sol
        # Accept if better (prefer smaller n on ties)
        if score < best_score - 1e-10 or (abs(score - best_score) < 1e-10 and n_trial < best_n):
            best_h = h_sm.copy()
            best_n = n_trial
            best_score = score
            print(f"  N={n_trial}: C5={best_score:.6f}")

    # Phase 4: Random restarts at best_n
    for restart in range(5):
        if time.time() > end_time - 30:
            break
        h_pert = best_h + rng.randn(best_n) * 0.08
        h_pert = repair_feasible(h_pert, best_n)
        h_sm = solve_smooth(h_pert, best_n, T=80.0, max_iter=60)
        score = exact_c5(h_pert)
        if score < best_score - 1e-12:
            h_pert = h_pert
        tau_init = exact_c5(h_pert) + 0.005
        active = find_active_lags(h_pert, best_n, k_max=14)
        h_sol, tau_sol = solve_slsqp(h_pert, best_n, active, tau_init,
                                     max_iter=50, ftol=1e-8)
        if h_sol is not None:
            score2 = exact_c5(h_sol)
            if score2 < score:
                h_pert = h_sol
                score = score2
        if score < best_score - 1e-12:
            best_h = h_pert.copy()
            best_score = score
            print(f"  Restart {restart}: C5={best_score:.6f}")

    # Final: one more round of refinement at best_n
    if time.time() < end_time - 20:
        h_sm = solve_smooth(best_h, best_n, T=200.0, max_iter=80)
        score = exact_c5(h_sm)
        if score < best_score - 1e-12:
            best_h = h_sm.copy()
            best_score = score
            print(f"  Final smooth: C5={best_score:.6f}")

        tau_init = best_score + 0.003
        active = find_active_lags(best_h, best_n, k_max=22)
        h_sol, tau_sol = solve_slsqp(best_h, best_n, active, tau_init,
                                     max_iter=60, ftol=1e-10)
        if h_sol is not None:
            score = exact_c5(h_sol)
            if score < best_score - 1e-12:
                best_h = h_sol.copy()
                best_score = score
                print(f"  Final SLSQP: C5={best_score:.6f}")

    # Dyadic alignment
    best_h_dyad = dyadic_align(best_h, best_n, rng)
    score_dyad = exact_c5(best_h_dyad)
    # Keep the better of dyadic and non-dyadic
    if score_dyad <= best_score + 5e-5:
        best_h = best_h_dyad
        best_score = score_dyad
    else:
        # Use non-dyadic but ensure sum is as close as possible
        best_h = repair_feasible(best_h, best_n)
        best_score = exact_c5(best_h)

    # Final validation
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(float(np.sum(best_h)) - best_n / 2.0) < 1e-9

    print(f"Final: N={best_n}, C5={best_score:.6f}")
    return (best_h, best_score, best_n)
