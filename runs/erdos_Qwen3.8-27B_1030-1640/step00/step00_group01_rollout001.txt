import numpy as np
from scipy.optimize import minimize
import time

def score_exact(h):
    """Compute the exact C5 score."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_objective_and_grad(x, tau):
    """Compute the smooth log-sum-exp objective and its gradient."""
    n = len(x)
    dx = 2.0 / n
    q = 1.0 - x
    corr = np.correlate(x, q, mode='full')  # length 2n-1
    a = corr * dx / tau
    a_max = np.max(a)
    exp_vals = np.exp(a - a_max)
    Z = np.sum(exp_vals)
    F = tau * (a_max + np.log(Z))
    p = exp_vals / Z  # softmax weights, length 2n-1
    
    # Gradient computation
    # dF/dx[k] = dx * sum_m p[m] * dV_m/dx[k]
    # where V_m = corr[m + n - 1] (raw correlation at lag m = m_idx - (n-1))
    # dV_m/dx[k] = (1-x[k-m]) * [0<=k-m<n] - x[k+m] * [0<=k+m<n]
    
    G1 = np.zeros(n)
    G2 = np.zeros(n)
    for m_idx in range(2 * n - 1):
        m = m_idx - (n - 1)
        pm = p[m_idx]
        if pm < 1e-15:
            continue
        # First term: (1-x[k-m]) for 0 <= k-m < n
        k_lo = max(0, m)
        k_hi = min(n - 1, m + n - 1)
        if k_lo <= k_hi:
            G1[k_lo:k_hi+1] += pm * (1.0 - x[k_lo - m:k_hi - m + 1])
        # Second term: x[k+m] for 0 <= k+m < n
        k_lo2 = max(0, -m)
        k_hi2 = min(n - 1, n - 1 - m)
        if k_lo2 <= k_hi2:
            G2[k_lo2:k_hi2+1] += pm * x[k_lo2 + m:k_hi2 + m + 1]
    
    grad = dx * (G1 - G2)
    return F, grad

def project_box_sum(z, target):
    """Project z onto {x : 0 <= x_i <= 1, sum(x) = target} using bisection."""
    n = len(z)
    # Find alpha such that sum(clip(z - alpha, 0, 1)) = target
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix sum exactly
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - result
            mask = room > 1e-10
            if mask.any():
                add = diff / room[mask].sum()
                result[mask] = np.minimum(result[mask] + add, 1.0)
        else:
            mask = result > 1e-10
            if mask.any():
                sub = (-diff) / result[mask].sum()
                result[mask] = np.maximum(result[mask] - sub, 0.0)
    
    # Final clip
    result = np.clip(result, 0.0, 1.0)
    return result

def repair_sum(h, target, tol=1e-10):
    """Repair the sum of h to be exactly target."""
    diff = target - h.sum()
    if abs(diff) <= tol:
        return h
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-8

            h[mask] += diff / h[mask].sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, tau):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')

    p = exp_a / Z
    
    G1 = np.zeros(n)
    G2 = np.zeros(n)

def subgradient_polish(h, target, n_steps=100, lr=0.01):

    n = len(h)
    dx = 2.0 / n
    h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        m_star = np.argmax(corr)
        m = m_star - (n - 1)

                val -= h[k + m]

def subgradient_polish(h, target, n_steps=200, lr=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_star = np.argmax(corr)
        m = m_star - (n - 1)
        
        sg = np.zeros(n)
        # (1-h[k-m]) for valid k
        if m >= 0:
            # k-m in [0,n-1] means k in [m, m+n-1], intersect [0,n-1] = [m, n-1]
            if m < n:
                sg[m:n] += (1.0 - h[0:n-m])
        else:
            # k-m = k+|m| in [0,n-1] means k in [0, n-1-|m|]
            t = -m
            sg[0:n-t] += (1.0 - h[t:n])
        
        # -h[k+m] for valid k
        if m >= 0:
            # k+m in [0,n-1] means k in [0, n-1-m]
            if m < n:
                sg[0:n-m] -= h[m:n]
        else:
            # k+m = k-|m| in [0,n-1] means k in [|m|, n-1]
            t = -m
            sg[t:n] -= h[0:n-t]
        
        sg *= dx
        h = project_box_sum(h - lr * sg, target)
    
    return h
