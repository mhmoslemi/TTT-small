import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def thue_morse(n, num_ones):
    """Generate a Thue-Morse-like balanced binary sequence of length n with num_ones ones."""
    # Generate Thue-Morse sequence
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        # Count bits
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        tm[i] = c % 2
    
    # If we need exactly num_ones ones, adjust
    ones_count = tm.sum()
    if ones_count > num_ones:
        # Flip some 1s to 0s (choose the ones that are most "surrounded by 1s")
        one_positions = np.where(tm == 1)[0]
        # Pick the ones to flip
        to_flip = one_positions[len(one_positions)-num_ones:]
        tm[to_flip] = 0
    elif ones_count < num_ones:
        zero_positions = np.where(tm == 0)[0]
        to_flip = zero_positions[:num_ones - ones_count]
        tm[to_flip] = 1
    
    return tm.astype(float)

def evenly_spaced(n, num_ones):
    """Place num_ones ones at evenly spaced positions."""
    h = np.zeros(n)
    positions = np.linspace(0, n-1, num_ones, dtype=int)
    h[positions] = 1
    return h

                h[j] = 1.0

                h[j] = 0.0

    return float(np.max(c) * dx)

def make_balanced_binary(n, start_pattern=None, seed=42):
    """Create a balanced binary sequence of length n (must be even)."""
    num_ones = n // 2
    if start_pattern is not None:
        h = start_pattern[:n].astype(float)
        if len(h) < n:
            h = np.pad(h, (0, n-len(h)))
        # Binarize: keep top n/2 as 1
        sorted_idx = np.argsort(h)[::-1]
        result = np.zeros(n)
        result[sorted_idx[:num_ones]] = 1.0
        return result
    
    # Default: use Thue-Morse
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = c % 2
    # Ensure exactly n/2 ones
    ones_count = tm.sum()
    if ones_count != num_ones:
        # Adjust
        if ones_count > num_ones:
            one_pos = np.where(tm == 1)[0]
            tm[one_pos[-(ones_count-num_ones):]] = 0
        else:
            zero_pos = np.where(tm == 0)[0]
            tm[zero_pos[:num_ones-ones_count]] = 1
    return tm.astype(float)

def greedy_optimize(h, max_rounds=5000, deadline=None):
    """Greedy: try all swaps, take best improvement each round."""
    n = len(h)

    current_c = evaluate_c5(h)

        if deadline is not None and time.perf_counter() > deadline:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c_new = evaluate_c5(h)
                imp = current_c - c_new
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (int(i), int(j))
                h[i] = 1.0

                best_c = c_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c

    n_src = len(h)
    if n_src == n_target:
        return h.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    return np.interp(x_tgt, x_src, h)

import time

def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def check_feasible(h, tol=1e-9):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    target = n / 2.0
    if abs(h.sum() - target) > 1e-6 * max(1, target):
        return False
    return True

def project_feasible(h):
    """Project h to the feasible set [0,1]^n with sum = n/2."""
    h = h.copy()
    n = len(h)
    target = n / 2.0
    
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    
    # If sum is already correct, done
    if abs(h.sum() - target) < 1e-12:
        return h
    
    # Binary search for lambda such that sum(clip(h + lambda)) = target
    # clip(h + lambda) is monotonically increasing in lambda
    lo = -max(h.max() - 0, 0) - 2
    hi = max(1.0 - h.min(), 0) + 2
    
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(h + mid, 0.0, 1.0).sum()
        if val < target:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Fix any remaining rounding error
    diff = target - h_proj.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference
        if diff > 0:
            # Need to add more; add to entries that are < 1
            adjustable = h_proj < 1.0
            n_adj = adjustable.sum()
            if n_adj > 0:
                h_proj[adjustable] += diff / n_adj
                h_proj = np.clip(h_proj, 0.0, 1.0)
        else:
            adjustable = h_proj > 0.0
            n_adj = adjustable.sum()
            if n_adj > 0:
                h_proj[adjustable] += diff / n_adj
                h_proj = np.clip(h_proj, 0.0, 1.0)
    
    return h_proj

def thue_morse_binary(n):
    """Generate Thue-Morse based binary sequence of length n with n/2 ones (n even)."""
    num_ones = n // 2
    tm = np.zeros(n, dtype=np.float64)
    for i in range(n):
        cnt = bin(i).count('1')
        tm[i] = float(cnt % 2)
    
    # Adjust to have exactly n/2 ones
    ones = np.where(tm == 1.0)[0]
    zeros = np.where(tm == 0.0)[0]
    if len(ones) > num_ones:
        # Flip some 1s to 0s
        for idx in ones[num_ones:]:
            tm[idx] = 0.0
    elif len(ones) < num_ones:
        for idx in zeros[:num_ones - len(ones)]:
            tm[idx] = 1.0
    
    return tm

def evenly_spaced_binary(n, offset=0):
    """Generate evenly spaced binary sequence."""
    h = np.zeros(n)
    num_ones = n // 2
    step = n / num_ones  # = 2 for balanced
    for k in range(num_ones):
        pos = int((k + offset) * step) % n
        h[pos] = 1.0
    # Ensure exactly num_ones ones (handle collisions)
    while h.sum() < num_ones:
        # Find a zero position and set to 1
        zeros = np.where(h == 0)[0]
        if len(zeros) == 0:
            break
        h[zeros[0]] = 1.0
    return h

def random_binary(n, seed=42):
    """Generate random balanced binary sequence."""
    rng = np.random.default_rng(seed)
    num_ones = n // 2
    positions = rng.choice(n, size=num_ones, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h
