def _adjust_sum(y, S):
    y = np.asarray(y, dtype=float).copy()
    S = float(S)
    n = y.size
    if n == 0: return y
    for _ in range(8):
        res = S - float(y.sum())
        if abs(res) < 1e-14: break
        if res > 0.0:
            order = np.argsort(1.0 - y)[::-1]
            for idx in order:
                room = 1.0 - y[idx]
                if room <= 0.0: continue
                step = min(res, room)
                if step <= 1e-16: continue
                y[idx] += step
                res -= step
                if abs(res) < 1e-14: break
        else:
            order = np.argsort(y)[::-1]
            for idx in order:
                if y[idx] <= 0.0: continue
                step = min(-res, y[idx])
                if step <= 1e-16: continue
                y[idx] -= step
                res += step
                if abs(res) < 1e-14: break
    return np.clip(y, 0.0, 1.0)

import time
import numpy as np
from scipy import optimize as _optimize

_BEST_H = None
_BEST_C = float("inf")
_BEST_N = 0
_CURRENT_TAU = 1e-3
_EP_N = 0
_EP_INDICES = None
_PAIR_CACHE = {}


def _overlap_components(h):
    n = len(h)
    dx = 2.0 / float(n)
    return dx * np.correlate(h, 1.0 - h, mode="full")


def _c5(h):
    return float(np.max(_overlap_components(h)))


def _project_sum_box(v, S):
    x = np.asarray(v, dtype=float)
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    S = float(S)
    if S <= 1e-15:
        return np.zeros(n, dtype=float)
    if S >= float(n) - 1e-15:
        return np.ones(n, dtype=float)

    lo = -100.0
    hi = 100.0
    for _ in range(60):
        flo = np.clip(x - lo, 0.0, 1.0).sum()
        fhi = np.clip(x - hi, 0.0, 1.0).sum()
        if flo >= S and fhi <= S:
            break
        if flo < S:
            lo -= 100.0
        if fhi > S:
            hi += 100.0

    y = np.zeros(n, dtype=float)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        sm = y.sum()
        if sm > S:
            lo = mid
        else:
            hi = mid
    return y


def _adjust_sum(y, S):
    y = np.asarray(y, dtype=float).copy()
    S = float(S)
    n = y.size
    if n == 0:
        return y
    for _ in range(10):
        res = S - float(y.sum())
        if abs(res) < 1e-14:
            break
        if res > 0.0:
            order = np.argsort(1.0 - y)[::-1]
            for idx in order:
                room = 1.0 - float(y[idx])
                if room <= 0.0:
                    continue
                step = min(res, room)
                if step <= 1e-16:
                    continue
                y[idx] += step
                res -= step
                if abs(res) < 1e-14:
                    break
        else:
            order = np.argsort(y)[::-1]
            for idx in order:
                room = float(y[idx])
                if room <= 0.0:
                    continue
                step = min(-res, room)
                if step <= 1e-16:
                    continue
                y[idx] -= step
                res += step
                if abs(res) < 1e-14:
                    break
    return np.clip(y, 0.0, 1.0)


def _make_feasible(h):
    x = np.asarray(h, dtype=float)
    if x.size == 0:
        return None
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    S = 0.5 * x.size
    y = _project_sum_box(x, S)
    y = _adjust_sum(y, S)
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(y.sum()) - S) > 1e-9:
        y = np.full(x.size, 0.5, dtype=float)
        y = _adjust_sum(y, S)
    return y

def _get_pairs(n):
    n = int(n)
    cache = _PAIR_CACHE
    if n in cache:
        return cache[n]
    L = 2 * n - 1
    pairs = []
    for m in range(L):
        i0 = max(0, m - n + 1)
        i1 = min(m, n - 1)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1, dtype=int)
            j = idx + (n - 1 - m)
        else:
            idx = np.empty(0, dtype=int)
            j = np.empty(0, dtype=int)
        pairs.append((idx, j))
    cache[n] = pairs
    return cache[n]


def _overlap_grad_matrix(h):
    n = len(h)
    pairs = _get_pairs(n)
    G = np.zeros((len(pairs), n), dtype=float)
    dx = 2.0 / float(n)
    for m in range(len(pairs)):
        idx, j = pairs[m]
        if idx.size:
            row = G[m]
            row[idx] += dx * (1.0 - h[j])
            row[j] -= dx * h[idx]
    return G


def _soft_objective(h):
    tau = float(_CURRENT_TAU)
    if tau <= 0.0:
        tau = 1e-6
    c = _overlap_components(h)
    s = c / tau
    m = s.max()
    z = np.exp(s - m)
    zsum = z.sum()
    if zsum <= 0.0 or not np.isfinite(zsum):
        return float(c.max())
    return float(tau * (m + np.log(zsum)))


def _soft_grad(h):
    tau = float(_CURRENT_TAU)
    if tau <= 0.0:
        tau = 1e-6
    c = _overlap_components(h)
    s = c / tau
    m = s.max()
    z = np.exp(s - m)
    zsum = z.sum()
    if zsum <= 0.0 or not np.isfinite(zsum):
        return np.zeros(len(h), dtype=float)
    p = z / zsum
    n = len(h)
    g = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    pairs = _get_pairs(n)
    tol = 1e-15
    for mi in range(len(pairs)):
        w = float(p[mi])
        if w > tol:
            idx, j = pairs[mi]
            if idx.size:
                g[idx] += w * dx * (1.0 - h[j])
                g[j] -= w * dx * h[idx]
    return g


def _sum_eq_fun(h):
    return float(np.sum(h) - 0.5 * len(h))


def _sum_eq_jac(h):
    return np.ones(len(h), dtype=float)


def _ep_sum_eq_fun(x):
    n = int(_EP_N)
    return float(np.sum(x[:n]) - 0.5 * n)


def _ep_sum_eq_jac(x):
    g = np.zeros(len(x), dtype=float)
    g[:int(_EP_N)] = 1.0
    return g


def _make_bounds(n):
    bounds = []
    for _ in range(int(n)):
        bounds.append((0.0, 1.0))
    return bounds


def _solve_soft(h0, tau, maxiter):
    global _CURRENT_TAU
    h = _make_feasible(h0)
    if h is None:
        return None
    _CURRENT_TAU = float(tau)
    n = len(h)
    try:
        res = _optimize.minimize(
            _soft_objective,
            h,
            jac=_soft_grad,
            method="SLSQP",
            bounds=_make_bounds(n),

            options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": False},
        )
    except Exception:
        return None
    if not hasattr(res, "x"):
        return None
    return _make_feasible(res.x)


def _ep_objective(x):
    return float(x[-1])


def _ep_jac(x):
    g = np.zeros(len(x), dtype=float)
    g[-1] = 1.0
    return g

    h = x[:n]

    h = x[:n]
    idx = _EP_INDICES
    G = _overlap_grad_matrix(h)
    J = -G[idx].copy()
    J[:, -1] = 1.0
    return J


def _solve_sub_epigraph(h0, indices, slack, maxiter):

    h = _make_feasible(h0)
    if h is None:
        return None
    n = len(h)
    L = 2 * n - 1

    indices = indices[(indices >= 0) & (indices < L)]

    if indices.size == 0:

    cmax = _c5(h)
    s0 = max(1e-12, cmax + float(slack))
    if not np.isfinite(s0):
        return None

    _EP_N = n
    _EP_INDICES = indices
    x = np.empty(n + 1, dtype=float)
    x[:n] = h
    x[n] = s0
    bounds = _make_bounds(n)
    bounds.append((0.0, max(2.0, s0 + 1.0)))
    try:

            _ep_objective,

            jac=_ep_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=[
                {"type": "ineq", "fun": _ep_ineq_fun, "jac": _ep_ineq_jac},
                {"type": "eq", "fun": _ep_sum_eq_fun, "jac": _ep_sum_eq_jac},
            ],
            options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": False},
        )
    except Exception:
        return None
    if not hasattr(res, "x"):
        return None
    return _make_feasible(res.x[:n])


def _top_indices(h, k):
    c = _overlap_components(h)
    L = len(c)
    k = int(max(1, min(k, L)))
    order = np.argsort(c, kind="mergesort")[::-1]
    return order[:k].astype(int, copy=False)


def _soft_maxiter(n):

    return max(30, min(100, int(4500.0 / float(n))))


def _ep_maxiter(n):

def _elapsed(t0):
    return float(time.monotonic()) - float(t0)


def _sequ_soft(h0, taus, maxiter, t0, budget_s, reserve):

            return
        slack = float(slacks[si])
        k = int(ks[si]) if si < len(ks) else int(ks[-1])
        h = _BEST_H.copy()
        for _ in range(3):
            if _elapsed(t0) > budget - reserve:
                return
            k_use = k
            if k_use < 0:
                L = 2 * _BEST_N - 1
                k_use = L if L <= 100 else 80
            idx = _top_indices(h, k_use)
            hnew = _solve_sub_epigraph(h, idx, slack, maxiter)
            if hnew is None:
                continue
            old_c = float(_BEST_C)
            _consider(hnew)
            if float(_BEST_C) < old_c - 1e-12:
                h = _BEST_H.copy()
            else:
                break


def _resample(h, new_n):
    new_n = int(new_n)
    old_n = len(h)
    if new_n <= 0:
        return np.empty(0, dtype=float)
    if old_n <= 0:
        return np.full(new_n, 0.5, dtype=float)
    if old_n == new_n:
        return np.array(h, copy=True)
    x_old = (np.arange(old_n) + 0.5) * (2.0 / float(old_n))
    x_new = (np.arange(new_n) + 0.5) * (2.0 / float(new_n))
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def _sequ_resample(ns, taus, t0, budget_s, reserve, do_ep):
    if _BEST_H is None:
        return
    budget = float(budget_s)
    reserve = float(reserve)
    for nt in ns:
        nt = int(nt)
        if nt <= 0:
            continue
        if _elapsed(t0) > budget - reserve - 25.0:
            return
        if nt > _BEST_N:
            y = _resample(_BEST_H, nt)
            _consider(y)
            _sequ_soft(_BEST_H, taus, _soft_maxiter(nt), t0, budget_s, reserve)
            if do_ep and nt <= 64:
                if _elapsed(t0) <= budget - reserve - 50.0:
                    _sequ_ep((0.005, 0.002), (20, -1), _ep_maxiter(nt), t0, budget_s, reserve)


def _wave_profile(n, rng):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    x = (np.arange(n) + 0.5) / float(n)
    f1 = int(rng.integers(1, 9))
    ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
    amp1 = float(rng.uniform(0.10, 0.42))
    f2 = int(rng.integers(2, 9))
    ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
    amp2 = float(rng.uniform(0.03, 0.09))
    y = 0.5 + amp1 * np.cos(2.0 * np.pi * f1 * x + ph1)
    y += amp2 * np.cos(2.0 * np.pi * f2 * x + ph2)
    y += 0.02 * rng.standard_normal(n)
    return y


def _binary_profile(n, rng):
    n = int(n)
    y = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = rng.permutation(n)[:k]
        y[idx] = 1.0
    if n % 2 == 1:
        cand = np.where(y < 1.0)[0]
        if cand.size > 0:
            pos = int(rng.integers(0, cand.size))
            y[cand[pos]] += 0.5
    return y


def _sequ_random(count, target_n, taus, maxiter, t0, budget_s, reserve, rng):
    if _BEST_H is None:
        return
    budget = float(budget_s)
    reserve = float(reserve)
    for i in range(int(count)):
        if _elapsed(t0) > budget - reserve - 20.0:
            return
        nt = int(target_n)
        if _BEST_N > 0:
            nt = _BEST_N
        mode = i % 4
        if mode == 0:
            sigma = 0.01 + 0.015 * (i % 4)
            h = _BEST_H + sigma * rng.standard_normal(nt)
        elif mode == 1:
            h = _wave_profile(nt, rng)
        elif mode == 2:
            h = _binary_profile(nt, rng)
        else:
            h = _resample(_BEST_H, nt) + 0.03 * rng.standard_normal(nt)
        _consider(h)
        _sequ_soft(_BEST_H, taus, maxiter, t0, budget_s, reserve)


def run(seed=42, budget_s=1100, **kwargs):
    global _BEST_H, _BEST_C, _BEST_N
    t0 = float(time.monotonic())
    rng = np.random.default_rng(seed)

    _BEST_H = None
    _BEST_C = float("inf")
    _BEST_N = 0

    parent = globals().get("initial_h_values")
    if parent is not None:
        p = np.asarray(parent, dtype=float)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            _consider(p)

    if _BEST_H is None:
        _consider(np.full(47, 0.5, dtype=float))

    if _BEST_H is not None:
        _consider(0.5 * (_BEST_H + _BEST_H[::-1]))

    if float(budget_s) - _elapsed(t0) < 60.0:
        return _finalize_best()

    _sequ_soft(
        _BEST_H,
        (0.08, 0.035, 0.015, 0.006, 0.0025, 0.001, 0.0004),
        _soft_maxiter(_BEST_N),
        t0,
        budget_s,
        45.0,
    )

    _sequ_ep(
        (0.02, 0.008, 0.003, 0.0012, 0.0005),
        (20, 25, 35, -1, -1),
        _ep_maxiter(_BEST_N),
        t0,
        budget_s,
        60.0,
    )

    _sequ_random(
        8,

        t0,
        budget_s,
        50.0,

    _sequ_resample(

        t0,

        70.0,
        True,

    if _elapsed(t0) <= float(budget_s) - 120.0:

            (30, 50, -1),
            _ep_maxiter(_BEST_N),
            t0,
            budget_s,
            60.0,

    if _elapsed(t0) <= float(budget_s) - 60.0:

            _soft_maxiter(_BEST_N),
            t0,

            45.0,
            rng,
        )

    if _elapsed(t0) <= float(budget_s) - 35.0:
        _sequ_soft(
            _BEST_H,
            (0.002, 0.0008, 0.0003),
            _soft_maxiter(_BEST_N),
            t0,
            budget_s,
            30.0,
        )

    return _finalize_best()
