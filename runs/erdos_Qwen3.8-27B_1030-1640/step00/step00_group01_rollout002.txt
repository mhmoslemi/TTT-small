            n_list.append(nt)
    n_list.sort()

    tau_scheds = {
        'full': [0.1, 0.05, 0.02, 0.008, 0.003, 0.001],
        'med': [0.05, 0.02, 0.008, 0.003, 0.001],
        'short': [0.02, 0.008, 0.003, 0.001],
    }

    for n in n_list:
        if elapsed() > budget_s - 50:
            break
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        time_left = budget_s - elapsed() - 40
        # Allocate time based on how many resolutions remain
        frac = max(0.15, min(0.4, time_left * 0.3 / (sum(1 for x in n_list if n_list.index(x) >= n_list.index(n)) * 1.0)))
        n_starts = 4 if n <= 100 else 3
        tau_sched = tau_scheds['full'] if time_left > 200 else (tau_scheds['med'] if time_left > 100 else tau_scheds['short'])
        max_iter_each = 200 if n <= 100 else 120

        # Build starts
        starts = []
        if best_h is not None:
            if len(best_h) == n:
                starts.append(best_h.copy())
            else:
                x_old = np.linspace(0, 2, len(best_h), endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                interp = np.interp(x_new, x_old, best_h)
                starts.append(project_box_sum(interp, target))

        # Random
        for _ in range(2):
            z = rng.random(n)
            starts.append(project_box_sum(z, target))

        # Binary
        ones_count = int(round(n / 2.0))
        binary = np.zeros(n)
        idx = rng.choice(n, ones_count, replace=False)
        binary[idx] = 1.0
        starts.append(binary)

        # Smooth
        x = np.linspace(0, 2, n, endpoint=False)
        z = 0.5 + 0.35 * np.sin(np.pi * 3 * x + rng.uniform(0, 2 * np.pi))
        starts.append(project_box_sum(z, target))

        for x0 in starts[:n_starts]:
            if elapsed() > budget_s - 25:
                break
            x_res, s_res = _try_optimize(x0, n, target, bounds, tau_sched, max_iter_each, t0, budget_s)
            if s_res < best_score:
                best_score = s_res
                best_h = x_res.copy()

    # Final polish on best
    if best_h is not None and elapsed() < budget_s - 20:
        n_best = len(best_h)
        target = n_best / 2.0
        bounds = [(0.0, 1.0)] * n_best

        # Very fine tau
        x_pol, s_pol = _try_optimize(best_h, n_best, target, bounds,
                                     [0.001, 0.0005, 0.0002], 250, t0, budget_s)
        if s_pol < best_score:
            best_score = s_pol
            best_h = x_pol

    # Local refine
    if best_h is not None and elapsed() < budget_s - 10:
        n_best = len(best_h)
        target = n_best / 2.0
        best_h, best_score = _local_refine(best_h, target, t0, budget_s, max_rounds=3)

    # Try binarization (round values > 0.5 to 1, < 0.5 to 0, keep some fractional)
    if best_h is not None and elapsed() < budget_s - 5:
        n_best = len(best_h)
        target = n_best / 2.0
        # Try: round to nearest 0.1
        h_round = np.round(best_h * 10.0) / 10.0
        h_round = project_box_sum(h_round, target)
        s_round = score_exact(h_round)
        if s_round < best_score:
            best_score = s_round
            best_h = h_round

        # Try: hard binary
        h_bin = (best_h > 0.5).astype(np.float64)
        # Fix count
        ones = int(h_bin.sum())
        target_ones = int(round(n_best / 2.0))
        if ones > target_ones:
            # Remove some ones (those with lowest "importance")
            idx5 = np.where(h_bin == 1)[0]
            order = np.argsort(-best_h[idx5])  # remove those closest to 0.5 first
            for oi in order[:ones - target_ones]:
                h_bin[idx5[oi]] = 0.0
        elif ones < target_ones:
            idx0 = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[idx0])  # promote those closest to 0.5
            for oi in order[:target_ones - ones]:
                h_bin[idx0[oi]] = 1.0
        s_bin = score_exact(h_bin)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin

    if best_h is None:
        n = 63
        best_h = np.full(n, 0.5)
        best_score = score_exact(best_h)

    # Final validation
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_box_sum(best_h, target_final)
    best_score = score_exact(best_h)

    return best_h, best_score, n_final
