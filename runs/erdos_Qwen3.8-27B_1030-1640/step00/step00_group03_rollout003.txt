import time

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += (diff / s) * room
            else:
                h[:] = 0.5
        else:
            s = h.sum()
            if s > 1e-14:
                h += (diff / s) * h
            else:
                h[:] = 0.5
        h = np.clip(h, 0.0, 1.0)
        # Final correction
        resid = target - h.sum()
        if abs(resid) > 1e-14:
            idx = np.argmax(1.0 - h) if resid > 0 else np.argmax(h)
            if resid > 0 and h[idx] < 1.0:
                h[idx] += min(resid, 1.0 - h[idx])
            elif resid < 0 and h[idx] > 0.0:
                h[idx] += max(resid, -h[idx])
    return h

                h += (diff / s) * room
        else:
            s = h.sum()
            if s > 1e-13:
                h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
        r = target - h.sum()
        if abs(r) > 1e-13:
            if r > 0:
                order = np.argsort(1.0 - h)[::-1]
            else:
                order = np.argsort(h)[::-1]
            for i in order:
                if r > 0:
                    adj = min(r, 1.0 - h[i])
                    h[i] += adj
                    r -= adj
                else:
                    adj = max(r, -h[i])
                    h[i] += adj
                    r -= adj
                if abs(r) < 1e-15:
                    break
    return h

def _make_feasible(n, rng):
    h = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    h += rng.uniform(-0.3, 0.3, n)
    return _repair(h, n)

def _transfer(n_new, h_old):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return _repair(h_new, n_new)

def _obj_z(z):
    return z[0]

def _obj_jac_z(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[0] = 1.0
    return g

def _eq_fun(z):
    return z[1:].sum() - len(z[1:]) * 0.5

def _eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[1:] = 1.0
    return g

def _lag_val(h, k, dx):
    n = len(h)
    lo = max(0, -k)
    hi = min(n, n - k)
    if lo >= hi:
        return 0.0
    idx = np.arange(lo, hi)
    return dx * float(np.sum(h[idx + k] * (1.0 - h[idx])))

def _ineq_fun(z, k, dx):
    return _lag_val(z[1:], k, dx) - z[0]

def _ineq_jac(z, k, dx):
    h = z[1:]
    n = len(h)
    lo = max(0, -k)
    hi = min(n, n - k)
    g = np.zeros(n + 1)
    g[0] = -1.0
    ja_lo = max(0, lo + k)
    ja_hi = min(n, hi + k)
    if ja_lo < ja_hi:
        ja = np.arange(ja_lo, ja_hi)
        g[1:][ja] += (1.0 - h[ja - k]) * dx
    if lo < hi:
        jb = np.arange(lo, hi)
        g[1:][jb] -= h[jb + k] * dx
    return g

def _top_lags(h, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_lags = 2 * n - 1
    K = min(K, n_lags)
    top_idx = np.argsort(corr)[-K:]
    lags = (top_idx - (n - 1)).astype(int)
    return sorted(set(lags.tolist()))

def _solve_epigraph(h0, n, lags, maxiter=100, ftol=1e-9):
    dx = 2.0 / n
    x0 = np.zeros(n + 1)
    x0[0] = _c5(h0) + 0.003
    x0[1:] = h0
    cons = []
    for k in lags:
        cons.append({'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac,
                     'args': (k, dx)})
    cons.append({'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac})
    bounds = [(-0.5, 2.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_z, x0, method='SLSQP', jac=_obj_jac_z,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _repair(res.x[1:].copy(), n)
            return h_new
    except Exception:
        pass
    return h0

def _subgradient_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = int(max_idx - (n - 1))
    lo = max(0, -k)
    hi = min(n, n - k)
    idx = np.arange(lo, hi)
    grad = np.zeros(n)
    ja_lo = max(0, lo + k)
    ja_hi = min(n, hi + k)
    ja = np.arange(ja_lo, ja_hi)
    grad[ja] += (1.0 - h[ja - k])
    grad[idx] -= h[idx + k]
    grad *= dx
    h_new = h - alpha * grad
    h_new -= (h_new.sum() - 0.5 * n) / n
    h_new = _repair(h_new, n)
    return h_new

def _solve_sumsq(h0, n, maxiter=80, ftol=1e-9):
    """Minimize sum of C_k^2 as proxy"""
    dx = 2.0 / n

    def _ss_obj(z):
        h = z
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return float(np.sum(corr * corr))

    def _ss_jac(z):
        h = z
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        g = np.zeros(n)
        n_lags = 2 * n - 1
        for m in range(n_lags):
            k = m - (n - 1)
            if corr[m] == 0.0:
                continue
            lo = max(0, -k)
            hi = min(n, n - k)
            idx = np.arange(lo, hi)
            ja_lo = max(0, lo + k)
            ja_hi = min(n, hi + k)
            ja = np.arange(ja_lo, ja_hi)
            g[ja] += 2.0 * corr[m] * (1.0 - h[ja - k])
            g[idx] += 2.0 * corr[m] * (-h[idx + k])
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(_ss_obj, h0, method='SLSQP', jac=_ss_jac,
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - 0.5*n,
                                  'jac': lambda z: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return _repair(res.x.copy(), n)
    return h0

# Need top-level equivalents for the sumsq constraints
def _ss_eq_fun(z):
    return z.sum() - len(z) * 0.5

def _ss_eq_jac(z):
    return np.ones(len(z))

def _ss_obj_func(z):
    h = z
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(corr * corr))

def _ss_jac_func(z):
    h = z
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    g = np.zeros(n)
    n_lags = 2 * n - 1
    for m in range(n_lags):
        k = m - (n - 1)
        cm = corr[m]
        if cm == 0.0:
            continue
        lo = max(0, -k)
        hi = min(n, n - k)
        idx = np.arange(lo, hi)
        ja_lo = max(0, lo + k)
        ja_hi = min(n, hi + k)
        ja = np.arange(ja_lo, ja_hi)
        g[ja] += 2.0 * cm * (1.0 - h[ja - k])
        g[idx] -= 2.0 * cm * h[idx + k]
    return g

def _solve_sumsq2(h0, n, maxiter=80, ftol=1e-9):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ss_eq_fun, 'jac': _ss_eq_jac}]
    try:
        res = minimize(_ss_obj_func, h0, method='SLSQP', jac=_ss_jac_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x.copy(), n)
    except Exception:
        pass
    return h0

def _optimize_sequence(h_init, n, min_time, deadline, rng):
    h_best = h_init.copy()
    c5_best = _c5(h_best)
    t_start = time.time()

    # Phase 1: Epigraph SLSQP with growing active set
    K_start = 10
    K_max = 2 * n - 1
    h_cur = h_best.copy()
    while time.time() < deadline:
        elapsed = time.time() - t_start
        if elapsed > min_time:
            break
        K = min(K_start + int(elapsed * 2), K_max)
        lags = _top_lags(h_cur, K)
        h_new = _solve_epigraph(h_cur, n, lags, maxiter=60 + K, ftol=1e-10)
        c5_new = _c5(h_new)
        if c5_new < c5_best - 1e-11:
            c5_best = c5_new
            h_best = h_new.copy()
        h_cur = h_new
        # Check for convergence
        if K >= K_max:
            break
    h_cur = h_best.copy()

    # Phase 2: Subgradient polishing
    alpha = 0.3
    stall = 0
    for it in range(200):
        if time.time() > deadline:
            break
        h_new = _subgradient_step(h_cur, n, alpha)
        c5_new = _c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                alpha *= 0.7
                stall = 0
    h_cur = h_best.copy()

    # Phase 3: Sum-of-squares refinement
    if time.time() < deadline:
        h_new = _solve_sumsq2(h_cur, n, maxiter=60, ftol=1e-9)
        c5_new = _c5(h_new)
        if c5_new < c5_best - 1e-11:
            c5_best = c5_new
            h_best = h_new.copy()

    # Phase 4: Final epigraph polish with all lags
    if time.time() < deadline:
        lags_all = list(range(-(n-1), n))
        h_new = _solve_epigraph(h_best, n, lags_all, maxiter=100, ftol=1e-10)
        c5_new = _c5(h_new)
        if c5_new < c5_best - 1e-11:
            c5_best = c5_new
            h_best = h_new.copy()

    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.91
    rng = np.random.default_rng(seed)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = _make_feasible(45, rng)

    n0 = len(h_init)
    h0 = _repair(h_init, n0)
    c5_0 = _c5(h0)

    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n0

    # Optimize at initial n
    n_time = min(300, (deadline - t0) * 0.4)
    h_opt, c5_opt = _optimize_sequence(h0, n0, n_time, t0 + n_time, rng)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = n0

    # Try other n values
    n_candidates = []
    for n_try in [28, 32, 36, 40, 44, 48, 52, 56, 60, 64]:
        if n_try != n0:
            n_candidates.append(n_try)

    remaining = deadline - time.time()
    if remaining > 30:
        per_n = max(15, remaining / (len(n_candidates) + 1) * 0.7)
        for n_try in n_candidates:
            if time.time() > deadline:
                break
            if time.time() - t0 > budget_s * 0.88:
                break
            h_tr = _transfer(n_try, best_h)
            t_sub_deadline = min(time.time() + per_n, deadline)
            h_opt, c5_opt = _optimize_sequence(h_tr, n_try, per_n * 0.8,
                                               t_sub_deadline, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Random restarts at best n if time remains
    if time.time() < deadline:
        n_restarts = 5
        per_restart = (deadline - time.time()) / (n_restarts + 0.5)
        for r in range(n_restarts):
            if time.time() > deadline:
                break
            if per_restart < 5:
                break
            h_pert = best_h + rng.uniform(-0.08, 0.08, best_n)
            h_pert = _repair(h_pert, best_n)
            t_rd = min(time.time() + per_restart, deadline)
            h_opt, c5_opt = _optimize_sequence(h_pert, best_n, per_restart * 0.7,
                                               t_rd, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final repair and evaluation
    final_h = _repair(best_h, best_n)
    final_c5 = _c5(final_h)

    return (final_h, final_c5, best_n)
