def _adjust_sum(p, diff, target):
    # If diff is small, distribute
    if abs(diff) <= 1e-13: return p
    if diff > 0:
        slack = 1.0 - p
        total = slack.sum()
        if total <= 0: return p
        p += diff * (slack/total)
        p = np.clip(p,0.0,1.0)
    else:
        slack = p
        total = slack.sum()
        if total <= 0: return p
        p += diff * (slack/total)
        p = np.clip(p,0.0,1.0)
    diff2 = target - p.sum()
    if abs(diff2) > 1e-12:
       # single coordinate
       if diff2 > 0:
           idx = np.argmax(1.0 - p)
           add = min(diff2, 1.0 - p[idx])
           if add != 0: p[idx] += add
       else:
           idx = np.argmin(p)
           sub = min(-diff2, p[idx])
           if sub != 0: p[idx] -= sub
       # if still, distribute again
       diff3 = target - p.sum()
       if abs(diff3) > 1e-14:
          if diff3>0:
             slack=1-p; tot=slack.sum(); if tot>0: p=np.clip(p+diff3*slack/tot,0,1)
          else:
             slack=p; tot=slack.sum(); if tot>0: p=np.clip(p+diff3*slack/tot,0,1)
    return p
```

    def h_part(self,z): return np.asarray(z,dtype=float).ravel()[:self.n]
    def obj(self,z): return float(np.asarray(z,dtype=float).ravel()[-1])
    def obj_jac(self,z):

def _adjust_sum(p, diff, target):
    if abs(diff) < 1e-14: return p
    for _ in range(5):
       diff = target - p.sum()
       if abs(diff) < 1e-14: break
       if diff > 0:
          idxs = np.where(p < 1.0 - abs(diff) - 1e-16)[0]
          if idxs.size >0:
             p[idxs[0]] += diff
          else:
             slack = 1.0 - p
             if slack.sum() >0:
                p += diff*slack/slack.sum()
                p = np.clip(p,0,1)
       else:
          idxs=np.where(p > abs(diff)+1e-16)[0]
          if idxs.size>0: p[idxs[0]] += diff
          else:
             slack=p
             if slack.sum()>0: p += diff*slack/slack.sum(); p=np.clip(p,0,1)
    # enforce bounds
    return np.clip(p,0,1)
```

def _adjust_sum(p, diff, target):
    if not np.all(np.isfinite(p)): return p
    for _ in range(6):
        diff = target - float(np.sum(p))
        if abs(diff) < 1e-14: break
        if diff > 0:
            # single
            candidates = np.where(p < 1.0 - diff - 1e-15)[0]
            if candidates.size >0:
                p = p.copy(); p[candidates[0]] += diff
            else:
                slack = 1.0 - p
                tot=float(np.sum(slack))
                if tot <=0: break
                p = p + diff*(slack/tot)
                p = np.clip(p,0.0,1.0)
        else:
            candidates = np.where(p > -diff - 1e-15)[0] # since p > abs(diff)
            if candidates.size >0:
                p=p.copy(); p[candidates[0]] += diff
            else:
                slack=p
                tot=float(np.sum(slack))
                if tot <=0: break
                p=p + diff*(slack/tot)
                p=np.clip(p,0.0,1.0)
    return np.clip(p,0.0,1.0)

import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint, NonlinearConstraint

PAIR_CACHE = {}


def get_pair_data(n):
    key = int(n)
    if key <= 0:
        return None
    if key in PAIR_CACHE:
        return PAIR_CACHE[key]
    I, J = np.indices((key, key), dtype=np.intp)
    M = (key - 1) + J - I
    data = (I.ravel(), J.ravel(), M.ravel())
    if len(PAIR_CACHE) > 18:
        PAIR_CACHE.pop(next(iter(PAIR_CACHE)))
    PAIR_CACHE[key] = data
    return data


def _adjust_sum(p, diff, target):
    if p.size == 0:
        return p
    if not np.all(np.isfinite(p)):
        return p
    for _ in range(8):
        diff = target - float(np.sum(p))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(p < 1.0 - diff - 1e-15)[0]
            if idx.size > 0:
                p = p.copy()
                p[idx[0]] += diff
            else:
                slack = 1.0 - p
                tot = float(np.sum(slack))
                if tot <= 0.0:
                    break
                p = p + diff * (slack / tot)
                p = np.clip(p, 0.0, 1.0)
        else:
            idx = np.where(p > -diff - 1e-15)[0]
            if idx.size > 0:
                p = p.copy()
                p[idx[0]] += diff
            else:
                slack = p
                tot = float(np.sum(slack))
                if tot <= 0.0:
                    break
                p = p + diff * (slack / tot)
                p = np.clip(p, 0.0, 1.0)
    return np.clip(p, 0.0, 1.0)


def project_box_sum(y, target=None):
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    if target is None:
        target = 0.5 * float(n)
    if n == 0:
        return y
    if not np.all(np.isfinite(y)):
        y = np.full(n, float(target) / float(n), dtype=float)
    clipped = np.clip(y, 0.0, 1.0)
    s = float(np.sum(clipped))
    if abs(s - float(target)) <= 1e-9:
        if abs(s - float(target)) > 1e-13:
            clipped = _adjust_sum(clipped, float(target) - s, float(target))
        return clipped
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        ss = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if ss > float(target):
            lo = mid
        else:
            hi = mid
    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    return _adjust_sum(p, float(target) - float(np.sum(p)), float(target))


def exact_c5(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n <= 0:
        return float("inf")
    dx = 2.0 / float(n)
    vals = np.correlate(h, 1.0 - h, mode="full")
    if vals.size == 0:
        return float("inf")
    return float(dx * float(np.max(vals)))


def grad_c_matrix(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    data = get_pair_data(n)
    if data is None:
        return np.zeros((max(0, 2 * n - 1), n), dtype=float)
    i, j, m = data
    gm = np.zeros((2 * n - 1, n), dtype=float)
    if i.size:
        np.add.at(gm, (m, i), 1.0 - h[j])
        np.add.at(gm, (m, j), -h[i])
    return gm


def make_ramp(n):
    n = max(1, int(n))
    y = (np.arange(n, dtype=float) + 0.5) / float(n)
    return project_box_sum(y)


def make_rev_ramp(n):
    n = max(1, int(n))
    y = 1.0 - (np.arange(n, dtype=float) + 0.5) / float(n)
    return project_box_sum(y)


def make_quad_profile(n):
    n = max(1, int(n))
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    y = 0.5 + 0.45 * (2.0 * t - 1.0)
    return project_box_sum(y)


def make_sin_profile(n, freq, amp, phase):
    n = max(1, int(n))
    t = np.arange(n, dtype=float) / float(n)
    y = 0.5 + float(amp) * np.sin(2.0 * np.pi * float(freq) * t + float(phase))
    return project_box_sum(y)


def make_power_profile(n, p):
    n = max(1, int(n))
    p = max(0.1, float(p))
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    y = ((p + 1.0) / 2.0) * np.power(t, p)
    return project_box_sum(y)


def resample_vector(old, n_new):
    old = np.asarray(old, dtype=float).ravel()
    n_new = max(1, int(n_new))
    n_old = old.size
    if n_old == 0:
        return project_box_sum(np.full(n_new, 0.5))
    if n_old == n_new:
        return project_box_sum(old)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, old, left=float(old[0]), right=float(old[-1]))
    return project_box_sum(y)


def mix_vectors(a, b, alpha):

    b = np.asarray(b, dtype=float).ravel()

def sym_improve(h):

    if h.size <= 1:
        return h
    s = 0.5 * (h + h[::-1])
    s = project_box_sum(s)

class SmoothTask(object):
    def __init__(self, n, tau):
        self.n = int(n)
        self.tau = float(tau)
        self._last_x = None
        self._last_grad = None

    def value_grad(self, x):
        h = np.asarray(x, dtype=float).ravel()[:self.n]
        n = self.n
        dx = 2.0 / float(n)
        c = np.correlate(h, 1.0 - h, mode="full")
        cmax = float(np.max(c))
        z = (c - cmax) / self.tau
        ez = np.exp(z)
        s = float(np.sum(ez))
        if s <= 0.0 or not np.isfinite(s):
            s = 1.0
            ez = ez / s
        w = ez / s
        val = dx * (cmax + self.tau * float(np.log(s)))
        gm = grad_c_matrix(h)
        grad = (w @ gm) * dx
        return float(val), np.asarray(grad, dtype=float)

    def fun(self, x):
        val, grad = self.value_grad(x)
        self._last_x = np.asarray(x, dtype=float).copy()
        self._last_grad = grad
        return val

    def grad(self, x):
        if self._last_grad is not None and self._last_x is not None:
            xv = np.asarray(x, dtype=float)
            if xv.shape == self._last_x.shape and np.array_equal(xv, self._last_x):
                return self._last_grad
        return self.value_grad(x)[1]


class EqCon(object):
    def __init__(self, n):
        self.n = int(n)

    def __call__(self, x):
        x = np.asarray(x, dtype=float).ravel()
        return float(np.sum(x[:self.n]) - 0.5 * float(self.n))

    def jac(self, x):
        return np.ones(self.n, dtype=float)


class EpigraphTask(object):
    def __init__(self, n, sel=None):
        self.n = int(n)
        self.dx = 2.0 / float(self.n)
        if sel is None:
            self.sel = np.arange(2 * self.n - 1, dtype=np.intp)
        else:
            self.sel = np.asarray(sel, dtype=np.intp).ravel()

    def obj(self, z):
        return float(np.asarray(z, dtype=float).ravel()[-1])

    def obj_jac(self, z):
        g = np.zeros(self.n + 1, dtype=float)
        g[-1] = 1.0
        return g

    def ineq(self, z):
        z = np.asarray(z, dtype=float).ravel()
        h = z[:self.n]
        c = np.correlate(h, 1.0 - h, mode="full")
        return z[-1] - self.dx * c[self.sel]

    def ineq_jac(self, z):
        z = np.asarray(z, dtype=float).ravel()
        h = z[:self.n]
        gm = grad_c_matrix(h)
        rows = gm[self.sel]
        J = np.empty(rows.shape[0] + 0, dtype=float) if False else np.empty((rows.shape[0], self.n + 1), dtype=float)
        J[:, :self.n] = -self.dx * rows
        J[:, self.n] = 1.0
        return J

    def eq(self, z):
        z = np.asarray(z, dtype=float).ravel()
        return float(np.sum(z[:self.n]) - 0.5 * float(self.n))

    def eq_jac(self, z):
        g = np.zeros(self.n + 1, dtype=float)
        g[:self.n] = 1.0
        return g


def make_bounds_pairs(n, extra_upper=None):
    pairs = []
    for _ in range(int(n)):
        pairs.append((0.0, 1.0))
    if extra_upper is not None:
        pairs.append((0.0, float(extra_upper)))
    return pairs


def solve_epigraph_slsqp(task, z0, bounds, maxiter):
    cons = [
        {"type": "ineq", "fun": task.ineq, "jac": task.ineq_jac},
        {"type": "eq", "fun": task.eq, "jac": task.eq_jac},
    ]
    try:
        res = minimize(
            fun=task.obj,
            x0=np.asarray(z0, dtype=float),
            jac=task.obj_jac,
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            tol=1e-11,
            options={"maxiter": int(maxiter), "ftol": 1e-13},
        )
    except Exception:
        return None
    if res is not None and hasattr(res, "x") and res.x is not None:
        x = np.asarray(res.x, dtype=float)
        if x.size == z0.size and np.all(np.isfinite(x)):
            return x
    return None


def solve_epigraph_trust(task, z0, n, upper):
    try:
        lo = np.zeros(n + 1, dtype=float)
        hi = np.empty(n + 1, dtype=float)
        hi[:n] = 1.0
        hi[-1] = float(upper)
        bds = Bounds(low=lo, ub=hi)
        A = np.zeros((1, n + 1), dtype=float)
        A[0, :n] = 1.0
        S = 0.5 * float(n)
        cons = [
            NonlinearConstraint(task.ineq, 0.0, np.inf, jac=task.ineq_jac),
            LinearConstraint(A, np.array([S]), np.array([S])),
        ]
        res = minimize(
            fun=task.obj,
            x0=np.asarray(z0, dtype=float),
            jac=task.obj_jac,
            bounds=bds,
            constraints=cons,
            method="trust-constr",
            tol=1e-8,
            options={"maxiter": 180},
        )
    except Exception:
        return None
    if res is not None and hasattr(res, "x") and res.x is not None:
        x = np.asarray(res.x, dtype=float)
        if x.size == z0.size and np.all(np.isfinite(x)):
            return x
    return None

def store_candidate(state, h):

    else:
        h1 = cur
    note_candidate(state, h1)
    v1 = exact_c5(h1)

    if v1 > 0.45:
        slack1 = 0.09
    elif v1 > 0.40:
        slack1 = 0.045
    elif v1 > 0.385:
        slack1 = 0.018
    else:
        slack1 = max(0.0022, 0.28 * (v1 - 0.37))
    slack1 = min(max(slack1, 1e-4), 0.09)

    if can_do(state, 50):
        h2 = improve_epigraph(h1, state, slack1)
    else:
        h2 = h1
    note_candidate(state, h2)

    mid, _ = choose_best_of([h1, h2])
    if mid is None:
        mid = h1

    if can_do(state, 35):
        h3 = improve_smooth(mid, n, state)
    else:
        h3 = mid
    note_candidate(state, h3)
    v3 = exact_c5(h3)

    if v3 > 0.40:
        slack2 = 0.04
    else:
        slack2 = max(0.004, min(0.018, 0.22 * v3))

    if can_do(state, 50):
        h4 = improve_epigraph(h3, state, slack2)
    else:
        h4 = h3
    note_candidate(state, h4)

    pool = [cur, h1, h2, h3, h4]
    if state[0] is not None and int(state[2]) == int(n):
        pool.append(state[0])
    late, _ = choose_best_of(pool)
    h5 = None
    if late is not None and can_do(state, 35):
        vlate = exact_c5(late)
        slack3 = max(0.0008, min(0.0045, 0.22 * vlate))
        h5 = improve_epigraph(late, state, slack3)
        note_candidate(state, h5)
        if h5 is not None:
            pool.append(h5)
            if state[0] is not None and int(state[2]) == int(n):
                pool.append(state[0])

    best, _ = choose_best_of(pool)
    if best is None:
        best = late if late is not None else h4
    best = adopt_global(state, best, n)
    note_candidate(state, best)
    return best


def profile_dims(base):
    base = int(base)
    vals = [
        base, base + 1, base + 2, base + 3, base + 5, base + 8, base + 12,
        base - 1, base - 2, base - 4, base - 6,
        25, 30, 35, 40, 45, 50, 55, 60, 65, 70, 80, 90, 100, 115, 135
    ]
    out = []
    for d in vals:
        d = int(d)
        if 8 <= d <= 160 and d not in out:
            out.append(d)
    return out


def add_start(starts, c, force):
    if c is None:
        return
    c = np.asarray(c, dtype=float).ravel()
    if c.size == 0:
        return
    if not np.all(np.isfinite(c)):
        return
    c = project_box_sum(c)
    if c.size == 0:
        return
    v = exact_c5(c)
    starts.append((float(v), c.copy(), bool(force)))


def start_sort_key(item):
    return (0 if item[2] else 1, float(item[0]))


def build_starts(state, parent, rng):
    starts = []
    base = int(state[2]) if state[2] is not None and int(state[2]) > 0 else 47
    if base < 8:
        base = 47

    if parent is not None:
        add_start(starts, parent, True)
    if state[0] is not None:
        add_start(starts, state[0], True)

    add_start(starts, make_ramp(base), True)
    add_start(starts, make_rev_ramp(base), True)
    add_start(starts, make_quad_profile(base), True)
    add_start(starts, make_power_profile(base, 0.5), True)
    add_start(starts, make_power_profile(base, 2.0), True)
    add_start(starts, make_sin_profile(base, 1, 0.45, 0.0), True)

    dims = profile_dims(base)
    for d in dims:
        add_start(starts, make_ramp(d), False)
        if state[0] is not None:
            add_start(starts, resample_vector(state[0], d), False)
        if d in (base, base + 3, base + 8):
            add_start(starts, make_power_profile(d, 0.5), False)
            add_start(starts, make_power_profile(d, 2.0), False)

    if parent is not None and state[0] is not None:
        par = resample_vector(parent, base)
        cur = resample_vector(state[0], base)
        add_start(starts, mix_vectors(par, cur, 0.25), False)
        add_start(starts, mix_vectors(par, cur, 0.5), False)
        add_start(starts, mix_vectors(par, cur, 0.75), False)

    if state[0] is not None and rng is not None:
        add_start(starts, perturb_h(state[0], rng, 0.02), False)
        add_start(starts, perturb_h(state[0], rng, 0.06), False)

    starts.sort(key=start_sort_key)
    return starts


def note_initial(state, parent):
    base = 47
    if parent is not None:
        parent = project_box_sum(parent)
        if parent.size > 0:
            base = int(parent.size)
            note_candidate(state, parent)
    note_candidate(state, make_ramp(base))
    note_candidate(state, make_rev_ramp(base))
    note_candidate(state, make_quad_profile(base))
    note_candidate(state, make_power_profile(base, 0.5))
    note_candidate(state, make_power_profile(base, 2.0))
    note_candidate(state, make_sin_profile(base, 1, 0.45, 0.0))
    if state[0] is None:
        note_candidate(state, np.full(base, 0.5, dtype=float))


def run(seed=42, budget_s=1100, **kwargs):
    state = init_state(seed, budget_s)
    parent = None
    try:
        arr = np.asarray(initial_h_values, dtype=float).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            parent = project_box_sum(arr)
    except Exception:
        parent = None

    note_initial(state, parent)
    if state[0] is None:
        note_candidate(state, np.full(47, 0.5, dtype=float))

    rng = state[4]
    starts = build_starts(state, parent, rng)
    for item in starts:
        note_candidate(state, item[1])

    tl = state[3] - time.monotonic()
    max_starts = max(3, int(max(0.0, tl - 80.0) / 95.0) + 1)
    if max_starts > len(starts):
        max_starts = len(starts)

    processed = 0
    for item in starts[:max_starts]:
        if not can_do(state, 85):
            break
        val, cand, forced = item
        if (not forced) and val > float(state[1]) + 0.045:
            continue
        one_refinement(cand, state)
        processed += 1

    if can_do(state, 70) and state[0] is not None:
        one_refinement(state[0].copy(), state)

    if can_do(state, 30) and state[0] is not None:
        h = improve_epigraph(state[0], state, 1.2e-4)
        note_candidate(state, h)

    if state[0] is None:
        h = np.full(47, 0.5, dtype=float)
        state[0] = project_box_sum(h)
        state[1] = exact_c5(state[0])
        state[2] = int(state[0].size)

    h = project_box_sum(state[0])
    val = exact_c5(h)
    return h, float(val), int(h.size)
