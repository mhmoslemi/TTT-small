def compute_lags(h):
    """Compute all correlation values c(l) for l = -(n-1), ..., n-1."""
    n = len(h)
    vals = np.correlate(h, 1.0 - h, mode='full')

def lse_objective(h, tau):
    """LSE smoothing of max(c(l)) * dx."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    a = dx * vals / tau
    a_max = np.max(a)
    if a_max > 500:
        return tau * a_max  # overflow protection
    z = np.sum(np.exp(a - a_max))
    return tau * (a_max + np.log(z))

    a_max = np.max(a)

        k_max = np.argmax(vals)
        p = np.zeros(len(vals))
        p[k_max] = 1.0
    else:
        exp_a = np.exp(a - a_max)
        p = exp_a / np.sum(exp_a)
    
    grad = np.zeros(n)
    for k in range(len(vals)):
        if p[k] < 1e-14:
            continue
        ell = k - (n - 1)
        w = p[k] * dx
        if ell >= 0:
            if ell == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[:n-ell] += w * (1.0 - h[ell:])
                grad[ell:] -= w * h[:n-ell]
        else:
            t = -ell
            grad[t:] += w * (1.0 - h[:n-t])
            grad[:n-t] -= w * h[t:]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_g_tau = [0.1]
_g_n = [63]

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(vals) * dx)

def project_box_sum(z, target):
    n = len(z)
    z = np.asarray(z, dtype=np.float64).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = 0.5 * (lo + hi)
    h = np.clip(z - alpha, 0.0, 1.0)
    residual = target - h.sum()
    for _ in range(50):
        if abs(residual) < 1e-13:
            break
        if residual > 0:
            room = np.maximum(0.0, 1.0 - h)
            tr = room.sum()
            if tr > 1e-15:
                h += residual * room / tr
            else:
                break
        else:
            avail = np.maximum(0.0, h)
            ta = avail.sum()
            if ta > 1e-15:
                h += residual * avail / ta
            else:
                break
        residual = target - h.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full')
    a = dx * vals / tau
    a_max = a.max()
    if a_max > 500:
        return float(tau * a_max)
    exp_a = np.exp(a - a_max)
    z = exp_a.sum()
    return float(tau * (a_max + np.log(z)))

def lse_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full')
    a = dx * vals / tau
    a_max = a.max()
    if a_max > 500:
        k_max = int(np.argmax(vals))
        p = np.zeros(len(vals))
        p[k_max] = 1.0
    else:
        exp_a = np.exp(a - a_max)
        p = exp_a / exp_a.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-14:
            continue
        ell = k - (n - 1)
        if ell == 0:
            grad += w * (1.0 - 2.0 * x)
        elif ell > 0:
            grad[:n - ell] += w * (1.0 - x[ell:])
            grad[ell:] -= w * x[:n - ell]
        else:
            t = -ell
            grad[t:] += w * (1.0 - x[:n - t])
            grad[:n - t] -= w * x[t:]
    return grad

def _slsqp_obj(x):
    return lse_obj(x, _g_tau[0])

def _slsqp_grad(x):
    return lse_grad(x, _g_tau[0])

def _slsqp_eq(x):
    return float(np.sum(x) - _g_n[0] * 0.5)

def _slsqp_eq_jac(x):
    return np.ones(_g_n[0])

def optimize_projected(h0, n, tau_list, time_limit):
    target = n * 0.5
    h = project_box_sum(h0.copy(), target)
    t_start = time.time()
    for tau in tau_list:
        f_curr = lse_obj(h, tau)
        lr = 2.0 / (n + 10)
        for it in range(500):
            if time.time() - t_start > time_limit:
                break
            g = lse_grad(h, tau)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            step = lr
            improved = False
            for _ in range(60):
                h_new = project_box_sum(h - step * g, target)
                f_new = lse_obj(h_new, tau)
                if f_new < f_curr - 1e-5 * step * gn * gn:
                    h = h_new
                    f_curr = f_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break
    return project_box_sum(h, target)

def optimize_slsqp(h0, n, tau_list, maxiter=200):
    target = n * 0.5
    h = project_box_sum(h0.copy(), target)
    _g_n[0] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    for tau in tau_list:
        _g_tau[0] = tau
        try:
            res = minimize(
                fun=_slsqp_obj, x0=h, jac=_slsqp_grad,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            h = project_box_sum(res.x, target)
        except Exception:
            pass
    return h

def subgrad_refine(h0, n, n_iters=2000, lr_init=0.8):
    target = n * 0.5
    dx = 2.0 / n
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()
    best_sc = score_exact(h)
    for it in range(n_iters):
        vals = np.correlate(h, 1.0 - h, mode='full')
        v_max = vals.max()
        active = np.where(vals >= v_max - 1e-8)[0]
        grad = np.zeros(n)
        na = len(active)
        for k in active:
            ell = k - (n - 1)
            if ell == 0:
                grad += (1.0 - 2.0 * h) / na
            elif ell > 0:
                grad[:n - ell] += (1.0 - h[ell:]) / na
                grad[ell:] -= h[:n - ell] / na
            else:
                t = -ell
                grad[t:] += (1.0 - h[:n - t]) / na
                grad[:n - t] -= h[t:] / na
        grad *= dx
        step = lr_init / (np.sqrt(it + 1.0) + 1.0)
        h = project_box_sum(h - step * grad, target)
        sc = score_exact(h)
        if sc < best_sc:
            best_sc = sc
            best_h = h.copy()
    return best_h

def make_starts(n, parent, rng):
    target = n * 0.5
    starts = []
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            x_old = np.linspace(0.5/len(parent), 2.0 - 0.5/len(parent), len(parent))
            x_new = np.linspace(0.5/n, 2.0 - 0.5/n, n)
            interp = np.interp(x_new, x_old, parent)
            starts.append(project_box_sum(interp, target))
    starts.append(np.full(n, 0.5))
    binary = np.zeros(n)
    binary[:n//2] = 1.0
    s = binary.sum()
    if s > target:
        idx = np.where(binary == 1)[0]
        for i in range(int(s - target)):
            binary[idx[i]] = 0.0
    starts.append(binary.copy())
    alt = np.zeros(n)
    alt[0::2] = 1.0
    s = alt.sum()
    if s > target:
        idx = np.where(alt == 1)[0]
        for i in range(int(s - target)):
            alt[idx[i]] = 0.0
    elif s < target:
        idx = np.where(alt == 0)[0]
        for i in range(int(target - s)):
            alt[idx[i]] = 1.0
    starts.append(alt.copy())
    for r in range(6):
        rh = rng.random(n)
        starts.append(project_box_sum(rh, target))
    for r in range(3):
        t = np.linspace(0, 2*np.pi, n, endpoint=False)
        phase = rng.uniform(0, 2*np.pi)
        freq = rng.integers(1, max(2, n//4) + 1)
        sw = 0.5 + 0.35 * np.sin(freq * t + phase)
        starts.append(project_box_sum(np.clip(sw, 0, 1), target))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    parent = None
    try:

            if pv is not None:

    best_h = None
    best_score = 1.0

    if parent is not None:

            ps = score_exact(parent)
            if ps < best_score:
                best_score = ps
                best_h = parent.copy()
        except Exception:
            pass

    n_list = []
    if parent is not None:
        np_ = len(parent)
        n_list.append(np_)
    for nv in [32, 48, 63, 96, 128]:
        if nv not in n_list:
            n_list.append(nv)

    tau_coarse = [0.3, 0.15, 0.07, 0.03]
    tau_fine = [0.015, 0.007, 0.003]

    for n in n_list:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break

        if n <= 80:

        elif n <= 128:
            n_starts = 5
        else:
            n_starts = 3

        starts = make_starts(n, parent, rng)[:n_starts + 3]

        time_per_n = remaining * 0.85 / max(1, len(n_list) - n_list.index(n))
        time_per_start = time_per_n / max(1, len(starts))

        for h0 in starts:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            t_limit = min(time_per_start * 0.7, remaining * 0.3)

            sc = score_exact(h_pg)
            if sc < best_score:

            if n <= 96:

                    h_sqp = optimize_slsqp(h_pg, n, tau_fine, maxiter=150)
                    sc = score_exact(h_sqp)

            best_h = h_final.copy()

    if best_h is None:
        best_h = np.full(64, 0.5)
