import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_set(v, target_sum):
    """Project v onto {h : 0<=h<=1, sum(h)=target_sum}"""
    n = len(v)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def compute_active_gradient(h, n):
    """Compute gradient of the active (max) correlation function"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    
    # Find all active (near-max) indices for numerical stability
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    
    grad = np.zeros(n)
    for m in active:
        L = m - (n - 1)
        if L >= 0:
            # grad[j] = (1-h[j-L])*[L<=j<n] - h[j+L]*[0<=j<n-L]
            grad[L:] += (1 - h[:n-L])
            grad[:n-L] -= h[L:]
        else:
            M = -L
            # grad[j] = (1-h[j+M])*[0<=j<n-M] - h[j-M]*[M<=j<n]
            grad[:n-M] += (1 - h[M:])
            grad[M:] -= h[:n-M]
    
    # Average over active set
    grad /= len(active)
    return grad * dx, np.max(corr) * dx

def projected_subgradient(h0, n, max_iter=10000, step_init=None):
    """Projected subgradient descent"""
    target_sum = n / 2.0
    h = h0.copy()
    
    if step_init is None:
        step_init = 0.1
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        grad, current_c5 = compute_active_gradient(h, n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        step = step_init / (1.0 + 0.001 * it)
        h = h - step * grad
        h = project_onto_set(h, target_sum)
    
    if evaluate_c5(h) < best_c5:
        best_c5 = evaluate_c5(h)
        best_h = h.copy()
    
    return best_h, best_c5

    dx = 2.0 / n

            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
    else:
        M = -L
        if M < n:
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
    return g * dx
