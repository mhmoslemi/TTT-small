def compute_G(h):
    """Compute all correlation values (the components of the max)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # shape (2n-1,)

def smooth_max_and_grad(h, theta):
    """Compute smooth max of correlation values times dx, and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = corr * dx  # shape (2n-1,)
    
    # Softmax weights
    G_max = G.max()
    exp_G = np.exp((G - G_max) / theta)
    w = exp_G / exp_G.sum()
    
    S = theta * (G_max + np.log(exp_G.sum()))
    
    # Gradient: sum_k w_k * dG_k/dh[i]
    # G_k = dx * corr_k
    # corr_k = sum_n h[n+k]*(1-h[n]) (valid n)
    
    grad = np.zeros(n)
    # For each lag k (in the full correlation, k goes from -(n-1) to n-1)
    # Position in array: k + (n-1)
    for k in range(-(n-1), n):
        wk = w[k + n - 1] * dx
        if wk < 1e-15:
            continue
        if k == 0:
            # d/dh[i] of sum_n h[n](1-h[n]) = 1 - 2*h[i]
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            # d/dh[i] of sum_n h[n+k](1-h[n])
            # For i in [k, n-1]: contribution (1-h[i-k])
            # For i in [0, n-1-k]: contribution -h[i+k]
            contrib = np.zeros(n)
            contrib[k:] += (1.0 - h[:n-k])
            contrib[:n-k] -= h[k:]
            grad += wk * contrib
        else:
            # k < 0, let l = -k
            l = -k
            # d/dh[i] of sum_n h[n](1-h[n+l])
            # For i in [0, n-1-l]: contribution (1-h[i+l])
            # For i in [l, n-1]: contribution -h[i-l]
            contrib = np.zeros(n)
            contrib[:n-l] += (1.0 - h[l:])
            contrib[l:] -= h[:n-l]
            grad += wk * contrib
    
    return S, grad

def evaluate_C(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(y):

    lo = -float(np.abs(y).max()) - 3.0
    hi = float(np.abs(y).max()) + 3.0

        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    G = corr * dx
    Gmax = G.max()
    expG = np.exp((G - Gmax) / theta)
    w = expG / expG.sum()
    S = theta * (Gmax + np.log(expG.sum()))
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = k + n - 1
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            # c_k = sum_{n=0}^{n-1-k} h[n+k](1-h[n])
            # d/dh[i]: (1-h[i-k]) for i in [k, n-1], -h[i+k] for i in [0, n-1-k]
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            l = -k
            # c_{-l} = sum_{n=0}^{n-1-l} h[n](1-h[n+l])
            # d/dh[i]: (1-h[i+l]) for i in [0, n-1-l], -h[i-l] for i in [l, n-1]
            grad[:n-l] += wk * (1.0 - h[l:])
            grad[l:] -= wk * h[:n-l]
    
    return S, grad

def pair_line_search(h, s, t, tol=1e-10):
    """Minimize C along h[s]-=tau, h[t]+=tau."""
    T = min(h[s], 1.0 - h[t])
    if T < 1e-12:
        return h, evaluate_C(h)
    
    def obj(tau):
        h_new = h.copy()
        h_new[s] -= tau
        h_new[t] += tau
        return evaluate_C(h_new)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(60):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    mid = (a + b) / 2.0
    fmid = obj(mid)
    
    # Also check endpoints
    f0 = obj(0.0)
    fT = obj(T)
    
    best_tau = mid
    best_f = fmid
    if f0 < best_f:
        best_tau = 0.0
        best_f = f0
    if fT < best_f:
        best_tau = T
        best_f = fT
    
    h_new = h.copy()
    h_new[s] -= best_tau
    h_new[t] += best_tau
    return h_new, best_f

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(y):
    n = len(y)
    target = n / 2.0
    lo = -float(np.abs(y).max()) - 3.0
    hi = float(np.abs(y).max()) + 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = corr * dx
    Gmax = G.max()
    expG = np.exp((G - Gmax) / theta)
    exp_sum = expG.sum()
    w = expG / exp_sum
    S = theta * (Gmax + np.log(exp_sum))
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            l = -k
            grad[:n - l] += wk * (1.0 - h[l:])
            grad[l:] -= wk * h[:n - l]
    return S, grad

def smooth_gradient_descent(h, theta, max_iters=300, deadline=None):
    best_h = h.copy()
    best_c = evaluate_C(h)
    alpha = 0.3
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        S, grad = smooth_max_grad(h, theta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        for _ in range(40):
            h_trial = project_h(h - alpha * grad)
            S_trial, _ = smooth_max_grad(h_trial, theta)
            if S_trial <= S + 1e-10:
                break
            alpha *= 0.5
        else:
            break
        h = h_trial
        c = evaluate_C(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if S_trial < S:
            alpha = min(alpha * 1.15, 3.0)
        else:
            alpha *= 0.5
    return best_h, best_c

def pair_line_search(h, s, t):
    n = len(h)
    T = min(h[s], 1.0 - h[t])
    if T < 1e-12:
        return h.copy(), evaluate_C(h)
    hs = h[s]
    ht = h[t]

    def obj(tau):
        h[s] = hs - tau
        h[t] = ht + tau
        val = evaluate_C(h)
        h[s] = hs
        h[t] = ht
        return val

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, T
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = obj(c_pt)
    fd = obj(d_pt)
    for _ in range(50):
        if b - a < 1e-10 * max(1.0, T):
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj(d_pt)
    mid = (a + b) / 2.0
    cand_taus = [0.0, T, mid, a, b]
    best_tau = 0.0
    best_f = obj(0.0)
    for tc in cand_taus:
        tc = max(0.0, min(T, tc))
        fc = obj(tc)
        if fc < best_f:
            best_f = fc
            best_tau = tc
    h_new = h.copy()
    h_new[s] -= best_tau
    h_new[t] += best_tau
    return h_new, best_f

def pair_descent_phase(h, n_pairs, deadline, rng, gradient_bias=0.5):
    n = len(h)
    best_h = h.copy()
    best_c = evaluate_C(h)
    theta = max(0.01 * best_c, 1e-4)
    no_improve = 0
    for i in range(n_pairs):
        if time.time() > deadline:
            break
        if no_improve > 200:
            _, grad = smooth_max_grad(best_h, theta)
            no_improve = 0
        if rng.random() < gradient_bias:
            _, grad = smooth_max_grad(best_h, theta)
            heights = np.abs(grad)
            heights /= heights.sum() + 1e-15
            s = rng.choice(n, p=heights)
            t = rng.choice(n, p=heights)
            if s == t:
                s = (s + 1) % n
        else:
            s = rng.integers(n)
            t = rng.integers(n)
            if s == t:
                s = (s + 1) % n
        h_new, c_new = pair_line_search(best_h, int(s), int(t))
        if c_new < best_c - 1e-12:
            best_h = h_new
            best_c = c_new
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c

def make_sine_start(n, freq, phase, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_h(h)

def make_binary_zigzag(n, rng):
    h = np.zeros(n)
    block = max(2, n // (n // 3 + 1))
    for i in range(0, n // block + 1):
        start = i * block
        end = min(start + block, n)
        if i % 2 == 0:
            h[start:end] = 1.0
        else:
            h[start:end] = 0.0
    return project_h(h)

def make_random_start(n, rng):
    h = rng.uniform(0, 1, n)
    return project_h(h)

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new)
