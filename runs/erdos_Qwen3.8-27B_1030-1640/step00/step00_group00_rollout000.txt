def evaluate_C(h):
    """Exact objective: max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y):
    """Project onto {h : 0<=h_i<=1, sum(h)=n/2}"""
    n = len(y)
    target = n / 2.0
    # Bisection for lambda
    lo = -2.0
    hi = 2.0
    # Expand bracket
    while np.sum(np.clip(y + lo, 0, 1)) > target:
        lo *= 2
    while np.sum(np.clip(y + hi, 0, 1)) < target:
        hi *= 2
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(y + (lo + hi) / 2, 0, 1)
    # Fix to exact target
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Distribute the difference
        idx = np.argsort(np.clip(result + 0.1, 0, 1) - result)  # indices with most room
        # Actually simpler: just add/subtract uniform fractional amount
        # Better: find indices where we can add/subtract
        if diff > 0:
            candidates = np.where(result < 1 - 1e-10)[0]
        else:
            candidates = np.where(result > 1e-10)[0]
        # Spread diff
        per = diff / len(candidates)
        result[candidates] += per
        result = np.clip(result, 0, 1)
    return result

    """Compute all 2n-1 correlation terms (unweighted, before *dx)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr  # length 2n-1

def smoothed_objective_grad(h, theta):
    """Compute softmax-smoothed objective and its gradient.
    Returns (S, grad) where S = dx * softmax_C, grad = dS/dh
    """
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation_terms(h)  # all terms, length 2n-1
    
    # Softmax weights (numerically stable)
    max_c = np.max(corr)
    exps = np.exp((corr - max_c) / theta)
    weights = exps / np.sum(exps)
    
    S = dx * theta * np.log(np.sum(exps)) + dx * max_c
    
    # Compute gradient: sum_k w_k * dT_k/dh_j
    # For each lag k (index in corr array), compute gradient contribution
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        w = weights[k]
        lag = k - (n - 1)  # lag can be negative, zero, or positive
        
        if lag == 0:
            # T_0 = sum_i h_i(1-h_i), dT_0/dh_j = 1-2h_j
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # out[n-1+lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # d/dh_j: if j>=lag: +1-h[j-lag]; if j<=n-1-lag: -h[j+lag]
            # j in [lag, n-1] from first part
            first_part = np.zeros(n)
            first_part[lag:] = 1.0 - h[:n-lag]
            # j in [0, n-1-lag] from second part  
            second_part = np.zeros(n)
            second_part[:n-lag] = -h[lag:]
            grad += w * (first_part + second_part)
        else:
            # lag < 0, let l = -lag > 0
            # out[n-1-l] = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l])
            l = -lag
            # d/dh_j: if j<=n-1-l: +1-h[j+l]; if j>=l: -h[j-l]
            first_part = np.zeros(n)
            first_part[:n-l] = 1.0 - h[l:]
            second_part = np.zeros(n)
            second_part[l:] = -h[:n-l]
            grad += w * (first_part + second_part)
    
    return S, grad

def pair_line_search(h, s, t, max_evals=100):
    """Exact ternary search over tau for pair (s,t). Returns (best_h, best_C)."""
    n = len(h)
    h_s = h[s]
    h_t = h[t]
    T_max = min(h_s, 1.0 - h_t)
    if T_max < 1e-12:
        return h.copy(), evaluate_C(h)
    
    def line_val(tau):
        h_new = h.copy()
        h_new[s] = h_s - tau
        h_new[t] = h_t + tau
        return evaluate_C(h_new)
    
    # Ternary search
    a, b = 0.0, T_max
    best_val = min(line_val(0), line_val(T_max))
    best_tau = 0.0 if line_val(0) <= line_val(T_max) else T_max
    # Re-evaluate to avoid double compute
    f0 = line_val(0)
    fT = line_val(T_max)
    best_tau = 0.0 if f0 <= fT else T_max
    best_val = min(f0, fT)
    
    evals = 3
    for _ in range(max_evals - 3):
        if b - a < 1e-10:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = line_val(m1)
        f2 = line_val(m2)
        evals += 2
        if f1 < f2:
            b = m2
            if f1 < best_val:
                best_val = f1
                best_tau = m1
        else:
            a = m1
            if f2 < best_val:
                best_val = f2
                best_tau = m2
        if evals >= max_evals:
            break

    h_best[s] = h_s - best_tau
    h_best[t] = h_t + best_tau
    h_best = project_h(h_best)  # ensure feasibility
    return h_best, evaluate_C(h_best)

import numpy as np
import time

def evaluate_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(y):
    n = len(y)
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(60):
        if np.sum(np.clip(y + lo, 0.0, 1.0)) > target:
            lo *= 2.0
        else:
            break
    for _ in range(60):
        if np.sum(np.clip(y + hi, 0.0, 1.0)) < target:
            hi *= 2.0
        else:
            break
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def smoothed_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    exps = np.exp((corr - mc) / theta)
    tot = np.sum(exps)
    if tot < 1e-300 or not np.isfinite(tot):
        return 0.0, np.zeros(n)
    w = exps / tot
    S = dx * (mc + theta * np.log(tot))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            l = lag
            grad[l:] += wk * (1.0 - h[:n - l])
            grad[:n - l] -= wk * h[l:]
        else:
            l = -lag
            grad[:n - l] += wk * (1.0 - h[l:])
            grad[l:] -= wk * h[:n - l]
    return S, grad

def pair_line_search(h, s, t, max_iters=80):
    n = len(h)
    hs, ht = h[s], h[t]
    T = min(hs, 1.0 - ht)
    if T < 1e-12:
        return h, evaluate_C(h)

    def line_val(tau):
        hn = h.copy()
        hn[s] = hs - tau
        hn[t] = ht + tau
        return evaluate_C(hn)

    f0 = line_val(0.0)
    fT = line_val(T)
    best_val = f0 if f0 <= fT else fT
    best_tau = 0.0 if f0 <= fT else T
    a, b = 0.0, T
    for _ in range(max_iters):
        width = b - a
        if width < 1e-11 * (1.0 + T):
            break
        m1 = a + width / 3.0
        m2 = b - width / 3.0
        f1 = line_val(m1)
        f2 = line_val(m2)
        if not np.isfinite(f1) or not np.isfinite(f2):
            break
        if f1 < f2:
            b = m2
            if f1 < best_val:
                best_val = f1
                best_tau = m1
        else:
            a = m1
            if f2 < best_val:
                best_val = f2
                best_tau = m2
    hn = h.copy()
    hn[s] = hs - best_tau
    hn[t] = ht + best_tau
    hn = project_h(hn)
    return hn, evaluate_C(hn)

def grad_descent(h, time_limit, start_time, theta_init=None):
    n = len(h)
    cur = h.copy()
    cur_C = evaluate_C(cur)
    best_h = cur.copy()
    best_C = cur_C
    if theta_init is None:
        theta_init = max(5e-4, 0.1 * cur_C)
    thetas = theta_init * np.array([1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    for theta in thetas:
        if time.time() - start_time > time_limit:
            break
        alpha = 0.05
        for _ in range(300):
            if time.time() - start_time > time_limit:
                break
            S, grad = smoothed_grad(cur, theta)
            if not np.isfinite(S):
                break
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-12:
                break
            improved = False
            for _ in range(40):
                trial = project_h(cur - alpha * grad)
                trial_C = evaluate_C(trial)
                if trial_C < cur_C - 1e-12:
                    cur = trial
                    cur_C = trial_C
                    if cur_C < best_C:
                        best_C = cur_C
                        best_h = cur.copy()
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                break
    return best_h, best_C

def pair_descent(h, max_rounds, time_limit, start_time, rng):
    n = len(h)
    cur = h.copy()
    cur_C = evaluate_C(cur)
    best_h = cur.copy()
    best_C = cur_C
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        theta = max(5e-4, 0.05 * cur_C)
        S, grad = smoothed_grad(cur, theta)
        if not np.isfinite(S):
            break
        # Pick pair: move mass from index with high grad to low grad
        if rng.random() < 0.4:
            s = int(rng.integers(0, n))
            t = int(rng.integers(0, n))
        else:
            s = int(np.argmax(grad))
            t = int(np.argmin(grad))
        if s == t:
            s = (s + 1) % n
        if s == t:
            continue
        hn, Cn = pair_line_search(cur, s, t, max_iters=60)
        if Cn < cur_C - 1e-11:
            cur = hn
            cur_C = Cn
            if cur_C < best_C:
                best_C = cur_C
                best_h = cur.copy()
        else:
            hn2, Cn2 = pair_line_search(cur, t, s, max_iters=60)
            if Cn2 < cur_C - 1e-11:
                cur = hn2
                cur_C = Cn2
                if cur_C < best_C:
                    best_C = Cn2
                    best_h = cur.copy()
    return best_h, best_C

def make_wave_start(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    freq = rng.uniform(1.0, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.2, 0.4)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    return project_h(h)

def make_random_start(n, rng):
    h = rng.random(n)
    return project_h(h)

def make_binary_start(n, rng):
    base = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    h = base + rng.normal(0, 0.12, n)
    return project_h(h)

def resample_parent(parent, new_n):
    old_n = len(parent)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    h = np.interp(new_x, old_x, parent)
    return project_h(h)
