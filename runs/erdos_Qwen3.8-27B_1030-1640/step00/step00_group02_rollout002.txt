import numpy as np
import time

def c5_eval(h):
    """Compute C5 = max(dx * correlate(h, 1-h, 'full'))"""
    n = len(h)
    dx = 2.0 / n
    overlap = dx * np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(overlap))

def overlap_values(h):
    """Return the full overlap vector (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    """Repair h to be feasible: in [0,1], sum = n/2."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    # Distribute correction
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add mass; add to entries below 1
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room < diff:
                return None
            h += diff * room / total_room
        else:
            # Need to remove mass; remove from entries above 0
            space = h.copy()
            total_space = np.sum(space)
            if total_space < -diff:
                return None
            h += diff * space / total_space
    h = np.clip(h, 0.0, 1.0)
    # Final check
    if abs(np.sum(h) - target) > 1e-9:
        # Fallback: iterative
        for _ in range(100):
            diff = target - np.sum(h)
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = h < 1.0
                if not np.any(mask):
                    return None
                h[mask] += diff / np.sum(mask)
            else:
                mask = h > 0.0
                if not np.any(mask):
                    return None
                h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    if np.sum(h) - target > 1e-9 or np.sum(h) - target < -1e-9:
        return None
    return h

def compute_abq(h, d):
    """Compute A, B, Q arrays for the quadratic model along direction d.
    c_m(t) = A_m + B_m*t + Q_m*t^2
    Assumes sum(d) = 0.
    """
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    A = np.zeros(L)
    B = np.zeros(L)
    Q = np.zeros(L)
    
    for m in range(L):
        i_lo = max(0, m - n + 1)
        i_hi = min(m, n - 1)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + n - 1 - m
        A[m] = dx * np.sum(h[idx] * (1.0 - h[jdx]))
        B[m] = dx * np.sum(d[idx] * (1.0 - h[jdx]) - h[idx] * d[jdx])
        Q[m] = -dx * np.sum(d[idx] * d[jdx])
    
    return A, B, Q

def feasible_interval(h, d):
    """Compute the feasible interval [t_lo, t_hi] for h + t*d."""
    eps = 1e-14
    pos = d > eps
    neg = d < -eps
    
    t_lo = -np.inf
    t_hi = np.inf
    
    if np.any(pos):
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
        t_lo = max(t_lo, np.max(-h[pos] / d[pos]))
    if np.any(neg):
        t_hi = min(t_hi, np.max(h[neg] / (-d[neg])))
        t_lo = max(t_lo, np.min((1.0 - h[neg]) / (-d[neg])))
    
    return t_lo, t_hi

def line_solve(h, d, c5_cur):

def active_gradient(h, cvals, tol_frac=0.003):
    """Compute gradient of the max-norm objective using hard active set."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    cmax = np.max(cvals)
    tol = max(1e-10, tol_frac * cmax)
    active = cvals >= cmax - tol
    if not np.any(active):
        return None
    weights = np.zeros(L)
    weights[active] = 1.0 / np.sum(active)
    return weighted_gradient(h, weights, dx)

def smooth_gradient(h, cvals, tau):
    """Compute gradient using softmax weights."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    z = (cvals - np.max(cvals)) / tau
    w = np.exp(z)
    w = w / w.sum()
    return weighted_gradient(h, w, dx)

def weighted_gradient(h, weights, dx):
    """Compute gradient of sum_m weights[m] * c_m with respect to h."""
    n = len(h)
    L = 2 * n - 1
    g = np.zeros(n)
    
    for m in range(L):
        if weights[m] < 1e-15:
            continue
        i_lo = max(0, m - n + 1)
        i_hi = min(m, n - 1)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + n - 1 - m
        alpha = weights[m] * dx
        # d c_m / d h_r: for r = idx[i], contribution is alpha*(1-h[jdx[i]])
        #                for r = jdx[i], contribution is alpha*(-h[idx[i]])
        np.add.at(g, idx, alpha * (1.0 - h[jdx]))
        np.add.at(g, jdx, -alpha * h[idx])
    
    return g

def make_direction_from_gradient(g):
    """Make a sum-zero direction from gradient (negative gradient)."""
    d = -g
    d = d - np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm

def random_direction(rng, n):
    d = rng.standard_normal(n)
    d = d - np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm

def pair_direction(i, j, n):
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def transfer_h(h_old, n_new):
    """Transfer h from old dimension to new dimension via interpolation."""
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
    return t_lo, t_hi


def line_solve(h, d, c5_cur):
    n = len(h)
    d = d - np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return None
    d = d / dn
    t_lo, t_hi = feasible_interval(h, d)
    if t_hi - t_lo < 1e-10:
        return None
    A, B, Q = compute_abq(h, d)
    L = len(A)
    candidates = [t_lo, t_hi, 0.0]
    for m in range(L):
        if abs(Q[m]) > 1e-14:
            tv = -B[m] / (2.0 * Q[m])
            if t_lo <= tv <= t_hi:
                candidates.append(tv)
    if L <= 200:
        idx_list = list(range(L))
    else:
        idx_list = list(np.argsort(A)[::-1][:100])
    for ii in range(len(idx_list)):
        m1 = idx_list[ii]
        for jj in range(ii + 1, len(idx_list)):
            m2 = idx_list[jj]
            DQ = Q[m1] - Q[m2]
            DB = B[m1] - B[m2]
            DA = A[m1] - A[m2]
            if abs(DQ) > 1e-14:
                disc = DB * DB - 4.0 * DQ * DA
                if disc >= 0:
                    sq = np.sqrt(disc)
                    t1 = (-DB + sq) / (2.0 * DQ)
                    t2 = (-DB - sq) / (2.0 * DQ)
                    if t_lo <= t1 <= t_hi:
                        candidates.append(t1)
                    if t_lo <= t2 <= t_hi:
                        candidates.append(t2)
            elif abs(DB) > 1e-14:
                t = -DA / DB
                if t_lo <= t <= t_hi:
                    candidates.append(t)
    candidates.sort()
    deduped = [candidates[0]]
    for c in candidates[1:]:
        if c - deduped[-1] > 1e-10:
            deduped.append(c)
    candidates = deduped
    n_cands = len(candidates)
    if n_cands > 2000:
        # Pre-filter with surrogate
        t_arr = np.array(candidates)
        chunk = 256
        surr = np.empty(n_cands)
        for s in range(0, n_cands, chunk):
            e = min(s + chunk, n_cands)
            tc = t_arr[s:e]
            vals = A[None, :] + tc[:, None] * B[None, :] + tc[:, None] ** 2 * Q[None, :]
            surr[s:e] = vals.max(axis=1)
        n_eval = min(n_cands, 100)
        order = np.argsort(surr)
        keep = set(order[:n_eval].tolist())
        keep.add(0)
        keep.add(n_cands - 1)
        for k in range(n_cands):
            if abs(candidates[k]) < 1e-13:
                keep.add(k)
        candidates = [candidates[k] for k in sorted(keep)]
    best_c5 = c5_cur
    best_h = None
    for t in candidates:
        hc = h + t * d
        if hc.min() < -1e-7 or hc.max() > 1.0 + 1e-7:
            continue
        hc = np.clip(hc, 0.0, 1.0)
        cs = c5_eval(hc)
        if cs < best_c5 - 1e-12:
            best_c5 = cs
            best_h = hc.copy()
    if best_h is not None:
        return best_h, best_c5
    return None


def weighted_gradient(h, weights, dx):
    n = len(h)
    L = 2 * n - 1
    g = np.zeros(n)
    for m in range(L):
        w = weights[m]
        if w < 1e-15:
            continue
        ilo = max(0, m - n + 1)
        ihi = min(m, n - 1)
        if ilo > ihi:
            continue
        idx = np.arange(ilo, ihi + 1)
        jdx = idx + n - 1 - m
        alpha = w * dx
        np.add.at(g, idx, alpha * (1.0 - h[jdx]))
        np.add.at(g, jdx, -alpha * h[idx])
    return g


def active_gradient(h, cvals, tol_frac=0.002):
    cmax = float(np.max(cvals))
    tol = max(1e-10, tol_frac * cmax)
    active = cvals >= cmax - tol
    if not np.any(active):
        return None
    w = np.zeros(len(cvals))
    w[active] = 1.0 / float(np.sum(active))
    return weighted_gradient(h, w, 2.0 / len(h))


def smooth_gradient(h, cvals, tau):
    n = len(h)
    z = (cvals - np.max(cvals)) / tau
    w = np.exp(z)
    w = w / w.sum()
    return weighted_gradient(h, w, 2.0 / n)


def grad_direction(g):
    d = -g
    d = d - np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return None
    return d / dn


def rand_direction(rng, n):
    d = rng.standard_normal(n)
    d = d - np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return None
    return d / dn


def transfer_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)
