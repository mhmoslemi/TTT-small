                best_h, best_c = h_opt, c_opt
        if n_try % 2 == 0 and time.perf_counter() < deadline - 20:
            h_bin, c_bin = binary_search_opt(h_try, deadline, max_rounds=200)
            if c_bin < best_c:
                best_h, best_c = h_bin, c_bin

    # Phase 4: Final polish on best
    if time.perf_counter() < deadline - 10:
        h_opt, c_opt = optimize_lbfgs(best_h, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    if time.perf_counter() < deadline - 5:
        h_opt, c_opt = subgradient_optimize(best_h, deadline, steps=3000)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    best_h = project_feasible(best_h)
    best_c = evaluate_c5(best_h)
    return best_h, best_c, len(best_h)
