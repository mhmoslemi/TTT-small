import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def check_feasible(h, tol=1e-8):
    n = len(h)
    target = n / 2.0
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1 + tol):
        return False
    if abs(h.sum() - target) > 1e-6:
        return False
    return True

def project_feasible(h):
    """Project h to be feasible: 0<=h<=1, sum(h)=n/2"""
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    # Binary search on shift to satisfy both constraints
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_try = np.clip(h + mid, 0.0, 1.0)
        s = h_try.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Fine-tune to get exact sum
    diff = target - h.sum()
    # Distribute the difference
    if abs(diff) > 1e-12:
        # Add/subtract from interior points
        interior = (h > 1e-6) & (h < 1 - 1e-6)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # Fallback: just adjust one element
            if diff > 0:
                idx = np.argmax(h < 1.0)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmin(h > 0.0)
                h[idx] = max(0.0, h[idx] + diff)
    return h

def resample(parent, target_n):
    """Resample parent array to target_n points via linear interpolation."""
    n_parent = len(parent)
    # Parent samples are at centers: (i+0.5)*2/n_parent
    # Target samples are at centers: (j+0.5)*2/target_n
    x_parent = (np.arange(n_parent) + 0.5) * 2.0 / n_parent
    x_target = (np.arange(target_n) + 0.5) * 2.0 / target_n
    interp = np.interp(x_target, x_parent, parent)
    return interp

def binary_from_continuous(h_cont, n):

    target = n // 2
    idx = np.argsort(h_cont)[::-1][:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def check_feasible(h, tol=1e-6):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(h.sum() - n / 2.0) > 1e-4:
        return False
    return True

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
        else:
            idx = int(np.argmax(h < 1.0)) if diff > 0 else int(np.argmin(h > 0.0))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_to(parent, target_n):
    n_p = len(parent)
    x_p = (np.arange(n_p) + 0.5) * 2.0 / n_p
    x_t = (np.arange(target_n) + 0.5) * 2.0 / target_n
    return np.interp(x_t, x_p, parent)

def make_binary_from_values(vals, n):
    k = n // 2
    idx = np.argpartition(vals, -k)[-k:]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def random_binary(n, rng):
    picks = rng.choice(n, size=n // 2, replace=False)

def thue_morse_binary(n, rng):
    def tm_val(i):
        c = 0
        while i:
            c ^= 1
            i >>= 1
        return float(c)
    h = np.array([tm_val(i) for i in range(n)])
    if h.sum() != n / 2:
        return random_binary(n, rng)
    return h

def quadratic_residue_binary(n, rng):
    # Use a prime-based construction
    p = n  # need n to be prime, else fall back
    h = np.zeros(n)
    # Simple: use quadratic residues mod p as 1s
    if p > 2 and (p - 1) % 4 == 3:
        residues = set()
        for x in range(1, p):
            residues.add((x * x) % p)
        residues.discard(0)
        if len(residues) == (p - 1) // 2:
            h[list(residues)] = 1.0
            if len(residues) == n // 2:
                return h
    return random_binary(n, rng)

def make_odd_candidate(parent, n, rng):

def continuous_polish(h, deadline, rng, max_iters=500):
    """Small continuous polish via random pair transfers."""
    n = len(h)

    best_h = h.copy()
    best_c = cur_c
    step = 0.05

    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        s = int(rng.integers(n))
        t = int(rng.integers(n))
        if s == t:
            continue
        dt = step * rng.random()
        if h[s] - dt < 0 or h[t] + dt > 1:
            dt = min(dt, h[s], 1.0 - h[t])
            if dt < 1e-8:
                continue
        h[s] -= dt
        h[t] += dt
        c = evaluate_c5(h)
        if c < cur_c - 1e-10:
            cur_c = c
            step = min(step * 1.1, 0.2)
            if c < best_c - 1e-12:

                best_h = h.copy()

            h[t] -= dt
            step = max(step * 0.9, 0.01)

    # Project

    best_c = evaluate_c5(best_h)
    return best_h, best_c
