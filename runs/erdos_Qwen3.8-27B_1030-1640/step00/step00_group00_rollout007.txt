def compute_G(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0/n
    one = 1.0 - h
    G = np.empty(2*n-1, dtype=float)
    if n == 1:
       G[0] = np.dot(h, one)*dx
       return G
    G[0] = np.dot(h, one)*dx
    idx=1

       m = n-l
       G[idx] = np.dot(h[:m], one[l:])*dx
       idx += 1
       G[idx] = np.dot(h[l:], one[:m])*dx
       idx += 1

def soft_value(G, theta):
    G = np.asarray(G, dtype=float)
    if G.size == 0: return 0.0
    if not np.all(np.isfinite(G)): return float('inf')
    if theta <= 0: return float(np.max(G))
    x = G/theta
    mx = np.max(x)
    if not np.isfinite(mx): return float(np.max(G))
    ex = np.exp(x-mx)
    s = np.sum(ex)
    if s <=0 or not np.isfinite(s): return float(np.max(G))
    return float(theta*(mx + np.log(s)))

def er_eval(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / n
    if not np.all(np.isfinite(h)):
        return float("inf")
    v = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(v)):
        return float("inf")
    return float(np.max(v) * dx)


def er_project(y, target):
    y = np.asarray(y, dtype=float)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    n = y.size
    if n == 0:
        return y
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)

    yc = np.clip(y, 0.0, 1.0)
    if abs(float(yc.sum()) - target) <= 1e-13:
        return yc

    lo = -1.0
    hi = 1.0
    for _ in range(40):
        slo = float(np.sum(np.clip(y + lo, 0.0, 1.0)))
        shi = float(np.sum(np.clip(y + hi, 0.0, 1.0)))
        if slo <= target and shi >= target:
            break
        lo *= 2.0
        hi *= 2.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        smid = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if smid < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)


def er_G(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    dx = 2.0 / n
    one = 1.0 - h
    G = np.empty(2 * n - 1, dtype=float)
    G[0] = float(np.dot(h, one) * dx)
    idx = 1
    for l in range(1, n):
        m = n - l
        G[idx] = float(np.dot(h[:m], one[l:]) * dx)
        idx += 1
        G[idx] = float(np.dot(h[l:], one[:m]) * dx)
        idx += 1
    return G


def er_soft_value(G, theta):
    G = np.asarray(G, dtype=float)
    if G.size == 0:
        return 0.0
    if not np.all(np.isfinite(G)):
        return float("inf")
    if theta <= 0.0 or not np.isfinite(theta):
        return float(np.max(G))
    x = G / theta
    mx = float(np.max(x))
    if not np.isfinite(mx):
        return float(np.max(G))
    s = float(np.sum(np.exp(x - mx)))
    if s <= 0.0 or not np.isfinite(s):
        return float(np.max(G))
    return float(theta * (mx + np.log(s)))


def er_softmax(G, theta):
    G = np.asarray(G, dtype=float)
    if G.size == 0:
        return np.empty(0, dtype=float)
    if not np.all(np.isfinite(G)):
        return np.full(G.size, 1.0 / G.size)
    if theta <= 0.0 or not np.isfinite(theta):
        i = int(np.argmax(G))
        w = np.zeros(G.size)
        w[i] = 1.0
        return w
    x = G / theta
    mx = float(np.max(x))
    ex = np.exp(x - mx)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(G.size, 1.0 / G.size)
    return ex / s


def er_weighted_grad(w, h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    expected = 2 * n - 1
    w = np.asarray(w, dtype=float)
    if w.size != expected:
        tmp = np.zeros(expected, dtype=float)
        m = min(w.size, expected)
        tmp[:m] = w[:m]
        w = tmp
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    w[w < 0.0] = 0.0
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=float)
    w = w / s

    dx = 2.0 / n
    g = np.zeros(n, dtype=float)
    if w[0] != 0.0:
        g += w[0] * dx * (1.0 - 2.0 * h)

    idx = 1
    for l in range(1, n):
        m = n - l
        wa = w[idx]
        idx += 1
        if wa != 0.0:
            g[:m] += wa * dx * (1.0 - h[l:])
            g[l:] -= wa * dx * h[:m]
        wb = w[idx]
        idx += 1
        if wb != 0.0:
            g[l:] += wb * dx * (1.0 - h[:m])
            g[:m] -= wb * dx * h[l:]
    return g

def er_pair_eval(h, s, t, tau):
    T = min(float(h[s]), float(1.0 - h[t]))
    if tau < 0.0:
        tau = 0.0
    if tau > T:
        tau = T
    hs = h[s]
    ht = h[t]
    h[s] = hs - tau
    h[t] = ht + tau
    val = er_eval(h)
    h[s] = hs
    h[t] = ht
    return val


def er_pair_golden(h, s, t, a, b, max_iter=28):
    a = float(a)
    b = float(b)

    c = b - rho * (b - a)
    d = a + rho * (b - a)
    fc = er_pair_eval(h, s, t, c)
    fd = er_pair_eval(h, s, t, d)

    for _ in range(int(max_iter)):
        if b - a <= 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - rho * (b - a)
            fc = er_pair_eval(h, s, t, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + rho * (b - a)
            fd = er_pair_eval(h, s, t, d)

    best_tau = 0.5 * (a + b)
    best_c = er_pair_eval(h, s, t, best_tau)
    for tau in (a, b, c, d, 0.0, 0.5 * (a + b)):
        vv = er_pair_eval(h, s, t, tau)
        if np.isfinite(vv) and vv < best_c:
            best_c = vv
            best_tau = tau
    return float(best_tau), float(best_c)


def er_try_pair(h, s, t, T, current_c, accept_tol=1e-10):
    if T <= 1e-10:
        return None
    step = T / 8.0
    best_tau = 0.0
    best_c = float(current_c)

    for i in range(1, 9):
        tau = i * step
        vv = er_pair_eval(h, s, t, tau)
        if np.isfinite(vv) and vv < best_c - accept_tol:
            best_c = vv
            best_tau = tau

    if not np.isfinite(best_c) or best_c >= current_c - accept_tol:
        return None

    idx = int(best_tau / step + 0.5)
    if idx < 1:
        idx = 1
    if idx > 8:
        idx = 8
    a = (idx - 1) * step
    b = (idx + 1) * step
    if a < 0.0:
        a = 0.0
    if b > T:
        b = T

    gt, gc = er_pair_golden(h, s, t, a, b, 28)
    if np.isfinite(gc) and gc < best_c - 1e-14:
        best_c = gc
        best_tau = gt

    if best_c < current_c - accept_tol:
        return float(best_tau), float(best_c)
    return None

def er_random_start(n, seed, kind):
    if n <= 0:
        return np.empty(0, dtype=float)
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    k = int(kind) % 4

    if k == 0:
        h = (rng.random(n) < 0.5).astype(float)
    elif k == 1:
        h = (
            0.5
            + 0.45 * np.sin(2.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
            + 0.18 * np.sin(4.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
        )
    elif k == 2:
        h = np.zeros(n, dtype=float)
        pos = 0
        upper = max(3, min(n, n // 5 + 2))
        while pos < n:
            length = int(rng.integers(2, upper + 1))
            if length <= 0:
                length = 1
            end = min(n, pos + length)
            h[pos:end] = 1.0 if rng.random() < 0.5 else 0.0
            pos = end
    else:
        h = np.clip(0.5 + 0.25 * rng.standard_normal(n), 0.0, 1.0)

    return er_project(h, n / 2.0)


def er_resample(h0, new_n):
    h0 = np.asarray(h0, dtype=float)
    old_n = h0.size
    if old_n == 0 or new_n <= 0:
        return h0.copy()
    if new_n == old_n:
        return h0.copy()
    xo = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xn = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    hnew = np.interp(xn, xo, h0, left=float(h0[0]), right=float(h0[-1]))
    return er_project(hnew, new_n / 2.0)

def er_sum_adjust(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    target = float(target)
    err = target - float(h.sum())
    if abs(err) < 1e-14:
        return h

    sign = 1.0 if err > 0.0 else -1.0
    for _ in range(6):
        if abs(err) < 1e-14:
            break
        if sign > 0.0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        delta = sign * min(abs(err) / cnt, 0.25)
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
        err = target - float(h.sum())

    if abs(err) > 1e-11:
        h = er_project(h, target)
    return h

    best_h = np.asarray(best_h, dtype=float)
    target = best_h.size / 2.0

    if not er_feasible_ok(best_h, target):
        cand = er_project(best_h, target)
        cc = er_eval(cand)
        if np.isfinite(cc):
            best_h = cand
            best_c = float(cc)
    else:
        best_c = float(er_eval(best_h))

    adj = er_sum_adjust(best_h, target)
    if er_feasible_ok(adj, target):
        ca = er_eval(adj)
        sa = abs(float(adj.sum()) - target)
        sb = abs(float(best_h.sum()) - target)
        if np.isfinite(ca) and ca <= best_c + 1e-12 and sa <= sb + 1e-14:
            best_h = adj
            best_c = float(ca)

    if np.min(best_h) >= -1e-12 and np.max(best_h) <= 1.0 + 1e-12:
        cl = np.clip(best_h, 0.0, 1.0)
        if er_feasible_ok(cl, target):
            cc = er_eval(cl)
            if np.isfinite(cc) and cc < best_c + 1e-12:
                best_h = cl
                best_c = float(cc)
        else:
            proj = er_project(cl, target)
            cc = er_eval(proj)
            if np.isfinite(cc) and cc < best_c + 1e-10:
                best_h = proj
                best_c = float(cc)

    return best_h, float(best_c), int(best_h.size)
