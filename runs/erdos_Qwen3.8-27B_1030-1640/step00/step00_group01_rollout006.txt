import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(z, target):
    n = len(z)
    z = np.asarray(z, dtype=np.float64).copy()
    
    def _cs(alpha):
        return np.sum(np.clip(z - alpha, 0.0, 1.0))
    
    low = np.min(z) - 2.0
    high = np.max(z) + 2.0
    for _ in range(200):
        mid = (low + high) / 2.0
        if _cs(mid) > target:
            low = mid
        else:
            high = mid
    
    h = np.clip(z - (low + high) / 2.0, 0.0, 1.0)
    # Fix floating point
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-15
            if mask.any():
                h[mask] += diff * room[mask] / room[mask].sum()
        else:
            mask = h > 1e-15
            if mask.any():
                h[mask] += diff * h[mask] / h[mask].sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    
    a = corr * dx / tau
    a_max = a.max()
    u = np.exp(np.clip(a - a_max, -500, 0))
    Z = u.sum()
    if Z < 1e-300:
        return float(a_max * tau), np.zeros(n)
    p = u / Z
    
    obj = float(tau * (a_max + np.log(Z)))

    return np.ones(len(x))

    h = np.clip(z - (low + high) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            if mask.any():
                h[mask] += diff * room[mask] / room[mask].sum()

            mask = h > 1e-14
            if mask.any():
                h[mask] += diff * h[mask] / h[mask].sum()

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    a = corr * dx / tau
    a_max = float(a.max())
    shifted = a - a_max
    shifted = np.where(shifted < -500, -500.0, shifted)
    u = np.exp(shifted)
    Z = float(u.sum())
    if Z < 1e-300:
        return a_max * tau, np.zeros(n)
    p = u / Z
    obj = tau * (a_max + np.log(Z))
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = dx * p[m]
        if abs(w) < 1e-16:
            continue
        lo1 = max(0, -k)
        hi1 = min(n - 1, n - k - 1)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - x[lo1+k:hi1+k+1])
        lo2 = max(0, k)
        hi2 = min(n - 1, n + k - 1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * x[lo2-k:hi2-k+1]
    return obj, grad

    return smooth_obj_and_grad(x, _tau_global)[0]

def _grad_wrapper(x):
    return smooth_obj_and_grad(x, _tau_global)[1]

    return x.sum() - len(x) / 2.0

def _sum_eq_jac(x):
    return np.ones(len(x))

    sorted_vals = np.sort(corr * dx)[::-1]
    # tau based on gap between top values
    if len(sorted_vals) > 5:
        tau_init = max(0.05 * max_corr, (sorted_vals[0] - sorted_vals[min(5, len(sorted_vals)-1)]) / min(5, len(sorted_vals)-1) * 3)
    else:
        tau_init = 0.1 * max_corr
    tau_init = max(tau_init, 0.01)
    
    # Schedule: geometric decrease
    tau = tau_init
    ntaus = 12
    for i in range(ntaus):
        _tau_global = tau
        
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
        
        try:
            res = minimize(
                _obj_wrapper, h0, jac=_grad_wrapper,
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-14}
            )
        except Exception:
            tau *= 0.7
            continue
        
        cand = project_box_sum(res.x, target)
        s = score_exact(cand)
        if s < best_score:
            best_score = s
            best_h = cand.copy()
        h0 = cand.copy()
        tau *= 0.7
        if tau < 0.001:
            break
    
    return best_h, best_score
