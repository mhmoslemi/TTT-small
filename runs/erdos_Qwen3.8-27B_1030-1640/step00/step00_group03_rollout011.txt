    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return tau - corr * dx

def _all_lags_jac(z):
    n = _global_n
    h = z[1:]
    dx = 2.0 / n
    num_lags = 2 * n - 1
    g = np.zeros((num_lags, n + 1))
    g[:, 0] = 1.0
    for k in range(-(n - 1), n):
        m = k + (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        idx = np.arange(lo, hi)
        jidx = idx + k
        g[m, 1 + idx] += (1.0 - h[jidx]) * dx
        g[m, 1 + jidx] -= h[idx] * dx
    return g

def _eq_fun(z):
    return z[1:].sum() - _global_n / 2.0

def _eq_jac(z):
    g = np.zeros(len(z))
    g[1:] = 1.0
    return g

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(v, n):
    v = np.array(v, dtype=float).copy()

    target = n / 2.0

        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def transfer_to_new_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return make_feasible(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    # Nearest neighbor interpolation
    idx = np.searchsorted(x_old, x_new)
    idx = np.clip(idx, 1, n_old - 1)
    # Check if previous is closer
    prev = idx - 1
    d_curr = np.abs(x_new - x_old[np.clip(idx, 0, n_old - 1)])
    d_prev = np.abs(x_new - x_old[prev])
    use_prev = d_prev < d_curr
    idx[use_prev] = prev[use_prev]
    h_new = h_old[np.clip(idx, 0, n_old - 1)]
    return make_feasible(h_new, n_new)

def solve_epigraph(h_init, n, max_iter=200, ftol=1e-10):
    """Solve the epigraph problem using SLSQP"""
    global _global_n
    _global_n = n
    h = make_feasible(h_init, n)
    c5_init = exact_c5(h)
    c5_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    tau0 = float(np.max(c5_vals))
    
    z0 = np.concatenate([[tau0], h])
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'ineq', 'fun': _all_lags_fun, 'jac': _all_lags_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
    ]
    
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': ftol})
    except Exception:
        return h, c5_init
    
    h_opt = make_feasible(res.x[1:], n)
    c5_opt = exact_c5(h_opt)
    
    if c5_opt < c5_init:
        return h_opt, c5_opt
    return h, c5_init

    return v


def transfer_to_new_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return make_feasible(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    idx = np.searchsorted(x_old, x_new)
    idx = np.clip(idx, 1, n_old - 1)
    prev = idx - 1
    d_curr = np.abs(x_new - x_old[idx])
    d_prev = np.abs(x_new - x_old[prev])
    use_prev = d_prev < d_curr
    idx[use_prev] = prev[use_prev]
    h_new = h_old[idx]
    return make_feasible(h_new, n_new)


def solve_epigraph(h_init, n, max_iter=200, ftol=1e-10):
    _setup_indices(n)
    h = make_feasible(h_init, n)
    c5_init = exact_c5(h)
    c5_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    tau0 = float(np.max(c5_vals))
    z0 = np.concatenate([[tau0], h])
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
    except Exception:
        return h, c5_init
    h_opt = make_feasible(res.x[1:], n)
    c5_opt = exact_c5(h_opt)
    if c5_opt < c5_init:
        return h_opt, c5_opt
    return h, c5_init


def make_binary_seed(n, rng):
    v = np.zeros(n)
    idx = rng.permutation(n)
    if n % 2 == 0:
        v[idx[:n // 2]] = 1.0
    else:
        v[idx[:n // 2]] = 1.0
        v[idx[n // 2]] = 1.0
        v[idx[n // 2]] = 0.5
    return make_feasible(v, n)


def make_periodic_seed(n, rng):
    p = int(rng.integers(3, max(4, n // 2)))
    a = max(1, int(0.5 / (1.0 / p) * p) // 2)
    a = max(1, min(p - 1, p // 2))
    pattern = np.zeros(p)
    ones_pos = rng.choice(p, a, replace=False)
    pattern[ones_pos] = 1.0
    v = np.tile(pattern, n // p + 1)[:n]
    return make_feasible(v, n)


def make_smooth_seed(n, rng):
    p = int(rng.integers(4, max(5, n // 3)))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.2, 0.45)
    t = np.arange(n)
    v = 0.5 + amp * np.sin(2 * np.pi * t / p + phase)
    return make_feasible(v, n)


def perturb_seed(h, n, rng, scale=0.1):
    v = h + rng.uniform(-scale, scale, n)
    return make_feasible(v, n)


def local_polish(h, n, rng, steps=200):
    h = make_feasible(h, n)
    c5_cur = exact_c5(h)
    dx = 2.0 / n
    c5_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    for _ in range(steps):
        i = int(rng.integers(n))
        delta = rng.choice([-1, 1]) * rng.uniform(0.01, 0.15)
        h_new = h.copy()
        h_new[i] += delta
        if h_new[i] < 0 or h_new[i] > 1:
            continue
        h_new = make_feasible(h_new, n)
        c5_new = exact_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h = h_new
            c5_cur = c5_new
            c5_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return h, c5_cur


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    t_end = t_start + budget_s * 0.92
    rng = np.random.default_rng(seed)

    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    if h0 is None:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    best_h = make_feasible(h0, n0)
    best_c5 = exact_c5(best_h)
    best_n = n0

    def _elapsed_ok():
        return time.time() < t_end

    # Phase 1: SLSQP from initial
    h_s, c5_s = solve_epigraph(best_h, n0, max_iter=400, ftol=1e-11)
    if c5_s < best_c5:
        best_h, best_c5 = h_s, c5_s

    # Phase 2: Local polish on best
    h_s, c5_s = local_polish(best_h, best_n, rng, steps=500)
    if c5_s < best_c5 - 1e-12:
        best_h, best_c5 = h_s, c5_s

    # Phase 3: Random binary seeds
    for i in range(50):
        if not _elapsed_ok():
            break
        v = make_binary_seed(n0, rng)
        h_s, c5_s = solve_epigraph(v, n0, max_iter=200, ftol=1e-9)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5 = h_s, c5_s

    # Phase 4: Periodic seeds
    for i in range(20):
        if not _elapsed_ok():
            break
        v = make_periodic_seed(n0, rng)
        h_s, c5_s = solve_epigraph(v, n0, max_iter=200, ftol=1e-9)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5 = h_s, c5_s

    # Phase 5: Smooth (sinusoidal) seeds
    for i in range(20):
        if not _elapsed_ok():
            break
        v = make_smooth_seed(n0, rng)
        h_s, c5_s = solve_epigraph(v, n0, max_iter=200, ftol=1e-9)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5 = h_s, c5_s

    # Phase 6: Perturbations of current best
    for i in range(20):
        if not _elapsed_ok():
            break
        scale = 0.03 + 0.07 * (i / 20.0)
        v = perturb_seed(best_h, best_n, rng, scale=scale)
        h_s, c5_s = solve_epigraph(v, best_n, max_iter=200, ftol=1e-9)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5 = h_s, c5_s

    # Phase 7: Try different n values
    n_candidates = [44, 46, 48, 50, 52, 40, 36, 60, 64, 32]
    for n_try in n_candidates:
        if not _elapsed_ok():
            break
        h_try = transfer_to_new_n(best_h, n_try)
        h_s, c5_s = solve_epigraph(h_try, n_try, max_iter=200, ftol=1e-9)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_s, c5_s, n_try
        if not _elapsed_ok():
            break
        v = make_binary_seed(n_try, rng)
        h_s, c5_s = solve_epigraph(v, n_try, max_iter=150, ftol=1e-8)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_s, c5_s, n_try

    # Phase 8: More restarts at best n
    for i in range(50):
        if not _elapsed_ok():
            break
        if rng.random() < 0.5:
            v = make_binary_seed(best_n, rng)
        elif rng.random() < 0.5:
            v = make_periodic_seed(best_n, rng)
        else:
            v = make_smooth_seed(best_n, rng)
        h_s, c5_s = solve_epigraph(v, best_n, max_iter=200, ftol=1e-10)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5 = h_s, c5_s

    # Phase 9: Polish the best solution
    if _elapsed_ok():
        h_s, c5_s = local_polish(best_h, best_n, rng, steps=1000)
        if c5_s < best_c5 - 1e-12:
            best_h, best_c5 = h_s, c5_s

    # Phase 10: Final SLSQP refinement
    if _elapsed_ok():
        h_s, c5_s = solve_epigraph(best_h, best_n, max_iter=500, ftol=1e-13)
        if c5_s < best_c5 - 1e-13:
            best_h, best_c5 = h_s, c5_s

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = exact_c5(best_h)

    return (best_h, best_c5, best_n)
