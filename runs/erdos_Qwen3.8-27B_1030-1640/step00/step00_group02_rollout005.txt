def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = y.size
    target = float(target)
    if not np.all(np.isfinite(y)):
        y = np.zeros(n)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # if already within box and close, fast adjust
    h = np.clip(y, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    # find lambda
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    # If target near 0/n, shrink
    for _ in range(80):
        mid = 0.5*(lo+hi)
        val = np.clip(y - mid, 0.0, 1.0).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    # residual distribution
    for _ in range(n + 10):
        resid = target - h.sum()
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            free = np.arange(n)
        if resid > 0:
            cap = float(np.sum(1.0 - h[free]))
            if cap <= 0: break
            amount = min(resid, cap)
            step = amount / free.size
            # To respect caps, if step might exceed some caps, distributed uniform okay but some may exceed 1. If step <= min cap? Not guaranteed. Could add min(cap_i, amount/freesize) individually and repeat.
            # More robust: add equal, then clip, repeat.
            h[free] += step
            h = np.clip(h,0,1)
        else:
            cap = float(np.sum(h[free]))
            if cap <=0: break
            amount = min(-resid, cap)
            step = amount / free.size
            h[free] -= step
            h = np.clip(h,0,1)
    # final serial adjustment
    h = np.clip(h,0,1)
    resid = target - h.sum()
    for _ in range(n+5):
        if abs(resid) < 1e-15:
            break
        if resid > 0:
            idx = int(np.argmax(1.0 - h))
            inc = min(resid, 1.0 - h[idx])
        else:
            idx = int(np.argmin(h))
            inc = max(resid, -h[idx])
        h[idx] += inc
        h = np.clip(h,0,1)
        resid = target - h.sum()
    return h

def get_lag_indices(n):
    if n in _LAG_CACHE: return _LAG_CACHE[n]
    L=2*n-1
    idx_arr=[]
    d_list=[]
    for m in range(L):
        d = m - (n-1)
        i_start = max(0, n-1-m)
        i_end = min(n-1, 2*n-2-m)
        if i_start > i_end:
            idx = np.empty(0, dtype=int)
        else:
            idx = np.arange(i_start, i_end+1, dtype=int)
        idx_arr.append(idx)
        d_list.append(d)
    _LAG_CACHE[n] = (idx_arr, d_list)
    return _LAG_CACHE[n]

```python
import time
import numpy as np
from scipy.optimize import minimize

_LAG_CACHE = {}


def time_ok(deadline, reserve):
    try:
        return time.monotonic() < (float(deadline) - float(reserve))
    except Exception:
        return False


def get_lag_indices(n):
    if n in _LAG_CACHE:
        return _LAG_CACHE[n]
    L = 2 * n - 1
    idx_list = []
    d_list = []
    for m in range(L):
        d = m - (n - 1)
        i_start = max(0, n - 1 - m)
        i_end = min(n - 1, 2 * n - 2 - m)
        if i_start > i_end:
            idx = np.empty(0, dtype=int)
        else:
            idx = np.arange(i_start, i_end + 1, dtype=int)
        idx_list.append(idx)
        d_list.append(d)
    _LAG_CACHE[n] = (idx_list, d_list)
    return _LAG_CACHE[n]


def overlap_gradient_matrix(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros((0, 0), dtype=float)
    idx_list, d_list = get_lag_indices(n)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    for m in range(L):
        idx = idx_list[m]
        if idx.size == 0:
            continue
        d = d_list[m]
        j = idx + d
        G[m, j] += dx * (1.0 - h[idx])
        G[m, idx] -= dx * h[j]
    return G


def evaluate_exact(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(c) * dx)
    if not np.isfinite(val):
        return float("inf")
    return val


def maybe_update(best, h, c):
    h = np.asarray(h, dtype=float)
    c = float(c)
    if h.size == 0 or not np.all(np.isfinite(h)) or not np.isfinite(c):
        return best
    if best is None:
        return (h.copy(), c, int(h.size))
    bh, bc, bn = best
    if c < bc - 1e-12:
        return (h.copy(), c, int(h.size))
    if abs(c - bc) <= 1e-12 and int(h.size) < int(bn):
        return (h.copy(), c, int(h.size))
    return best


def make_bounds(n):
    return [(0.0, 1.0) for _ in range(int(n))]


def make_epi_bounds(n):
    return [(0.0, 1.0) for _ in range(int(n))] + [(0.0, 1.0)]


def sum_constraint(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return 0.0
    return float(x.sum() - 0.5 * x.size)


def sum_jac(x):
    x = np.asarray(x, dtype=float)
    return np.ones(x.size, dtype=float)


def epi_sum_constraint(x):
    x = np.asarray(x, dtype=float)
    if x.size <= 1:
        return 0.0
    return float(x[:-1].sum() - 0.5 * (x.size - 1))


def epi_sum_jac(x):
    x = np.asarray(x, dtype=float)
    n = x.size
    j = np.zeros(n, dtype=float)
    if n > 1:
        j[:-1] = 1.0
    return j


def epi_objective(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return 0.0
    return float(x[-1])


def epi_objective_jac(x):
    x = np.asarray(x, dtype=float)
    j = np.zeros(x.size, dtype=float)
    if x.size > 0:
        j[-1] = 1.0
    return j


def epi_ineq_fun(x):
    x = np.asarray(x, dtype=float)
    if x.size <= 1:
        return np.array([float(x[-1]) if x.size else 0.0], dtype=float)
    n = x.size - 1
    h = x[:-1]
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, mode="full")
    return float(x[-1]) - c


def epi_ineq_jac(x):
    x = np.asarray(x, dtype=float)
    if x.size <= 1:
        return np.ones((1, max(1, x.size)), dtype=float)
    n = x.size - 1
    L = 2 * n - 1
    G = overlap_gradient_matrix(x[:-1])
    J = np.empty((L, x.size), dtype=float)
    J[:, :n] = -G
    J[:, n:] = -0.0
    J[:, n] = 1.0
    return J


def lse_objective(x, tau):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0 or not np.all(np.isfinite(x)):
        return 1e3
    dx = 2.0 / n
    c = dx * np.correlate(x, 1.0 - x, mode="full")
    if not np.all(np.isfinite(c)):
        return 1e3
    tau = float(tau)
    if tau <= 0.0:
        tau = 1e-12
    z = c / tau
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return 1e3
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return 1e3
    return float(tau * (zmax + np.log(s)))


def lse_jac(x, tau):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    dx = 2.0 / n
    c = dx * np.correlate(x, 1.0 - x, mode="full")
    if not np.all(np.isfinite(c)):
        return np.zeros(n, dtype=float)
    tau = float(tau)
    if tau <= 0.0:
        tau = 1e-12
    z = c / tau
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return np.zeros(n, dtype=float)
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=float)
    w = e / s
    G = overlap_gradient_matrix(x)
    g = G.T.dot(w)
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=float)
    return g


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).reshape(-1)
    n = y.size
    target = float(target)
    if n == 0:
        return np.empty(0, dtype=float)
    if not np.all(np.isfinite(y)):
        y = np.zeros(n, dtype=float)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    h = np.clip(y, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.clip(y - mid, 0.0, 1.0).sum())
        if val > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(n + 10):
        resid = target - float(h.sum())
        if abs(resid) < 1e-14:
            break
        if resid > 0.0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            free = np.arange(n)
        if resid > 0.0:
            cap = float(np.sum(1.0 - h[free]))
            if cap <= 0.0:
                break
            amount = min(resid, cap)
            h[free] += amount / float(free.size)
        else:
            cap = float(np.sum(h[free]))
            if cap <= 0.0:
                break
            amount = min(-resid, cap)
            h[free] -= amount / float(free.size)
        h = np.clip(h, 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)
    resid = target - float(h.sum())
    for _ in range(n + 5):
        if abs(resid) < 1e-15:
            break
        if resid > 0.0:
            idx = int(np.argmax(1.0 - h))
            inc = min(resid, 1.0 - float(h[idx]))
        else:
            idx = int(np.argmin(h))
            inc = max(resid, -float(h[idx]))
        h[idx] += inc
        h = np.clip(h, 0.0, 1.0)
        resid = target - float(h.sum())
    return h


def make_feasible(h, n=None):
    h = np.asarray(h, dtype=float).reshape(-1)
    if n is None:
        n = h.size
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    if h.size != n:
        if h.size == 0:
            h = np.full(n, 0.5, dtype=float)
        else:
            x_old = (np.arange(h.size) + 0.5) * (2.0 / h.size)
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)
    return project_box_sum(h, 0.5 * n)


def symmetrize(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h.copy()
    return 0.5 * (h + h[::-1])


def uniform_start(n):
    return np.full(int(n), 0.5, dtype=float)


def add_gaussian(h, sigma, rng):
    h = np.asarray(h, dtype=float)
    return h + rng.normal(0.0, float(sigma), h.size)


def binary_start(n, rng):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    h = np.zeros(n, dtype=float)
    S = 0.5 * n
    base = int(np.floor(S))
    frac = S - base
    if base > 0:
        if base >= n:
            h[:] = 1.0
        else:
            pos = rng.choice(n, size=base, replace=False)
            h[pos] = 1.0
    if frac > 0.0 and base < n:
        rem = np.where(h < 1.0)[0]
        if rem.size > 0:
            h[rem[int(rng.randint(rem.size))]] = frac
    return h


def wave_start(n, rng):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n, dtype=float)
    nterms = int(rng.randint(2, 5))
    amp_total = float(rng.uniform(0.2, 0.7))
    for _ in range(nterms):
        freq = int(rng.randint(1, 8))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        h += (amp_total / float(nterms)) * np.cos(2.0 * np.pi * freq * x + phase)
    return h


def resample_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_new = int(n_new)
    if n_new <= 0:
        return np.empty(0, dtype=float)
    if h_old.size == 0:
        return np.full(n_new, 0.5, dtype=float)
    if h_old.size == n_new:
        return h_old.copy()
    if h_old.size == 1:
        return np.full(n_new, float(h_old[0]), dtype=float)
    x_old = (np.arange(h_old.size) + 0.5) * (2.0 / h_old.size)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])


def solve_maxiter(n, remaining, cap):
    n = int(n)
    cap = int(cap)
    if n <= 50:
        base = 150
    elif n <= 70:
        base = 100
    else:
        base = 60
    if remaining < 300.0:
        base = min(base, 90)
    if remaining < 150.0:
        base = min(base, 60)
    if remaining < 80.0:
        base = min(base, 35)
    return int(max(20, min(base, cap)))


def refine_lse_once(h, tau, deadline, reserve, cap, best):
    h = make_feasible(h, h.size if h.size > 0 else 47)
    if h.size == 0:
        return h, float("inf"), best
    c = evaluate_exact(h)
    best = maybe_update(best, h, c)
    if not time_ok(deadline, reserve):
        return h, c, best
    rem = max(0.0, float(deadline - time.monotonic() - reserve))
    mi = solve_maxiter(h.size, rem, cap)
    if mi < 20:
        return h, c, best
    try:
        res = minimize(
            lse_objective,
            h,
            jac=lse_jac,
            args=(float(tau),),
            method="SLSQP",
            bounds=make_bounds(h.size),
            constraints=[{"type": "eq", "fun": sum_constraint, "jac": sum_jac}],
            tol=1e-10,
            maxiter=mi,
            iprint=0,
        )
    except Exception:
        return h, c, best
    resx = getattr(res, "x", None)
    if resx is not None:
        resx = np.asarray(resx, dtype=float)
        if resx.size == h.size and np.all(np.isfinite(resx)):
            h1 = make_feasible(resx, h.size)
            c1 = evaluate_exact(h1)
            if c1 < c - 1e-12:
                h = h1
                c = c1
                best = maybe_update(best, h, c)
            hs = make_feasible(symmetrize(h), h.size)
            cs = evaluate_exact(hs)
            if cs < c - 1e-12:
                h = hs
                c = cs
                best = maybe_update(best, h, c)
    return h, c, best


def polish_lse(h0, deadline, reserve, best, max_taus=4, cap=100):
    h0 = make_feasible(h0, h0.size if h0.size > 0 else 47)
    if h0.size == 0:
        return best
    h = h0
    c = evaluate_exact(h)
    best = maybe_update(best, h, c)
    tau = 0.05
    for _ in range(int(max_taus)):
        if not time_ok(deadline, reserve):
            break
        h, c, best = refine_lse_once(h, tau, deadline, reserve, cap, best)
        tau *= 0.5
    return best


def try_epigraph_once(h, deadline, reserve, cap, best):
    h = make_feasible(h, h.size if h.size > 0 else 47)
    if h.size <= 1:
        return h, evaluate_exact(h), best
    c = evaluate_exact(h)
    best = maybe_update(best, h, c)
    if not time_ok(deadline, reserve):
        return h, c, best
    rem = max(0.0, float(deadline - time.monotonic() - reserve))
    mi = solve_maxiter(h.size, rem, cap)
    if mi < 15:
        return h, c, best
    s0 = min(0.98, max(0.0, c + 0.02))
    x0 = np.concatenate((h, np.array([s0], dtype=float)))
    constraints = [
        {"type": "eq", "fun": epi_sum_constraint, "jac": epi_sum_jac},
        {"type": "ineq", "fun": epi_ineq_fun, "jac": epi_ineq_jac},
    ]
    try:
        res = minimize(
            epi_objective,
            x0,
            jac=epi_objective_jac,
            method="SLSQP",
            bounds=make_epi_bounds(h.size),
            constraints=constraints,
            tol=1e-10,
            maxiter=mi,
            iprint=0,
        )
    except Exception:
        return h, c, best
    resx = getattr(res, "x", None)
    if resx is not None:
        resx = np.asarray(resx, dtype=float)
        if resx.size == h.size + 1 and np.all(np.isfinite(resx)):
            h1 = make_feasible(resx[:-1], h.size)
            c1 = evaluate_exact(h1)
            if c1 < c - 1e-12:
                h = h1
                c = c1
                best = maybe_update(best, h, c)
            hs = make_feasible(symmetrize(h), h.size)
            cs = evaluate_exact(hs)
            if cs < c - 1e-12:
                h = hs
                c = cs
                best = maybe_update(best, h, c)
    return h, c, best


def load_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except Exception:
        return None
    if p.size == 0 or not np.all(np.isfinite(p)):
        return None
    return p


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s))
    rng = np.random.RandomState(seed)
    best = None

    parent = load_parent()
    if parent is not None and parent.size > 0:
        parent = make_feasible(parent, parent.size)
        best = maybe_update(best, parent, evaluate_exact(parent))
        psym = make_feasible(symmetrize(parent), parent.size)
        best = maybe_update(best, psym, evaluate_exact(psym))

    if best is None:
        hu = uniform_start(47)
        best = maybe_update(best, hu, evaluate_exact(hu))

    reserve = 60.0

    if time_ok(deadline, reserve + 120.0):
        best = polish_lse(best[0], deadline, reserve, best, max_taus=7, cap=140)

    if time_ok(deadline, reserve + 240.0):
        _, _, best = try_epigraph_once(best[0], deadline, reserve, 140, best)

    last_best = best[1]
    stale = 0
    it = 0

    while time_ok(deadline, reserve + 120.0):
        nbest = int(best[2])
        if it % 4 == 0:
            sigma = 0.008 + 0.004 * (it % 3)
            hstart = add_gaussian(best[0], sigma, rng)
        elif it % 4 == 1:
            nn = nbest + (it % 3) * 2
            if nn < 12 or nn > 90:
                nn = nbest
            hstart = binary_start(nn, rng)
        elif it % 4 == 2:
            nn = nbest + (it % 3) * 3
            if nn < 12 or nn > 90:
                nn = nbest
            hstart = wave_start(nn, rng)
        else:
            nn = nbest + 1 + (it % 5) * 2
            if nn > 90:
                nn = nbest
            hstart = resample_h(best[0], nn)

        hstart = make_feasible(hstart, max(2, int(hstart.size)))
        if hstart.size > 100:
            hstart = make_feasible(resample_h(hstart, 80), 80)

        max_taus = 2
        cap = 70
        if time_ok(deadline, reserve + 240.0) and (it % 3 == 0):
            max_taus = 3
            cap = 100

        best = polish_lse(hstart, deadline, reserve, best, max_taus=max_taus, cap=cap)

        if it % 6 == 5 and time_ok(deadline, reserve + 240.0):
            _, _, best = try_epigraph_once(best[0], deadline, reserve, 120, best)

        it += 1
        if best[1] < last_best - 1e-12:
            last_best = best[1]
            stale = 0
        else:
            stale += 1
        if stale >= 30:
            break

    for tau in (5e-4, 2e-4, 1e-4, 5e-5):
        if time_ok(deadline, 40.0):
            _, _, best = refine_lse_once(best[0], tau, deadline, 40.0, 100, best)

    if time_ok(deadline, 120.0):
        _, _, best = try_epigraph_once(best[0], deadline, 40.0, 160, best)

    if best is None:
        h = uniform_start(47)
    else:
        h = best[0]

    h = make_feasible(h, h.size if h.size > 0 else 47)
    c = evaluate_exact(h)

    if not np.isfinite(c):
        h = uniform_start(47)
        c = evaluate_exact(h)

    return np.asarray(h, dtype=float), float(c), int(h.size)
```
