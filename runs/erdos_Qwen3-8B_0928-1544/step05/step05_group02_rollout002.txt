    t = cp.Variable((n_points, n_points))
    C = cp.Variable()
    constraints = []
    for i in range(n_points):
        for j in range(n_points):
            constraints.append(t[i, j] <= h[i])
            constraints.append(t[i, j] <= h[j])
            constraints.append(t[i, j] >= h[i] + h[j] - 1)
            constraints.append(t[i, j] >= 0)
    for m in range(2 * n_points - 1):
        shift = (m - n_points + 1) * dx
        lag_idx = (np.arange(n_points) + n_points - 1 + m) % n_points
        lag_idx = lag_idx.astype(int)
        sum_val = 0
        for i in range(n_points):
            j = (i + m - n_points + 1) % n_points
            sum_val += h[i] * h[j]
        sum_val = sum_val * dx
        constraints.append(sum_val >= 1 - C)
    obj = cp.Minimize(C)
    prob = cp.Problem(obj, constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=1100 - (time.time() - start_time))
    except:
        prob.solve(solver=cp.ECOS, verbose=False, time_limit=1100 - (time.time() - start_time))
    h_values = h.value
    c5_bound = C.value
    return h_values, c5_bound, n_points
