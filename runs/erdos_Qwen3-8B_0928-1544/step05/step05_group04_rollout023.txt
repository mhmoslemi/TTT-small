import numpy as np
from scipy.optimize import minimize
import copy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy().astype(float)
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        dx = 2.0 / n_points
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(h, k)
            term = np.sum(h * (1 - shifted)) * dx
            overlaps.append(term)
        return np.max(overlaps)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s * 0.7)
    
    for _ in range(max_iter):
        sigma = 0.05 * (1 - _ / max_iter)
        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
        
        initial_h = result.x.copy()
        time_used += 1
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
