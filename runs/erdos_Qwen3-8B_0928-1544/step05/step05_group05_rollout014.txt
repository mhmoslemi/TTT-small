def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

        h = np.full(70, 0.5)

        corr = np.correlate(h_vec, 1 - h_vec, mode='full')

            h_perturb_plus[j] += grad_epsilon

            c5_plus = compute_c5(h_perturb_plus)
            
            h_perturb_minus = h.copy()
            h_perturb_minus[j] -= grad_epsilon
            h_perturb_minus = np.clip(h_perturb_minus, 0.0, 1.0)
            h_perturb_minus_sum = np.sum(h_perturb_minus)

        h = np.clip(h, 0.0, 1.0)
