from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    else:
        h_initial = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    initial_population = [h_initial.copy()]
    for _ in range(20):
        perturbed = h_initial + np.random.normal(0, 0.15, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        initial_population.append(perturbed)
    
    max_iter = int(budget_s / 100)
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:

            best_h = result.x
