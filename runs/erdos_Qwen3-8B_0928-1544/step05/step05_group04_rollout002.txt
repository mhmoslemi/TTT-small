from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    def feasible_projection(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        return h * (n_points / 2) / total
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    
    for _ in range(500):
        time_used += 1
        if time_used >= budget_s:
            break
        h_perturbed = feasible_projection(initial_h + 0.05 * np.random.randn(n_points))
        h_perturbed = feasible_projection(h_perturbed * (ones_count / np.sum(h_perturbed)))
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
