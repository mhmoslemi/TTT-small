    # Compute the correlation between h and (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The dx is 2.0 / len(h)

    return max_corr

import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

        h = initial_h_values.copy()
    else:
        h = np.zeros(n)
        idx = np.random.choice(n, 35, replace=False)
        h[idx] = 1.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n
        return np.max(corr) * dx
    
    def repair_sum(h):
        target = 35
        diff = np.sum(h) - target
        if diff == 0:
            return h
        if diff > 0:
            sz = int(min(20, diff))
            idx = np.where(h == 1)[0][:sz]
            h[idx] = 0
        else:
            sz = int(-min(20, diff))
            idx = np.where(h == 0)[0][:sz]
            h[idx] = 1.0
        return h
    
    def perturb(h):
        n_flip = int(n * 0.05)
        for _ in range(n_flip):
            idx = np.random.randint(n)
            val = h[idx]
            if val == 0:
                h[idx] = np.random.choice([1.0, 0.5])
            elif val == 1:
                h[idx] = np.random.choice([0.0, 0.5])
            else:
                h[idx] = np.random.choice([0.0, 1.0])
        return h
    
    def shift_window(h):
        window_size = np.random.randint(5, 15)
        start = np.random.randint(n - window_size + 1)
        vals = h[start:start+window_size]
        shift = np.random.choice([-1, 1])
        h[start:start+window_size] = np.roll(vals, shift)
        return h
    
    def optimize_even(h):
        for i in range(0, n, 2):
            h1 = h.copy()
            h1[i] = 0.0

                h[i] = 0.0
            elif c2 < c1 and c2 < c3:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    def optimize_odd(h):
        for i in range(1, n, 2):

            h1[i] = 0.0

                h[i] = 0.0

                h[i] = 1.0
