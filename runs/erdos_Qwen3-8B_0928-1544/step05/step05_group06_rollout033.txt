def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    dx = 2.0 / n_points
    ones_count = int(n_points / 2)
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def feasibility_proj(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) > 0:
            scale = ones_count / np.sum(h)
            h *= scale
        return h

    def optimize():
        result = minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_points, jac=None, options={'maxiter': 150})
        if result.success:
            return feasibility_proj(result.x), result.fun
        return initial_h, objective(initial_h)

    h, c5 = optimize()
    for _ in range(5):
        h = feasibility_proj(h + np.random.normal(0, 0.1, n_points))
        new_h, new_c5 = optimize()
