import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    if not np.isclose(np.sum(initial_h), ones_count):
        initial_h = initial_h * (ones_count / np.sum(initial_h)) if np.sum(initial_h) > 0 else np.zeros(n_points)
    best_h = initial_h.copy()
    best_c5 = np.inf
    max_iter = int(budget_s / 4)
    K = 30  # Number of shifts to target

    for _ in range(max_iter):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_indices = np.argsort(corr)[-K:]
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            selected = corr[top_indices]
            return np.sum(selected ** 2)
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
        if result.success:
            current_corr = np.correlate(result.x, 1 - result.x, mode='full')
            current_c5 = np.max(current_corr) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    return (best_h, best_c5, n_points)
