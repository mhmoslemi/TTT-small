import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    # Generate diverse initial guesses
    num_initial = 50
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):
        # Randomly perturb 20% to 0.75 and 20% to 0.25
        perturbed = initial_h.copy()
        indices = np.random.choice(n_points, size=20, replace=False)
        perturbed[indices] = 0.75
        indices = np.random.choice(n_points, size=20, replace=False)
        perturbed[indices] = 0.25
        initial_population.append(perturbed)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    max_iter = int(budget_s / 20)
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=Bounds(0.0, 1.0),
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_population.append(result.x)
                initial_population.pop(np.random.randint(len(initial_population)))
        else:
            # Fallback to SLSQP for difficult cases
            result = minimize(
                fun=objective,
                x0=current_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
                initial_population.append(result.x)
                initial_population.pop(np.random.randint(len(initial_population)))
