                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    h[i] = 1
            elif val == 0:
                h[i] = 0.5
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    h[i] = 0
        
        # Optimize odd indices
        for i in range(1, n_points, 2):
            val = h[i]
            if val == 1:
                h[i] = 0.5
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    h[i] = 1

                h[i] = 0.5
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    h[i] = 0
        
        # Final check
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
