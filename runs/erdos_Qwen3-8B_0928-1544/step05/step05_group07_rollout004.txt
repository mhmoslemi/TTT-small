import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def optimize_step(h):
        corr = np.correlate(h, 1 - h, mode='full')
        K = 50
        top_indices = np.argsort(corr)[-K:]
        top_corr = corr[top_indices]
        
        def objective(x):
            x = np.array(x)
            corr_new = np.correlate(x, 1 - x, mode='full')
            top_corr_new = corr_new[top_indices]
            return np.sum(top_corr_new**2)
        
        def constraint_sum(x):
            return np.sum(x) - ones_count
        
        bounds = [(0.0, 1.0)] * n_points
        res = minimize(objective, h, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, method='L-BFGS-B', tol=1e-8)
        return res.x if res.success else h
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    max_iter = int(budget_s / 8)
    for _ in range(max_iter):
        new_h = optimize_step(best_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return best_h, best_c5, n_points
