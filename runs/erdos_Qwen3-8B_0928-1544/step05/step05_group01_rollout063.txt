import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Generate remaining fireflies
    for _ in range(num_fireflies - 1):
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=ones_count, replace=False)
        h[idx] = 1.0
        fireflies.append(h)

        h_repaired[indices[:ones_count]] = 1.0
        return h_repaired

    best_h = None

    alpha = 0.2
    max_iter = 300
    perturb_prob = 0.05

    for _ in range(max_iter):
        fireflies_sorted = sorted(fireflies, key=compute_obj)

        for i, h in enumerate(fireflies_sorted):
