import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    swarm_size = 50
    max_iter = 1000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    vel_max = 0.3
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        particles = np.zeros((swarm_size, n_points))
        particles[0] = h_initial
        for i in range(1, swarm_size):
            noise = np.random.normal(0, 0.05, n_points)
            particles[i] = h_initial + noise
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= target_sum / np.sum(particles[i])
        for _ in range(int(swarm_size * 0.3)):
            h_rand = np.random.rand(n_points)
            h_rand *= target_sum / np.sum(h_rand)
            particles = np.vstack([particles, h_rand])
    else:
        particles = np.random.rand(swarm_size, n_points)
        particles *= target_sum / np.sum(particles, axis=1, keepdims=True)
        particles = np.clip(particles, 0, 1)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    velocities = np.random.uniform(-vel_max, vel_max, (swarm_size, n_points))
    personal_best = particles.copy()
    personal_best_c5 = np.array([evaluate_c5(p) for p in particles])
    global_best = np.min(personal_best_c5)
    global_h = particles[np.argmin(personal_best_c5)]

    for _ in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_h - particles[i])
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= target_sum / np.sum(particles[i])
        new_c5 = np.array([evaluate_c5(p) for p in particles])
        improve_mask = new_c5 < personal_best_c5

            best_c5 = global_best
            best_h = global_h

    return best_h, best_c5, n_points
