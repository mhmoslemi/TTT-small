import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = int(n_points / 2)
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = h * (ones_count / np.sum(h))

    time_used = 0
    max_iter = int(budget_s // 10)
    
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        corr_scaled = corr * (2.0 / n_points)
        max_index = np.argmax(corr_scaled)
        k = max_index - (n_points - 1)
        valid_indices = []
        for i in range(n_points):
            j = i + int(k)
            if 0 <= j < n_points:
                valid_indices.append(i)
        
        variables = cp.Variable(n_points)
        objective_terms = []
        for i in valid_indices:
            j = i + int(k)
            objective_terms.append(variables[i] * variables[j])
        objective = cp.Maximize(cp.sum(objective_terms))
        constraints = [cp.sum(variables) == ones_count, variables >= 0, variables <= 1]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
                h = variables.value
        except:

            break

    c5 = np.max(corr) * (2.0 / n_points)
