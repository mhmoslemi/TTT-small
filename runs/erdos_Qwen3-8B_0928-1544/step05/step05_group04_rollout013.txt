import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = np.full(n_points, 0.5)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    best_h = initial_h.copy()
    best_c5 = 1.0

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    total_time = 0
    start_time = time.time()
    max_iter = int(budget_s * 0.6)

    for iter in range(max_iter):
        h_perturbed = initial_h + np.random.normal(0, 0.1 * (0.8 ** iter), n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        initial_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return best_h, best_c5, n_points
