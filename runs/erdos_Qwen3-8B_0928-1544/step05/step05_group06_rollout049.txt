import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def feasibility_proj(h):
        h = np.clip(h, 0.0, 1.0)
        return h * (ones_count / np.sum(h)) if np.sum(h) > 0 else h
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count})
    
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-8})
    best_h = feasibility_proj(result.x)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
