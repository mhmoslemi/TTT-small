from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    lambda_penalty = 1e-4

    def objective(h):

        max_overlap = np.max(corr) * dx

        return max_overlap + penalty

    bounds = [(0.0, 1.0)] * n_points
