import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        initial_pop = [h.copy() for _ in range(100)]
    else:
        initial_pop = []
        for _ in range(100):
            idx = np.random.rand(3)
            if idx[0] < 0.5:
                h = np.random.rand(n_points) * (1 - np.random.rand())
            elif idx[1] < 0.75:
                t = np.minimum(np.arange(n_points), n_points - np.arange(n_points)) / (n_points / 2)
                h = np.random.rand(n_points) * t
            else:
                h = np.random.rand(n_points)
            h = h / np.sum(h) * ones_count
            initial_pop.append(h)
        for i in range(100):
            initial_pop[i] = np.clip(initial_pop[i], 0, 1) * ones_count / np.sum(initial_pop[i])
    positions = [p.copy() for p in initial_pop]
    velocities = [np.zeros(n_points) for _ in range(100)]
    personal_best = [p.copy() for p in positions]
    global_best = positions[0].copy()
    best_c5 = float('inf')
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    max_iter = int(budget_s * 0.7)
    for i in range(max_iter):
        for j in range(100):
            r1, r2 = np.random.rand(2)
            cognitive = 2 * r1 * (personal_best[j] - positions[j])
            social = 2 * r2 * (global_best - positions[j])
            velocities[j] = 0.7 * velocities[j] + cognitive + social
            velocities[j] = np.clip(velocities[j], -0.5, 0.5)
            new_pos = positions[j] + velocities[j]
            new_pos = np.clip(new_pos, 0, 1) * ones_count / np.sum(new_pos)
            positions[j] = new_pos
            curr = compute_c5(positions[j])
            if curr < compute_c5(personal_best[j]):
                personal_best[j] = positions[j].copy()
                if curr < best_c5:
                    global_best = positions[j].copy()
                    best_c5 = curr
        if i % 10 == 0:
            if i > 0 and np.isnan(best_c5):
                break
    return global_best, best_c5, n_points
