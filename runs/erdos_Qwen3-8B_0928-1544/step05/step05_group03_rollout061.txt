def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    scale = ones_count / sum_h
    h = h * scale
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        h = h * scale
        return evaluate(h)
    
    bounds = np.array([[0.0, 1.0]] * n_points)
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), recombination=0.9, 
                                   tol=1e-8, maxiter=int(budget_s * 0.8))
