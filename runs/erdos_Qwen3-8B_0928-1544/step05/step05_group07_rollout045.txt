    for _ in range(max_iter):
        # Direct optimization
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
        if result.success:
            candidate_h = result.x
            if np.all(candidate_h >= 0) and np.all(candidate_h <= 1) and np.isclose(np.sum(candidate_h), target_sum):
                current_c5 = obj(candidate_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = candidate_h
        if time_used >= budget_s:
            break
        time_used += 2
    
    return best_h, best_c5, n_points
