import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def project_h(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) > 0:
            return h * (ones_count / np.sum(h))
        return h
    
    max_iter = int(budget_s / 100)
    num_starts = 10
    
    for _ in range(num_starts):
        perturb = np.random.normal(0, 0.1, n_points)
        h_start = initial_h + perturb
        h_start = project_h(h_start)
        result = minimize(compute_c5, h_start, method='Nelder-Mead', options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
