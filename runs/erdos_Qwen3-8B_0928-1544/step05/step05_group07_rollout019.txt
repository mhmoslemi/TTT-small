import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    for i in range(n_points // 2):
        initial_h[i*2] = 1.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    def perturb(h):
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        total = np.sum(h_perturbed)
        if total == 0:
            return np.random.rand(n_points)
        scale = ones_count / total
        return h_perturbed * scale
    
    def crossover(parent1, parent2):
        child = (parent1 + parent2) / 2
        total = np.sum(child)
        if total == 0:
            return np.random.rand(n_points)
        scale = ones_count / total
        return child * scale
    
    population = [initial_h.copy() for _ in range(100)]
    for i in range(100):
        population[i] = perturb(population[i])
    
    start_time = time.time()
    max_generations = 500
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
        fitness = [objective(ind) for ind in population]
        idx = np.argsort(fitness)[:int(0.3 * len(population))]
        selected = [population[i] for i in idx]
        
        new_population = []
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(selected, 2, replace=True)
            child = crossover(p1, p2)
            if np.random.rand() < 0.3:
                child = perturb(child)
            new_population.append(child)
        
        population = new_population
        for ind in population:
            curr_c5 = objective(ind)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = ind.copy()
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
