        initial = initial_h_values.copy()
    else:
        initial = np.zeros(n_points)
        initial[::2] = 1.0
        sum_h = np.sum(initial)
        if sum_h != ones_count:
            idx = np.where(initial == 0)[0]
            initial[idx[:ones_count - sum_h]] = 1.0 if sum_h < ones_count else 0.0
    
    num_fireflies = 100
    fireflies = np.zeros((num_fireflies, n_points))
    fireflies[0, :] = initial
    
    for i in range(1, num_fireflies):
        bits = np.zeros(n_points, dtype=int)
        bits[:ones_count] = 1
        np.random.shuffle(bits)
        fireflies[i, :] = bits
    
    def compute_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    max_iterations = 300
    alpha = 0.2
    beta = 0.2
    gamma = 0.5
    perturb_prob = 0.05
    
    best_h = np.copy(fireflies[0])
    best_c5 = compute_objective(best_h)
    
    for _ in range(max_iterations):
        objectives = np.array([compute_objective(f) for f in fireflies])
        sorted_idx = np.argsort(objectives)
        sorted_fireflies = fireflies[sorted_idx]
        
        for i in range(num_fireflies):
            firefly = sorted_fireflies[i]
            for j in range(i):
                other = sorted_fireflies[j]
                distance = np.linalg.norm(firefly - other)
                attractiveness = beta * np.exp(-gamma * distance)
                new_pos = firefly + attractiveness * (other - firefly) + alpha * np.random.randn(n_points)
                new_pos = repair(new_pos)
                fireflies[i, :] = new_pos
        
        for i in range(num_fireflies):
            if np.random.rand() < perturb_prob:
                perturbation = alpha * np.random.randn(n_points)
                fireflies[i, :] = repair(fireflies[i, :] + perturbation)
        
        for i in range(num_fireflies):
            h_candidate = fireflies[i]
            c5_candidate = objectives[i]
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    
    return best_h, best_c5, n_points
