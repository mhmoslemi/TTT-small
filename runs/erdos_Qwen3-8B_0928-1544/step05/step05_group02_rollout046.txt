import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    max_iterations = 800
    alpha = 0.1
    decay = 0.995
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        x = np.linspace(0, 2, n_points)
        h = (1 + np.sin(np.pi * x)) / 2

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (target_sum / h_sum)
        return h

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        h = clip_rescale(h)
        H = np.fft.fft(h)
        H_shift = np.fft.fftshift(H)
        H_r = np.fft.ifftshift(H_shift)
        conv = np.fft.ifft(H * np.conj(H_r)).real
        conv = np.roll(conv, n_points // 2)
        c5 = np.max(conv)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        H += alpha * np.random.normal(0, 1, len(H))
        H = np.clip(H, np.fft.fft([0]*n_points), np.fft.fft([1]*n_points))
        h = np.fft.ifft(H).real
        alpha *= decay

    best_h = clip_rescale(best_h)
    return best_h, evaluate_c5(best_h), n_points
