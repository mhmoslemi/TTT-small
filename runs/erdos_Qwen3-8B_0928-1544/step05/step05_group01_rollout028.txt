        new_objectives = np.array([objective(p) for p in particles])
        for i in range(swarm_size):
            if new_objectives[i] < pbest_values[i]:
                pbest[i, :] = particles[i, :]
                pbest_values[i] = new_objectives[i]
        if new_objectives.min() < gbest_value:
            gbest = pbest[pbest_values.argmin()].copy()
            gbest_value = new_objectives.min()
        if _ % 10 == 0:
            top_indices = np.argsort(new_objectives)[:10]
            for i in top_indices:
                h_initial = particles[i, :].copy()
                res = minimize(
                    fun=objective,
                    x0=h_initial,
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * n_points,
                    constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
                    options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
                )
                if res.success:
                    particles[i, :] = res.x.copy()

    return gbest, gbest_value, n_points
