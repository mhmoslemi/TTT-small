import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

        bounds=bounds,

        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,

        maxiter=int(budget_s / 10),
