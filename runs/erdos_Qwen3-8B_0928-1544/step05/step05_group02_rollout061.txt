                overlap += h[i] * (1 - h[(i + k) % n_points]) * (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k

        gradient = np.zeros(n_points)
        for i in range(n_points):
            gradient[i] = 1.0 - h[(i + best_k) % n_points]

        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h *= (target_sum / h_sum) if h_sum != 0 else 1.0

        current_c5 = max_overlap
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= 0.995

    max_overlap = -np.inf
    best_k = 0
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            overlap += best_h[i] * (1 - best_h[(i + k) % n_points]) * (2.0 / n_points)
        if overlap > max_overlap:
            max_overlap = overlap
            best_k = k

    return best_h, max_overlap, n_points
