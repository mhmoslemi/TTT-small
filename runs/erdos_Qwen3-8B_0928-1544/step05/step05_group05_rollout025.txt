def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0
    dx = 2.0 / n_points
    max_iterations = int(budget_s / 10)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
        initial_h = initial_h * required_sum / np.sum(initial_h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = []
    for _ in range(30):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * required_sum / np.sum(h)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    start_time = time.time()

    for _ in range(max_iterations):
        fitness = np.array([compute_c5(h) for h in population])
        best_idx = np.argmin(fitness)

            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        offspring = []
