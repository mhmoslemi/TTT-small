from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Perturb current best solution
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (ones_count / np.sum(perturbed_h))
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )

            best_h = result.x
