from scipy.optimize import minimize
import random

            h_initial[idx[:ones_count - sum_h]] = 1.0
    
    def obj(h):

    def eval_h(h):
        return obj(h)
    
    def repair(h):
        h_sorted = np.sort(h)

        h_repaired[-ones_count:] = 1.0
        return h_repaired
    
    def firefly_algorithm():
        m = 100
        fireflies = [np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) for _ in range(m)]
        for f in fireflies:
            if np.sum(f) != ones_count:
                f[:] = repair(f)
        best_h = min(fireflies, key=eval_h)
