    # Multi-start SLSQP optimization
    def optimize_from_guess(guess):
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 35})
        return result.x if result.success else guess
    
    # Generate perturbed initial guesses
    perturbations = [initial_h_values + np.random.normal(0, 0.1, n_points) for _ in range(5)]
    candidates = [project_feasible(p) for p in perturbations]
    
    # Evaluate and refine each candidate
    refined = [optimize_from_guess(c) for c in candidates]
    refined.append(initial_h_values)
    
    # Find best solution
    best_c5 = float('inf')
    best_h = refined[0]
    for h in refined:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
