import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    h = cp.Variable(n_points)
    t = cp.Variable((n_points, n_points))
    C = cp.Variable()

    # Constraints for t[i][j] representing h[i] * h[j]
    for i in range(n_points):
        for j in range(n_points):
            constraints = [
                t[i, j] <= h[i],
                t[i, j] <= h[j],
                t[i, j] >= h[i] + h[j] - 1,
                t[i, j] >= 0
            ]
            for c in constraints:

    # Objective: minimize C
    objective = cp.Minimize(C)

    # Constraints: sum of h = target_sum
    constraints = [cp.sum(h) == target_sum]
