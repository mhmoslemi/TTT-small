import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Optimized steps for faster convergence
    num_pop = 30
    num_gen = int(budget_s / 100)
    population = [initial_h.copy() for _ in range(num_pop)]
    
    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    
    for _ in range(num_gen):
        scores = [evaluate(h) for h in population]
        sort_idx = np.argsort(scores)
        top = sort_idx[:int(0.1 * num_pop)]
        
        # Blend top solutions for diversity
        new_pop = [np.mean([population[i] for i in top], axis=0) for _ in range(num_pop // 2)]
        
        # Apply local optimization
        for h in new_pop:
            res = minimize(lambda x: evaluate(x), h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-5)
            new_pop.append(res.x)
        
        # Replace worst with random perturbations
        for _ in range(num_pop - len(new_pop)):
            h = population[np.random.choice(num_pop)]
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h *= (ones_count / sum_h)
            new_pop.append(h)
        
        population = new_pop
        current_best = min(zip(scores, population), key=lambda x: x[0])[1]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    return best_h, best_c5, n_points
