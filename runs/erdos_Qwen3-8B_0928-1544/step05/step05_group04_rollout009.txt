    global_best = personal_best[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)
    
    start_time = time.time()
    for _ in range(200):
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 2.0 * r1 * (personal_best[i] - particles[i]) + 2.0 * r2 * (global_best - particles[i])
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] *= (0.5 * n_points) / sum_p
        current_values = np.array([objective(p) for p in particles])
        for i in range(n_particles):
            if current_values[i] < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_values[i]
        best_idx = np.argmin(personal_best_values)
        if personal_best_values[best_idx] < global_best_value:
            global_best = personal_best[best_idx].copy()
            global_best_value = personal_best_values[best_idx]
        if time.time() - start_time >= budget_s:
            break
    return global_best, global_best_value, n_points
