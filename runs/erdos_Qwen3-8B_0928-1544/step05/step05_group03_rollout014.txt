from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    time_used = 0
    max_iter = int(budget_s // 5)
    
    for _ in range(max_iter):
        scores = [evaluate(h) for h in population]
        sort_indices = np.argsort(scores)
        sorted_pop = [population[i] for i in sort_indices]
        parents = sorted_pop[:int(0.2 * population_size)]
        
        new_candidates = []
        for h in parents:
            if np.random.rand() < 0.4:
                perturb = np.random.normal(0, 0.1, n_points)
                h_perturb = h + perturb
                h_perturb = np.clip(h_perturb, 0, 1)
                sum_h = np.sum(h_perturb)
                scale = ones_count / sum_h
                new_h = h_perturb * scale
                new_candidates.append(new_h)
            else:
                x = np.linspace(0, 2, n_points)
                h_perturb = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2)
