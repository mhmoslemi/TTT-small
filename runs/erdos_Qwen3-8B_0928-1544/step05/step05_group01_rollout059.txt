    iterations = 0
    while True:
        for step in [0, 1]:
            for i in range(n_points):
                if step == 0 and i % 2 == 1:
                    current_val = h[i]
                    possible_vals = []
                    if current_val == 1.0:
                        possible_vals = [0.0, 0.5]
                    elif current_val == 0.0:
                        possible_vals = [1.0, 0.5]
                    else:
                        possible_vals = [1.0, 0.0]
                    for new_val in possible_vals:
                        h_tmp = h.copy()
                        h_tmp[i] = new_val
                        repair(h_tmp)
                        c5 = compute_c5(h_tmp)
                        if c5 < best_c5:
                            best_c5 = c5
                            h = h_tmp.copy()
                elif step == 1 and i % 2 == 0:
                    current_val = h[i]
                    possible_vals = []
                    if current_val == 1.0:
                        possible_vals = [0.0, 0.5]
                    elif current_val == 0.0:
                        possible_vals = [1.0, 0.5]
                    else:
                        possible_vals = [1.0, 0.0]
                    for new_val in possible_vals:
                        h_tmp = h.copy()
                        h_tmp[i] = new_val
                        repair(h_tmp)
                        c5 = compute_c5(h_tmp)
                        if c5 < best_c5:
                            best_c5 = c5
                            h = h_tmp.copy()
        for i in range(n_points):
            if random.random() < 0.1:
                current_val = h[i]
                if current_val == 1.0:
                    h[i] = 0.0
                elif current_val == 0.0:
                    h[i] = 1.0
                else:
                    h[i] = 1.0 if random.random() < 0.5 else 0.0
        repair(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        iterations += 1
        if iterations > 1000 or time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
