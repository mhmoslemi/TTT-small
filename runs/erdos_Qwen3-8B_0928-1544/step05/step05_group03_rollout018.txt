import numpy as np
from scipy.optimize import differential_evolution
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def fitness(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    result = differential_evolution(fitness, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1.0), recombination=0.9, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
