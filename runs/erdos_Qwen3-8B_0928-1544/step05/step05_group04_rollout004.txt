from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    # Initial optimization with SLSQP
    res = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,

        options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
    )
