    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    time_used = 0
    max_iter = int(budget_s // 10)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m = np.argmax(corr)
        return corr[m]

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            new_h = result.x
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * (2.0 / n_points)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                h = new_h.copy()
            else:
                perturb = np.random.normal(0, 0.1, n_points)
                new_h = np.clip(h + perturb, 0.0, 1.0)
                sum_new = np.sum(new_h)
                if abs(sum_new - n_points / 2) > 1e-5:
                    delta = (n_points / 2 - sum_new) / n_points
                    new_h += delta
                    new_h = np.clip(new_h, 0.0, 1.0)
                h = new_h.copy()
        time_used += 10
        if time_used >= budget_s:
            break

    h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-5:
        delta = (n_points / 2 - sum_h) / n_points
        h += delta
        h = np.clip(h, 0.0, 1.0)
    return h, best_c5, n_points
