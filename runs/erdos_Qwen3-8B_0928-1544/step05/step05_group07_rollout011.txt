from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Reduced for faster computation
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        population[i] = np.clip(pop, 0.0, 1.0)

        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    elapsed = 0
    max_iters = int(budget_s / 0.15)  # Estimating 0.15s per iteration
    
    for _ in range(max_iters):
        # Evaluate and sort
        fitness = np.array([compute_c5(h) for h in population])

        new_population = []

            if sum_child == 0:
