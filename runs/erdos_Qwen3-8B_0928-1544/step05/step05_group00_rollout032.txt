import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h_values = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Generate diverse initial guesses using Halton sequence
    sampler = qmc.Halton(d=1, scramble=True)
    positions = np.sort(sampler.random(n_points))
    h_initial_binary = np.zeros(n_points)
    h_initial_binary[0:ones_count] = 1
    np.random.shuffle(h_initial_binary)
    h_initial = np.zeros(n_points)
    h_initial[positions.argsort()] = h_initial_binary
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    # More extensive population and better optimiser configuration
    initial_population = [h_initial]
    for _ in range(24):
        perturbed = h_initial + np.random.normal(0, 0.05, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = np.copy(h_initial)
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 15)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(best_h)
            initial_population.pop(np.random.randint(len(initial_population)))
    
    return best_h, best_c5, n_points
