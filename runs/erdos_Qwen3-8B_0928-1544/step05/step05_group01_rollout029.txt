import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points // 2
    swarm_size = 50

    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        diff = sum_h - target_sum
        if diff != 0:
            indices = np.where((h > 0) & (h < 1))[0]
            if len(indices) > 0:
                delta = diff / len(indices)
                h[indices[:abs(diff)]] += delta if diff > 0 else -delta
        return h

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        initial_h[np.random.choice(n_points, size=target_sum, replace=False)] = 1.0
        indices = np.random.choice(n_points, size=5, replace=False)
        initial_h[indices] = 0.5

    swarm = [initial_h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    personal_best = [h.copy() for h in swarm]
    global_best = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):

            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            swarm[i] += velocities[i]
        for i in range(swarm_size):
            if np.random.rand() < 0.1:

    return global_best, best_c5, n_points
