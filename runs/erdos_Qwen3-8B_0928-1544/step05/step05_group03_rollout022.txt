            a.append(a_k)
            b.append(b_k)
        return [a0] + a + b
    
    def rebuild_h(coeffs):
        a0 = coeffs[0]
        a = coeffs[1:1+K]
        b = coeffs[1+K:]
        prediction = a0 / 2.0
        for k in range(K):
            prediction += a[k] * np.cos((k+1)*np.pi * x_freq)
            prediction += b[k] * np.sin((k+1)*np.pi * x_freq)
        return prediction
    
    def objective(coeffs):
        pred = rebuild_h(coeffs)
        h_clipped = np.clip(pred, 0, 1)
        sum_h = np.sum(h_clipped) * dx
        h_scaled = h_clipped * (1 / sum_h)
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    initial_coeffs = compute_coeffs(h)
    result = minimize(objective, initial_coeffs, method='L-BFGS-B', options={'maxiter': 200, 'ftol': 1e-7})
    best_coeffs = result.x
    
    final_h = rebuild_h(best_coeffs)
    h_clipped = np.clip(final_h, 0, 1)
    sum_h = np.sum(h_clipped) * dx
    h_scaled = h_clipped * (1 / sum_h)
    best_c5 = objective(best_coeffs)
    
    return h_scaled, best_c5, n_points
