                if len(ones_in_mask) >= delta:
                    h_candidate[mask[ones_in_mask[:delta]]] = 0
            elif delta < 0:
                zeros_in_mask = np.where(h_candidate[mask] == 0)[0]
                if len(zeros_in_mask) >= -delta:
                    h_candidate[mask[zeros_in_mask[: -delta]]] = 1
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
                h = h_candidate.copy()
    return best_h, compute_c5(best_h), 70
