            sorted_idx = np.argsort(np.abs(h - 0.5))

                idx = np.where(h == 0.5)[0]
                if not idx.size:
                    h[0] += np.sign(diff)

                    h[idx[0]] += np.sign(diff)
                h = np.clip(h, 0, 1)
        return h
