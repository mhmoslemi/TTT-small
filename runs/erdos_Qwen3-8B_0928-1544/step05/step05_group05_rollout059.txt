            c5_new = compute_c5(new_h)
            c5_old = compute_c5(h)
            if c5_new < c5_old:
                return new_h, alpha
            alpha *= 0.5
            if alpha < 1e-10:
                return h, 1.0
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(5000):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(best_h)
        new_h, alpha = line_search(best_h, grad)
        best_h = new_h
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        if _ % 200 == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed = best_h + perturbation
            perturbed = clip_and_scale(perturbed)
            if compute_c5(perturbed) < best_c5:
                best_h = perturbed
                best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
