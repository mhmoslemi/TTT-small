import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i % 2 == 0:
                initial_h[i] = 1.0  # Alternating 1 and 0 pattern

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_overlap = np.max(corr)
        return max_overlap * (2.0 / n_points)

    def constraint_func(h):
        return np.sum(h) - target_sum

    bounds = Bounds([0.0] * n_points, [1.0] * n_points)
    constraints = [{'type': 'eq', 'fun': constraint_func}]

    num_initial = 25
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):
        perturbed = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_population.append(perturbed)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    max_iter = int(budget_s / 100)
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,

            constraints=constraints,
