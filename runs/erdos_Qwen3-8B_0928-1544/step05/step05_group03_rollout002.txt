from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

        h += perturb
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        h *= scale
        population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    time_used = 0
    max_iter = int(budget_s // 10)
    
    for _ in range(max_iter):
        scores = np.array([evaluate(h) for h in population])
        indices = np.argsort(scores)
        selected = population[indices[:population_size // 2]]
        
        new_population = []
        for h in selected:
            perturb = np.random.normal(0, 0.2, n_points)
            h_perturb = h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            sum_h = np.sum(h_perturb)
            scale = ones_count / sum_h
            new_h = h_perturb * scale
            new_population.append(new_h)
        
        population = selected + new_population
        population = [np.clip(p, 0, 1) for p in population]
