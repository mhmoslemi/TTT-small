import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0.0, 1.0)] * n_points
    candidates = [h]
    for _ in range(4):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (required_sum / np.sum(perturbed))
        candidates.append(perturbed)
    
    while time.time() - start_time < budget_s:
        h = candidates[np.random.randint(len(candidates))]
        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        candidates.append(result.x)
        if len(candidates) > 10:
            candidates.pop(0)
    
    return best_h, best_c5, n_points
