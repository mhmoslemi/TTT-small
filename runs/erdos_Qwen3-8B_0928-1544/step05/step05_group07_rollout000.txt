from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    target_sum = n_points / 2

    if not np.allclose(np.sum(initial_h), target_sum):
        initial_h = initial_h / np.sum(initial_h) * target_sum
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 4)
    
    for iter in range(max_iter):
        h_perturbed = best_h + np.random.normal(0, 0.05 * (1 - iter / max_iter), n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (target_sum / sum_perturbed if sum_perturbed != 0 else 1.0)
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,

        )
