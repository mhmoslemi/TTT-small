import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def feasibility_proj(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) == 0:
            return h
        return h * (ones_count / np.sum(h))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s // 40)
    
    for _ in range(max_iter):
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturb = feasibility_proj(best_h + perturb)
        res = minimize(objective, h_perturb, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'ftol': 1e-6, 'maxiter': 50})
        if res.success:
            candidate = res.x
            c5 = objective(candidate)
            if c5 < best_c5:
                best_h = candidate
                best_c5 = c5
        time_used += 40
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
