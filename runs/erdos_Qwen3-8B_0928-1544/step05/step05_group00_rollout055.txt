import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    λ = 1e-4  # Penalty coefficient

    if 'initial_h_values' in globals():
        initial_population = [initial_h_values]
    else:
        def generate_initial_pop(num_initial):
            pop = []
            for _ in range(num_initial):
                h = np.random.uniform(0.25, 0.75, n_points - 1)
                last = (n_points / 2 - np.sum(h))
                last = np.clip(last, 0.25, 0.75)
                h = np.append(h, last)
                pop.append(h)
            return pop
        initial_population = generate_initial_pop(25)

    best_h = initial_population[0].copy()
    best_c5 = float('inf')

    bounds = [(0.0, 1.0)] * n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * (2.0 / n_points)
        penalty = λ * np.sum(h**2 * (1 - h)**2)
        return max_corr + penalty

    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)

    for _ in range(200):
        idx = np.random.randint(len(initial_population))
        current_h = initial_population[idx].copy()
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun - λ * np.sum(result.x**2 * (1 - result.x)**2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_population.append(result.x)
                del initial_population[np.random.randint(len(initial_population))]
    
    return best_h, best_c5, n_points
