import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Evolve population
    pop_size = 30
    population = [initial_h.copy() for _ in range(pop_size)]
    
    # DE parameters
    F = 0.8
    cr = 0.9
    max_steps = int(budget_s // 10)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for _ in range(max_steps):
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < cr:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            scale = ones_count / trial_sum
            trial *= scale
            if evaluate(trial) < evaluate(population[i]):
                population[i] = trial
    
    best = min(population, key=evaluate)
