import numpy as np
from cvxpy import Variable, Problem, Maximize, solvers

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    h = initial_h.copy()
    time_used = 0
    max_iter = int(budget_s // 2)
    
    for _ in range(max_iter):
        max_overlap_val = -np.inf
        best_k = 0
        possible_shifts = list(range(-(n_points - 1), n_points))
        
        for k in possible_shifts:
            current_sum = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    current_sum += h[i] * h[j]
            current_overlap = current_sum * dx
            if current_overlap > max_overlap_val:
                max_overlap_val = current_overlap
                best_k = k
        
        h_var = Variable(n_points)
        terms = [h_var[i] * h_var[i + best_k] for i in range(n_points) if 0 <= i + best_k < n_points]
        objective = Maximize(sum(terms) * dx)
        constraints = [sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]

        problem.solve(solver=solvers.ECOS)

    cross_corr = np.correlate(h, 1 - h, mode='full')
    overlap = cross_corr * dx
    c5_bound = np.max(overlap)
    return h, c5_bound, n_points
