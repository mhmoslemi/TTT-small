from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140  # Double the sample size for better resolution
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Generate initial binary pattern
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0  # Evenly spaced 1s

    # Normalize to ensure sum(h) = n_points / 2
    initial_h /= np.sum(initial_h)

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s * 0.8)

    for iter in range(max_iter):
        # Adaptive perturbation
        sigma = 0.1 * (1 - iter / max_iter)
        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)

        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))

        result = minimize(
