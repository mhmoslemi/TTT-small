import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 70
        initial_h = np.full(n_points, 0.5)
    required_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def perturb(h, required_sum, sigma=0.1, fraction=0.05):
        n = len(h)
        indices = np.random.choice(n, size=int(fraction * n), replace=False)
        h_perturbed = h.copy()
        h_perturbed[indices] += np.random.normal(0, sigma, len(indices))
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        scale = required_sum / np.sum(h_perturbed)
        h_perturbed *= scale
        return h_perturbed

    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)
    temp = 1.0
    cooling = 0.995
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_h = perturb(current_h, required_sum)
        new_c5 = compute_c5(new_h)
        delta = best_c5 - new_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        temp *= cooling

    return best_h, best_c5, 70
