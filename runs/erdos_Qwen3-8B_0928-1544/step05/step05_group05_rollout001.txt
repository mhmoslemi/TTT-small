    required_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        population.append(initial_h)
        for _ in range(4):

            population.append(perturbed)

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        # Selection, crossover, mutation
        new_pop = []
        for _ in range(len(population)):

            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[idx1], population[idx2]
            # Crossover
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0.0, 1.0)
            child_sum = np.sum(child)
            if abs(child_sum) > 1e-10:
