import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Construct a starting vector as a stepwise constant function
        initial_h = np.zeros(n_points)
        initial_h[:ones_count] = 1.0
        sum_h = np.sum(initial_h)
        scale = ones_count / sum_h
        initial_h *= scale
    initial_h = np.clip(initial_h, 0, 1)

    # Fitness function for evaluation
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    # Optimization parameters
    population_size = 60
    population = [initial_h.copy() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    time_used = 0
    max_iter = int(budget_s // 5)

    for _ in range(max_iter):
        # Evaluate the population
        scores = [evaluate(h) for h in population]
        sorted_idx = np.argsort(scores)
        # Select top 30% of the population as parents
        parents = sorted_idx[:int(0.3 * population_size)]
        # Crossover and mutation
        children = []
        for i in range(len(parents)):
            h_parent = population[parents[i]]
            # Apply Gaussian mutation
            noise = np.random.normal(0, 0.05, n_points)
            h_mutation = h_parent + noise
            h_mutation = np.clip(h_mutation, 0, 1)
            sum_h = np.sum(h_mutation)
            scale = ones_count / sum_h
            h_mutation *= scale
            children.append(h_mutation)
        # Replace older population with new candidates
        population = [population[i] for i in sorted_idx[int(0.3 * population_size):]] + children
        # Update best solution
        current_best = population[np.argmin(scores)]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        time_used += 5
        if time_used >= budget_s:
            break
    return best_h, best_c5, n_points
