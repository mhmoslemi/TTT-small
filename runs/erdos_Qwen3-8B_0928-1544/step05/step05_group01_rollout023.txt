            diff = sorted_h[i] - sorted_h[j]
            distance = np.sum(np.abs(diff))
            if distance > 0:
                attractiveness = np.exp(-gamma * distance)
                new_h = sorted_h[i] + beta * attractiveness * (sorted_h[j] - sorted_h[i]) * alpha
                new_h = repair_h(new_h)
                if np.sum(new_h) == required_sum:
                    fireflies[i] = new_h
                    current_c5[i] = compute_c5(new_h)
                    if current_c5[i] < best_c5:
                        best_c5 = current_c5[i]
                        best_h = new_h.copy()
                        best_idx = i

        if _ % 5 == 0:
            for i in range(len(fireflies)):
                if np.random.rand() < 0.1:
                    new_h = fireflies[i] + np.random.uniform(-0.2, 0.2, size=n_points)
                    new_h = np.clip(new_h, 0, 1)
                    new_h = repair_h(new_h)
                    if np.sum(new_h) == required_sum:
                        fireflies[i] = new_h
                        current_c5[i] = compute_c5(new_h)
                        if current_c5[i] < best_c5:
                            best_c5 = current_c5[i]
                            best_h = new_h.copy()
                            best_idx = i

        if _ == max_iter - 1:
            break

    return best_h, best_c5, n_points
