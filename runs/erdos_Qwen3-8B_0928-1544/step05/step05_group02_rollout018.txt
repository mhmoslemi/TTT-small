import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([0.5] * n_points)  # Default if not provided

    if 'initial_h_values' in globals():
        initial_h_values = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points)
        sine_h = (1 + np.sin(np.pi * x)) / 2
        initial_h_values = sine_h

    h = cp.Variable(n_points)
    t = cp.Variable((n_points, n_points))
    C = cp.Variable()

    # Constraints for t[i][j] = h[i] * h[j]
    for i in range(n_points):
        for j in range(n_points):
            t[i, j] <= h[i]
            t[i, j] <= h[j]
            t[i, j] >= h[i] + h[j] - 1
            t[i, j] >= 0

    # Constraints for h
    constraints = [cp.sum(h) == target_sum, 0 <= h, h <= 1]

    # Constraints for maximum overlap
    for k in range(-n_points + 1, n_points):
        sum_t = 0
        for i in range(n_points):
            j = (i + k) % n_points
            sum_t += t[i, j]
        constraints.append(sum_t * dx >= 1 - C)

    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.MOSEK)
    except:
        prob.solve(solver=cp.ECOS)

    best_h = h.value
    c5_bound = C.value
    return best_h, c5_bound, n_points
