import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    max_iterations = 2000
    learning_rate = 0.05
    decay_rate = 0.999
    start_time = time.time()
    
    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        
        dx = 2.0 / n_points
        k = lag
        i_plus_k = (np.arange(n_points) + k) % n_points
        i_minus_k = (np.arange(n_points) - k) % n_points
        grad = (1 - h[i_plus_k] - h[i_minus_k]) * dx
        
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 1e-12:
            h *= target_sum / current_sum
        
        current_c5 = max_val * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        learning_rate *= decay_rate
    
    return best_h, best_c5, n_points
