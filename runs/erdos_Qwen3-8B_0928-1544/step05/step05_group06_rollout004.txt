def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    time_used = 0
    max_iter = int(budget_s // 10)
    dx = 2.0 / n_points

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift_k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + shift_k < n_points:
                grad[i] += (1 - h[i + shift_k])
            if 0 <= i - shift_k < n_points:
                grad[i] -= h[i - shift_k]
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= ones_count / sum_h
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        time_used += 10
        if time_used >= budget_s:
            break
    return best_h, best_c5, n_points
