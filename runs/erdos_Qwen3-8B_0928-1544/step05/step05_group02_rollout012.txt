def run(seed=42, budget_s=1100, **kwargs):

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    for _ in range(1000):

        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        k_star = lag % n_points
