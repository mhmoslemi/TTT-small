import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0

    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        norm = np.sum(h_clipped)
        if norm == 0:
            return np.zeros_like(h)
        scale = required_sum / norm
        return h_clipped * scale

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial = initial_h_values
        pop = [initial]
        for _ in range(4):
            h_perturbed = initial + np.random.normal(0, 0.08, n_points)
            h_perturbed = project(h_perturbed)
            pop.append(h_perturbed)
    else:
        pop = [np.full(n_points, 0.5)]

    while len(pop) < 5:
        new_h = np.full(n_points, 0.5)
        new_h = project(new_h)
        pop.append(new_h)
    pop = pop[:30]

    best_h = pop[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(30):
            a, b = np.random.choice(pop, size=2, replace=False)
            alpha = np.random.rand()
            child = alpha * a + (1 - alpha) * b
            child = project(child)
            new_pop.append(child)

        scores = [evaluate(h) for h in new_pop]
        scores, new_pop = zip(*sorted(zip(scores, new_pop)))
        new_pop = new_pop[:20] + [np.random.normal(0, 0.05, n_points) for _ in range(10)]
        new_pop = [project(h) for h in new_pop]
        new_pop_scores = [evaluate(h) for h in new_pop]
        new_pop = [h for _, h in sorted(zip(new_pop_scores, new_pop))]

        if evaluate(new_pop[0]) < best_c5:
            best_c5 = evaluate(new_pop[0])
            best_h = new_pop[0]
        pop = new_pop

    return best_h, best_c5, n_points
