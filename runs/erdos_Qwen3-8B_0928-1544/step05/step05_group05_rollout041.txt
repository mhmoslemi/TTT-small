            perturb = np.random.normal(0, 0.05, n_points)
            h_perturb = initial_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb * (required_sum / np.sum(h_perturb))
            population.append(h_perturb)

    while len(population) < 30:
        new_h = np.random.uniform(0, 1, n_points)
        new_h = new_h * (required_sum / np.sum(new_h))
        population.append(new_h)
    population = population[:30]

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(30):
            indices = np.random.choice(len(population), size=2, replace=False)
            a, b = population[indices[0]], population[indices[1]]
            alpha = np.random.rand()
            child = alpha * a + (1 - alpha) * b
            noise = np.random.normal(0, 0.03, n_points)
            child += noise
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            if sum_child != 0:
                child = child * (required_sum / sum_child)
            new_population.append(child)
        fitness_new = np.array([evaluate(h) for h in new_population])
        best_idx = np.argmin(fitness_new)
        if fitness_new[best_idx] < best_c5:
            best_c5 = fitness_new[best_idx]
            best_h = new_population[best_idx].copy()
        population = new_population

    return best_h, best_c5, n_points
