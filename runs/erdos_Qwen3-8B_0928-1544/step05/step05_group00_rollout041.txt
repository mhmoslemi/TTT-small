import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Generate initial_h using Halton sequence
        def halton(n, base=2):
            sequence = []
            for i in range(n):
                v = 0.0
                i_copy = i
                while i_copy > 0:
                    v += (i_copy % base) / (base ** (1 + (i_copy // base)))
                    i_copy = i_copy // base
                sequence.append(v)
            return sequence
        
        # Generate 50 points in [0, 1)
        halton_ones = halton(ones_count, 2)
        scaled_points = [point * 2 for point in halton_ones]  # scale to [0, 2)
        bin_indices = set()
        for point in scaled_points:
            bin_index = int(point * n_points / 2)
            bin_indices.add(bin_index)
        initial_h = np.zeros(n_points)
        for idx in bin_indices:
            initial_h[idx] = 1.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    num_initial = 30
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        initial_population.append(perturbed)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 150)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)
            initial_population.pop(np.random.randint(len(initial_population)))
    
    return best_h, best_c5, n_points
