import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    perturbation = np.random.normal(0, 0.1, n_points)
    initial_population = [initial_h.copy() + perturbation]
    for _ in range(24):
        perturbation = np.random.uniform(-0.2, 0.2, n_points)
        initial_population.append(initial_h.copy() + perturbation)
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    constraint_sum = lambda h: np.sum(h) - ones_count
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        idx = np.random.randint(len(initial_population))
        current_h = initial_population[idx]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)
            initial_population.pop(np.random.choice(len(initial_population)))
            if best_c5 < 0.380876:
                return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
