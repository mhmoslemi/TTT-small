import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else None

    if initial_h is not None:
        h = initial_h.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i * 2] = 1.0

    # Normalize and clamp
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = _compute_c5(h, n_points)
    beta = 1e-4
    max_iter = 100
    start_time = time.time()
    iter_count = 0

    while iter_count < max_iter and time.time() - start_time < budget_s:
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            max_corr = np.max(corr) * (2.0 / n_points)
            sum_neg_log_h = -np.sum(np.log(x))
            sum_neg_log_1h = -np.sum(np.log(1 - x))
            sum_h = np.sum(x)
            penalty = (sum_h - target_sum) ** 2
            return max_corr + beta * (sum_neg_log_h + sum_neg_log_1h + penalty)

        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=[(1e-8, 1 - 1e-8) for _ in range(n_points)],
            options={'maxiter': 100}
        )

        if not result.success:
            break

        h = np.clip(result.x, 1e-8, 1 - 1e-8)
        sum_h = np.sum(h)
        factor = target_sum / sum_h
        h = h * factor

        current_c5 = _compute_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        beta *= 1.1
        iter_count += 1

    return best_h, best_c5, n_points

def _compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * (2.0 / n_points)
    return max_corr
