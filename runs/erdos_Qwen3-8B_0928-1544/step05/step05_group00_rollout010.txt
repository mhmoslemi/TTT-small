from scipy.optimize import minimize

    def objective(h):

    def constraint_sum(h):

    bounds = [(0.0, 1.0)] * n_points
    
    initial_population = [initial_h.copy()]

        perturbed = initial_h + np.random.uniform(-0.2, 0.2, n_points)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
