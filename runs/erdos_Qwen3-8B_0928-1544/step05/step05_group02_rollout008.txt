import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    swarm_size = 50
    max_iters = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    if 'initial_h_values' in globals():
        initial = initial_h_values.copy()
    else:
        initial = np.random.rand(n_points)
        initial = np.clip(initial, 0, 1)
        sum_initial = np.sum(initial)
        if sum_initial != 0:
            initial *= (target_sum / sum_initial)
        initial = initial.copy()
    swarm = [initial.copy() for _ in range(swarm_size)]
    velocities = np.random.uniform(-0.3, 0.3, (swarm_size, n_points))
    personal_best = [h.copy() for h in swarm]
    personal_vals = [np.max(np.correlate(h, 1-h, mode='full')) * (2.0 / n_points) for h in swarm]
    global_best = swarm[0].copy()
    global_val = personal_vals[0]
    def project(h):
        clamped = np.clip(h, 0, 1)
        s = np.sum(clamped)
        if s == 0:
            return clamped
        return clamped * (target_sum / s)
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > budget_s:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            vel = velocities[i] + inertia * (personal_best[i] - swarm[i]) * r1 + c2 * (global_best - swarm[i]) * r2
            new_pos = swarm[i] + vel
            new_pos = project(new_pos)
            swarm[i] = new_pos
            velocities[i] = vel
            curr_val = np.max(np.correlate(new_pos, 1 - new_pos, mode='full')) * (2.0 / n_points)
            if curr_val < personal_vals[i]:
                personal_vals[i] = curr_val
                personal_best[i] = new_pos.copy()
                if curr_val < global_val:
                    global_val = curr_val
                    global_best = new_pos.copy()
    return global_best, global_val, n_points
