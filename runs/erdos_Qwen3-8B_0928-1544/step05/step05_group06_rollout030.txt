        new_h = optimize_perturbation(hpert)
        current_c5 = objective(new_h)
        if current_c5 < best_c5:
            best_h = new_h
            best_c5 = current_c5
            print(f"New best C5: {best_c5}")
        else:
            new_h = optimize_perturbation(best_h + np.random.normal(0, 0.05, n_points))
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = new_h * (ones_count / np.sum(new_h)) if np.sum(new_h) > 0 else new_h
            current_c5 = objective(new_h)
            if current_c5 < best_c5:
                best_h = new_h
                best_c5 = current_c5
        if np.abs(best_c5 - 0.38085) < 1e-5:
            break
    return best_h, best_c5, n_points
