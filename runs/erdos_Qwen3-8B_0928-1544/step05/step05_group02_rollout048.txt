import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2  # 35.0
    swarm_size = 50
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0

    # Initialize particles
    particles = []
    velocities = []
    if 'initial_h_values' in globals():
        # Start with initial_h_values
        particles.append(initial_h_values)
        velocities.append(np.random.normal(0, 0.1, n_points))
        # Add other particles by perturbing initial_h_values
        for _ in range(swarm_size - 1):
            noise = np.random.normal(0, 0.2, n_points)
            new_h = initial_h_values + noise
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum != 0:
                new_h = new_h * (target_sum / new_h_sum)
            particles.append(new_h)
            velocities.append(np.random.normal(0, 0.1, n_points))
    else:
        # Generate random particles
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h))
            particles.append(h)
            velocities.append(np.random.normal(0, 0.1, n_points))

    # Initialize personal best and global best
    personal_best = [np.copy(p) for p in particles]

    best_c5 = float('inf')

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    for i in range(swarm_size):
        current_c5 = evaluate_c5(particles[i])
        if current_c5 < best_c5:
            best_c5 = current_c5
            global_best = np.copy(particles[i])

def run(seed=42, budget_s=1100, **kwargs):

            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum != 0:

            particles.append(new_h)
