                h_new = h + attraction * (j_h - h) + alpha * np.random.normal(size=n_points)
                sorted_new = np.sort(h_new)[::-1]
                indices = np.argsort(-h_new)
                h_repaired = np.zeros(n_points)
                h_repaired[indices[:ones_count]] = 1.0
                h_new = h_repaired
                if compute_c5(h_new) < compute_c5(h):
                    h = h_new
            if np.random.rand() < perturb_prob:
                perturb = np.random.normal(0, 0.1, size=n_points)
                h_new = h + perturb
                sorted_new = np.sort(h_new)[::-1]
                indices = np.argsort(-h_new)
                h_repaired = np.zeros(n_points)
                h_repaired[indices[:ones_count]] = 1.0
                h_new = h_repaired
                if compute_c5(h_new) < compute_c5(h):
                    h = h_new
            population[sorted_indices[idx]] = h
            if compute_c5(h) < best_c5:
                best_c5 = compute_c5(h)
                best_h = h
    
    return best_h, best_c5, n_points
