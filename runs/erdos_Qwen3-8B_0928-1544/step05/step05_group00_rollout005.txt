import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    initial_population = [initial_h.copy()]
    for _ in range(30):
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        initial_population.append(perturbed)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    initial_population = np.array(initial_population)
    initial_population = np.vstack([initial_population, initial_h.copy() for _ in range(10)])
    initial_population = initial_population[:40]
    initial_population = [h.tolist() for h in initial_population]
    
    max_iter = int(budget_s / 10)
    elapsed = 0
    
    for _ in range(max_iter):
        start = time.time()
        current_h = initial_population[np.random.randint(len(initial_population))]
        h = np.array(current_h)
        # Bounds
        bnds = Bounds([0.0]*n_points, [1.0]*n_points)
        # Constraints
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        res = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bnds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        elapsed += time.time() - start
        if elapsed >= budget_s:
            break
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
            initial_population.append(res.x)
            if len(initial_population) > 40:
                initial_population.pop(np.random.randint(len(initial_population)))
    
    return best_h, best_c5, n_points
