            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (particles[i] - particles[i]) + c2 * r2 * (best_h - particles[i])
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            particles[i] += velocities[i]
            h_clipped = np.clip(particles[i], 0, 1)
            sum_h = np.sum(h_clipped)
            if sum_h == 0:
                h_clipped = np.random.rand(n_points)
                h_clipped = h_clipped * target_sum / np.sum(h_clipped)
            else:
                h_clipped = h_clipped * target_sum / sum_h
            particles[i] = h_clipped
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * (2. / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = particles[i].copy()
        time_used += 1
    
    return best_h, best_c5, n_points
