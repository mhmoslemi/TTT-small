def halton_sequence(n, base):
    # Generate n points in [0, 1) using the given base
    sequence = []
    for i in range(1, n+1):
        x = 0
        denom = base
        num = i
        while num > 0:
            x += (num % base) / denom
            num = num // base
            denom *= base
        sequence.append(x)
    return np.array(sequence)

def run(seed=42, budget_s=1100, **kwargs):

        # Generate binary initial guess with Halton sequence
        def halton(n, base):
            seq = []
            for i in range(1, n+1):
                x = 0.0
                denom = base
                num = i
                while num > 0:
                    x += (num % base) / denom
                    num = num // base
                    denom *= base
                seq.append(x)
            return np.array(seq)
        base = 2
        positions = halton(ones_count, base)
        positions *= 2.0
        dx = 2.0 / n_points

        indices = np.clip(indices, 0, n_points - 1)
        counts = np.zeros(n_points, dtype=bool)
        for i in range(ones_count):
            if not counts[indices[i]]:
                counts[indices[i]] = True
        initial_h = counts.astype(float)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    initial_population = [initial_h.copy()]

        initial_population.append(perturbed)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    max_iter = int(budget_s / 100)
