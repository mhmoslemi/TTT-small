        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h.copy()
        
        # Perturbation every 100 iterations
        if iter % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h_perturb = h + noise
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
            new_c5_perturb = compute_c5(h_perturb)
            if new_c5_perturb < best_c5:
                best_c5 = new_c5_perturb
                best_h = h_perturb.copy()
                h = h_perturb.copy()
    
    return best_h, best_c5, n_points
