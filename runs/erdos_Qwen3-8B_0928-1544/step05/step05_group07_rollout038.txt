import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Ensure initial h meets constraints
    h = initial_h.copy()

    h /= np.sum(h)
    h *= ones_count
    h = np.clip(h, 0.0, 1.0)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s / 4)
    
    for _ in range(max_iter):
        # Perturb current best
        sigma = 0.05 * (1.0 - _ / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
