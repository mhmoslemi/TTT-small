            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        initial_candidates.append(result.x)
        # Keep top 10 candidates to maintain diversity
        initial_candidates = initial_candidates[-10:]

    return best_h, best_c5, n_points
