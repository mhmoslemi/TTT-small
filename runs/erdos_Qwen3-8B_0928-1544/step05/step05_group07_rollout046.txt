    final_c5 = _calculate_c5(best_h, n_points)
    return best_h, final_c5, n_points

def _calculate_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n_points)
