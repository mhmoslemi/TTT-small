import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()
    best_h = h
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    def objective(x):
        h_curr = x.copy()
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def feasibility(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped > 0:
            return x_clipped * (ones_count / sum_clipped)
        return x_clipped
    
    def perturb(x, temp):
        noise = np.random.normal(0, 0.1 * temp, n_points)
        return feasibility(x + noise)
    
    T = 1.0
    alpha = 0.995
    time_used = 0
    max_iter = int(budget_s // 10)
    
    for _ in range(max_iter):
        h_perturbed = perturb(h, T)
        res = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=[(0,1)]*n_points, options={'ftol':1e-8, 'maxiter':100})
        if res.success:
            h_new = res.x
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h = h_new
        T *= alpha
        time_used += 10
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
