        if current_c5_shift < best_c5:
            best_c5 = current_c5_shift
            best_h = shifted_right.copy()
        
        # Odd step
        h_candidate = h.copy()
        for i in range(1, n_points, 2):
            if np.random.rand() < 0.1:
                h_candidate[i] ^= 1
        h_candidate = repair(h_candidate)
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()
        
        # Shifted versions
        shifted_left = np.roll(h_candidate, shift=-1)
        current_c5_shift = compute_c5(shifted_left)
        if current_c5_shift < best_c5:
            best_c5 = current_c5_shift
            best_h = shifted_left.copy()
        shifted_right = np.roll(h_candidate, shift=1)
        current_c5_shift = compute_c5(shifted_right)
        if current_c5_shift < best_c5:
            best_c5 = current_c5_shift
            best_h = shifted_right.copy()
        
        # Update h to the best found so far
        h = best_h.copy()
    
    return best_h, best_c5, n_points
