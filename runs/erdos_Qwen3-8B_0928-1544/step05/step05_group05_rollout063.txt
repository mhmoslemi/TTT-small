import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0
    if 'initial_h_values' in globals():

    else:
        h = np.full(n_points, 0.5)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def compute_gradient(h):
        grad = np.zeros_like(h)
        epsilon = 1e-6
        for i in range(len(h)):
            h_perturb = h.copy()
            h_perturb[i] += epsilon
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            max_corr_perturb = np.max(corr_perturb) * (2.0 / n_points)

            max_corr_original = np.max(corr_original) * (2.0 / n_points)
            grad[i] = (max_corr_perturb - max_corr_original) / epsilon
        return grad
    
    best_h = h.copy()
