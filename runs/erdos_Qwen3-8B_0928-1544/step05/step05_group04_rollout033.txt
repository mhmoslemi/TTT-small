def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = float('inf')

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')

    start_time = time.time()

            h = particles[i]
            curr = compute_c5(h)
            if curr < best_c5:
                best_c5 = curr
