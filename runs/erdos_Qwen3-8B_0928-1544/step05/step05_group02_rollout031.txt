import numpy as np
import time
import scipy.fft as fft

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 30
    sigma = 0.25
    max_iterations = 1000
    sigma_decay = 0.995
    start_time = time.time()
    population = []

    if 'initial_h_values' in dir(globals()):
        population.append(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        sine_h = (1 + np.sin(np.pi * x)) / 2
        population.append(sine_h)
        for _ in range(population_size - 1):
            phase = np.random.uniform(0, 2 * np.pi)
            shifted_x = np.mod(x + phase, 2)
            sine_h_shifted = (1 + np.sin(np.pi * shifted_x)) / 2
            population.append(sine_h_shifted)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = float('inf')
    best_h = None

    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        evaluations = [evaluate_c5(h) for h in population]
        sorted_indices = np.argsort(evaluations)
        num_best = int(population_size * 0.2)
        best_candidates = [population[i] for i in sorted_indices[:num_best]]

        new_candidates = []
        for candidate in best_candidates:
            noise = np.random.normal(0, sigma, n_points)
            new_h = candidate + noise
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum != 0:
                new_h = new_h * (target_sum / new_h_sum)
            new_candidates.append(new_h)

        combined = population + new_candidates
        combined_evals = [evaluate_c5(h) for h in combined]
        sorted_combined = sorted(zip(combined, combined_evals), key=lambda x: x[1])
        population = [x[0] for x in sorted_combined[:population_size]]

        current_best = sorted_combined[0][0]
        current_c5 = evaluate_c5(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best

        sigma *= sigma_decay

    return best_h, best_c5, n_points
