    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    initial_population = [initial_h.copy()]

        perturb = np.random.normal(0, 0.08, n_points)
        initial_population.append(initial_h + perturb)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 25)
    
    for _ in range(max_iter):

        current_h = np.clip(current_h, 0, 1)
        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
