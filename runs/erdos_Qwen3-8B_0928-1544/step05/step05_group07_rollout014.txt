import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    initial_h = np.array(initial_h, dtype=np.float64)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 5)
    
    for iter in range(max_iter):
        if time_used >= budget_s:
            break
        sigma = 0.1 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping
        time_used += 3
        if time_used >= budget_s:
            break
    
    # Final validation
    best_h = np.clip(best_h, 1e-8, 1 - 1e-8)
    total = np.sum(best_h)
    best_h = best_h * (ones_count / total)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
