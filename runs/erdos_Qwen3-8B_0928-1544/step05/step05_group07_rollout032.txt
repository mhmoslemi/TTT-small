            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 4
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
