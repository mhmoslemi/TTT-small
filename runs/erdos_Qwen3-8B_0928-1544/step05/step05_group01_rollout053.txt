        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35}
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons)
        return res.x
    
    global_best = refine(global_best)
    return (global_best, best_c5, n)
