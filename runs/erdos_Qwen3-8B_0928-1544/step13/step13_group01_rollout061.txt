import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    assert np.isclose(np.sum(h), ones_count), "Initial h sum mismatch"

    def compute_c5(h, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    tabu_list = set()
    tabu_size = 20
    start_time = time.time()

    while time.time() - start_time < budget_s:
        while True:
            i, j = np.random.choice(n_points, size=2, replace=False)
            pair = tuple(sorted((i, j)))
            if pair not in tabu_list:
                delta = 0.05
                max_delta_i = 1 - h[i]
                max_delta_j = h[j]
                possible_delta = min(delta, max_delta_i, max_delta_j)
                new_h_i = h[i] + possible_delta
                new_h_j = h[j] - possible_delta
                if 0 <= new_h_i <= 1 and 0 <= new_h_j <= 1:

                    new_h[j] = new_h_j

                        tabu_list.add(pair)
