import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    max_generations = 1000
    time_budget = budget_s

    def clip_and_normalize(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        population = [clip_and_normalize(initial_h)]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            population.append(clip_and_normalize(h))
    else:
        population = [np.random.rand(n_points) * 0.5 for _ in range(population_size)]
        for i in range(len(population)):
            population[i] = clip_and_normalize(population[i])

    best_h = population[0].copy()
    best_c5 = float('inf')
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > time_budget:
            break

        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            fitness.append(current_c5)

        best_idx = np.argmin(fitness)
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_h = population[best_idx].copy()

            candidates = np.random.choice(population, tournament_size)
            selected_parents.append(min(candidates, key=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx))

        new_population = []
        for _ in range(population_size):
            parent1 = selected_parents[np.random.randint(len(selected_parents))]
            parent2 = selected_parents[np.random.randint(len(selected_parents))]
