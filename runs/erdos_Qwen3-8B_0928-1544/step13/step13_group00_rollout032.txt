            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[_]):
                population[_] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

            if time.time() - start_time >= budget_s:
                break

        population = [np.clip(h, 0.0, 1.0) for h in population]
        for i in range(len(population)):
            population[i] = np.clip(population[i], 0.0, 1.0) * (ones_count / np.sum(population[i])) if np.sum(population[i]) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    return best_h, best_c5, n_points
