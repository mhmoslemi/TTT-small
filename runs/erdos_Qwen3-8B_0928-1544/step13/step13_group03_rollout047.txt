import time

def run(seed=42, budget_s=1100, **kwargs):

    n_coarse = 35
    n_fine = 70
    ones_count = n_coarse // 2
    temp_coarse = 0.5
    temp_fine = 0.1
    max_iter = 500
    std_coarse = 0.1
    std_fine = 0.05
    decay = 0.995

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.block([np.ones(8.75), np.zeros(17.5), np.ones(8.75)])
        initial_h = np.clip(initial_h, 0, 1)
        initial_sum = np.sum(initial_h)
        initial_h = initial_h * (ones_count / initial_sum)

    def resample(h, n_new):
        return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    def anneal(h, n_new, temp, std, max_iter, decay):
        h = resample(h, n_new)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_new // 2) / sum_h
        best = h.copy()
        best_obj = objective(best)
        t = temp
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, std, n_new)
            trial = h + perturb
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            trial = trial * (n_new // 2) / sum_trial
            trial_obj = objective(trial)
            delta = trial_obj - best_obj
            if delta < 0 or np.random.rand() < np.exp(-delta / t):
                h = trial.copy()
                best = h.copy()
                best_obj = trial_obj
            t *= decay
        return best, best_obj

    start_time = time.time()
