            vel += omega * vel + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            vel = np.clip(vel, -max_vel, max_vel)
            velocities[i] = vel
            particles[i] += vel
            clipped = np.clip(particles[i], 0.0, 1.0)
            current_sum = np.sum(clipped)
            if current_sum == 0:
                particles[i] = np.full_like(clipped, 0.5)
            else:
                particles[i] = clipped * (target_sum / current_sum)
            new_c5 = compute_C5(particles[i])
            if new_c5 < best_c5[i]:
                pbest[i] = np.copy(particles[i])
                best_c5[i] = new_c5
    
    if np.min(best_c5) < compute_C5(gbest):
        gbest = np.copy(particles[np.argmin(best_c5)])
    
    clipped = np.clip(gbest, 0.0, 1.0)
    current_sum = np.sum(clipped)
    if current_sum == 0:
        gbest = np.full_like(clipped, 0.5)
    else:
        gbest = clipped * (target_sum / current_sum)
    
    return gbest, compute_C5(gbest), n_points
