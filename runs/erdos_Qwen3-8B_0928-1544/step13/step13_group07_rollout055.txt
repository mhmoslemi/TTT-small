    start_time = time.time()
    max_iterations = 1000
    perturb_count = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        m = idx_max - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + m < n_points:
                gradient[j] += 1 - h[j + m]
            if 0 <= j - m < n_points:
                gradient[j] -= h[j - m]
        best_alpha = 0.0
        best_new_c5 = best_c5
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005, 0.001, 0.0001]:
            h_candidate = h + alpha * (-gradient)

            if h_candidate_sum == 0:
                continue
            h_candidate = h_candidate * (n_points / 2) / h_candidate_sum
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            new_c5 = np.max(corr_candidate) * dx
            if new_c5 < best_new_c5:
                best_alpha = alpha

        if best_alpha > 0:
            h = h + best_alpha * (-gradient)

            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            corr_new = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_c5:
