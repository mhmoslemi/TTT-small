import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    max_iterations = 100
    elapsed = 0
    best_c5 = 1e10
    best_h = np.random.rand(n_points) * 0.5 * (ones_count / np.sum(np.random.rand(n_points) * 0.5))
    best_h = best_h * (ones_count / np.sum(best_h))

    if 'initial_h_values' in globals():
        best_h = initial_h_values.copy()
        best_h = best_h * (ones_count / np.sum(best_h))

    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        # Compute overlaps for all shifts
        overlaps = []
        for k in range(n_points):
            total = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                total += best_h[i] * (1 - best_h[j])
            overlaps.append(total)
        current_max = max(overlaps)
        if current_max < best_c5:
            best_c5 = current_max
        k_max = np.argmax(overlaps)

        # Construct Q for the selected shift
        Q = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k_max) % n_points
            Q[i][j] = 1
            Q[j][i] = 1

        x = cp.Variable(n_points)
        objective = cp.Maximize((x.T @ Q @ x) / 2)
        constraints = [
            cp.sum(x) == ones_count,
            0 <= x,
            x <= 1
        ]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS)
        except:
            continue

        if prob.status == 'optimal':
            h_val = x.value
            h_val = h_val * (ones_count / np.sum(h_val))
            h_val = np.clip(h_val, 0.0, 1.0)
            best_h = h_val
        else:
            break

    # Final c5 calculation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n_points)
    c5 = np.max(corr)
    return best_h, c5, n_points
