            continue
        coarse_h_new = coarse_h_new * (sum_coarse / coarse_h_new_sum)
        new_C5_c = compute_C5_c(coarse_h_new)
        current_C5_c = compute_C5_c(coarse_h)
        if new_C5_c < current_C5_c:
            coarse_h = coarse_h_new.copy()
            step_size_coarse = 0.1
        else:
            step_size_coarse *= 0.8

    fine_h = np.repeat(coarse_h, k)

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def compute_gradient_fine(h, lag):
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j])
                gradient[j] -= h[i]
        return gradient

    max_iter_fine = 500
    step_size_fine = 0.1
    best_h = fine_h.copy()
    best_c5 = compute_C5(fine_h)
    start_time = time.time()
    for _ in range(max_iter_fine):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(fine_h, 1 - fine_h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        grad = compute_gradient_fine(fine_h, lag)
        u = np.ones(n_points)
        g_dot_u = np.dot(grad, u)
        projected = (g_dot_u * u) / np.dot(u, u)
        adjusted_grad = grad - projected
        fine_h_new = fine_h - step_size_fine * adjusted_grad
        fine_h_new = np.clip(fine_h_new, 0.0, 1.0)
        fine_h_new_sum = np.sum(fine_h_new)
        if fine_h_new_sum == 0:
            continue
        fine_h_new = fine_h_new * (sum_fine / fine_h_new_sum)
        new_c5 = compute_C5(fine_h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = fine_h_new.copy()
            fine_h = fine_h_new.copy()
        else:
            step_size_fine *= 0.8

    fine_h_sum = np.sum(fine_h)
    if fine_h_sum == 0:
        fine_h = np.zeros(n_points)
    else:
        fine_h = fine_h * (sum_fine / fine_h_sum)

    return fine_h, best_c5, n_points
