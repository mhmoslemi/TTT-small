import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    population = [initial_h_scaled]
    for _ in range(199):
        h = initial_h_scaled + np.random.rand(n_points) * 0.1 - 0.05
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / np.sum(h))
        population.append(h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    population_size = 200
    mutation_rate = 0.15
    crossover_rate = 0.8
    stagnation = 0

    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + np.random.rand(n_points) * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                u = np.random.rand(n_points) * 0.5
            u = u * (ones_count / np.sum(u))
            new_c5 = objective(u)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = u.copy()
                stagnation = 0
            new_population.append(u)
        population = new_population
        if stagnation > 100:
            mutation_rate = min(0.25, mutation_rate + 0.01)
            crossover_rate = max(0.5, crossover_rate - 0.01)
