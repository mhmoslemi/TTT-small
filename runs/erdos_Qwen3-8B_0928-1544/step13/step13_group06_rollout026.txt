    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)

        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        gradients = np.zeros(n_points)

            gradients[i] = term1 + term2

        h += velocity
