        idx_max = np.argmax(corr)

        gradient = np.zeros(n_points)

            if 0 <= j + m < n_points:
                gradient[j] += (1 - h[j + m]) * dx
            if 0 <= j - m < n_points:
                gradient[j] -= h[j - m] * dx
        best_alpha = 0.0

            new_c5 = np.max(corr_candidate) * dx
