import numpy as np
import time
import scipy.ndimage

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points_coarse = 35
    n_points_fine = 70
    ones_count_coarse = n_points_coarse // 2
    ones_count_fine = n_points_fine // 2

    def upscale(h):
        x_old = np.linspace(0, n_points_coarse - 1, n_points_coarse)
        x_new = np.linspace(0, n_points_coarse - 1, n_points_fine)
        h_new = np.interp(x_new, x_old, h)
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h = np.sum(h_new)
        return h_new * (n_points_fine / 2) / sum_h if sum_h > 0 else h_new

    if 'initial_h_values' in globals():
        coarse_h = np.interp(np.linspace(0, n_points_coarse-1, n_points_coarse),
                             np.linspace(0, len(initial_h_values)-1, len(initial_h_values)),
                             initial_h_values)
        coarse_h = np.clip(coarse_h, 0.0, 1.0)
        sum_coarse = np.sum(coarse_h)
        coarse_h = coarse_h * (n_points_coarse / 2) / sum_coarse
        coarse_h = scipy.ndimage.gaussian_filter(coarse_h, sigma=1.5)
    else:
        coarse_h = np.zeros(n_points_coarse)
        coarse_h[:17] = 1.0
        sum_coarse = np.sum(coarse_h)
        coarse_h = coarse_h * (n_points_coarse / 2) / sum_coarse
        coarse_h = scipy.ndimage.gaussian_filter(coarse_h, sigma=1.5)

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    # Coarse grid SA
    best_coarse = coarse_h.copy()
    best_c5 = compute_c5(best_coarse)
    temp = 0.5
    iter_limit = 500

        sum_new = np.sum(new_h)
        if sum_new == 0:

        new_h = new_h * (n_points_coarse / 2) / sum_new
        current_c5 = compute_c5(new_h)
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_coarse = new_h.copy()
            best_c5 = current_c5

    # Fine grid SA

    temp = 0.1

        if time.time() - start_time > budget_s:

        perturbation = np.random.normal(0, 0.05, n_points_fine)
        new_h = fine_h + perturbation
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new == 0:
