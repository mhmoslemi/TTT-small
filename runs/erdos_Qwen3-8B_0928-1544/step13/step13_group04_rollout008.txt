    best_h = None
    simplex = [np.array(s) for s in simplex]
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate C5 for all points in the simplex
        c5_values = [compute_C5(s) for s in simplex]
        # Order the points by their C5 values
        ordered = sorted(zip(simplex, c5_values), key=lambda x: x[1])
        simplex = [s for s, _ in ordered]
        c5_values = [cv for _, cv in ordered]
        # Find the worst and best points
        worst_point = simplex[0]
        worst_c5 = c5_values[0]
        best_point = simplex[-1]
        best_c5 = c5_values[-1]
        # Compute centroid
        centroid = np.mean(simplex[:-1], axis=0)
        # Reflect the worst point
        reflected = centroid + (centroid - worst_point)
        # Evaluate reflected point
        reflected = repair(reflected, n_points)
        reflected_c5 = compute_C5(reflected)
        # Check if reflected is better than worst
        # If reflected is better than worst but worse than best
        if reflected_c5 < worst_c5 and reflected_c5 > best_c5:
            simplex[0] = reflected
        # If reflected is better than best
        elif reflected_c5 < best_c5:
            # Expand

            expansion = repair(expansion, n_points)
