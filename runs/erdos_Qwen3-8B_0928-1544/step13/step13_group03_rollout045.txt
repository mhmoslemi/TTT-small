import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 100
    generations = 100
    crossover_rate = 0.8
    mutation_rate = 0.2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))

    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            h = initial_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        else:
            h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h))
        population.append(h)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(-current_c5)
        sorted_indices = np.argsort(fitness)
        selected = population[sorted_indices[:population_size//2]]

        new_population = []
        for _ in range(population_size//2):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (target_sum / np.sum(child))
            new_population.append(child)
        population[:population_size//2] = new_population

        for h in population:
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    return best_h, best_c5, n_points
