            new_value = objective(population[i])
            if new_value < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = new_value
                if new_value < global_best_value:
                    global_best = population[i].copy()
                    global_best_value = new_value
                    best_h = global_best.copy()
                    best_c5 = global_best_value
        if iteration % 50 == 0 and iteration > 0:
            if np.allclose(personal_best_values, best_c5, atol=1e-6):
                max_v += 0.02
                if max_v > 0.2:
                    max_v = 0.2
                num_replace = int(population_size * 0.1)
                indices = np.argsort(personal_best_values)[-num_replace:]
                for idx in indices:
                    perturb = np.random.uniform(0.9, 1.1, n_points)
                    candidate = np.random.rand(n_points) * 0.5
                    candidate = np.clip(candidate, 0, 1)
                    candidate_sum = np.sum(candidate)
                    candidate_scaled = candidate * (ones_count / candidate_sum)
                    population[idx] = candidate_scaled
                    velocities[idx] = np.zeros(n_points)

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum)
    best_c5 = objective(best_h_scaled)
    return best_h_scaled, best_c5, n_points
