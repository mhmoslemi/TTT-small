import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 100
    max_iterations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    time_budget = budget_s

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)

    def generate_feasible():
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h)
        return h

    population = []
    for i in range(population_size):
        if 'initial_h_values' in globals() and i < population_size // 2:
            h = initial_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = generate_feasible()
            else:
                h = h * (target_sum / sum_h)
        else:
            h = generate_feasible()
        population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > time_budget:
            break
        fitness = np.array([compute_c5(p) for p in population])
        sorted_indices = np.argsort(-fitness)
        parents = [population[i] for i in sorted_indices[:population_size // 2]]
        new_population = []

        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points)
                child = np.concatenate([p1[:point], p2[point:]])
            else:
                child = p1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            if sum_c == 0:
                child = generate_feasible()
            else:
                child = child * (target_sum / sum_c)
            new_population.append(child)

        population = new_population
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
