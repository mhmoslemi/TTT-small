import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-7
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    T = 1.0
    cooling = 0.9995
    iterations = 0
    
    while time.time() - start_time < budget_s and iterations < 10000:
        iterations += 1
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            h_perturb_sum = np.sum(h_perturb)
            if h_perturb_sum == 0:
                h_perturb = h.copy()
            else:
                h_perturb = h_perturb * (ones_count / h_perturb_sum)
            new_c5 = compute_c5(h_perturb)
            grad[i] = (new_c5 - compute_c5(h)) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1)
        v_hat = v / (1 - beta2)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            continue
        h_new = h_new * (ones_count / h_new_sum)
        
        # Simulated Annealing
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
        
        # Cool down
        T *= cooling
    
    return best_h, best_c5, n_points
