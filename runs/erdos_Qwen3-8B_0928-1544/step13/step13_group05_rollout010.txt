from scipy.optimize import Bounds, minimize, LinearConstraint

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h * (ones_count / np.sum(initial_h))
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (ones_count / np.sum(initial_h))

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    bounds = Bounds(0, 1)
    constraint = LinearConstraint(np.ones(n_points), [ones_count], [ones_count])

    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-4, 'disp': False}
    )

    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
