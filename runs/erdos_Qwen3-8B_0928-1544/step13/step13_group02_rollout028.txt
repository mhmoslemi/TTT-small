        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_coarse
        return max_corr * dx

    pop_size = 50
    population = [coarse_h_initial.copy() for _ in range(pop_size)]
    for h in population:
        h_sum = np.sum(h)
        if h_sum == 0:
            h[:] = 0.5
        h[:] = h * (n_coarse_half / h_sum)
        h[:] = np.clip(h, 0.0, 1.0)

    best_coarse = coarse_h_initial.copy()
    best_c5_coarse = objective_coarse(best_coarse)
    start_time = time.time()
    max_generations = 200

    while time.time() - start_time < budget_s * 0.5:
        parent1, parent2 = np.random.choice(population, size=2, replace=False)
        child = parent1.copy()
        idx = np.random.randint(n_coarse)
        child[idx:] = parent2[idx:]
        if np.random.rand() < 0.1:
            for i in range(n_coarse):
                if np.random.rand() < 0.1:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0.0, 1.0)
        total = np.sum(child)
        if total == 0:
            child[:] = 0.5
        child[:] = child * (n_coarse_half / total)
        child[:] = np.clip(child, 0.0, 1.0)
        fitness = objective_coarse(child)
        if fitness < objective_coarse(population[np.random.randint(pop_size)]):
            population[np.random.randint(pop_size)] = child.copy()
        if fitness < best_c5_coarse:
            best_c5_coarse = fitness
            best_coarse = child.copy()
        if np.random.rand() < 0.1:
            best_coarse = np.random.rand(n_coarse) * (n_coarse_half / np.sum(np.random.rand(n_coarse)))

    fine_h_initial = np.zeros(n_fine)
    for i in range(n_fine // 2):
        fine_h_initial[2*i] = best_coarse[i]
        fine_h_initial[2*i + 1] = best_coarse[i]
    total = np.sum(fine_h_initial)
    fine_h_initial = fine_h_initial * (n_fine_half / total)
    fine_h_initial = np.clip(fine_h_initial, 0.0, 1.0)

    def objective_fine(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_fine
        return max_corr * dx

    h_fine = fine_h_initial.copy()
    learning_rate = 0.01
    max_iterations = 500
    start_time_fine = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time_fine > budget_s * 0.5:
            break
        grad = np.zeros(n_fine)
        epsilon = 1e-5
        for i in range(n_fine):
            h_plus = h_fine.copy()
            h_plus[i] += epsilon
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum == 0:
                h_plus[:] = 0.5
            else:
                h_plus = h_plus * (n_fine_half / h_plus_sum)
                h_plus = np.clip(h_plus, 0.0, 1.0)
            obj_plus = objective_fine(h_plus)
            h_minus = h_fine.copy()
            h_minus[i] -= epsilon
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum == 0:
                h_minus[:] = 0.5
            else:
                h_minus = h_minus * (n_fine_half / h_minus_sum)
                h_minus = np.clip(h_minus, 0.0, 1.0)
            obj_minus = objective_fine(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * epsilon)
        h_fine -= learning_rate * grad
        total = np.sum(h_fine)
        if total == 0:
            h_fine[:] = 0.5
        else:
            h_fine = h_fine * (n_fine_half / total)
            h_fine = np.clip(h_fine, 0.0, 1.0)
        current_c5 = objective_fine(h_fine)
        if current_c5 < best_c5_coarse:
            best_c5_coarse = current_c5

    return h_fine, best_c5_coarse, n_fine
