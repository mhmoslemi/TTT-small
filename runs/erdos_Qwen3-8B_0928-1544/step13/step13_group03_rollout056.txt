import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():

    bounds = [(0.0, 1.0) for _ in range(n_points)]
