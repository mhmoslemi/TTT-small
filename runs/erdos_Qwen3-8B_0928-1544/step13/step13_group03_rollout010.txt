            scale = target_sum / second_sum
            second_half *= scale
        h[35:] = second_half
        # Evaluate
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
    return h, best_c5, n_points
