import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    target_sum = n_points / 2
    population_size = 100

        if _ < population_size // 2:
            h = initial_h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h[i] += np.random.normal(0, 0.05)
            h = np.clip(h, 0, 1)

            h = np.zeros(n_points)
            h[indices] = 1.0
            population.append(h)
