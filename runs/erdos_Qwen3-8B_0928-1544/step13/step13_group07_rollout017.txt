                h = h * (n_points / 2) / h_sum
            corr_new = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        prev_grad = gradient
    return best_h, best_c5, n_points
