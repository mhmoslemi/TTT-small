import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    h = h / np.sum(h) * ones_count
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    velocity = np.zeros(n_points)
    learning_rate = 1.0
    momentum = 0.9
    max_iterations = 10000
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            term1 = 0.0
            if 0 <= i_plus_k < n_points:
                term1 = 1.0 - h[i_plus_k]
            i_minus_k = i - k
            term2 = 0.0
            if 0 <= i_minus_k < n_points:
                term2 = -h[i_minus_k]
            gradient[i] = term1 + term2
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * ones_count
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= 0.995
    h = np.clip(best_h, 0, 1)
    h = h / np.sum(h) * ones_count
    return h, best_c5, n_points
