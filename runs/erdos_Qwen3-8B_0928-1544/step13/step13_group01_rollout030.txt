    population = h0[:n_points].copy()
    population_size = 200
    F_start, F_end = 0.4, 0.9

    mutation_rate = 0.25
    tournament_size = 5

    best_h = population[0].copy()
    best_c5 = 0.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    start_time = time.time()
    F, CR = F_start, CR_start
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            idxs = np.random.choice(population_size, tournament_size, replace=False)
            parents = population[idxs]
            parent1, parent2 = parents[np.argmin(objective(parents))], parents[np.argmin(objective(parents))]

            child = alpha * parent1 + (1 - alpha) * parent2
            u = np.copy(child)

                u += perturb

            if u_sum != 0:
                u = u * (ones_count / u_sum)
            new_c5 = objective(u)
            if new_c5 < objective(population[_]):
