import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points / 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.75
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
    
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
    
    best_h = result.x
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time
    
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
