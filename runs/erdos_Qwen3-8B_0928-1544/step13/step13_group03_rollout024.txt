import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    if 'initial_h_values' in globals():
        original_n = len(initial_h_values)
        original_x = np.linspace(0, 2, original_n)
        new_x = np.linspace(0, 2, n_points)
        h_initial = np.interp(new_x, original_x, initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = 0.5 * (1 + np.cos(np.pi * x / 2))
    
    sum_initial = np.sum(h_initial)
    if not np.isclose(sum_initial, n_points / 2):
        h_initial = h_initial * (n_points / 2) / sum_initial
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    start_time = time.time()
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-7, 'eps': 1e-8})
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-7, 'eps': 1e-8})
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
