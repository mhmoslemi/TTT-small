import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h0 = initial_h_values
    else:
        h0 = np.zeros(n_points)
        h0[:ones_count] = 1.0
        h0 = h0 + np.random.rand(n_points) * 0.1
        h0 = h0 / np.sum(h0) * ones_count
    population = [h0]
    population_size = 200
    F = 0.9
    CR = 0.95
    max_iter = 100
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    
    def compute_c5(h, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * ones_count
        population.append(h)
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        for idx in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            v = population[a] + F * (population[b] - population[c])
            u = np.copy(population[idx])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0, 1)
            u = u * (ones_count / np.sum(u))
            current_c5 = compute_c5(u, n_points)

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = u.copy()
        F = max(0.3, F - 0.01)
        CR = min(1.0, CR + 0.01)
    
    return best_h, best_c5, n_points
