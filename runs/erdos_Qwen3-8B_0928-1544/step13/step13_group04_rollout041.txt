    start_time = time.time()
    
    alpha = 1.0
    gamma = 2.0
    rho = 0.5
    sigma = 0.5
    
    while time.time() - start_time < budget_s:
        simplex_c5 = [compute_C5(p) for p in simplex]
        sorted_indices = np.argsort(simplex_c5)
        sorted_simplex = [simplex[i].copy() for i in sorted_indices]
        sorted_c5 = np.array(simplex_c5)[sorted_indices]
        
        worst_point = sorted_simplex[-1]
        worst_c5 = sorted_c5[-1]
        best_point = sorted_simplex[0]
        best_c5_val = sorted_c5[0]
        
        centroid = np.mean(sorted_simplex[:-1], axis=0)
        
        reflected_point = centroid + alpha * (centroid - worst_point)
        reflected_point = feasible_point(reflected_point)
        reflected_c5 = compute_C5(reflected_point)
        
        if reflected_c5 < worst_c5:
            new_simplex = [p.copy() for p in sorted_simplex[:-1]]
            new_simplex.append(reflected_point)
            simplex = new_simplex
            if reflected_c5 < best_c5_val:
                best_c5 = reflected_c5
                best_h = reflected_point.copy()
        elif reflected_c5 < best_c5_val:
            expanded_point = centroid + gamma * (centroid - worst_point)
            expanded_point = feasible_point(expanded_point)
            expanded_c5 = compute_C5(expanded_point)
            if expanded_c5 < worst_c5:
                new_simplex = [p.copy() for p in sorted_simplex[:-1]]
                new_simplex.append(expanded_point)
                simplex = new_simplex
                if expanded_c5 < best_c5_val:
                    best_c5 = expanded_c5
                    best_h = expanded_point.copy()
            else:
                new_simplex = [p.copy() for p in sorted_simplex[:-1]]
                new_simplex.append(reflected_point)
                simplex = new_simplex
        else:
            contracted_point = centroid + rho * (centroid - worst_point)
            contracted_point = feasible_point(contracted_point)
            contracted_c5 = compute_C5(contracted_point)
            if contracted_c5 < worst_c5:
                new_simplex = [p.copy() for p in sorted_simplex[:-1]]
                new_simplex.append(contracted_point)
                simplex = new_simplex
                if contracted_c5 < best_c5_val:
                    best_c5 = contracted_c5
                    best_h = contracted_point.copy()
            else:
                shrinked_points = [best_point + sigma * (p - best_point) for p in sorted_simplex[:-1]]
                shrinked_points = [feasible_point(p) for p in shrinked_points]
                new_simplex = shrinked_points.copy()
                new_simplex.append(best_point)
                simplex = new_simplex
    
    return best_h, best_c5, n_points
