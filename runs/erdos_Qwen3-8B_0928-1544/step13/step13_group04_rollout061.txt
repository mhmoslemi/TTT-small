        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j])
                gradient[j] -= h[i]
        h -= step_size * gradient
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h = h * (ones_count / h_sum)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            step_size *= 0.8

    h_sum = np.sum(best_h)
    if h_sum == 0:
        best_h = np.zeros(n_points)
    else:
        best_h = best_h * (ones_count / h_sum)
    return best_h, best_c5, n_points
