import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_half = n_points // 2

    # Initialize h
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * n_half

    # Bounds: each h_i ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Constraint: sum(h) = 35
    def constraint_sum(x):
        return np.sum(x) - n_half
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    # Objective function
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Run optimizer
    result = minimize(
        objective,
        h_initial,
        method='COBYLA',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000}
    )

    # Post-processing: scale to meet sum constraint
    best_h = result.x
    sum_h = np.sum(best_h)
    if np.abs(sum_h) < 1e-10:
        best_h = np.zeros_like(best_h)
    else:
        best_h = best_h * (n_half / sum_h)

    # Final evaluation
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * (2.0 / n_points)
    return best_h, best_c5, n_points
