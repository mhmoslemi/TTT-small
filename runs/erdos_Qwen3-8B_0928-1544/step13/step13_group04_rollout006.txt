import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)

    h = initial_h_scaled.copy()
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_C5(h)
    start_time = time.time()
    max_iterations = 10000
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            gradient[i] += (1 - h[j])
            gradient[j] -= h[i]
        velocity = momentum * velocity + gradient

        h_new_sum = np.sum(h_new)
