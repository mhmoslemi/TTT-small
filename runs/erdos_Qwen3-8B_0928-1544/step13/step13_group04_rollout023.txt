        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_C5(h)
    start_time = time.time()
    max_iterations = 2000
    step_size = 0.1
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            gradient[i] += (1 - h[j])
            gradient[j] -= h[i]
        u = np.ones(n_points)
        g_dot_u = np.dot(gradient, u)
        projected = (g_dot_u * u) / np.dot(u, u)
        adjusted_gradient = gradient - projected
        
        h_new = h - step_size * adjusted_gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            continue
        h_new_scaled = h_new * (ones_count / h_new_sum)
        
        new_c5 = compute_C5(h_new_scaled)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new_scaled.copy()
            h = h_new_scaled.copy()
        else:
            step_size *= 0.995
    
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.zeros(n_points)
    else:
        h = h * (ones_count / h_sum)
    return h, best_c5, n_points
