    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=max_iterations,
        popsize=population_size,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )

    best_h = scale_pop(result.x)
    best_c5 = result.fun
    return best_h, best_c5, n_points
