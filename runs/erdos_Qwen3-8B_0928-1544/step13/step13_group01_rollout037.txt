import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
        scaling = (n_points / 2) / h_sum
        h = h * scaling
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / h.size
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    max_tabu = 20
    tabu_list = []
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        if i == j:
            continue
        delta_max_i = 1.0 - h[i]
        delta_max_j = h[j]
        delta_max = min(delta_max_i, delta_max_j)
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0, delta_max)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        pair = tuple(sorted((i, j)))
        if pair in tabu_list:
            continue
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            tabu_list.append(pair)
            if len(tabu_list) > max_tabu:
                tabu_list.pop(0)
    
    return best_h, best_c5, n_points
