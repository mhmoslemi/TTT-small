import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 30
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        population.append(initial_h)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
            population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0]
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()
        new_population = [current_best.copy()]
        for _ in range(population_size - 1):
            indices = np.random.choice(len(population), size=3, replace=False)
            parent1_idx = indices[np.argmin(fitness[indices])]
            parent1 = population[parent1_idx]
            indices = np.random.choice(len(population), size=3, replace=False)
            parent2_idx = indices[np.argmin(fitness[indices])]
            parent2 = population[parent2_idx]
            crossover_point = np.random.randint(1, n_points)
            child = np.zeros(n_points)
            child[:crossover_point] = parent1[:crossover_point]
            child[crossover_point:] = parent2[crossover_point:]
            mutate = np.random.rand() < 0.2
            if mutate:
                noise = np.random.normal(0, 0.05, n_points)

                    child = np.zeros(n_points)
