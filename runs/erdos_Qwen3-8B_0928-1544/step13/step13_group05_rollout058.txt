            if new_pos_sum == 0:
                continue
            new_pos_scaled = new_pos * (ones_count / new_pos_sum)
            population[i] = new_pos_scaled
            new_value = objective(population[i])
            if new_value < objective(personal_best[i]):
                personal_best[i] = population[i].copy()
                if new_value < best_c5:
                    best_c5 = new_value
                    best_h = population[i].copy()
                    global_best = population[i].copy()
        if iteration % 20 == 0 and iteration > 0:
            improvement = np.min([objective(p) for p in personal_best]) - best_c5
            if improvement < 1e-6:
                max_v += 0.02
                if max_v > 0.25:
                    max_v = 0.25
                num_replace = int(population_size * 0.2)
                indices = np.argsort([objective(p) for p in personal_best])[-num_replace:]
                for idx in indices:
                    perturb = np.random.uniform(0.9, 1.1, n_points)
                    candidate = population[idx] * perturb
                    candidate = np.clip(candidate, 0, 1)
                    candidate_sum = np.sum(candidate)
                    if candidate_sum == 0:
                        continue
                    candidate_scaled = candidate * (ones_count / candidate_sum)
                    population[idx] = candidate_scaled
                    velocities[idx] = np.zeros(n_points)

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if best_h_sum == 0:
        best_h = np.zeros(n_points)
        best_h_sum = 0
    best_h_scaled = best_h * (ones_count / best_h_sum) if best_h_sum != 0 else np.zeros(n_points)
    best_c5 = objective(best_h_scaled)
    return best_h_scaled, best_c5, n_points
