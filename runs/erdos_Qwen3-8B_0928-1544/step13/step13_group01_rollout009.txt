import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
    else:
        h0 = np.random.rand(n_points) * 0.5
        h0_sum = np.sum(h0)
        h0 = h0 * (ones_count / h0_sum)
    population = [h0]
    population_size = 200
    F = 0.5
    CR = 0.8
    best_h = h0.copy()
    best_c5 = np.max(np.correlate(h0, 1 - h0, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    stale_count = 0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(population_size):
            a, b, c = random.sample(range(population_size), 3)
            v = population[a] + F * (population[b] - population[c])
            u = population[_].copy()
            for j in range(n_points):
                if random.random() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0, 1)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = obj(u_scaled)
            if new_c5 < obj(population[_]):
                new_pop.append(u_scaled)
            else:
                new_pop.append(population[_])
        population = new_pop
        current_best = min(population, key=obj)
        current_c5 = obj(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            stale_count = 0
        else:
            stale_count += 1
        if stale_count > 200:
            F = max(0.4, F - 0.01)
            CR = min(0.95, CR + 0.01)
            stale_count = 0
    return best_h, best_c5, n_points
