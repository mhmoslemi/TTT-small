def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

            rolled = np.roll(h, k)
            integrals.append(np.sum(h * rolled))
        k_max = np.argmin(integrals)
        grad = np.roll(h, k_max) + np.roll(h, -k_max)

        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            alpha *= 0.95
    return best_h, best_c5, n_points

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    max_overlap = 0.0

    return max_overlap
