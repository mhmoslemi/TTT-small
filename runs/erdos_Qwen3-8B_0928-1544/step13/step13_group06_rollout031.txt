import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = h / np.sum(h) * ones_count
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    max_iterations = 2000
    learning_rate = 0.1
    start_time = time.time()
    decay_rate = 0.999
        
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        derivatives = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            term1 = 0
            if 0 <= i_plus_k < n_points:
                term1 = 1 - h[i_plus_k]
            term2 = 0
            if 0 <= i_minus_k < n_points:
                term2 = -h[i_minus_k]
            derivatives[i] = term1 + term2
        idx = np.argmin(derivatives)
        direction = 1 if derivatives[idx] < 0 else -1
        delta = learning_rate * direction
        h[idx] += delta
        h[idx] = np.clip(h[idx], 0, 1)
        h = h / np.sum(h) * ones_count
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
    return best_h, best_c5, n_points
