import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mut_rate = 0.1
    tour_size = 5
    max_gen = 100
    
    def _eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    pop = []
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
        h_init = np.clip(h_init, 0, 1)
        sum_h = np.sum(h_init)
        if sum_h > 0:
            h_init = h_init * (n_points / 2) / sum_h
        pop.append(h_init)
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            pop.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            pop.append(h)
    
    best_h = pop[0]
    best_c5 = _eval_c5(best_h)
    start = time.time()
    
    for gen in range(max_gen):
        if time.time() - start > budget_s:
            break
        
        fit = [_eval_c5(h) for h in pop]
        parents = []
        for _ in range(pop_size // 2):
            candidates = np.random.choice(pop, tour_size, replace=False)
            best = min(candidates, key=lambda x: _eval_c5(x))
            parents.append(best)
        
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            pt = np.random.randint(1, n_points)
            c1 = np.concatenate([p1[:pt], p2[pt:]])
            c2 = np.concatenate([p2[:pt], p1[pt:]])
            offspring.extend([c1, c2])
        
        for j in range(len(offspring)):
            h_cand = offspring[j].copy()
            for k in range(n_points):
                if np.random.rand() < mut_rate:
                    h_cand[k] += np.random.normal(0, 0.05)
                    h_cand[k] = np.clip(h_cand[k], 0, 1)
            sum_h = np.sum(h_cand)
            if sum_h > 0:
                h_cand = h_cand * (n_points / 2) / sum_h
            offspring[j] = h_cand
        
        new_pop = offspring
        best_idx = np.argmin(fit)
        new_pop.append(pop[best_idx])
        fits = [_eval_c5(h) for h in new_pop]
        sorted_idx = np.argsort(fits)
        pop = [new_pop[i] for i in sorted_idx[:pop_size]]
        
        current_fit = [_eval_c5(h) for h in pop]
        best_idx = np.argmin(current_fit)
        current_best = pop[best_idx]
        current_c5 = current_fit[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    return best_h, best_c5, n_points
