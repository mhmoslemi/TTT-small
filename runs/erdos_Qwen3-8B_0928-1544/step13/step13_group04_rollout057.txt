            if reflection_c5 < centroid_c5:
                expansion = centroid + (reflection_scaled - centroid) * 1.5
                expansion = np.clip(expansion, 0, 1)
                current_sum = np.sum(expansion)
                if current_sum == 0:
                    expansion_scaled = np.zeros(n_points)
                else:
                    scale = target_sum / current_sum
                    expansion_scaled = expansion * scale
                expansion_scaled = np.clip(expansion_scaled, 0, 1)
                expansion_c5 = compute_C5(expansion_scaled)
                if expansion_c5 < reflection_c5:
                    new_s = sorted_s[:-1] + [expansion_scaled]
                else:
                    new_s = sorted_s[:-1] + [reflection_scaled]
            else:
                contraction = centroid + (reflection_scaled - centroid) * 0.5
                contraction = np.clip(contraction, 0, 1)
                current_sum = np.sum(contraction)
                if current_sum == 0:
                    contraction_scaled = np.zeros(n_points)
                else:
                    scale = target_sum / current_sum
                    contraction_scaled = contraction * scale
                contraction_scaled = np.clip(contraction_scaled, 0, 1)
                contraction_c5 = compute_C5(contraction_scaled)
                if contraction_c5 < compute_C5(worst):
                    new_s = sorted_s[:-1] + [contraction_scaled]
                else:
                    new_s = [sorted_s[0]]
                    for p in sorted_s[1:]:
                        new_p = sorted_s[0] + 0.5 * (p - sorted_s[0])
                        new_p = np.clip(new_p, 0, 1)
                        current_sum = np.sum(new_p)
                        if current_sum == 0:
                            new_p_scaled = np.zeros(n_points)
                        else:
                            scale = target_sum / current_sum
                            new_p_scaled = new_p * scale
                        new_p_scaled = np.clip(new_p_scaled, 0, 1)
                        new_s.append(new_p_scaled)
                    new_s.append(sorted_s[-1])
        simplex = new_s
        
        current_c5 = compute_C5(sorted_s[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = sorted_s[0].copy()
    
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    if current_sum == 0:
        best_h = np.zeros(n_points)
    else:
        best_h = best_h * (target_sum / current_sum)
    best_h = np.clip(best_h, 0, 1)
    return best_h, best_c5, n_points
