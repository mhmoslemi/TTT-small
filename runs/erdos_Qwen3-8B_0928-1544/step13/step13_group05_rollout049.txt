import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    ones_count = n_points // 2
    population_size = 150
    w = 0.8
    c1 = 2.0
    c2 = 2.0
    max_v = 0.2
    max_iterations = 800
    stagnation_threshold = 100
    early_stop_threshold = 0.002

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum)

    def scale_to_sum(h):
        return h * (ones_count / np.sum(h))

    population = [initial_h_scaled.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.uniform(0.8, 1.2, n_points)
        candidate = initial_h_scaled * perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = scale_to_sum(candidate)
        population.append(candidate)

    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in population]
    personal_best_values = [objective(p) for p in population]
    global_best = population[0].copy()
    global_best_value = objective(population[0])
    best_h = global_best.copy()
    best_c5 = global_best_value
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 60:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = velocities[i] + w * velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            new_velocity = np.clip(new_velocity, -max_v, max_v)
            velocities[i] = new_velocity
            new_pos = population[i] + new_velocity
            new_pos = np.clip(new_pos, 0, 1)
            new_pos_scaled = scale_to_sum(new_pos)
            population[i] = new_pos_scaled
            new_value = objective(population[i])
            if new_value < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = new_value
                if new_value < global_best_value:
                    global_best = population[i].copy()
                    global_best_value = new_value
                    best_h = global_best.copy()
                    best_c5 = global_best_value
        if iteration % 50 == 0 and iteration > 0:
            if np.max(np.abs(np.array(personal_best_values) - best_c5)) < early_stop_threshold:

                    max_v = 0.4
                num_replace = int(population_size * 0.2)

                    candidate = np.random.rand(n_points) * 0.5
                    candidate = np.clip(candidate, 0, 1)
