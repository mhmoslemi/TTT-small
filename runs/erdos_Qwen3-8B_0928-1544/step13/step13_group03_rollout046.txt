        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.1, n_coarse)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        if sum_p == 0:
            continue
        perturbed = perturbed * (ones_coarse / sum_p)
        new_c5 = compute_c5(perturbed)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = perturbed.copy()
            best_c5 = new_c5
        temp *= 0.99

    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, best_h, kind='cubic', fill_value='extrapolate')
    fine_h = f(x_fine)
    fine_h = np.clip(fine_h, 0, 1)
    sum_f = np.sum(fine_h)
    if sum_f == 0:
        fine_h = np.zeros(n_fine)
    else:
        fine_h = fine_h * (ones_fine / sum_f)

    best_h_fine = fine_h.copy()
    best_c5_fine = compute_c5(best_h_fine)
    temp_f = 0.1
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_fine)
        perturbed = best_h_fine + noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        if sum_p == 0:
            continue
        perturbed = perturbed * (ones_fine / sum_p)
        new_c5 = compute_c5(perturbed)
        delta = new_c5 - best_c5_fine
        if delta < 0 or np.random.rand() < np.exp(-delta / temp_f):
            best_h_fine = perturbed.copy()
            best_c5_fine = new_c5
        temp_f *= 0.99

    return best_h_fine, best_c5_fine, n_fine
