import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    h = h / np.sum(h) * ones_count
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    max_iterations = 10000
    learning_rate = 0.1
    decay_rate = 0.999
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        derivatives = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            term1 = 0
            if 0 <= i_plus_k < n_points:
                term1 = 1 - h[i_plus_k]
            term2 = 0
            if 0 <= i_minus_k < n_points:
                term2 = -h[i_minus_k]
            derivatives[i] = term1 + term2
        idx = np.argmin(derivatives)
        direction = 1 if derivatives[idx] < 0 else -1
        delta = learning_rate * direction / (iteration + 1)
        h[idx] += delta
        h[idx] = np.clip(h[idx], 0, 1)
        h = h / np.sum(h) * ones_count
        current_c5 = np.max(corr) * (2.0 / n_points)

            best_h = h.copy()
        learning_rate *= decay_rate
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
