import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    dx = 2.0 / n_points

    # Initialize particles
    if 'initial_h_values' in globals():
        particles = [initial_h_values.copy()]
        for _ in range(49):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * ones_count / np.sum(h)
            particles.append(h)
    else:
        particles = []
        for _ in range(50):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * ones_count / np.sum(h)
            particles.append(h)
    particles = np.array(particles, dtype=np.float64)
    
    # Initialize velocities
    velocities = np.random.rand(50, n_points) * 0.1

    # Fitness function
    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Initial fitness
    fitness = np.array([compute_fitness(p) for p in particles])
    best_fitness = np.min(fitness)
    gbest = np.copy(particles[np.argmin(fitness)])
    
    # PSO parameters
    inertia = 0.9
    inertia_decay = 0.999
    c1 = c2 = 2.0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(50):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (particles[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -2.0, 2.0)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            if sum_h == 0:
                particles[i] = np.full(n_points, 0.5)
            else:
                particles[i] = particles[i] * ones_count / sum_h
        
        # Compute new fitness
        new_fitness = np.array([compute_fitness(p) for p in particles])
        for i in range(50):
            if new_fitness[i] < fitness[i]:
                fitness[i] = new_fitness[i]
                particles[i] = np.copy(particles[i])
        
        # Update global best
        if np.min(fitness) < best_fitness:
            best_fitness = np.min(fitness)
            gbest = np.copy(particles[np.argmin(fitness)])
            best_h = np.copy(gbest)
        
        # Decay inertia
        inertia *= inertia_decay
    
    best_h = np.copy(gbest)
    return best_h, best_fitness, n_points
