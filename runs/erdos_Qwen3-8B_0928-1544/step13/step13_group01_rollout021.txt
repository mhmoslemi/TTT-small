import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tabu_list = []
    tabu_size = 20
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        max_delta_i = 1.0 - h[i]
        max_delta_j = h[j]
        max_delta = min(max_delta_i, max_delta_j)
        if max_delta <= 0:
            continue
        delta = np.random.uniform(-max_delta, max_delta)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        move = (i, j)
        if move not in tabu_list:
            current_c5 = compute_c5(new_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
            tabu_list.append(move)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
    
    return best_h, best_c5, n_points
