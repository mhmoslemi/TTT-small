import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=target_sum, replace=False)
        h[indices] = 1.0
        h = h.astype(float)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    learning_rate = 0.01
    max_iter = int((budget_s - 5) / 0.1)
    
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = corr.max() * (2.0 / n_points)
        k_max = corr.argmax()
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k_max) % n_points
            j_minus = (i - k_max) % n_points
            gradient[i] = (1 - h[j_plus]) - h[j_minus]
        
        h = h - learning_rate * gradient * (2.0 / n_points)
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif current_c5 >= best_c5 - 1e-5:
            learning_rate *= 0.5
            if learning_rate < 1e-6:
                break
        
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
