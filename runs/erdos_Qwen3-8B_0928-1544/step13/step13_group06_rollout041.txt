        h_initial = np.clip(h_initial, 0, 1)
        scale = ones_count / np.sum(h_initial)
        h_initial = h_initial * scale
        particles.append(h_initial)
        # Generate other particles
        for _ in range(n_particles - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * ones_count / np.sum(h)
            particles.append(h)
    else:
        # Generate all particles randomly
        for _ in range(n_particles):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * ones_count / np.sum(h)
            particles.append(h)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [np.copy(p) for p in particles]
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    gbest_idx = np.argmin([compute_c5(p) for p in particles])
    gbest = particles[gbest_idx]
    best_c5 = compute_c5(gbest)
    start_time = time.time()
    c1 = 2.0
    c2 = 2.0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        omega = 0.9 - (elapsed / budget_s) * 0.5
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            v = velocities[i] * omega + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            v = np.clip(v, -2, 2)
            new_pos = particles[i] + v
            new_pos = np.clip(new_pos, 0, 1)
            total = np.sum(new_pos)
            if total == 0:
                new_pos = np.zeros_like(new_pos)
            else:
                new_pos = new_pos * ones_count / total
            particles[i] = new_pos
            current_fitness = compute_c5(particles[i])
            if current_fitness < compute_c5(pbest[i]):
                pbest[i] = particles[i].copy()
            if current_fitness < compute_c5(gbest):
                gbest = particles[i].copy()
                best_c5 = current_fitness
    best_h = gbest
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
