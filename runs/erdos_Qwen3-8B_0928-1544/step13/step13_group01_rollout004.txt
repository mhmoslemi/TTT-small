import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    
    def compute_c5(current_h):
        corr = np.correlate(current_h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tabu_list = []
    tabu_size = 20
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        delta_i = 1.0 - h[i]
        delta_j = h[j]
        delta_max = min(delta_i, delta_j)
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0, delta_max)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        move = tuple(sorted((i, j)))
        if move in tabu_list:
            continue
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            tabu_list = []
            h = h_new
        else:
            tabu_list.append(move)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
    return best_h, best_c5, n_points
