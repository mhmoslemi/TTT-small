import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:17] = 1.0
        h[-17:] = 1.0

        h = np.clip(h, 0.0, 1.0)
    best_c5 = float('inf')
    global_best = h.copy()
    start_time = time.time()
    max_iterations = 200
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Optimize first half
        fixed_part = h[35:]
        S = required_sum - np.sum(fixed_part)
        bounds = [(0.0, 1.0) for _ in range(35)]
        def obj1(x):
            h_candidate = np.concatenate([x, fixed_part])
            corr = np.correlate(h_candidate, 1.0 - h_candidate, mode='full')
            return np.max(corr) * (2.0 / n_points)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S}]
        res = minimize(obj1, h[:35], bounds=bounds, constraints=cons, method='SLSQP')
