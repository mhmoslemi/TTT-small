import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
    else:
        h0 = np.random.rand(n_points) * 0.5
        h0_sum = np.sum(h0)
        h0 = h0 * (ones_count / h0_sum)
    population = [h0]
    population_size = 200
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
        population.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = 0.6
    CR = 0.9
    mutation_rate = 0.1
    stagnation = 0

    best_c5 = float('inf')
    best_h = h0.copy()

    while time.time() - start_time < budget_s:

            break
        new_population = []
        for _ in range(population_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            v = a + F * (b - c)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u = u * (ones_count / u_sum)
            new_c5 = objective(u)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = u.copy()
            new_population.append(u)
        population = new_population
        stagnation = 0
        for h in population:
            if objective(h) < best_c5:
                stagnation = 0
                break
        else:
            stagnation += 1
            if stagnation > 100:

    return best_h, best_c5, n_points
