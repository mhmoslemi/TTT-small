import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    if initial_h_sum == 0:
        initial_h = np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / initial_h_sum)
    h = initial_h.copy()

    time_start = time.time()
    max_iterations = 50
    best_h = h.copy()
    best_c5 = float('inf')

    for _ in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        # Compute all k's
        constraints = []
        h_var = cp.Variable(n_points)
        C_var = cp.Variable()
        constraints.append(cp.sum(h_var) == ones_count)
        for k in range(-n_points + 1, n_points):
            overlap = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    overlap += h[i] * (1 - h[j]) * dx
            dO_k = np.zeros(n_points)
            for j in range(n_points):
                term1 = (1 - h[j + k]) if 0 <= j + k < n_points else 0.0
                term2 = h[j - k] if 0 <= j - k < n_points else 0.0
                dO_k[j] = dx * (term1 - term2)
            A_k = np.dot(dO_k, h)
            constraints.append(C_var >= cp.sum(cp.multiply(dO_k, h_var)) - A_k + overlap)
        for j in range(n_points):
            constraints.append(h_var[j] >= 0)
            constraints.append(h_var[j] <= 1)
        problem = cp.Problem(cp.Minimize(C_var), constraints)
        try:
            problem.solve()
            if problem.status == 'OPTIMAL':
                h = h_var.value
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum != 0:
                    h = h * (ones_count / h_sum)
                # Compute C5
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:

    c5_bound = np.max(corr) * dx
