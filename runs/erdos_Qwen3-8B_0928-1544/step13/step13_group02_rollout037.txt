    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    if np.sum(h) == 0:
        h = np.zeros_like(h) + 0.5
    h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = 0.0
    def calculate_c5(h):
        n = len(h)
        dx = 2.0 / n
        max_val = 0.0
        best_k = 0
        for k in range(n):
            shifted = np.roll(h, k)
            val = np.dot(h, 1 - shifted)
            val *= dx
            if val > max_val:
                max_val = val
                best_k = k
        return max_val, best_k
    current_c5, best_k = calculate_c5(h)
    best_c5 = current_c5
    start_time = time.time()
    learning_rate = 0.1
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        current_c5, best_k = calculate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        shifted = np.roll(h, best_k)
        gradient = (1 - shifted) * (2.0 / n_points)
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            continue
        h = h * (ones_count / sum_h)
    return best_h, best_c5, n_points
