import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / np.sum(h))
    best_c5 = np.inf
    best_h = h.copy()
    dx = 2.0 / n_points
    learning_rate = 0.1
    decay_rate = 0.995
    start_time = time.time()
    
    for _ in range(10000):
        # Compute all overlaps
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlaps[k] = np.dot(h, 1 - shifted) * dx
        max_overlap = np.max(overlaps)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        # Gradient for the current maximum overlap
        k_idx = np.argmax(overlaps)
        grad = 1 - np.roll(h, k_idx)
        # Update h
        h -= learning_rate * grad
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / np.sum(h))
            continue
        h = h * (ones_count / h_sum)
        # Decay learning rate
        learning_rate *= decay_rate
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
