import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.ndimage import gaussian_filter
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        original_h = initial_h_values
        x_original = np.linspace(0, 2, len(original_h))
        x_new = np.linspace(0, 2, n_points)
        new_h = np.interp(x_new, x_original, original_h)
    else:
        h = np.zeros(n_points)
        h[:35] = 1.0
        h[35:70] = 0.0
        h[70:105] = 1.0
        h[105:] = 0.0
        new_h = gaussian_filter(h, sigma=5)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        new_h = new_h * (target_sum / new_h_sum) if new_h_sum > 0 else np.zeros(n_points)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)

    constraint = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-10, options={'maxiter': 200})

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(10):
            result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-10, options={'maxiter': 200})
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
