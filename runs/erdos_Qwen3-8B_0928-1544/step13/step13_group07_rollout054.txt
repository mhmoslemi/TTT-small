def run(seed=42, budget_s=1100, **kwargs):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iterations = 1000
    for idx in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        m = idx_max - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + m < n_points:
                gradient[j] += 1 - h[j + m]
            if 0 <= j - m < n_points:
                gradient[j] -= h[j - m]
        best_alpha = 0.0
        best_new_c5 = best_c5
        alphas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]
        for alpha in alphas:

            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            h_candidate_sum = np.sum(h_candidate)
            if h_candidate_sum == 0:
                continue
            h_candidate = h_candidate * (n_points / 2) / h_candidate_sum
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            new_c5 = np.max(corr_candidate) * dx

        if best_alpha > 0:

            if h_sum > 0:

            corr_new = np.correlate(h, 1 - h, mode='full')
