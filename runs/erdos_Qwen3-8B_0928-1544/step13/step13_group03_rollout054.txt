        perturb = np.random.normal(0, 0.05, n_fine)
        h_new = h_fine + perturb
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new != 0:
            h_new = h_new * (n_fine / 2) / sum_new
        else:
            h_new = np.zeros(n_fine)
        c5_new = compute_c5_fine(h_new)
        delta = c5_new - best_c5_fine
        if delta < 0 or np.random.rand() < np.exp(-delta / temp_fine):
            best_c5_fine = c5_new
            best_h_fine = h_new.copy()
        temp_fine *= cooling_rate_fine
    
    return best_h_fine, best_c5_fine, n_fine
