def objective(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx

def clip_and_scale(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    sum_h = np.sum(h_clipped)
    if sum_h == 0:
        return np.zeros_like(h)
    scale = (n_points // 2) / sum_h
    return h_clipped * scale

    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
