        strategy='best1bin',
        mutation=(0.5, 1.0),
        crossover=0.9,
        popsize=50,
        tol=1e-6,
        maxiter=500,
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
