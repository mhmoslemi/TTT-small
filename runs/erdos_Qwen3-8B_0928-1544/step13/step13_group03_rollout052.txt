            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            clipped = np.clip(child, 0, 1)
            sum_clipped = np.sum(clipped)
            if sum_clipped == 0:
                h = np.random.rand(n_points)
                h = h * (target_sum / np.sum(h))
                clipped = h
            else:
                clipped = clipped * (target_sum / sum_clipped)
            new_population.append(clipped)
        
        population[:population_size // 2] = top_parents
        population[population_size // 2:] = new_population
        
        for h in population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
