import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    ones_count = n_points // 2
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_resampled = interp1d(x_initial, initial_h_values, kind='linear', fill_value='extrapolate')(x_new)
        h_resampled = np.clip(h_resampled, 0, 1)
        sum_h = np.sum(h_resampled)
        scaling = ones_count / sum_h
        h_resampled *= scaling
    else:
        # Create symmetric initial guess
        h_resampled = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Constraints
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization options
    result = minimize(
        objective,
        h_resampled,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8, 'eps': 1e-8}
    )
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:  # Leave 10s buffer
        return result.x, float('inf'), n_points
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
