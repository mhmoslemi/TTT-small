import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    
    # Adam optimizer parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 1
    
    # Simulated annealing parameters
    T = 1.0
    cooling_rate = 0.9995
    
    for _ in range(10000):
        if time.time() - start_time >= budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        index = np.argmax(corr)
        k = index - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            if 0 <= i_plus_k < n_points:
                term1 = 1.0 - h[i_plus_k]
            else:
                term1 = 0.0
            if 0 <= i_minus_k < n_points:
                term2 = h[i_minus_k]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and scale constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * (ones_count / n_points)
        
        # Simulated annealing
        new_corr = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if np.random.rand() < np.exp((best_c5 - new_corr) / T):
            best_h = h.copy()
            best_c5 = new_corr
        else:
            if new_corr < best_c5:
                best_h = h.copy()
                best_c5 = new_corr
        
        T *= cooling_rate
    
    return best_h, best_c5, n_points
