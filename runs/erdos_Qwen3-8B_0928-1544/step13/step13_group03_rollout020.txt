import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.8
    max_generations = 100

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.5
        initial_sum = np.sum(initial_h)
        initial_h = initial_h * (ones_count / initial_sum)
    
    # Initialize population with initial_h and perturbations
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            new_h = new_h * (ones_count / new_sum)
        population.append(new_h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [-compute_c5(h) for h in population]
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        # Select top 50% and generate new population
        top_half = sorted_pop[:population_size // 2]
        new_population = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(top_half, 2, replace=False)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            # Apply mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child != 0:
                child = child * (ones_count / sum_child)
            new_population.append(child)
        # Replace bottom half
        population = sorted_pop[:population_size // 2] + new_population
        # Update best solution
        for candidate in population:
            current_c5 = compute_c5(candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate.copy()
    
    return best_h, best_c5, n_points
