import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    population_size = 150
    ones_count = n_points // 2
    w = 0.8
    c1 = 2.0
    c2 = 2.0
    max_v = 0.2
    max_iterations = 1000
    stagnation_threshold = 20
    evaluation_count = 0

    def objective(h):
        nonlocal evaluation_count
        evaluation_count += 1
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
    else:
        initial_h_scaled = np.random.rand(n_points) * 0.5
        initial_h_scaled = initial_h_scaled * (ones_count / np.sum(initial_h_scaled))

    population = [initial_h_scaled.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.uniform(0.9, 1.1, n_points)
        candidate = initial_h_scaled * perturb
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        candidate_scaled = candidate * (ones_count / candidate_sum)
        population.append(candidate_scaled)

    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in population]
    personal_best_values = [objective(p) for p in population]
    global_best = population[0].copy()
    global_best_value = objective(population[0])
    best_h = global_best.copy()
    best_c5 = global_best_value
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = velocities[i] + w * velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            new_velocity = np.clip(new_velocity, -max_v, max_v)
            velocities[i] = new_velocity
            new_pos = population[i] + new_velocity
            new_pos = np.clip(new_pos, 0, 1)
            new_pos_sum = np.sum(new_pos)
            if new_pos_sum == 0:

            new_pos_scaled = new_pos * (ones_count / new_pos_sum)
            population[i] = new_pos_scaled
            new_value = objective(population[i])
            if new_value < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = new_value
                if new_value < global_best_value:
                    global_best = population[i].copy()
                    global_best_value = new_value
                    best_h = global_best.copy()
                    best_c5 = global_best_value
        if iteration % stagnation_threshold == 0 and iteration > 0:
            if np.max(np.abs(np.array(personal_best_values) - best_c5)) < 1e-6:

                if max_v > 0.2:

                num_replace = int(population_size * 0.1)
                indices = np.argsort(personal_best_values)[-num_replace:]
                for idx in indices:
