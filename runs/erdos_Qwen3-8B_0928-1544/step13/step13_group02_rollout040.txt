        h_candidate = h - learning_rate * grad
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_sum_candidate = np.sum(h_candidate)
        if h_sum_candidate != 0:
            scaling_factor = (ones_count) / h_sum_candidate
            h_candidate = h_candidate * scaling_factor
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h = h_candidate
        # Decay learning rate
        learning_rate *= 0.9995
    
    # Final evaluation
    final_c5 = float('inf')
    for k in range(n_points):
        shifted_h = np.roll(best_h, k)
        integral = np.dot(best_h, 1 - shifted_h) * dx
        if integral > final_c5:
            final_c5 = integral
    return best_h, final_c5, n_points
