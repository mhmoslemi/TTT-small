import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    max_iterations = 400
    crossover_rate = 0.8
    mutation_rate = 0.2
    std_dev = 0.1

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        sum_initial = np.sum(initial_h)
        if sum_initial != 0:
            initial_h = initial_h * (ones_count / sum_initial)
        population = [initial_h]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / sum_h)
            population.append(h)
    else:
        population = [np.random.rand(n_points) * 0.5 for _ in range(population_size)]
        for h in population:
            sum_h = np.sum(h)
            if sum_h == 0:
                h[:] = np.random.rand(n_points) * 0.5
            h[:] = h * (ones_count / sum_h)

    best_h, best_c5 = None, float('inf')
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(-c5)
        indices = np.argsort(fitness)
        selected = [population[i] for i in indices[:population_size // 2]]
        new_pop = []
        for _ in range(population_size // 2):
            p1, p2 = selected[np.random.randint(len(selected))], selected[np.random.randint(len(selected))]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.normal(0, std_dev)
                    child[j] = np.clip(child[j], 0, 1)
            sum_child = np.sum(child)
            child = child * (ones_count / sum_child) if sum_child != 0 else child
            new_pop.append(child)
        population = selected + new_pop
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
