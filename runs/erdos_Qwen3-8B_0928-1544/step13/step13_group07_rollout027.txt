import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_population = [initial_h_values.copy() for _ in range(50)]
    else:
        initial_population = [np.random.rand(n_points) * 0.5 for _ in range(50)]
        for i in range(len(initial_population)):
            h = initial_population[i]
            h_sum = np.sum(h)
            initial_population[i] = h * (n_points / 2) / h_sum

    best_h = initial_population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate_erdos_solution(h) for h in initial_population])
        selected = []
        for _ in range(50):
            indices = np.random.choice(50, 5, replace=False)
            best_idx = indices[np.argmin(fitness[indices])]
            selected.append(initial_population[best_idx].copy())

        new_population = []
        for i in range(0, 50, 2):
            p1, p2 = selected[i], selected[i+1]
            idx = np.random.randint(1, n_points-1)
            c1 = np.concatenate([p1[:idx], p2[idx:]])
            c2 = np.concatenate([p2[:idx], p1[idx:]])
            new_population.extend([c1, c2])

        for i in range(len(new_population)):
            h = new_population[i]
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            h_sum = np.sum(h)
            new_population[i] = h * (n_points / 2) / h_sum if h_sum > 0 else h

        initial_population = new_population
        for h in initial_population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
