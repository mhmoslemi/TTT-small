import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)

    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    T = 0.1 * current_C5
    alpha = 0.995
    min_T = 1e-5
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        epsilon = 0.1 * (T / 10)  # Adjust based on temperature
        delta = np.random.uniform(0, epsilon) * np.min([h[i], 1 - h[j]])
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        new_C5 = compute_C5(h_new)
        delta_C5 = new_C5 - current_C5
        
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new
            current_C5 = new_C5

            best_C5 = current_C5
            best_h = h.copy()
        
        T *= alpha
        if T < min_T:
            break
    
    return best_h, best_C5, n_points
