            new_c5 = np.max(new_corr) * dx
            if new_c5 < current_c5 - alpha_current * np.dot(grad, grad):
                break
            alpha_current *= beta
        
        h = new_h_projected
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if np.abs(best_c5 - current_c5) < tol:
            break
    
    return best_h, best_c5, n_points
