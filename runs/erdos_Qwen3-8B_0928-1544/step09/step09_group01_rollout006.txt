import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    ones_count = n_points // 2
    population_size = 50
    
    # Initialize population
    def create_initial_population():
        if 'initial_h_values' in globals():
            initial_h = initial_h_values.copy()
            initial_h_sum = np.sum(initial_h)
            initial_h_scaled = initial_h * (ones_count / initial_h_sum)
            population = [initial_h_scaled]
        else:
            population = []
            for _ in range(population_size):
                h = np.random.rand(n_points) * 0.5
                h_sum = np.sum(h)
                if h_sum == 0:
                    h = np.random.rand(n_points) * 0.5
                h_scaled = h * (ones_count / h_sum)
                population.append(h_scaled)
        return population
    
    population = create_initial_population()
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def evaluate_candidate(h):
        nonlocal best_c5
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
        return current_c5
    
    start_time = time.time()

    F = 0.8  # Mutation factor
    CR = 0.95  # Crossover probability

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            
            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)
            
            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            
            # Clip to [0,1]
            u = np.clip(u, 0.0, 1.0)
            
            # Scale to meet sum constraint
            u_sum = np.sum(u)
            if u_sum == 0:
                continue  # skip if sum is zero (unlikely)
            u_scaled = u * (ones_count / u_sum)
            
            # Evaluate
            evaluate_candidate(u_scaled)
