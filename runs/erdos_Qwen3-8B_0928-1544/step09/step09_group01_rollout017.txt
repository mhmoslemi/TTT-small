            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            current_c5 = objective(population[i])
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
        generation_count += 1
        
        if best_c5 < prev_best_c5:
            prev_best_c5 = best_c5
            gen_without_improvement = 0
        else:
            gen_without_improvement += 1
        
        if gen_without_improvement > 100:
            F = np.clip(F + 0.05, 0.5, 1.0)
            CR = np.clip(CR - 0.05, 0.5, 1.0)
            gen_without_improvement = 0
        
        if generation_count % 10 == 0:
            sorted_indices = sorted(range(len(population)), key=lambda i: objective(population[i]))
            top_5 = sorted_indices[:5]
            for idx in top_5:
                h = population[idx].copy()
                def obj(x):
                    x = np.clip(x, 0, 1)
                    x_sum = np.sum(x)
                    if x_sum == 0:
                        return np.inf
                    x_scaled = x * (ones_count / x_sum)
                    return objective(x_scaled)
                result = scipy.optimize.minimize(obj, h, method='Nelder-Mead', tol=1e-6, options={'maxiter': 100})
                if result.fun < objective(h):
                    population[idx] = result.x
        
    return best_h, best_c5, n_points
