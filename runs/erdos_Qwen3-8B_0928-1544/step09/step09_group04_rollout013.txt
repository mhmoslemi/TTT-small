import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full')

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate(h) for h in population])
        indices = np.argsort(fitness)
        new_population = []
        for _ in range(population_size // 2):
            idx1 = indices[np.random.choice(10)]
