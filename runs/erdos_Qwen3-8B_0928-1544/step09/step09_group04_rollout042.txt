import time

def run(seed=42, budget_s=1100, **kwargs):

    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    
    # Initialize population
    pop_size = 200
    h_pop = np.tile(initial_h, (pop_size, 1))
    noise = np.random.normal(0, 0.15, (pop_size, n_points))
    h_pop = np.clip(h_pop + noise, 0, 1)
    h_pop = h_pop * target_sum / np.sum(h_pop, axis=1, keepdims=True)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_pop[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([evaluate(h) for h in h_pop])
        # Tournament selection
        tournament_size = 5
        indices = np.random.randint(0, pop_size, (pop_size, tournament_size))
        winners = np.argmin(np.take_along_axis(fitness[indices], indices, axis=1), axis=1)
        parents = np.take_along_axis(h_pop, indices[:, winners], axis=0)
        # Crossover
        crossover_rate = 0.8
        children = []
        for i in range(pop_size // 2):
            p1, p2 = parents[2*i], parents[2*i+1]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
            else:
                c1, c2 = p1, p2
            children.extend([c1, c2])
        # Mutation
        mutation_rate = 0.1
        for i in range(len(children)):
            h = children[i]
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
            children[i] = h
        # Replace worst individuals
        combined = np.vstack([h_pop, children])
        fitness_combined = np.array([evaluate(h) for h in combined])
        sort_idx = np.argsort(fitness_combined)
        h_pop = combined[sort_idx[:pop_size]]
        # Update best
        current_best = h_pop[0]
        if evaluate(current_best) < best_c5:
            best_h, best_c5 = current_best, evaluate(current_best)
