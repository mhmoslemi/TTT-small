import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Default initial pattern: alternating 0 and 1
        initial_h = np.zeros(n_points, dtype=np.float64)
        initial_h[::2] = 1.0
        initial_h[1::2] = 0.0
        initial_h = np.clip(initial_h, 0.0, 1.0)
    
    def objective(h):
        if np.sum(h) == 0:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Initial population with perturbations
    def generate_population(pop_size):
        return [initial_h + np.random.normal(0, 0.2, n_points) for _ in range(pop_size)]
    
    # Differential Evolution
    def de_optimize():
        result = differential_evolution(
            fun=objective,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            popsize=30,
            maxiter=500,
            tol=1e-7,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            disp=False
        )
        return result.x
    
    # SLSQP refinement
    def slsqp_refine(h):
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return result.x
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Run DE
    de_start = time.time()
    while time.time() - de_start < budget_s * 0.6:
        try:
            de_result = de_optimize()
            if objective(de_result) < best_c5:
                best_c5 = objective(de_result)
                best_h = de_result.copy()
        except:
            break
    
    # Run SLSQP
    slsqp_start = time.time()
    while time.time() - slsqp_start < budget_s * 0.4:
        try:
            slsqp_result = slsqp_refine(best_h)
            if objective(slsqp_result) < best_c5:
                best_c5 = objective(slsqp_result)
                best_h = slsqp_result.copy()
        except:
            break
    
    return best_h, best_c5, n_points
