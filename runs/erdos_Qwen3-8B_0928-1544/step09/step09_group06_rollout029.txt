import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    n_particles = 100
    max_iterations = 600
    inertia_weight = 0.9
    cognitive_coeff = 2.0
    social_coeff = 2.0
    vel_limit = 0.1
    
    particles = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    velocities = [np.random.uniform(-vel_limit, vel_limit, n_points) for _ in range(n_particles)]
    
    personal_best = [p.copy() for p in particles]
    personal_best_values = [objective(p) for p in particles]
    global_best = initial_h.copy()
    global_best_value = objective(initial_h)
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = (inertia_weight * velocities[i] + 
                   cognitive_coeff * r1 * (personal_best[i] - particles[i]) + 
                   social_coeff * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(vel, -vel_limit, vel_limit)
            particles[i] = particles[i] + velocities[i]
            
            h = np.clip(particles[i], 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * target_sum / h_sum
            else:
                h = np.full(n_points, target_sum / n_points)
            particles[i] = h
            
            curr_val = objective(particles[i])
            if curr_val < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = curr_val
                if curr_val < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = curr_val
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(fun=objective, x0=global_best, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
    final_h = result.x if result.success else global_best
    final_c5 = result.fun if result.success else global_best_value
    
    return final_h, final_c5, n_points
