        total = np.sum(h)

            return np.full(n_points, 1.0 / n_points) * ones_count
        return h * (ones_count / total)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def get_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_k = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - max_k) % n_points
            grad[i] += (1 - h[j]) * 2.0 / n_points
            j = (i + max_k) % n_points
            grad[i] -= h[j] * 2.0 / n_points
        return grad
    
    h = clip_and_scale(initial_h.copy())
    best_h = h.copy()
