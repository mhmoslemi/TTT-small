                particles[i] = particles[i] * (target_sum / sum_p)
        
        for i in range(n_particles):
            current_c5 = compute_C5(particles[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = np.copy(particles[i])
                pbest_values[i] = current_c5
                if current_c5 < gbest_value:
                    gbest = np.copy(particles[i])
                    gbest_value = current_c5
        
        if np.min(pbest_values) == gbest_value:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                mutated_indices = np.random.choice(n_particles, size=int(n_particles * 0.2), replace=False)
                for idx in mutated_indices:
                    h = np.random.rand(n_points)
                    h = h * (target_sum / np.sum(h))
                    particles[idx] = np.clip(h, 0.0, 1.0)
                    particles[idx] = particles[idx] * (target_sum / np.sum(particles[idx]))
                stagnation_count = 0
        else:
            stagnation_count = 0
        
        omega = omega_start - (iteration * (omega_start - omega_end) / 1000)
    
    return gbest, gbest_value, n_points
