import time

        population.append(initial_h)

            h = stats.dirichlet([1.0]*n_points).rvs(1)[0]

            population.append(h)

        for _ in range(population_size):
            h = stats.dirichlet([1.0]*n_points).rvs(1)[0]
            h *= target_sum
            population.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Evaluate initial population
    fitness = [compute_C5(h) for h in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]

    start_time = time.time()
    
    for gen in range(max_generations):

        # Tournament selection
        parents = []

            tournament = np.random.choice(population, tournament_size)
            fits = [compute_C5(h) for h in tournament]

        # Crossover
        offspring = []
        for i in range(0, population_size, 2):

                p1, p2 = parents[i], parents[i+1]

                child1 = np.concatenate((p1[:split], p2[split:]))
