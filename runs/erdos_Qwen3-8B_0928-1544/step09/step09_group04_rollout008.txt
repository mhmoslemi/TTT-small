from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    initial_h_sum = np.sum(initial_h)
    init_h = initial_h * target_sum / initial_h_sum

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points

    best_h = init_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial local refinement
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False}
    )
    final_h = result.x
    final_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_sum
    best_c5 = result.fun
    best_h = final_h

        perturb = np.random.normal(0.0, 0.1, n_points)
