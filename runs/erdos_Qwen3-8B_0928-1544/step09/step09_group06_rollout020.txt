from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70

    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')

    c1, c2 = 2.0, 2.0
