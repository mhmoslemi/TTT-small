import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2.0
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        a = initial_h - 0.5
        a = a - a.mean()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        a = h - 0.5
        a = a - a.mean()
    num_particles = 30
    particles = [a.copy() for _ in range(num_particles)]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(num_particles)]
    best_fitness = float('inf')
    best_a = a.copy()
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time >= budget_s:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            r_other = np.random.randint(0, num_particles)
            velocity = velocities[i] + 0.7 * (2 * r1 * (particles[i] - particles[r_other]) + 2 * r2 * (np.random.choice(particles, 1)[0] - particles[i]))
            velocities[i] = velocity
            new_a = particles[i] + velocity
            clipped = np.clip(new_a, -0.5, 0.5)
            S = np.sum(clipped)
            delta = (-S) / n_points
            adjusted = clipped + delta
            adjusted = np.clip(adjusted, -0.5, 0.5)
            h_current = 0.5 + adjusted
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_fitness:
                best_fitness = c5
                best_a = adjusted.copy()
            particles[i] = adjusted.copy()
    def objective(x):
        h = 0.5 + x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x)}]
    res = minimize(objective, best_a, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
    if res.success:
        best_a = res.x
    best_h = 0.5 + best_a
    return best_h, res.fun, n_points
