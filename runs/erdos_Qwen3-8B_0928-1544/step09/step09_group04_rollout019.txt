import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    pop_size = 200

    population = []
    for _ in range(pop_size):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum) if h_sum > 1e-10 else np.zeros_like(h)
        population.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    fitness = [evaluate(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        parents = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(range(pop_size), size=5, replace=False)
            best = idxs[np.argmin(fitness[idxs])]
            parents.append(population[best])
        children = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                children.extend([child1, child2])
        mutated = []
        for child in children:
            noise = np.random.normal(0, 0.1, n_points)
            m = np.clip(child + noise, 0.0, 1.0)
            m_sum = np.sum(m)
            m = m * (target_sum / m_sum) if m_sum > 1e-10 else np.zeros_like(m)
            mutated.append(m)
        all_candidates = population + mutated
        all_fit = [evaluate(h) for h in all_candidates]
        indices = np.argsort(all_fit)
        population = [all_candidates[i] for i in indices[:pop_size]]
        curr_fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(curr_fitness)
        if curr_fitness[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx], curr_fitness[best_idx]

    return best_h, best_c5, n_points
