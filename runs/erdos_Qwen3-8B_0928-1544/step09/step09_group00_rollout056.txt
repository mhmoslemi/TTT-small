from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2
    if 'initial_h_values' in globals():

        population[i] = h.copy()

            h = np.zeros_like(h)

    while time.time() - start_time < budget_s:

            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)

                    trial[j] = mutant[j]

                    trial[j] = population[_][j]
            trial = np.clip(trial, 0, 1)

    def loc_obj(x):
        return objective(x)

    res = minimize(loc_obj, best_h, bounds=bounds, constraints=cons,
