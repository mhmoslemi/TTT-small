        return np.sum(h) - target_sum_high
    
    bounds = [(0.0, 1.0)] * n_high
    
    pop_size = 100
    pop = [initial_h + np.random.normal(0, 0.1, n_high) for _ in range(pop_size)]
    for i in range(pop_size):
        current_sum = np.sum(pop[i])
        if current_sum != 0:
            pop[i] = pop[i] * (target_sum_high / current_sum)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    F = 0.8
    CR = 0.9
    max_generations = 500
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_high):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:
                trial = trial * (target_sum_high / trial_sum)
            trial_c5 = objective(trial)
            if trial_c5 < objective(pop[i]):
                pop[i] = trial
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_high
