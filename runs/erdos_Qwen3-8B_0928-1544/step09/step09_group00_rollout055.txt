import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_a = initial_h - 0.5
    initial_a = np.clip(initial_a, -0.5, 0.5)
    sum_initial = np.sum(initial_a)
    delta = -sum_initial / n_points
    initial_a += delta
    initial_a = np.clip(initial_a, -0.5, 0.5)
    
    n_particles = 30
    particles = [initial_a.copy() for _ in range(n_particles)]
    velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
    
    def evaluate(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        return np.max(corr)
    
    pbest_pos = [a.copy() for a in particles]
    pbest_val = [evaluate(a) for a in particles]
    gbest_pos = particles[0].copy()
    gbest_val = pbest_val[0]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 2.0 * r1 * (pbest_pos[i] - particles[i]) + 2.0 * r2 * (gbest_pos - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            sum_p = np.sum(particles[i])
            delta_p = -sum_p / n_points
            particles[i] += delta_p
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            val = evaluate(particles[i])
            if val < pbest_val[i]:
                pbest_val[i] = val
                pbest_pos[i] = particles[i].copy()
        gbest_val = min(gbest_val, min(pbest_val))
        gbest_pos = pbest_pos[pbest_val.index(gbest_val)]
    
    best_h = 0.5 + gbest_pos
    best_c5 = gbest_val
    return best_h, best_c5, n_points
