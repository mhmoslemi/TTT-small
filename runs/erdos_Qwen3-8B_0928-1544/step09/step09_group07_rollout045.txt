import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    def generate_initial_population():
        max_pop_size = 100
        pop = [initial_h.copy()]
        for _ in range(max_pop_size - 1):
            # Create shifted and mutated versions
            shifted = np.roll(initial_h, np.random.randint(0, n_points))
            mutated = shifted + np.random.normal(0, 0.05, n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated_sum = np.sum(mutated)
            if mutated_sum != 0:
                mutated = mutated * (target_sum / mutated_sum)
            pop.append(mutated)
        return pop
    
    pop = generate_initial_population()
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            mutant = pop[a] + 0.7 * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.8:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:
                trial = trial * (target_sum / trial_sum)
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial.copy()
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
