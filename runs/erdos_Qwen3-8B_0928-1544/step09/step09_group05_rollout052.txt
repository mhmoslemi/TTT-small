import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        s = target_sum / np.sum(h) if np.sum(h) != 0 else 1.0
        return h * s
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize PSO
    pop_size = 100
    pop = []
    for _ in range(pop_size // 2):
        h = initial_h + np.random.normal(0, 0.1, n_points)
        pop.append(repair(h))
    for _ in range(pop_size // 2):
        h = np.random.uniform(0, 1, n_points)
        pop.append(repair(h))
    pop = np.array(pop)
    vel = np.random.uniform(-0.05, 0.05, (pop_size, n_points))
    
    best_h = initial_h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            r1, r2 = np.random.rand(2)
            vel[i] += 2 * r1 * (pop[a] - pop[i]) + 2 * r2 * (pop[b] - pop[c])
            pop[i] += vel[i]
            pop[i] = repair(pop[i])
            curr_c5 = obj(pop[i])
            if curr_c5 < obj(pop[i]):
                pop[i] = pop[i]
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = pop[i].copy()
        
        # Quasiperiodic mutation
        freq = 0.01
        amp = 0.05
        for i in range(pop_size):
            pop[i] += amp * np.sin(gen * freq + np.random.rand()) * np.random.choice([1, -1])
            pop[i] = repair(pop[i])
        
        if gen % 10 == 0:
            best_h = repair(best_h)
    
    # Local refinement
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
