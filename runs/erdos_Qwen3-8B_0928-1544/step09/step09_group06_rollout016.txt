import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def clip_and_normalize(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        return h_clipped * target_sum / sum_clipped
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Differential Evolution Setup
    pop_size = 50
    max_generations = 200
    F = 0.8
    CR = 0.6
    
    pop = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    pop = [clip_and_normalize(h) for h in pop]
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = clip_and_normalize(trial)
            trial_c5 = objective(trial)
            if trial_c5 < objective(pop[i]):
                pop[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    
    # Refine with SLSQP
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
    )
    
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
