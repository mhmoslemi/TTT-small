import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 30

        if i == 0:
            continue
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population[i] = h.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.7
    CR = 0.95
    generations_without_improvement = 0
    max_generations = 100
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
                generations_without_improvement = 0
            elif np.isclose(current_c5, best_c5, atol=1e-7):
                generations_without_improvement = 0
            else:
                generations_without_improvement += 1
            
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
        
        if generations_without_improvement >= max_generations:
            F = max(F - 0.01, 0.5)
            CR = max(CR + 0.02, 0.98)
            generations_without_improvement = 0
    
    if 'initial_h_values' in globals():
        h_opt = best_h.copy()
        h_opt = h_opt.astype(np.float64)
        n_vars = n_points
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :] = 1.0
        b_eq = np.array([ones_count], dtype=np.float64)
