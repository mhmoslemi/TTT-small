        h = 0.5 + a_vec
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    n_particles = 20
    particles = [a.copy() for _ in range(n_particles)]
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_obj = [obj(p) for p in pbest]
    gbest = particles[0].copy()
    gbest_obj = pbest_obj[0]

    start_time = time.time()
    w, c1, c2 = 0.7, 1.5, 1.5

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            particles[i] -= np.mean(particles[i])
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            current_obj = obj(particles[i])
            if current_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = current_obj
                if current_obj < gbest_obj:
                    gbest = particles[i].copy()
                    gbest_obj = current_obj

        top_indices = np.argsort(pbest_obj)[:5]
        for idx in top_indices:
            a_candidate = pbest[idx].copy()
            bounds = [(-0.5, 0.5) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda a: np.sum(a) - 0}
            result = minimize(obj, a_candidate, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
            if result.success and result.fun < pbest_obj[idx]:
                pbest[idx] = result.x
                pbest_obj[idx] = result.fun
                if result.fun < gbest_obj:
                    gbest = result.x
                    gbest_obj = result.fun

    best_h = 0.5 + gbest
    best_h = np.clip(best_h, 0, 1)
    return best_h, gbest_obj, n_points
