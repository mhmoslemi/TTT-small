from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 105
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Initialize with given or uniform
    init_h = initial_h.copy()
    init_h_sum = np.sum(init_h)
    if init_h_sum == 0:
        init_h = np.full(n_points, 1.0 / n_points)
    else:
        init_h = init_h * target_sum / init_h_sum
    
    def optimize_candidate(h_initial):
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={
                'maxiter': 300,
                'ftol': 1e-9,
                'eps': 1e-8,
                'disp': False
            }
        )
        final_h = result.x
        final_sum = np.sum(final_h)
        final_h = final_h * target_sum / final_sum
        return final_h, result.fun
    
    best_h, best_c5 = optimize_candidate(init_h)
    start_time = time.time()
    eval_time = 0
    
    # Adaptive perturbation with step decay
    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        sigma = 0.15 * (1 - _/15)
        perturb = np.random.normal(0.0, sigma, n_points)
