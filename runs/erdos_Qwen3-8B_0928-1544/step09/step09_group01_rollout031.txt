import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better resolution
    ones_count = n_points // 2
    population_size = 20
    max_iterations = int(budget_s / 50)  # Limit iterations based on time

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    # Parameters
    F = 0.8  # Mutation factor
    CR = 0.6  # Crossover probability
    elite_ratio = 0.2  # Retain best portion as elite

    for _ in range(max_iterations):
        # Create new population
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]

            # Mutate
            v = a_vec + F * (b_vec - c_vec)
            # Crossover
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            # Clip and scale
            u_sum = np.sum(u)
            u = np.clip(u, 0.0, 1.0)
            u_scaled = u * (ones_count / u_sum) if u_sum > 0 else np.zeros(n_points)

            # Evaluate
            current_c5 = objective(population[i])
            new_c5 = objective(u_scaled)
            if new_c5 < current_c5:
                new_population.append(u_scaled)
            else:
                new_population.append(population[i])
        # Elitism
        new_population.sort(key=objective)
        population = new_population[:int(population_size * elite_ratio)] + new_population[-int(population_size * (1 - elite_ratio)):]
        if objective(population[0]) < best_c5:
            best_c5 = objective(population[0])
            best_h = population[0].copy()
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points
