    n_points = 70
    ones_count = n_points // 2
    population_size = 50

    if 'initial_h_values' in globals():

        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])

    start_time = time.time()

    F = 0.8  # Mutation factor
    CR = 0.95  # Crossover probability
    generations = int((budget_s - 30) / 5)  # Allow 30s for setup

    for gen in range(generations):
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]

            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)

            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]

            # Clip to [0,1]
            u = np.clip(u, 0.0, 1.0)

            # Scale to meet sum constraint
            u_sum = np.sum(u)
            if u_sum == 0:
                continue  # skip if sum is zero (unlikely)

                if new_c5 < best_c5:

                    best_h = u_scaled.copy()
