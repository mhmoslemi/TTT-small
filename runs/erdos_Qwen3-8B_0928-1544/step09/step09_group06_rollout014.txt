        max_val = np.max(corr)
        m = np.argmax(corr)
        k = m - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            idx1 = (i + k) % n_points
            idx2 = (i - k) % n_points
            grad_i = (1 - h[idx1]) - h[idx2]
            gradient[i] = grad_i * dx
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        if no_improve_count > 200:
            learning_rate *= 0.5
            no_improve_count = 0
        if np.random.random() < 0.01:
            h += 1e-3 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(fun=compute_c5, x0=best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
