        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0]
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    mutation_rate = 0.2
    crossover_rate = 0.8
    elite_size = int(pop_size * 0.1)
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_C5(h) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx]
            best_c5 = fitness[current_best_idx]
        
        # Select parents via tournament
        parents = []
        for _ in range(pop_size):
            tournament = np.random.choice(population, size=5)
            selected = tournament[np.argmin([compute_C5(h) for h in tournament])]
            parents.append(selected)
        
        # Generate offspring
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = parents[i]
            parent2 = parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            # Mutate
            mutation = np.random.normal(0, 0.05, n_points)
            child1 += mutation
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (ones_count / np.sum(child1))
            mutation = np.random.normal(0, 0.05, n_points)
            child2 += mutation
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (ones_count / np.sum(child2))
            offspring.append(child1)
            offspring.append(child2)
        
        population = offspring
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        population = elite + [np.random.rand(n_points) * (ones_count / np.sum(np.random.rand(n_points))) for _ in range(pop_size - elite_size)]
    
    return best_h, best_c5, n_points
