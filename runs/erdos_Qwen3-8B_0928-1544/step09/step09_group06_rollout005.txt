from scipy.optimize import minimize
import time

        return np.max(corr) * dx

    n_particles = 100
    max_iter = 600
    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    vel_limits = (0.0, 0.2)

    particles = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    velocities = np.random.uniform(*vel_limits, size=(n_particles, n_points))
    pbest = particles.copy()
    pbest_obj = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_obj)]
    gbest_obj = np.min(pbest_obj)
    
    start_time = time.time()
