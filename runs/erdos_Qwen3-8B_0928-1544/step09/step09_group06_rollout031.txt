        return gradient
    
    # Main loop
    learning_rate = 1e-3
    best_improvement = 0

        h -= learning_rate * gradient

        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        
        # Update best solution
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:

            best_h = h.copy()

            best_improvement += 1
        
        if best_improvement >= 50:
            learning_rate *= 0.5
            best_improvement = 0

        if iteration % 100 == 0:
            noise = 1e-3 * np.random.randn(n_points)
