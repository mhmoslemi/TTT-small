        pop[i] = h * target_sum / h_sum
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial.copy()
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
                      options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-9})
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    return final_h, final_c5, n_points
