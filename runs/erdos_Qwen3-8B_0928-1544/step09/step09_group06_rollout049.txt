from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    def objective(h):

        return np.max(corr) * dx

        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Initial simplex
    def simplex_initial_guess(h):
        return np.clip(h + np.random.normal(0, 0.05, n_points), 0.0, 1.0) * target_sum / np.sum(np.clip(h + np.random.normal(0, 0.05, n_points), 0.0, 1.0))
    
    # Optimization
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Initial simplex-based optimization
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        simplex = [simplex_initial_guess(best_h) for _ in range(15)]
        result = minimize(objective, simplex[0], method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9})
