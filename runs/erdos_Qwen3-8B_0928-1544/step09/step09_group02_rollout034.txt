import time

def run(seed=42, budget_s=1100, **kwargs):

    pop_size = 50

            h = initial_h.copy()

            h += noise
            h = np.clip(h, 0, 1)
            h = h * (ones_count / np.sum(h))
        else:
            h = np.random.rand(n_points)
            h = h * (ones_count / np.sum(h))
        population.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = np.copy(population[0])
    best_value = compute_C5(best_h)
    
    def tournament_selection():
        indices = np.random.choice(len(population), 5, replace=False)
        candidates = [population[i] for i in indices]
        return min(candidates, key=compute_C5)
    
    mutation_rate = 0.2
    crossover_rate = 0.8
    gen = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection()
            parent2 = tournament_selection()

                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                child[mutate_idx] += np.random.normal(0, 0.1, len(mutate_idx))
                child = np.clip(child, 0, 1)
                child = child * (ones_count / np.sum(child))
            
            new_population.append(child)
        
        for i in range(len(new_population)):
