import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Ensure initial_h meets the sum constraint
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) != 0 else initial_h
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Parameters for DE
    pop_size = 100
    F = 0.8
    CR = 0.9
    max_gen = 800
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # DE optimization
    pop = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        pop[i] = np.clip(pop[i], 0.0, 1.0)
        pop[i] = pop[i] * target_sum / np.sum(pop[i])
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:
                trial = trial * (target_sum / trial_sum)
            trial = np.clip(trial, 0.0, 1.0)
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    # SLSQP refinement
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
