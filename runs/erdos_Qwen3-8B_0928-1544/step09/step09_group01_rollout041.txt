def run(seed=42, budget_s=1100, **kwargs):

    F = 0.7

        for _ in range(population_size - 10):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum != 0:
                h_scaled = h * (ones_count / h_sum)
                population.append(h_scaled)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum != 0:
                h_scaled = h * (ones_count / h_sum)
                population.append(h_scaled)
    
    # Evaluate initial best
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    gen = 0
    
    while time.time() - start_time < budget_s:
