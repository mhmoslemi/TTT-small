            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(fun=objective, x0=global_best, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
    final_h = result.x if result.success else global_best
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
