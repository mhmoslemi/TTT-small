import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (ones_count / np.sum(h_initial))
    a_initial = h_initial - 0.5

    pop_size = 20
    particles = []
    velocities = []
    for _ in range(pop_size):
        a = a_initial + np.random.uniform(-0.2, 0.2, n_points)
        a = np.clip(a, -0.5, 0.5)
        s = np.sum(a)
        a[-1] += (-s)
        a[-1] = np.clip(a[-1], -0.5, 0.5)
        particles.append(a)
        velocities.append(np.zeros(n_points))

    def compute_objective(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_a = a_initial.copy()
    best_c5 = compute_objective(best_a)
    start_time = time.time()

    w = 0.7
    c1 = 2.0
    c2 = 2.0

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            vel = velocities[i] + c1 * r1 * (particles[i] - best_a) + c2 * r2 * (best_a - particles[i])
            velocities[i] = w * velocities[i] + vel
            a_new = particles[i] + vel
            a_new = np.clip(a_new, -0.5, 0.5)
            s = np.sum(a_new)

                best_a = a_new.copy()
            particles[i] = a_new.copy()
            velocities[i] = vel.copy()

    best_h = 0.5 + best_a
    return best_h, best_c5, n_points
