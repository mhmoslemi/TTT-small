import time

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2

    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = [np.copy(p) for p in particles]
    pbest_values = [compute_C5(p) for p in pbest]
    gbest = np.copy(particles[np.argmin(pbest_values)])
    gbest_value = np.min(pbest_values)
    
    omega = 0.8
    c1, c2 = 1.5, 1.5
    max_stagnation = 50
    stagnation_count = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])

            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
        
        for i in range(n_particles):
            current_c5 = compute_C5(particles[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = np.copy(particles[i])
                pbest_values[i] = current_c5
                if current_c5 < gbest_value:
                    gbest = np.copy(particles[i])
                    gbest_value = current_c5
        
        if np.min(pbest_values) == gbest_value:
