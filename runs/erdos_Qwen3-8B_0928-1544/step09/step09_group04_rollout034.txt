import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    pop_size = 200
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = []
    for _ in range(pop_size):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = float('inf')

    for _ in range(generations):
        # Sort by fitness
        population.sort(key=evaluate)
        # Select parents
        parents = population[:pop_size // 2]
        # Crossover
        new_population = []
        for i in range(0, len(parents), 2):
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((parents[i][:split], parents[i+1][split:]))
                child2 = np.concatenate((parents[i+1][:split], parents[i][split:]))
                new_population.extend([child1, child2])
            else:
                new_population.extend([parents[i].copy(), parents[i+1].copy()])
        # Mutation
        for j in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                new_population[j] += noise
                new_population[j] = np.clip(new_population[j], 0, 1)
                new_population[j] = new_population[j] * target_sum / np.sum(new_population[j])
        # Replace worst
        population = population[:pop_size // 2] + new_population[:pop_size // 2]
        # Update best
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
