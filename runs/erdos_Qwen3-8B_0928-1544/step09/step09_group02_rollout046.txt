import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    
    T = 0.1 * current_C5
    T_min = 1e-5
    cooling_factor = 0.998
    steps_per_temp = 100
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for _ in range(steps_per_temp):
            i, j = np.random.randint(0, n_points, size=2)
            if i == j:
                continue
            epsilon = np.random.rand() * 0.1
            delta = epsilon * min(h[i], 1 - h[j])
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = np.clip(h_new, 0.0, 1.0)
            
            new_C5 = compute_C5(h_new)
            
            if new_C5 < current_C5 or np.random.rand() < np.exp((current_C5 - new_C5) / T):
                h[:] = h_new[:]
                current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_h[:] = h[:]
                best_C5 = current_C5
        
        if T < T_min:
            break
        T *= cooling_factor
    
    return best_h, best_C5, n_points
