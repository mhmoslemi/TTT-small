                    if x_sum == 0:
                        return float('inf')
                    x_scaled = x * (ones_count / x_sum)
                    return objective(x_scaled)
                res = opt.minimize(obj, h, method='Nelder-Mead', options={'xatol': 1e-6})
                new_h = res.x
                new_sum = np.sum(new_h)
                if new_sum == 0:
                    continue
                new_scaled = new_h * (ones_count / new_sum)
                new_c5 = objective(new_scaled)
                for i, h_i in enumerate(population):
                    if new_c5 < objective(h_i):
                        population[i] = new_scaled
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = new_scaled.copy()
    
    return best_h, best_c5, n_points
