import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_low = 35
    n_high = 70
    dx_low = 2.0 / n_low
    dx_high = 2.0 / n_high
    target_sum_low = n_low / 2
    target_sum_high = n_high / 2

    if 'initial_h_values' in globals():
        h_low_initial = initial_h_values[:n_low] * (target_sum_low / np.sum(initial_h_values[:n_low]))
    else:
        h_low_initial = np.full(n_low, 0.5)

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_low(h_initial):
        max_pop = 50
        F, CR = 0.6, 0.7
        pop = [np.clip(h_initial + np.random.normal(0, 0.05, n_low), 0, 1) for _ in range(max_pop)]
        for i in range(len(pop)):
            h = pop[i]
            h_sum = np.sum(h)
            pop[i] = h * target_sum_low / h_sum

        best_h, best_c5 = h_initial.copy(), compute_c5(h_initial, dx_low)
        start_time = time.time()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            for i in range(len(pop)):
                a, b, c = np.random.choice(len(pop), 3, replace=False)
                mutant = pop[a] + F * (pop[a] - pop[b])
                trial = pop[i].copy()
                for j in range(n_low):
                    if np.random.rand() < CR:
                        trial[j] = mutant[j]
                trial_sum = np.sum(trial)
                if trial_sum:
                    trial = trial * target_sum_low / trial_sum
                trial_c5 = compute_c5(trial, dx_low)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    pop[i] = trial
        return best_h, best_c5

    best_h_low, best_c5_low = optimize_low(h_low_initial)
    h_high_initial = np.repeat(best_h_low, 2)

    def optimize_high(h_initial):
        def obj(h):
            return compute_c5(h, dx_high)
        def cons(h):
            return np.sum(h) - target_sum_high
        bounds = [(0., 1.)] * n_high
        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'maxiter': 200})
        return res.x if res.success else h_initial, res.fun if res.success else compute_c5(h_initial, dx_high)

    best_h_high, best_c5_high = optimize_high(h_high_initial)
    return best_h_high, best_c5_high, n_high
