        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    a = h - 0.5
    a -= np.mean(a)
    a = np.clip(a, -0.5, 0.5)
    h = 0.5 + a
    m = n_points // 2
    left_a = a[:m]
    right_a = a[m:]
    best_a = a.copy()
    best_c5 = evaluate_erdos_solution(a)
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        
        # Optimize left
        target_sum = -np.sum(right_a)
        bounds = [(-0.5, 0.5) for _ in range(len(left_a))]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        def obj_left(x):
            full = np.concatenate([x, right_a])
            c = np.correlate(0.5 + full, 1 - (0.5 + full), mode='full') * dx
            return np.max(c)
        res = minimize(obj_left, left_a, bounds=bounds, constraints=cons)
        if res.success:
            left_a = res.x
            a = np.concatenate([left_a, right_a])
            a -= np.mean(a)
            a = np.clip(a, -0.5, 0.5)
            current_c5 = evaluate_erdos_solution(a)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = a.copy()
        
        # Optimize right
        target_sum = -np.sum(left_a)
        bounds = [(-0.5, 0.5) for _ in range(len(right_a))]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        def obj_right(x):
            full = np.concatenate([left_a, x])
            c = np.correlate(0.5 + full, 1 - (0.5 + full), mode='full') * dx
            return np.max(c)
        res = minimize(obj_right, right_a, bounds=bounds, constraints=cons)
        if res.success:
            right_a = res.x
            a = np.concatenate([left_a, right_a])
            a -= np.mean(a)
            a = np.clip(a, -0.5, 0.5)
            current_c5 = evaluate_erdos_solution(a)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = a.copy()
    
    best_h = 0.5 + best_a
    return best_h, best_c5, n_points
