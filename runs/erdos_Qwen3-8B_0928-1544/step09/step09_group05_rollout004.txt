import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100

    target_sum = n_points / 2

        return np.max(corr) * dx

    population_size = 50
    particles = [initial_h.copy() for _ in range(population_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]

    global_best = particles[0].copy()
    global_best_fitness = obj(global_best)

    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    mutation_amp = 0.2
    mutation_freq = 0.5
    mutation_phase = np.random.rand(population_size) * 2 * np.pi
    mutation_prob = 0.1
    max_iterations = 500
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (particles[i] - particles[i]) + social * r2 * (global_best - particles[i])
            
            # Quasiperiodic mutation
            angle = iteration * mutation_freq + mutation_phase[i]
            mutation = mutation_amp * np.sin(angle)
            velocities[i] += mutation
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and scale
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            current_sum = np.sum(particles[i])
            if current_sum != 0:

            # Evaluate and update
            fitness = obj(particles[i])
            if fitness < obj(global_best):
                global_best = particles[i].copy()
                global_best_fitness = fitness
    
    # Local refinement
