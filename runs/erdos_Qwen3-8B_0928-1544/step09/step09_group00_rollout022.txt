    velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_points))
    pbest = [p.copy() for p in particles]
    gbest = a.copy()
    best_c5 = compute_c5(gbest)
    
    inertia = 0.8
    cognitive = 2.0
    social = 2.0
    max_iter = int((budget_s * 0.8) // 1)
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            s = np.sum(particles[i])
            if s != 0:
                delta = -s
                particles[i][-1] += delta
                particles[i][-1] = np.clip(particles[i][-1], -0.5, 0.5)
            
            c5 = compute_c5(particles[i])
            if c5 < compute_c5(pbest[i]):
                pbest[i] = particles[i].copy()
            if c5 < best_c5:
                best_c5 = c5
                gbest = particles[i].copy()
    
    def compute_k_max(a):
        h = a + 0.5
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        return max_corr_idx
    
    k_max = compute_k_max(gbest)
    
    def convex_objective(a_flat):
        a = a_flat.reshape(n_points)
        overlap = 0.0
        for j in range(n_points):
            idx = (j + k_max) % n_points
            overlap += (0.5 + a[j]) * (0.5 - a[idx])
        return overlap
    
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x)})
    
    result = minimize(convex_objective, gbest, method='SLSQP', bounds=bounds, constraints=cons)
    best_a = result.x
    
    best_h = best_a + 0.5
    final_c5 = compute_c5(best_a)
    
    return best_h, final_c5, n_points
