            term1 = 1.0 - h[(j + k) % n_points]
            term2 = h[(j - k) % n_points]
            grad[j] = (term1 - term2) * dx
        return grad

    def backtrack_line_search(h, grad, compute_C5, max_backtracks=20, beta=0.5, alpha_init=0.1):
        current_value = compute_C5(h)
        alpha = alpha_init
        for _ in range(max_backtracks):
            trial_h = h - alpha * grad
            trial_h_clipped = np.clip(trial_h, 0.0, 1.0)
            sum_trial = np.sum(trial_h_clipped)
            if sum_trial == 0:
                return 0.0
            scale = ones_count / sum_trial
            trial_h_projected = trial_h_clipped * scale
            trial_value = compute_C5(trial_h_projected)
            if trial_value <= current_value - alpha * np.dot(grad, grad) * 0.5:
                return alpha
            alpha *= beta
        return alpha

    start_time = time.time()
    max_iter = 10000
    convergence_threshold = 1e-5
    best_h = h.copy()
    best_c5 = compute_C5(h)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        k = max_corr_index - (n_points - 1)
        grad = compute_gradient(h, k)
        alpha = backtrack_line_search(h, grad, compute_C5)
        trial_h = h - alpha * grad
        trial_h_clipped = np.clip(trial_h, 0.0, 1.0)
        sum_trial = np.sum(trial_h_clipped)
        scale = ones_count / sum_trial
        h = trial_h_clipped * scale
        current_c5 = compute_C5(h)
        if np.abs(current_c5 - best_c5) < convergence_threshold:
            break
        best_c5 = current_c5
        best_h = h.copy()

    return best_h, best_c5, n_points
