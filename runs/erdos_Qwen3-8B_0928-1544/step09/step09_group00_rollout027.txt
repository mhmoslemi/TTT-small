import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    a = initial_h - 0.5
    a = a - np.mean(a)
    
    def compute_c5(a):
        h = a + 0.5
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        a_clipped = np.clip(x, -0.5, 0.5)
        a_normalized = a_clipped - np.mean(a_clipped)
        return compute_c5(a_normalized)
    
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.mean(x)})
    
    best_a = a.copy()
    best_c5 = compute_c5(best_a)
    
    start_time = time.time()
    max_iter = 2000
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        rand_idx = np.random.choice(range(n_points), 3, replace=False)
        mutant = best_a + np.random.rand(n_points) * (best_a - best_a)
        mutant = np.clip(mutant, -0.5, 0.5)
        mutant = mutant - np.mean(mutant)
        current_c5 = compute_c5(mutant)
        if current_c5 < best_c5:
            best_a = mutant.copy()
            best_c5 = current_c5
    
    result = minimize(objective, best_a, method='L-BFGS-B', bounds=bounds, tol=1e-6)
    best_a = result.x
    best_a = best_a - np.mean(best_a)
    best_c5 = compute_c5(best_a)
    
    h_values = best_a + 0.5
    return h_values, best_c5, n_points
