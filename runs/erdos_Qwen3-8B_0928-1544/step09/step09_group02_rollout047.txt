            new_c5 = compute_C5(h_new)
            if new_c5 < current_c5:
                h = h_new.copy()
                current_c5 = new_c5
            else:
                prob = np.exp((new_c5 - current_c5) / T)
                if np.random.rand() < prob:
                    h = h_new.copy()
                    current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        T *= cooling_factor
        iteration_count += iterations_per_temp
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
