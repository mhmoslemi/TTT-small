                a, b = population[i], population[i + 1]
                child = crossover(a, b)
                new_population.append(child)
        combined = population + new_population
        combined_c5 = current_c5 + [evaluate(p) for p in new_population]
        combined = sorted(zip(combined, combined_c5), key=lambda x: x[1])
        population = [c[0] for c in combined[:pop_size]]
        current_c5 = [c[1] for c in combined[:pop_size]]
        mutation_strength *= mutation_decay
    return best_h, best_c5, n_points
