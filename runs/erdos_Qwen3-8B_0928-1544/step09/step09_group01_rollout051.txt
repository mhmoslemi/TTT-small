        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        while h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
        h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    alpha = 0.1
    max_epochs = 10000
    patience = 100
    no_improvement = 0

    for _ in range(max_epochs):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        s_max = np.argmax(corr)
        k = s_max - (n_points - 1)

        grad = np.zeros(n_points)
        for j in range(n_points):
            i_plus_k = (j + k) % n_points
            i_minus_k = (j - k) % n_points
            grad[j] = (1 - h[i_plus_k]) - h[i_minus_k]

        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h = h * (ones_count / h_sum)

        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= patience:
                alpha *= 0.5
                no_improvement = 0

    return best_h, best_c5, n_points
