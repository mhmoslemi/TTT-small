def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    left = initial_h[:35]
    right = initial_h[35:]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def optimize_left(right, sum_left):
        c = [1.0 - right[i + 35] for i in range(35)]
        A_eq = np.array([[1.0]*35])
        b_eq = [sum_left]
        bounds = [(0, 1)]*35
        result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        return result.x if result.success else left
    
    def optimize_right(left, sum_right):
        c = [1.0 - left[i - 35] for i in range(35, 70)]
        A_eq = np.array([[1.0]*35])
        b_eq = [sum_right]
        bounds = [(0, 1)]*35
        result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        return result.x if result.success else right
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        sum_left = 1.0 - np.sum(right)
        left = optimize_left(right, sum_left)
        sum_right = 1.0 - np.sum(left)
        right = optimize_right(left, sum_right)
        h = np.concatenate([left, right])
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
