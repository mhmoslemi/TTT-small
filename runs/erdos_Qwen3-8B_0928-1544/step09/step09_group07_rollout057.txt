import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_low = 35
    n_high = 70
    dx_low = 2.0 / n_low
    dx_high = 2.0 / n_high
    target_sum_high = n_high / 2

    # Initialize with coarse grid or given initial_h_values
    if 'initial_h_values' in globals():
        h_high_initial = np.array(initial_h_values)
        # Downsample to low-res for initialization
        h_low_initial = np.repeat(h_high_initial[:n_low], 2)[:n_low]
    else:
        # Binary initial guess for low-res
        n_ones_low = n_low // 2
        h_low_initial = np.zeros(n_low)
        h_low_initial[:n_ones_low] = 1.0
        h_low_initial[np.random.choice(n_low, size=n_low - n_ones_low, replace=False)] = 0.0
        np.random.shuffle(h_low_initial)

    # Objective for low-res grid
    def objective_low(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_low

    bounds_low = [(0.0, 1.0)] * n_low

    # Differential evolution on low-res
    start_time = time.time()
    result_low = differential_evolution(
        objective_low, bounds_low,
        strategy='best1bin', popsize=15, tol=1e-6, seed=seed,
        maxiter=100, polish=True, mutation=(0.6, 1.0), recombination=0.9
    )
    elapsed_time = time.time() - start_time

    best_h_low = result_low.x

    # Upsample to high-res
    h_high_initial = np.repeat(best_h_low, 2)[:n_high]
    sum_high = np.sum(h_high_initial)
    h_high_initial = h_high_initial * target_sum_high / sum_high

    # Objective for high-res grid
    def objective_high(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_high

    # Constraints for high-res
    def constraint_high(h):
        return np.sum(h) - target_sum_high

    bounds_high = [(0.0, 1.0)] * n_high

    # SLSQP refinement
    result_high = minimize(
        fun=objective_high, x0=h_high_initial,
        method='SLSQP', bounds=bounds_high,
        constraints={'type': 'eq', 'fun': constraint_high},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )

    best_h = result_high.x
    best_c5 = result_high.fun

    return best_h, best_c5, n_high
