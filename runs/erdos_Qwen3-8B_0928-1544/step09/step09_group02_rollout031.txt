import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    h = h * (ones_count / np.sum(h))

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5

    T = 1.0
    alpha = 0.995
    T_min = 1e-5
    start_time = time.time()

    while time.time() - start_time < budget_s:
        epsilon = max(0.01, 0.1 * T)
        i, j = np.random.randint(n_points), np.random.randint(n_points)
        while i == j:
            j = np.random.randint(n_points)
        delta = epsilon * min(h[i], 1 - h[j])
        h_new = h.copy()
        h_new[i] -= delta
