        cons_high = ({'type': 'eq', 'fun': lambda h: np.sum(h) - (n_high / 2)})
        bounds_high = [(0.0, 1.0)] * n_high
        res_high = minimize(obj_high, h_high, method='SLSQP', bounds=bounds_high, constraints=cons_high, options={'maxiter': 150})
        h_high_opt = res_high.x
        c5_high = obj_high(h_high_opt)
        
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h_high = h_high_opt

    return best_h_high, best_c5, n_high
