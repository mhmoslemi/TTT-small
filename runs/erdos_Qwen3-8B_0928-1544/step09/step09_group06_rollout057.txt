from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Initialize population with small noise
    pop_size = 50
    F = 0.6
    CR = 0.8
    pop = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        h = pop[i]
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        pop[i] = h * target_sum / h_sum
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Genetic algorithm phase
    max_generations = 300
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):

            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial.copy()
                if trial_obj < best_c5:
                    best_c5 = trial_obj
