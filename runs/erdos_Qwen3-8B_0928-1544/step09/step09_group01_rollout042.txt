            best_h = h_scaled.copy()
            patience = 0
            alpha *= 1.1
        else:
            patience += 1
            if patience > max_patience:
                alpha *= 0.5
                patience = 0
        
        h = h_scaled.copy()
    
    return best_h, best_c5, n_points
