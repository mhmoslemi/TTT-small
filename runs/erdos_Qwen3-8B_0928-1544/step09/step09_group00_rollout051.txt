                pbest[i] = pop[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gb = pop[i].copy()
        
        if best_c5 < float('inf'):
            def obj(x):
                x = np.clip(x, -0.5, 0.5)
                x = x - np.mean(x)
                x = np.clip(x, -0.5, 0.5)
                h = 0.5 + x
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * (2.0 / n_points)
            
            constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x)})
            bounds = [(-0.5, 0.5) for _ in range(n_points)]
            res = minimize(obj, gb, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
            if res.success:
                gb = res.x
                best_c5 = res.fun
    
    best_h = 0.5 + gb
    return best_h, best_c5, n_points
