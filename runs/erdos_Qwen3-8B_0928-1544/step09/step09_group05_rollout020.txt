from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        s = target_sum / np.sum(h) if np.sum(h) != 0 else 1.0
        return h * s
    
    class Particle:
        def __init__(self, pos):
            self.pos = pos.copy()
            self.vel = np.random.uniform(-0.05, 0.05, n_points)
            self.best_pos = pos.copy()
            self.best_obj = obj(pos)
    
    # PSO parameters
    pop_size = 100
    max_iter = 300
    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    amp = 0.4
    freq = 0.005
    decay = 0.999
    mutation_prob = 0.1
    particles = [Particle(repair(initial_h + np.random.normal(0, 0.1, n_points))) for _ in range(pop_size)]
    
    best_h = particles[0].pos
    best_c5 = obj(best_h)
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for p in particles:
            r1, r2 = np.random.rand(), np.random.rand()
            p.vel = inertia * p.vel + cognitive * r1 * (p.best_pos - p.pos) + social * r2 * (best_h - p.pos)
            if np.random.rand() < mutation_prob:
                p.pos += amp * np.sin(i * freq + np.random.rand() * 2*np.pi)
            p.pos = repair(p.pos)
            p.obj = obj(p.pos)
            if p.obj < p.best_obj:
                p.best_pos = p.pos.copy()
                p.best_obj = p.obj
        if np.min([p.best_obj for p in particles]) < best_c5:

    # Local refinement
