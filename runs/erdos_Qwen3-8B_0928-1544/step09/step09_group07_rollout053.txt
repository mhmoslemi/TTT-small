import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_high = 70
    dx = 2.0 / n_high
    target_sum = n_high / 2
    
    if 'initial_h_values' in globals():
        initial_high = initial_h_values.copy()
    else:
        pattern = np.tile([1, 1, 0, 0, 0], 7)
        high_res = np.repeat(pattern, 2)
        scale_factor = target_sum / np.sum(high_res)
        initial_high = high_res * scale_factor
        initial_high = np.clip(initial_high, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_high
    
    population_size = 100
    population = [initial_high + np.random.normal(0, 0.05, n_high) for _ in range(population_size)]
    for i in range(len(population)):
        population[i] = np.clip(population[i], 0, 1)
        population[i] *= target_sum / np.sum(population[i])
    
    best_h = initial_high.copy()
    best_c5 = objective(initial_high)
    start_time = time.time()
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + np.random.normal(0, 0.1, n_high)
            mutant = np.clip(mutant, 0, 1)
            trial = population[i].copy()
            for j in range(n_high):
                if np.random.rand() < 0.8:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum if trial_sum != 0 else trial
            trial_obj = objective(trial)
            if trial_obj < objective(population[i]):
                population[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_high
