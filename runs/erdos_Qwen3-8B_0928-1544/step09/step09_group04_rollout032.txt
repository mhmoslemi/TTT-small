    init_h = init_h * target_sum / init_h_sum

            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={
                'maxiter': 500,
                'ftol': 1e-8,

                'disp': False
            }

        final_sum = np.sum(final_h)
