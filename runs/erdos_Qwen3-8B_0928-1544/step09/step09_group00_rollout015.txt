    particles = [a.copy() for _ in range(num_particles)]
    velocities = np.random.rand(num_particles, n_points) * 0.1
    personal_best = [a.copy() for _ in range(num_particles)]
    personal_best_values = [np.inf] * num_particles
    global_best = a.copy()
    global_best_value = np.inf
    w, c1, c2 = 0.7, 2.0, 2.0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            sum_p = np.sum(particles[i])
            particles[i] -= sum_p / n_points
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            h = 0.5 + particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_c5
                if current_c5 < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_c5
    best_h = 0.5 + global_best
    return best_h, global_best_value, n_points
