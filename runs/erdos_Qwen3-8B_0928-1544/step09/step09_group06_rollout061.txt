    gbest_fitness = compute_c5(gbest)

    start_time = time.time()

    for gen in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (pbest[i] - particles[i]) +
                            social * r2 * (gbest - particles[i]))

            # Update position
            particles[i] += velocities[i]

            # Clip and scale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])

            # Compute fitness
            current_c5 = compute_c5(particles[i])

            # Update personal best
            if current_c5 < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_c5

            # Update global best
            if current_c5 < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = current_c5

    # Refinement with SLSQP
    def obj_func(h):
        return compute_c5(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = minimize(obj_func, x0=gbest, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})

    final_h = result.x if result.success else gbest
    final_c5 = result.fun if result.success else gbest_fitness

    return final_h, final_c5, n_points
