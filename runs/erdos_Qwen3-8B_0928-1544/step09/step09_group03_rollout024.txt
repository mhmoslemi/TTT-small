    start_time = time.time()

        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Find the max value and its index
        max_idx = np.argmax(corr)
        # Convert to lag k_max
        n = n_points
        lag = max_idx - (n -1)
        # Compute the gradient
        grad = np.zeros(n)
        for j in range(n):
            i_plus_k = (j + lag) % n
            i_minus_k = (j - lag) % n
            grad[j] = (1 - h[i_plus_k]) - h[i_minus_k]
        grad *= 2.0 / n  # multiply by dx
        
        # Update h
        h -= learning_rate * grad
        
        # Clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Normalize sum to n_points / 2
        h = h * (ones_count / np.sum(h))
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    # Evaluate the final h
    corr_final = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr_final) * (2.0 / n_points)
    
    return h, c5, n_points
