    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.999
    noise_scale = 0.05
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        # Compute correlation and find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        k_max_idx = np.argmax(corr)
        k_max = k_max_idx - (n_points - 1)  # convert to actual lag (negative to positive)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        if k_max >= 0:
            for j in range(n_points):
                if j + k_max < n_points:
                    grad[j] += (1 - h[j + k_max])
                if j - k_max >= 0:
                    grad[j] -= h[j - k_max]
        else:
            m = -k_max
            for j in range(n_points):
                if j + m < n_points:
                    grad[j] += (1 - h[j + m])
                if j - m >= 0:
                    grad[j] -= h[j - m]
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        
        # Apply noise
        if iter_count % 100 == 0:
            h += np.random.normal(0, noise_scale, n_points)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
        
        # Evaluate C5
        current_c5 = max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Decay learning rate
        learning_rate *= decay_rate
        iter_count += 1
    
    return best_h, best_c5, n_points
