import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    population_size = 50
    population = [initial_h.copy() if 'initial_h_values' in globals() else np.random.rand(n_points)]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * (ones_count / np.sum(np.random.rand(n_points)))
        population.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    fitness = np.array([compute_C5(h) for h in population])
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx]
    
    start_time = time.time()
    
    params = {
        'crossover_rate': 0.8,
        'mutation_rate': 0.1,
        'tournament_size': 5,
        'elitism_ratio': 0.2
    }
    
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, params['tournament_size'], replace=False)
            best_tour = tournament[np.argmin(fitness[tournament])]
            new_population.append(population[best_tour].copy())
        
        for i in range(population_size):
            if np.random.rand() < params['crossover_rate']:
                j = np.random.randint(population_size)
                parent1, parent2 = new_population[i], new_population[j]
                split = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = new_population[i].copy()
            
            if np.random.rand() < params['mutation_rate']:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * (ones_count / np.sum(child))
            
            new_population[i] = child
        
        new_fitness = np.array([compute_C5(h) for h in new_population])
        combined = np.vstack((population, new_population))
        combined_fitness = np.concatenate((fitness, new_fitness))
        
        sorted_indices = np.argsort(combined_fitness)
        top_k = int(params['elitism_ratio'] * population_size)
        population = combined[sorted_indices[:population_size], :]
        fitness = combined_fitness[sorted_indices[:population_size]]
        
        if np.min(fitness) < best_c5:
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
    
    return best_h, best_c5, n_points
