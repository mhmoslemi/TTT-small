import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h_high = initial_h_values.copy()
    else:
        h_high = np.full(n_points, 0.5)
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Feasibility check
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # DE optimization with structured pattern
    # Create structured initial pattern for low-res
    n_low = 35
    period = 5
    pattern = np.array([1.0, 1.0, 0.0, 0.0, 0.0])
    h_low = np.tile(pattern, n_low // len(pattern))
    h_low = h_low[:n_low]
    h_low_sum = np.sum(h_low)
    scale_factor = (n_low / 2) / h_low_sum
    h_low = h_low * scale_factor
    
    # Upsample to high-res
    h_high = np.repeat(h_low, 2)
    h_high_sum = np.sum(h_high)
    scale_factor = target_sum / h_high_sum
    h_high = h_high * scale_factor
    
    # DE optimization
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=50,
        tol=1e-10,
        mutation=(0.7, 1),
        recombination=0.9,
        maxiter=500,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # SLSQP optimization
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}

        bounds=bounds,
        constraints=constraints,

    )
    
    final_h = result_slsqp.x
    final_c5 = result_slsqp.fun
