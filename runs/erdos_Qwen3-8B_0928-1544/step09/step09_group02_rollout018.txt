import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2
    population_size = 50
    if 'initial_h_values' in globals():
        base_h = initial_h_values.copy()
        population = [np.clip(base_h + np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(population_size)]
    else:
        population = [np.random.rand(n_points) for _ in range(population_size)]
    
    def normalize(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if total == 0:
            return np.zeros_like(h_clip)
        factor = ones_count / total
        return h_clip * factor
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0]
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_C5(h) for h in population]
        current_best_idx = np.argmin(fitness)
        current_best = fitness[current_best_idx]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[current_best_idx]
        
        new_pop = []
        for _ in range(population_size):
            idxs = np.random.choice(population_size, 5, replace=False)
            best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            parent = population[best_idx]
            
            if np.random.rand() < 0.8:
                other_idx = np.random.choice(population_size, 1, replace=False)[0]
                other = population[other_idx]
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent[:split], other[split:]])
            else:
                child = parent.copy()
            
            if np.random.rand() < 0.2:
                mutate_idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                child[mutate_idx] += np.random.normal(0, 0.05, len(mutate_idx))
                child = normalize(child)
            
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_pop.append(child)
        
        population = new_pop
    
    return best_h, best_c5, n_points
