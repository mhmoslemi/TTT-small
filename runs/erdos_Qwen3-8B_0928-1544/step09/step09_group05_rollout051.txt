                delta = beta * (sorted_pop[j]-sorted_pop[i]) + alpha * np.random.normal(0, 0.1, n_points)
                new_h = sorted_pop[i] + delta
                new_h = clip_and_scale(new_h)
                new_c5 = evaluate_erdos_solution(new_h)
                if new_c5 < brightness[i]:
                    pop[i] = new_h
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
        alpha *= 0.99
        beta0 *= 0.99
        if time.time() - start_time > transition_time:
            n_points = 140
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            new_best_h = np.repeat(best_h, 2)
            new_best_h = clip_and_scale(new_best_h)
            pop = [new_best_h + np.random.normal(0, 0.1, n_points) for _ in range(pop_size)]
            pop = [clip_and_scale(h) for h in pop]
            alpha = 1.0
            beta0 = 0.05
            gamma = 0.5

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=lambda x: evaluate_erdos_solution(x),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5

    return final_h, final_c5, n_points
