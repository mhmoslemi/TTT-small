            CR = min(CR + 0.1, 0.95)
        elif best_c5 > prev_best + 1e-5:
            F = min(F + 0.1, 1.0)
            CR = max(CR - 0.1, 0.5)

    best_h_sum = np.sum(best_h)
    if best_h_sum != 0:
        best_h = best_h * (ones_count / best_h_sum)
    return best_h, best_c5, n_points
