import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    
    pop_size = 200
    population = np.random.normal(0.0, 0.15, (pop_size, n_points))
    population = initial_h + population
    population = np.clip(population, 0.0, 1.0)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(h) for h in population])
        tournament_size = 5
        parents = []
        for _ in range(pop_size // 2):
            contestants = np.random.choice(pop_size, tournament_size, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parents.append(population[winner])
        new_pop = []
        for i in range(0, pop_size // 2, 2):
            p1, p2 = parents[i], parents[i+1]
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((p1[:split], p2[split:]))
            child2 = np.concatenate((p2[:split], p1[split:]))
            new_pop.append(child1)
            new_pop.append(child2)
        for i in range(len(new_pop)):
            h = new_pop[i]
            noise = np.random.normal(0.0, 0.1, n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
            new_pop[i] = h
        population = np.array(new_pop)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx], fitness[best_idx]
    
    return best_h, best_c5, n_points
