import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Genetic algorithm parameters
    pop_size = 100
    max_gen = 500
    mut_rate = 0.05
    cross_rate = 0.8
    elite_frac = 0.1
    
    # Initialize population
    def create_population(base):
        pop = []
        for _ in range(pop_size):
            h = base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * target_sum / np.sum(h)
            pop.append(h)
        return pop
    
    pop = create_population(initial_h)
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Genetic algorithm loop
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [objective(ind) for ind in pop]
        elites = np.argsort(fitness)[:int(elite_frac * pop_size)]
        # Select parents
        parent_indices = np.random.choice(range(pop_size), size=3, replace=False)
        a, b, c = parent_indices
        # Mutation
        mutant = pop[a] + np.random.normal(0, mut_rate, n_points)
        mutant = np.clip(mutant, 0.0, 1.0)
        mutant = mutant * target_sum / np.sum(mutant)
        # Crossover
        trial = pop[np.random.choice(range(pop_size))]
        for i in range(n_points):
            if np.random.rand() < cross_rate:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0.0, 1.0)
        trial = trial * target_sum / np.sum(trial)
        trial_obj = objective(trial)
        if trial_obj < fitness[parent_indices[0]]:
            pop[parent_indices[0]] = trial
            if trial_obj < best_c5:
                best_c5 = trial_obj
                best_h = trial.copy()
    
    # Local optimization
    res = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = res.x if res.success else best_h
    final_c5 = res.fun if res.success else best_c5
    
    return final_h, final_c5, n_points
