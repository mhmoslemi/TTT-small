from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * dx
    
    pop_size = 100
    max_gen = 300
    F, CR = 0.6, 0.85
    pop = [np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1) * (target_sum / np.sum(np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1))) for _ in range(pop_size)]
    
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = np.clip(trial * (target_sum / trial_sum), 0, 1) if trial_sum != 0 else trial
            trial_obj = obj(trial)
            if trial_obj < obj(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    def constraint_sum(h):

    bounds = [(0.0, 1.0)] * n_points
