def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    M = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    population_size = 30
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = initial_h.copy()
        perturb = np.random.normal(0, 0.05, n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * (M / np.sum(h))
        population.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.8
    CR = 0.9
    generations = 0
    
    while True:
        if time.time() - start_time >= budget_s:
            break
        generations += 1
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (M / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
        
        if generations % 100 == 0 or generations == 1:
            h = best_h.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            max_k = np.argmax(corr)
            k = max_k
            def local_objective(x):
                return -np.sum(x * x[(k + np.arange(n_points)) % n_points])
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - M}]
            bounds = [(0, 1) for _ in range(n_points)]
            try:
                result = minimize(local_objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8})
                if result.success:
                    new_h = result.x
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
            except:
                pass
    
    return best_h, best_c5, n_points
