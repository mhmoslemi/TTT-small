            h = np.zeros(n_points)
        else:
            h = h * (ones_count / h_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Use differential evolution to minimize the objective
    bounds = [(min(0.5, val), max(1.0, val)) for val in np.zeros(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=1000,
        seed=seed,
        disp=False
    )
    
    best_h = 0.5 + result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (ones_count / np.sum(best_h))
    return best_h, result.fun, n_points
