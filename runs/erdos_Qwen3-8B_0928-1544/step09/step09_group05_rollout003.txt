            j_idx = np.random.randint(0, i)
            h_i = sorted_pop[i]
            h_j = sorted_pop[j_idx]
            distance = np.linalg.norm(h_i - h_j)
            delta = (h_j - h_i) * (1 - np.exp(-gamma * distance))
            new_h = h_i + beta * delta + alpha * np.random.normal(0, 0.1, n_points)
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum != 0:
                new_h = new_h * (target_sum / new_h_sum)
            else:
                new_h = np.zeros_like(new_h)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < sorted_brightness[i]:
                sorted_pop[i] = new_h
                sorted_brightness[i] = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        alpha *= 0.99
        beta *= 0.99
        gamma *= 0.99
        pop = sorted_pop
    
    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        return compute_c5(h, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=cons)
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return (final_h, final_c5, n_points)
