import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    pop_size = 100
    particles = []
    phases = []
    for _ in range(pop_size):
        if np.random.rand() < 0.5:
            h = initial_h + np.random.normal(0, 0.1, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= target_sum / sum_h
        particles.append(h)
        phases.append(np.random.uniform(0, 2 * np.pi))
    particles = np.array(particles)
    phases = np.array(phases)
    pbest = particles.copy()
    pbest_obj = np.array([obj(p) for p in particles])
    global_best = particles[np.argmin(pbest_obj)].copy()
    global_best_obj = np.min(pbest_obj)
    velocities = np.random.normal(0, 0.05, (pop_size, n_points))
    generation = 0
    start_time = time.time()
    max_gen = 1000
    
    while generation < max_gen and time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            # Update velocity
            vel = velocities[i] * 0.9 + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            # Update position
            particles[i] += vel
            # Apply quasiperiodic mutation
            amp = 0.5 * np.exp(-0.001 * generation)
            freq = 0.1
            mutation = amp * np.sin(generation * freq + phases[i])
            particles[i] += mutation
            # Repair
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                particles[i] *= target_sum / sum_h
            # Evaluate
            current_obj = obj(particles[i])
            if current_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = current_obj
            if current_obj < global_best_obj:
                global_best = particles[i].copy()
                global_best_obj = current_obj
        generation += 1
    
    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=global_best,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else global_best
    final_c5 = result.fun if result.success else global_best_obj
    
    return final_h, final_c5, n_points
