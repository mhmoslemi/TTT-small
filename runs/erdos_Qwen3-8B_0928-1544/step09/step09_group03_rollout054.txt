import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    population = [initial_h.copy()]
    for _ in range(19):
        noise = np.random.normal(0, 0.1, n_points)
        new_h = initial_h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (ones_count / np.sum(new_h))
        population.append(new_h)
    
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h
    best_c5 = fitness(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_pop = []
        for h in population:
            noise = np.random.normal(0, 0.1, n_points)
            mutant = h + noise
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (ones_count / np.sum(mutant))
            new_pop.append(mutant)
        
        top = sorted(population, key=fitness)[:5]
        for i in range(len(top)):
            for j in range(i+1, len(top)):
                if np.random.rand() < 0.5:
                    child = (top[i] + top[j]) / 2
                    child = child * (ones_count / np.sum(child))
                    new_pop.append(child)
        
        new_pop = sorted(new_pop, key=fitness)[:20]
        for h in new_pop:
            if fitness(h) < best_c5:
                best_c5 = fitness(h)
                best_h = h
        population = new_pop
    
    return best_h, best_c5, n_points
