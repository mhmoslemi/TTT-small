import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    initial_h = initial_h * (target_sum / np.sum(initial_h)) if np.sum(initial_h) > 0 else initial_h
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s * 0.8)
    
    for iter in range(max_iter):
        # Perturb current best with smaller Gaussian noise
        sigma = 0.05 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum constraint with clipping and element-wise adjustment
        def adjust_sum(h):
            delta = target_sum - np.sum(h)
            if abs(delta) < 1e-7:
                return h
            if delta > 0:
                idx = np.argsort(h)
                h[idx[:int(delta)]] += 1e-5
            else:
                idx = np.argsort(-h)
                h[idx[:int(-delta)]] -= 1e-5
            return h
        
        h_perturbed = adjust_sum(h_perturbed)
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping
        time_used += 15
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
