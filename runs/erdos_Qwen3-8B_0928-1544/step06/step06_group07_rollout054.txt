import numpy as np
from cvxpy import Variable, Problem, Minimize, Constraint, sum_entries
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        h_initial[:ones_count] = 1.0
        np.random.shuffle(h_initial)
        h_initial += np.random.uniform(-0.1, 0.1, size=n_points)
        h_initial = np.clip(h_initial, 0.0, 1.0)
    
    def compute_overlaps(h):
        overlaps = []
        for k in range(n_points):
            valid_length = n_points - k
            if valid_length <= 0:
                continue
            val = 0.0
            for i in range(valid_length):
                val += h[i] * (1 - h[i + k])
            overlaps.append(val * dx)
        return overlaps
    
    def optimize_with_constraints(active_k, h_prev):
        h = Variable(n_points)
        tau = Variable()
        constraints = [sum_entries(h) == ones_count]
        for i in range(n_points):
            constraints.append(h[i] >= 0.0)
            constraints.append(h[i] <= 1.0)
        for k in active_k:
            valid_length = n_points - k
            expr = 0.0
            for i in range(valid_length):
                expr += h[i] * h[i + k]
            constraints.append(expr >= (1 - tau) * ones_count)
        objective = Minimize(tau)
        problem = Problem(objective, constraints)
        result = problem.solve(solver='ECOS', verbose=False)
        if result and problem.status == 'OPTIMAL':
            return h.value, tau.value
        return h_prev, np.inf
    
    active_k = []
    h = h_initial.copy()
    c5 = np.inf
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        overlaps = compute_overlaps(h)
        current_max = np.max(overlaps)
        if current_max < c5:
            break
        new_k = np.argmax(overlaps)
        if new_k not in active_k:
            active_k.append(new_k)
            h_new, tau_new = optimize_with_constraints(active_k, h)
            if tau_new < c5:
                h = h_new
                c5 = tau_new
            else:
                break
        else:
            break
    
    return h, c5, n_points
