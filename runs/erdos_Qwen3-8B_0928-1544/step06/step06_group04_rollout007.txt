import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    pop_size = 50
    mut_rate = 0.1
    tour_size = 5
    start_time = time.time()
    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    
    def init_pop():
        if 'initial_h_values' in globals():
            pop = [np.array(initial_h_values)]
        else:
            pop = [np.random.rand(n_points)]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            pop.append(h)
        return np.array(pop)
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def select_parent(pop, fitness):
        idxs = np.random.choice(len(pop), tour_size, replace=False)
        best_idx = idxs[np.argmin(fitness[idxs])]
        return pop[best_idx].copy()
    
    def crossover(a, b):
        child = a.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = b[i]
        return child
    
    def mutate(ind):
        for i in range(n_points):
            if np.random.rand() < mut_rate:
                ind[i] += np.random.uniform(-0.1, 0.1)
                ind[i] = np.clip(ind[i], 0, 1)
        s = np.sum(ind)
        if s != 0:
            ind *= (target_sum / s)
        return ind
    
    pop = init_pop()
    fitness = np.array([eval_c5(h) for h in pop])
    best_idx = np.argmin(fitness)
    best_h, best_c5 = pop[best_idx].copy(), fitness[best_idx]
    
    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(pop_size):
            p1 = select_parent(pop, fitness)
            p2 = select_parent(pop, fitness)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        new_fitness = np.array([eval_c5(h) for h in new_pop])
        combined = np.vstack((pop, new_pop))
        combined_fitness = np.hstack((fitness, new_fitness))
        indices = np.argsort(combined_fitness)[:pop_size]

        fitness = combined_fitness[indices]

    return best_h, best_c5, n_points
