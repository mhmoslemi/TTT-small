import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy() * (target_sum / np.sum(initial_h)) if np.sum(initial_h) > 0 else initial_h
    best_c5 = objective(best_h)
    time_used = 0
    
    for _ in range(100):
        h = best_h.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
        
        def adjust_sum(h):
            while np.sum(h) > target_sum:
                h[np.argmax(h)] -= 1e-5
            while np.sum(h) < target_sum:
                h[np.argmin(h)] += 1e-5
            return h
        
        h = adjust_sum(h)
        
        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        time_used += 100
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
