def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    def objective(h):
        max_corr = 0
        dx = 2.0 / n_points
        for k in range(n_points):
            shift = k * dx
            corr = 0.0
            for i in range(n_points):
                x = i * dx
                x_k = x + shift
                if 0 <= x_k < 2:
                    j = int(x_k // dx)
                    corr += h[i] * (1 - h[j]) * dx
            max_corr = max(max_corr, corr)
        return max_corr
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0

        sigma = 0.05 * (1 - iter / max_iter)
