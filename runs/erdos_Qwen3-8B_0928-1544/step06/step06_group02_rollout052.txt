    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for _ in range(300):
        # Perturb current best with Gaussian noise
        sigma = 0.15 * (1 - _ / 300)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum constraint with linear interpolation
        def adjust_sum(h):
            delta = target_sum - np.sum(h)
            h = h + delta / n_points
            return np.clip(h, 0.0, 1.0)
        
        h_perturbed = adjust_sum(h_perturbed)
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping based on time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
