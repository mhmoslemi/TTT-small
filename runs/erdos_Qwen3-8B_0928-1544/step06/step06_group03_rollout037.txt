import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Coarser grid
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    h = initial_h.copy()
    h /= np.sum(h) if np.sum(h) > 0 else 1.0
    h *= ones_count
    h = np.clip(h, 0.0, 1.0)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s / 5)
    
    for _ in range(max_iter):
        sigma = 0.1 * (1.0 - _ / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 5
        if time_used >= budget_s:
            break
    
    # Upscale to 70 points
    n_points = 70
    ones_count = n_points // 2
    h_final = np.repeat(best_h[:n_points//2], 2) if n_points % 2 == 0 else np.repeat(best_h[:n_points//2], 2)
    h_final = h_final * (ones_count / np.sum(h_final))
    h_final = np.clip(h_final, 1e-8, 1.0 - 1e-8)
    
    corr = np.correlate(h_final, 1.0 - h_final, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return h_final, best_c5, n_points
