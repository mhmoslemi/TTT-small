import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased points for finer resolution
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Generate a binary initial guess with structured placement
        initial_h = np.zeros(n_points)
        initial_h[:ones_count] = 1.0
        np.random.shuffle(initial_h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = Bounds(lb=0.0, ub=1.0)
    
    # Generate a larger initial population with diverse perturbations
    num_initial = 100
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):
        perturbed = initial_h + np.random.normal(0, 0.15, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 100)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)

    return best_h, best_c5, n_points
