from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        initial_h = np.clip(initial_h, 0.0, 1.0)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def compute_overlaps(h):
        dx = 2.0 / n_points
        overlaps = []
        for k in range(1, n_points):
            start = 0
            end = n_points - k
            overlap = np.sum(h[:end] * (1 - h[k:end + k])) * dx
            overlaps.append(overlap)
        return overlaps

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_start = time.time()

    for _ in range(200):
        if time.time() - time_start > budget_s:
            break
        overlaps = compute_overlaps(best_h)
        worst_k = np.argsort(overlaps)[-10:]
        constraints = []
        for k in worst_k:
            start_idx = 0

            def cons(h_val):
                return np.sum(h_val[:end_idx] * (1 - h_val[k:end_idx + k])) - best_c5 * (n_points / 2)
            constraints.append({'type': 'ineq', 'fun': lambda h, k=k: cons(h)})
        
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
