def compute_overlap_all(h, n_points, dx):
    overlaps = []
    for k in range(1, n_points):
        start = 0
        end = n_points - k
        overlap = np.sum(h[:end] * (1 - h[k:end + k])) * dx
        overlaps.append(overlap)
    return np.array(overlaps)

def compute_overlap(h, n_points, dx):
    return np.max(compute_overlap_all(h, n_points, dx))

def compute_gradient(h, k, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points - k):
        grad[i] += (1 - h[i + k])
        grad[i + k] -= h[i]
    return grad
