        h = h * (ones_count / np.sum(h))
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        learning_rate *= decay_rate
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
