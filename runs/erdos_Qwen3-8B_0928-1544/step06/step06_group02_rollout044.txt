import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    initial_h = initial_h * (target_sum / np.sum(initial_h)) if np.sum(initial_h) > 0 else initial_h

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 2)

    for _ in range(max_iter):
        perturbation = 0.03 * np.random.randn(n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        h_perturbed = np.clip(h_perturbed + (target_sum - np.sum(h_perturbed)) / n_points, 0.0, 1.0)

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )

        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        time_used += 0.5
        if time_used >= budget_s:

    return best_h, best_c5, n_points
