    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    for gen in range(1000):
        if time.time() - start_time >= budget_s:
            break
        fitness = [evaluate_erdos_solution(h) for h in population]
        sorted_idx = np.argsort(fitness)
        parents = []
        for _ in range(pop_size // 2):
            t = np.random.choice(sorted_idx, 5, replace=False)
            winners = t[np.argmin(fitness[t])]
            parents.append(population[winners])
        
        new_population = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            child = (p1 + p2) / 2
            mutation_rate = 0.1
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.uniform(-0.05, 0.05)
                    child[j] = np.clip(child[j], 0, 1)
            sum_c = np.sum(child)
            scale = ones_count / sum_c if sum_c != 0 else 1
            child = np.clip(child * scale, 0, 1)
            new_population.append(child)
        population = new_population[:pop_size]
        
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
