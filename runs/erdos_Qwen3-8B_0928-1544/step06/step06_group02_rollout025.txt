from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Efficient with smaller n_points
    target_sum = n_points / 2
    
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)

    if np.sum(initial_h) > 0:
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        initial_h = np.zeros(n_points)

        corr = np.correlate(h, 1 - h, mode='full')

        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Use L-BFGS-B with gradient and projection

        corr = np.correlate(h, 1 - h, mode='full')

        h_shifted = np.roll(h, shift=-k_max)

    result = minimize(
        fun=objective,
        x0=initial_h,

    return best_h, best_c5, n_points
