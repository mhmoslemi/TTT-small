            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            initial_h = result.x
    
    return best_h, best_c5, n_points
