        h = np.zeros(n_points)
        h[:35] = 1.0
        h[np.random.choice(n_points, 10, replace=False)] = 0.0
        h = np.clip(h, 0.0, 1.0)

    def compute_overlaps(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr[1:-1] * dx

    def compute_max_overlap(h):
        return np.max(compute_overlaps(h))

    def compute_gradient(k, h):
        grad = np.zeros(n_points)
        grad[:n_points - k] += 1 - h[k:]
        grad[k:] -= h[:n_points - k]
        return grad

    best_h = h.copy()
    best_c5 = compute_max_overlap(best_h)
    time_start = time.time()

    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        overlaps = compute_overlaps(best_h)
        max_k = np.argmax(overlaps) + 1
        k = max_k
        grad = compute_gradient(k, best_h)
        alpha = 1.0
        for _ in range(20):
            h_candidate = best_h - alpha * grad
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            h_sum = np.sum(h_candidate)
            delta = T - h_sum
            if delta != 0:
                h_candidate += delta / n_points
                h_candidate = np.clip(h_candidate, 0.0, 1.0)
            new_c5 = compute_max_overlap(h_candidate)
            if new_c5 < compute_max_overlap(best_h):
                break
            alpha *= 0.5
        h_candidate = best_h - alpha * grad
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_sum = np.sum(h_candidate)
        delta = T - h_sum
        if delta != 0:
            h_candidate += delta / n_points
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
        new_c5 = compute_max_overlap(h_candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate

    return best_h, best_c5, n_points
