from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    required_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        base = np.zeros(n_points)
        base[::2] = 1.0
        initial_h = base.copy()
        perturb = np.random.randn(n_points) * 0.1
        initial_h += perturb
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * required_sum / np.sum(initial_h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0.0, 1.0)] * n_points
    initial_population = [initial_h.copy()]
    num_initial = 100
    
    for _ in range(num_initial - 1):
        perturb = np.random.randn(n_points) * 0.1
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        scale = required_sum / new_sum
        new_h = new_h * scale
        new_h = np.clip(new_h, 0, 1)
        initial_population.append(new_h)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 8)
    
    start_time = time.time()
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(

        if result.success and result.fun < best_c5:

            initial_population.pop(np.random.randint(len(initial_population)))
