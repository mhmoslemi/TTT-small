import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0)] * n_points
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        def objective_for_k(h):
            total = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    total += h[i] * (1 - h[j])
            return total * dx
        
        # Optimize for the worst shift
        result = minimize(
            fun=objective_for_k,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return best_h, best_c5, n_points
