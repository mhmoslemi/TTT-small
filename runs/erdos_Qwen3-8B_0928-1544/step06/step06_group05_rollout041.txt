        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    def compute_gradient(h):
        conv = np.correlate(h, 1 - h, mode='full')
        max_k_idx = np.argmax(conv)
        n = len(h)
        k = max_k_idx - (n - 1)
        grad = np.zeros_like(h)
        for j in range(n):
            j_plus_k = j + k
            if 0 <= j_plus_k < n:
                grad[j] += 1 - h[j_plus_k]
            j_minus_k = j - k
            if 0 <= j_minus_k < n:
                grad[j] -= h[j_minus_k]
        return grad
    
    initial_population = [initial_h.copy()]
    initial_population.append(np.flip(initial_h))
    initial_population.append(np.roll(initial_h, 2))
    initial_population.append(np.roll(initial_h, 10))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 10)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        res = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            jac=compute_gradient,
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
            initial_population.append(res.x)
            initial_population.pop(np.random.randint(len(initial_population)))
    
    return best_h, best_c5, n_points
