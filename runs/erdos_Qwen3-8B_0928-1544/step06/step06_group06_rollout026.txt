import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = Bounds(0.0, 1.0)
    
    # Start with current best C5
    best_c5 = 0.381003
    best_h = initial_h.copy()
    
    # Critical shifts to check (evenly spaced)
    shift_count = 20
    shifts = np.linspace(0, 2, shift_count + 1)[1:-1]
    
    # Binary search over gamma
    gamma_low = 0.0
    gamma_high = 0.381
    max_iter = int(budget_s / 300)
    
    for _ in range(max_iter):
        gamma_mid = (gamma_low + gamma_high) / 2.0
        constraints = [
            {'type': 'ineq', 'fun': lambda h, k=shift: np.correlate(h, 1 - h, mode='full')[k] * (2.0 / n_points) - gamma_mid}
            for shift in shifts
        ]
        
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints + [{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            gamma_low = gamma_mid  # Try smaller gamma
        else:
            gamma_high = gamma_mid  # Try larger gamma
    
    return best_h, best_c5, n_points
