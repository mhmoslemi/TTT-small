import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Normalize initial_h
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    # Objective function
    def objective(h):
        h_norm = h.copy()
        h_norm = np.clip(h_norm, 0.0, 1.0)
        if np.sum(h_norm) != 0:
            h_norm = h_norm * (ones_count / np.sum(h_norm))
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Bounds
    bounds = [(0.0, 1.0)] * n_points
    
    start_time = time.time()
    
    # Differential Evolution
    if time.time() - start_time < budget_s * 0.7:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=100, popsize=25, tol=1e-6,
            mutation=(0.5, 1), recombination=0.8,
            seed=seed
        )
        elapsed = time.time() - start_time
    else:
        result_de = minimize(
            objective, initial_h, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': 100, 'ftol': 1e-9}
        )
        elapsed = time.time() - start_time
    
    # SLSQP Refinement
    best_h = result_de.x
    best_c5 = result_de.fun
    
    if elapsed < budget_s:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (ones_count / np.sum(best_h))
        best_c5 = objective(best_h)
        
        def constraint_sum(h):
            return np.sum(h) - ones_count
        
        result_slsqp = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (ones_count / np.sum(best_h))
    
    return best_h, best_c5, n_points
