from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
        diff = np.random.choice([-1, 1], size=n_points, p=[0.2, 0.8])
        initial_h += diff * 0.25
        initial_h = np.clip(initial_h, 0.0, 1.0)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def compute_overlaps(h):
        dx = 2.0 / n_points
        overlaps = []
        for k in range(1, n_points):
            start = 0
            end = n_points - k
            overlap = np.sum(h[:end] * (1 - h[k:end + k])) * dx
            overlaps.append(overlap)
        return overlaps
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_start = time.time()
    
    for _ in range(200):
        if time.time() - time_start > budget_s:
            break
        overlaps = compute_overlaps(best_h)
        worst_k = np.argsort(overlaps)[-5:]
        constraints = [lambda h, k=k: np.sum(h[:n_points-k] * (1 - h[k:])) - best_c5 * (n_points / 2) for k in worst_k]
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(

        )
