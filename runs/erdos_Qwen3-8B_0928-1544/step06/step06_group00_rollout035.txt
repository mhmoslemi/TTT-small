def run(seed=42, budget_s=1100, **kwargs):

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    best_coarse = population[0]
    best_c5 = evaluate(best_coarse)
    
    start_time = time.time()
    for _ in range(n_generations):

                child = (p1 + p2) / 2

                child = np.clip(p + perturb, 0, 1)
                child = child * (N_coarse / 2) / np.sum(child)
                new_population.append(child)
        population = new_population

            curr_c5 = evaluate(h)
