    best_c5 = evaluate_erdos_solution(initial_h)
    for gen in range(generations):
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        new_population = []
        for _ in range(n_population):
            parent1 = np.random.choice(population, size=3, replace=False)
            parent1 = min(parent1, key=lambda x: evaluate_erdos_solution(x))
            parent2 = np.random.choice(population, size=3, replace=False)
            parent2 = min(parent2, key=lambda x: evaluate_erdos_solution(x))
            point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:point], parent2[point:]])
            indices = np.random.choice(n_points, size=int(0.25 * n_points), replace=False)
            noise = np.random.normal(0, mutation_rate, len(indices))
            child[indices] += noise
            child = np.clip(child, 0, 1)
            child = child * (ones_count / np.sum(child))
            new_population.append(child)
        population = new_population
        mutation_rate *= 0.995
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
