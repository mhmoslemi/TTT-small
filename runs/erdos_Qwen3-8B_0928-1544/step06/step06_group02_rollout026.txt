from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)

    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)

            initial_h[i * 2] = 1.0
    else:
        initial_h = np.clip(initial_h, 0.0, 1.0)
        if np.sum(initial_h) > 0:
            initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 2)
    patience = 5
    
    for iter in range(max_iter):
        # Perturb current best with noise
        sigma = 0.1 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum
        current_sum = np.sum(h_perturbed)
        delta = target_sum - current_sum
        if np.abs(delta) > 1e-7:
            h_perturbed += delta / len(h_perturbed)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

                break
        
        time_used += 2
