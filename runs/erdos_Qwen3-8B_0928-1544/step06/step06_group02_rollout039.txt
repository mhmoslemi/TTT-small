from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Ensure initial_h meets sum constraint
    if np.sum(initial_h) > 0:
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        initial_h = np.zeros(n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for iter in range(max_iter):
        # Perturb current best with adaptive noise
        sigma = 0.1 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Ensure sum constraint with adjustment
        def adjust_sum(h):
            diff = target_sum - np.sum(h)
            if diff > 0:
                idx = np.argwhere(h < 1).ravel()
                h[idx[:int(diff)]] += 1.0
            elif diff < 0:
                idx = np.argwhere(h > 0).ravel()
                h[idx[:int(-diff)]] -= 1.0
            return h
        
        h_perturbed = adjust_sum(h_perturbed)
        
        # Local optimization with bounds and constraints
        result = minimize(
