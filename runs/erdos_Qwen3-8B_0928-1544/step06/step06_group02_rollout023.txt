import numpy as np
from cvxpy import Problem, Minimize, Variable, Binary, Constraint, sum as cvxsum
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
    target_sum = n_points / 2

    h = Variable(n_points, integer=True, bounds=(0, 1))
    C = Variable()

    # Constraints for overlapping shifts
    for k in range(n_points):
        expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h[i] * (1 - h[j])
        Constraint(expr <= C)

    sum_h = cvxsum(h)
    Constraint(sum_h == target_sum)

    prob = Problem(Minimize(C), [])
    try:
        prob.solve(solver='CBC', verbose=False, time_limit=budget_s)
    except:

    if prob.status == 'optimal':
        best_h = np.round(h.value).astype(float)

        return best_h, best_c5, n_points
