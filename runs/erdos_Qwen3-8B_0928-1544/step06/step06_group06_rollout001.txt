from scipy.optimize import minimize, differential_evolution

        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    def evaluate(h):
        return objective(h), constraint_sum(h)

        for _ in range(pop_size - 1):
            perturb = np.random.normal(0, 0.25, n_points)
            candidate = np.clip(initial_h + perturb, 0, 1)
            if np.abs(np.sum(candidate) - ones_count) < 1e-6:
                pop.append(candidate)
        return pop
    
    initial_population = generate_population(50)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 20)

    for _ in range(max_iter):

            if np.abs(np.sum(h) - ones_count) > 1e-3:
                continue
            res = differential_evolution(
                fun=objective,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
