import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points_coarse = 35
    n_points_fine = 70
    ones_count = n_points_coarse // 2
    if 'initial_h_values' in globals():
        initial_h_coarse = initial_h_values[:n_points_coarse]
    else:
        initial_h_coarse = np.full(n_points_coarse, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    best_h_coarse = initial_h_coarse.copy()
    best_c5 = objective(best_h_coarse)
    time_used = 0
    
    for _ in range(500):
        sigma = 0.05 * (1 - _ / 500)
        h_perturbed = initial_h_coarse + np.random.normal(0, sigma, n_points_coarse)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n_points_coarse,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h_coarse = result.x
        
        initial_h_coarse = result.x
        time_used += time.time() - time.time()
        if time_used >= budget_s:
            break
    
    initial_h_fine = np.repeat(best_h_coarse, 2)
    ones_count_fine = n_points_fine // 2
    
    def objective_fine(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def constraint_sum_fine(h):
        return np.sum(h) - ones_count_fine
    
    best_h_fine = initial_h_fine.copy()
    best_c5_fine = objective_fine(best_h_fine)
    
    time_used = 0

        h_perturbed = initial_h_fine + np.random.normal(0, sigma, n_points_fine)
