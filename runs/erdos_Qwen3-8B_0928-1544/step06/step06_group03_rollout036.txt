import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    def clip_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (ones_count / total)
    
    # Initial guess
    h = clip_rescale(initial_h)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(150):
        # Perturb current best with noise
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = clip_rescale(h_perturbed)
        
        # Local optimization
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 30, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 5:
            break
    
    # Final validation
    h_final = clip_rescale(best_h)
    corr = np.correlate(h_final, 1.0 - h_final, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return h_final, best_c5, n_points
