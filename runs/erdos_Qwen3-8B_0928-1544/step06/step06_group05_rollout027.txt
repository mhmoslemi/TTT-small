import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    bounds = [(0., 1.) for _ in range(n_points)]
    
    # Initialize from provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(n_points, 0.5)

    def repair(h):
        h = np.clip(h, 0., 1.)
        s = np.sum(h)
        if np.isclose(s, target_sum):
            return h
        delta = (target_sum - s) / n_points
        return np.clip(h + delta, 0., 1.)
    
    # Ensure initial repair
    h = repair(h)
    
    # Objective function to minimize max overlap
    def objective(x):
        h_repaired = repair(x)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx
    
    # Run differential evolution
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=50,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
