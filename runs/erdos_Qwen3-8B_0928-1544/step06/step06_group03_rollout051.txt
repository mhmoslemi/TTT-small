import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    desired_sum = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    population_size = 30
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (desired_sum / np.sum(new_h))
        population.append(new_h)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    F = 0.8
    CR = 0.9
    max_iter = int(1100 / 0.1)
    
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        np.random.shuffle(population)
        for i in range(len(population)):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + F * (b - c)
            rand_idx = np.random.rand(n_points) < CR
            trial = np.where(rand_idx, mutant, population[i])
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros_like(trial)
            else:
                trial = trial * (desired_sum / trial_sum)
            current_val = objective(population[i])
            trial_val = objective(trial)
            if trial_val < current_val:
                population[i] = trial
    
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    best_h = np.clip(best_h, 1e-8, 1.0 - 1e-8)
    best_h = best_h * (desired_sum / np.sum(best_h))
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return best_h, best_c5, n_points
