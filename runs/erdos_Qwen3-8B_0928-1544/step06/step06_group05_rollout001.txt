import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    if not isinstance(initial_h, np.ndarray):
        initial_h = np.full(n_points, 0.5)
    # Add symmetric perturbation
    if n_points % 2 == 0:
        half = n_points // 2
        perturb = np.random.rand(half) * 0.2 - 0.1
        initial_h[:half] += perturb
        initial_h[half:] = initial_h[:half][::-1]
    else:
        half = n_points // 2
        perturb = np.random.rand(half) * 0.2 - 0.1
        initial_h[:half] += perturb
        initial_h[half+1:] = initial_h[:half][::-1]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), [ones_count], [ones_count])

    initial_population = [initial_h.copy()]
    for _ in range(49):
        if np.random.rand() > 0.5:
            perturb = np.random.rand(n_points) * 0.1 - 0.05
            new_h = initial_h + perturb
        else:
            half = n_points // 2
            new_h = np.zeros(n_points)
            new_h[:half] = np.random.rand(half) * 0.5
            new_h[half:] = 1.0 - new_h[:half][::-1]
        initial_population.append(new_h)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 50)

    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        res = minimize(
            fun=objective,
            x0=current_h,
            method='trust-constr',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 20, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
            initial_population.append(res.x)
            initial_population.pop(np.random.randint(len(initial_population)))

    return best_h, best_c5, n_points
