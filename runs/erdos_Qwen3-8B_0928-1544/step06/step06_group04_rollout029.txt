import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:

    def compute_correlations(h):
        return np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
    best_h = h.copy()
    best_c5 = np.max(compute_correlations(h))
    beta = 1.0
    eta = 0.1
    start_time = time.time()
    while time.time() - start_time < budget_s:
        c = compute_correlations(h)
        current_c5 = np.max(c)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        Z = np.sum(np.exp(beta * c))
        grad = (np.exp(beta * c) / Z) * (1 - 2 * h)
        h += eta * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:

        beta *= 0.99
        eta *= 0.999
    return best_h, best_c5, n_points
