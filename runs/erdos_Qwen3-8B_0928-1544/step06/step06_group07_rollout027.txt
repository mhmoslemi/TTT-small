import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_pop = 20
    initial_h_values = None
    if 'initial_h_values' in globals():
        initial_h_values = initial_h_values
    else:
        initial_h_values = np.full(n_points, 0.5)
        initial_h_values += np.random.normal(0, 0.1, n_points)
        initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
        total = np.sum(initial_h_values)
        scaling = (n_points / 2) / total
        initial_h_values = initial_h_values * scaling

    population = [initial_h_values.copy()]
    for _ in range(n_pop - 1):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        scaling = (n_points / 2) / total
        h = h * scaling
        population.append(h)

    best_h = initial_h_values.copy()
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    time_start = time.time()

    while time.time() - time_start < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5

        top_idx = np.argsort(fitness)[:int(n_pop / 2)]
        parents = [population[i] for i in top_idx]

        new_population = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i + 1]
                child = (p1 + p2) / 2
                mutate_mask = np.random.rand(n_points) < 0.05
                child[mutate_mask] += np.random.normal(0, 0.1, np.sum(mutate_mask))
                child = np.clip(child, 0.0, 1.0)
                total = np.sum(child)
                scaling = (n_points / 2) / total
                child = child * scaling
                new_population.append(child)
            else:
                new_population.append(parents[i])
        new_population.append(current_best)
        population = new_population

        top_idx = np.argsort(fitness)[:2]
        for idx in top_idx:
            h = population[idx].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr[n_points:2 * n_points - 1])
            worst_k = max_index + 1
            grad = np.zeros_like(h)
            for i in range(n_points - worst_k):
                grad[i] += (1 - h[i + worst_k])
            for i in range(worst_k, n_points):
                grad[i] -= h[i - worst_k]
            learning_rate = 0.01
            h = h - learning_rate * grad
            total = np.sum(h)
            scaling = (n_points / 2) / total
            h = h * scaling
            h = np.clip(h, 0.0, 1.0)
            population[idx] = h

    return best_h, best_c5, n_points
