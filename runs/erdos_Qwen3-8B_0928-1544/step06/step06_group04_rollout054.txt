import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    start_time = time.time()
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
        population.append(initial_h)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            candidates = np.random.choice(population, tournament_size, replace=False)
            parent1 = min(candidates, key=lambda h: evaluate_c5(h))
            candidates2 = np.random.choice(population, tournament_size, replace=False)
            parent2 = min(candidates2, key=lambda h: evaluate_c5(h))
            # Uniform crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.uniform(-0.1, 0.1)
                        child[i] = np.clip(child[i], 0, 1)
            # Normalize sum
            child_sum = np.sum(child)
            if child_sum != 0:
                child = child * (target_sum / child_sum)
            new_population.append(child)
        
        # Evaluate new population
        new_fitness = [evaluate_c5(h) for h in new_population]
        # Combine old and new and keep best
        combined = population + new_population
        combined_fitness = [evaluate_c5(h) for h in combined]
        combined_idx = np.argsort(combined_fitness)[:population_size]
        population = [combined[i] for i in combined_idx]
        # Update best
        current_best_idx = np.argmin([evaluate_c5(h) for h in population])
        current_best_c5 = evaluate_c5(population[current_best_idx])
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx]
            best_c5 = current_best_c5
    
    return best_h, best_c5, n_points
