import numpy as np

import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
        noise = np.random.normal(0, 0.05, n_points)
        initial_h += noise
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_h = adjust_sum(initial_h, target_sum)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr[1:-1])
        return max_val * dx

    def adjust_sum(h, target_sum):
        current = np.sum(h)
        if abs(current - target_sum) < 1e-8:
            return h
        delta = target_sum - current
        h += delta / len(h)
        h = np.clip(h, 0.0, 1.0)
        return h

    pop_size = 20
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum(h, target_sum)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()

    while time.time() - time_start < budget_s:
        fitness = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        selected = [population[i] for i in sorted_idx[:int(len(population)*0.5)]]
        new_pop = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                p1, p2 = selected[i], selected[i+1]
                child = (p1 + p2) / 2.0
                if random.random() < 0.1:
                    noise = np.random.normal(0, 0.05, n_points)
                    child += noise
                    child = np.clip(child, 0.0, 1.0)
                    child = adjust_sum(child, target_sum)
                new_pop.append(child)
            else:
                new_pop.append(selected[i])
        new_pop += [population[i] for i in sorted_idx[:max(0, len(population) - len(new_pop))]]
        population = new_pop
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
