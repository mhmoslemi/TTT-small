import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased resolution for better precision
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Ensure initial h meets constraints
    h = initial_h.copy()
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total > 0:
        h = h * (ones_count / total)
    h = np.clip(h, 0.0, 1.0)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s * 0.8)  # Adjusted to allow for more iterations
    
    for iteration in range(max_iter):
        # Perturb current best with decreasing noise
        sigma = 0.05 * (1.0 - iteration / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 1.5  # Estimated time per iteration
        if time_used >= budget_s:
            break
    
    # Final validation
    h_final = np.clip(best_h, 1e-8, 1.0 - 1e-8)
    corr = np.correlate(h_final, 1.0 - h_final, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return h_final, best_c5, n_points
