import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

        return h_clipped * (n_points / 2.0 / total)

        corr = np.correlate(h_norm, 1 - h_norm, mode='full')

        objective,

        strategy='best1bin',

        maxiter=50,
