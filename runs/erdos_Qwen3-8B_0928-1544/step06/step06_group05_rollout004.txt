    for _ in range(max_iter):
        if time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = c5
        for i in range(swarm_size):
            if pbest_fitness[i] < best_fitness:
                best_fitness = pbest_fitness[i]
                best_h = pbest[i].copy()
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel = velocities[i] * w + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (best_h - particles[i])
            vel = np.clip(vel, -max_velocity, max_velocity)
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            s_new = np.sum(new_pos)
            delta = (ones_count - s_new) / n_points
            new_pos += delta
            new_pos = np.clip(new_pos, 0, 1)
            particles[i] = new_pos
    
    return best_h, best_fitness, n_points
