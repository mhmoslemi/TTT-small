    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    num_particles = 20
    particles = []
    velocities = []
    pbest = []
    pbest_obj = []
    
    # Initialize particles
    for _ in range(num_particles):
        if _ == 0:
            h = initial_h.copy()
        else:
            h = initial_h + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (R / np.sum(h))
        particles.append(h)
        velocities.append(np.random.uniform(-0.2, 0.2, n_points))
        pbest.append(h.copy())
        pbest_obj.append(objective(particles[-1], n_points))
    
    gbest = initial_h.copy()
    gbest_obj = objective(gbest, n_points)
    
    # PSO parameters
    w = 0.9
    c1 = c2 = 2.0
    max_iter = 500
    iterations = 0
    
    while iterations < max_iter:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            vel = np.clip(vel, -0.2, 0.2)
            velocities[i] = vel
            # Update position
            particles[i] += vel
            # Clip to [0,1]
            new_h = np.clip(particles[i], 0, 1)
            # Scale to meet sum
            sum_h = np.sum(new_h)
            if sum_h == 0:
                new_h = np.zeros(n_points)
            else:
                scale = R / sum_h
                new_h = np.clip(new_h * scale, 0, 1)
            particles[i] = new_h
            # Compute objective
            current_obj = objective(particles[i], n_points)
            # Update pbest
            if current_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = current_obj
                # Update gbest
                if current_obj < gbest_obj:
                    gbest = particles[i].copy()
                    gbest_obj = current_obj
        iterations += 1
    
    # Final validation
    final_h = gbest.copy()
    # Ensure sum is exactly R
    sum_h = np.sum(final_h)
    if sum_h != R:
        scale = R / sum_h
        final_h = np.clip(final_h * scale, 0, 1)
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    c5 = np.max(corr) * (2.0 / n_points)
    
    return final_h, c5, n_points

def objective(h_vec, n_points):
    corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
    return np.max(corr) * (2.0 / n_points)
