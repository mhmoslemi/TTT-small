import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    n_points = 70
    ones_count = n_points // 2
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    elapsed = 0
    
    def genetic_search(n_gen=100, pop_size=50, n_seg=20):
        nonlocal best_c5, best_h
        segs = np.linspace(0, 2, n_seg+1)
        pop = []
        for _ in range(pop_size):
            h = np.random.uniform(0, 1, n_seg)
            h = h * (ones_count / np.sum(h))
            pop.append(h)
        for gen in range(n_gen):
            scores = [objective(h) for h in pop]
            inds = np.argsort(scores)
            top = pop[inds[:pop_size//10]]
            new_pop = []
            for _ in range(pop_size//2):
                p1, p2 = np.random.choice(top, 2, replace=False)
                child = (p1 + p2) / 2
                child = child * (ones_count / np.sum(child))
                new_pop.append(child)
                if np.random.rand() < 0.1:
                    child += np.random.normal(0, 0.1, n_seg)
                    child = np.clip(child, 0, 1)
                    child = child * (ones_count / np.sum(child))
                new_pop.append(child)
            pop = new_pop
            scores = [objective(h) for h in pop]
            best_idx = np.argmin(scores)
            if scores[best_idx] < best_c5:
                best_c5 = scores[best_idx]
                best_h = pop[best_idx]
    
    genetic_search(n_gen=50, pop_size=50, n_seg=20)
    
    while elapsed < budget_s - 100:
        h_perturbed = best_h + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        if result.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        elapsed = time.time() - start_time
