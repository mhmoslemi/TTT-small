import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points_low = 35
    n_points_high = 70
    ones_low = n_points_low // 2
    ones_high = n_points_high // 2

    # Initial setup
    if 'initial_h_values' in globals():
        h_low = initial_h_values.copy()
    else:
        h_low = np.zeros(n_points_low)
    
    h_low = np.clip(h_low, 0.0, 1.0)
    h_low = h_low * (ones_low / np.sum(h_low))
    h_low = np.clip(h_low, 0.0, 1.0)

    # Perturb and optimize low-res candidates
    candidates_low = [h_low + np.random.normal(0, 0.1, n_points_low) for _ in range(5)]
    candidates_low = [np.clip(c, 0, 1) * (ones_low / np.sum(c)) for c in candidates_low]

    best_h_low = h_low
    best_c5 = float('inf')

    for h_candidate in candidates_low:
        def obj_low(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points_low)
        
        def cons_low(x):
            return np.sum(x) - ones_low
        
        res = minimize(obj_low, h_candidate, method='SLSQP',
                       bounds=[(0,1)]*n_points_low,
                       constraints={'type':'eq', 'fun':cons_low},
                       options={'maxiter':100, 'ftol':1e-8})
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h_low = res.x

    # Upscale and optimize high-res
    h_high = np.repeat(best_h_low, 2)
    h_high = np.clip(h_high, 0, 1) * (ones_high / np.sum(h_high))
    h_high = np.clip(h_high, 0, 1)

    candidates_high = [h_high + np.random.normal(0, 0.05, n_points_high) for _ in range(3)]
    candidates_high = [np.clip(c, 0, 1) * (ones_high / np.sum(c)) for c in candidates_high]

    best_h_high = h_high
    best_c5_high = float('inf')

    for h_candidate in candidates_high:
        def obj_high(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points_high)
        
        def cons_high(x):
            return np.sum(x) - ones_high
        
        res = minimize(obj_high, h_candidate, method='SLSQP',
                       bounds=[(0,1)]*n_points_high,
                       constraints={'type':'eq', 'fun':cons_high},
                       options={'maxiter':100, 'ftol':1e-8})
        
        if res.success and res.fun < best_c5_high:
            best_c5_high = res.fun
            best_h_high = res.x

    # Final evaluation
    h_final = np.clip(best_h_high, 1e-8, 1 - 1e-8)

    corr = np.correlate(h_final, 1 - h_final, mode='full')
