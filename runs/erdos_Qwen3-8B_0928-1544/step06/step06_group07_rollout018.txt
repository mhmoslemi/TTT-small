            c5 = compute_c5(h)
            population_fitness.append(c5)
        sorted_indices = np.argsort(population_fitness)
        top_candidates = [population[i] for i in sorted_indices[:10]]

        new_population = []
        for i in range(0, len(top_candidates), 2):
            if i + 1 < len(top_candidates):
                parent1 = top_candidates[i]
                parent2 = top_candidates[i + 1]
                child = (parent1 + parent2) / 2
                new_population.append(child)
            else:
                new_population.append(top_candidates[i])

        mutated_population = []
        for h in new_population:
            mut = np.random.normal(0, 0.1, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)
            current_sum = np.sum(h_mut)
            delta = current_sum - S_desired
            if delta != 0:
                h_mut[-1] += delta
                h_mut = np.clip(h_mut, 0, 1)
            mutated_population.append(h_mut)

        refined_population = []
        for h in mutated_population:
            if np.all(0 <= h <= 1) and np.isclose(np.sum(h), S_desired, atol=1e-5):
                refined_population.append(h)
        for h in refined_population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        population = refined_population

    return best_h, best_c5, n_points
