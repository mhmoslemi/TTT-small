            overlaps.append(overlap)
        return np.max(overlaps)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_start = time.time()
    
    for _ in range(200):
        if time.time() - time_start > budget_s:
            break
        overlaps = []
        k_values = np.arange(1, n_points)
        for k in k_values:
            valid_indices = np.arange(n_points - k)
            overlap = np.sum(best_h[valid_indices] * (1 - best_h[k + valid_indices])) * (2.0 / n_points)
            overlaps.append(overlap)
        worst_k = np.argmax(overlaps)
        worst_overlap = overlaps[worst_k]
        
        def constraint(h):
            valid_indices = np.arange(n_points - k_values[worst_k])
            return np.sum(h[valid_indices] * (1 - h[k_values[worst_k] + valid_indices])) * (2.0 / n_points) - best_c5 * (n_points / 2)
        
        constraints = [{'type': 'ineq', 'fun': constraint}]
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        else:
            break
    
    return best_h, best_c5, n_points
