import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    n_points = 70
    ones_count = n_points // 2
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    
    def objective(x):
        h_val = x.copy()
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    for _ in range(10):
        sigma = 0.2 * (1.0 - (time.time() - start_time) / budget_s)
        h_perturbed = h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h = result.x
    
    return best_h, best_c5, n_points
