        # Adaptive perturbation
        sigma = 0.05 * np.exp(-iter / 100)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Introduce diversity to avoid local optima
        if iter % 20 == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            best_h = np.clip(best_h + perturbation, 0, 1)
            best_h = best_h * (ones_count / np.sum(best_h))
        
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
