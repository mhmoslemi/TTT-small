            tournament_fitness = [evaluate_c5(h) for h in tournament]
            best_tour_index = np.argmin(tournament_fitness)
            parents.append(tournament[best_tour_index])

        offspring = []
        for _ in range(pop_size):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            current_sum = np.sum(child)
            if current_sum != 0:
                child = child * (target_sum / current_sum)
            offspring.append(child)

        population = [current_best] + offspring[:-1]

    return best_h, best_c5, n_points
