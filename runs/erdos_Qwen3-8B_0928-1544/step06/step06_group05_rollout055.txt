from scipy.optimize import minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
        perturb = np.random.normal(0, 0.1, n_points)
        initial_h += perturb
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    initial_population = [initial_h.copy()]
    num_initial = 100
    
    for _ in range(num_initial - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        initial_population.append(initial_h + perturb)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 10)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        res = minimize(
