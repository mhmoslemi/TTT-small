import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    h = h / np.sum(h) * ones
    particles = [h]
    for _ in range(19):
        p = particles[0] + np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * (ones / np.sum(p))
        particles.append(p)
    velocities = [np.random.uniform(-0.2, 0.2, n_points) for _ in range(20)]
    
    def obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        return np.max(c) * (2.0 / n_points)
    
    pb = [p.copy() for p in particles]
    gb = particles[0].copy()
    best = obj(gb)
    
    w, c1, c2 = 0.9, 2.0, 2.0
    max_iter = int(budget_s / 2)
    vel_lim = 0.2
    
    for _ in range(max_iter):
        for i in range(20):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1*r1*(pb[i] - particles[i]) + c2*r2*(gb - particles[i])
            vel = np.clip(vel, -vel_lim, vel_lim)
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            s = np.sum(particles[i])
            if s > 0:
                particles[i] = particles[i] * (ones / s)
        for i in range(20):
            curr = obj(particles[i])
            if curr < obj(pb[i]):
                pb[i] = particles[i].copy()
            if curr < best:
                gb = particles[i].copy()
                best = curr
    
    h_final = np.clip(gb, 1e-8, 1 - 1e-8)
    s = np.sum(h_final)
    if s > 0:
        h_final = h_final * (ones / s)
    c = np.correlate(h_final, 1 - h_final, mode='full')
    best_c5 = np.max(c) * (2.0 / n_points)
    return h_final, best_c5, n_points
