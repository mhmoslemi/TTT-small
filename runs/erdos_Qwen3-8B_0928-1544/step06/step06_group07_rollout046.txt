                {'type': 'ineq', 'fun': c} for c in constraint_exprs
            ],
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        else:
            break

    return best_h, best_c5, n_points
