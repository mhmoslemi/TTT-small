        initial_population.append(perturbed)
    
    best_h = initial_population[0].copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 150)
    for _ in range(max_iter):
        parent1 = initial_population[np.random.randint(len(initial_population))]
        parent2 = initial_population[np.random.randint(len(initial_population))]
        crossover = np.random.randint(1, n_points)
        child = np.concatenate([parent1[:crossover], parent2[crossover:]])
        mutation_rate = 0.2
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                child[i] += np.random.normal(0, 0.1)
                child[i] = np.clip(child[i], 0, 1)
        sum_child = np.sum(child)
        if sum_child != 0:
            child = child * (ones_count / sum_child)
        current_c5 = objective(child)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = child.copy()
        idx = np.random.randint(len(initial_population))
        initial_population[idx] = child
    
    sum_best = np.sum(best_h)
    if sum_best != 0:
        best_h = best_h * (ones_count / sum_best)
    return best_h, best_c5, n_points
