                fun=objective,
                x0=h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        
        initial_h = best_h
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
