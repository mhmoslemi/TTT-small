import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values

        initial_h = np.full(n_points, 0.5)

    best_h = initial_h.copy()

    max_iter = int(budget_s / 300)
    
    for _ in range(max_iter):

        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

                idx = np.random.randint(len(initial_population))
                initial_population[idx] = result.x
