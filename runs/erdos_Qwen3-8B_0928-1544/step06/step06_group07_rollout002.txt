        new_population = []
        while len(new_population) < len(population):
            p1 = np.random.choice(top_parents)
            p2 = np.random.choice(top_parents)
            child = crossover(p1, p2)
            new_population.append(mutate(child))
        population = new_population
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx]
        if best_h is not None:
            def obj_func(x):
                h = x.reshape(-1)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = minimize(
                fun=obj_func,
                x0=best_h.flatten(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                best_h = result.x.reshape(-1)
                best_h = np.clip(best_h, 0.0, 1.0)
                sum_h = best_h.sum()
                scale = target_sum / sum_h
                best_h = best_h * scale
                best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n_points
