from scipy.optimize import minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.full(n_points, 0.5)
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h += (np.sum(h) - ones_count) / n_points
        h = np.clip(h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    max_iter = int(budget_s / 5)
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(max_iter):
        res = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        else:
            break
    
    return best_h, best_c5, n_points
