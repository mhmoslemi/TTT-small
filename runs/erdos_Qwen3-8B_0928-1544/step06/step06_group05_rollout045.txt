import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, target_sum):
            return h
        delta = (target_sum - total) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        return h

    swarm_size = 50
    swarm = []
    for _ in range(10):

        h = initial_h + perturb
        h = repair(h)
        swarm.append(h)

        h = np.random.rand(n_points)
        h = h / h.sum() * target_sum
        h = repair(h)
        swarm.append(h)

        h = h / h.sum() * target_sum
        h = repair(h)

    swarm = np.array(swarm)
