import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def compute_overlaps(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_overlaps(best_h).max()
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlaps = corr * (2.0 / n_points)
        worst_index = np.argmax(overlaps)
        target_val = overlaps[worst_index]
        
        def objective(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode='full')
            return corr[worst_index] * (2.0 / n_points)
        
        bounds = [(0.0, 1.0)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            break
    
    return best_h, best_c5, n_points
