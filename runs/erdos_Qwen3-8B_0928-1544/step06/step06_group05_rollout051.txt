import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_points_opt = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    initial_pop_size = 100
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Structured initial population
        initial_h = np.full(n_points, 0.5)
        # Add periodic patterns
        for _ in range(6):
            pattern = np.zeros(n_points)
            pattern[:n_points//2] = 0.8
            pattern[n_points//2:] = 0.2
            initial_h = np.vstack((initial_h, pattern))
        # Add random perturbations
        for _ in range(6):
            perturb = np.random.uniform(-0.2, 0.2, n_points)
            initial_h = np.vstack((initial_h, initial_h[0] + perturb))
        # Normalize and clip
        initial_h = (initial_h - initial_h.min()) / (initial_h.max() - initial_h.min())
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h[:initial_pop_size]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    initial_population = [initial_h[i].copy() for i in range(initial_pop_size)]

    best_h = initial_population[0].copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 10)

    for _ in range(max_iter):
        pop = [h.copy() for h in initial_population]
        evals = [objective(h) for h in pop]
        idx = np.argmin(evals)
        best_h = pop[idx]
        best_c5 = evals[idx]

        # EVOLUTION STEP
        for _ in range(2):
            h = pop[idx].copy()
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else h
            res = minimize(
                fun=objective,
                x0=h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
                pop.append(res.x)
                pop.pop(np.random.randint(len(pop)))
        initial_population = pop

    return best_h, best_c5, n_points_opt
