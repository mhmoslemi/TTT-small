import numpy as np
from scipy.optimize import minimize
import time

def adjust_sum(h, target):
    s = np.sum(h)
    if np.isclose(s, target):
        return h
    delta = target - s
    if delta == 0:
        return h
    h = h.copy()
    if delta > 0:
        # Increase some values without exceeding 1.0
        indices = np.argsort(h)
        for i in indices:
            if h[i] < 1.0:
                add = min(delta, 1.0 - h[i])
                h[i] += add
                delta -= add
                if delta == 0:
                    break
    else:
        # Decrease some values without going below 0
        indices = np.argsort(h)[::-1]
        for i in indices:
            if h[i] > 0.0:
                sub = min(-delta, h[i])
                h[i] -= sub
                delta += sub
                if delta == 0:
                    break
    return h

def eval_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        best_h = initial_h_values.copy()
    else:
        best_h = np.full(n_points, 0.5)
    
    def objective(h):
        return eval_c5(h, n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_c5 = eval_c5(best_h, n_points)
    start_time = time.time()
    
    for _ in range(100):  # Limit iterations to avoid excessive runtime
        if time.time() - start_time > budget_s:
            break
        h_perturb = best_h + np.random.normal(0, 0.3, n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = adjust_sum(h_perturb, ones_count)
        try:
            result = minimize(
                fun=objective,
                x0=h_perturb,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
