        initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    initial_population = [initial_h.copy()]
    for _ in range(49):
        perturb = np.random.rand(n_points) * 0.2 - 0.1
        candidate = initial_h + perturb
        candidate = np.clip(candidate, 0, 1)
        # Adjust sum
        diff = (ones_count - np.sum(candidate)) / n_points
        candidate += diff
        candidate = np.clip(candidate, 0, 1)
        initial_population.append(candidate)

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    max_iter = int(budget_s / 10)

    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        res = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
            # Replace a random point in population
            initial_population[np.random.randint(len(initial_population))] = res.x

    return best_h, best_c5, n_points
