import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    desired_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.choice([0, 1], size=n_points)
        h = np.where(h, 1.0, 0.0)
        h[:n_points // 2] = 1.0
        h[n_points // 2:] = 0.0
        np.random.shuffle(h)
    
    def compute_overlaps(h, k):
        if k >= n_points:
            return 0.0
        return np.dot(h[:n_points - k], 1 - h[k:]) * (2.0 / n_points)
    
    def compute_gradient(h, k):
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k < n_points:
                grad[j] += 1.0 - h[j + k]
            if j >= k:
                grad[j] -= h[j - k]
        return grad
    
    def adjust_sum(h):
        current_sum = np.sum(h)
        scale = desired_sum / current_sum
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        return h_scaled
    
    best_h = h.copy()
    best_c5 = compute_overlaps(best_h, 1)
    time_start = time.time()
    max_iterations = 1000
    
    for _ in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        current_c5 = compute_overlaps(best_h, 1)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        max_overlap = 0.0
        k_max = 1
        for k in range(1, n_points):
            overlap = compute_overlaps(best_h, k)
            if overlap > max_overlap:
                max_overlap = overlap
                k_max = k
        if max_overlap == 0.0:
            break
        grad = compute_gradient(best_h, k_max)
        direction = -grad
        alpha = 0.1
        found = False
        for _ in range(10):
            step = alpha * direction
            new_h = best_h + step
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = adjust_sum(new_h)
            new_overlap = compute_overlaps(new_h, k_max)
            if new_overlap < max_overlap:
                best_h = new_h
                found = True
                break
            alpha *= 0.5
        if not found:
            break
    
    return best_h, best_c5, n_points
