        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=500,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-12,
        polish=True,
        seed=seed
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
