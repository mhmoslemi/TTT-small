import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = n_points / 2
    
    if 'initial_h_values' in globals():

        return np.sum(h[:n_points - k] * (1 - h[k:])) * dx
    
    best_h = initial_h.copy()
    best_c5 = max(compute_overlap(best_h, k) for k in range(1, n_points))
