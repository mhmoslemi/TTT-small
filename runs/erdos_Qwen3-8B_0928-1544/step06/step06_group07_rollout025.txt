        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[1:n_points]
        max_overlap = np.max(overlaps)
        max_k_idx = np.argmax(overlaps) + 1
        if max_overlap <= best_c5:
            break
        k_val = max_k_idx
        grad = np.zeros(n_points)
        for i in range(n_points - k_val):
            grad[i] += 1.0 - h[i + k_val]
            grad[i + k_val] -= h[i]
        alpha = 0.1
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        new_h *= target_sum / current_sum
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_overlaps = new_corr[1:n_points]
        new_max = np.max(new_overlaps)
        if new_max < max_overlap:
            h = new_h
            best_c5 = new_max
        else:
            alpha *= 0.5
            new_h = h - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h)
            new_h *= target_sum / current_sum
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_overlaps = new_corr[1:n_points]
            new_max = np.max(new_overlaps)
            if new_max < max_overlap:
                h = new_h
                best_c5 = new_max
            else:
                break

    return h, best_c5, n_points

def calculate_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n_points
    return np.max(corr[1:n_points]) * dx
