import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Coarse genetic search
    n_coarse = 20
    pop_size = 50
    gen_limit = 100
    population = [np.random.choice([0,1], size=n_coarse, p=[0.5,0.5]) for _ in range(pop_size)]
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_coarse)
    
    for _ in range(gen_limit):
        fit = [eval_c5(h) for h in population]
        fit_idx = np.argsort(fit)
        elite = population[fit_idx[:int(len(population)*0.1)]]
        new_pop = []
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = np.zeros(n_coarse)
            for i in range(n_coarse):
                child[i] = p1[i] if np.random.rand() < 0.5 else p2[i]
            for i in range(n_coarse):
                if np.random.rand() < 0.1:
                    child[i] ^= 1
            new_pop.append(child)
        population = new_pop
    
    # Best coarse candidate
    best_h_coarse = population[np.argmin([eval_c5(h) for h in population])]
    
    # Upscale to n_points
    h_fine = np.repeat(best_h_coarse, 3)[:n_points]
    if len(h_fine) < n_points:
        h_fine = np.pad(h_fine, (0, n_points - len(h_fine)), 'constant')
    h_fine = h_fine * (ones_count / np.sum(h_fine))
    
    # Fine SLSQP
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h_perturbed = h_fine + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h_fine = result.x
    
    return best_h, best_c5, n_points
