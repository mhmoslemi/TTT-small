        h_adjusted = h_clipped + delta
        h_clipped = np.clip(h_adjusted, 0, 1)
        return h_clipped

    def objective(h_vec):
        h = repair(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    initial_population = []
    for _ in range(5):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = initial_h + perturb
        candidate = np.clip(candidate, 0, 1)
        initial_population.append(candidate)
    for _ in range(5):
        pattern = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
        initial_population.append(pattern)
    for _ in range(5):
        candidate = np.random.rand(n_points)
        candidate_clipped = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate_clipped)
        delta = (target_sum - current_sum) / n_points
        candidate_repaired = candidate_clipped + delta
        candidate_repaired = np.clip(candidate_repaired, 0, 1)
        initial_population.append(candidate_repaired)

    initial_population = [np.clip(p, 0, 1) for p in initial_population]
    mean = np.mean(initial_population, axis=0)
    es = cma.CMAEvolutionStrategy(mean, 0.5, {'popsize': 50})
    best_h = initial_population[0].copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        solutions = es.ask()
        for x in solutions:
            x_repaired = repair(x)
            val = objective(x_repaired)
            es.tell(x, val)
            if val < best_c5:
                best_c5 = val
                best_h = x_repaired

    return best_h, best_c5, n_points
