import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Genetic search parameters
    generations = 30
    pop_size = 50
    elite_frac = 0.1
    mutation_rate = 0.05
    
    def generate_candidate():
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        return h
    
    population = [generate_candidate() for _ in range(pop_size)]
    
    for _ in range(generations):
        fitness = [objective(p) for p in population]
        elites = np.argsort(fitness)[:int(pop_size * elite_frac)]
        new_pop = [population[i] for i in elites]
        
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (population[p1] + population[p2]) / 2
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(len(child))
                child[idx] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (ones_count / np.sum(child))
            new_pop.append(child)
        population = new_pop
    
    best_candidate = min(population, key=objective)
    best_c5 = objective(best_candidate)
    best_h = best_candidate
    
    # Gradient refinement
    for _ in range(100):
        perturb = np.random.normal(0, 0.05, n_points)
        trial = np.clip(best_h + perturb, 0, 1)
        trial = trial * (ones_count / np.sum(trial))
        
        result = minimize(
            fun=objective,
            x0=trial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    # Final safety check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (ones_count / np.sum(best_h))
    
    return best_h, best_c5, n_points
