        max_val = corr.max()
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        if lag < 0:
            lag = -lag
        lag = min(lag, n_points - 1)
        grad = compute_gradient(h, lag)
        
        # Line search
        def objective(x):
            new_h = h - x * grad
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            scale = ones_count / sum_new if sum_new != 0 else 1.0
            scaled_h = new_h * scale
            corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
            return np.max(corr)
        
        alpha, _, _ = line_search(objective, lambda x: -grad, 0.0, 1.0)
        alpha = np.clip(alpha, 0, 1.0)
        h = update_h(h, grad, alpha)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Perturbation
        perturbation = 0.2 * np.sin(h)
        h = np.clip(h + perturbation, 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h if sum_h != 0 else 1.0
        h = h * scale
    
    return best_h, best_c5, n_points
