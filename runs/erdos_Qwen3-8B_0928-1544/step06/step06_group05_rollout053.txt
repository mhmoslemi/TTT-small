            h = np.clip(h, 0, 1)
        population.append(h)
        velocities.append(np.zeros(n_points))
        pbests.append(h.copy())
        corr = np.correlate(h, 1 - h, 'full')
        pbest_values.append(np.max(corr) * dx)

    gbest_idx = np.argmin(pbest_values)
    gbest = pbests[gbest_idx].copy()
    best_c5 = pbest_values[gbest_idx]

    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.5
    t0 = time.time()

    while time.time() - t0 < budget_s:
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (pbests[i] - population[i])
            social = c2 * r2 * (gbest - population[i])
            vel = velocities[i] * w + cognitive + social
            vel = np.clip(vel, -max_vel, max_vel)
            new_pos = population[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = np.sum(new_pos)
            delta = (target_sum - current_sum) / n_points
            new_pos += delta
            new_pos = np.clip(new_pos, 0, 1)
            population[i] = new_pos.copy()
            corr = np.correlate(population[i], 1 - population[i], 'full')
            current_c5 = np.max(corr) * dx
            if current_c5 < pbest_values[i]:
                pbests[i] = population[i].copy()
                pbest_values[i] = current_c5
                if current_c5 < best_c5:
                    gbest = pbests[i].copy()
                    best_c5 = current_c5
    return gbest, best_c5, n_points
