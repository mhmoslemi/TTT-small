import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better precision without exceeding 1k samples
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Fallback to a binary random initialization
        initial_h = np.zeros(n_points)
        initial_h[:ones_count] = 1.0
        np.random.shuffle(initial_h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def compute_overlaps(h):
        dx = 2.0 / n_points
        overlaps = []
        for k in range(1, n_points):
            start = 0
            end = n_points - k
            overlap = np.sum(h[:end] * (1 - h[k:end + k])) * dx
            overlaps.append(overlap)
        return overlaps

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_start = time.time()

    for _ in range(200):  # More iterations for better exploration
        if time.time() - time_start > budget_s:
            break
        overlaps = compute_overlaps(best_h)
        worst_k_idx = np.argmax(overlaps)
        k = worst_k_idx + 1  # k ranges from 1 to n_points -1
