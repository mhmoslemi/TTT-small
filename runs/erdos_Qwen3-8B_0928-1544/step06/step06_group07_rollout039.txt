            if problem.status == 'optimal' and tau.value < best_c5:
                best_c5 = tau.value
                best_h = h.value
        except:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
    if abs(total - n_points / 2) > 1e-6:
        diff = (total - n_points / 2) / n_points
        best_h += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n_points
