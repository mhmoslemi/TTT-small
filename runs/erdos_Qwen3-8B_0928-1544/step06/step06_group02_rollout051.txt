from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        # Perturb current best with shrinking noise
        sigma = 0.2 * (1 - _ / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum constraint with clipping and element-wise adjustment
        def adjust_sum(h):
            while np.sum(h) > target_sum:
                h[np.argmax(h)] -= 1e-5
            while np.sum(h) < target_sum:
                h[np.argmin(h)] += 1e-5
            return h
        
        h_perturbed = adjust_sum(h_perturbed)
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
