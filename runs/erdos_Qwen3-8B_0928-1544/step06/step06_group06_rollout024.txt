import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        initial_h[:ones_count] = 1.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    initial_population = [initial_h.copy()]
    for _ in range(100):
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(100):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)
            initial_population.pop(np.random.randint(len(initial_population)))
        if time.time() - start_time > budget_s - 100:
            break
    
    return best_h, best_c5, n_points
