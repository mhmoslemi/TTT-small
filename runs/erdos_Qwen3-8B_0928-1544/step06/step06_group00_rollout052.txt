from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    def obj(h):

        return np.max(corr) * (2.0 / n_points)
    
    def norm(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        return h * (ones_count / s) if s != 0 else h
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    # Simulated Annealing
    temp = 1.0
    cooling = 0.997
    for _ in range(int(budget_s * 0.25)):
        if time.time() - start_time > budget_s:
            break
        i = np.random.randint(n_points)
        new_h = best_h.copy()
        new_h[i] = 1 - new_h[i]
        j = np.random.randint(n_points)
        new_h[j] = np.clip(new_h[j] + (ones_count - np.sum(new_h)) / n_points, 0, 1)
        new_c5 = obj(new_h)
        if np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            best_h, best_c5 = new_h, new_c5
        temp *= cooling
    
    # Particle Swarm Optimization
    pop_size = 50
    particles = np.random.rand(pop_size, n_points) * 2 - 1
    particles = norm(particles)
    velocities = np.random.rand(pop_size, n_points) * 0.1
    pbest = particles.copy()
    pbest_scores = np.array([obj(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_scores)]
    for _ in range(int(budget_s * 0.25)):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 2 * r1 * (pbest[i] - particles[i]) + 2 * r2 * (gbest - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = norm(particles[i])
            curr = obj(particles[i])
            if curr < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = curr
        gbest = pbest[np.argmin(pbest_scores)]
        if obj(gbest) < best_c5:
            best_h, best_c5 = gbest, obj(gbest)
    
    # Genetic Algorithm
    pop = np.random.rand(50, n_points) * 2 - 1

    for _ in range(int(budget_s * 0.25)):
        if time.time() - start_time > budget_s:

        offspring = []
        for _ in range(25):
            a, b = np.random.choice(50, 2, replace=False)
            child = (pop[a] + pop[b]) / 2
            child = norm(child)
            offspring.append(child)
        pop = np.vstack([pop, offspring])
