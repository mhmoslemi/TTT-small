import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    n_half = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_half] = 1
        noise = np.random.normal(0, 0.05, size=n_points)
        h += noise
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h *= n_half / current_sum

    def compute_c5(h_in):
        max_k = 0
        for k in range(1, n_points):
            end = n_points - k
            val = np.dot(h_in[:end], 1 - h_in[k:end + k]) * dx
            if val > max_k:
                max_k = val
        return max_k

    def grad_c5(h_in):
        k = np.argmax([np.dot(h_in[:n_points - k], 1 - h_in[k:n_points]) * dx for k in range(1, n_points)])
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            grad[i] += dx * (1 - h_in[i + k])
            grad[i + k] -= dx * h_in[i]
        return grad

    population = [h.copy() for _ in range(20)]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()

    for gen in range(1000):
        if time.time() - time_start > budget_s:
            break
        scores = [compute_c5(ind) for ind in population]
        sorted_inds = np.argsort(scores)
        top = sorted_inds[:10]
        new_pop = []
        for i in range(0, len(top), 2):
            if i + 1 < len(top):
                new_pop.append((population[top[i]] + population[top[i + 1]]) / 2)
            else:
                new_pop.append(population[top[i]])
        for h_mut in new_pop:
            if np.random.rand() < 0.05:
                idx = np.random.choice(n_points)
                h_mut[idx] += np.random.normal(0, 0.1)
                h_mut = np.clip(h_mut, 0, 1)
                total = np.sum(h_mut)
                h_mut *= n_half / total
        combined = sorted(population, key=compute_c5)[:2] + new_pop
        combined = sorted(combined, key=compute_c5)
        population = combined
        for candidate in population:
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = c5
        grad = grad_c5(best_h)
        best_h -= grad * 0.01
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        best_h *= n_half / total

    return best_h, best_c5, n_points
