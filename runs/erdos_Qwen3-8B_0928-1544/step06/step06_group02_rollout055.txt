from scipy.optimize import minimize
from scipy.stats import norm

    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = np.zeros(n_points)

        initial_h = initial_h_values.copy()

            for i in range(n_points // 2):
                initial_h[i*2] = 1.0
    else:
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        # Perturb with smaller steps
        sigma = 0.05
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum constraint using linear interpolation
        def adjust_sum(h):
            current_sum = np.sum(h)
            if current_sum > target_sum:
                excess = current_sum - target_sum
                idx = np.argsort(h)
                for i in range(int(excess)):
                    h[idx[i]] -= 1e-5
            elif current_sum < target_sum:
                deficit = target_sum - current_sum
                idx = np.argsort(-h)
                for i in range(int(deficit)):
                    h[idx[i]] += 1e-5
            return h
        
        h_perturbed = adjust_sum(h_perturbed)
        
        # Local optimization with COBYLA for robustness
        result = minimize(
