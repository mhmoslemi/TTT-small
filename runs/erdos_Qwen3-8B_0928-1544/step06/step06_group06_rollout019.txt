from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140  # Increased number of points for better resolution
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Generate a simple periodic initial guess if no initial provided
        initial_h = np.sin(np.linspace(0, np.pi, n_points)) ** 2  # Sinusoidal shape
        initial_h = np.clip(initial_h, 0, 1)
        # Normalize to ensure the total is n_points / 2
        norm_factor = (n_points / 2) / np.sum(initial_h)
        initial_h *= norm_factor
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Generate initial population with more diversity
    initial_population = [initial_h.copy()]
    for _ in range(40):  # More initial guesses
        perturbed = initial_h + np.random.normal(0, 0.15, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    max_iter = 80  # Number of iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 100:
            break
        current_h = initial_population[np.random.randint(len(initial_population))]
        current_h = np.clip(current_h, 0, 1)
        
        # Prepare optimization
        initial_guess = current_h.copy()
        res = minimize(
            fun=objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[
                LinearConstraint(np.ones(n_points), [ones_count], [ones_count])
            ],
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
                initial_population.append(res.x)
                initial_population.pop(np.random.randint(len(initial_population)))
    
    return best_h, best_c5, n_points
