import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    start_time = time.time()
    best_h = None
    best_c5 = np.inf

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

            initial_h = initial_h * (target_sum / initial_h_sum)
        h = initial_h.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)

    def compute_c5(h):
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * (2.0 / n_points)

    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()

    T = 1.0
    cooling_rate = 0.9999
    steps = 0

    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum != 0:
            h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)

        new_c5 = compute_c5(h_perturbed)

        if new_c5 < current_c5:
            h = h_perturbed.copy()
            current_c5 = new_c5
        else:
            delta = new_c5 - current_c5
            prob = np.exp(-delta / T)

                h = h_perturbed.copy()
                current_c5 = new_c5

        if current_c5 < best_c5:
            best_h = h.copy()

        T *= cooling_rate
        steps += 1
