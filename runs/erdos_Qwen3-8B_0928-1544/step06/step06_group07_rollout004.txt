        if h_sum > 0:
            h = h * (target_sum / h_sum)
        initial_points.append(h)

    best_h = initial_points[0].copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()

    for initial in initial_points:
        h = initial.copy()
        for _ in range(200):
            if time.time() - time_start > budget_s:
                break
            overlaps = compute_overlaps(h)
            k = np.argmax(overlaps)
            grad = compute_gradient(h, k)
            h -= 0.01 * grad
            h = np.clip(h, 0.0, 1.0)
            current_sum = np.sum(h)
            if abs(current_sum - target_sum) > 1e-6:
                h = h * (target_sum / current_sum)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    return best_h, best_c5, n_points
