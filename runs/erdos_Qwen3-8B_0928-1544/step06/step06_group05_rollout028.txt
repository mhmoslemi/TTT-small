import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = 35
    
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.full(n_points, 0.5)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h += (ones_count - current_sum) / n_points
        h = np.clip(h, 0, 1)
    
    particles = [h.copy() for _ in range(50)]
    velocities = [np.zeros(n_points) for _ in range(50)]
    pbest = [p.copy() for p in particles]
    pbest_vals = [np.max(np.correlate(p, 1 - p, 'full')) * dx for p in particles]
    gbest = particles[0].copy()
    gbest_val = pbest_vals[0]
    
    w, c1, c2 = 0.7, 2.0, 2.0
    start_time = time.time()
    
    for _ in range(int(budget_s / 10)):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        for i in range(50):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            particles[i] += (ones_count - current_sum) / n_points

                pbest_vals[i] = val
