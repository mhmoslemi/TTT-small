from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    
    h = initial_h * (target_sum / np.sum(initial_h)) if np.sum(initial_h) > 0 else initial_h
    h = np.clip(h, 0.0, 1.0)
    
    def objective(h_flat):
        h_array = h_flat.reshape(n_points)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = Bounds(0.0, 1.0)
    best_h = h
    best_c5 = objective(h)
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        sigma = 0.2 * (1 - _ / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        def adjust_sum(h):
            while np.sum(h) > target_sum:
                idx = np.argmax(h)
                h[idx] -= 1e-5
            while np.sum(h) < target_sum:
                idx = np.argmin(h)
                h[idx] += 1e-5
            return h
        
        h_perturbed = adjust_sum(h_perturbed)
        
        result = minimize(

            x0=h_perturbed,
            method='Nelder-Mead',
