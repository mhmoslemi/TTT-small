from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        initial_h = np.full(n_points, 0.5)
    
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    time_used = 0
    generations = 300
    
    for gen in range(generations):
        start_time = time.time()
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        
        parent_indices = np.argsort(fitness)[:int(population_size * 0.3)]
        parents = [population[i] for i in parent_indices]
        
        offspring = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            mutate_rate = 0.2 * (1 - gen / generations)
            for i in range(n_points):
                if np.random.rand() < mutate_rate:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            child = child * (target_sum / np.sum(child))
            offspring.append(child)
        
        combined = population + offspring
        combined_fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points) for h in combined]
        sorted_indices = np.argsort(combined_fitness)
        population = [combined[i] for i in sorted_indices[:population_size]]
        
        for h in population:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        time_used += time.time() - start_time
        if time_used >= budget_s:
