import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    total_iterations = 0
    max_iterations = 50
    sigma = 0.25
    while time.time() - start_time < budget_s:
        total_iterations += 1
        if total_iterations > max_iterations:
            break
        # Generate perturbed initial guess through swapping values
        h_perturbed = initial_h.copy()
        if np.random.rand() < 0.5:
            idx = np.random.choice(n_points, 2, replace=False)
            h_perturbed[idx[0]] = np.clip(h_perturbed[idx[0]] + np.random.normal(0, sigma), 0.0, 1.0)
            h_perturbed[idx[1]] = np.clip(h_perturbed[idx[1]] + np.random.normal(0, sigma), 0.0, 1.0)
        else:
            idx = np.random.choice(n_points, 2, replace=False)
            h_perturbed[idx[0]] = np.clip(h_perturbed[idx[0]] + np.random.normal(0, sigma), 0.0, 1.0)
            h_perturbed[idx[1]] = np.clip(h_perturbed[idx[1]] - np.random.normal(0, sigma), 0.0, 1.0)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

                best_h = result.x
