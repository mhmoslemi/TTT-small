import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    pop_size = 50
    max_gen = 1000
    mutation_rate = 0.1
    tournament_size = 3

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
        population = [initial_h]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            population.append(h)
    else:
        population = [np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points))) for _ in range(pop_size)]

    for _ in range(max_gen):
        if time.time() > budget_s:
            break
        fitness = [evaluate_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < evaluate_c5(population[0]):
            population[0] = population[best_idx].copy()

        parents = []
        for _ in range(pop_size // 2):
            t = np.random.choice(pop_size, tournament_size, replace=False)
            winner = t[np.argmin([fitness[i] for i in t])]
            parents.append(population[winner])

        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[0]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum != 0:
                child *= target_sum / child_sum
            offspring.append(child)

        for j in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                offspring[j][idx] += np.random.uniform(-0.1, 0.1)
                offspring[j][idx] = np.clip(offspring[j][idx], 0, 1)
                offspring[j] = np.clip(offspring[j], 0, 1)
                child_sum = np.sum(offspring[j])
                if child_sum != 0:
                    offspring[j] *= target_sum / child_sum

        new_pop = population + offspring
        new_pop.sort(key=lambda x: evaluate_c5(x))
        population = new_pop[:pop_size]

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_points
