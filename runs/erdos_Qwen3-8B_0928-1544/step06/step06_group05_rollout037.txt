import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.full(n_points, 0.5)

        h = np.clip(h, 0, 1)
        delta = (target_sum - np.sum(h)) / n_points
        h += delta
        h = np.clip(h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = (target_sum - current_sum) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        return h
    
    def evaluate(h):
        return objective(h)
    
    def update_velocity(v, pbest, gbest, w=0.7, c1=2.0, c2=2.0):
        r1, r2 = np.random.rand(), np.random.rand()
        v = w * v + c1 * r1 * (pbest - h) + c2 * r2 * (gbest - h)
        return np.clip(v, -0.5, 0.5)
    
    particles = [h.copy() for _ in range(50)]
    pbest = [p.copy() for p in particles]
    gbest = min(particles, key=objective)
    best_c5 = objective(gbest)
    
    start_time = time.time()
