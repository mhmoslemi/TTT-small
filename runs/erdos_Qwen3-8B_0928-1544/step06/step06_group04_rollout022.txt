import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 50
    tournament_size = 3
    mutation_rate = 0.1
    generations = 200
    start_time = time.time()

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
        population = [initial_h]
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            population.append(h)

    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    for gen in range(generations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        new_population = []
        for _ in range(population_size):
            tournament = np.random.choice(population, size=tournament_size)
            tournament_fitness = [evaluate_c5(h) for h in tournament]
            winner = tournament[np.argmin(tournament_fitness)]
            new_population.append(winner)
        
        for i in range(0, population_size, 2):
            a = new_population[i]
            b = new_population[i + 1] if i + 1 < population_size else a
            mask = np.random.rand(n_points) < 0.5
            child1 = np.where(mask, a, b)
            child2 = np.where(mask, b, a)
            new_population[i] = child1
            new_population[i + 1] = child2
        
        for i in range(population_size):
            h = new_population[i]
