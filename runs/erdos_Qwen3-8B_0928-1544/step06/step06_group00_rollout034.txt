from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def cons(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    # Coarse genetic search with 20 segments
    if 'initial_h_values' not in globals():
        n_genetic = 20
        target_genetic = n_genetic // 2
        population = [np.random.rand(n_genetic) * (target_genetic / np.sum(np.random.rand(n_genetic))) for _ in range(30)]
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            fitness = np.array([obj(h) for h in population])
            elite = population[np.argmin(fitness)]
            new_pop = []
            for _ in range(len(population)):
                p1, p2 = np.random.choice(population, 2, replace=False)
                child = (p1 + p2) / 2
                if np.random.rand() < 0.1:

                    child += noise
                    child = np.clip(child, 0, 1)
                    child = child * (target_genetic / np.sum(child))
                new_pop.append(child)
            population = new_pop
        best_h = np.zeros(n_points)

            x_i = i * (2 / n_points)
            segment = int(x_i / (2 / n_genetic))

        best_h = best_h * (target_sum / np.sum(best_h))

    best_h = res.x

    return best_h, best_c5, n_points
