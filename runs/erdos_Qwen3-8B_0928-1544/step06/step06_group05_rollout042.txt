def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values

        initial_h = np.full(n_points, 0.5)

        initial_h += perturb
        initial_h = np.clip(initial_h, 0, 1)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        delta = (n_points / 2) - total
        h += delta / n_points

        return h
    
    def objective(h):

    def constraint_sum(h):

    def generate_initial_population(n_pop):
