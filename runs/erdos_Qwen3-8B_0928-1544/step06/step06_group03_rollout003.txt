import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points // 2
    n_pop = 30
    F = 0.8
    CR = 0.9
    max_gen = 1000
    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    start_time = time.time()

    # Initialize population
    pop = np.zeros((n_pop, n_points))
    if 'initial_h_values' in globals():
        pop[0] = initial_h_values
    else:
        pop[0] = np.random.rand(n_points)
    
    for i in range(n_pop):
        vec = pop[i].copy()
        vec = np.clip(vec, 0, 1)
        vec_sum = np.sum(vec)
        vec = vec * (required_sum / vec_sum)
        pop[i] = vec

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_pop):
            a, b, c = np.random.choice(n_pop, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = pop[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * (required_sum / trial_sum)
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
            if c5 < np.max(np.correlate(pop[i], 1 - pop[i], mode='full')) * (2.0 / n_points):
                pop[i] = trial

    best_h = np.clip(best_h, 1e-8, 1 - 1e-8)
    best_h = best_h * (required_sum / np.sum(best_h))
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    return best_h, best_c5, n_points
