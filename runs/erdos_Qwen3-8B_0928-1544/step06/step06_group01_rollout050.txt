import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = 1000
    perturb_freq = 10
    improve_threshold = 1e-5

    for iter in range(max_iter):
        start_time = time.time()
        if iter % perturb_freq == 0:
            sigma = 0.05 * (1 - iter / max_iter)
            h_perturbed = initial_h + np.random.normal(0, sigma, n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        else:
            h_perturbed = initial_h.copy()
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5 - improve_threshold:
            best_c5 = result.fun
            best_h = result.x
            
        initial_h = result.x
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
