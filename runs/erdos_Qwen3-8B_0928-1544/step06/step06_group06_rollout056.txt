from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values

        initial_h = np.random.rand(n_points)
        initial_h = np.sort(initial_h)

        initial_h[initial_h >= (ones_count - 1) / n_points] = 1
        initial_h[:ones_count] = 1
        initial_h[ones_count:] = 0
        initial_h = np.where(initial_h > 1, 1, initial_h)
        initial_h = np.where(initial_h < 0, 0, initial_h)

        # Adjust to ensure sum is correct
        initial_h = [1.0 if i < ones_count else 0.0 for i in range(n_points)]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
