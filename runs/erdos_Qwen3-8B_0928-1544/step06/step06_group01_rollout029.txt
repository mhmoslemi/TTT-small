import numpy as np
from numpy.random import rand, normal, choice
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    population = [initial_h.copy()]
    for _ in range(49):
        h = rand(n_points)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    np.random.shuffle(population)
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(initial_h)
    time_used = 0
    max_gen = 300
    for gen in range(max_gen):
        start_time = time.time()
        fit = np.array([evaluate_erdos_solution(h) for h in population])
        parents = []
        for _ in range(len(population) // 2):
            idxs = choice(len(population), 5, replace=False)
            winner = idxs[np.argmin(fit[idxs])]
            parents.append(population[winner])
        offspring = []
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i+1]
            child = (p1 + p2) / 2
            m_rate = 0.2 * (1 - gen / max_gen)
            for j in range(n_points):
                if np.random.rand() < m_rate:
                    child[j] += normal(0, 0.05)
                    child[j] = np.clip(child[j], 0, 1)
            child = child * (ones_count / np.sum(child))
            offspring.append(child)
        new_pop = []
        for _ in range(len(population)):
            if np.random.rand() < 0.1:
                new_pop.append(offspring.pop())
            else:
                new_pop.append(population.pop())
        population = new_pop
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    return best_h, best_c5, n_points
