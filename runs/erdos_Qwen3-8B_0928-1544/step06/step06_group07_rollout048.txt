        h[:n_points//2] = 1.0
        perturbation = np.random.normal(0, 0.1, n_points)
        h += perturbation
        h = np.clip(h, 0.0, 1.0)
        h *= required_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points, dx)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        overlaps = []
        for k in range(1, n_points):
            end = n_points - k
            overlap = np.sum(h[:end] * (1 - h[k:end + k])) * dx
            overlaps.append(overlap)
        worst_k = np.argmax(overlaps) + 1
        k = worst_k
        grad = np.zeros(n_points)
        grad[:n_points - k] += 1 - h[k:]
        grad[k:] -= h[:n_points - k]
        direction = -grad

        alpha = 0.1
        new_h = h + alpha * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scale = required_sum / new_sum if new_sum != 0 else 1.0
        new_h *= scale
        new_h = np.clip(new_h, 0.0, 1.0)

        h = new_h
        current_c5 = compute_c5(h, n_points, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points

def compute_c5(h, n_points, dx):
    overlaps = []
    for k in range(1, n_points):
        end = n_points - k
        overlap = np.sum(h[:end] * (1 - h[k:end + k])) * dx
        overlaps.append(overlap)
    return max(overlaps) if overlaps else 0.0
