    pop_size = 50
    gen_max = 350

    best_h = population[0].copy()

    time_used = 0
    gen = 0

    while gen < gen_max and time_used < budget_s:

            tournament = np.random.choice(pop_size, 3, replace=False)
            p1, p2 = tournament[np.argmin(fitness[tournament])], tournament[np.argmin(fitness[tournament])]
            cross = np.random.randint(1, n_points)
            c1 = np.concatenate((population[p1, :cross], population[p2, cross:]))
