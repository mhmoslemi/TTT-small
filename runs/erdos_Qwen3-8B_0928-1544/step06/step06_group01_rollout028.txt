import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 50
    tournament_size = 5
    max_generations = 200
    mutation_rate = 0.2
    best_h = None
    best_c5 = float('inf')
    time_used = 0
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Initialize population
    population = [initial_h]
    while len(population) < population_size:
        indices = np.random.choice(n_points, size=35, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Evaluate initial population
    for h in population:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    for gen in range(max_generations):
        start_time = time.time()
        # Evaluate all individuals
        fitness = [evaluate(h) for h in population]
        # Select parents with tournament selection
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population, size=tournament_size, replace=False)
            idx = np.argmin([evaluate(h) for h in tournament])
            parents.append(tournament[idx])
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            a = parents[i]
            b = parents[i+1] if i+1 < len(parents) else a
            child = (a + b) / 2
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            if sum_child > 0:
                child = child * (target_sum / sum_child)
            offspring.append(child)
        
        # Mutate
        mutated = []
        for child in offspring:
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    delta = np.random.normal(0, 0.05)
                    child[j] += delta
                    child[j] = np.clip(child[j], 0.0, 1.0)
            sum_child = np.sum(child)
            if sum_child > 0:
                child = child * (target_sum / sum_child)
            mutated.append(child)
        
        # Replace population
        combined = population + mutated
        combined.sort(key=lambda x: evaluate(x))
        population = combined[:population_size]
        
        # Update best solution
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
