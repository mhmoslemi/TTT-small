    s = n_points / 2.0
    pop_size = 50
    max_gen = 200
    mut_rate = 0.05
    elitism = 20
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        sum_rest = np.sum(initial_h[:-1])
        last = s - sum_rest
        initial_h = np.concatenate([initial_h[:-1], [last]])
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        sum_rest = np.sum(h[:-1])
        last = s - sum_rest
        if 0 <= last <= 1:
            pop.append(np.concatenate([h[:-1], [last]]))
    best_h = pop[0]
    best_c5 = float('inf')
    for _ in range(max_gen):
        fitness = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            fit = np.max(corr) * (2.0 / n_points)
            fitness.append(fit)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx].copy()
        parents = []
        for _ in range(pop_size // 2):
            t_indices = np.random.choice(pop_size, 5)
            best_t = t_indices[np.argmin([fitness[i] for i in t_indices])]
            parents.append(pop[best_t].copy())
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else p1
            child = (p1 + p2) / 2
            offspring.append(child)
        mutated = []
        for h in offspring:
            delta = np.random.normal(0, mut_rate, n_points)
            h += delta
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = s / sum_h if sum_h != 0 else 1.0
            h *= scale
            mutated.append(h)
        pop = mutated
        if elitism > 0:
            elite_idx = np.argsort(fitness)[:elitism]
            for idx in elite_idx:
                pop.append(pop[idx].copy())
            pop = pop[:pop_size]
        time_used = time.time()
        if time.time() - time_used >= budget_s:
            break
    return best_h, best_c5, n_points
