    while time.time() - time_start < budget_s:
        overlaps = compute_c5(best_h)
        worst_k = np.argmax(overlaps)
        grad = compute_gradient(best_h, worst_k)
        step_size = 0.01
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        delta = target_sum - current_sum
        if abs(delta) > 1e-6:
            new_h += delta / n_points
            new_h = np.clip(new_h, 0, 1)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return best_h, best_c5, n_points
