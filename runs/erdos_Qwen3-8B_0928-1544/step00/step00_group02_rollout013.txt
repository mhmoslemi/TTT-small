import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Initial setup
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    best_c5 = float('inf')
    best_h = h.copy()
    time_start = time.time()
    MAX_ITER = 200
    for _ in range(MAX_ITER):
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        if time.time() - time_start > budget_s:
            break
        # Find active k's
        max_corr = np.max(correlation)
        active_k_indices = np.where(np.abs(correlation - max_corr) < 1e-6)[0]
        # Compute gradient
        g = np.zeros(n_points)
        for idx in active_k_indices:
            k = idx - (n_points - 1)
            for j in range(n_points):
                j_plus_k = j + k
                term1 = 1.0 - h[j_plus_k] if 0 <= j_plus_k < n_points else 0.0
                j_minus_k = j - k
                term2 = h[j_minus_k] if 0 <= j_minus_k < n_points else 0.0
                g[j] += (term1 - term2) * dx
        # Solve LP
        h_var = cp.Variable(n_points)
        obj = cp.Maximize(cp.sum(g * h_var))
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            break
        if prob.status != 'optimal':
            break
        h_prime = h_var.value
        # Line search
        candidates = [0.0, 0.5, 1.0]
        min_c5 = float('inf')
        best_alpha = 0.0
        for a in candidates:
            h_candidate = h + a * (h_prime - h)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - n_points / 2) > 1e-6:
                h_candidate = h_candidate * (n_points / 2) / sum_candidate

            if c5 < min_c5:
                min_c5 = c5
                best_alpha = a
        h = h + best_alpha * (h_prime - h)
