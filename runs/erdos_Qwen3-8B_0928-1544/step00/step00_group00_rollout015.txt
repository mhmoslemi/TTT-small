import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = len(h)

    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    iterations = 0
    while time.time() - start < budget_s:
        iterations += 1
        grad = compute_gradient(h, dx, n_points)
        h -= 0.01 * grad
        h = project(h, n_points)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, dx, n_points):
    grad = np.zeros(n_points)
    eps = 1e-4
    for i in range(n_points):
        e = np.zeros(n_points)
        e[i] = 1.0
        h_plus = h + eps * e
        h_minus = h - eps * e
        h_plus = project(h_plus, n_points)
        h_minus = project(h_minus, n_points)
        c_plus = compute_c5(h_plus, dx)
        c_minus = compute_c5(h_minus, dx)
        grad[i] = (c_plus - c_minus) / (2 * eps)
    return grad

def project(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    sum_clipped = np.sum(h_clipped)
    if sum_clipped == 0:
        return np.zeros(n_points)
    scale = (n_points / 2) / sum_clipped
    return h_clipped * scale
