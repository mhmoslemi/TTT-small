    start_time = time.time()
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], options={'maxiter': 200, 'ftol': 1e-6})
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], options={'maxiter': 100})
    
    if result.success:
        best_h = result.x
    else:
        best_h = initial_h
    
    best_h = np.clip(best_h, 0.0, 1.0)
    adjustment = (n_points / 2 - np.sum(best_h)) * 1.0
    best_h[-1] += adjustment
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h[-1] = (n_points / 2 - np.sum(best_h[:-1])) * 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, objective(best_h), n_points
