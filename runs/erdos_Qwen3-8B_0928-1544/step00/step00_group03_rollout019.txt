        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h *= (n_points / 2) / h_sum
        initial_h = h.copy()

    population_size = 50

    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        # Generate a new candidate by mutating initial_h
        new_h = initial_h.copy()
        # Add random delta
        delta = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h += delta
        new_h = np.clip(new_h, 0, 1)
        # Scale to sum n/2
        new_h_sum = np.sum(new_h)
        if new_h_sum != 0:
            scale = (n_points / 2) / new_h_sum
            new_h *= scale
        population.append(new_h)
    best_c5 = np.inf
    best_h = initial_h.copy()
    start_time = time.time()

    for gen in range(1000):
        if time.time() - start_time > budget_s:
