import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        # Fallback: binary vector with n_points=47
        n_points = 47
        h_initial = np.zeros(n_points)
        h_initial[:n_points//2] = 1.0
    
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Objective function
    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Bounds for h: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Equality constraint: sum(h) == n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess
    h_initial = h_initial.astype(float)

    # Optimization with early stopping
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Handle optimization failure
    if not result.success:
        return (h_initial, objective(h_initial), n_points)

    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)

    # Enforce sum constraint
    sum_h = np.sum(h_opt)
    if abs(sum_h - n_points / 2) > 1e-6:

    # Final C5 value
