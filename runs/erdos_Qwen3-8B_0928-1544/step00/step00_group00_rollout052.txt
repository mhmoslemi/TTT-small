def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        target_sum = n_points / 2
        if current_sum > target_sum:
            excess = current_sum - target_sum
            indices = np.argsort(h_clipped)[::-1]
            for i in indices:
                if excess <= 0:
                    break
                delta = min(excess, h_clipped[i])
                h_clipped[i] -= delta
                excess -= delta
        elif current_sum < target_sum:
            deficit = target_sum - current_sum
            indices = np.argsort(h_clipped)
            for i in indices:
                if deficit <= 0:
                    break
                delta = min(deficit, 1 - h_clipped[i])
                h_clipped[i] += delta
                deficit -= delta
        return h_clipped

    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_particles = 100
    if initial_h is not None:
        initial_population = [initial_h.copy() for _ in range(n_particles // 2)]
        random_population = []
        for _ in range(n_particles - len(initial_population)):
            h = np.random.dirichlet(np.ones(n_points), 1)[0] * (n_points / 2)
            random_population.append(h)
        particles = initial_population + random_population
    else:
        particles = [np.random.dirichlet(np.ones(n_points), 1)[0] * (n_points / 2) for _ in range(n_particles)]

    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    gbest = min(pbest, key=lambda x: evaluate_erdos_solution(x))

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            x = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            w, c1, c2 = 0.8, 1.2, 1.2
            v_new = w * v + c1 * r1 * (pbest[i] - x) + c2 * r2 * (gbest - x)
            x_new = x + v_new
            x_projected = project_h(x_new)
            particles[i] = x_projected
            velocities[i] = v_new
            c5 = evaluate_erdos_solution(particles[i])
            if c5 < evaluate_erdos_solution(pbest[i]):
                pbest[i] = particles[i].copy()
            if c5 < evaluate_erdos_solution(gbest):
                gbest = particles[i].copy()
    best_c5 = evaluate_erdos_solution(gbest)
    return (gbest, best_c5, n_points)
