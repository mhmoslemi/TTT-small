    tabu_list = []
    tabu_tenure = 20
    start_time = time.time()
    while time.time() - start_time < budget_s:
        candidates = []
        for _ in range(5):
            perturb = np.random.normal(0, 0.1, n_points)
            new_h = h + perturb
            new_h = np.clip(new_h, 0, 1)
            sum_h = np.sum(new_h)
            scaling = (n_points / 2) / sum_h
            new_h = new_h * scaling
            candidates.append(new_h.copy())
        for candidate in candidates:
            c5 = evaluate_erdos_solution(candidate)
            is_in_tabu = False
            for t_h, t_c5 in tabu_list:
                if np.allclose(candidate, t_h, atol=1e-4):
                    is_in_tabu = True
                    break
            if not is_in_tabu:
                tabu_list.append((candidate.copy(), c5))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = candidate.copy()
        current_h = best_h.copy()
        ones_minus_h = 1 - current_h
        corr = np.correlate(current_h, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        corr *= dx
        max_k_index = np.argmax(corr)
        grad = np.zeros(n_points)
        k_max = max_k_index
        for j in range(n_points):
            if j <= n_points - 1 - k_max:
                grad[j] = (1 - current_h[j + k_max])
            elif j >= k_max:
                grad[j] = -current_h[j - k_max]
            else:
                grad[j] = 0.0
        learning_rate = 0.01
        new_h = current_h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        sum_h = np.sum(new_h)
        scaling = (n_points / 2) / sum_h
        new_h = new_h * scaling
        c5_new = evaluate_erdos_solution(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
        if len(tabu_list) >= tabu_tenure:
            tabu_list.pop(0)
    return best_h, best_c5, n_points
