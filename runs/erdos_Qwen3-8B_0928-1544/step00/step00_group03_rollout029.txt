import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            m_minus_k = (m - k) % n_points
            m_plus_k = (m + k) % n_points
            grad[m] = dx * (1.0 - h[m_minus_k] - h[m_plus_k])
        return grad

    cons = [{'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    options = {'maxiter': 200, 'ftol': 1e-7, 'disp': False}
    result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, options=options)

    if result.success:

        c5_bound = result.fun
    else:
        best_h = h
        c5_bound = objective(h)
