        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles[i, :] = h

    velocities = np.random.rand(swarm_size, n_points) * 0.01

    personal_bests = particles.copy()
    personal_best_fitness = np.zeros(swarm_size)
    for i in range(swarm_size):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        personal_best_fitness[i] = np.max(corr) * dx

    global_best = particles[0].copy()
    global_best_fitness = personal_best_fitness[0]

    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            inertia = 0.7
            cognitive = 2.0
            social = 2.0
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_bests[i] - particles[i]) + social * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            total = np.sum(particles[i])
            particles[i] = particles[i] * (n_points / 2) / total
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            new_fitness = np.max(corr) * dx
            if new_fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = new_fitness
                personal_bests[i] = particles[i].copy()
            if new_fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = new_fitness

    return (global_best, global_best_fitness, n_points)
