def normalize(h):
    total = np.sum(h)
    scale = (n_points / 2) / total
    return h * scale

def compute_c5(h, dx):
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, size):
    candidates = np.random.choice(len(population), size=size)
    best_idx = np.argmin([compute_c5(population[i], dx) for i in candidates])
    return population[candidates[best_idx]].copy(), population[np.random.choice(len(population), 1, replace=False)[0]].copy()

def crossover(p1, p2):
    mask = np.random.rand(len(p1)) < 0.5
    child1 = np.where(mask, p1, p2)
    child2 = np.where(~mask, p1, p2)
    return child1, child2

def mutate(h, rate, scale, n_points):
    mask = np.random.rand(len(h)) < rate
    h[mask] += np.random.normal(0, scale, sum(mask))
    h = np.clip(h, 0, 1)
    h = normalize(h)
    return h

def replace_population(old, new):
    combined = old + new
    combined.sort(key=lambda x: compute_c5(x, dx))
