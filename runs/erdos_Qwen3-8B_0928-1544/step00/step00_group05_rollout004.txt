import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h_values = evaluate_erdos_solution.initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    
    # Ensure initial h satisfies sum constraint
    if not np.isclose(np.sum(initial_h_values), sum_h):
        raise ValueError("Initial h does not satisfy sum constraint")
    
    def compute_c5(h):
        h = np.array(h)
        if not np.isclose(np.sum(h), sum_h):
            return np.inf
        cor = np.correlate(h, 1 - h, mode="full")
        return np.max(cor) * dx
    
    def perturb(h, idx1, idx2):
        h = np.array(h)
        h[idx1], h[idx2] = h[idx2], h[idx1]
        return h.tolist()
    
    # Initial solution
    best_h = np.array(initial_h_values).copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Local search with swaps
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        improved = False
        for i in range(n_points):
            for j in range(i + 1, n_points):
                if best_h[i] == 0 and best_h[j] == 1:
                    new_h = perturb(best_h, i, j)
                    new_c5 = compute_c5(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
                        improved = True
        if not improved:
            break
    
    # Additional fine-tuning with gradient descent
    def objective(x):
        return compute_c5(x)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, best_h, bounds=bounds, method='L-BFGS-B', tol=1e-6)
    best_h = result.x
    best_c5 = compute_c5(best_h)

        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
