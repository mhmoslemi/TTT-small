from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)

    best_c5 = np.inf
    best_h = h_initial
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_once(x0):
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 200})
        if time.time() - start_time > budget_s:
            return None
        if result.success and result.fun < best_c5:
            nonlocal best_c5, best_h
            best_c5 = result.fun
            best_h = result.x
        return result
    
    # Initial optimization
    result = optimize_once(h_initial)
    if result is None:
        return (best_h, best_c5, n_points)
    
    # Local improvements
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturb = np.random.uniform(-0.15, 0.15, n_points)
        perturb = np.clip(perturb, -0.15, 0.15)

        new_h = new_h - (np.sum(new_h) - n_points/2) / n_points
        result = optimize_once(new_h)
