        initial_h = np.random.rand(n_points)
    # Adjust initial_h to meet constraints
    h = np.clip(initial_h, 0, 1)
    S = np.sum(h)
    if np.abs(S - T) > 1e-6:
        if S < T:
            deficit = T - S
            indices = np.argsort(h)
            for i in indices:
                if h[i] >= 1:
                    continue
                add = min(deficit, 1 - h[i])
                h[i] += add
                deficit -= add
                if deficit <= 0:
                    break
        else:
            excess = S - T
            indices = np.argsort(-h)
            for i in indices:
                if h[i] <= 0:
                    continue
                sub = min(excess, h[i])
                h[i] -= sub
                excess -= sub
                if excess <= 0:
                    break
    initial_h = h
    n_particles = 100
    particles = [initial_h.copy() for _ in range(n_particles)]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    personal_best_vals = [float('inf')] * n_particles
    global_best_val = float('inf')
    global_best = initial_h.copy()
    start_time = time.time()
    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    max_iter = 500
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            v = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = inertia * v + cognitive * r1 * (particles[i] - particles[i]) + social * r2 * (global_best - particles[i])
            velocities[i] = v
            particles[i] += v
            # Project
            h = np.clip(particles[i], 0, 1)
            S = np.sum(h)
            if np.abs(S - T) > 1e-6:
                if S < T:
                    deficit = T - S
                    indices = np.argsort(h)
                    for idx in indices:
                        if h[idx] >= 1:
                            continue
                        add = min(deficit, 1 - h[idx])
                        h[idx] += add
                        deficit -= add
                        if deficit <= 0:
                            break
                else:
                    excess = S - T
                    indices = np.argsort(-h)
                    for idx in indices:
                        if h[idx] <= 0:
                            continue
                        sub = min(excess, h[idx])
                        h[idx] -= sub
                        excess -= sub
                        if excess <= 0:
                            break
            particles[i] = h
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i])
            if c5 < personal_best_vals[i]:
                personal_best_vals[i] = c5
                if c5 < global_best_val:
                    global_best_val = c5
                    global_best = particles[i].copy()
    return (global_best, global_best_val, n_points)
