def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    n_points = 45
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    delta = n_points / 2.0 - sum_h
    if abs(delta) > 1e-6:
        h += delta / n_points
    dx = 2.0 / n_points

    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def grad_objective(h_vec):
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            obj_plus = objective(h_plus)
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad_objective, tol=1e-10)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad_objective, tol=1e-10, options={'maxiter': 100})
    if elapsed > budget_s:
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad_objective, tol=1e-10, options={'maxiter': 50})
    if elapsed > budget_s:
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad_objective, tol=1e-10, options={'maxiter': 20})

    return (result.x, result.fun, n_points)
