        h = np.random.rand(n_points)
        h = h / h.sum() * (n_points / 2)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    
    p_best = [p.copy() for p in particles]
    g_best = particles[0].copy()
    
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        fft_h = np.fft.fft(h)
        fft_1mh = np.fft.fft(1.0 - h)
        cross_corr = np.fft.ifft(fft_h * fft_1mh).real
        return np.max(cross_corr) * dx
    
    best_c5 = compute_c5(particles[0])
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            v = velocities[i]
            r1, r2 = np.random.rand(2)
            v += w * v + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            v = np.clip(v, -velocity_clamp, velocity_clamp)
            
            new_h = particles[i] + v
            new_h = np.clip(new_h, 0, 1)
            total = np.sum(new_h)
            if total == 0:
                new_h = np.zeros(n_points) + 0.5
            else:
                new_h = new_h * (n_points / 2) / total
            particles[i] = new_h
            
            current_c5 = compute_c5(particles[i])
            if current_c5 < compute_c5(p_best[i]):
                p_best[i] = particles[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                g_best = particles[i].copy()
    
    return (g_best, best_c5, n_points)
