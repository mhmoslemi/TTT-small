import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 47
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.zeros(n_points)
