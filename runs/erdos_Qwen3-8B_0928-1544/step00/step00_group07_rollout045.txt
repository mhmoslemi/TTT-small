import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)

        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

    )

        h_opt = result.x
        c5 = objective(h_opt)
    else:
        h_opt = h_start
        c5 = objective(h_opt)
