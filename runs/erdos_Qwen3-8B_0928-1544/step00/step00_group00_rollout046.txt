            i = np.random.randint(n_points)
            p[i] += np.random.uniform(-0.1, 0.1)
        p = project(p, n_points)
        if any(np.allclose(p, t, atol=1e-2) for t in tabu):
            continue
        eps = 1e-5
        grad = np.zeros(n_points)
        c = compute_C5(p, 2.0 / n_points)
        for j in range(n_points):
            tmp = p.copy()
            tmp[j] += eps
            tmp = project(tmp, n_points)
            grad[j] = (compute_C5(tmp, 2.0 / n_points) - c) / eps
        new_h = p - 0.01 * grad
        new_h = project(new_h, n_points)
        nc5 = compute_C5(new_h, 2.0 / n_points)
        if nc5 < best_c5:
            best_c5 = nc5
            best_h = new_h.copy()
        tabu.append(new_h)
        if len(tabu) > tabu_len:
            tabu.pop(0)
        h = new_h
    return (best_h, best_c5, n_points)
