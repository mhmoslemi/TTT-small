            c2 = np.concatenate([p2[:idx], p1[idx:]])
            new_population.append(mutate(c1, target_sum_initial, n_points_initial))
            new_population.append(mutate(c2, target_sum_initial, n_points_initial))
        population = new_population
        current_fitness = np.array([fitness(ind) for ind in population])
        best_idx = np.argmin(current_fitness)
        best_h = population[best_idx].copy()
        best_fitness = current_fitness[best_idx]
    
    n_points_fine = 128
    target_sum_fine = n_points_fine / 2
    x_initial = np.linspace(0, 2, n_points_initial)
    x_fine = np.linspace(0, 2, n_points_fine)
    h_fine = np.interp(x_fine, x_initial, best_h)
    h_fine = h_fine * target_sum_fine / np.sum(h_fine)
    
    def objective(x):
        return compute_c5(x, n_points_fine)
    
    bounds = [(0, 1)] * n_points_fine
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_fine}
    res = minimize(objective, h_fine, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    if res.success:
        final_h = res.x
        c5 = res.fun
    else:
        final_h = h_fine
        c5 = compute_c5(final_h, n_points_fine)
    
    c5_coarse = best_fitness
    if c5 < c5_coarse:
        return (final_h.tolist(), c5, n_points_fine)
    else:
        return (best_h.tolist(), c5_coarse, n_points_initial)
