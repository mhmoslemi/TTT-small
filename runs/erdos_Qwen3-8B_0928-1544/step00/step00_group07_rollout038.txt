import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_fine = 128
    time_coarse = 300
    time_fine = 800

    def generate_initial(n):
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h, n):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n
        return np.max(corr) * dx

    def interpolate_h(h_coarse, n_coarse, n_fine):
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine = np.interp(x_fine, x_coarse, h_coarse)
        return h_fine

    # Initial setup

        h_coarse = initial_h_values.copy()

        h_coarse = generate_initial(n_coarse)

    # GA on coarse grid

    gen = 300
    tournament_size = 5
    elite = int(pop_size * 0.4)
    mut_std = 0.1

    population = [h_coarse.copy() for _ in range(pop_size)]
    best_h = h_coarse
    best_c5 = compute_c5(h_coarse, n_coarse)
    while time.time() - start < time_coarse and gen > 0:
        fitness = [compute_c5(ind, n_coarse) for ind in population]
        parents = []
        for _ in range(pop_size):
            t = np.random.choice(population, tournament_size)
            selected = t[np.argmin([compute_c5(ind, n_coarse) for ind in t])]
            parents.append(selected)
        offspring = []
        for i in range(0, pop_size, 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_coarse-1)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                offspring.append(child1)

            else:
                offspring.append(p1)
                offspring.append(p2)

            mu = np.random.normal(0, mut_std, n_coarse)
