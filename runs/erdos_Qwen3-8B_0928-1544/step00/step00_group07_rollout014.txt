        h = h * (n_points / 2) / np.sum(h)
        initial_h_values = h
    
    n_points_coarse = len(initial_h_values)
    n_points_fine = 128
    
    def evaluate_fitness(h):
        dx = 2.0 / len(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    population_size = 50
    max_generations = 200
    tournament_size = 5
    mutation_std = 0.1
    
    initial_h = initial_h_values.copy()
    population = [initial_h.copy() for _ in range(population_size)]
    
    for gen in range(max_generations):
        if gen % 10 == 0 and gen > 0:
            current_time = time.time()
            if current_time - start_time > budget_s:
                break
        
        fitnesses = [evaluate_fitness(ind) for ind in population]
        parents = []
        for _ in range(population_size):
            candidates = np.random.choice(population, tournament_size, replace=False)
            winner = min(candidates, key=evaluate_fitness)
            parents.append(winner)
        
        new_population = []
        for i in range(0, population_size, 2):
            parent1, parent2 = parents[i], parents[i+1]
            point = np.random.randint(1, len(parent1)-1)
            child1 = np.concatenate((parent1[:point], parent2[point:]))
            child2 = np.concatenate((parent2[:point], parent1[point:]))
            child1 += np.random.normal(0, mutation_std, len(child1))
            child2 += np.random.normal(0, mutation_std, len(child2))
            child1 = np.clip(child1, 0, 1)
            current_sum = np.sum(child1)
            child1 = child1 * (len(child1) / 2) / current_sum
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            current_sum = np.sum(child2)
            child2 = child2 * (len(child2) / 2) / current_sum
            child2 = np.clip(child2, 0, 1)
            new_population.extend([child1, child2])
        
        population = new_population
    
    best_h_coarse = min(population, key=evaluate_fitness)
    x_coarse = np.linspace(0, 2, n_points_coarse)
    x_fine = np.linspace(0, 2, n_points_fine)
    h_fine = np.interp(x_fine, x_coarse, best_h_coarse)
    sum_h = np.sum(h_fine)
    h_fine = h_fine * (n_points_fine / 2) / sum_h
    h_fine = np.clip(h_fine, 0, 1)
    
    bounds = [(0.0, 1.0) for _ in range(n_points_fine)]
    def objective(h):
        return evaluate_fitness(h)
    
    try:
        result = opt.minimize(objective, h_fine, method='L-BFGS-B', bounds=bounds)
    except:
        result = opt.minimize(objective, h_fine, method='BFGS', bounds=bounds)
    
    best_h_fine = result.x
    sum_h = np.sum(best_h_fine)
    best_h_fine = best_h_fine * (n_points_fine / 2) / sum_h
    best_h_fine = np.clip(best_h_fine, 0, 1)
    final_c5 = evaluate_fitness(best_h_fine)
    
    return best_h_fine, final_c5, n_points_fine
