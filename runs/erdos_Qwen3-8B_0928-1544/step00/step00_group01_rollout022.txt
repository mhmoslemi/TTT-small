        delta = target_sum - current_sum
        if delta != 0:
            delta_per_element = delta / n_points
            new_h += delta_per_element
            new_h = np.clip(new_h, 0, 1)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - best_c5) / T)
            if random.random() < prob:
                h = new_h.copy()
                best_c5 = new_c5
        T *= alpha
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (h, best_c5, n_points)
