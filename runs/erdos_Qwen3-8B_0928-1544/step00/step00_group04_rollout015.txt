                mut_mask = np.random.rand(n_points) < mutation_rate
                c1[mut_mask] += np.random.normal(0, mutation_scale, sum(mut_mask))
            c1 = np.clip(c1, 0, 1)
            c1 = c1 * (n_points / 2) / np.sum(c1)
            
            if np.random.rand() < mutation_rate:
                mut_mask = np.random.rand(n_points) < mutation_rate
                c2[mut_mask] += np.random.normal(0, mutation_scale, sum(mut_mask))
            c2 = np.clip(c2, 0, 1)
            c2 = c2 * (n_points / 2) / np.sum(c2)
            
            new_pop.append(c1)
            new_pop.append(c2)
        
        # Combine and select top candidates
        combined = population + new_pop
        fitness_combined = [evaluate_erdos_solution(h) for h in combined]
        sorted_indices = np.argsort(fitness_combined)
        population = [combined[i] for i in sorted_indices[:pop_size]]
        
        # Update best solution
        current_best = population[0]
        current_c5 = evaluate_erdos_solution(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
