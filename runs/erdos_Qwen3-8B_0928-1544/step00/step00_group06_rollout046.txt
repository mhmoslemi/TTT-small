    result = minimize(obj, h_initial, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_sum}], tol=1e-8)
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
