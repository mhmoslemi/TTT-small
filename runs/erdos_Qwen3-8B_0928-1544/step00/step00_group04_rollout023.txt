        initial_h = initial_h * (n_points / 2) / initial_sum
    
    population = [initial_h]
    for _ in range(49):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = initial_h
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    
    pop_size = 50
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h, n_points) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx]
        
        new_population = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(len(population), 5, replace=False)
            parent1 = population[np.argmin([fitness[i] for i in idxs])]
            idxs = np.random.choice(len(population), 5, replace=False)
            parent2 = population[np.argmin([fitness[i] for i in idxs])]
            
            mask = np.random.rand(n_points) < 0.5
            child1 = parent1 * mask + parent2 * (1 - mask)
            child2 = parent2 * mask + parent1 * (1 - mask)
            
            mutation = np.random.normal(0, 0.01, n_points)
            child1 += mutation
            child2 += mutation
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            
            sum_c1 = np.sum(child1)
            if sum_c1 > 0:
                child1 *= (n_points / 2) / sum_c1
            sum_c2 = np.sum(child2)
            if sum_c2 > 0:
                child2 *= (n_points / 2) / sum_c2
            
            new_population.extend([child1, child2])
        
        population = new_population
    
    return (best_h, best_c5, n_points)
