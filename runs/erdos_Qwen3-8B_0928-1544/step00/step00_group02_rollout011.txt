def run(seed=42, budget_s=1100):

        h *= (n_points / 2) / sum_h

    learning_rate = 1e-3

    step_count = 0
    time_start = time.time()
    no_improvement = 0
    
    while time.time() - time_start < budget_s and step_count < max_steps:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 20:
                break
        
        max_corr = np.max(corr)
        indices = np.where(np.isclose(corr, max_corr))[0]
        active_ks = [j - (n_points - 1) for j in indices]

        for k in active_ks:
            for i in range(n_points):
                i_plus_k = i + k
                if 0 <= i_plus_k < n_points:
                    gradient[i] += (1 - h[i_plus_k])
                i_minus_k = i - k
                if 0 <= i_minus_k < n_points:
                    gradient[i] -= h[i_minus_k]
        gradient *= dx  # Correct for the scaling in C5
