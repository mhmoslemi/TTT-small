    if abs(total - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / total
    conv = np.correlate(h, 1 - h, mode='full')
    current_max = conv.max()
    k_max_idx = np.argmax(conv)
    k_actual = k_max_idx - (n_points - 1)
    best_c5 = current_max * dx
    best_h = h.copy()

    for _ in range(1000):
        conv = np.correlate(h, 1 - h, mode='full')
        current_max = conv.max()
        k_max_idx = np.argmax(conv)
        k_actual = k_max_idx - (n_points - 1)
        gradient = np.zeros_like(h)
        for m in range(n_points):
            t1 = 1.0
            t2 = 0.0
            if 0 <= m + k_actual < n_points:
                t1 = 1.0 - h[m + k_actual]
            if 0 <= m - k_actual < n_points:
                t2 = h[m - k_actual]
            gradient[m] = t1 - t2
        h_new = h - 0.1 * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        total_new = np.sum(h_new)
        if total_new > 0:
            h_new = h_new * (n_points / 2) / total_new
        conv_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = conv_new.max() * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return (best_h.tolist(), best_c5, n_points)
