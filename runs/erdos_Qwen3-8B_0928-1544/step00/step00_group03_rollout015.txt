import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 70
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    swarm_size = 100
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        particles.append(initial_h.copy())
        velocities.append(np.zeros_like(initial_h))
    else:
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            particles.append(h)
            velocities.append(np.zeros_like(h))
    
    def compute_c5(h):
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    pbest = [np.copy(p) for p in particles]
    pbest_cost = [compute_c5(p) for p in particles]
    
    gbest = np.copy(particles[pbest_cost.index(min(pbest_cost))])
    gbest_cost = min(pbest_cost)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            current_h = particles[i]
            current_v = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            new_v = inertia_weight * current_v + c1 * r1 * (pbest[i] - current_h) + c2 * r2 * (gbest - current_h)
            new_h = current_h + new_v
            new_h_clipped = np.clip(new_h, 0.0, 1.0)
            sum_clipped = np.sum(new_h_clipped)
            if sum_clipped == 0:
                scale = 1.0
            else:
                scale = (n_points / 2) / sum_clipped
            new_h_normalized = new_h_clipped * scale
            velocities[i] = new_v
            particles[i] = new_h_normalized
        
        for i in range(swarm_size):
            current_h = particles[i]
            cost = compute_c5(current_h)
            if cost < pbest_cost[i]:
                pbest[i] = np.copy(current_h)
                pbest_cost[i] = cost
            if cost < gbest_cost:
                gbest = np.copy(current_h)
                gbest_cost = cost
    
    return (gbest.tolist(), gbest_cost, n_points)
