        for _ in range(20):
            p = random.choice(parents)
            mut_mask = np.random.rand(n_points) < 0.1
            p = p.copy()
            p[mut_mask] += np.random.uniform(-0.1, 0.1, np.sum(mut_mask))
            p_sum = np.sum(p)
            scaling = target_sum / p_sum
            p = p * scaling
            p = np.clip(p, 0, 1)
            mutated.append(p)
        
        # Combine and select next population
        new_pop = offspring + mutated
        new_pop.extend([population[i] for i in range(50) if i not in range(40)])
        new_pop = np.array(new_pop) if new_pop else population
        candidates = np.array(new_pop)
        
        # Evaluate all
        c5s = [evaluate_c5(h) for h in candidates]
        idx = np.argsort(c5s)
        population = candidates[idx[:50]]
        
        # Track best
        current_min = np.min(c5s)
        if current_min < best_c5:
            best_c5 = current_min
            best_h = candidates[np.argmin(c5s)]
    
    return (best_h, best_c5, n_points)
