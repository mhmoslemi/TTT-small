import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    best_h = h.copy()
    best_c5 = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = best_c5.max()
    
    start_time = time.time()
    
    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = corr.max()
        if c5 <= best_c5 + 1e-6:
            break
        
        max_val = c5
        active_ks = [k for k in range(len(corr)) if abs(corr[k] - max_val) < 1e-6]
        
        g = np.zeros(n_points)
        for k in active_ks:
            for m in range(n_points):
                if m + k < n_points:
                    g[m] += (1 - h[m + k])
                if m - k >= 0:
                    g[m] -= h[m - k]
        
        h_new = cp.Variable(n_points)
        constraints = [cp.sum(h_new) == required_sum, 0 <= h_new, h_new <= 1]
        objective = cp.Minimize(g @ h_new)
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status != 'OPTIMAL':
                break
            
            alpha_low, alpha_high = 0.0, 1.0
            for _ in range(50):
                alpha_mid = (alpha_low + alpha_high) / 2
                new_h = (1 - alpha_mid) * h + alpha_mid * h_new.value
                corr_new = np.correlate(new_h, 1 - new_h, mode='full') * dx
                c5_mid = corr_new.max()
                if c5_mid < corr_new[np.argmax(corr_new)]:
                    alpha_high = alpha_mid
                else:
                    alpha_low = alpha_mid
            
            new_h = (1 - alpha_high) * h + alpha_high * h_new.value
            corr_new = np.correlate(new_h, 1 - new_h, mode='full') * dx
            new_c5 = corr_new.max()
            if new_c5 < c5:
                h = new_h.copy()
                best_c5 = new_c5
                best_h = h.copy()
        except:
            break
    
    return (best_h, best_c5, n_points)
