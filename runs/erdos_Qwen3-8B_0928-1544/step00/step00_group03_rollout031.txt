from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_initial = initial_h_values.copy()
    else:
        n_points = 70
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / (np.sum(h_initial) / (n_points / 2.0))
    n_particles = 100
    swarm = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    if 'initial_h_values' in globals():
        swarm[0, :] = h_initial
        for i in range(1, n_particles):
            h = np.random.rand(n_points)
            h = h / (np.sum(h) / (n_points / 2.0))
            swarm[i, :] = h
    else:
        for i in range(n_particles):
            h = np.random.rand(n_points)
            h = h / (np.sum(h) / (n_points / 2.0))
            swarm[i, :] = h
    pbest = swarm.copy()
    gbest = swarm[0, :]
    best_c5 = np.inf
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_c5 = compute_c5(gbest)
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    time_start = time.time()
    iter_count = 0
    max_iter = 1000
    while time.time() - time_start < budget_s and iter_count < max_iter:
        iter_count += 1
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i, :] = omega * velocities[i, :] + c1*r1*(pbest[i, :] - swarm[i, :]) + c2*r2*(gbest - swarm[i, :])
            swarm[i, :] += velocities[i, :]
            swarm[i, :] = np.clip(swarm[i, :], 0.0, 1.0)
            sum_swarm = np.sum(swarm[i, :])
            if sum_swarm == 0:
                swarm[i, :] = np.zeros(n_points)
            else:
                swarm[i, :] = swarm[i, :] * (n_points / 2.0) / sum_swarm
        for i in range(n_particles):
            h = swarm[i, :]
            current_c5 = compute_c5(h)
            if current_c5 < compute_c5(pbest[i, :]):
                pbest[i, :] = h.copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = h.copy()
    return (gbest, best_c5, n_points)
