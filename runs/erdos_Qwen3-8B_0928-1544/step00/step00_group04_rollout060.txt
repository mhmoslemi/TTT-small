        n_points = 100
        h = np.ones(n_points) * 0.5

    # Ensure constraints
    if 'initial_h_values' in globals():
        if not np.all((h >= 0) & (h <= 1)):
            raise ValueError("Initial h values out of bounds")
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            raise ValueError("Initial sum not equal to n_points / 2")
    else:
        # Sum is already correct for uniform distribution
        pass

    dx = 2.0 / n_points

    def compute_c5(h_val):
        corr = correlate(h_val, 1 - h_val, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for _ in range(10000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            k = (i - lag) % n_points
            grad[i] = (1 - h[j]) - h[k]

        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)

        sum_h = np.sum(h_new)
        if sum_h == 0:
            h_new = h.copy()
        else:
            h_new = h_new * (n_points / 2) / sum_h

        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()
        else:
            h = h.copy()

        if np.abs(new_c5 - best_c5) < 1e-6:
            break

    return (best_h.tolist(), best_c5, n_points)
