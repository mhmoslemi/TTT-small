import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    np.random.seed(seed)
    
    def objective(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        x0=initial_h_values,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    if result.success:
        h_values = result.x.copy()
        c5_bound = result.fun
    else:
        h_values = initial_h_values.copy()
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5_bound = np.max(corr) * (2.0 / n_points)
    
    h_values = np.clip(h_values, 0.0, 1.0)
    return (h_values, c5_bound, n_points)
