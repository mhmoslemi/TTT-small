        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    p_best_c5 = np.array([compute_c5(p) for p in p_best])
    g_best_c5 = compute_c5(g_best)
    
    w, c1, c2, max_iter, vel_clamp = 0.7, 2.0, 2.0, 100, 0.2
    time_start = time.time()
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            velocities[i] = np.clip(velocities[i], -vel_clamp, vel_clamp)
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] *= (n_points / 2.0) / np.sum(particles[i])
            
            c5_candidate = compute_c5(particles[i])
            if c5_candidate < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = c5_candidate
            if c5_candidate < g_best_c5:
                g_best = particles[i].copy()
                g_best_c5 = c5_candidate
    
    best_h = g_best
    best_c5 = g_best_c5
    return (best_h, best_c5, n_points)
