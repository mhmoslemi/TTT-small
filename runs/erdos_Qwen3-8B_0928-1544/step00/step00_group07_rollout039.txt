        if h_sum == 0:
            return child
        scale = (n_points / 2) / h_sum
        return child * scale
    
    # Genetic algorithm setup
    pop_size = 50
    generations = 100
    population = [initial_h_coarse] + [generate_individual(n_points_coarse) for _ in range(pop_size - 1)]
    best_h = initial_h_coarse
    best_c5 = evaluate_c5(best_h, n_points_coarse)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(ind, n_points_coarse) for ind in population]
        top_indices = np.argsort(fitness)[:int(pop_size * 0.6)]
        top_parents = [population[i] for i in top_indices]
        new_population = []
        for i in range(0, len(top_parents), 2):
            if i + 1 < len(top_parents):
                p1, p2 = top_parents[i], top_parents[i+1]
                c1 = crossover(p1, p2)
                c1 = mutate(c1)
                new_population.append(c1)
                c2 = crossover(p2, p1)
                c2 = mutate(c2)
                new_population.append(c2)
            else:
                new_population.append(mutate(top_parents[i]))
        population = new_population
        current_best = min(population, key=lambda x: evaluate_c5(x, n_points_coarse))
        current_val = evaluate_c5(current_best, n_points_coarse)
        if current_val < best_c5:
            best_h = current_best
            best_c5 = current_val
    
    # Interpolate to fine grid
    x_coarse = np.linspace(0, 2, n_points_coarse)
    x_fine = np.linspace(0, 2, n_points_fine)
    h_fine = np.interp(x_fine, x_coarse, best_h)
    h_fine = h_fine / np.sum(h_fine) * (n_points_fine / 2)
    
    def objective(x):
        h = x.copy()
        dx = 2.0 / n_points_fine
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points_fine)]
    result = scipy.optimize.minimize(objective, h_fine, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    h_fine_opt = result.x
    h_fine_opt = h_fine_opt / np.sum(h_fine_opt) * (n_points_fine / 2)
    opt_c5 = result.fun
    
    # Select best solution
    if opt_c5 < best_c5:
        final_h = h_fine_opt
        final_c5 = opt_c5
        final_n = n_points_fine
    else:
        final_h = best_h
        final_c5 = best_c5
        final_n = n_points_coarse
    
    return (final_h, final_c5, final_n)
