            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.ones(n_points) * 0.5
            else:
                h = h * (n_points / 2) / h_sum
            particles.append(h)
    else:
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.ones(n_points) * 0.5
            else:
                h = h * (n_points / 2) / h_sum
            particles.append(h)
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = [p.copy() for p in particles]
    pbest_fitness = [evaluate_erdos_solution(p) for p in particles]
    gbest = particles[np.argmin(pbest_fitness)].copy()
    gbest_fit = np.min(pbest_fitness)
    iters = 0
    while time.time() - start_time < budget_s:
        iters += 1
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.7 * velocities[i] + 2 * r1 * (pbest[i] - particles[i]) + 2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            h_sum = np.sum(particles[i])
            if h_sum == 0:
                particles[i] = np.ones(n_points) * 0.5
            else:
                particles[i] = particles[i] * (n_points / 2) / h_sum
            fit = evaluate_erdos_solution(particles[i])
            if fit < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fit
            if fit < gbest_fit:
                gbest = particles[i].copy()
                gbest_fit = fit
    return (gbest, gbest_fit, n_points)
