    if not np.isclose(np.sum(h) * dx, 1.0):
        h = h * (1.0 / (np.sum(h) * dx))

    max_iterations = 100
    elapsed = 0
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    for _ in range(max_iterations):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        idx = np.argmax(corr)
        shift = idx - (n_points - 1)
        max_c5 = max_val * dx

        # Check if current C5 is better
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()

        # Skip if no improvement
        if max_c5 == best_c5:
            continue

        # Formulate QP to maximize cross-correlation for the selected shift
        h_var = cp.Variable(n_points)
        obj = cp.Maximize(cp.sum([h_var[i] * h_var[(i + shift) % n_points] for i in range(n_points)]) * dx)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]

        try:
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.SCS, max_iter=1000)
            h = h_var.value
        except:
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.ECOS, max_iter=1000)
            h = h_var.value

        # Check elapsed time and convergence
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    corr_final = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr_final) * dx
    return best_h, final_c5, n_points
