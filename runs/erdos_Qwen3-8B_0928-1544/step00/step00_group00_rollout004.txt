def project(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 1e-8:
        return h
    if current_sum > target_sum:
        # Reduce
        excess = current_sum - target_sum
        # Sort indices in descending order of values
        indices = np.argsort(h)[::-1]
        # Iterate through the indices and subtract from the values
        for i in indices:
            if h[i] == 0:
                continue
            # How much can we subtract?
            sub = min(excess, h[i])
            h[i] -= sub
            excess -= sub
            if excess == 0:
                break
    else:
        # Add
        deficit = target_sum - current_sum
        indices = np.argsort(h)
        for i in indices:
            if h[i] == 1:
                continue
            add = min(deficit, 1 - h[i])
            h[i] += add
            deficit -= add
            if deficit == 0:
                break
    return h
