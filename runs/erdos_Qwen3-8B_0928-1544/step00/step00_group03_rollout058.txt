import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 70
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 1
    # Ensure initial_h is valid
    assert np.all(initial_h >= 0) and np.all(initial_h <= 1)
    assert np.isclose(np.sum(initial_h), n_points / 2)
    
    pop_size = 50
    max_generations = 200
    mutation_rate = 0.1
    tournament_size = 10
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    def rescale(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros(n_points)
        scale = target_sum / current_sum
        h_rescaled = h * scale
        h_clamped = np.clip(h_rescaled, 0, 1)
        sum_clamped = np.sum(h_clamped)
        if sum_clamped < target_sum:
            delta = target_sum - sum_clamped
            idx = np.where(h_clamped < 1)[0]
            if idx.size > 0:
                h_clamped[idx[-1]] = 1
        return h_clamped
    
    best_fitness = float('inf')
    best_h = initial_h.copy()
    generation = 0
    
    while time.time() - start_time < budget_s:
        fits = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fits)
        best_fitness = fits[best_idx]
        best_h = population[best_idx].copy()
        
        new_pop = []
        while len(new_pop) < pop_size:
            t1 = random.sample(range(pop_size), tournament_size)
            t2 = random.sample(range(pop_size), tournament_size)
            p1, p2 = population[t1[np.argmin(fits[t1])]], population[t2[np.argmin(fits[t2])]]
            
            if random.random() < 0.8:
                cp = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:cp], p2[cp:]])
            else:
                child = p1.copy()
            
            if random.random() < mutation_rate:
                m_idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                child[m_idx] = np.random.rand(len(m_idx))
            
            child = rescale(child, n_points / 2)
            new_pop.append(child)
            if len(new_pop) >= pop_size:
                break
        
        population = new_pop
        generation += 1
        if generation > max_generations:
            break
    
    return best_h, best_fitness, n_points
