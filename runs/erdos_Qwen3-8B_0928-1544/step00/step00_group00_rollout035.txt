        h_var = cp.Variable(n_points)
        objective = cp.Maximize(0.5 * cp.quad_form(h_var, Q))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, max_iter=100)
            if prob.status == 'optimal':
                h = np.clip(h_var.value, 0, 1)
                # Ensure sum constraint
                total = np.sum(h) * dx
                if not np.isclose(total, 1.0):
                    delta = (1.0 - total) / n_points
                    h += delta * np.ones(n_points)
                    h = np.clip(h, 0, 1)
        
        except:
            pass
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
