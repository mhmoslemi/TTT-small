        rand_h = np.random.uniform(0, 1, n_coarse)
        rand_h = normalize_h(rand_h, target_sum_coarse)
        initial_population.append(rand_h)
    start_time = time.time()
    for gen in range(n_generations):
        fitness = [compute_c5(h, n_coarse) for h in initial_population]
        top_indices = np.argsort(fitness)[:5]
        new_population = []
        for _ in range(pop_size // 2):
            p1_idx, p2_idx = np.random.choice(top_indices, 2, replace=False)
            p1 = initial_population[p1_idx]
            p2 = initial_population[p2_idx]
            split = n_coarse // 2
            child1 = np.concatenate([p1[:split], p2[split:]])
            child2 = np.concatenate([p2[:split], p1[split:]])
            child1 += np.random.normal(0, mutation_std, n_coarse)
            child2 += np.random.normal(0, mutation_std, n_coarse)
            child1 = normalize_h(child1, target_sum_coarse)
            child2 = normalize_h(child2, target_sum_coarse)
            new_population.extend([child1, child2])
        initial_population = new_population
        fitness = [compute_c5(h, n_coarse) for h in initial_population]
        sorted_indices = np.argsort(fitness)
        initial_population = [initial_population[i] for i in sorted_indices[:pop_size]]
        if time.time() - start_time > budget_s:
            break
    best_coarse = initial_population[0]
    # Upsample to fine grid (128 points)
    n_fine = 128
    target_sum_fine = n_fine / 2
    upsampled = np.repeat(best_coarse, 2)
    # Gradient optimization
    def objective(x):
        h = x.reshape(n_fine)
        return compute_c5(h, n_fine)
    def constraint(x):
        h = x.reshape(n_fine)
        return np.sum(h) - target_sum_fine
    bounds = [(0, 1)] * n_fine
    x0 = upsampled
    result = minimize(objective, x0, bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, method='L-BFGS-B')
    best_fine = result.x
    c5_final = compute_c5(best_fine, n_fine)
    return best_fine, c5_final, n_fine
