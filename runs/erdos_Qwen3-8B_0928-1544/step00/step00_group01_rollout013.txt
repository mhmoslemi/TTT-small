import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    required_sum = n_points / 2.0
    current_sum = np.sum(initial_h_values)
    diff = required_sum - current_sum
    adjusted = np.array(initial_h_values, dtype=float)
    if np.abs(diff) > 1e-10:
        adjustment = diff / n_points
        adjusted += adjustment
        adjusted = np.clip(adjusted, 0.0, 1.0)
    dx = 2.0 / n_points
    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
        return np.max(corr) * dx
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - required_sum})
    best_c5 = float('inf')
    best_h = adjusted
    for _ in range(10):
        pert = 0.01 * np.random.randn(n_points)
        new_guess = np.clip(adjusted + pert, 0.0, 1.0)
        new_sum = np.sum(new_guess)
        diff = required_sum - new_sum
        if np.abs(diff) > 1e-10:
            adj = diff / n_points
            new_guess += adj
            new_guess = np.clip(new_guess, 0.0, 1.0)
        res = minimize(objective, new_guess, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            c5 = res.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x
    return (best_h.tolist(), best_c5, n_points)
