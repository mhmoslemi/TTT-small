    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 22.5 / n_points)
    current_h = initial_h
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx)
    start_time = time.time()
    tabu_list = []
    tabu_tenure = 20
    
    def perturb(h):
        h_perturbed = h.copy()
        n_perturb = np.random.randint(1, 5)
        indices = np.random.choice(len(h), size=n_perturb, replace=False)
        delta = np.random.uniform(-0.1, 0.1, size=n_perturb)
        h_perturbed[indices] += delta
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        total = np.sum(h_perturbed)
        if total == 0.0:
            return np.zeros_like(h_perturbed)
        factor = 22.5 / total
        h_perturbed *= factor
        return h_perturbed
    
    def is_tabu(h, tabu_list, tolerance=1e-4):
        for t in tabu_list:
            if np.all(np.abs(h - t) < tolerance):
                return True
        return False
    
    def compute_c5(h, dx):
        h_1_minus = 1 - h
        corr = correlate(h, h_1_minus, mode='full')
        return np.max(corr) * dx
    
    while time.time() - start_time < budget_s:
        perturbed_h = perturb(current_h)
        c5_perturbed = compute_c5(perturbed_h, dx)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = perturbed_h.copy()
        if not is_tabu(perturbed_h, tabu_list):
            tabu_list.append(perturbed_h)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        current_h = perturbed_h
    return (best_h, best_c5, n_points)
