        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        delta = target_sum - current_sum
        new_h[-1] += delta
        new_h = np.clip(new_h, 0, 1)
        new_c5 = compute_c5(new_h, 2.0 / n_points, n_points)
        if new_c5 < best_c5:
            current_h = new_h
            best_c5 = new_c5
        else:
            prob = np.exp((best_c5 - new_c5) / T)
            if np.random.rand() < prob:
                current_h = new_h
        T *= cooling_rate
    return (current_h, best_c5, n_points)
