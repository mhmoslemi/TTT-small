    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb current_h
        perturb_h = current_h + np.random.uniform(-0.05, 0.05, size=n_points)
        perturb_h = np.clip(perturb_h, 0, 1)
        sum_perturb = np.sum(perturb_h)
        if sum_perturb == 0:
            perturb_h = np.zeros(n_points)
        else:
            scale_perturb = (n_points / 2) / sum_perturb
            perturb_h = perturb_h * scale_perturb
        perturb_h = np.clip(perturb_h, 0, 1)
        
        # Tabu check
        is_tabu = False
        for t in tabu_list:
            if np.allclose(perturb_h, t, atol=1e-4):
                is_tabu = True
                break
        if is_tabu:
            continue
        
        # Compute C5
        c5 = compute_c5(perturb_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturb_h.copy()
            tabu_list.append(perturb_h)
            if len(tabu_list) > max_tabu_size:
                tabu_list.pop(0)
        
        current_h = perturb_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
