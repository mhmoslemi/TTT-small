import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    h_initial = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h_initial = np.clip(h_initial[:n_points], 0.0, 1.0)
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    result = minimize(

        h_initial, 
        method='COBYLA', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 500}

    if result.success:

        c5_bound = result.fun
