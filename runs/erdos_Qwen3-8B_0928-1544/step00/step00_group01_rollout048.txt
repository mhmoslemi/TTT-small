from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values

    dx = 2.0 / n_points
    
    def obj(h_flat):

        max_corr = np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    cons = [{'type': 'eq', 'fun': cons_func}]
    
    x0 = np.array(initial_h_values, dtype=float)

    result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
