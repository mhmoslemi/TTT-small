import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = np.array(globals().get('initial_h_values', [0.5]*47))
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    res = minimize(
        objective, initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
    )

    if res.success:
        h_values = res.x
        c5_bound = res.fun
    else:
        h_values = initial_h
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5_bound = np.max(corr) * dx

    h_values = np.clip(h_values, 0.0, 1.0)
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, n_points / 2):
        adj = (n_points / 2 - sum_h)
        h_values[-1] += adj
        h_values = np.clip(h_values, 0.0, 1.0)

    return h_values, c5_bound, n_points
