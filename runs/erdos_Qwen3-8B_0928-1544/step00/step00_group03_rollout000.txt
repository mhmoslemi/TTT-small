        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        if abs(sum_h_new) < 1e-10:
            h_new = np.zeros_like(h_new)
        else:
            h_new = h_new * (n_points / 2) / sum_h_new
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        elapsed = time.time() - time_start
        if elapsed > budget_s - 1e-2:
            break
        h = h_new
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return best_h, final_c5, n_points
