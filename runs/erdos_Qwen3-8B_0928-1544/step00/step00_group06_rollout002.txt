def run(seed=42, budget_s=1100, **kwargs):

        h = (np.random.rand(n_points) * 2 - 1) * 0.5
        h /= (h.sum() * dx)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def compute_c5(h):
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        corr *= dx
        return np.max(corr)
    
    def adjust_h(h, k_index):
        n = len(h)
        lag = k_index - (n - 1)
        overlapping = []
        for x in range(n):
            x_val = x * dx
            shifted_val = x_val + lag
            if 0 <= shifted_val <= 2:
                overlapping.append(x)
        if not overlapping:
            return h
        len_over = len(overlapping)
        non_overlapping = [i for i in range(n) if i not in overlapping]
        len_non = len(non_overlapping)
        if len_non == 0:
            return h
        # Compute current overlapping sum
        sum_over = np.sum(h[overlapping])
        sum_non = np.sum(h[non_overlapping])

        delta = 0.02
        # Decrease overlapping
        h[overlapping] -= delta
        # Increase non-overlapping
        h[non_overlapping] += (delta * len_over) / len_non
        # Clamp values to [0, 1]
        h = np.clip(h, 0, 1)
        # Renormalize to maintain sum
        current_sum = np.sum(h) * dx
        if abs(current_sum - 1.0) > 1e-6:
            scale = (1.0 / (current_sum * dx)) if current_sum != 0 else 1.0
            h *= scale
        return h
    
    for _ in range(5000):
