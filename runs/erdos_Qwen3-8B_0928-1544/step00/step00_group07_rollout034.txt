        s = np.sum(initial_h)
        scaling = target_sum / s
        initial_h = np.clip(initial_h * scaling, 0.0, 1.0)

    n_particles = 50
    particles = [initial_h.copy()]
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        s = np.sum(h)
        scaling = target_sum / s
        h = np.clip(h * scaling, 0.0, 1.0)
        particles.append(h.copy())
    
    velocities = [np.random.uniform(-0.2, 0.2, n_points) for _ in particles]
    personal_best = [h.copy() for h in particles]
    best_h = particles[0].copy()
    best_c5 = np.inf

    def evaluate_c5(h):
        weights = 1 - h
        corr = np.correlate(h, weights, mode='full')
        return np.max(corr) * dx

    def update_particle(h, v, pbest, global_best):
        r1, r2 = np.random.rand(), np.random.rand()
        v_new = 0.7 * v + 2*r1*(pbest - h) + 2*r2*(global_best - h)
        v_new = np.clip(v_new, -0.2, 0.2)
        h_new = h + v_new
        h_new = np.clip(h_new, 0.0, 1.0)
        s = np.sum(h_new)
        scaling = target_sum / s if s != 0 else 1.0
        h_new = np.clip(h_new * scaling, 0.0, 1.0)
        return h_new, v_new

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            h_new, v_new = update_particle(h, v, personal_best[i], best_h)
            particles[i] = h_new
            velocities[i] = v_new
            c_new = evaluate_c5(particles[i])
            if c_new < evaluate_c5(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = particles[i].copy()
            if c_new < best_c5:
                best_c5 = c_new
                best_h = particles[i].copy()
    return best_h, best_c5, n_points
