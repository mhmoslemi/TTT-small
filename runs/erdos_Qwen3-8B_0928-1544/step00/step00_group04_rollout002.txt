        h_clipped = np.clip(h_prelim, 0, 1)
        # Compute sum and scale
        sum_clipped = np.sum(h_clipped) * dx
        scale_factor = (n_points / 2) / sum_clipped if sum_clipped != 0 else 1.0
        h_new = h_clipped * scale_factor
        
        # Update h
        h = h_new.copy()
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute new c5
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = corr_new.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
        
        # Adjust step size
        eta *= 0.9995
    
    return (best_h, best_c5, n_points)
