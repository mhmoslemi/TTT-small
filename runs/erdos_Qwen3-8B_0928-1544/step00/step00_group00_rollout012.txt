def project_h(h, n_points):
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-8:
        return h
    if current_sum < target_sum:
        deficit = target_sum - current_sum
        indices = np.where(h < 1)[0]
        sorted_indices = np.argsort(h[indices])
        adjust = indices[sorted_indices]
        for idx in adjust:
            add = min(deficit, 1 - h[idx])
            h[idx] += add
            deficit -= add
            if abs(deficit) < 1e-8:
                break
    else:
        excess = current_sum - target_sum
        indices = np.where(h > 0)[0]
        sorted_indices = np.argsort(-h[indices])
        adjust = indices[sorted_indices]
        for idx in adjust:
            subtract = min(excess, h[idx])
            h[idx] -= subtract
            excess -= subtract
            if abs(excess) < 1e-8:
                break
