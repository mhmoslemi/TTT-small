import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 128
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.full(n_points, 0.5)

    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def equality_constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        fun=objective,
        x0=h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': equality_constraint}],
        options={'maxiter': 500, 'disp': False}
    )

    best_h = result.x
    best_c5 = result.fun

    if not (np.all(best_h <= 1) and np.all(best_h >= 0)):
        best_h = h_start
        best_c5 = objective(best_h)

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = h_start
        best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
