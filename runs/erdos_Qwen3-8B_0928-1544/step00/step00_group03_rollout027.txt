        h = np.random.uniform(0, 1, n_points)
        s = np.sum(h)
        h *= (n_points / 2) / s
        h = np.clip(h, 0, 1)
        pop.append(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    def crossover(a, b):
        split = np.random.randint(1, len(a))
        child = np.concatenate([a[:split], b[split:]])
        s = np.sum(child)
        scale = (n_points / 2) / s
        child *= scale
        child = np.clip(child, 0, 1)
        return child
    
    def mutate(h):
        h = h.copy()
        m = int(len(h) * mutation_rate)
        idxs = np.random.choice(len(h), size=m, replace=False)
        for i in idxs:
            h[i] += np.random.uniform(-0.1, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        s = np.sum(h)
        scale = (n_points / 2) / s
        h *= scale
        h = np.clip(h, 0, 1)
        return h
    
    for gen in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        fit_pop = [(compute_c5(h), h) for h in pop]
        fit_pop.sort()
        parents = []
        for _ in range(25):
            t = random.sample(fit_pop, 10)
            parents.append(min(t)[1])
        next_pop = []
        for i in range(20):
            a = parents[i % len(parents)]
            b = parents[(i+1) % len(parents)]
            child = crossover(a, b)
            child = mutate(child)
            next_pop.append(child)
        for _ in range(5):
            best = min(fit_pop)[1]
            next_pop.append(mutate(best))
        combined = pop + next_pop
        combined.sort()
        pop = [c[1] for c in combined[:50]]
        current_best = min(fit_pop)[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = min(fit_pop)[1]
    
    return (best_h, best_c5, n_points)
