        cross_corr = np.correlate(h, 1 - h, mode="full")
        return np.max(cross_corr) * dx
    
    def generate_individual(n_points):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    population = [generate_individual(n_coarse) for _ in range(30)]
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    for _ in range(100):
        fitness = np.array([compute_c5(ind) for ind in population])
        parents = []
        for _ in range(30):
            tournament = np.random.choice(30, 4, replace=False)
            best_tournament = tournament[np.argmin(fitness[tournament])]
            parents.append(population[best_tournament])
        
        new_population = []
        for i in range(0, 30, 2):
            if i + 1 < 30:
                parent1 = parents[i]
                parent2 = parents[i + 1]
                split = np.random.randint(1, n_coarse)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                new_population.append(child1)
                new_population.append(child2)
        
        for i in range(len(new_population)):
            child = new_population[i]
            perturb = np.random.normal(0, 0.1, n_coarse)
            child += perturb
            child = np.clip(child, 0, 1)
            total = np.sum(child)
            if total != 0:
                child = child * (n_coarse / 2) / total
            new_population[i] = child
        
        fitness_new = np.array([compute_c5(ind) for ind in new_population])
        population = new_population
        current_best = np.argmin(fitness_new)
        if fitness_new[current_best] < best_c5:
            best_h = population[current_best]
            best_c5 = fitness_new[current_best]
    
    n_fine = 128
    dx_fine = 2.0 / n_fine
    
    def upscale(h, new_n):
        x = np.linspace(0, 2, len(h))
        f = interp1d(x, h, kind='linear', bounds_error=False, fill_value=0)
        x_new = np.linspace(0, 2, new_n)
        h_upscaled = f(x_new)
        h_upscaled = np.clip(h_upscaled, 0, 1)
        total = np.sum(h_upscaled)
        if total != 0:
            h_upscaled = h_upscaled * (new_n / 2) / total
        return h_upscaled
    
    h_fine = upscale(best_h, n_fine)
    
    def obj_func(x):
        h = x.copy()
        cross_corr = np.correlate(h, 1 - h, mode="full")
        return np.max(cross_corr) * dx_fine
    
    bounds = [(0, 1) for _ in range(n_fine)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_fine / 2})
    
    result = minimize(obj_func, h_fine, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    
    final_h = result.x
    final_c5 = result.fun
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (best_h, best_c5, n_fine)
