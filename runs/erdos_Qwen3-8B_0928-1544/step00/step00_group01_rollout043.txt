        return max_corr * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraint = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    
    best_h = np.array(initial_h_values)
    best_c5 = objective(best_h)
    
    # Multi-start optimization
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        # Perturb initial guess
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturb = np.clip(initial_h_values + perturbation, 0.0, 1.0)
        # Adjust sum
        adjustment = (target_sum - np.sum(h_perturb)) / n_points
        h_perturb += adjustment
        # Optimize
        result = minimize(objective, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    elapsed = time.time() - start_time
    # Final adjustment to meet sum constraint
    final_sum = np.sum(best_h)
    adjustment = (target_sum - final_sum) / n_points
    best_h += adjustment
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return (best_h, best_c5, n_points)
