    cooling_rate = 0.995
    start_time = time.time()
    
    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(conv) * dx
    
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        delta_sum = sum_new - target_sum
        if delta_sum != 0:
            idx = np.random.randint(n_points)
            new_h[idx] += delta_sum
            new_h = np.clip(new_h, 0.0, 1.0)
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5:
            h = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            prob = np.exp(-(new_c5 - current_c5) / T)
            if np.random.rand() < prob:
                h = new_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        T *= cooling_rate
    return (best_h.tolist(), best_c5, n_points)
