import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = None

    population_size = 50
    population = []
    for _ in range(population_size):
        if initial_h is not None and np.random.rand() < 0.1:
            h = initial_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (ones_count / h_sum)
            else:
                h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
        population.append(h)

    best_h = population[0]
    best_c5 = float('inf')
    start_time = time.time()
    mutation_rate = 0.1
    selection_ratio = 0.2

    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(current_c5)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        num_keep = int(population_size * selection_ratio)
        top_indices = np.argsort(fitness)[:num_keep]
        top_h = [population[i] for i in top_indices]
        new_population = []
        for _ in range(population_size):
            parent = np.random.choice(top_h)
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                mutated = parent + noise
                mutated = np.clip(mutated, 0, 1)
                mutated_sum = np.sum(mutated)
                if mutated_sum != 0:
                    mutated = mutated * (ones_count / mutated_sum)
                else:
                    mutated = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            else:
                mutated = parent.copy()
            new_population.append(mutated)
        population = new_population

    return best_h, best_c5, n_points
