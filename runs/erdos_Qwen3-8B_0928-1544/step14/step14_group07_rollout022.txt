import numpy as np
import scipy.fft as fft
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = float('inf')
    alpha = 0.01
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        h_fft = fft.fft(h)
        g = 1 - h
        g_fft = fft.fft(g)
        corr_fft = h_fft * np.conj(g_fft)
        corr = fft.ifft(corr_fft).real
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        if k < 0:
            k += n_points
        shifted_forward = np.roll(h, -k)
        shifted_backward = np.roll(h, k)
        gradient = shifted_forward + shifted_backward
        gradient *= (2.0 / n_points)
        
        h -= alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (ones_count / sum_h)
        else:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        
        corr = fft.ifft(h_fft * np.conj(g_fft)).real
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, n_points
