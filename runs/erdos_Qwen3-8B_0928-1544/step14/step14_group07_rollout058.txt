    def gradient_func(hvec):
        corr = np.correlate(hvec, 1 - hvec, mode='full')
        k_max_lag = np.argmax(corr)
        n = len(hvec)
        k = k_max_lag - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            i_plus_k = i + k
            term1 = 1 - hvec[i_plus_k] if 0 <= i_plus_k < n else 0.0
            i_minus_k = i - k
            term2 = -hvec[i_minus_k] if 0 <= i_minus_k < n else 0.0
            grad[i] = term1 + term2
        return grad
    
    result = scipy.optimize.minimize(objective, h, method='trust-constr', jac=gradient_func, bounds=bounds, constraints=constraints)
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
