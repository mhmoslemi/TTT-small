import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.6

    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points) * 0.6

    h = h * (ones_count / h_sum)
    h = np.clip(h, 0, 1)

    def objective(h_flat):
        h = h_flat.copy()
        if np.sum(h) != ones_count:
            h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else h * (ones_count / 0.6)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h_flat):
        return np.sum(h_flat) - ones_count

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    def optimize():
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=1000)
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt_sum = np.sum(h_opt)
        if h_opt_sum == 0:
            h_opt = np.random.rand(n_points) * 0.6 * (ones_count / 0.6)
        else:
            h_opt = h_opt * (ones_count / h_opt_sum)
        c5 = objective(h_opt)
        return h_opt, c5

    best_h, best_c5 = optimize()
    elapsed = 0
    start = time.time()
    while elapsed < budget_s:
        new_h, new_c5 = optimize()
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
        elapsed = time.time() - start

    return best_h, best_c5, n_points
