    start_time = time.time()
    T = 1.0
    alpha = 0.99
    max_iter = 10000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        if not ones_indices.size or not zeros_indices.size:
            break
        
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        h_candidate = h.copy()
        h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
        
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * (2.0 / n_points)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        else:
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()
        
        T *= alpha
    
    return best_h, best_c5, n_points
