import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256  # Increased resolution for better precision
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / np.sum(h)) if np.sum(h) > 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    max_iter = 1000
    iter_count = 0
    alpha = 0.1
    decay = 0.999
    improvement_threshold = 1e-6
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        indices = np.where(np.isclose(corr, current_max, atol=1e-9))[0]
        k_values = indices - (n_points - 1)
        k_values = np.unique(k_values)
        
        grad = np.zeros(n_points)
        for k in k_values:
            if not np.isclose(k, 0):
                for i in range(n_points):
                    i_plus_k = (i + k) % n_points
                    i_minus_k = (i - k) % n_points
                    grad[i] += (1 - h[i_plus_k]) - h[i_minus_k]
        
        h += alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
        
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * (2.0 / n_points)
        
        if new_max < best_c5 - improvement_threshold:
            best_c5 = new_max
            best_h = h.copy()
            alpha *= 1.0
        else:
            alpha *= decay
        
        iter_count += 1
    
    return best_h, best_c5, n_points
