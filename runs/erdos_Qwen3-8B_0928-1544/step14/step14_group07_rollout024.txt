import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    ones_count = n_points / 2
    swarm_size = 100
    max_iterations = 1000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
        if len(h_initial) != n_points:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (ones_count / np.sum(h_initial))
        particles = [h_initial]
        velocities = [np.zeros(n_points)]
        personal_best_h = [h_initial.copy()]
        personal_best_c5 = [float('inf')]
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        initial_c5 = np.max(corr) * (2.0 / n_points)
        personal_best_c5[0] = initial_c5
    else:
        particles = []
        velocities = []
        personal_best_h = []
        personal_best_c5 = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * (ones_count / 0.5)
            else:
                h = h * (ones_count / h_sum)
            particles.append(h)
            velocities.append(np.zeros(n_points))
            corr = np.correlate(h, 1 - h, mode='full')
            initial_c5 = np.max(corr) * (2.0 / n_points)
            personal_best_h.append(h.copy())
            personal_best_c5.append(initial_c5)
    
    global_best_c5 = min(personal_best_c5)
    global_best_h = personal_best_h[personal_best_c5.index(global_best_c5)]
    
    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s and iterations < max_iterations:
        for i in range(len(particles)):
            h = particles[i].copy()
            velocity = velocities[i].copy()
            
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            
            if current_c5 < personal_best_c5[i]:
                personal_best_c5[i] = current_c5
                personal_best_h[i] = h.copy()
            
            if personal_best_c5[i] < global_best_c5:
                global_best_c5 = personal_best_c5[i]
                global_best_h = personal_best_h[i]
            
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocity = w * velocity + c1 * r1 * (personal_best_h[i] - h) + c2 * r2 * (global_best_h - h)
            
            h_new = h + velocity
            h_new = np.clip(h_new, 0.0, 1.0)
            h_sum_new = np.sum(h_new)
            if h_sum_new != 0:
                h_new = h_new * (ones_count / h_sum_new)
            else:
                h_new = np.random.rand(n_points) * (ones_count / 0.5)
            
            particles[i] = h_new
            velocities[i] = velocity
        
        iterations += 1
    
    best_h = personal_best_h[personal_best_c5.index(min(personal_best_c5))]
    best_c5 = min(personal_best_c5)
    
    return best_h, best_c5, n_points
