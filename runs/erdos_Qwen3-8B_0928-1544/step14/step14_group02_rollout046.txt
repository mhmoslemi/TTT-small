        if h_sum != 0:
            h = h * (ones_count / h_sum)
        new_c5 = compute_c5(h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            if iterations % 50 == 0:
                perturb = np.random.rand(n_points) * 0.15
                h += perturb
                h = np.clip(h, 0.0, 1.0)
                h_sum = np.sum(h)
                if h_sum != 0:
                    h = h * (ones_count / h_sum)
                new_c5 = compute_c5(h, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    conv = np.correlate(h, 1 - h, mode='full')
    max_conv = np.max(conv)
    dx = 2.0 / n_points
    return max_conv * dx
