import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        else:
            h = h * (target_sum / h_sum)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.1
    max_iterations = int(budget_s / 1)
    num_iterations = 0
    while num_iterations < max_iterations:
        # Compute cross correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        k_max = max_index - (n_points - 1)
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k_max < n_points:
                grad[i] += (1 - h[i + k_max]) * dx
            if 0 <= i - k_max < n_points:
                grad[i] -= h[i - k_max] * dx
        # Update h
        h = h - learning_rate * grad
        # Clip h
        h = np.clip(h, 0.0, 1.0)
        # Rescale to target sum
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        # Compute current C5
        current_c5 = np.max(cross_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Adapt learning rate and check for progress
        num_iterations += 1
        # Optionally perturb and adjust learning rate
        # For example, after a certain number of steps
        if num_iterations % 100 == 0:
            learning_rate *= 0.95
        # Check time budget
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points
