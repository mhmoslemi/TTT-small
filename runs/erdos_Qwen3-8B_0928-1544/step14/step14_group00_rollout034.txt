def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = 35
    population_size = 200
    w = 0.7
    c1 = 2.0
    c2 = 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (35.0 / total)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.5

    particles = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = initial_h + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points) * 0.5
        h = project(h)
        particles.append({'h': h, 'pbest': h.copy(), 'pbest_c5': objective(h)})

    gbest = particles[0]['h'].copy()
    gbest_c5 = particles[0]['pbest_c5']
    velocities = np.zeros((population_size, n_points))

    start_time = time.time()
    stagnation = 0
    prev_gbest_c5 = gbest_c5
    while time.time() - start_time < budget_s:
        # Process all particles in this iteration
        for i in range(population_size):
            current_h = particles[i]['h']
            current_c5 = objective(current_h)
            if current_c5 < particles[i]['pbest_c5']:

                particles[i]['pbest_c5'] = current_c5

                    gbest_c5 = current_c5

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    population_size = 200
    w = 0.7
    c1 = 2.0
    c2 = 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * (35.0 / total) if total != 0 else np.zeros_like(h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.5

    particles = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = initial_h + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points) * 0.5
        h = project(h)
        particles.append({'h': h, 'pbest': h.copy(), 'pbest_c5': objective(h)})

    gbest = particles[0]['h'].copy()
    gbest_c5 = particles[0]['pbest_c5']
    velocities = np.zeros((population_size, n_points))

    start_time = time.time()
    stagnation = 0
    prev_gbest_c5 = gbest_c5

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            current_h = particles[i]['h']
            current_c5 = objective(current_h)
            if current_c5 < particles[i]['pbest_c5']:
                particles[i]['pbest'] = current_h.copy()
                particles[i]['pbest_c5'] = current_c5
                if current_c5 < gbest_c5:
                    gbest = current_h.copy()
                    gbest_c5 = current_c5
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (particles[i]['pbest'] - current_h) + c2 * r2 * (gbest - current_h)
            new_h = current_h + velocities[i]
            new_h = project(new_h)
            particles[i]['h'] = new_h

        if gbest_c5 == prev_gbest_c5:
            stagnation += 1
        else:
            stagnation = 0
        prev_gbest_c5 = gbest_c5
        if stagnation > 50:
            w = max(w - 0.02, 0.4)
            stagnation = 0

    return gbest, gbest_c5, n_points
