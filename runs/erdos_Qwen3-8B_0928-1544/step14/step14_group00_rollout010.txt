import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = 35
    pop_size = 200
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    w_min = 0.4
    stagnation_limit = 100
    max_iter = 10000  # arbitrary, adjusts based on time

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.5
    particles = []
    velocities = []
    for _ in range(pop_size):
        h = initial_h.copy() + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Objective function
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Personal and global best
    pbest = [h.copy() for h in particles]
    pbest_values = [compute_c5(h) for h in pbest]
    gbest = pbest[0].copy()
    gbest_value = pbest_values[0]
    best_h = gbest.copy()
    best_c5 = gbest_value
    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = vel
            # Update position
            particles[i] += vel
            # Clip and normalize
            h = particles[i]
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            else:
                h = np.random.rand(n_points) * 0.5 * (target_sum / 0.5)
            particles[i] = h
            # Compute objective
            current_c5 = compute_c5(h)
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = h.copy()
                pbest_values[i] = current_c5
                if current_c5 < gbest_value:
                    gbest = h.copy()
                    gbest_value = current_c5
                    best_h = h.copy()
                    best_c5 = current_c5
                    stagnation = 0
            else:
                stagnation += 1
        # Adjust inertia weight
        if stagnation > stagnation_limit:
            w = max(w_min, w - 0.02)
            stagnation = 0

    return best_h, best_c5, n_points
