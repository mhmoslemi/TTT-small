import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    # Scale initial solution to satisfy the integral constraint
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def objective(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return float('inf')
        h_scaled = h_clipped * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Coarse search with multiple DE runs
    for i in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=50,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + i
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    # Local refinement with L-BFGS-B
    bounds_l_bfgs = [(0.0, 1.0) for _ in range(n_points)]
    result_l_bfgs = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds_l_bfgs,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if result_l_bfgs.fun < best_c5:
        best_h = result_l_bfgs.x
        best_c5 = result_l_bfgs.fun

    # Final checks and scaling
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum) if best_h_sum != 0 else initial_h_scaled
    final_c5 = objective(best_h_scaled)

    return best_h_scaled, final_c5, n_points
