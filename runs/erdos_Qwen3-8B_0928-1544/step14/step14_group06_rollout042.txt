import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    # Normalize initial solution to satisfy the integral constraint
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Reinforced differential evolution with adaptive mutation
    for _ in range(2):
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=200,
            popsize=80,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

        if time.time() - start_time > budget_s:
            break

    # Fine-tune with SLSQP and bound projection
    def constraint(h):
        return np.sum(h) - ones_count

    def project(h):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        return h * (ones_count / s) if s != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def refined_objective(h):
        return objective(project(h))

    result = opt.minimize(
        refined_objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points

    return result.x, objective(project(result.x)), n_points
