def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    target_sum = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points) * 0.6
        h_initial = h_initial * (target_sum / np.sum(h_initial))
    def obj(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.inf
        h_proj = h_clipped * (target_sum / sum_clipped)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = opt.differential_evolution(obj, bounds, maxiter=100, popsize=15, mutation=(0.5, 1.0), recombination=0.8, tol=1e-6, polish=True, seed=seed)
    best_h = result.x
    best_c5 = obj(best_h)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = opt.minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    best_h = res.x
    best_c5 = obj(best_h)
    return best_h, best_c5, n_points
