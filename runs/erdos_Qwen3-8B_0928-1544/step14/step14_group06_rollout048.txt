import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
    h_sum = np.sum(h)
    h = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    
    def objective(x):
        h = np.clip(x, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Coarse search with more restarts and varied parameters
    for restart in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=30,
            tol=1e-5,
            mutation=(0.6, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + restart
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        
        if time.time() - start_time > budget_s:
            break
    
    # Fine-tuning with L-BFGS-B
    def objective_with_bounds(x):
        x = np.clip(x, 0.0, 1.0)
        x_sum = np.sum(x)
        if x_sum == 0:
            return float('inf')
        x_scaled = x * (ones_count / x_sum)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds_lsq = Bounds(0.0, 1.0)
    result_lsq = opt.minimize(
        objective_with_bounds,
        best_h,
        method='L-BFGS-B',
        bounds=bounds_lsq,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    best_h = result_lsq.x
    best_c5 = objective_with_bounds(best_h)
    
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h_sum = np.sum(final_h)
    final_h_scaled = final_h * (ones_count / final_h_sum) if final_h_sum != 0 else best_h
    
    return final_h_scaled, objective_with_bounds(final_h_scaled), n_points
