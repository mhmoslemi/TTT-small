        f_fft = fft(ones_minus_h)
        corr_fft = h_fft * conjugate(f_fft)
        corr = ifft(corr_fft).real
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        shifted_forward = np.roll(h, lag)
        shifted_backward = np.roll(h, -lag)
        gradient = shifted_forward + shifted_backward
        # Update h
        h = h - alpha * gradient
        # Clip h
        h = np.clip(h, 0.0, 1.0)
        # Rescale to maintain sum(h) = ones_count
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        # Recompute cross-correlation after update
        h_fft = fft(h)
        ones_minus_h = 1 - h
        f_fft = fft(ones_minus_h)
        corr_fft = h_fft * conjugate(f_fft)
        corr = ifft(corr_fft).real
        current_c5 = np.max(corr) * (2.0 / n_points)
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Adjust alpha
        if current_c5 < best_c5:
            alpha *= alpha_factor
        else:
            alpha *= 0.95
        if best_c5 < 0.38085:
            break
        iteration += 1
    return best_h, best_c5, n_points
