import numpy as np
import time
import scipy.optimize as opt
import cma

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.6
    if np.sum(initial_h) == 0:
        initial_h = np.random.rand(n_points) * 0.6
    initial_h_scaled = initial_h * (ones_count / np.sum(initial_h))
    bounds = np.array([[0.0]*n_points, [1.0]*n_points])

    def obj_CMA(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h_scaled = h * (ones_count / sum_h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    es = cma.CMAES(obj_CMA, initial_h_scaled, bounds=bounds, popsize=30, sigma0=0.1)
    elapsed = 0
    while elapsed < budget_s * 0.7:
        if time.time() - start_time > budget_s:
            break
        es.optimize()
        elapsed = time.time() - start_time

    best_cma_h = es.result.x.copy()

    def obj_SLSQP(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h_scaled = h * (ones_count / sum_h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
    bounds_SLSQP = [(0.0, 1.0)] * n_points

    res = opt.minimize(obj_SLSQP, best_cma_h, method='SLSQP', bounds=bounds_SLSQP, constraints=cons, tol=1e-6)
    best_h = res.x
    best_c5 = obj_SLSQP(best_h)
    if np.isnan(best_c5):
        best_h = best_cma_h
        best_c5 = obj_SLSQP(best_h)

    return best_h, best_c5, n_points
