import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    
    # Ensure average is 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else initial_h * 0.5
    initial_h_scaled = np.clip(initial_h_scaled, 0.0, 1.0)
    
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return initial_h_scaled.copy()
        return h_clipped * (ones_count / h_sum)
    
    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Use differential evolution with adaptive mutation
    for _ in range(3):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(objective, bounds, strategy='best1bin', 
                                           maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1.0), 
                                           recombination=0.8, polish=True, disp=False, seed=seed + _)
        
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        
        if time.time() - start_time > budget_s:
            break
    
    # Final projection and evaluation
    best_h = project(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
