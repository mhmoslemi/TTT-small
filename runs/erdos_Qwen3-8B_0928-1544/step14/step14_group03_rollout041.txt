import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    def generate_random_h():
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h[:] = 0.5
        scale = ones_count / h_sum
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h
    
    population_size = 50
    population = []
    if 'initial_h_values' in globals():
        initial_candidates = [initial_h_values.copy() for _ in range(5)]

        population = [generate_random_h() for _ in range(population_size)]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        return max_overlap * (2.0 / n_points)
    
    best_c5 = float('inf')

    start_time = time.time()
