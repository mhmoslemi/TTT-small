import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    c1 = 2.0
    c2 = 2.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')

    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    gbest = None
    gbest_value = float('inf')

    for _ in range(population_size):
