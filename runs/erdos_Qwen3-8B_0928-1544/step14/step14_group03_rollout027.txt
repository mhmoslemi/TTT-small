import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    learning_rate = 0.05
    decay_rate = 0.999
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        max_index = np.argmax(corr)
        shift = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            term1 = (i + shift) % n_points
            term2 = (i - shift) % n_points
            grad[i] = (1 - h[term1]) - h[term2]
        
        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        
        learning_rate *= decay_rate
    
    return best_h, best_c5, n_points
