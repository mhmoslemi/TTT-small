import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)

        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.zeros_like(h) + (sum_required / n_points)
        scale = sum_required / h_sum
        return h_clipped * scale

        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
