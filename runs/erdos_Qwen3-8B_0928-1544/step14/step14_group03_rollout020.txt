        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j_plus_k = (i + k) % n_points
            j_minus_k = (i - k) % n_points
            grad[i] = (1 - h_vec[j_plus_k]) - h_vec[j_minus_k]
        grad *= dx
        return grad
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, constraints=constraint, tol=1e-6)
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
