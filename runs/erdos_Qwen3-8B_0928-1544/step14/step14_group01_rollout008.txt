import numpy as np
import time
import scipy.optimize as opt
import scipy.signal as signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    ones_count = n_points // 2

    # Load initial solution if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.6

    def normalize(h):
        h_sum = np.sum(h)
        return h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.6 * (ones_count / 0.6)

    pop = [normalize(initial_h)]
    for _ in range(49):
        h = np.random.rand(n_points) * 0.6
        if np.sum(h) == 0:
            h = np.random.rand(n_points) * 0.6
        pop.append(normalize(h))

    def obj(h):
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0]
    best_c5 = obj(best_h)
    start_time = time.time()

    # Global search
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        new_pop = []
        for _ in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            v = pop[a] + np.random.rand() * (pop[b] - pop[c])
            u = np.clip(v, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            new_pop.append(u * (ones_count / u_sum))
        pop = new_pop
        for h in pop:
            if obj(h) < best_c5:
                best_c5 = obj(h)
                best_h = h.copy()

    # Local refinement
    def slsqp_obj(h):
        return obj(h)

    def slsqp_cons(x):
        return np.sum(x) - ones_count

    bounds = [(0., 1.) for _ in range(n_points)]
    res = opt.minimize(slsqp_obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': slsqp_cons})
    best_h = res.x
    best_c5 = res.fun

    return best_h, best_c5, n_points
