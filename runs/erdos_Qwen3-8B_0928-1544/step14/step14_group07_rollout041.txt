import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2
    n_particles = 200
    if 'initial_h_values' in globals():

    else:
        particles = [np.random.rand(n_points) for _ in range(n_particles)]
        for p in particles:
            p *= (ones_count / np.sum(p))
    velocities = [np.random.rand(n_points) * 0.01 for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_values = [np.max(np.correlate(p, 1 - p, mode='full')) * (2.0 / n_points) for p in particles]
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    previous_global_value = global_best_value
    
    start_time = time.time()
    improvement_threshold = 1e-6
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_value = np.max(corr) * (2.0 / n_points)
            if current_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_value
        
        current_global_idx = np.argmin(personal_best_values)
        current_global_value = personal_best_values[current_global_idx]
        if current_global_value < global_best_value - improvement_threshold:
            global_best_value = current_global_value
            global_best = personal_best[current_global_idx].copy()
            previous_global_value = global_best_value
        
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (personal_best[i] - particles[i]) + 1.5 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]

            if sum_p != 0:

            else:
                particles[i] = np.random.rand(n_points) * (ones_count / 0.5)
        
        if abs(global_best_value - previous_global_value) < improvement_threshold:
            break
    
    return global_best, global_best_value, n_points
