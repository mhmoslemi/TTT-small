import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    # Initialize h from provided initial_h_values or random
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    h = initial_h
    # Normalize h to satisfy the integral constraint
    h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    alpha = 0.1
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        # Compute gradient for the maximum shift k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1.0 - h[i_plus_k]) - h[i_minus_k]
        # Update h to reduce the overlap for shift k
        h -= alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        # Ensure sum constraint is met
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
        # Check if improvement is achieved
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * (2.0 / n_points)
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
            alpha = 0.1
        else:
            alpha *= 0.97
        if alpha < 1e-6:
            break
    return best_h, best_c5, n_points
