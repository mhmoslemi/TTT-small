    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse_sum = np.sum(h_coarse)
    h_coarse = h_coarse * (n_coarse / 2) / h_coarse_sum if h_coarse_sum else h_coarse

    def obj_pso(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return float('inf')
        h_scaled = h * (n_coarse / 2) / np.sum(h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2 / n_coarse)

    n_particles = 50
    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    max_iter = 200
    elite = 10

    particles = np.array([h_coarse + np.random.rand(n_coarse) * 0.1 for _ in range(n_particles)])
    velocities = np.random.rand(n_particles, n_coarse) * 0.1
    pbest = particles.copy()
    pbest_vals = np.array([obj_pso(p) for p in particles])
    gbest = particles[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            curr_val = obj_pso(particles[i])
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
        curr_gbest = np.min(pbest_vals)
        if curr_gbest < gbest_val:
            gbest_val = curr_gbest
            gbest = pbest[np.argmin(pbest_vals)]
    top = pbest[np.argsort(pbest_vals)[:elite]]

    upscaled = []
    for p in top:
        x = np.interp(x_fine, x_coarse, p)
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum > 0:
            x = x * ones_count / x_sum
        upscaled.append(x)

    best_h = upscaled[0]
    best_c5 = float('inf')
    for h in upscaled:
        def obj_sqp(x):
            x = np.clip(x, 0, 1)
            if np.sum(x) == 0:
                return float('inf')
            x_scaled = x * ones_count / np.sum(x)
            corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return np.max(corr) * (2 / n_fine)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
        bnds = [(0, 1) for _ in range(n_fine)]
        res = opt.minimize(obj_sqp, h, method='SLSQP', bounds=bnds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum == 0:
        final_h = h
    else:
        final_h = final_h * ones_count / final_sum
    return final_h, best_c5, n_fine
