import time

def run(seed=42, budget_s=1100, **kwargs):

    swarm_size = 200
    particles = np.array([np.random.rand(n_points) * 2 for _ in range(swarm_size)])
    particles = particles * (ones_count / np.sum(particles, axis=1, keepdims=True))
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    pbest = particles.copy()
    pbest_values = np.full(swarm_size, np.inf)
    gbest = particles[0].copy()
    gbest_value = np.inf
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    start_time = time.time()
    no_improvement = 0
    max_iterations = 1000

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * (2.0 / n_points)
        return max_corr

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            h = particles[i]
            current_val = evaluate(h)
            if current_val < pbest_values[i]:
                pbest_values[i] = current_val
                pbest[i] = h.copy()

                gbest_value = current_val
                gbest = h.copy()
        for i in range(swarm_size):

            pbest_h = pbest[i]

        if np.abs(gbest_value - np.min(pbest_values)) < 1e-8:
