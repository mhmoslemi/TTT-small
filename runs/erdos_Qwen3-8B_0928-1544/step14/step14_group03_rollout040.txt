            h = np.random.rand(n_points)

            h = h * (target_sum / h_sum) if h_sum != 0 else np.random.rand(n_points) * (target_sum / 0.5)
            pop.append(h)
    # Compute C5
    def compute_c5(h):

        return np.max(corr) * 2.0 / n_points
    # Evaluate initial population
    scores = [compute_c5(h) for h in pop]
    best_idx = np.argmin(scores)
