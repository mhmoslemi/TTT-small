        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        population.append(h * (ones_count / h_sum))

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_low
    CR = CR_low
    stagnation = 0

    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[_]):
                population[_] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
                F = max(F_low, F - 0.01)
                CR = min(CR_high, CR + 0.01)
                stagnation = 0
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points
