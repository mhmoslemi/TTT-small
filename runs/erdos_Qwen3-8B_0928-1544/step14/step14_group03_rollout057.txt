import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 50
    mutation_rate = 0.1
    selection_ratio = 0.2
    dx = 2.0 / n_points

    def normalize(h):
        h_sum = np.sum(h)
        if h_sum != 0:
            return h * (ones_count / h_sum)
        return np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    if 'initial_h_values' in globals():
        population = [initial_h_values.copy()]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            population.append(normalize(h))
    else:
        population = [normalize(np.random.rand(n_points)) for _ in range(pop_size)]

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        scores = [compute_c5(h) for h in population]
        indices = np.argsort(scores)
        selected = [population[i] for i in indices[:int(selection_ratio * pop_size)]]

        new_pop = []
        for h in selected:
            mutated = h.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    mutated[i] += np.random.normal(0, 0.05)
            mutated = np.clip(mutated, 0, 1)
            mutated = normalize(mutated)
            new_pop.append(mutated)
        while len(new_pop) < pop_size:
            new_pop.append(normalize(np.random.rand(n_points)))
        
        population = new_pop
        current_c5 = compute_c5(population[0])
        if current_c5 < best_c5:
            best_h = population[0]
            best_c5 = current_c5

    return best_h, best_c5, n_points
