def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)

    # Scale initial solution to satisfy the integral constraint
    def repair(h_candidate):

        h_sum = np.sum(h_clipped)

        scale = (ones_count) / h_sum
        return h_clipped * scale

    def objective(h_candidate):
        h_repaired = repair(h_candidate)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * (2.0 / n_points)

        h = np.random.rand(n_points)

        if h_sum != 0:

    best_h = h.copy()

    try:
