            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    candidates = [(evaluate(h), h) for h in population]
    candidates.sort()
    best_c5, best_h = candidates[0]
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        selected = candidates[:num_keep]
        new_candidates = []
        for c5, h in selected:
            mutated = h.copy()
            noise = np.random.normal(0, mutation_std, size=n_points)
            mutated += noise
            mutated = np.clip(mutated, 0.0, 1.0)
            mutated_sum = np.sum(mutated)
            if mutated_sum != 0:
                mutated = mutated * (ones_count / mutated_sum)
            else:
                mutated = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            new_c5 = evaluate(mutated)
            new_candidates.append((new_c5, mutated))
        
        combined = [(c5, h) for c5, h in candidates] + new_candidates
        combined.sort()
        candidates = combined[:population_size]
        
        if candidates[0][0] < best_c5:
            best_c5 = candidates[0][0]
            best_h = candidates[0][1].copy()
    
    return best_h, best_c5, n_points
