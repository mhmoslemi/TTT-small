import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    alpha = 0.1

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        if max_corr <= best_c5:
            alpha *= 0.98
            continue
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h[j]) * (2.0 / n_points)
        h -= alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
