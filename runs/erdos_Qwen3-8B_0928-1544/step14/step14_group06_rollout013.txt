import numpy as np
import time
from scipy.optimize import Bounds, LinearConstraint, minimize, differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum == 0:
            initial_h = np.zeros(n_points) + 0.5
        else:
            initial_h = initial_h * ones_count / initial_h_sum
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * ones_count / h_sum
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5 * ones_count / 0.5
        return h_clipped * ones_count / h_sum
    
    def evaluate(h):
        h = repair(h)
        return objective(h)
    
    initial_h = repair(initial_h)
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    # Coarse search with differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=50,
        tol=1e-5,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Fine search with SQP
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    cons = LinearConstraint(np.ones(n_points), [ones_count], [ones_count])
    bounds_sqp = Bounds(0.0, 1.0)
    
    result_sqp = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds_sqp,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    best_h = result_sqp.x
    best_h = repair(best_h)
    final_c5 = evaluate(best_h)
    
    return best_h, final_c5, n_points
