import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / np.sum(h))
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        overlaps = []
        for k in range(n_points):
            roll_h = np.roll(h, k)
            overlap = np.sum(h * (1.0 - roll_h))
            overlaps.append(overlap)
        max_overlap = max(overlaps)
        c5 = max_overlap * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 50:
                perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                for i in perturb_indices:
                    h[i] += np.random.uniform(-0.05, 0.05)
                    h[i] = np.clip(h[i], 0.0, 1.0)
                current_sum = np.sum(h)
                if current_sum != 0:
                    scale = target_sum / current_sum
                    h = h * scale
                    h = np.clip(h, 0.0, 1.0)
                else:
                    h = np.random.rand(n_points) * 0.5
                    h = h * (target_sum / np.sum(h))
                stagnation = 0
        
        max_overlap_index = overlaps.index(max_overlap)
        k = max_overlap_index
        roll_h = np.roll(h, k)
        contributions = h * (1.0 - roll_h)
        top_percent = 0.1
        top_count = max(1, int(n_points * top_percent))
        sorted_indices = np.argsort(contributions)[::-1]
        top_indices = sorted_indices[:top_count]
        
        for i in top_indices:
            if h[i] > 0.5:
                h[i] = max(0.0, h[i] - 0.01)
            else:

            h = h * (target_sum / np.sum(h))
