            term1 = 0.0
            if 0 <= i_plus_k < n_points:
                term1 = 1.0 - h[i_plus_k]
            term2 = 0.0
            if 0 <= i_minus_k < n_points:
                term2 = h[i_minus_k]
            gradient[i] = term1 - term2
        
        h += alpha * (-gradient)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
        
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * (2.0 / n_points)
        
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
        
        if new_max < best_c5:
            alpha *= 1.01
        else:
            alpha *= 0.95
        
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, n_points
