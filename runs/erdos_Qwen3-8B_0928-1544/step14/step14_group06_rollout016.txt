import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70

        h_sum = np.sum(h)

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(5):  # Increased number of restarts
        current_h = best_h.copy()

        # Differential evolution parameters
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,

            maxiter=150,
            popsize=100,
            tol=1e-5,

            disp=False,

        )

        if result.fun < best_c5:

        if time.time() - start_time > budget_s:
            break

    # Local refinement with L-BFGS-B
    def objective_local(h, ones_count, n_points):
        h = np.clip(h, 0.0, 1.0)

        h_scaled = h * (ones_count / h_sum)
