import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = 35
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / h.sum()) if h.sum() != 0 else np.random.rand(n_points) * (target_sum / 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / h.sum()) if h.sum() != 0 else np.zeros_like(h) + target_sum / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[np.arange(n_points)]) * (2.0 / n_points)
    start_time = time.time()
    learning_rate = 0.01
    max_iterations_without_improvement = 0
    max_stagnation = 100
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        valid_indices = np.arange(n_points)
        k_max = valid_indices[corr[valid_indices].argmax()]
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k_max < n_points:
                term1 = 1.0 - h[j + k_max]
            else:
                term1 = 0.0
            if j >= k_max:
                term2 = -h[j - k_max]
            else:
                term2 = 0.0
            grad[j] = term1 + term2
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new_sum = h_new.sum()
        if h_new_sum == 0:
            h_new = np.zeros_like(h_new) + target_sum / n_points
        else:
            h_new = h_new * (target_sum / h_new_sum)
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new[valid_indices]) * (2.0 / n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            max_iterations_without_improvement = 0
        else:
            max_iterations_without_improvement += 1
            if max_iterations_without_improvement > max_stagnation:
                learning_rate *= 0.5
                max_iterations_without_improvement = 0
        h = h_new.copy()
    return best_h, best_c5, n_points
