def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    N = n_points
    sum_required = N / 2.0
    swarm_size = 200
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iterations = 1000
    
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values, dtype=np.float64)
        particles = [np.copy(initial) for _ in range(swarm_size)]
    else:
        particles = [np.random.rand(N) for _ in range(swarm_size)]
    
    # Scale initial particles
    for i in range(swarm_size):
        h = particles[i]
        h_sum = np.sum(h)
        if h_sum == 0:
            particles[i] = np.random.rand(N) * (sum_required / 0.5)
        else:
            particles[i] = h * (sum_required / h_sum)
    
    velocities = [np.zeros(N) for _ in range(swarm_size)]
    pbest = [np.copy(p) for p in particles]
    pbest_error = [0.0 for _ in range(swarm_size)]
    
    # Evaluate initial personal best
    for i in range(swarm_size):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        max_corr = np.max(corr)
        c5_val = max_corr * (2.0 / n_points)
        pbest_error[i] = c5_val
    
    gbest_index = np.argmin(pbest_error)
    gbest = pbest[gbest_index]
    gbest_error = pbest_error[gbest_index]
    best_h = gbest
    best_c5 = gbest_error
    
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate each particle
        for i in range(swarm_size):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5_val = max_corr * (2.0 / n_points)
            if c5_val < pbest_error[i]:
                pbest[i] = np.copy(h)
                pbest_error[i] = c5_val
        
        # Update global best
        current_min = np.min(pbest_error)
        if current_min < gbest_error:
            gbest_error = current_min
            gbest_index = np.argmin(pbest_error)
            gbest = pbest[gbest_index]
            best_h = gbest
            best_c5 = gbest_error
            if best_c5 < 0.38085:
                break
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1 = np.random.rand(N)
            r2 = np.random.rand(N)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            h_sum = np.sum(particles[i])
            if h_sum == 0:
                particles[i] = np.random.rand(N) * (sum_required / 0.5)
            else:
                particles[i] = particles[i] * (sum_required / h_sum)
    
    return best_h, best_c5, n_points
