    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points)

    # Normalize to ensure sum is correct
    h = h * (required_sum / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * 0.5 * (required_sum / 0.5)

    def compute_c5(h, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        shift_k = max_index - (n_points - 1)

        contributions = h * (1 - h[(np.arange(n_points) + shift_k) % n_points])
        top_indices = np.argsort(contributions)[::-1][:n_points // 10]

        for i in top_indices:
            if h[i] > 0.5:
                h[i] = max(0.0, h[i] - 0.1)
            else:
                j = (i + shift_k) % n_points
                h[j] = min(1.0, h[j] + 0.1)

        current_sum = np.sum(h)
        if current_sum != 0:
            scale = required_sum / current_sum
            h = h * scale
        else:
            h = np.random.rand(n_points) * 0.5 * (required_sum / 0.5)

        new_c5 = compute_c5(h, n_points)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 50:
            perturb_amount = 0.15
            perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[perturb_indices] += np.random.normal(0, perturb_amount, size=len(perturb_indices))
            h = np.clip(h, 0.0, 1.0)
            current_sum = np.sum(h)
            if current_sum != 0:
                scale = required_sum / current_sum
                h = h * scale
            else:
                h = np.random.rand(n_points) * 0.5 * (required_sum / 0.5)
            stagnation = 0

    return best_h, best_c5, n_points
