import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    def scale_h(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5
        return h * (ones_count / h_sum)

    initial_h_scaled = scale_h(initial_h)

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = scale_h(h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(3):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=150,
            popsize=70,
            tol=1e-6,
            mutation=(0.6, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
