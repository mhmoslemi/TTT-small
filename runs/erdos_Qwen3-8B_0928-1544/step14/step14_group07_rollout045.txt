        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        lag = max_corr_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            j_plus_lag = j + lag
            j_minus_lag = j - lag
            term1 = 1.0 - h[j_plus_lag] if 0 <= j_plus_lag < n_points else 0.0
            term2 = h[j_minus_lag] if 0 <= j_minus_lag < n_points else 0.0
            gradient[j] = (term1 - term2) * dx
        h_new = h - alpha * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            h_new = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h_new = h_new * (ones_count / h_new_sum)
        h = h_new
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max_overlap = np.max(new_corr) * dx
        if new_max_overlap < best_c5:
            best_c5 = new_max_overlap
            best_h = h.copy()
            alpha *= 1.05
            if alpha > 0.1:
                alpha = 0.1
        else:
            alpha *= 0.95
            if alpha < 0.001:
                alpha = 0.001
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, n_points
