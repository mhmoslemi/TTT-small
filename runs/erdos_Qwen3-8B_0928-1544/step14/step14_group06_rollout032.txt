    ones_count_coarse = n_points_coarse // 2
    ones_count_fine = n_points_fine // 2

    if 'initial_h_values' in globals():
        initial_h_coarse = initial_h_values
        initial_h_coarse_sum = np.sum(initial_h_coarse)
        initial_h_coarse_scaled = initial_h_coarse * (ones_count_coarse / initial_h_coarse_sum)
    else:
        initial_h_coarse = np.random.rand(n_points_coarse) * 0.5
        initial_h_coarse_sum = np.sum(initial_h_coarse)
        initial_h_coarse_scaled = initial_h_coarse * (ones_count_coarse / initial_h_coarse_sum)

    def objective_coarse(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count_coarse / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points_coarse)

    best_h_coarse = initial_h_coarse_scaled.copy()
    best_c5_coarse = objective_coarse(best_h_coarse)

    start_time = time.time()
    for _ in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points_coarse)]
        result = opt.differential_evolution(
            objective_coarse,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.7,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5_coarse:
            best_h_coarse = result.x
            best_c5_coarse = result.fun
        if time.time() - start_time > budget_s:
            break

    x_coarse = np.linspace(0, 2, n_points_coarse)
    x_fine = np.linspace(0, 2, n_points_fine)
    h_fine_initial = np.interp(x_fine, x_coarse, best_h_coarse)

    h_fine_initial_sum = np.sum(h_fine_initial)
    h_fine_initial_scaled = h_fine_initial * (ones_count_fine / h_fine_initial_sum)

    def objective_fine(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count_fine / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points_fine)

    best_h_fine = h_fine_initial_scaled.copy()
    best_c5_fine = objective_fine(best_h_fine)
    start_fine = time.time()

    for _ in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points_fine)]
        result = opt.differential_evolution(
            objective_fine,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=40,
            tol=1e-5,
            mutation=(0.3, 0.7),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _ + 5
        )
        if result.fun < best_c5_fine:
            best_h_fine = result.x
            best_c5_fine = result.fun
        if time.time() - start_fine > budget_s - (time.time() - start_time):
            break

    def objective_refined(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count_fine / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points_fine)

    best_h_fine = np.clip(best_h_fine, 0.0, 1.0)
    h_fine_sum = np.sum(best_h_fine)
    h_fine_scaled = best_h_fine * (ones_count_fine / h_fine_sum) if h_fine_sum != 0 else h_fine_initial_scaled

    result_refined = opt.minimize(
        objective_refined,
        h_fine_scaled,
        bounds=[(0.0, 1.0) for _ in range(n_points_fine)],
        method='L-BFGS-B',
        tol=1e-5,
        options={'maxiter': 100}
    )

    final_h = result_refined.x
    final_c5 = objective_refined(final_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    h_sum = np.sum(final_h)
    final_h_scaled = final_h * (ones_count_fine / h_sum) if h_sum != 0 else h_fine_initial_scaled

    return final_h_scaled, final_c5, n_points_fine
