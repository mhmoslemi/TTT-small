import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    def scale_h(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5
        return h * (ones_count / h_sum)

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h = scale_h(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = scale_h(initial_h)
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=100, popsize=50,
            mutation=(0.5, 1.0), recombination=0.8, tol=1e-5, polish=True,
            disp=False, seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    # Local refinement
    def lbfgs_objective(h):
        h = np.clip(h, 0.0, 1.0)
        h = scale_h(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    result = opt.minimize(
        lbfgs_objective, best_h, method='L-BFGS-B', bounds=bounds, jac=None,
        tol=1e-6, options={'maxiter': 100}
    )
    best_h = result.x
    best_c5 = lbfgs_objective(best_h)

    return best_h, best_c5, n_points
