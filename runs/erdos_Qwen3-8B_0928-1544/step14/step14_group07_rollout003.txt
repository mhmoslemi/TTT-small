import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (ones_count / np.sum(h_initial))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
