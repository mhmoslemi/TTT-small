import time

def run(seed=42, budget_s=1100, **kwargs):

    target_sum = n_points // 2
    population_size = 200
    w = 0.7
    c1, c2 = 2.0, 2.0
    max_iter = 5000
    stagnation = 0

        initial_h = np.random.rand(n_points) * 0.5

    velocities = np.zeros((population_size, n_points))
    pbest = particles[:]
    pbest_scores = np.array([evaluate(h, n_points) for h in pbest])
    gbest = pbest[np.argmin(pbest_scores)]
    gbest_score = np.min(pbest_scores)
    best_h = gbest.copy()
    best_c5 = float('inf')
    start_time = time.time()

            # Update position

            particles[i] = particles[i] * (target_sum / np.sum(particles[i])) if np.sum(particles[i]) != 0 else particles[i]

def evaluate(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
