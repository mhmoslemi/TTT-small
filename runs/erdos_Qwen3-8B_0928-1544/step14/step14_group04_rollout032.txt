    h = h * (required_sum / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * 0.5 * (required_sum / 0.5)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    alpha = 0.1
    start_time = time.time()
    dx = 2.0 / n_points
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        indices = np.where(corr == max_corr)[0]
        k_list = [(i - (n_points - 1)) for i in indices]
        grad = np.zeros(n_points)
        for k in k_list:
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] += (1 - h[j]) * dx
        grad /= len(k_list) if len(k_list) > 0 else 1
        h += alpha * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.random.rand(n_points) * 0.5 * (required_sum / 0.5)
        else:
            h *= required_sum / sum_h
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            alpha *= 1.1
        else:
            alpha *= 0.95
        if alpha < 1e-6:
            break
    return best_h, best_c5, n_points
