import numpy as np
import numpy.fft as fft
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = float('inf')
    alpha = 0.01
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # FFT-based cross-correlation
        H = fft.fft(h)
        F = fft.fft(1.0 - h)
        corr = fft.ifft(H * np.conj(F)).real
        max_overlap = np.max(corr) * (2.0 / n_points)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Determine lag with maximum overlap
        idx = np.argmax(corr)
        lag = idx - (n_points - 1)
        shifted_forward = np.roll(h, lag)
        shifted_backward = np.roll(h, -lag)
        gradient = shifted_forward + shifted_backward
        gradient *= (2.0 / n_points)
        
        # Update h
        h -= alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (ones_count / sum_h)
        else:
            h = np.random.rand(n_points) * (ones_count / 0.5)

        if new_c5 < best_c5:

            best_h = h.copy()
            alpha *= 1.05
        else:
            alpha *= 0.95
        
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, n_points
