import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    h = h * (ones_count / h_sum) if h_sum != 0 else h
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    alpha = 0.1
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        h += alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * (2.0 / n_points)
        if new_max < best_c5:

            best_h = h.copy()
            alpha = 0.1
        else:

        if alpha < 1e-6:
