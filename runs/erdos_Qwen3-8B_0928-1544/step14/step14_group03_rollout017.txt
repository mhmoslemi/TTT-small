    ones_count = n_points // 2
    population_size = 50
    std_dev = 0.05
    replacement_ratio = 0.8
    replace_count = int(population_size * replacement_ratio)

    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        for _ in range(5):
            h = initial_h.copy()
            h_sum = np.sum(h)
            scale = (n_points / 2) / h_sum if h_sum != 0 else 1.0
            h = h * scale
            h = np.clip(h, 0.0, 1.0)
            population.append(h)
        for _ in range(population_size - 5):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            scale = (n_points / 2) / h_sum if h_sum != 0 else 1.0
            h = h * scale
            h = np.clip(h, 0.0, 1.0)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            scale = (n_points / 2) / h_sum if h_sum != 0 else 1.0
            h = h * scale
            h = np.clip(h, 0.0, 1.0)
            population.append(h)

    best_h = population[0].copy()
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s:
        evaluations = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            evaluations.append(c5)
        
        best_index = np.argmin(evaluations)
        if evaluations[best_index] < best_c5:
            best_c5 = evaluations[best_index]
            best_h = population[best_index].copy()
        
        sorted_indices = np.argsort(evaluations)
        top_candidates = [population[i] for i in sorted_indices[:population_size // 5]]
        new_candidates = []
        for h in top_candidates:
            for _ in range(5):
                noise = np.random.normal(0, std_dev, n_points)
                h_mutated = h + noise
                h_mutated = np.clip(h_mutated, 0.0, 1.0)
                h_sum = np.sum(h_mutated)
                if h_sum == 0:
                    h_mutated = np.full(n_points, 0.5)
                else:
                    scale = (n_points / 2) / h_sum
                    h_mutated = h_mutated * scale
                new_candidates.append(h_mutated)
        new_candidates = new_candidates[:replace_count]
        for i in range(replace_count):
            population[sorted_indices[-replace_count + i]] = new_candidates[i]
    
    return best_h, best_c5, n_points
