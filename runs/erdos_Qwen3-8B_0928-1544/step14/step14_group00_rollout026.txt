import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    population_size = 200
    inertia = 0.7
    c1, c2 = 2.0, 2.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    population = []
    velocities = []
    pbest = []
    for _ in range(population_size):
        h = initial_h + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)
        velocities.append(np.zeros(n_points))
        pbest.append(h_scaled.copy())
    gbest = population[0].copy()
    best_c5 = np.inf
    for i in range(population_size):
        c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
            gbest = population[i].copy()
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] * inertia + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            velocities[i] = v
            population[i] += v
            population[i] = np.clip(population[i], 0, 1)
            h_sum = np.sum(population[i])
            if h_sum == 0:
                population[i] = np.random.rand(n_points) * 0.5
            else:
                population[i] = population[i] * (ones_count / h_sum)
            current_c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx
            if current_c5 < np.max(np.correlate(pbest[i], 1 - pbest[i], mode='full')) * dx:
                pbest[i] = population[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = pbest[i].copy()
    return gbest, best_c5, n_points
