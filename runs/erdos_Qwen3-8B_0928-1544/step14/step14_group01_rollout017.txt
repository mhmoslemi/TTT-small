            es.tell(x, cost)
            if cost < best_c5:
                best_c5 = cost
                best_h = x.copy()
        if time.time() - start_time >= budget_s:
            break

    def objective_slsqp(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-6)
    best_h = result.x
    best_c5 = objective_slsqp(best_h)

    return best_h, best_c5, n_points
