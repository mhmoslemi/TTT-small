import numpy as np

import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    # Initialize with provided initial_h_values
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:ones_count] = 1.0  # Sparse initial guess
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points)
    h = h * (ones_count / h_sum)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints
    def constraint_sum(h):
        return np.sum(h) - ones_count

    cons = {'type': 'eq', 'fun': constraint_sum}

    # Bounds
    bounds = Bounds(0.0, 1.0)

    # Optimization with L-BFGS-B
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 500, 'ftol': 1e-7})
    best_h = result.x
    best_c5 = result.fun

    return best_h, best_c5, n_points
