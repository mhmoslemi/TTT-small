import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 50
    mutation_rate = 0.1
    selection_ratio = 0.2
    num_selection = int(pop_size * selection_ratio)
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    initial_h = None

    if 'initial_h_values' in globals():
        initial_h = initial_h_values

    def init_population():
        pop = []
        if initial_h is not None:
            for _ in range(5):
                h = initial_h.copy()
                h_sum = np.sum(h)
                if h_sum != 0:
                    h = h * (ones_count / h_sum)
                pop.append(h)
        while len(pop) < pop_size:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (ones_count / h_sum)
            pop.append(h)
        return [np.array(h) for h in pop]

    population = init_population()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            current_c5s.append(c5)
        
        sorted_indices = np.argsort(current_c5s)
        selected_indices = sorted_indices[:num_selection]

        new_pop = []
        for idx in selected_indices:
            h = population[idx].copy()

            h += noise

            if h_sum != 0:
                h = h * (ones_count / h_sum)
            new_pop.append(h)

        non_selected = sorted_indices[num_selection:]
        non_selected = non_selected[:pop_size - num_selection]
        for idx in non_selected:
            new_pop.append(population[idx].copy())

        population = new_pop

        current_c5s = []
