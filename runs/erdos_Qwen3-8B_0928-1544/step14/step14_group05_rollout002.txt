import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    n = n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n) * 0.5
    h = initial_h.copy()
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n) * 0.5 * (ones_count / 0.5)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)
    start_time = time.time()
    learning_rate = 0.1
    iter_no_improve = 0
    max_iter_no_improve = 100

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        for x in range(n):

        h = h - learning_rate * grad

        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)

            h = np.random.rand(n) * 0.5 * (ones_count / 0.5)
        current_c5 = np.max(corr) * (2.0 / n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
