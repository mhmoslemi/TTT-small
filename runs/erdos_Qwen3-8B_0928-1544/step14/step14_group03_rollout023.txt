import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = candidate_h.copy()
            current_c5 = candidate_c5
