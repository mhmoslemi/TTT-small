            h_clip = np.clip(particles[i], 0.0, 1.0)
            h_sum = np.sum(h_clip)
            if h_sum == 0:
                continue
            h_scaled = h_clip * (ones_count / h_sum)
            particles[i] = h_scaled
            current_c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * (2.0 / n_points)
            if current_c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_c5
                if current_c5 < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_c5
    
    h = gbest.copy()
    learning_rate = 0.01
    max_steps = 1000
    for step in range(max_steps):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0.0, 1.0)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum == 0:
                continue
            h_plus_scaled = h_plus * (ones_count / h_plus_sum)
            c_plus = np.max(np.correlate(h_plus_scaled, 1 - h_plus_scaled, mode='full')) * (2.0 / n_points)
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0.0, 1.0)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum == 0:
                continue
            h_minus_scaled = h_minus * (ones_count / h_minus_sum)
            c_minus = np.max(np.correlate(h_minus_scaled, 1 - h_minus_scaled, mode='full')) * (2.0 / n_points)
            grad[i] = (c_plus - c_minus) / (2 * epsilon)
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h = h * (ones_count / h_sum)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if new_c5 < gbest_val:
            gbest_val = new_c5
            gbest = h.copy()
        elif abs(new_c5 - gbest_val) < 1e-6:
            break
    
    return gbest, gbest_val, n_points
