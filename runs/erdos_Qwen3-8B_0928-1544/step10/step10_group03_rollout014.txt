import numpy as np
import time

def compute_overlap_integral(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_total_gradient(h, n_points, dx):
    gradient = np.zeros(n_points)
    k_values = np.arange(-n_points + 1, n_points)
    for k in k_values:
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j]) * dx
                gradient[j] -= h[i] * dx
    return gradient
