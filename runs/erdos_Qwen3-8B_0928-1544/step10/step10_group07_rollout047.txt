    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=30, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        tol=1e-8, 
        maxiter=200, 
        polish=True, 
        dispatch='parallel'
    )
    
    if time.time() - start_time > budget_s - 5:
        best_h = result.x
        best_c5 = objective(best_h)
        return best_h, best_c5, n_points
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Fallback to local refinement
    def local_refine(current_h):
        h = current_h.copy()
        for _ in range(100):
            idx = np.random.randint(0, n_points)
            h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (sum_h / h_sum)
            trial_c5 = objective(h)
            if trial_c5 < objective(current_h):
                current_h = h.copy()
        return current_h
    
    best_h = local_refine(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
