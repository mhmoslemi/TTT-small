import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    sum_h = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective_surrogate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        epsilon = 0.1
        return np.log(np.sum(np.exp(epsilon * overlaps))) / epsilon

    def constraint_sum(h_vec):
        return np.sum(h_vec) - sum_h

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(objective_surrogate, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    best_h = result.x
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb_sum = np.sum(h_perturb)
        if h_perturb_sum > 0:
            h_perturb = h_perturb * (sum_h / h_perturb_sum)
        current_c5 = compute_c5(h_perturb)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturb.copy()

    return best_h, best_c5, n_points
