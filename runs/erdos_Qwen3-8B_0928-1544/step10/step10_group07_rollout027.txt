            result = minimize(compute_c5, h, method='L-BFGS-B', jac=compute_gradient, bounds=bounds)
            h = result.x
        except:
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (sum_h / h_sum)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if _ % 50 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (sum_h / h_sum)
            current_c5_perturbed = compute_c5(h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = h.copy()
    
    return best_h, best_c5, n_points
