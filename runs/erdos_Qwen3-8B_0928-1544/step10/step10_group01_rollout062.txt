            break
        h_new = best_h - learning_rate * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = h_new * (ones_count / np.sum(h_new))
        new_c5 = evaluate(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
    
    return best_h, best_c5, n_points

def evaluate(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
