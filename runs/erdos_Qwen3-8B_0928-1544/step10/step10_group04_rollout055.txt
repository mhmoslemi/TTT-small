from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum) if h_sum > 0 else np.full(n_points, 0.5)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap_idx = np.argmax(corr)
        k = (max_overlap_idx - (n_points - 1)) % n_points
        overlap = corr[max_overlap_idx] * dx

        if overlap < best_c5:
            best_c5 = overlap
            best_h = h.copy()

        def objective(x):
            h_new = x
            total = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                total += h_new[i] * (1.0 - h_new[j]) * dx
            return total

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})

        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
