            i_plus_k = i + k_lag

            else:
                term1 = 0.0
            i_minus_k = i - k_lag

            gradient[i] = term1 + term2

        h_proposed = h - learning_rate * gradient

        perturbation = temperature * np.random.randn(n_points)
