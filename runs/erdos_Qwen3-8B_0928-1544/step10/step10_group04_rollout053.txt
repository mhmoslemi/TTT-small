import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (target_sum / h_sum)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    iterations = 0

    while iterations < 100 and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        shift_k = k_index - (n_points - 1)
        shift_k = np.clip(shift_k, -n_points + 1, n_points - 1)

        def objective(x):
            shifted = np.roll(1 - x, shift_k)
            return np.dot(x, shifted) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if res.success:

            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        iterations += 1

    return best_h, best_c5, n_points
