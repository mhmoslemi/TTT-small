    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    F = 0.8
    Cr = 0.9
    start_time = time.time()

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    for gen in range(300):
        if time.time() - start_time >= budget_s:
            break
        for i in range(population_size):
            idxs = np.random.choice(population_size, 3, replace=False)
            a1, a2, a3 = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a1 + F * (a2 - a3)
            trial = np.copy(pop[i])
            for j in range(n_points):
                if np.random.rand() < Cr:
                    trial[j] = mutant[j]
            trial = np.clip(trial, -0.5, 0.5)
            trial_sum = np.sum(trial)
            trial += (-trial_sum) / n_points
            h_trial = 0.5 + trial
            h_trial = np.clip(h_trial, 0, 1)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            h_current = 0.5 + pop[i]
            h_current = np.clip(h_current, 0, 1)
            c5_current = compute_c5(h_current)
            if c5_trial < c5_current:
                pop[i] = trial.copy()

    h_best = 0.5 + best_h
    h_best = np.clip(h_best, 0, 1)
    best_c5 = compute_c5(h_best)
    return best_h, best_c5, n_points
