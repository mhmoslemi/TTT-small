        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        best_h = h
    else:
        clamped = np.clip(a_initial, -0.5, 0.5)
        sum_clamped = np.sum(clamped)
        delta = -sum_clamped / len(clamped)
        adjusted = clamped + delta
        adjusted = np.clip(adjusted, -0.5, 0.5)
        h = 0.5 + adjusted
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        best_h = h
    
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        delta = (1.0 - sum_h) / (n_points * dx)
        best_h += delta
        best_h = np.clip(best_h, 0, 1)
    
    return best_h, final_c5, n_points
