import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for finer resolution
    sum_h = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (sum_h / h_sum)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - sum_h
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        return compute_c5(h_vec)
    
    try:
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            jac=None,
            options={'maxiter': 500, 'ftol': 1e-6}
        )
    except:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500}
        )
    
    best_h = result.x
    c5 = compute_c5(best_h)
    return best_h, c5, n_points
