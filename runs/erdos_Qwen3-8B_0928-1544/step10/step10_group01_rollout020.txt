import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    ones_count = n_points // 2
    population_size = 50
    F = 0.7
    CR = 0.8

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population.append(initial_h_scaled)
        for _ in range(population_size - 1):
            h = initial_h_scaled + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                continue
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        for _ in range(population_size):

            h_sum = np.sum(h)

                h = np.random.rand(n_points)

            population.append(h_scaled)
