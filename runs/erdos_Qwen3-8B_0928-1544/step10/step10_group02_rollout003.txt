import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.full(n_points, 0.5) * (ones_count / n_points)
        return h * (ones_count / s)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = normalize(initial_h)
    best_c5 = objective(best_h)
    temp = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        trial_h = best_h + noise
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = normalize(trial_h)
        trial_c5 = objective(trial_h)
        
        if trial_c5 < best_c5 or np.random.rand() < np.exp(-(trial_c5 - best_c5) / temp):
            best_c5 = trial_c5
            best_h = trial_h
        
        temp *= cooling_rate
    
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
