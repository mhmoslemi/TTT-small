import numpy as np

    # Function to project h to feasible domain
    def project_h(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scaling = (n_points / 2) / sum_h
        return h * scaling

    # Project initial guess
    h = project_h(h)

    # Objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)

    # Gradient function
    def gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_index = np.argmax(corr)
        k_shift = max_index - (n_points - 1)
        grad = np.zeros_like(h_vec)

    # Optimization bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
