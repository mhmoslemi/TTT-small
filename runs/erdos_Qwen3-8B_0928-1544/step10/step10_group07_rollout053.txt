    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_h}]
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    result = minimize(compute_c5, h_initial, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
