    # Convert to a symmetric function if possible
    if n_points % 2 == 0:

    # Compute initial C₅
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()

        if time.time() - start_time >= budget_s:

        worst_k = np.argmax(corr) - (n_points - 1)

        # Split h into left and right halves
        mid = n_points // 2
        left = h[:mid]
        right = h[mid:]

        # Optimize left block
        def obj_left(x):

        bounds_left = [(0, 1) for _ in range(mid)]

        if res_left.success:

        # Optimize right block
        def obj_right(x):
            tmp = np.concatenate([left, x])
            corr = np.correlate(tmp, 1 - tmp, mode='full')
            return corr[worst_k + (n_points - 1)] * dx

        bounds_right = [(0, 1) for _ in range(n_points - mid)]
