import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        eps = 0.1
        sum_exp = np.sum(np.exp(eps * corr))
        return np.log(sum_exp) / eps
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_h}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
