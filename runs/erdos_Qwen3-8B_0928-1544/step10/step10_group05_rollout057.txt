        return np.max(corr) * 2.0 / n_points
    
    best_a = a.copy()
    best_c5 = compute_c5(best_a)
    initial_temperature = 1.0
    cooling_rate = 0.999
    perturbation_scale = 0.05
    max_iterations = 100000
    start_time = time.time()
    
    temperature = initial_temperature
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        temperature *= cooling_rate
        delta = np.random.normal(0, perturbation_scale, n_points)
        candidate_a = best_a + delta
        candidate_a = np.clip(candidate_a, -0.5, 0.5)
        clipped_sum = candidate_a.sum()
        candidate_a -= clipped_sum / n_points
        if np.any(np.abs(candidate_a) > 0.5):
            continue
        current_c5 = compute_c5(candidate_a)
        delta_c5 = current_c5 - best_c5
        if delta_c5 < 0:
            best_a = candidate_a.copy()
            best_c5 = current_c5
        else:
            if np.random.rand() < np.exp(-delta_c5 / temperature):
                best_a = candidate_a.copy()
                best_c5 = current_c5
    
    h_values = best_a + 0.5
    return h_values, best_c5, n_points
