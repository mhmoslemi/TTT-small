import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n_points, 0.5)
    
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess))
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / total)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    max_iter = 500
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=max_iter, popsize=20, tol=1e-6,

        seed=seed, polish=True, disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=100, popsize=15, tol=1e-6,
            mutation=(0.5, 1), recombination=0.9,

        )
    
    best_h = result.x
    best_c5 = objective(best_h)
