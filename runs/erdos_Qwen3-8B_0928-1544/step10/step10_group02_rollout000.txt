        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
