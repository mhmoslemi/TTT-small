import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:

    if initial_sum != 0:
        initial_h = initial_h * (ones_count / initial_sum)
    else:
        initial_h = np.zeros(n_points)
    current_h = initial_h.copy()
    best_h = current_h.copy()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_c5 = compute_c5(current_h)
    current_c5 = best_c5
    
    initial_temperature = 50.0
    cooling_rate = 0.9995
    min_temperature = 1e-6
    start_time = time.time()
    temperature = initial_temperature
    
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum == 0:
            perturbed_h = np.zeros(n_points)
        else:
            perturbed_h = perturbed * (ones_count / perturbed_sum)
        new_c5 = compute_c5(perturbed_h)
