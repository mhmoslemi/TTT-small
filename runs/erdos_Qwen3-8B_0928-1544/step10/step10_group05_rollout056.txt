import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        a = h_initial - 0.5
    else:
        a = np.zeros(n_points)
        a += np.random.normal(0, 0.05, n_points)
        a = np.clip(a, -0.5, 0.5)
        a -= a.mean()
    
    def compute_c5(a_vec):
        h = a_vec + 0.5
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, 'full')
        return np.max(corr) * 2.0 / n_points
    
    a = a - a.mean()
    best_a = a.copy()
    best_c5 = compute_c5(a)
    
    temp = 1.0
    cool_rate = 0.999
    max_iter = 100000
    perturb_scale = 0.01
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, perturb_scale, n_points)
        a_candidate = a + perturb
        a_candidate = np.clip(a_candidate, -0.5, 0.5)
        a_candidate -= a_candidate.mean()
        a_candidate = np.clip(a_candidate, -0.5, 0.5)
        new_c5 = compute_c5(a_candidate)
        if new_c5 < best_c5:
            a = a_candidate
            best_c5 = new_c5
            best_a = a_candidate.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                a = a_candidate
                best_c5 = new_c5
                best_a = a_candidate.copy()
        temp *= cool_rate
    
    best_h = best_a + 0.5
    return best_h, best_c5, n_points
