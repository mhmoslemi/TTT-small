        if h_sum > 0:
            h = h * (ones_count / h_sum)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        
        # Find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        worst_k = np.argmax(corr)
        
        # Define objective to minimize overlap for worst_k
        def objective(x):
            h_new = x
            overlap = 0
            for i in range(n_points):
                j = (i + worst_k) % n_points
                overlap += h_new[i] * (1 - h_new[j]) * dx
            return overlap
        
        # Constraints: sum(h) = n_points / 2
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
        
        # Bounds and initial guess
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        
        if res.success:
            h = res.x
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    return best_h, best_c5, n_points
