    R = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == 1, 0 <= h <= 1]
    
    # Add quadratic constraints for each lag
    for m in range(1, n_points):
        sum_expr = 0.0
        for i in range(n_points - m):
            sum_expr += h[i] * h[i + m]
        sum_expr *= dx
        constraints.append(sum_expr + R >= 1)
    
    # Objective: maximize R
    prob = cp.Problem(cp.Maximize(R), constraints)
    prob.solve(solver='MOSEK', verbose=False)
    
    if prob.status != 'OPTIMAL':
        # Fallback to initial guess if solver fails
        h_opt = h_initial
    else:
        h_opt = h.value
    
    # Compute C5 as 1 - R
    c5_bound = 1.0 - R.value if prob.status == 'OPTIMAL' else 0.380919
    
    return h_opt, c5_bound, n_points
