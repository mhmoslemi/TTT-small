        relevant_corr = corr[n_points - 1 : 2 * n_points - 1]
        max_overlap_idx = np.argmax(relevant_corr)
        k = max_overlap_idx

        def obj_left(x):
            new_a_left = x
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                a_part = a[n_left + (j - n_left) % (n_points - n_left)] if j >= n_left else a[j]
                val_left = 0.5 + new_a_left[i % n_left]
                val_right = 0.5 - a_part
                overlap += val_left * val_right * dx
            return overlap

        bounds_left = [(-0.5, 0.5) for _ in range(n_left)]
        cons_left = ({'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(a[n_left:])})
        res_left = minimize(obj_left, a[:n_left], method='SLSQP', bounds=bounds_left, constraints=cons_left)
        if res_left.success:
            a[:n_left] = res_left.x

        def obj_right(x):
            new_a_right = x
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                a_part = a[j] if j < n_left else new_a_right[j - n_left]
                val_left = 0.5 + a[j]
                val_right = 0.5 - new_a_right[(j - n_left) % (n_points - n_left)]
                overlap += val_left * val_right * dx
            return overlap

        bounds_right = [(-0.5, 0.5) for _ in range(n_points - n_left)]
        cons_right = ({'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(a[:n_left])})
        res_right = minimize(obj_right, a[n_left:], method='SLSQP', bounds=bounds_right, constraints=cons_right)
        if res_right.success:
            a[n_left:] = res_right.x

        h = 0.5 + a
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        perturb = np.random.uniform(-0.01, 0.01, size=n_points)
        a = np.clip(a + perturb, -0.5, 0.5)

    return best_h, best_c5, n_points
