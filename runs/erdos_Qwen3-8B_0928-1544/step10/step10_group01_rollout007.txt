    rho2 = 2.0
    inertia_decay = 0.99
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(len(swarm))]
    personal_best = [h.copy() for h in swarm]
    global_best = swarm[0].copy()
    best_c5 = evaluate_erdos_solution(global_best)
    start_time = time.time()
    while time.time() - start_time < budget_s * 0.8:
        for i in range(len(swarm)):
            v = velocities[i]
            pbest = personal_best[i]
            gbest = global_best
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v += omega * (rho1 * r1 * (pbest - swarm[i]) + rho2 * r2 * (gbest - swarm[i]))
            velocities[i] = v
            swarm[i] += v
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm_sum = np.sum(swarm[i])
            if swarm_sum == 0:
                continue
            swarm[i] = swarm[i] * (ones_count / swarm_sum)
            new_c5 = evaluate_erdos_solution(swarm[i])
            if new_c5 < evaluate_erdos_solution(personal_best[i]):
                personal_best[i] = swarm[i].copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    global_best = swarm[i].copy()
        omega *= inertia_decay
    current_h = global_best.copy()
    for _ in range(1000):
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k = np.argmax(corr)
        grad = np.zeros_like(current_h)
        eps = 1e-5
        for i in range(n_points):
            h_perturbed = current_h.copy()
            h_perturbed[i] += eps
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum == 0:
                continue
            h_perturbed_scaled = h_perturbed * (ones_count / h_perturbed_sum)
            corr_perturbed = np.correlate(h_perturbed_scaled, 1 - h_perturbed_scaled, mode='full')
            val_perturbed = corr_perturbed[k] * (2.0 / n_points)
            corr_original = np.correlate(current_h, 1 - current_h, mode='full')
            val_original = corr_original[k] * (2.0 / n_points)
            grad[i] = (val_perturbed - val_original) / eps
        current_h -= 0.01 * grad
        current_h = np.clip(current_h, 0, 1)
        current_h_sum = np.sum(current_h)
        if current_h_sum == 0:
            break
        current_h = current_h * (ones_count / current_h_sum)
        new_c5 = evaluate_erdos_solution(current_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            global_best = current_h.copy()
        if abs(new_c5 - best_c5) < 1e-6:
            break
    return global_best, best_c5, n_points
