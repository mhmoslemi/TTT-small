import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    sum_h = n_points / 2.0

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (sum_h / h_sum)

    def compute_actual_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        values = corr * dx
        eps = 1e-3
        log_sum = np.log(np.sum(np.exp(eps * values)))
        return log_sum / eps

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_actual_c5(best_h)

    for _ in range(10):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum > 0:
            scale = sum_h / h_perturbed_sum
            h_perturbed = h_perturbed * scale
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        current_h = result.x
        current_c5 = compute_actual_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
