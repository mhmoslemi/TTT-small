import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    perturb = 0.01 * np.random.randn(n_points)
    h += perturb
    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2
    sum_current = np.sum(h)
    if not np.isclose(sum_current, target_sum):
        scaling_factor = target_sum / sum_current
        h = h * scaling_factor
        h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    beta = 0.5
    decay_rate = 0.995
    max_iter = 500
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        C = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(C)
        max_indices = np.where(C >= max_val - 1e-8)[0]
        grad = np.zeros(n_points)
        for m in max_indices:
            k = m - (n_points - 1)
            for i in range(n_points):
                i_plus_k = i + k
                i_minus_k = i - k
                term1 = 0.0
                if 0 <= i_plus_k < n_points:
                    term1 = 1.0 - h[i_plus_k]
                term2 = 0.0
                if 0 <= i_minus_k < n_points:
                    term2 = h[i_minus_k]
                grad[i] += term1 - term2
        h -= beta * grad
        h = np.clip(h, 0.0, 1.0)
        sum_current = np.sum(h)
        if not np.isclose(sum_current, target_sum):
            scaling_factor = target_sum / sum_current
            h = h * scaling_factor
            h = np.clip(h, 0.0, 1.0)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        beta *= decay_rate
    
    return best_h, best_c5, n_points

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * 2.0 / n_points
