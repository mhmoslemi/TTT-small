import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    def compute_overlap_integral(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def compute_total_gradient(h):
        gradient = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')

            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j]) * (2.0 / n_points)
                    gradient[j] -= h[i] * (2.0 / n_points)
        return gradient

    start_time = time.time()
    alpha = 0.1
    decay = 0.998
