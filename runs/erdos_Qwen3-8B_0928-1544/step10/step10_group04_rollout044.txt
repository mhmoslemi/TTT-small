        cons_right = {'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(a_left) - 0}
        res_right = minimize(obj_right, a_right, method='L-BFGS-B', bounds=bounds_right, constraints=cons_right)
        if res_right.success:
            a_right = res_right.x
            a[n_left:] = a_right
        
        h = a + 0.5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    return best_h, best_c5, n_points
