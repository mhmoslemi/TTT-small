        x_clip = np.clip(x, 0, 1)
        x_norm = x_clip * (sum_h / np.sum(x_clip))
        return np.sum(x_norm) - sum_h
    
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=20, mutation=(0.5, 1), recombination=0.9,
        tol=1e-6, maxiter=200, disp=False,
        polish=True, seed=seed
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
