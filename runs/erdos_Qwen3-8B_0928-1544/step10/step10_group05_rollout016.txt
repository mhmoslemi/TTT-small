    perturbation_scale = 0.1
    min_temp = 1e-6
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        delta = np.random.normal(0, perturbation_scale, n_points)
        candidate_a = current_a + delta
        candidate_a = candidate_a - candidate_a.mean()
        candidate_a = np.clip(candidate_a, -0.5, 0.5)
        sum_clipped = candidate_a.sum()
        candidate_a[-1] += -sum_clipped / n_points
        candidate_a = np.clip(candidate_a, -0.5, 0.5)
        c5_candidate = compute_c5(candidate_a)
        if c5_candidate < best_c5:
            best_a = candidate_a.copy()
            best_c5 = c5_candidate
        elif np.random.rand() < np.exp((best_c5 - c5_candidate) / initial_temp):
            current_a = candidate_a.copy()
        initial_temp *= cooling_rate
    
    h_values = best_a + 0.5
    return h_values, best_c5, n_points
