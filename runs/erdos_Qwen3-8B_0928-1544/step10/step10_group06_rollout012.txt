        h = np.full(n_points, 0.5)
    h = h.astype(float)
    h = np.clip(h, 0, 1)
    required_sum = n_points / 2
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.full(n_points, 0.5)
    else:
        scale_factor = required_sum / sum_h
        h = h * scale_factor
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    step_size = 0.01
    no_improvement = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j])
                gradient[j] -= h[i]
        h -= step_size * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full(n_points, 0.5)
        else:
            scale_factor = required_sum / sum_h
            h = h * scale_factor
            h = np.clip(h, 0, 1)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 100:
                step_size *= 0.5
                no_improvement = 0
        if step_size < 1e-6:
            break

    return (best_h, best_c5, n_points)
