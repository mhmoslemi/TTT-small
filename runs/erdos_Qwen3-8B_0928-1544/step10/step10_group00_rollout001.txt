import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    F_start = 0.7
    CR_start = 0.9
    F_decay = 0.01
    CR_inc = 0.005
    F_min = 0.5
    CR_max = 0.95
    stagnation = 50

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start
    iter_count = 0
    no_improve = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        if iter_count % 100 == 0:
            F = max(F - F_decay, F_min)
            CR = min(CR + CR_inc, CR_max)
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            if objective(u_scaled) < best_c5:
                population[i] = u_scaled
                no_improve = 0
            else:
                no_improve += 1
            if objective(u_scaled) < best_c5:
                best_c5 = objective(u_scaled)
                best_h = u_scaled.copy()
        if no_improve >= stagnation:
            F = F_start
            CR = CR_start
            no_improve = 0

    return best_h, best_c5, n_points
