import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [h.copy() for h in population]
    global_best = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            h = population[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)

                best_c5 = current_c5
                global_best = h.copy()
        
        omega = 0.9 - (0.8 * (time.time() - start_time) / budget_s)
        T = max(0.1, 1.0 - (time.time() - start_time) / budget_s)

            vel = np.clip(vel, -0.5, 0.5)

            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum != 0:
                new_h = new_h * (ones_count / new_h_sum)
            else:
                new_h = np.zeros(n_points)
            
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                perturbed_h = new_h + noise
                perturbed_h = np.clip(perturbed_h, 0, 1)

                if perturbed_h_sum != 0:
                    perturbed_h = perturbed_h * (ones_count / perturbed_h_sum)
