import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (ones_count / h_sum)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = objective(h)
    best_h = h.copy()

    start_time = time.time()
    iterations = 0
    perturb_size = 0.05
    while time.time() - start_time < budget_s:
        iterations += 1
        if iterations > 100000:
            break
        i = np.random.randint(n_points)
        delta = np.random.uniform(-perturb_size, perturb_size)
        new_h = h.copy()
        new_h[i] += delta
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_h = np.zeros(n_points)
        else:
            scale = (ones_count) / new_sum
            new_h = new_h * scale
        new_h = np.clip(new_h, 0, 1)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        h = new_h

    return best_h, best_c5, n_points
