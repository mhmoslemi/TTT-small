import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    dx = 2.0 / n_points

    def evaluate(h):
        n = len(h)
        cross_corr = np.zeros(n)
        for k in range(n):
            total = 0.0
            for i in range(n - k):
                total += h[i] * (1 - h[i + k]) * dx
            cross_corr[k] = total
        return np.max(cross_corr)

    def gradient(h):
        n = len(h)
        cross_corr = np.zeros(n)
        for k in range(n):
            total = 0.0
            for i in range(n - k):
                total += h[i] * (1 - h[i + k]) * dx
            cross_corr[k] = total
        max_k = np.argmax(cross_corr)
        grad = np.zeros(n)
        for j in range(n):
            j_add = (j + max_k) % n
            j_sub = (j - max_k) % n
            grad[j] = dx * (1 - h[j_add] - h[j_sub])
        return grad

    best_h = h.copy()
    best_c5 = evaluate(h)
    best_time = time.time()

    learning_rate = 0.1
    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - best_time > budget_s:
            break
        grad = gradient(h)
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        if sum_h_new == 0:
            h_new = np.zeros(n_points)

        if new_c5 < best_c5:

            best_c5 = new_c5
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9

    # Final scaling to ensure constraint
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (ones_count / sum_h)
    return h, best_c5, n_points
