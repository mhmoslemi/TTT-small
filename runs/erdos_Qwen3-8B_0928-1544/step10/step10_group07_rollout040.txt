            grad[i] = (c_plus - compute_c5(h)) / epsilon
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    result = minimize(
        fun=compute_c5,
        x0=h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        options={
            'maxiter': 500,
            'ftol': 1e-7,
            'gtol': 1e-7,
            'eps': 1e-8,
            'disp': False
        }
    )
    
    h = np.clip(result.x, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (sum_h / h_sum)
    c5 = compute_c5(h)
    return h, c5, n_points
