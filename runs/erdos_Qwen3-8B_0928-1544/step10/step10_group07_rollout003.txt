import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    sum_h = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h})
    
    def optimize(start):
        result = minimize(objective, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 200})
        return result.x, result.fun
    
    best_h, best_c5 = h.copy(), objective(h)
    start_time = time.time()
    
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(h + perturbation, 0, 1)
        perturbed = perturbed * (sum_h / np.sum(perturbed))
        current_h, current_c5 = optimize(perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    return best_h, best_c5, n_points
