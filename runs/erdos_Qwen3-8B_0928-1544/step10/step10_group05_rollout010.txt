import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def normalize(a):
        a = a - a.sum() / n_points
        a = np.clip(a, -0.5, 0.5)
        return a
    
    def compute_c5(a_vec):
        h = 0.5 + a_vec
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    def tournament_select(pop, fit, size=5):
        idxs = np.random.choice(len(pop), size, replace=False)
        best = idxs[np.argmin(fit[idxs])]
        return pop[best]
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points)
        child = np.concatenate([parent1[:point], parent2[point:]])
        child = normalize(child)
        return child
    
    def mutate(a):
        if np.random.random() < 0.1:
            idx = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            a[idx] += delta
        a = normalize(a)
        return a
    
    a_initial = initial_h - 0.5
    a_initial = normalize(a_initial)
    population = [a_initial]
    for _ in range(49):
        a = np.random.uniform(-0.5, 0.5, n_points)
        a = normalize(a)
        population.append(a)
    
    fitness = np.array([compute_c5(a) for a in population])
    best_idx = np.argmin(fitness)
    best_fitness = fitness[best_idx]
    best_a = population[best_idx].copy()
    
    max_iter = 500
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        new_pop = []
        for _ in range(len(population)):
            p1 = tournament_select(population, fitness)
            p2 = tournament_select(population, fitness)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        
        new_fitness = np.array([compute_c5(a) for a in new_pop])
        combined = np.concatenate([population, new_pop])
        combined_fitness = np.concatenate([fitness, new_fitness])
        order = np.argsort(combined_fitness)
        population = combined[order][:len(population)]
        fitness = combined_fitness[order][:len(population)]
        
        if np.min(fitness) < best_fitness:
            best_fitness = np.min(fitness)
            best_a = population[np.argmin(fitness)]
    
    h_values = best_a + 0.5
    return h_values, best_fitness, n_points
