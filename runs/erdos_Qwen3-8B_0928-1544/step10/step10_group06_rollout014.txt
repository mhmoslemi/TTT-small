import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = 0.5 * np.ones(n_points)
        for _ in range(10):
            h += np.random.rand(n_points) * 0.1 - 0.05
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    scale = (n_points / 2) / s
    h = h * scale
    h = np.clip(h, 0.0, 1.0)

    def compute_c5_and_lag(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        idx = np.argmax(corr)
        lag = idx - (n_points - 1)
        return max_val, lag

    def compute_gradient(h, lag):
        grad = np.zeros_like(h)
        if lag > 0:
            for i in range(len(h) - lag):
                grad[i] += (1 - h[i + lag])
                grad[i + lag] -= h[i]
        else:
            lag_abs = -lag
            for i in range(lag_abs, len(h)):
                grad[i] += (1 - h[i - lag_abs])
                grad[i - lag_abs] -= h[i]
        return grad

    def optimize(h, steps=10, step_size=0.01):
        for _ in range(steps):
            max_val, lag = compute_c5_and_lag(h)
            grad = compute_gradient(h, lag)
            h = h - step_size * grad
            h = np.clip(h, 0.0, 1.0)
            s = np.sum(h)
            h = h * (n_points / 2) / s
            h = np.clip(h, 0.0, 1.0)
        return h

    best_h = h.copy()
    best_c5 = compute_c5_and_lag(h)[0]
    start_time = time.time()

    while time.time() - start_time < budget_s:
        h = optimize(best_h, steps=10, step_size=0.015)
        current_c5 = compute_c5_and_lag(h)[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        h = best_h.copy()
        # Random perturbation
        h += np.random.rand(n_points) * 0.05 - 0.025
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        h = h * (n_points / 2) / s
        h = np.clip(h, 0.0, 1.0)
        current_c5 = compute_c5_and_lag(h)[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
