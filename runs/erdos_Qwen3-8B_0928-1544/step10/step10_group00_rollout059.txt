import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    population_size = 50
    max_iter = 1000
    T_initial = 100.0
    alpha = 0.995
    SA_prob = 0.1
    omega_initial = 0.9
    omega_final = 0.4
    phi1 = 1.4
    phi2 = 1.4
    vel_limit = 0.5

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_h_sum = np.sum(initial_h)
        scaled_h = initial_h * (target_sum / initial_h_sum) if initial_h_sum != 0 else np.zeros(n_points)
        population = [scaled_h]
        velocities = [np.zeros(n_points)]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            scaled_h = h * (target_sum / h_sum)
            population.append(scaled_h)
            velocities.append(np.zeros(n_points))
    else:
        population = []
        velocities = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            scaled_h = h * (target_sum / h_sum)
            population.append(scaled_h)
            velocities.append(np.zeros(n_points))

    pbests = [h.copy() for h in population]
    gbest = population[0].copy()
    best_c5 = 1.0
    best_h = gbest.copy()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = objective(gbest)
    best_h = gbest.copy()

    start_time = time.time()
    current_iter = 0

    while time.time() - start_time < budget_s:
        current_iter += 1
        T = T_initial * (alpha ** current_iter)
        omega = omega_initial - (omega_initial - omega_final) * (current_iter / max_iter)

        for i in range(population_size):
            h = population[i]
            v = velocities[i]
            r1 = np.random.rand()
            r2 = np.random.rand()
            new_velocity = omega * v + phi1 * r1 * (pbests[i] - h) + phi2 * r2 * (gbest - h)
            new_velocity = np.clip(new_velocity, -vel_limit, vel_limit)
            velocities[i] = new_velocity
            new_h = h + new_velocity
            new_h_clipped = np.clip(new_h, 0.0, 1.0)
            new_h_sum = np.sum(new_h_clipped)
            if new_h_sum == 0:
                new_h_scaled = np.zeros(n_points)
            else:
                scaling_factor = target_sum / new_h_sum
                new_h_scaled = new_h_clipped * scaling_factor
            population[i] = new_h_scaled

            new_c5 = objective(population[i])
            if new_c5 < objective(pbests[i]):
                pbests[i] = population[i].copy()
            if new_c5 < objective(gbest):
                gbest = population[i].copy()
                best_c5 = new_c5
                best_h = gbest.copy()

            if np.random.rand() < SA_prob:
                perturbed_h = population[i].copy()
                num_perturb = np.random.randint(1, n_points // 2)
                indices = np.random.choice(n_points, num_perturb, replace=False)
                for idx in indices:
                    perturbed_h[idx] += np.random.uniform(-0.1, 0.1)
                perturbed_h_clipped = np.clip(perturbed_h, 0.0, 1.0)
                perturbed_h_sum = np.sum(perturbed_h_clipped)
                if perturbed_h_sum == 0:
                    perturbed_h_scaled = np.zeros(n_points)

                    if new_c5_perturb < objective(gbest):
                        gbest = perturbed_h_scaled.copy()
                        best_c5 = new_c5_perturb
                        best_h = gbest.copy()
                population[i] = perturbed_h_scaled

    return best_h, best_c5, n_points
