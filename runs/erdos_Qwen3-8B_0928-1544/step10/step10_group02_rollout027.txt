    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        initial_h = h

    if not np.isclose(np.sum(initial_h), target_sum):
        initial_h = initial_h * target_sum / np.sum(initial_h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, n_points)

    T_initial = 100.0
    cooling_rate = 0.995
    start_time = time.time()

    T = T_initial
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.05, n_points)
        new_h = best_h + noise
        new_h = np.clip(new_h, 0.0, 1.0)

        sum_new = np.sum(new_h)
        if sum_new == 0:
            new_h = np.zeros_like(new_h)
        else:
            scale = target_sum / sum_new
            new_h = new_h * scale

        new_c5 = compute_c5(new_h, n_points)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            best_c5 = new_c5

        T *= cooling_rate

    return best_h, best_c5, n_points
