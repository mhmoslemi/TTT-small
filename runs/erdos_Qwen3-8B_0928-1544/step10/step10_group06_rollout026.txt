            c = compute_c5(candidate)
            if c < best_c5:
                best_c5 = c
                best_h = candidate
        
        sigma *= (1 - (time.time() - start_time) / budget_s)
        sigma = max(sigma, 0.001)
    
    h_values = best_h
    return h_values, best_c5, n_points
