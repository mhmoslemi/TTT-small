def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 100
    F_initial = 0.8
    F_final = 0.6
    CR_initial = 0.9
    CR_final = 0.95

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
    else:
        population = []

    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    while time.time() - start_time < budget_s:
        F = F_initial - (F_initial - F_final) * (time.time() - start_time) / budget_s
        CR = CR_initial + (CR_final - CR_initial) * (time.time() - start_time) / budget_s
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
