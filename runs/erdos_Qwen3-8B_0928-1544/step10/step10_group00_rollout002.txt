                else:
                    prob = np.exp((current_c5 - new_c5) / temp)
                    if np.random.rand() < prob:
                        population[i] = perturbed_h.copy()
        
        # Cool temperature
        temp *= cooling_rate
        
        # Update best solution
        for i in range(len(population)):
            current_c5 = objective(population[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[i].copy()
    
    return best_h, best_c5, n_points
