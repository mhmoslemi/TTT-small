        segs = []
        start = 0
        while start < n_points:
            end = start
            while end < n_points and h[end] == h[start]:
                end += 1
            if end > start:
                segs.append((start, end, h[start]))
            start = end
        return segs
    
    segments = to_segments(h, n_points)
    n_segments = len(segments)
    
    def segments_to_array(segs, n_points):

            arr[start:end] = h_val
        return arr
    
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * (2.0 / n_points)

        def objective(w):
            new_segs = []
            pos = 0
            for width, h_val in zip(w[:-1], h_vals[:-1]):

            h_array = segments_to_array(new_segs, n_points)
            return compute_c5(h_array)

        w = np.array([s[1] - s[0] for s in segs])
        result = minimize(objective, w, bounds=bounds, method='L-BFGS-B')
        return result.x
    
    def optimize_heights(segs, w):
