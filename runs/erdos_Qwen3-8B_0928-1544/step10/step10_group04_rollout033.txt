import numpy as np
from scipy.optimize import minimize
import time
from scipy.optimize import COBYLA

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        h = h.clip(0, 1)
    
    a = h - 0.5
    a_sum = np.sum(a)
    if not np.isclose(a_sum, 0):
        adjustment = -a_sum / len(a)
        a += adjustment
    
    n_left = n_points // 2
    a_left = a[:n_left]
    a_right = a[n_left:]
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    
    def calc_overlap(a_left, a_right, k_val):
        val_left = 0.5 + a_left
        val_right = 0.5 - np.roll(a_left, k_val)
        overlap = np.sum(val_left * val_right * dx)
        return overlap
    
    def objective(a_params):
        a_left_new = a_params[:n_left]
        a_right_new = a_params[n_left:]
        max_overlap = -np.inf
        for k in range(n_points):
            overlap = calc_overlap(a_left_new, a_right_new, k)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    bounds = [(-0.5, 0.5) for _ in range(n_points)]
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time >= budget_s:
            break
        
        # Optimize all parameters
        res = minimize(objective, np.concatenate((a_left, a_right)), 
                      method='COBYLA', bounds=bounds, 
                      options={'maxiter': 50, 'disp': False})
        if res.success:
            a = res.x
            h = a + 0.5
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h) / n_points
            if np.abs(h_sum - 0.5) > 1e-4:
                h = h * (0.5 / h_sum)
            h = np.clip(h, 0, 1)
            
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    return best_h, best_c5, n_points
