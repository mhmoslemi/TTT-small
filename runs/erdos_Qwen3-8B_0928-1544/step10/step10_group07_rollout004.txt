            h = h * (sum_h / np.sum(h))
    
    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        eps = 0.01
        sum_exp = np.sum(np.exp(eps * corr))
        return np.log(sum_exp) / eps * (2.0 / n_points)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h}]
    bounds = [(0., 1.) for _ in range(n_points)]
    
    start_time = time.time()
    result = minimize(objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + noise, 0, 1)
        if np.sum(perturbed) > 0:
            perturbed = perturbed * (sum_h / np.sum(perturbed))
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < result.fun:
            result = res
            best_h = result.x
            best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
