    h = h * scale_factor
    best_h = h.copy()

    learning_rate = 0.1
    temperature = 0.5
    decay_lr = 0.999
    decay_temp = 0.99
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += 1 - h[j + k]
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k]
        # Update h
        noise = temperature * np.random.randn(n_points)
        h_proposed = h - learning_rate * grad + noise
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed_sum = np.sum(h_proposed)
        if h_proposed_sum == 0:
            continue
        scale_factor = (ones_count) / h_proposed_sum
        h_proposed = h_proposed * scale_factor
        # Compute new C5
        new_c5 = np.max(np.correlate(h_proposed, 1 - h_proposed, mode='full')) * (2.0 / n_points)
        # Acceptance
        if new_c5 < best_c5:
            h = h_proposed
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Simulated annealing
            if np.random.rand() < np.exp((best_c5 - new_c5) / temperature):

                best_c5 = new_c5
                best_h = h.copy()
        # Decay parameters
        learning_rate *= decay_lr
        temperature *= decay_temp
    return best_h, best_c5, n_points
