    sum_h = np.sum(h)
    h *= (n_points / 2.0) / sum_h
    h = np.clip(h, 0.0, 1.0)
    
    best_c5 = float('inf')
    best_h = h.copy()
    learning_rate = 0.1
    decay_rate = 0.98
    max_iter = 10000
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1.0 - h[j]) * (2.0 / n_points)
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h_current = np.sum(h)
        scale_factor = (n_points / 2.0) / sum_h_current
        h *= scale_factor
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
    
    return best_h, best_c5, n_points
