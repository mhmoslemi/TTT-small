                if sum_h == 0:
                    h_new = np.zeros(n_points)
                else:
                    h_new = h_new * (ones_count / sum_h)
                new_c5 = compute_c5(h_new)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
    
    return best_h, best_c5, n_points
