        if time.time() - start > budget_s:
            break
        # Small random perturbation
        perturb = np.random.normal(0, 0.1, size=a_initial.shape)
        a = a_initial + perturb
        result = minimize(evaluate, a, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(a.shape[0])])
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = result.x
    
    h_values = 0.5 + M @ best_a
    h_values = np.clip(h_values, 0, 1)
    return h_values, best_c5, n_points
