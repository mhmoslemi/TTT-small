import numpy as np
import time

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == 0:
        return np.array([1.0] + [0.0] * (n_points - 1))
    alpha = (n_points / 2) / s
    h = alpha * h
    return h

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else None
    
    if initial_h is not None:
        population = [initial_h.copy()]
        for _ in range(49):
            perturbation = np.random.normal(0, 0.1, n_points)
            h = initial_h + perturbation
            population.append(repair(h, n_points))
    else:
        population = []
        for _ in range(50):
            h = np.random.rand(n_points)
            population.append(repair(h, n_points))
    
    best_h = population[0]
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    mutation_rate = 0.1
    
    while time.time() - start_time < budget_s:
        new_population = []
        for h in population:
            perturbation = np.random.normal(0, mutation_rate, n_points)
            mutated_h = h + perturbation
            repaired_h = repair(mutated_h, n_points)
            current_c5 = compute_c5(repaired_h, n_points)
            new_population.append((current_c5, repaired_h))
        
        new_population.sort()
        top_candidates = [h for (c5, h) in new_population[:50]]
        
        for c5, h in new_population:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        mutation_rate *= (1 - (time.time() - start_time) / budget_s)
        
        population = top_candidates
    
    return best_h, best_c5, n_points
