import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    temperature = 0.5
    learning_rate = 0.1
    decay_lr, decay_temp = 0.999, 0.99
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        g = np.zeros(n_points)
        if k > 0:
            for i in range(n_points - k):
                g[i] = 1 - h[i + k]
        elif k < 0:
            m = -k
            for i in range(m, n_points):
                g[i] = 1 - h[i - m]
        else:
            g = 1 - 2 * h
        h_proposed = h - learning_rate * g
        noise = temperature * np.random.randn(n_points)
        h_proposed += noise
        h_proposed = np.clip(h_proposed, 0.0, 1.0)
        h_sum = np.sum(h_proposed)
        scale = ones_count / h_sum if h_sum != 0 else 1.0
        h_proposed *= scale
        new_corr = np.correlate(h_proposed, 1 - h_proposed, mode='full')
        new_c5 = np.max(new_corr) * (2.0 / n_points)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temperature):
            h = h_proposed.copy()
            best_c5 = new_c5
            best_h = h.copy()
        learning_rate *= decay_lr
        temperature *= decay_temp
    return best_h, best_c5, n_points
