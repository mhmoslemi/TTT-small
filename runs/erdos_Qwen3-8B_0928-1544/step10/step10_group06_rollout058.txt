    best_c5 = float('inf')
    best_h = parents[0].copy()
    
    start_time = time.time()
    initial_sigma = 0.15
    T = budget_s / 3
    mutation_rate = initial_sigma
    
    while time.time() - start_time < budget_s:
        new_parents = []
        for parent in parents:
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = repair(child)
            if np.random.rand() < 0.1:
                child = np.random.rand(n_points) * 0.5 - 0.25
                child = repair(child)
            new_parents.append(child)
        parents = new_parents
        for child in parents:
            c5 = compute_c5(child)
            if c5 < best_c5:
                best_c5 = c5
                best_h = child.copy()
        mutation_rate *= np.exp(-10 * (time.time() - start_time) / budget_s)
    
    return best_h, best_c5, n_points
