    perturb_scale = 0.1
    max_iter = 100000
    time_budget = budget_s
    
    start_time = time.time()
    T = initial_temp
    
    for _ in range(max_iter):
        if time.time() - start_time >= time_budget:
            break
        # Perturb
        perturb = np.random.normal(0, perturb_scale, n_points)
        new_a = current_a + perturb
        new_a = np.clip(new_a, -0.5, 0.5)
        sum_new = np.sum(new_a)
        adjust = -sum_new / n_points
        new_a += adjust
        new_a = np.clip(new_a, -0.5, 0.5)
        # Compute new_c5
        new_c5 = compute_c5(new_a)
        # Acceptance
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_a = new_a.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_a = current_a.copy()
                best_c5 = current_c5
        T *= cooling_rate
    
    h_values = best_a + 0.5
    return h_values, best_c5, n_points
