import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        personal_best = [h.copy() for h in population]
    else:
        population = []
        personal_best = []
        for _ in range(50):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
            population.append(h)
            personal_best.append(h.copy())
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    gbest = best_h.copy()
    velocities = [np.zeros(n_points) for _ in range(len(population))]
    start_time = time.time()
    
    rho1, rho2 = 2.0, 2.0
    inertia = 0.9
    
    while time.time() - start_time < budget_s:
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + rho1 * r1 * (personal_best[i] - population[i]) + rho2 * r2 * (gbest - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0.0, 1.0)
            h_sum = np.sum(population[i])
            population[i] *= (ones_count / h_sum)
            current_c5 = objective(population[i])
            if current_c5 < objective(personal_best[i]):
                personal_best[i] = population[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = population[i].copy()
                    best_h = population[i].copy()
    
    time_left = budget_s - (time.time() - start_time)
    eta = 0.01
    for _ in range(int(time_left / 0.1)):
        if time.time() - start_time >= budget_s:
            break
        h = best_h.copy()
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_perturb = h.copy()
            h_perturb[j] += eps
            h_perturb_sum = np.sum(h_perturb)
            h_perturb_scaled = h_perturb * (ones_count / h_perturb_sum)
            obj_perturb = objective(h_perturb_scaled)
            grad[j] = (obj_perturb - objective(h)) / eps
        h += -eta * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)

        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
