            start_idx = int(starts[i] / dx)
            end_idx = int(ends[i] / dx)
            height = heights[i]
            for j in range(start_idx, end_idx):
                if j < n_points:
                    h[j] = height
        # Normalize to ensure ∫h dx = 1
        total = np.sum(h) * dx
        h = h * (ones_count / total) if total > 0 else h
        return h
    
    # Initial heights for segments
    segment_heights = np.ones(n_segments)
    best_h = build_h_from_segments(start_points, end_points, segment_heights)
    best_c5 = objective(best_h)
    
    # Optimization loop
    for _ in range(1000):
        if time.time() - start_time >= budget_s:
            break
        # Optimize segment positions
        perturbation = np.random.normal(0, 0.05, n_segments)
        new_starts = start_points + perturbation
        new_ends = end_points + perturbation
        new_starts.sort()
        new_ends.sort()
        for i in range(n_segments):
            if new_starts[i] >= new_ends[i]:
                new_starts[i] = new_ends[i] - 0.1
        new_h = build_h_from_segments(new_starts, new_ends, segment_heights)
        current_c5 = objective(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
            start_points, end_points = new_starts, new_ends
        # Optimize segment heights
        perturbation = np.random.normal(0, 0.05, n_segments)
        new_heights = segment_heights * (1 + perturbation)
        new_heights = np.clip(new_heights, 0.1, 1)
        total = np.sum(new_heights) * (2.0 / n_points)
        new_heights = new_heights * (ones_count / total) if total > 0 else new_heights
        new_h = build_h_from_segments(start_points, end_points, new_heights)
        current_c5 = objective(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
            segment_heights = new_heights
    
    return best_h, best_c5, n_points
