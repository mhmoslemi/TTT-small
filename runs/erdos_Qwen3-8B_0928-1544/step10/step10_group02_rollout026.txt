import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    initial_sum = np.sum(initial_h)
    if initial_sum != 0:
        initial_h = initial_h * (ones_count / initial_sum)
    else:
        initial_h = np.zeros(n_points)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    
    T = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        new_h = best_h + noise
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_h = np.zeros(n_points)
        else:
            new_h = new_h * (ones_count / new_sum)
        
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * (2.0 / n_points)
        
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            best_c5 = new_c5
        
        T *= cooling_rate
    
    return best_h, best_c5, n_points
