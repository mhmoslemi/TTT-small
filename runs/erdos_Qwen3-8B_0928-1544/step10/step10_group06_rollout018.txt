def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    h_pop = []
    if initial_h is not None:
        h_pop.append(initial_h.copy())
        for _ in range(49):
            h = np.random.rand(n_points)
            s = np.sum(h)
            factor = (n_points / 2) / s
            h = h * factor
            h_pop.append(h)
    else:
        for _ in range(50):
            h = np.random.rand(n_points)
            s = np.sum(h)
            factor = (n_points / 2) / s
            h = h * factor
            h_pop.append(h)
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    sigma = 0.1
    while time.time() - start_time < budget_s:
        new_pop = []
        for h in h_pop:
            for _ in range(50):
                noise = np.random.normal(0, sigma, n_points)
                offspring = h + noise
                offspring = np.clip(offspring, 0, 1)
                s = np.sum(offspring)
                factor = (n_points / 2) / s
                offspring = offspring * factor
                new_pop.append(offspring)
        candidates = h_pop + new_pop
        c5s = []
        for h in candidates:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            c5s.append(c5)
        idxs = np.argsort(c5s)
        selected = idxs[:50]
        h_pop = [candidates[i] for i in selected]
        min_c5 = np.min(c5s)
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = candidates[np.argmin(c5s)]
        sigma = max(sigma * (1 - (time.time() - start_time) / budget_s), 0.001)
    return (best_h, best_c5, n_points)
