        h = np.full(n_points, 0.5)
    # Ensure sum is correct
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.full(n_points, ones_count / n_points)
    else:
        h = h * (ones_count / h_sum)
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    # Initial C5
    best_c5 = objective(h)
    best_h = h.copy()
    # SA parameters
    T = 1.0
    cooling_rate = 0.995
    # Time tracking
    start_time = time.time()
    # Main loop
    while time.time() - start_time < budget_s:
        # Generate perturbation
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize sum
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum == 0:
            h_perturbed = np.full(n_points, ones_count / n_points)
        else:
            h_perturbed = h_perturbed * (ones_count / h_perturbed_sum)
        # Compute new_c5
        new_c5 = objective(h_perturbed)
        # Acceptance criteria
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_perturbed.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_c5 = new_c5
                best_h = h_perturbed.copy()
        # Cool temperature
        T *= cooling_rate
    return best_h, best_c5, n_points
