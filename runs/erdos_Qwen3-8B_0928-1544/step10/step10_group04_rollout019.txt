    n_points = 70
    dx = 2.0 / n_points

    # Initialize population
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        a = h - 0.5
        a_sum = np.sum(a)
        adjustment = -a_sum / len(a)
        a += adjustment
        population = [a.copy()]
    else:
        N = 20
        population = []
        for _ in range(N):
            a = np.random.uniform(-0.5, 0.5, n_points)
            a_sum = np.sum(a)
            adjustment = -a_sum / len(a)
            a += adjustment
            population.append(a)
        population = np.array(population)

    def objective(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_a = population[0]

    F = 0.8
    CR = 0.9
    generations = 300
    time_start = time.time()

    for gen in range(generations):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            # Select three distinct indices
            indices = np.random.choice(len(population), 3, replace=False)
            j, k, l = indices
            a_target = population[i]
            a_j = population[j]
            a_k = population[k]
            a_l = population[l]
            # Mutation: a_j + F * (a_k - a_l)
            mutant = a_j + F * (a_k - a_l)
            # Crossover
            trial = a_target.copy()
            for m in range(n_points):
                if np.random.rand() < CR:
                    trial[m] = mutant[m]
            # Clip to bounds
            trial = np.clip(trial, -0.5, 0.5)
            # Adjust sum to zero
            trial_sum = np.sum(trial)
            adjustment = -trial_sum / n_points
            trial += adjustment
            # Evaluate
            current_c5 = objective(trial)
            if current_c5 < objective(a_target):
                new_population.append(trial)
            else:
                new_population.append(a_target)
        population = new_population

        # Update best
        for a_candidate in population:
            current_c5 = objective(a_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = a_candidate

    # Final adjustment
    best_a = np.clip(best_a, -0.5, 0.5)
    best_a_sum = np.sum(best_a)
    adjustment = -best_a_sum / n_points
    best_a += adjustment
    best_h = 0.5 + best_a
    return best_h, best_c5, n_points
