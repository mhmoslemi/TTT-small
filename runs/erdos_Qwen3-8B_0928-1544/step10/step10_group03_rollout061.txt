            break
        grad = compute_gradient(h)
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.full(n_points, 0.5)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
        else:
            h = h * (ones_count / h_sum)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < best_c5 - 1e-5:
            alpha *= 0.95
        else:
            alpha = max(alpha * 0.5, 1e-5)
    
    return best_h, best_c5, n_points
