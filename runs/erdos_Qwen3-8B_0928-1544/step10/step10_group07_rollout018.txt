import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()

        perturbed = current_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)

        current_h = perturbed
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
