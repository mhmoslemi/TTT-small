import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h_sum = np.sum(initial_h)
    if initial_h_sum > 0:
        initial_h = initial_h * (n_points / 2) / initial_h_sum
    pop_size = 20
    population = [initial_h.copy() for _ in range(pop_size)]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.8
    CR = 0.9
    elite = best_h.copy()
    
    while True:
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial_sum = np.sum(trial)
            trial = np.clip(trial, 0, 1)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (n_points / 2) / trial_sum
            current_c5 = objective(trial)
            if current_c5 < best_c5 or (np.abs(current_c5 - best_c5) < 1e-8 and np.random.rand() < 0.1):
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
        if np.array_equal(best_h, elite) and np.random.rand() < 0.1:
            break
        elite = best_h.copy()
    return best_h, best_c5, n_points
