from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    best_c5 = 0.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    current_h = best_h.copy()
    best_c5 = objective(current_h)
    
    learning_rate = 0.2
    max_iterations = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        k_max = np.argmax(corr)
        k_shift = k_max - n_points + 1
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_shift) % n_points
            grad[i] = (1 - current_h[j]) - current_h[(i - k_shift) % n_points]
        
        step = learning_rate * grad
        candidate_h = current_h - step
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        if np.sum(candidate_h) == 0:
            candidate_h = current_h
        else:
            candidate_h = candidate_h * (ones_count / np.sum(candidate_h))
        
        result = minimize(
            fun=objective,
            x0=candidate_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
