import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Initialize harmony memory
    HMs = []
    if 'initial_h_values' in globals():
        HMs.append(initial_h.copy())
    for _ in range(49):  # Add 49 more binary solutions
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:ones_count]] = 1.0
        HMs.append(h)
    
    # Evaluate C5 for all in HM
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Sort HM based on C5
    sorted_HMs = sorted(HMs, key=compute_c5)
    best_h = sorted_HMs[0]
    worst_h = sorted_HMs[-1]
    
    start_time = time.time()
    iterations = 0
    max_iterations = int(budget_s / 0.1)  # About 11000 iterations
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        # Generate new solution
        # Randomly select from HM with probability 0.5, or generate new
        if np.random.random() < 0.5:
            # Select from HM
            new_h = np.random.choice(HMs, 1)[0].copy()
        else:
            # Generate new binary solution
            indices = np.random.permutation(n_points)
            new_h = np.zeros(n_points)
            new_h[indices[:ones_count]] = 1.0
        
        # Perturb the new_h slightly
        perturb = np.random.normal(0, 0.05, n_points)
        new_h += perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        # Normalize sum to ones_count
        new_h = new_h * ones_count / np.sum(new_h)
        
        # Compute C5
        current_c5 = compute_c5(new_h)
        # Compare with worst in HM
        if current_c5 < compute_c5(worst_h):
            # Replace worst
            worst_index = np.argmin([compute_c5(h) for h in HMs])
            HMs[worst_index] = new_h.copy()
            sorted_HMs = sorted(HMs, key=compute_c5)
            best_h = sorted_HMs[0]
            worst_h = sorted_HMs[-1]
        
        iterations += 1
    
    # Now perform gradient-based refinement on best_h
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
    else:
        best_h = best_h
    
    return best_h, compute_c5(best_h), n_points

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        HMs = [initial_h]
    else:
        initial_h = np.full(n_points, 0.5)
        HMs = []
    
    # Initialize harmony memory with binary solutions
    for _ in range(50 - len(HMs)):
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:ones_count]] = 1.0
        HMs.append(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    sorted_HMs = sorted(HMs, key=compute_c5)
    best_h = sorted_HMs[0]
    worst_h = sorted_HMs[-1]
    
    start_time = time.time()
    iterations = 0
    max_iter = int(budget_s / 0.1)
    
    while iterations < max_iter and time.time() - start_time < budget_s:
        if np.random.random() < 0.5:
            new_h = np.random.choice(HMs, 1)[0].copy()
        else:
            indices = np.random.permutation(n_points)
            new_h = np.zeros(n_points)
            new_h[indices[:ones_count]] = 1.0
        
        perturb = np.random.normal(0, 0.05, n_points)
        new_h += perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            new_h = new_h * ones_count / new_sum
        
        curr_c5 = compute_c5(new_h)
        if curr_c5 < compute_c5(worst_h):
            worst_idx = np.argmin([compute_c5(h) for h in HMs])
            HMs[worst_idx] = new_h.copy()
            sorted_HMs = sorted(HMs, key=compute_c5)
            best_h = sorted_HMs[0]
            worst_h = sorted_HMs[-1]
        
        iterations += 1
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
    else:
        best_h = best_h
    
    return best_h, compute_c5(best_h), n_points
