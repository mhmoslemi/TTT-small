                h_medium = h_medium / np.sum(h_medium) * (n_points / 2)
                result = minimize(
                    fun=lambda h: objective(h, n_points),
                    x0=h_medium,
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * n_points,
                    constraints={'type': 'eq', 'fun': lambda h: constraint_sum(h, n_points)},
                    options={'maxiter': 200, 'ftol': 1e-9}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
                        current_h = result.x
            if time.time() - start_time > budget_s:
                return project_h(best_h, n_points), best_c5, n_points
        elif stage_name == 'high':
            if current_h is not None:
                h_high = upsample_h(current_h, 35, n_points)
                h_high = h_high / np.sum(h_high) * (n_points / 2)
                result = minimize(
                    fun=lambda h: objective(h, n_points),
                    x0=h_high,
                    method='COBYLA',
                    bounds=[(0.0, 1.0)] * n_points,
                    constraints={'type': 'eq', 'fun': lambda h: constraint_sum(h, n_points)},
                    options={'maxiter': 500, 'ftol': 1e-9}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
                        current_h = result.x
            if time.time() - start_time > budget_s:
                return project_h(best_h, n_points), best_c5, n_points
    
    final_h = project_h(best_h, n_points)
    return final_h, best_c5, n_points
