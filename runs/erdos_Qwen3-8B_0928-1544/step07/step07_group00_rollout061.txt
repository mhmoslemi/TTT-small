        particles = np.clip(particles, 0.0, 1.0)
        particles = np.array([h / h.sum() * (n_points / 2) for h in particles])
    
    # Refine best candidate with SLSQP
    result = minimize(
        fun=objective,
        x0=global_best,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
