        n = len(h)
        current_sum = np.sum(h)
        delta = (target_sum - current_sum) / n
        h_new = h + delta
        h_new = np.clip(h_new, 0, 1)
        new_sum = np.sum(h_new)
        diff = target_sum - new_sum
        if abs(diff) > 1e-8:
            # Adjust remaining diff by spreading it across all positions
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
        return h_new

    h = project(h, T)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    for _ in range(10):  # Increased iterations for better convergence
        if time.time() - start_time > budget_s:
            break
        # Coordinate-wise optimization
        for i in range(n_points):
            previous_h = h.copy()
            # Objective function for scalar optimization
            def objective(x):
                temp_h = previous_h.astype(float).copy()
                temp_h[i] = x
                corr = np.correlate(temp_h, 1 - temp_h, mode='full')
                return np.max(corr) * dx
            # Use bounded optimization
            res = minimize_scalar(objective, bounds=(0, 1), method='bounded', tol=1e-8)
            h[i] = res.x
        # Project to maintain sum constraint
        h = project(h, T)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return best_h, final_c5, n_points
