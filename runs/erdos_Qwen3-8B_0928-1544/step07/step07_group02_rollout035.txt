import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())

    # Generate structured initial guesses
    x_vals = np.linspace(0, 2, n_points)
    # Square wave: 1 on [0, 1], 0 on [1, 2]
    square_wave = np.zeros(n_points)
    square_wave[:ones_count] = 1.0
    initial_guesses.append(square_wave.copy())

    # Sine wave: 0.5 + 0.5 * sin(πx)
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x_vals)
    initial_guesses.append(sine_wave.copy())

    # Triangular wave: 1 on [0, 1], 0 on [1, 2] with linear ramps
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_vals[i]
        if x < 1.0:
            triangular_wave[i] = x
        else:
            triangular_wave[i] = 2.0 - x
    # Normalize to sum to ones_count
    triangular_wave = triangular_wave / np.max(triangular_wave) * ones_count
    initial_guesses.append(triangular_wave.copy())

    # Random guesses
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (ones_count / np.sum(h_rand))
        initial_guesses.append(h_rand)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 250, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
