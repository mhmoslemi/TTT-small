import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize population
    pop_size = 200
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        normalizer = np.sum(h)
        h = h * (ones_count / normalizer)
        h = np.clip(h, 0, 1)
        pop.append(h)

    start_time = time.time()
    max_gen = 50
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fit = [-eval_c5(h) for h in pop]
        idx = np.argsort(fit)
        top = pop[idx[:int(len(pop)*0.2)]]
        new_pop = []
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(top, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            if np.random.rand() < 0.1:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.05)
                        child[i] = np.clip(child[i], 0, 1)
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (ones_count / child_sum)
            else:
                child = np.zeros_like(child)
            child = np.clip(child, 0, 1)
            new_pop.append(child)
        pop = new_pop
    
    best_idx = np.argmin([-eval_c5(h) for h in pop])
    best_h = pop[best_idx]
    best_c5 = eval_c5(best_h)

    # Local refinement
    def objective(h):
        return eval_c5(h)
    def constraint_sum(h):
        return np.sum(h) - ones_count
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(objective, best_h, method='SLSQP',
                      bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
                      options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8})
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (ones_count / np.sum(best_h))
    return best_h, best_c5, n_points
