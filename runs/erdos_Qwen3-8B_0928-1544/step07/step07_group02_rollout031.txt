        np.random.shuffle(np.arange(n_points))
        for i in range(n_points):
            h_temp = best_h.copy()
            def obj(h_i):
                h_temp[i] = h_i
                return compute_c5(h_temp)
            res = minimize_scalar(obj, bounds=(0., 1.), method='bounded')
            if res.success:
                best_h[i] = res.x
        best_h = project_h(best_h, target_sum)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
