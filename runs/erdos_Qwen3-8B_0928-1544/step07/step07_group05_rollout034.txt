import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    def scale_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        factor = ones_count / total
        return h * factor
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    population_size = 50
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        population = [scale_h(initial_h)]
    else:
        population = [np.full(n_points, 0.5)]
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        population.append(scale_h(h))
    
    best_h = population[0]
    best_c5 = compute_c5(population[0])
    start_time = time.time()
    
    generations = 200
    tournament_size = 3
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([compute_c5(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        
        new_population = []
        for _ in range(population_size):
            tournament = random.sample(range(population_size), tournament_size)
            best_tourn = min(tournament, key=lambda x: fitness[x])
            parent1 = population[best_tourn]

                    if parent2_idx != best_tourn:

                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1
