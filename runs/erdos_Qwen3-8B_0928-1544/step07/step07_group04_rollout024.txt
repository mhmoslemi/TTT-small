import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    def optimize_with_perturbations(start_h):
        best_h_local = start_h.copy()
        best_c5_local = objective(start_h)
        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturbed = start_h + perturb
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (required_sum / np.sum(h_perturbed))
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_h = result.x
                current_sum = np.sum(current_h)
                if not np.isclose(current_sum, required_sum, atol=1e-6):
                    current_h = current_h * (required_sum / current_sum)
                current_h = np.clip(current_h, 0.0, 1.0)
                current_c5 = objective(current_h)
                if current_c5 < best_c5_local:
                    best_c5_local = current_c5
                    best_h_local = current_h
                start_h = best_h_local
        return best_h_local, best_c5_local
    
    res1, c1 = optimize_with_perturbations(best_h)
    res2, c2 = optimize_with_perturbations(res1)
    res3, c3 = optimize_with_perturbations(res2)
    
    best_h_final = res3
    best_c5_final = min(c1, c2, c3)
    
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.02, n_points)
        h_perturbed = best_h_final + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (required_sum / np.sum(h_perturbed))
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_sum = np.sum(current_h)
            if not np.isclose(current_sum, required_sum, atol=1e-6):

            if current_c5 < best_c5_final:
                best_c5_final = current_c5
                best_h_final = current_h
    
    return best_h_final, best_c5_final, n_points
