import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

    target_sum = len(h) // 2
    n_points = len(h)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')
        lag_index = np.argmax(corr)
        lag_k = lag_index - (n_points - 1)

        valid_mask = (idx >= 0) & (idx < n_points)
        coefficients = np.zeros(n_points)
        coefficients[valid_mask] = 1 - h[valid_mask]
        h_opt = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_opt * coefficients))
        constraints = [0 <= h_opt, h_opt <= 1, cp.sum(h_opt) == target_sum]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS)

            break
        if prob.status != 'OPTIMAL':

        scale = target_sum / sum_new

        h = h_new.copy()
        curr_c5 = compute_c5(h, n_points)
        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n_points)
