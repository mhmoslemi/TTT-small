import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    h = initial_h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    T = 10.0
    cooling_rate = 0.995
    noise_std = 0.1
    steps = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        steps += 1
        noise = np.random.normal(0, noise_std, n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            continue
        scaling = ones_count / sum_new
        new_h = new_h * scaling
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        T *= cooling_rate
        if steps % 100 == 0:
            noise_std = max(noise_std * 0.95, 0.01)
    
    return best_h, best_c5, n_points
