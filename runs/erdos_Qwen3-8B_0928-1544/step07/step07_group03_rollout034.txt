import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    bounds = [(0.0, 1.0)] * n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def cons(h):
        return np.sum(h) - target_sum

    # Particle Swarm Optimization
    n_particles = 50
    inertia = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 100
    particles = np.random.rand(n_particles, n_points)
    if 'initial_h_values' in globals():
        particles += np.random.normal(0, 0.1, particles.shape)
        particles = np.clip(particles, 0, 1)
        particles *= (target_sum / np.sum(particles, axis=1, keepdims=True))
    else:
        particles *= (target_sum / np.sum(particles, axis=1, keepdims=True))
    velocities = np.random.rand(n_particles, n_points) * 0.1
    personal_best = particles.copy()
    personal_best_values = np.array([obj(p) for p in particles])
    global_best = particles[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        r1, r2 = np.random.rand(n_particles, n_points), np.random.rand(n_particles, n_points)
        velocities = inertia * velocities + c1 * r1 * (personal_best - particles) + c2 * r2 * (global_best - particles)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            particles[i] *= (target_sum / np.sum(particles[i]))
        current_values = np.array([obj(p) for p in particles])
        improved = current_values < personal_best_values
        personal_best[improved] = particles[improved]
        personal_best_values[improved] = current_values[improved]
        if np.min(current_values) < global_best_value:
            global_best = particles[np.argmin(current_values)]
            global_best_value = np.min(current_values)

    best_h = global_best
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons},
                   options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
    best_c5 = res.fun
    return res.x, best_c5, n_points
