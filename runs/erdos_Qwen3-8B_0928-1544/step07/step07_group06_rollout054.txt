import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Hybrid optimization: GA + gradient refinement
    def evolve_population(pop, generations=100, mutation_rate=0.1):
        for _ in range(generations):
            p = np.random.permutation(len(pop))
            new_pop = []
            for i in range(0, len(pop), 2):
                p1, p2 = pop[p[i]], pop[p[i+1]]
                # Crossover
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                # Mutation
                child = np.clip(child + np.random.randn(n_points) * mutation_rate, 0, 1)
                child = child * (ones_count / np.sum(child))
                new_pop.append(child)
            pop = np.array(new_pop + [pop[i] for i in range(len(pop)) if np.random.rand() < 0.1])
            # Evaluate and sort
            fitness = [objective(ind) for ind in pop]
            pop = pop[np.argsort(fitness)][:len(pop)//2]
        return pop
    
    def refine_solution(h):
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        return result.x
    
    # Initial population
    pop_size = 50
    pop = np.random.rand(pop_size, n_points)
    pop = pop / np.sum(pop, axis=1, keepdims=True) * ones_count
    pop_fitness = [objective(ind) for ind in pop]
    
    # GA phase
    ga_pop = evolve_population(pop, generations=300)
    best_idx = np.argmin([objective(ind) for ind in ga_pop])
    best_h = ga_pop[best_idx]

    for _ in range(10):

    # Final refinement with gradient-based steps
    if time.time() - start_time < budget_s - 10:
