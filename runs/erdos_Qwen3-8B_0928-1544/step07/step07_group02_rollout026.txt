from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2

    h_sine = h_sine * (required_sum / current_sum)
    initial_guesses.append(h_sine)

    h_two_intervals = np.zeros(n_points)
    indices_low = np.where((x >= 0.5) & (x <= 1.0))[0]
    indices_high = np.where((x >= 1.5) & (x <= 2.0))[0]
    h_two_intervals[indices_low] = 1.0
    h_two_intervals[indices_high] = 1.0
    current_sum = np.sum(h_two_intervals)
