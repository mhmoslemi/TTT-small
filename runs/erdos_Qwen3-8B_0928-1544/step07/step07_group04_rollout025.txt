from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def optimize_with_perturbations(start_h):
        best_h_local = start_h.copy()
        best_c5_local = objective(start_h)
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            h_perturbed = start_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            norm = np.sum(h_perturbed)
            if norm != 0:
                h_perturbed = h_perturbed * (ones_count / norm)
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_local:
                    best_c5_local = current_c5

        return best_h_local, best_c5_local
    
    res1, c1 = optimize_with_perturbations(best_h)
    res2, c2 = optimize_with_perturbations(res1)

    best_c5_final = min(c1, c2, c3)
