def run(seed=42, budget_s=1100, **kwargs):

    target_sum = n_points / 2

    generations = 200

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        initial_pop[i] = h

        return np.max(corr) * (2.0 / n_points)
    best_h = initial_pop[0]
    best_c5 = evaluate_candidate(best_h)
    start_time = time.time()

        new_population = [best_h]
        for _ in range(population_size - 1):

            child += mutation

            child = child * (target_sum / np.sum(child))
            new_population.append(child)
        initial_pop = new_population
        for h in initial_pop:
