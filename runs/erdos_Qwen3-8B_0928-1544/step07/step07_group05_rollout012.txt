import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35  # 70/2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Enforce sum constraint
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    T = 1.0  # Initial temperature
    alpha = 0.995  # Cooling rate
    perturb_rate = 0.1  # 10% of points to flip
    patience_counter = 0
    
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 0.1 * budget_s:
            break
        
        # Perturb
        num_perturb = int(n_points * perturb_rate)
        flip_indices = np.random.choice(n_points, num_perturb, replace=False)
        new_h = best_h.copy()
        new_h[flip_indices] = 1 - new_h[flip_indices]
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * (ones_count / np.sum(new_h))
        
        # Evaluate
        new_c5 = objective(new_h)
        delta_e = new_c5 - best_c5
        
        # Acceptance
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            best_h = new_h.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adapt perturbation rate
        if patience_counter > 50:
            perturb_rate = min(0.2, perturb_rate * 1.1)
            patience_counter = 0
        
        T *= alpha
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (ones_count / np.sum(best_h))
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
