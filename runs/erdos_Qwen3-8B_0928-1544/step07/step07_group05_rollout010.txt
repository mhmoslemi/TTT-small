import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals():
        initial_pop = [initial_h_values.copy()]
    else:
        initial_pop = [np.full(n_points, 0.5)]

    # Generate remaining population with perturbed initial
    for _ in range(49):
        curr = initial_pop[0].copy()
        noise = np.random.normal(0, 0.3, n_points)
        curr += noise
        curr = np.clip(curr, 0, 1)
        curr = curr * (ones_count / np.sum(curr))
        initial_pop.append(curr)

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h))
        return h

    pop = np.array(initial_pop)
    start_time = time.time()
    gen = 0

    while time.time() - start_time < budget_s:
        gen += 1
        # Evaluate fitness
        fit = np.array([fitness(ind) for ind in pop])
        # Track best
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5 = fit[best_idx]
            best_h = pop[best_idx].copy()
        # Tournament selection
        new_pop = []
        for _ in range(len(pop)):
            a, b = np.random.randint(len(pop), size=2)
            candidates = [fit[a], fit[b], fit[np.random.randint(len(pop))]]
            winner = np.argmin(candidates)
            parent = pop[winner]
            new_pop.append(parent)
        # Crossover
        for i in range(0, len(new_pop), 2):
            if np.random.rand() < 0.8:
                p1, p2 = new_pop[i], new_pop[i+1]
                split = np.random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                new_pop[i], new_pop[i+1] = normalize(child1), normalize(child2)
        # Mutation
        for i in range(len(new_pop)):
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    new_pop[i][j] += np.random.normal(0, 0.1)
                    new_pop[i][j] = normalize(new_pop[i])[j]
        pop = np.array(new_pop)
        if gen > 200:
            break

    return best_h, best_c5, n_points
