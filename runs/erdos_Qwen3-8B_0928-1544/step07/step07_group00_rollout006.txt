from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    def project_feasible(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if np.isclose(sum_h, 0.0):
            return h_clipped
        scale = (n_points / 2) / sum_h
        return h_clipped * scale
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        step_size = 0.1 * (1.0 - (time.time() - start_time) / budget_s)
        h_perturbed = h + np.random.normal(0, step_size, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
