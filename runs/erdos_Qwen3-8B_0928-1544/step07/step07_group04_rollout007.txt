from scipy.optimize import minimize
import time

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h
    best_c5 = objective(best_h)
    start_time = time.time()

        best_h_local = start_h.copy()
        best_c5_local = objective(start_h)

            if time.time() - start_time > budget_s:
                break

            h_perturbed = start_h + perturb

                fun=objective,
                x0=h_perturbed,
                method='L-BFGS-B',

            )
