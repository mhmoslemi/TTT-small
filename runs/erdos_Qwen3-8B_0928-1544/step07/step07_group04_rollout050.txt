import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    def rescale(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (ones_count / total)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    population = [initial_h.copy() for _ in range(30)]
    for i in range(len(population)):
        perturb = np.random.normal(0, 0.1, n_points)
        population[i] += perturb
        population[i] = rescale(population[i])

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_c5 = [objective(ind) for ind in population]
        top_indices = np.argsort(current_c5)[:int(len(population)*0.3)]
        top_candidates = [population[i] for i in top_indices]

        new_population = top_candidates[:]
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(top_candidates, 2, replace=False)
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            mut = np.random.normal(0, 0.05, n_points)
            child += mut
            new_population.append(rescale(child))
        population = new_population

        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if time.time() - start_time >= budget_s:
            break

    return best_h, best_c5, n_points
