```python
import numpy as np
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    
    # Initialize h
    h = np.zeros(n_points)
    if 'initial_h_values' in locals():
        h = initial_h_values.copy()
    else:
        h[:int(n_points/2)] = 1.0
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for _ in range(5):
        indices = np.random.permutation(n_points)
        for i in indices:
            def obj(x):
                h_temp = h.copy()
                h_temp[i] = x
                return compute_c5(h_temp)
            res = minimize_scalar(obj, bounds=(0,1), method='bounded')
            h[i] = res.x
        
        # Projection
        h_proj = cp.Variable(n_points)
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_proj - h)),
                          [cp.sum(h_proj) == target_sum, 0 <= h_proj, h_proj <= 1])
        prob.solve()
        h = h_proj.value
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if time.time() - start_time > budget_s:
            break
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
```
