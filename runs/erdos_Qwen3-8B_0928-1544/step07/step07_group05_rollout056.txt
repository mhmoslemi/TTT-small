import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    max_iter = 1000
    alpha = 0.1
    decay_rate = 0.99
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if (j + k) < n_points:
                gradient[j] += (1 - h[j + k]) * dx
            if (j - k) >= 0:
                gradient[j] -= h[j - k] * dx
        # Update h
        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= (n_points / 2) / sum_h
        # Evaluate current C5
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Decay learning rate
        alpha *= decay_rate
    
    return best_h, best_c5, n_points
