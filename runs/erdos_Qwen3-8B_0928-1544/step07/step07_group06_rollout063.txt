import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (ones_count / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # GA Phase
    pop_size = 100
    gen_limit = 50
    population = [best_h.copy() for _ in range(pop_size)]
    best_idx = np.argsort([objective(ind) for ind in population])[:int(pop_size/2)]

    for gen in range(gen_limit):
        if time.time() - start_time > budget_s * 0.6:
            break
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(best_idx, 2, replace=False)
            child = (population[p1] + population[p2]) / 2
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                child[idx] += np.random.normal(0, 0.1)
            child = child * (ones_count / np.sum(child))
            new_pop.append(child)
        population = new_pop
        current_fit = [objective(ind) for ind in population]
        best_idx = np.argsort(current_fit)[:int(pop_size/2)]
        best_h = population[best_idx[0]].copy()
        best_c5 = objective(best_h)

    # Refinement
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    # Hybridization
    second = population[1].copy() if len(population) > 1 else best_h.copy()
    hybrid = (best_h + second) / 2
    hybrid = hybrid * (ones_count / np.sum(hybrid))
    hybrid_c5 = objective(hybrid)
    if hybrid_c5 < best_c5:
        best_h = hybrid
        best_c5 = hybrid_c5

    # Final normalization
    best_h = best_h * (ones_count / np.sum(best_h))
    return best_h, best_c5, n_points
