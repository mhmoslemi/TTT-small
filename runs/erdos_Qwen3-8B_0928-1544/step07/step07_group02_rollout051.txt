import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())

    # Checkerboard pattern
    structured_checkered = np.zeros(n_points)
    structured_checkered[::2] = 1.0
    structured_checkered = structured_checkered * (ones_count / np.sum(structured_checkered))
    initial_guesses.append(structured_checkered)

    # Sine wave pattern
    x = np.linspace(0, 2, n_points)
    sine_base = 0.5 * (1 + np.sin(2 * np.pi * x / (n_points / 2)))
    sine_clamped = np.clip(sine_base, 0, 1)
    sine_total = np.sum(sine_clamped)
    sine_scaled = sine_clamped * (ones_count / sine_total)
    initial_guesses.append(sine_scaled)

    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (ones_count / np.sum(h_rand))
        initial_guesses.append(h_rand)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
