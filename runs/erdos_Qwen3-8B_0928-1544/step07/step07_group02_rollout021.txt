    def objective(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2

    def optimize_stage(initial_guesses, n_points, maxiter=200):
        best_c5 = float('inf')
        best_h = None
        for guess in initial_guesses:
            result = minimize(
                fun=lambda x: objective(x, n_points),
                x0=guess,
                method='SLSQP',
                bounds=[(0,1)]*n_points,
                constraints={'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)},
                options={'maxiter': maxiter, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        return best_h, best_c5

    n_stages = 3
    stages = [14, 35, 70]
    best_h = None
    elapsed = 0
    for i in range(n_stages):
        n = stages[i]
        initial_guesses = []
        if i == 0 and initial_h_values is not None:
            initial_guesses.append(downsample(initial_h_values, n))
        for _ in range(5):
            h_rand = np.random.rand(n)
            h_rand = h_rand * (n // 2) / np.sum(h_rand)
            initial_guesses.append(h_rand)
        for phase in np.linspace(0, 2 * np.pi, 5):
            h_sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n + phase)
            h_sine = h_sine * (n // 2) / np.sum(h_sine)
            initial_guesses.append(h_sine)
        start = time.time()
        best_h_current, best_c5 = optimize_stage(initial_guesses, n)
        elapsed += time.time() - start
        if elapsed > budget_s:
            break
        if i == 2:
            best_h = best_h_current
            best_c5_bound = best_c5
            n_points = n
            break
        if i < 2:
            best_h = upsample(best_h_current, stages[i+1])
    return best_h, best_c5_bound, n_points
