import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)

        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (ones_count / np.sum(perturbed))

        result = minimize(
            fun=objective,

            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
