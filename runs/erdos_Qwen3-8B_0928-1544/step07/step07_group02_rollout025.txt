import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())

    # Generate 5 random and 5 structured initial guesses
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (ones_count / np.sum(h_rand))
        initial_guesses.append(h_rand)
    
    # Add structured patterns: cosine and alternating blocks
    for _ in range(5):
        # Cosine pattern
        h_cos = 0.5 * (1 + np.cos(2 * np.pi * np.linspace(0, 1, n_points)))
        h_cos = h_cos * (ones_count / np.sum(h_cos))
        initial_guesses.append(h_cos)
        
        # Alternating blocks
        block_size = n_points // 4
        h_block = np.zeros(n_points)
        h_block[:block_size] = 1.0
        h_block[block_size:2*block_size] = 0.0
        h_block[2*block_size:3*block_size] = 1.0
        h_block[3*block_size:] = 0.0
        h_block = h_block * (ones_count / np.sum(h_block))
        initial_guesses.append(h_block)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
