import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

    alpha = 0.01
    start_time = time.time()
    
    for _ in range(10000):
        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        grad = (1 - 2 * h) * np.sign(corr)

        h = np.clip(h, 0.0, 1.0)
        h = h * (ones_count / np.sum(h))

    return best_h, best_c5, n_points
