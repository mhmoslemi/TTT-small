import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def evaluate_erdos(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    swarm_size = 50
    particles = np.random.uniform(0, 1, (swarm_size, n_points))
    particles = np.array([h / h.sum() * (n_points / 2) for h in particles])
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    best_positions = particles.copy()
    best_costs = np.full(swarm_size, np.inf)
    
    global_best = initial_h.copy()
    global_cost = evaluate_erdos(global_best)
    max_iter = 500
    w = 0.9
    w_decrement = 0.9 / max_iter * 0.5
    c1, c2 = 2.0, 2.0
    start_time = time.time()
    
    for iteration in range(max_iter):
        w = max(0.4, w - w_decrement)
        costs = np.array([evaluate_erdos(p) for p in particles])
        mask = costs < best_costs
        best_positions[mask] = particles[mask]
        best_costs[mask] = costs[mask]
        if np.min(costs) < global_cost:
            global_cost = np.min(costs)
            global_best = particles[np.argmin(costs)]
        
        for i in range(swarm_size):
            rand1, rand2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * rand1 * (best_positions[i] - particles[i]) + c2 * rand2 * (global_best - particles[i])
        particles += velocities
        particles = np.clip(particles, 0, 1)
        particles = np.array([h / h.sum() * (n_points / 2) for h in particles])
        if time.time() - start_time > budget_s:
            break
    
    def objective(h):
        return evaluate_erdos(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=global_best,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    best_h = result.x if result.success else global_best
    best_c5 = result.fun if result.success else global_cost
