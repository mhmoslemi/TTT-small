import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Optimal small size for fast evaluation
    ones_count = n_points // 2
    
    # Use provided initial guess if exists
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()

    start_time = time.time()
    
    def optimize(start_h):
        best_h_loc = start_h.copy()

                x0=h_perturbed,
