        initial_guesses.append(h_sine)

    best_c5_coarse = float('inf')
    best_h_coarse = None
    start_time = time.time()
    for guess in initial_guesses:
        guess = guess * ones_count_coarse / np.sum(guess)
        res = minimize(
            fun=lambda h: compute_c5(h, n_coarse),
            x0=guess,
            method='COBYLA',
            bounds=[(0.0, 1.0)] * n_coarse,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count_coarse},
            options={'maxiter': 100, 'disp': False}
        )
        if res.success:
            h_proj = np.clip(res.x, 0, 1)
            h_proj = h_proj * ones_count_coarse / np.sum(h_proj)
            current_c5 = compute_c5(h_proj, n_coarse)
            if current_c5 < best_c5_coarse:
                best_c5_coarse = current_c5
                best_h_coarse = h_proj
        if time.time() - start_time > budget_s:
            break

    if best_h_coarse is None:
        best_h_coarse = np.zeros(n_coarse)
        best_h_coarse[0] = 1.0

    h_medium = upsample(best_h_coarse, n_coarse, n_medium)
    h_medium = h_medium * ones_count_medium / np.sum(h_medium)
    best_c5_medium = float('inf')
    best_h_medium = None
    start_time = time.time()
    res = minimize(
        fun=lambda h: compute_c5(h, n_medium),
        x0=h_medium,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n_medium,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count_medium},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    if res.success:
        h_proj = np.clip(res.x, 0, 1)
        h_proj = h_proj * ones_count_medium / np.sum(h_proj)
        best_c5_medium = compute_c5(h_proj, n_medium)
        best_h_medium = h_proj
    if time.time() - start_time > budget_s:
        return best_h_medium, best_c5_medium, n_medium

    h_high = upsample(best_h_medium, n_medium, n_high)
    h_high = h_high * ones_count_high / np.sum(h_high)
    best_c5_high = float('inf')
    best_h_high = None
    start_time = time.time()
    res = minimize(
        fun=lambda h: compute_c5(h, n_high),
        x0=h_high,
        method='COBYLA',
        bounds=[(0.0, 1.0)] * n_high,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count_high},
        options={'maxiter': 500, 'disp': False}
    )
    if res.success:
        h_proj = np.clip(res.x, 0, 1)
        h_proj = h_proj * ones_count_high / np.sum(h_proj)
        best_c5_high = compute_c5(h_proj, n_high)
        best_h_high = h_proj
        if best_c5_high < best_c5_medium:
            return best_h_high, best_c5_high, n_high

    return best_h_medium, best_c5_medium, n_medium
