import numpy as np
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in locals():
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.random.rand(n_points)
        initial_guess = initial_guess * (target_sum / np.sum(initial_guess))
    h = initial_guess.copy()
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    for pass_num in range(5):  # Number of passes
        permuted_indices = np.random.permutation(n_points)
        for i in permuted_indices:
            def objective(x):
                h_new = h.copy()
                h_new[i] = x
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                return np.max(corr) * dx
            
            result = minimize_scalar(objective, bounds=(0, 1), method='bounded', tol=1e-5)
            h[i] = result.x
        
        # Projection step
        current_sum = np.sum(h)
        delta = (target_sum - current_sum)
        if abs(delta) > 1e-5:
            h += delta / n_points
            h = np.clip(h, 0, 1)
        
        # Compute current C5 and update best_c5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if time.time() - start_time > budget_s:
            break
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return best_h, final_c5, n_points
