def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2

        h = initial_h_values.copy()

        h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def project(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (ones_count / total)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Gradient descent with adaptive learning rate
    def optimize_with_gradient_descent(start_h, max_iter=200):
        h = start_h.copy()
        learning_rate = 0.05
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            # Compute finite difference gradient
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project(h_plus)
                c_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project(h_minus)
                c_minus = compute_c5(h_minus)
                grad[i] = (c_plus - c_minus) / (2 * eps)
            # Update h
            h = h - learning_rate * grad
            h = project(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        return best_h, best_c5
    
    # Local optimization with perturbations and convex relaxation
    def optimize_with_convex(start_h, max_iter=100):

                for i in range(n_points):

                    integral += h[i] * (1 - h[j])
                integral *= (2.0 / n_points)
                constraints.append(tau >= integral)
