    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = (initial_h >= 0.5).astype(int)
        current_sum = np.sum(initial_h)
        if current_sum != ones_needed:
            diff = current_sum - ones_needed
            if diff > 0:
                indices = np.where(initial_h == 1)[0]
                np.random.shuffle(indices)
                for i in range(diff):
                    initial_h[indices[i]] = 0
            else:
                indices = np.where(initial_h == 0)[0]
                np.random.shuffle(indices)
                for i in range(-diff):
                    initial_h[indices[i]] = 1
        population.append(initial_h)
        for _ in range(pop_size - 1):
            h = np.zeros(n_points, dtype=int)
            h[:ones_needed] = 1
            np.random.shuffle(h)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.zeros(n_points, dtype=int)
            h[:ones_needed] = 1
            np.random.shuffle(h)
            population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = np.sort(fitness)

        λ = pop_size // 2
        parents = sorted_population[:λ]

        offspring = []
        for _ in range(pop_size - λ):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros(n_points, dtype=int)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]

            if np.random.rand() < 0.05:
                n_mutate = int(n_points * 0.05)
                indices = np.random.choice(n_points, n_mutate, replace=False)
                child[indices] = 1 - child[indices]

            def repair(h):
                current_sum = np.sum(h)
                if current_sum == ones_needed:
                    return h
                elif current_sum > ones_needed:
                    excess = current_sum - ones_needed
                    indices = np.where(h == 1)[0]
                    np.random.shuffle(indices)
                    for i in range(excess):
                        h[indices[i]] = 0
                else:
                    deficit = ones_needed - current_sum
                    indices = np.where(h == 0)[0]
                    np.random.shuffle(indices)
                    for i in range(deficit):
                        h[indices[i]] = 1
                return h
            child = repair(child)
            offspring.append(child)

        new_population = parents + offspring
        population = new_population

        current_best_fitness = sorted_fitness[0]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = sorted_population[0]

    return best_h.astype(float), best_c5, n_points
