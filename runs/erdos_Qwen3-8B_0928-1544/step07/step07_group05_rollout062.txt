import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.1
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        max_corr = np.max(corr)
        if max_corr == 0:
            break
        
        # Find the shift with maximum overlap
        shift_idx = np.argmax(corr)
        shift = shift_idx - (n_points - 1)
        
        # Adjust h to reduce overlap at the identified shift
        # The overlap at shift k is sum_x h[x] * (1 - h[x + k])
        # To reduce this, we can adjust h[x] and h[x + k]

        start = max(0, -shift)
        end = min(n_points, n_points - shift)
        overlap_region = h[start:end]
        overlap_region_1 = h[start:end + shift] if shift > 0 else h
        
        # Compute gradient direction: reduce overlap by adjusting h

        grad = np.zeros_like(h)
        grad[start:end] -= 0.5 * overlap_region
