import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []
    
    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())
    
    def generate_sinusoidal_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = 1.0 + 0.5 * np.sin(np.pi * (x - 0.5))
        h = np.clip(h, 0.0, 1.0)
        h = (h / np.sum(h)) * ones_count
        return h
    
    for _ in range(3):
        guess = generate_sinusoidal_guess(n_points)
        initial_guesses.append(guess)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
