def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2  # 35

    # Define initial guesses
    initial_guesses = []

    # Add initial_h_values if available
    if 'initial_h_values' in globals():

    # Generate random initial guesses
    for _ in range(5):
        # Generate a valid random guess
        while True:
            h_rand = np.random.rand(n_points)
            S = np.sum(h_rand)
            if S >= ones_count:
                h = h_rand * (ones_count / S)
                # Clip to [0,1] just in case
                h = np.clip(h, 0, 1)
                # Check that sum is approximately ones_count
                if np.isclose(np.sum(h), ones_count, atol=1e-6):
                    initial_guesses.append(h)
                break

    # Generate structured initial guesses (sinusoidal patterns)
    freqs = [0.5, 1.0, 2.0, 0.1, 0.2]
    for freq in freqs:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * freq * x))
        # Ensure the sum is approximately ones_count
        h_scaled = h * (ones_count / np.sum(h))
        h_scaled = np.clip(h_scaled, 0, 1)
        if np.isclose(np.sum(h_scaled), ones_count, atol=1e-6):
            initial_guesses.append(h_scaled)

    # Now, perform optimization for each initial guess
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=[(0.0, 1.0)] * n_points,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
