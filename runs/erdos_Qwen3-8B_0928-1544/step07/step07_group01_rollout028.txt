import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 100
    tournament_size = 5
    mutation_rate = 0.01
    n_generations = 1000
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.zeros(n_points, dtype=int)

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        binary_initial = np.round(initial_h).astype(int)
        diff = ones_count - np.sum(binary_initial)
        if diff != 0:
            zero_positions = np.where(binary_initial == 0)[0]
            one_positions = np.where(binary_initial == 1)[0]
            if diff > 0:
                if len(zero_positions) >= diff:
                    indices_to_flip = np.random.choice(zero_positions, size=diff, replace=False)
                    binary_initial[indices_to_flip] = 1
            else:
                if len(one_positions) >= -diff:
                    indices_to_flip = np.random.choice(one_positions, size=-diff, replace=False)
                    binary_initial[indices_to_flip] = 0
        population = [np.copy(binary_initial) for _ in range(pop_size)]
    else:
        population = []
        for _ in range(pop_size):
            positions = np.random.choice(n_points, size=ones_count, replace=False)
            binary_vec = np.zeros(n_points, dtype=int)
            binary_vec[positions] = 1
            population.append(binary_vec)
    
    # Main GA loop
    for generation in range(n_generations):
        # Evaluate fitness of all individuals
        fitness = []
        for ind in population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            c5 = max_corr * dx
            fitness.append(c5)
        # Selection: Tournament selection
        selected_parents = []
        for _ in range(pop_size):
            tournament = np.random.choice(range(pop_size), size=tournament_size, replace=False)
            best = tournament[np.argmin([fitness[i] for i in tournament])]
            selected_parents.append(population[best])
        # Crossover
        next_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected_parents[i]
            parent2 = selected_parents[i+1]
            crossover_point = np.random.randint(1, n_points)
            child1 = parent1.copy()
            child2 = parent2.copy()
            child1[crossover_point:] = parent2[crossover_point:]
            child2[crossover_point:] = parent1[crossover_point:]
            next_population.append(child1)
            next_population.append(child2)
        # Mutation
        for i in range(pop_size):
            child = next_population[i]
            if np.random.rand() < mutation_rate:
                zero_positions = np.where(child == 0)[0]
                one_positions = np.where(child == 1)[0]
                if len(zero_positions) > 0 and len(one_positions) > 0:
                    flip_zero = np.random.choice(zero_positions)
                    flip_one = np.random.choice(one_positions)
                    child[flip_zero], child[flip_one] = 1, 0
            next_population[i] = child
        population = next_population
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Evaluate the best individual
    fitness = []
    for ind in population:
        corr = np.correlate(ind, 1 - ind, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        c5 = max_corr * dx
        fitness.append(c5)
    best_index = np.argmin(fitness)
    best_h = population[best_index].astype(float)
    best_c5 = fitness[best_index]
    return best_h, best_c5, n_points
