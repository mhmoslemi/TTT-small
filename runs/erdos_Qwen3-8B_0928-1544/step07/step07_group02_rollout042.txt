import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())

    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (ones_count / np.sum(h_rand))
        initial_guesses.append(h_rand.copy())

    structured_guesses = []
    # Step function
    structured_guesses.append(np.concatenate([np.ones(35), np.zeros(35)]))
    # Alternating pattern
    structured_guesses.append(np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)]))
    # Periodic pattern with period 3
    structured_guesses.append(np.array([1.0 if (i % 3) < 2 else 0.0 for i in range(n_points)]))
    # Periodic pattern with period 4
    structured_guesses.append(np.array([1.0 if (i % 4) < 3 else 0.0 for i in range(n_points)]))
    # Sine wave
    sine = (np.sin(np.linspace(0, 2*np.pi, n_points)) + 1) / 2
    sum_sine = np.sum(sine)
    sine_scaled = sine * (35.0 / sum_sine)
    structured_guesses.append(sine_scaled)
    initial_guesses.extend(structured_guesses)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
