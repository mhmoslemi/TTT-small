            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5_coarse:
                best_c5_coarse = current_c5
                best_coarse = result.x

    def upsample(h, n_new):
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, h)
        h_new = h_new * (n_new / 2) / np.sum(h_new)
        return h_new

    h_medium = upsample(best_coarse, n_points_medium)
    bounds_medium = [(0.0, 1.0)] * n_points_medium
    best_medium = np.zeros(n_points_medium)
    best_c5_medium = float('inf')

    result_medium = minimize(
        fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points_medium),
        x0=h_medium,
        method='SLSQP',
        bounds=bounds_medium,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - ones_medium},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if result_medium.success:
        best_c5_medium = result_medium.fun
        best_medium = result_medium.x

    h_final = upsample(best_medium, n_points_final)
    bounds_final = [(0.0, 1.0)] * n_points_final

    result_final = minimize(
        fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points_final),
        x0=h_final,
        method='COBYLA',
        bounds=bounds_final,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - ones_final},
        options={'maxiter': 500, 'ftol': 1e-9}
    )

    if result_final.success:
        best_c5 = result_final.fun
        best_h = result_final.x
    else:
        best_c5 = best_c5_medium
        best_h = best_medium

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points_final)
    return (best_h, best_c5, n_points_final)
