import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    hm = [initial_h.copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        target_sum = ones_count
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum) if current_sum > 0 else h
        h = np.clip(h, 0, 1)
        hm.append(h.copy())

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = hm[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    max_iter = int(budget_s / 0.1)
    for _ in range(max_iter):
        if np.random.rand() < 0.2:
            new_h = best_h.copy()
        elif np.random.rand() < 0.5:
            new_h = hm[np.random.randint(len(hm))].copy()
        else:
            new_h = hm[np.random.randint(len(hm))].copy()
            perturb_idx = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            for i in perturb_idx:
                new_h[i] += np.random.uniform(-0.1, 0.1)
        current_sum = np.sum(new_h)
        scale = ones_count / current_sum if current_sum != 0 else 1.0
        new_h = new_h * scale
        new_h = np.clip(new_h, 0, 1)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
