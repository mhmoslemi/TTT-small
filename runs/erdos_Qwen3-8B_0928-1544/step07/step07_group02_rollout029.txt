            constraints={'type': 'eq', 'fun': lambda h: constraint_sum(h)},
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if result.success:
            h_clipped = np.clip(result.x, 0.0, 1.0)
            h_scaled = h_clipped * (n_medium / 2) / np.sum(h_clipped)
            c5 = objective(h_scaled, n_medium)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_scaled.copy()
        else:
            best_h = h_medium

        h_high = upsample(best_h, n_medium, n_high)
        result = minimize(
            fun=lambda h: objective(h, n_high),
            x0=h_high,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_high,
            constraints={'type': 'eq', 'fun': lambda h: constraint_sum(h)},
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if result.success:
            h_clipped = np.clip(result.x, 0.0, 1.0)
            h_scaled = h_clipped * (n_high / 2) / np.sum(h_clipped)
            c5 = objective(h_scaled, n_high)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_scaled.copy()
        if time.time() - start_time > budget_s:
            return best_h, best_c5, n_high

    return best_h, best_c5, n_high
