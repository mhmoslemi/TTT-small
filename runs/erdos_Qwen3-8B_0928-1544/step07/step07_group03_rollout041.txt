import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def cons(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    # Extended gradient descent with tighter constraints
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons},
                   options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9})
    
    best_h = res.x
    best_c5 = res.fun
    return best_h, best_c5, n_points
