import numpy as np

import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    population_size = 50
    population = []
    # Initialize population
    for _ in range(population_size):
        if _ == 0:
            h = initial_h.copy()
        else:
            h = initial_h.copy() + np.random.normal(0, 0.3, n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 3
    elitism_ratio = 0.1
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx]
        current_best_c5 = fitness[current_best_idx]

            best_c5 = current_best_c5

        new_population.extend([population[i] for i in elite_indices])
        
        while len(new_population) < population_size:
            # Tournament selection
            indices = np.random.choice(population_size, tournament_size)

            best_idx = np.argmin(selected_fitness)
            parent1 = selected[best_idx]
            
            indices2 = np.random.choice(population_size, tournament_size)
            selected2 = [population[i] for i in indices2]
            selected_fitness2 = [fitness[i] for i in indices2]
            best_idx2 = np.argmin(selected_fitness2)
