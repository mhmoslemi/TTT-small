        h = np.random.rand(n_points)
        h = np.clip(h * target_sum / np.sum(h), 0, 1)
        hamm.append(h)
    
    def c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def perturb(h, scale=0.05):
        h = h + np.random.uniform(-scale, scale, h.shape)
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        h = h * target_sum / s
        return h
    
    hamm = [perturb(h) for h in hamm]
    best_h, best_c = hamm[0].copy(), c5(hamm[0])
    
    start_time = time.time()
    iters = 0
    while time.time() - start_time < budget_s:
        new_h = np.random.choice(hamm)
        new_h = perturb(new_h)
        current_c = c5(new_h)
        if current_c < best_c:
            best_h, best_c = new_h, current_c
        c_list = [c5(h) for h in hamm]
        idx = np.argmax(c_list)
        if current_c < c_list[idx]:
            hamm[idx] = new_h
        iters += 1
    
    def obj(x):
        return c5(x)
    
    res = minimize(obj, best_h, bounds=[(0,1)]*n_points, constraints={'type':'eq','fun':lambda x: np.sum(x)-target_sum}, method='SLSQP')
    best_h = res.x
    best_c = res.fun
    return best_h, best_c, n_points
