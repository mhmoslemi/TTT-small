import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        # Enforce symmetry
        for i in range(n_points // 2):
            h[i], h[n_points - 1 - i] = (h[i] + h[n_points - 1 - i]) / 2, (h[i] + h[n_points - 1 - i]) / 2
    else:
        h = np.full(n_points, 0.5)
        for i in range(n_points // 2):
            h[i], h[n_points - 1 - i] = 0.5, 0.5
    
    def objective(x):
        h_full = np.zeros(n_points)
        h_full[:len(x)] = x
        for i in range(len(x)):
            h_full[n_points - 1 - i] = x[i]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - (n_points / 2)
    
    bounds = [(0.0, 1.0)] * (n_points // 2)
    best_h = h[:n_points // 2].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(2):
        if time.time() - start_time >= budget_s * 0.8:
            break
        n_points *= 2
        new_h = np.zeros(n_points // 2)
        new_h[:len(best_h)] = best_h
        # Interpolate to new grid
        for i in range(len(new_h)):
            new_h[i] = (best_h[i] + best_h[i + 1]) / 2 if i + 1 < len(best_h) else best_h[i]
        best_h = new_h
        best_c5 = objective(best_h)
    
    # Final optimization
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    
    if result.success:
        best_c5 = result.fun
        best_h = result.x
        n_points_final = len(best_h) * 2
    else:
        n_points_final = n_points // 2
    
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points_final
