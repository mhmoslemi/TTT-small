import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    def generate_candidate():
        c = np.random.rand(n_points)
        s = np.sum(c)
        if s == 0:
            return np.zeros(n_points)
        c = np.clip(c * (ones_count / s), 0.0, 1.0)
        return c

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    hm = [generate_candidate() for _ in range(50)]
    if 'initial_h_values' in globals() and not any(np.allclose(h, initial_h, atol=1e-6) for h in hm):
        hm.append(initial_h)
    hm = np.array(hm)
    hm_eval = np.array([evaluate(h) for h in hm])
    hm_sorted = np.argsort(hm_eval)
    best_idx = hm_sorted[0]
    best_h = hm[best_idx]
    best_c5 = hm_eval[best_idx]

    start_time = time.time()
    max_iter = int(budget_s / 0.1)
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

            h = np.random.choice(hm)

            h = generate_candidate()
        perturb = np.random.rand(n_points) * 0.2 - 0.1
        h += perturb

            h = np.clip(h * (ones_count / s), 0.0, 1.0)
        c = evaluate(h)

            hm[worst_idx] = h
