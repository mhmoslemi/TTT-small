from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
    
    result = minimize(

        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    # Additional perturbation-based fine-tuning
    best_h = result.x.copy()
    best_c5 = result.fun
