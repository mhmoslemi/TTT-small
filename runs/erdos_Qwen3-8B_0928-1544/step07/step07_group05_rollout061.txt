import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:

    best_c5 = current_c5
    best_h = current_h.copy()

    alpha = 0.95
    perturbation_rate = 0.1
    steps_without_improvement = 0

    start_time = time.time()
    
    for _ in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        num_perturb = int(perturbation_rate * n_points)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        h_perturbed = current_h.copy()
        h_perturbed[indices] += np.random.normal(0, 0.05, size=num_perturb)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))

        delta = new_c5 - current_c5
        
        if delta < 0:
            current_h = h_perturbed.copy()
            current_c5 = new_c5
            steps_without_improvement = 0
        else:

                current_h = h_perturbed.copy()
                current_c5 = new_c5
                steps_without_improvement = 0
            else:
                steps_without_improvement += 1

            best_h = current_h.copy()

            perturbation_rate = min(perturbation_rate * 1.2, 1.0)
            steps_without_improvement = 0
