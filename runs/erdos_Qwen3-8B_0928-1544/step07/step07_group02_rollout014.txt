import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points)
        h = np.sin(np.pi * x / 2) / 2 + 0.5
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(5):
        permuted = np.random.permutation(n_points)
        for i in permuted:
            def obj(x):
                h_temp = h.copy()
                h_temp[i] = x
                corr = np.correlate(h_temp, 1 - h_temp, mode='full')
                return np.max(corr) * (2.0 / n_points)
            res = opt.minimize_scalar(obj, bounds=(0, 1), method='bounded')
            h[i] = res.x
        h_proj = cp.Variable(n_points)
        constraints = [h_proj >= 0, h_proj <= 1, cp.sum(h_proj) == target_sum]
        obj = cp.Minimize(cp.norm(h_proj - h, 2))
        prob = cp.Problem(obj, constraints)
        prob.solve(verbose=False)
        h[:] = h_proj.value
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
