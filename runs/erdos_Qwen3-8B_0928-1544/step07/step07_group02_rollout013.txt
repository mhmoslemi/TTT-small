                method='COBYLA' if stage_idx == 0 else 'SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2},
                options={'maxiter': 100 if stage_idx == 0 else 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_h = project(result.x, n)
                current_c5 = objective(current_h, n)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            if time.time() - time_start > budget_s:
                return best_h, best_c5, n_high

    return best_h, best_c5, n_high
