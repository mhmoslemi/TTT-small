import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    h = initial_h.copy()

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        h += alpha * gradient
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h))

    return best_h, best_c5, n_points
