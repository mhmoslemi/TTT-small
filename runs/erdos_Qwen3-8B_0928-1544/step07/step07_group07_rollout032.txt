import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Initial optimization
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-6}
    )
    
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Iterative refinement
    if time.time() - start_time < budget_s * 0.8:
        for _ in range(3):
            sigma = 0.15 * (1 - (time.time() - start_time) / budget_s)
            h_perturbed = best_h + np.random.normal(0, sigma, n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
            
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-6}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
