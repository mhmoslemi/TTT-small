from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)

        h_perturbed = h + pert

        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        return h_perturbed
    
    def optimize(start_h):
        result = minimize(
            fun=objective,
            x0=start_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        return result.x if result.success else start_h
