import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    population_size = 30
    population = [best_h.copy() for _ in range(population_size)]
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate and select top candidates
        scores = np.array([objective(ind) for ind in population])
        top_idx = np.argsort(scores)[:population_size//2]
        selected = np.array([population[i] for i in top_idx])
        
        # Crossover
        new_population = []
        for i in range(population_size//2):
            for j in range(i+1, population_size//2):
                parent1, parent2 = selected[i], selected[j]
                child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
                new_population.append(child)
        # Mutation
        for i in range(population_size//2):
            parent = selected[i]
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = parent + noise
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total > 0:
                perturbed = perturbed * (ones_count / total)
            new_population.append(perturbed)
        # Update population
        population = new_population
        # Track best
        current_scores = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(current_scores)
        if current_scores[best_idx] < best_c5:
            best_c5 = current_scores[best_idx]
            best_h = population[best_idx].copy()
    
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        total = np.sum(h_perturbed)
        if total > 0:
            h_perturbed = h_perturbed * (ones_count / total)
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
