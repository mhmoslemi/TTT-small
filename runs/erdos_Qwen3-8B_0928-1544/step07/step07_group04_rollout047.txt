            try:
                problem.solve(solver=cp.SCS, eps=1e-8)
            except:
                pass
        if problem.status in ['OPTIMAL', 'OPT']:
            new_h = h_new.value
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = new_h * (ones_count / np.sum(new_h))
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            h = new_h.copy()
        else:
            perturb = np.random.normal(0, 0.05, n_points)
            h = h + perturb
            h = np.clip(h, 0.0, 1.0)
            h = h * (ones_count / np.sum(h))
    return best_h, best_c5, n_points
