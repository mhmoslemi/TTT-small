            candidate = generate_candidate()
        num_perturb = int(0.2 * n_points)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        perturb_ampl = 0.05
        candidate[indices] += np.random.uniform(-perturb_ampl, perturb_ampl, size=num_perturb)
        candidate = np.clip(candidate, 0.0, 1.0)
        current_sum = np.sum(candidate)
        delta = desired_sum - current_sum
        if delta:
            candidate[-1] += delta
            candidate = np.clip(candidate, 0.0, 1.0)
            current_sum = np.sum(candidate)
            delta = desired_sum - current_sum
            if delta:
                candidate[0] += delta
                candidate = np.clip(candidate, 0.0, 1.0)
        c5 = compute_c5(candidate)
        worst_idx = np.argmax([compute_c5(sol) for sol in harmony_memory])
        if c5 < compute_c5(harmony_memory[worst_idx]):
            harmony_memory[worst_idx] = candidate.copy()
        iteration += 1

    best_h = min(harmony_memory, key=compute_c5)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    k_max = np.argmax(corr)
    k_shift = k_max - n_points + 1
    grad = np.zeros(n_points)
    for i in range(n_points):
        j_plus = (i + k_shift) % n_points
        j_minus = (i - k_shift) % n_points
        grad[i] = (1.0 - best_h[j_plus]) - best_h[j_minus]
    learning_rate = 0.05
    for _ in range(50):
        step = learning_rate * grad
        candidate = best_h - step
        candidate = np.clip(candidate, 0.0, 1.0)
        current_sum = np.sum(candidate)
        delta = desired_sum - current_sum
        if delta:
            candidate[-1] += delta
            candidate = np.clip(candidate, 0.0, 1.0)
        new_c5 = compute_c5(candidate)
        if new_c5 < compute_c5(best_h):
            best_h = candidate.copy()
        learning_rate *= 0.99

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
