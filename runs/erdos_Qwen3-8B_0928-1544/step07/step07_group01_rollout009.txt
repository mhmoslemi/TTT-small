                best_c5 = current_c5
                best_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5_real = np.max(corr) * (2.0 / n_points)
    return best_h, best_c5_real, n_points
