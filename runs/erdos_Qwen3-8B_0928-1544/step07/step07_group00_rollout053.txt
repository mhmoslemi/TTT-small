import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2

    particles = np.array([h * target_sum / np.sum(h) for h in particles])
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    best_positions = particles.copy()
    best_costs = np.full(swarm_size, np.inf)
    global_best = initial_h.copy()
    global_cost = evaluate_erdos_solution(global_best)

    max_iter = 1000
    w = 0.9
    w_decrement = (0.9 - 0.4) / max_iter
    c1, c2 = 2.0, 2.0
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

        costs = np.array([evaluate_erdos_solution(p) for p in particles])
        mask = costs < best_costs

        best_costs[mask] = costs[mask]
        if np.min(costs) < global_cost:

        for i in range(swarm_size):
