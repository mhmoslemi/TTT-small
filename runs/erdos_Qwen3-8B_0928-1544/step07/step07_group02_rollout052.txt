            bounds=[(0.0, 1.0)] * len(h0),
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            return result.x, objective(result.x, n)
        return h0, float('inf')

    # Coarse optimization
    for guess in initial_guesses:
        current_h, current_c5 = optimize(guess, n_coarse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = upscale(current_h, n_coarse, n_points)

    # Medium optimization
    if best_c5 < float('inf'):
        best_h_med = upscale(best_h, n_points, n_med)
        h_med, c5_med = optimize(best_h_med, n_med)
        best_h = upscale(h_med, n_med, n_points)
        best_c5 = c5_med

    # Final optimization
    t0 = time.time()
    while time.time() - t0 < budget_s:
        result = minimize(
            lambda x: objective(x, n_points),
            x0=best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and objective(result.x, n_points) < best_c5:
            best_c5 = objective(result.x, n_points)
            best_h = result.x
        else:
            break

    return best_h, best_c5, n_points
