import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    alpha = 0.01
    decay = 0.999

    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        m = np.argmax(corr)
        kstar = m - (n_points - 1)
        gradient = np.zeros_like(best_h)
        for j in range(n_points):
            j_plus = j + kstar
            if 0 <= j_plus < n_points:
                gradient[j] += (1 - best_h[j_plus]) * dx
            j_minus = j - kstar
            if 0 <= j_minus < n_points:
                gradient[j] -= best_h[j_minus] * dx
        best_h = best_h - alpha * gradient
        best_h = np.clip(best_h, 0.0, 1.0)
        total = np.sum(best_h)
        best_h = best_h * (ones_count / total)
        corr_new = np.correlate(best_h, 1 - best_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
        alpha *= decay
    return best_h, best_c5, n_points
