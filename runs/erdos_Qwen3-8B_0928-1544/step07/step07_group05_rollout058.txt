                    child[j] += noise
                    child[j] = np.clip(child[j], 0, 1)
            
            temp_sum = np.sum(child)
            if temp_sum == 0:
                child = np.zeros(n_points)
            else:
                child = child * (ones_count / temp_sum)
            new_population[i] = child
        
        elite_indices = np.argsort(fitness)[:elitism_count]
        population = np.vstack([new_population, population[elite_indices]])
        population = population[:pop_size]
    
    return best_h, best_c5, n_points
