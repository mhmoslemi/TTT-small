    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    temp = 1.0
    cooling_rate = 0.995
    step = 0
    max_steps = 10000
    start_time = time.time()

    while step < max_steps and time.time() - start_time < budget_s:
        noise = np.random.normal(0, temp, n_points)
        perturbed = np.clip(h + noise, 0.0, 1.0)
        perturbed = perturbed * (ones_count / np.sum(perturbed))
        curr_c5 = objective(perturbed)

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = perturbed.copy()
            h = best_h
        else:
            prob = np.exp((curr_c5 - best_c5) / temp)
            if np.random.rand() < prob:
                best_c5 = curr_c5
                best_h = perturbed.copy()
                h = best_h

        temp *= cooling_rate
        step += 1

        if step % 100 == 0:
            res = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res.success:
                curr_c5 = res.fun
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = res.x
                    h = best_h

    return best_h, best_c5, n_points
