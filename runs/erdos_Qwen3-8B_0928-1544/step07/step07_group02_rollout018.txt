from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []

    if 'initial_h_values' in globals():

    best_h = np.zeros(n_points)
    start_time = time.time()

        return np.max(corr) * (2.0 / n_points)

        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
