import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    start_time = time.time()
    T = 1.0
    alpha = 0.995
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.15, n_points)
        h_perturbed = h + noise

        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))

            h = h_perturbed
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        T *= alpha
    return best_h, best_c5, n_points
