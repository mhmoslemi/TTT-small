from scipy.optimize import minimize
import time

    def objective(h):

        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    def optimize_with_perturbations(start_h):
        best_h_local = start_h.copy()
        best_c5_local = objective(start_h)
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break

            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
