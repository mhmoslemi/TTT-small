import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def cons(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    # Perform optimized SLSQP with advanced parameters
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons},
                       options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False})
        best_h = res.x
        best_c5 = res.fun
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    # Final check for feasibility
    best_h = np.clip(best_h, 0, 1)
    scale = target_sum / np.sum(best_h)
    best_h = best_h * scale
    best_c5 = obj(best_h)
    
    return best_h, best_c5, n_points
