import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    mutation_rate = 0.05

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        if np.sum(initial_h) == ones_count:
            population = [initial_h]
            for _ in range(population_size - 1):
                h = np.zeros(n_points, dtype=int)
                indices = np.random.choice(n_points, size=ones_count, replace=False)
                h[indices] = 1
                population.append(h)
        else:
            population = []
            for _ in range(population_size):
                h = np.zeros(n_points, dtype=int)
                indices = np.random.choice(n_points, size=ones_count, replace=False)
                h[indices] = 1
                population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.zeros(n_points, dtype=int)
            indices = np.random.choice(n_points, size=ones_count, replace=False)
            h[indices] = 1
            population.append(h)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        parents = sorted_pop[:population_size // 2]

        new_pop = []
        for _ in range(population_size // 2):
            p1, p2 = random.sample(parents, 2)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            delta = np.sum(child) - ones_count
            if delta != 0:
                if delta > 0:
                    ones_indices = np.where(child == 1)[0]
                    flip_indices = np.random.choice(ones_indices, size=delta, replace=False)
                    child[flip_indices] = 0
                else:
                    zeros_indices = np.where(child == 0)[0]
                    flip_indices = np.random.choice(zeros_indices, size=-delta, replace=False)
                    child[flip_indices] = 1
            
            flips = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
            child[flips] = 1 - child[flips]
            delta = np.sum(child) - ones_count
            if delta != 0:
                if delta > 0:
                    ones_indices = np.where(child == 1)[0]
                    flip_indices = np.random.choice(ones_indices, size=delta, replace=False)
                    child[flip_indices] = 0
                else:
                    zeros_indices = np.where(child == 0)[0]
                    flip_indices = np.random.choice(zeros_indices, size=-delta, replace=False)
                    child[flip_indices] = 1
            
            new_pop.append(child)
        
        population = parents + new_pop

        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    return best_h, best_c5, n_points
