import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, size=ones_count, replace=False)
        initial_h[indices] = 1
        initial_h = initial_h.astype(float)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    population_size = 50
    generations = 10
    population = np.zeros((population_size, n_points))
    for i in range(population_size):
        vec = np.zeros(n_points)
        if i == 0:
            vec[:] = initial_h
        else:
            indices = np.random.choice(n_points, size=ones_count, replace=False)
            vec[indices] = 1.0
        population[i] = vec

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for _ in range(generations):
        indices = np.argsort([objective(h) for h in population])[:population_size//2]
        selected = population[indices]
        new_population = []
        for i in range(population_size//2):
            parent1 = selected[i]
            parent2 = selected[np.random.randint(len(selected))]
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            new_population.append(child)
        for i in range(len(new_population)):
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    new_population[i][j] = 1.0 - new_population[i][j]
            new_population[i] = new_population[i] * (ones_count / np.sum(new_population[i]))
        population = np.array(new_population)
        current_best = min([objective(h) for h in population])
        if current_best < best_c5:
            best_h = population[np.argmin([objective(h) for h in population])]
            best_c5 = current_best

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if result.success:
        return result.x, result.fun, n_points
    return best_h, best_c5, n_points
