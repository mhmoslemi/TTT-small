import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    num_particles = 30
    particles = np.random.normal(0, 0.1, (num_particles, n_points))
    particles += h_initial
    particles = np.clip(particles, 0, 1)
    sum_p = np.sum(particles, axis=1, keepdims=True)
    particles *= (n_points / 2) / sum_p

    velocities = np.zeros((num_particles, n_points))
    best_h = particles.copy()
    best_c5 = np.inf

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def update_pbest(particles, pbest, pbest_vals, new_vals):
        for i in range(len(particles)):
            if new_vals[i] < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_vals[i]
        return pbest, pbest_vals

    def adjust_params(c1, c2, improvement):
        if improvement < 0.01:
            c1 += 0.1
            c2 -= 0.1
        else:
            c1 -= 0.1
            c2 += 0.1
        c1 = np.clip(c1, 0.5, 3.0)
        c2 = np.clip(c2, 0.5, 3.0)
        return c1, c2

    start_time = time.time()
    w, c1, c2 = 0.7, 2.0, 2.0
    pbest = particles.copy()
    pbest_vals = np.array([compute_C5(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    while time.time() - start_time < budget_s:
        r1, r2 = np.random.rand(num_particles), np.random.rand(num_particles)
        velocities = w * velocities + c1 * r1 * (pbest - particles) + c2 * r2 * (gbest - particles)
        particles += velocities

        particles = np.clip(particles, 0, 1)
        sum_p = np.sum(particles, axis=1, keepdims=True)
        particles *= (n_points / 2) / sum_p

        new_vals = np.array([compute_C5(p) for p in particles])
        pbest, pbest_vals = update_pbest(particles, pbest, pbest_vals, new_vals)
        new_gbest = pbest[np.argmin(pbest_vals)]
        new_gbest_val = np.min(pbest_vals)

        if new_gbest_val < gbest_val:
            gbest, gbest_val = new_gbest, new_gbest_val
            best_h, best_c5 = gbest, gbest_val

        improvement = (best_c5 - gbest_val) / best_c5 if best_c5 > 0 else 0.0
        c1, c2 = adjust_params(c1, c2, improvement)

    return best_h, best_c5, n_points
