import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import math

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        target = n_points / 2
        if sum_h == 0:
            return np.full(n_points, 0.5)
        scale = target / sum_h
        return np.clip(h * scale, 0, 1)
    
    best_h = h
    best_c5 = objective(h) * dx
    T = 1000.0
    alpha = 0.99
    iterations = 0
    max_iter = int(budget_s * 0.8)
    
    while iterations < max_iter and T > 1e-5:
        for _ in range(50):
            if iterations >= max_iter:
                break
            # Perturb 5 random points
            idxs = np.random.choice(n_points, 5, replace=False)
            perturb = np.random.uniform(-0.2, 0.2, 5)
            h_new = h.copy()
            h_new[idxs] += perturb
            h_new = project_to_feasible(h_new)
            
            current_cost = objective(h) * dx
            new_cost = objective(h_new) * dx
            delta = new_cost - current_cost
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if new_cost < best_c5:

                    best_h = h.copy()
            
            iterations += 1
            T *= alpha
    
    return (best_h, best_c5, n_points)
