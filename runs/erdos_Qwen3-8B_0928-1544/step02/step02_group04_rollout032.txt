    y_tri = 1 - np.abs(x - 1)
    y_tri = y_tri * target_sum / np.sum(y_tri)
    population.append(y_tri)
    
    x_sine = x
    y_sine = 0.5 * (1 + np.sin(np.pi * x_sine))
    y_sine = np.clip(y_sine, 0, 1)
    y_sine = y_sine * target_sum / np.sum(y_sine)
    population.append(y_sine)
    
    while len(population) < population_size:
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        h *= scale
        population.append(h)

    # Compute max overlap using FFT
    def compute_overlap(h):
        g = 1 - h
        N = len(h)
        h_pad = np.concatenate([h, np.zeros(2 * N - 1 - N)])
        g_pad = np.concatenate([g, np.zeros(2 * N - 1 - N)])
        fft_h = np.fft.fft(h_pad)
        fft_g = np.fft.fft(g_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_g)).real
        return np.max(corr) * dx

    # Evaluate
    def evaluate(individual):
        return compute_overlap(individual)

    best_h = population[0]
    best_c5 = evaluate(best_h)

    while time.time() - start_time < budget_s:
        fitness = [evaluate(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        selected = sorted_indices[:int(len(population) * elite_ratio)]
        new_population = [population[i] for i in selected]
        
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(population, 2, p=np.exp(-np.array(fitness)))
            c = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:c], p2[c:]])
            
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    child[j] += np.random.uniform(-mutation_scale, mutation_scale)
                    child[j] = np.clip(child[j], 0, 1)
            
            sum_child = np.sum(child)
            if sum_child != 0:
                child *= target_sum / sum_child
            else:
                child = np.random.uniform(0, 1, n_points)
                child *= target_sum / np.sum(child)
            
            new_population.append(child)
        population = new_population
        current_best = min(population, key=evaluate)
        if evaluate(current_best) < best_c5:
            best_c5 = evaluate(current_best)
            best_h = current_best

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    best_h *= target_sum / sum_best
    return best_h, best_c5, n_points
