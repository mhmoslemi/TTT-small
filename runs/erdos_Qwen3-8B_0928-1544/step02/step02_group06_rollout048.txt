        block_lengths = []
        for __ in range(k):
            length = base_length
            if remainder > 0:
                length += 1
                remainder -= 1
            block_lengths.append(length)
        h = np.zeros(n_points, dtype=np.float64)
        start = 0
        for length in block_lengths:
            end = start + int(length)
            h[start:end] = 1.0
            start = end

            adjustment = 1.0 - total_sum

                h[-1] -= -adjustment / dx
