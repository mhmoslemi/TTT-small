from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    pop_size = 30
    particles = np.random.normal(0, 0.1, (pop_size, n_points))
    particles += initial_h.reshape(1, -1)
    particles = np.clip(particles, 0, 1)
    for i in range(pop_size):
        particles[i] *= (n_points / 2) / np.sum(particles[i])
    
    pbest = particles.copy()
    gbest = particles[0].copy()
    velocities = np.zeros((pop_size, n_points))
    c1, c2 = 2.0, 2.0
    w = 0.7
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for _ in range(100000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            curr_c5 = evaluate(particles[i])
            if curr_c5 < evaluate(pbest[i]):
                pbest[i] = particles[i].copy()
                if curr_c5 < evaluate(gbest):
                    gbest = particles[i].copy()
        if np.any(evaluate(pbest) < best_c5):
            best_c5 = np.min(evaluate(pbest))

            c1 = np.clip(c1, 0.5, 3.0)
