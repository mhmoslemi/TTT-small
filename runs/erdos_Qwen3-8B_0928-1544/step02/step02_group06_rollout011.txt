from scipy.optimize import minimize
from copy import deepcopy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    evals = 0
    best_h = None
    best_c5 = float('inf')
    
    def objective(h):
        nonlocal evals
        evals += 1
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses = [initial_h_values]
    else:
        initial_guesses = [np.full(n_points, n_points / 2) / n_points]
    
    population_size = 50
    generations = 100
    mutation_rate = 0.2
    elite_ratio = 0.2
    
    # Initialize population
    pop = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for h in pop:
        h /= np.sum(h) * dx
        h = np.clip(h, 0, 1)
    
    # Genetic Algorithm
    for gen in range(generations):
        if evals > budget_s * 10:
            break
        # Evaluate fitness
        fitness = [objective(h) for h in pop]
        # Select elite
        elite_idx = np.argsort(fitness)[:int(population_size * elite_ratio)]
        elites = [pop[i] for i in elite_idx]
        # Create new population
        new_pop = elites
        for _ in range(population_size - len(elite_idx)):
            # Select parents
            p1, p2 = np.random.choice(elite_idx, 2, replace=False)
            parent1, parent2 = pop[p1], pop[p2]
            # Crossover
            child = parent1 * 0.5 + parent2 * 0.5
            # Mutation
            mutate = np.random.rand(n_points) < mutation_rate
            child[mutate] += np.random.uniform(-0.1, 0.1, np.sum(mutate))
            child = np.clip(child, 0, 1)
            child /= np.sum(child) * dx
            new_pop.append(child)
        pop = new_pop
    
    # Refine best candidates
    candidates = [h for h in pop if objective(h) < best_c5]
    if candidates:
        for h in candidates:
            result = minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            )
            if result.success:
                h_opt = result.x

                if c5 < best_c5:
                    best_h = h_opt

                h_opt = h
                c5 = objective(h_opt)
                if c5 < best_c5:
