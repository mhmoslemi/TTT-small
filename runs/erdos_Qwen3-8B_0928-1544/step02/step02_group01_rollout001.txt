import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h *= (n_points / 2) / total
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return h * scale
    
    pop_size = 30
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        h = initial_h + perturb
        h = project_feasible(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    mutation_std = 0.1
    mutation_decay = 0.995
    
    while time.time() - start_time < budget_s:
        offspring = []
        for h in population:
            perturb = np.random.normal(0, mutation_std, n_points)
            new_h = h + perturb
            new_h = project_feasible(new_h)
            offspring.append(new_h)
        
        candidates = population + offspring
        candidates_c5 = [compute_C5(h) for h in candidates]
        indices = np.argsort(candidates_c5)
        new_population = [candidates[i] for i in indices[:pop_size]]
        
        current_best = min(candidates_c5)
        if current_best < best_c5:
            best_h = candidates[indices[0]]
            best_c5 = current_best
        
        if len(np.unique(candidates_c5)) < 2:
            mutation_std *= mutation_decay
        else:
            mutation_std *= mutation_decay * (1 - 0.8 * (current_best - best_c5))
        
        population = new_population
