import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h0 = initial_h_values.copy()
    else:
        h0 = np.full(n, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=h0,
        method='COBYLA',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 2000, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h0
        best_c5 = objective(h0)
    
    if time.time() - start_time > budget_s:
        pass
    
    return best_h, best_c5, n
