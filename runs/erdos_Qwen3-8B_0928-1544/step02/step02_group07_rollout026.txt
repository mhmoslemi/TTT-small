import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        sum_h = np.sum(h)
        target_sum = n_points / 2
        penalty = 1e5 * (sum_h - target_sum) ** 2
        return compute_c5(h) + penalty

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        best_h = result.x if hasattr(result, 'x') else h_initial
        best_c5 = compute_c5(best_h)
    else:
        best_h = result.x
        best_c5 = compute_c5(best_h)

    # Ensure sum is correct
    sum_h = np.sum(best_h)
    target_sum = n_points / 2
    if abs(sum_h - target_sum) > 1e-6:
        diff = target_sum - sum_h
        best_h[-1] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n_points
