def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_h
        return h_clipped * scale
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    pop_size = 100
    tournament_size = 5
    mutation_rate = 0.1
    max_generations = 1000
    
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        population.append(normalize(perturbed))
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        fitnesses = np.array([compute_c5(ind) for ind in population])
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        new_population = []
        for _ in range(pop_size):
            tournament = np.random.choice(population, size=tournament_size, replace=False)
