import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def project_feasible(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        return h * scale

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, step):
        noise = np.random.normal(0, step, len(h))
        return project_feasible(h + noise)

    population_size = 30
    mutation_step = 0.1
    current_population = [initial_h.copy()]
    for _ in range(population_size - 1):
        candidate = initial_h.copy() + np.random.normal(0, 0.1, n_points)
        candidate = project_feasible(candidate)
        current_population.append(candidate)

    best_h = initial_h.copy()
    best_c5 = compute_C5(initial_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_population = []
        for h in current_population:
            mutated = mutate(h, mutation_step)
            new_population.append(mutated)
        evals = [compute_C5(h) for h in new_population]
        sort_indices = np.argsort(evals)
        top_k = int(population_size * 0.25)

        remaining = population_size - len(current_population)
        for _ in range(remaining):
            rand_h = np.random.rand(n_points)
            rand_h = project_feasible(rand_h)
            current_population.append(rand_h)
        for i, h in enumerate(new_population):
            if evals[i] < best_c5:
                best_c5 = evals[i]
                best_h = h
        if best_c5 < compute_C5(current_population[0]):
            mutation_step *= 0.95
        else:
            mutation_step *= 1.05
