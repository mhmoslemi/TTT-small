import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
        # Add perturbations
        for _ in range(5):
            perturbed = initial_h_values.copy()
            indices = np.random.randint(0, n_points, size=20)
            perturbed[indices] += np.random.uniform(-0.05, 0.05, size=20)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            initial_guesses.append(perturbed)
    else:
        num_initial = 10
        for _ in range(num_initial):
            a, b = np.random.randint(0, n_points-64), np.random.randint(a+64, n_points-64)
            h = np.zeros(n_points)
            h[a:a+64] = 1.0
            h[b:b+64] = 1.0
            initial_guesses.append(h)
        initial_guesses.append(np.full(n_points, n_points/2)/n_points)
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points/2,
        'jac': np.ones(n_points)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        guess = np.clip(guess, 0.0, 1.0)
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000})
        current_h = result.x if result.success else guess
        current_c5 = result.fun if result.success else objective(current_h)
        if np.abs(np.sum(current_h) - n_points/2) > 1e-6:
            current_h = guess
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    if best_h is not None:
        result_refine = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
        best_h = result_refine.x if result_refine.success else best_h
        best_c5 = result_refine.fun if result_refine.success else objective(best_h)
