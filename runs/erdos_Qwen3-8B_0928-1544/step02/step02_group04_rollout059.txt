        initial_h = initial_h_resampled
    else:
        new_n_points = 256
        initial_h = np.ones(new_n_points) * (new_n_points / 2) / new_n_points
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]

    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )

    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        h_values = initial_h
        c5_bound = objective(h_values)

    h_values = np.clip(h_values, 0, 1)
    if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
        h_values = initial_h
        c5_bound = objective(h_values)

    return h_values, c5_bound, n_points
