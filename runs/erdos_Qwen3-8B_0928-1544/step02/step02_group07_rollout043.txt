import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    def compute_C5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        delta = required_sum - sum_clipped
        if abs(delta) < 1e-10:
            return h_clipped
        h_clipped += delta / n_points
        h_clipped = np.clip(h_clipped, 0.0, 1.0)
        return h_clipped

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    sigma = 0.2
    population_size = 100
    max_iter = 500  # Increased for better exploration

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        population = []
        for _ in range(population_size):
            perturbation = np.random.normal(0, sigma, n_points)
            candidate = best_h + perturbation
            candidate = project(candidate)
            population.append(candidate)
        c5_values = [compute_C5(h) for h in population]
        best_idx = np.argmin(c5_values)
        current_h = population[best_idx]
        current_c5 = c5_values[best_idx]
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        # Adaptive sigma adjustment
        if current_c5 < best_c5:
            sigma *= 0.98
        else:
            sigma *= 1.02

    return best_h, best_c5, n_points
