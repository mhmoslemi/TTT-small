    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try a few perturbations if time remains
        for _ in range(2):
            perturb = np.random.uniform(-0.05, 0.05, size=n)
            new_guess = np.clip(current_h + perturb, 0, 1)
            res = minimize(
                fun=objective,
                x0=new_guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                tol=1e-8,
                options={'maxiter': 500, 'ftol': 1e-9}
            )
            if res.success and res.fun < current_c5:
                current_c5 = res.fun
                current_h = res.x
    
    return current_h, current_c5, n
