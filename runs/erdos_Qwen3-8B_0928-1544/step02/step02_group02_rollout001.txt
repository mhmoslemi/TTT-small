import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    n = 70
    dx = 2.0 / n
    initial_guess = np.array([])
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-11,
        options={'maxiter': 3000, 'ftol': 1e-11, 'disp': False}
    )
    
    if result.success:
        current_h = result.x
        current_c5 = result.fun
    else:
        current_h = initial_guess
        current_c5 = objective(current_h)
    
    # Fallback to initial guess if optimization failed
    if time.time() - start_time > budget_s or not result.success:
        current_h = initial_guess
        current_c5 = objective(current_h)
    
    return current_h, current_c5, n
