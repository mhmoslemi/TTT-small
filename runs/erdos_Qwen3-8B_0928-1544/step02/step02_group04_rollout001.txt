        evaluated = [(evaluate_c5(h, dx), h) for h in population]
        evaluated.sort()
        
        # Select top 50% parents
        elite_size = int(len(population) * 0.5)
        selected = [h for c5, h in evaluated[:elite_size]]
        
        # Generate new population
        new_population = []
        for _ in range(len(population)):
            # Crossover
            p1, p2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < cross_rate:
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = np.random.choice([p1, p2], size=n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.1
                child[mask] += np.random.normal(0, 0.05, sum(mask))
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum == 0:
                child = np.full(n_points, 0.5)
            else:
                child *= (n_points / 2) / child_sum
            new_population.append(child)
        
        population = new_population
        
        # Update best
        for c5, h in evaluated:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    return best_h, best_c5, n_points

def evaluate_c5(h, dx):
    corr = correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
