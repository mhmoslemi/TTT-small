import numpy as np
from scipy.optimize import minimize
from numpy.random import choice, uniform, randint, seed

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        num_initial = 20
        for _ in range(num_initial):
            start1 = np.random.randint(0, 128)
            start2 = start1 + 64
            h = np.zeros(n_points)
            h[start1:start1+64] = 1.0
            h[start2:start2+64] = 1.0
            initial_guesses.append(h)
        initial_guesses.append(np.full(n_points, n_points / 2) / n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def mutate(h):
        h = h.copy()
        idx = randint(0, n_points)
        h[idx] += uniform(-0.2, 0.2)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def crossover(p1, p2):
        h = (p1 + p2) / 2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def evaluate(h):
        return objective(h)
    
    best_h = None
    best_c5 = float('inf')
    pop_size = 50
    gen_max = 100
    time_used = 0
    
    pop = [h.copy() for h in initial_guesses]
    for _ in range(gen_max):
        fit = [evaluate(h) for h in pop]
        elites = np.argsort(fit)[:int(pop_size * 0.2)]
        new_pop = [pop[i] for i in elites]
        
        while len(new_pop) < pop_size:
            p1, p2 = choice(pop_size, 2, replace=False)
            child = crossover(pop[p1], pop[p2])
            if np.random.rand() < 0.2:
                child = mutate(child)
            new_pop.append(child)
        
        pop = new_pop
        current_time = time_used + (gen_max - _) * (budget_s / gen_max)
        if current_time > budget_s:
            break
        
    best_candidate = min(pop, key=evaluate)
    result_refine = minimize(
        fun=objective,
        x0=best_candidate,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )
    best_h = result_refine.x if result_refine.success else best_candidate
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    else:
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
