from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        num_initial = 30
        for _ in range(num_initial):
            start1 = np.random.randint(0, n_points - 128)
            start2 = start1 + 64
            h = np.zeros(n_points)
            h[start1:start1 + 64] = 1.0
            h[start2:start2 + 64] = 1.0
            initial_guesses.append(h)
        for _ in range(num_initial // 2):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            initial_guesses.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess
            current_c5 = objective(current_h)
        current_h = np.clip(current_h, 0.0, 1.0)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = guess
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    if best_h is not None:
        result_refine = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',

            constraints=cons,

        )
