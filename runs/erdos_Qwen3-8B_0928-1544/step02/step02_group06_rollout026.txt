import numpy as np
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def normalize(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        return h * scale

    initial_population = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
        sum_initial = np.sum(h_initial)
        scaled = h_initial * (target_sum / sum_initial)
        initial_population.append(scaled)
    else:
        for _ in range(5):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            scaled = h * (target_sum / sum_h)
            initial_population.append(scaled)

    population = initial_population
    for _ in range(50):
        fitness = [compute_c5(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        parents = sorted_pop[:int(len(sorted_pop) * 0.2)]
        offspring = []
        while len(offspring) < len(population) - len(parents):
            p1, p2 = random.sample(parents, 2)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            for i in range(n_points):
                if np.random.random() < 0.1:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            child = normalize(child)
            offspring.append(child)
        population = parents + offspring

    best_index = np.argmin([compute_c5(ind) for ind in population])
    best_h = population[best_index]
    best_h = np.clip(best_h, 0, 1)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
             'jac': np.ones_like(h)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(lambda h: compute_c5(h), best_h, method='SLSQP',
                      bounds=bounds, constraints=cons, options={'maxiter': 500,
                                                               'ftol': 1e-9})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - target_sum) > 1e-6:
        best_h = population[best_index]
    return (best_h, compute_c5(best_h), n_points)
