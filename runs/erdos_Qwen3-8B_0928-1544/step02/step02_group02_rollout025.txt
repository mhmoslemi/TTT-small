from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):

        max_overlap = np.max(corr) * dx
        return max_overlap

    n = 70
    initial_guess = get_initial_guess(n)
    dx = 2.0 / n

    bounds = Bounds([0.0]*n, [1.0]*n)

    result = minimize(
