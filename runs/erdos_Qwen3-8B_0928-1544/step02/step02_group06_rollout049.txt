        pos = 0
        for l in lengths:
            end = pos + l
            if end > n_points:
                end = n_points
                l = end - pos
            h[pos:end] = 1.0
            pos = end
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            continue
        initial_guesses.append(h)
    initial_guesses.append(np.full(n_points, n_points / 2) / n_points)
    
    best_h = None
    best_c5 = float('inf')
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    
    for guess in initial_guesses:
        result = minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full') * dx),
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess
            current_c5 = np.max(np.correlate(guess, 1 - guess, mode='full') * dx)
        current_h = np.clip(current_h, 0.0, 1.0)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = guess
            current_c5 = np.max(np.correlate(guess, 1 - guess, mode='full') * dx)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    if best_h is not None:
        result_refine = minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full') * dx),
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        if result_refine.success:
            best_h = result_refine.x
            best_c5 = result_refine.fun
        else:
            best_h = best_h
            best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    return (best_h, best_c5, n_points)
