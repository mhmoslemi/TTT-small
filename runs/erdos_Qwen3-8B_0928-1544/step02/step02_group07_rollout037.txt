import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    num_perturbations = 50
    max_iter_per_restart = 300
    np.random.seed(seed)

    for _ in range(num_perturbations):
        perturbation = np.random.normal(0, 0.01, n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed_h)
        scale = (n_points / 2) / sum_perturbed
        perturbed_h = perturbed_h * scale
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)

        result = minimize(
            compute_C5,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8}
        )

        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
