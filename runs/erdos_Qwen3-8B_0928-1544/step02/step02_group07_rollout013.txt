    # Objective function to minimize
    def compute_C5(h):

        return np.max(corr) * dx

    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0

    # Optimization parameters
    num_restarts = 50  # Increased restarts for broader exploration
    max_iter_per_restart = 200
    np.random.seed(seed)

    for _ in range(num_restarts):
        # Generate random initial guess

        sum_random = np.sum(random_h)
        scale = (n_points / 2) / sum_random

        # Adjust to meet sum constraint (ensure precision)
        sum_random = np.sum(random_h)
        if abs(sum_random - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_random
            random_h[-1] += diff
            random_h = np.clip(random_h, 0.0, 1.0)

        # Optimize from random guess
        result = minimize(
            compute_C5,
