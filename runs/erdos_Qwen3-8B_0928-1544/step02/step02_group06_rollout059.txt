import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        for _ in range(5):
            h = np.random.rand(n_points)
            h /= np.sum(h) * 128.0
            h = np.clip(h, 0.0, 1.0)
            initial_guesses.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    population = initial_guesses[:]
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h /= np.sum(h) * 128.0
        h = np.clip(h, 0.0, 1.0)
        population.append(h)
    
    def evaluate_all(pop):
        return [objective(ind) for ind in pop]
    
    fitness = evaluate_all(population)
    
    best_h = population[np.argmin(fitness)]
    best_c5 = objective(best_h)
    
    for gen in range(generations):
        if gen * 1000 > budget_s:
            break
        sorted_idx = np.argsort(fitness)
        parents = np.random.choice(sorted_idx[:int(len(sorted_idx)*0.2)], size=pop_size, replace=True)
        new_pop = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_rate:
                p1, p2 = population[np.random.choice(parents, 2, replace=False)]
                child = np.copy(p1)
                split = np.random.randint(1, n_points)
                child[:split] = p1[:split]

            child = np.clip(child, 0.0, 1.0)
            new_pop.append(child)
        
        population = new_pop
        fitness = evaluate_all(population)
        current_best = population[np.argmin(fitness)]
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
