import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time
    from scipy import signal

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    num_restarts = 30
    max_iter_per_restart = 200
    np.random.seed(seed)

    def generate_block_guess(m):
        block_size = n_points // m
        block_vals = []
        for i in range(m):
            start = i * block_size
            end = min((i + 1) * block_size, n_points)
            avg = np.mean(h_initial[start:end])
            block_vals.append(avg)
        total = np.sum(block_vals)
        scale = (n_points / 2) / total
        block_vals = [x * scale for x in block_vals]
        h = np.zeros(n_points)
        ptr = 0
        for val in block_vals:
            h[ptr:ptr + block_size] = val
            ptr += block_size
        return h

    for _ in range(num_restarts):
        if elapsed_time > budget_s:
            break
        # Generate block-based initial guess
        m = np.random.choice([20, 40, 60, 80])
        h_guess = generate_block_guess(m)
        h_guess = np.clip(h_guess, 0.0, 1.0)
        sum_guess = np.sum(h_guess)
        if abs(sum_guess - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_guess
            h_guess[-1] += diff
            h_guess = np.clip(h_guess, 0.0, 1.0)
        # Optimize
        result = minimize(
            compute_C5,
            h_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")
        elapsed_time = time.time() - start_time

    return best_h, best_c5, n_points
