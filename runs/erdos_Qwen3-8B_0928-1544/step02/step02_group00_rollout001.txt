def run(seed=42, budget_s=1100, **kwargs):

        h = h / np.sum(h) * (n_points / 2)
        pop.append(h)

    best_c5 = np.inf
    
    for gen in range(200):

            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        if min(fitness) < best_c5:

            best_h = pop[fitness.index(best_c5)]
