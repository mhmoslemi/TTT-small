import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values
    else:
        h_initial = np.full(70, 0.5)
    
    n = 70
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h_initial
        best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return best_h, best_c5, n
    
    # Try a few perturbations
    for _ in range(2):
        noise = np.random.uniform(-0.05, 0.05, n)
        perturbed = np.clip(best_h + noise, 0, 1)
        res = minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
    
    return best_h, best_c5, n
