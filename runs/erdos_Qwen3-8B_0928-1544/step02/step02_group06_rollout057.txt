import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points / 2
    np.random.seed(seed)
    
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        num_initial = 50
        for _ in range(num_initial):
            num_blocks = np.random.randint(3, 11)
            block_lengths = []
            for _ in range(num_blocks - 1):
                block_lengths.append(np.random.randint(2, 64))
            last_length = target_sum - sum(block_lengths)
            if last_length < 2:
                last_length = 2
                block_lengths[-1] = target_sum - sum(block_lengths)
            block_lengths.append(last_length)
            h = np.zeros(n_points, dtype=np.float64)
            pos = 0
            for length in block_lengths:
                end = int(pos + length)
                h[pos:end] = 1.0
                pos = end
            initial_guesses.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess
            current_c5 = objective(current_h)
        current_h = np.clip(current_h, 0.0, 1.0)
        if np.abs(np.sum(current_h) - target_sum) > 1e-6:
            current_h = guess
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    if best_h is not None:
        result_refine = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        if result_refine.success:
            best_h = result_refine.x
            best_c5 = result_refine.fun
        else:
            best_h = best_h
            best_c5 = objective(best_h)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - target_sum) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
