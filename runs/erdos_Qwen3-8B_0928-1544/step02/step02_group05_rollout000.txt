    current_temp = T_initial
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        num_perturb = int(n_points * 0.1)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        h_perturb = h.copy()
        h_perturb[indices] = np.random.rand(num_perturb)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        current_sum_perturb = np.sum(h_perturb)
        scale_factor = target_sum / current_sum_perturb
        h_perturb *= scale_factor
        candidate_cost = compute_cost(h_perturb)
        delta = candidate_cost - compute_cost(h)
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
            h = h_perturb.copy()
            if candidate_cost < best_cost:
                best_cost = candidate_cost
                best_h = h.copy()
        current_temp *= alpha
        current_cost = compute_cost(h)
        if current_cost < best_cost:
            best_cost = current_cost
            best_h = h.copy()
    return (best_h, best_cost, n_points)
