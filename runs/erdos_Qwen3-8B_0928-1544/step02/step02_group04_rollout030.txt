    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_initial = len(initial_h)
        new_n_points = 256
        x_initial = np.linspace(0, 2, n_initial)
        f = np.interp(np.linspace(0, 2, new_n_points), x_initial, initial_h)
        current_sum = np.sum(f)
        scale_factor = (new_n_points / 2) / current_sum
        initial_h_resampled = f * scale_factor
        current_h = initial_h_resampled
    else:
        new_n_points = 256
        current_h = np.ones(new_n_points) * (new_n_points / 2) / new_n_points

    n_points = len(current_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_h = current_h.copy()
    best_c5 = compute_c5(best_h)

    temp = 1.0
    alpha = 0.999
    start_time = time.time()

    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        if i == j:
            j = (j + 1) % n_points
        delta = np.random.uniform(-0.1, 0.1)
        new_h_i = min(max(current_h[i] + delta, 0.0), 1.0)
        delta_i = new_h_i - current_h[i]
        new_h_j = min(max(current_h[j] - delta_i, 0.0), 1.0)
        candidate = current_h.copy()
        candidate[i] = new_h_i
        candidate[j] = new_h_j
        current_sum_candidate = np.sum(candidate)
        delta_sum = (n_points / 2) - current_sum_candidate
        k = random.randint(0, n_points - 1)
        candidate[k] += delta_sum
        candidate[k] = min(max(candidate[k], 0.0), 1.0)
        new_c5 = compute_c5(candidate)
        if new_c5 < best_c5 or random.random() < np.exp((best_c5 - new_c5) / temp):
            current_h = candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = candidate
        temp *= alpha

    current_sum = np.sum(best_h)
    if abs(current_sum - n_points / 2) > 1e-6:
        best_h = current_h.copy()
        best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
