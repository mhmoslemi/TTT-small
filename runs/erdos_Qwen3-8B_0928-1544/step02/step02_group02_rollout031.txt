def run(seed=42, budget_s=1100, **kwargs):

            return initial_h_values.copy()
        return np.full(n, 0.5)
    
    n = 70
    dx = 2.0 / n
    initial_guess = get_initial_guess(n)
    
    def objective_and_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        max_idx = np.argmax(corr)
        k = max_idx - n + 1  # shift index
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j + k < n:

                grad[j] -= h[j - k]
        return max_overlap, grad * dx
