import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    n_particles = 20
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n, 0.5)
    
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        if np.random.rand() < 0.5 and 'initial_h_values' in globals():
            h = initial_h.copy()
        else:
            h = np.random.rand(n)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full(n, 0.5)
        else:
            scale = (n / 2) / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n))
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    pbest = [np.array(p) for p in particles]
    pbest_values = [evaluate(p) for p in particles]
    
    global_best = np.array(particles[0])
    global_best_value = pbest_values[0]
    
    w = 0.7
    c1, c2 = 2.0, 2.0
    start_time = time.time()
    
    for _ in range(10000):

            break

            r1, r2 = np.random.rand(), np.random.rand()
