import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    pop_size = 30
    mut_step = 0.1
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start = time.time()

    population = [project_feasible(h_initial + np.random.normal(0, mut_step, n_points)) for _ in range(pop_size)]

        best_idx = np.argmin(evals)
