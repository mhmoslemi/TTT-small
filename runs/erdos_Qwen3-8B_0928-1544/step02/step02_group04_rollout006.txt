        if candidate_c < current_c:
            current_h = candidate
            current_c = candidate_c
        else:
            prob = np.exp((current_c - candidate_c) / temp)
            if np.random.rand() < prob:
                current_h = candidate
                current_c = candidate_c
        if current_c < best_c:
            best_h = current_h.copy()
            best_c = current_c
        temp *= cooling_rate
    
    best_h = np.clip(best_h, 0, 1)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
