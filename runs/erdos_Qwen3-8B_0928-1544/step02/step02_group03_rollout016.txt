import numpy as np
from scipy.optimize import minimize, Bounds, basinhopping

def run(seed=42, budget_s=1100, **kwargs):
    # Check for initial_h_values
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
        n_points = 70
    else:
        h_initial = np.full(70, 0.5)
        n_points = 70

    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    def optimize_basin(start_h):
        result = basinhopping(
            fun=objective,
            x0=start_h,
            niter=20,
            T=1.0,
            minimizer_kwargs={
                'method': 'L-BFGS-B',
                'bounds': bounds,
                'constraints': {'type': 'eq', 'fun': constraint_sum},
                'tol': 1e-8,
                'options': {'maxiter': 50}
            },
            seed=seed
        )
        if result.success:
            return result.x
        return start_h

    def optimize_local(start_h):
        result = minimize(
            fun=objective,
            x0=start_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if result.success:
            return result.x
        return start_h

    h_basin = optimize_basin(h_initial)
    h_local = optimize_local(h_basin)

    # Clip and normalize
    h_clipped = np.clip(h_local, 0.0, 1.0)
    sum_clipped = np.sum(h_clipped)
    scale = (n_points / 2) / sum_clipped
    h_clipped *= scale

    c5 = objective(h_clipped)
    return h_clipped, c5, n_points
