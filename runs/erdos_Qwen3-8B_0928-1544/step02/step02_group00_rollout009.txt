import numpy as np
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial = initial_h_values.copy()
        initial = initial * (35.0 / np.sum(initial))
        initial = np.clip(initial, 0, 1)
        population = [initial]
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population = [h]
    
    for _ in range(49):
        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (35.0 / np.sum(h))
        population.append(h)
    
    for gen in range(200):
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            fitness.append(max_overlap)
        
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        
        top_indices = np.argsort(fitness)[:25]
        parents = [population[i] for i in top_indices]
        
        offspring = []
        for _ in range(25):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            if np.random.rand() < 0.7:
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = p1.copy()
                for i in range(n_points):
                    if np.random.rand() < 0.05:
                        child[i] = np.random.rand()
            child = np.clip(child, 0, 1)
            child = child * (35.0 / np.sum(child))
            offspring.append(child)
        
        population = parents + offspring
        np.random.shuffle(population)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (35.0 / np.sum(best_h))
    return best_h, best_c5, n_points
