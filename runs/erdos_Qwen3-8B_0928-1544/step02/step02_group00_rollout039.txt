def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf
    max_iter = 200
    delta = 0.02
    decay_rate = 0.995

    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        k_max = np.argmax(corr)
        s = k_max - (n_points - 1)
        contributions = np.zeros(n_points)
        for i in range(n_points):
            j = (i + s) % n_points
            contributions[i] = h[i] * (1 - h[j])
        top_indices = np.argsort(contributions)[::-1][:5]
        for i in top_indices:
            j = (i + s) % n_points
            h[i] -= delta
            h[j] += delta
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
        current_sum = np.sum(h)
        if not np.isclose(current_sum, n_points / 2, atol=1e-6):
            h = h / current_sum * (n_points / 2)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
        delta *= decay_rate

    final_h = best_h
    final_sum = np.sum(final_h)
    if abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    final_h = np.clip(final_h, 0.0, 1.0)
    return final_h, best_c5, n_points
