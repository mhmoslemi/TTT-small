import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import math

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 3000, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    best_h = result.x
    best_c5 = result.fun * dx
    current_n_points = n_points
    
    if budget_s > 1000:
        new_n_points = 280
        original_h = best_h
        original_x = np.linspace(0, 2, n_points)
        new_x = np.linspace(0, 2, new_n_points)
        new_h = np.interp(new_x, original_x, original_h)
        sum_h = np.sum(new_h)
        target_sum = new_n_points / 2
        scaling = target_sum / sum_h
        new_h_scaled = new_h * scaling
        new_h_clipped = np.clip(new_h_scaled, 0, 1)
        
        bounds_new = [(0.0, 1.0) for _ in range(new_n_points)]
        cons_new = ({'type': 'eq', 'fun': lambda h: np.sum(h) - new_n_points / 2})
        
        result2 = minimize(
            objective,
            new_h_clipped,
            method='SLSQP',
            bounds=bounds_new,
            constraints=cons_new,
            options={'maxiter': 2000, 'ftol': 1e-10, 'eps': 1e-9}
        )
        
        best_h2 = result2.x
        best_c52 = result2.fun * (2.0 / new_n_points)
        if best_c52 < best_c5:
            best_c5 = best_c52
            best_h = best_h2
            current_n_points = new_n_points
    
    return (best_h, best_c5, current_n_points)
