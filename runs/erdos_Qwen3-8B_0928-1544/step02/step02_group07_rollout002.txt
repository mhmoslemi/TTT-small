import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum(h):
        s = np.sum(h)
        delta = (n_points / 2) - s
        h += delta / n_points
        h = np.clip(h, 0, 1)
        return h

    def gen_candidate(mean, sigma):
        c = mean + sigma * np.random.randn(n_points)
        c = np.clip(c, 0, 1)
        c = adjust_sum(c)
        return c

    pop_size = 100
    sigma = 0.2
    pop = [gen_candidate(h_initial, sigma) for _ in range(pop_size)]
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed = 0.0
    start = time.time()

    for _ in range(500):
        fitness = np.array([compute_C5(h) for h in pop])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx]
        top = pop[np.argsort(fitness)[:int(pop_size * 0.2)]]
        new_mean = np.mean(top, axis=0)

        pop = [np.clip(new_mean + new_sigma * np.random.randn(n_points), 0, 1) for _ in range(pop_size)]
        pop = [adjust_sum(h) for h in pop]
        elapsed = time.time() - start
        if elapsed > budget_s:
