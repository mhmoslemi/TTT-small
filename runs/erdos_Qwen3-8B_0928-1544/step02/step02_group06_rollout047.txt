import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.random.rand(n_points)
        initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
    
    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return (h * (n_points / 2) / s)
    
    def smooth(h, sigma=0.5):
        if sigma < 0:
            return h
        h = gaussian_filter(h, sigma=sigma, mode='wrap')
        return rescale(h)
    
    h_pop = [initial_guess.copy() for _ in range(50)]
    for _ in range(100):
        brightness = np.array([obj(h) for h in h_pop])
        sorted_idx = np.argsort(-brightness)
        h_pop = [h_pop[i] for i in sorted_idx]
        for i in range(len(h_pop)):
            for j in range(i + 1, len(h_pop)):
                if brightness[j] < brightness[i]:
                    h_i = h_pop[i]
                    h_j = h_pop[j]
                    diff = h_j - h_i
                    r = np.linalg.norm(diff)
                    if r > 0:
                        h_i += beta * np.exp(-gamma * r) * diff + alpha * np.random.randn(n_points)
                        h_i = rescale(np.clip(h_i, 0, 1))
                        h_pop[i] = h_i
        h_pop = [smooth(h) for h in h_pop]
    
    best_h = min(h_pop, key=obj)
    best_c5 = obj(best_h)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        fun=obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    
    return (best_h, best_c5, n_points)
