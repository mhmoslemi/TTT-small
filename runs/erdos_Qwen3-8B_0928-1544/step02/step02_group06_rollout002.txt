from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256

    elite_ratio = 0.2
    mutation_rate = 0.15

    population = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        population.append(initial_h_values)
    else:
        for _ in range(pop_size):
            h = np.zeros(n_points)
            indices = choice(n_points, size=128, replace=False)
            h[indices] = 1.0
            population.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return np.max(corr)

    def mutate(parent):
        child = parent.copy()
        num_flips = int(mutation_rate * n_points)
        indices = choice(n_points, size=num_flips, replace=False)
        for i in indices:
            child[i] = 1 - child[i]
        # Ensure sum is exactly 128
        total = np.sum(child)
        if total > 128:
            diff = total - 128
            zeros = np.where(child == 0)[0]
            if zeros.size > 0:
                idx = zeros[np.random.choice(zeros.size)]
                child[idx] = 1
        elif total < 128:
            diff = 128 - total
            ones = np.where(child == 1)[0]
            if ones.size > 0:
                idx = ones[np.random.choice(ones.size)]
                child[idx] = 0
        return child

    best_h = None
    best_c5 = float('inf')

    for gen in range(max_gen):
        # Evaluate all
        fitness = np.array([evaluate(h) for h in population])
        order = np.argsort(fitness)
        sorted_pop = [population[i] for i in order]

        # Elites
        elite_count = int(pop_size * elite_ratio)
