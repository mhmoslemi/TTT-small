import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_start = initial_h_values.copy()
    else:
        h_start = np.full(70, 0.5)
    
    n = 70
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    def objective_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        obj = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 0
        if sum_h < n / 2 - 1e-8:
            penalty = 1e4 * (n / 2 - sum_h)
        elif sum_h > n / 2 + 1e-8:
            penalty = 1e4 * (sum_h - n / 2)
        return obj + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    result_de = differential_evolution(
        fun=objective_de,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed
    )
    
    h_de = result_de.x
    
    result_slsqp = minimize(
        fun=objective,
        x0=h_de,
        method='SLSQP',
        bounds=Bounds([0.0]*n, [1.0]*n),
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    
    best_c5 = result_slsqp.fun
    best_h = result_slsqp.x
    
    return best_h, best_c5, n
