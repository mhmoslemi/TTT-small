from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

    def objective(h):
        h = np.clip(h, 0, 1)

        max_overlap = np.max(corr) * dx
        return max_overlap

        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(

        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},

        options={'maxiter': 300, 'disp': False}
    )

        final_c5 = result.fun
