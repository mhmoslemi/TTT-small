from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 256
    dx = 2.0 / n_points
    initial_population = []
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_population.append(initial_h_values)
    else:

            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            initial_population.append(h)
    
    start_time = time.time()
    generations = 100
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        fitness = []
        for h in initial_population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr * dx)
            fitness.append(c5)
        sorted_indices = np.argsort(fitness)
        elite_size = int(len(initial_population) * 0.2)
        next_pop = [initial_population[i] for i in sorted_indices[:elite_size]]
        while len(next_pop) < len(initial_population):
            p1 = initial_population[sorted_indices[np.random.randint(len(sorted_indices))]]
