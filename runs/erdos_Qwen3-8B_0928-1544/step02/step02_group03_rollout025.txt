import numpy as np
from scipy.optimize import minimize, Bounds, basinhopping

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    
    n = 70
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result_slsqp = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-9,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    
    best_c5 = result_slsqp.fun
    best_h = result_slsqp.x
    
    def objective_penalty(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(h) - n / 2)
        return c5 + penalty
    
    result_bh = basinhopping(
        objective_penalty,
        x0=result_slsqp.x,
        niter=100,
        T=1.0,
        minimizer_kwargs={
            'method': 'L-BFGS-B',
            'bounds': bounds
        },
        disp=False
    )
    
    result_refined = minimize(
        fun=objective,
        x0=result_bh.x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-9,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    
    if result_refined.fun < best_c5:
        best_c5 = result_refined.fun
        best_h = result_refined.x
    
    return best_h, best_c5, n
