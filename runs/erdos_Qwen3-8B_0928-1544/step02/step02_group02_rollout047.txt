import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values
    else:
        initial_guess = np.full(n, 0.5)
    
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = initial_guess
        best_c5 = objective(initial_guess)
    
    if time.time() - start_time > budget_s:
        pass
    
    return best_h, best_c5, n
