import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    pop = []
    
    # Initialize population
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
        h = h * (35.0 / np.sum(h))
        pop.append(h)
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (35.0 / np.sum(h))
        pop.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate
        c5 = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5.append(np.max(corr) * dx)
        c5 = np.array(c5)
        idx = np.argsort(c5)
        
        # Update best
        if c5[idx[0]] < best_c5:
            best_c5 = c5[idx[0]]
            best_h = pop[idx[0]]
        
        # Selection
        parents = [pop[idx[0]], pop[idx[1]]]
        
        # Crossover
        idx_sp = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parents[0][:idx_sp], parents[1][idx_sp:]])
        child2 = np.concatenate([parents[1][:idx_sp], parents[0][idx_sp:]])
        
        # Mutation
        def mutate(h):
            h = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h[i] = np.random.rand()
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (35.0 / h_sum)
            return h
        
        child1 = mutate(child1)
        child2 = mutate(child2)
        
        # Replace worst
        pop[-2:] = [child1, child2]
    
    return best_h, best_c5, n_points
