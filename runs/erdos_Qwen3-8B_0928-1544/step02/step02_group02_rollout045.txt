import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n

    particles = np.random.rand(num_particles, n)

    def scale(h):

            return np.full(n, 0.5)
        return np.clip(h * (n/2) / s, 0, 1)
    
    for i in range(num_particles):

    # Initialize velocities

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Track best
    best_h = particles[0].copy()
    best_c5 = objective(best_h)
    personal_best = particles.copy()
    global_best = particles[0].copy()
    
    # PSO parameters

    c1, c2 = 2.0, 2.0
    
    iterations = 0
