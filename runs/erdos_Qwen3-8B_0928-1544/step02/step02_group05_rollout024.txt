import numpy as np
import random
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if initial_h_values is None:
        n_points = 140
        h = np.full(n_points, 0.5)
    else:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    random.seed(seed)
    np.random.seed(seed)
    
    def project_to_feasible(h):
        clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scale = (n_points / 2.0) / sum_clipped
        return clipped * scale
    
    dx = 2.0 / n_points
    
    def compute_C5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    h = project_to_feasible(h)
    current_h = h.copy()
    current_cost = compute_C5(current_h)
    best_h = current_h.copy()
    best_cost = current_cost
    T = 1000.0
    alpha = 0.99
    start_time = time.time()
    
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
        h_new = current_h.copy()
        for i in indices:
            h_new[i] = np.random.uniform(0, 1)
        h_new = project_to_feasible(h_new)
        new_cost = compute_C5(h_new)
        if new_cost < current_cost:
            current_h = h_new
            current_cost = new_cost
        else:
            prob = np.exp(-(new_cost - current_cost) / T)
            if random.random() < prob:
                current_h = h_new
                current_cost = new_cost
        if current_cost < best_cost:
            best_cost = current_cost
            best_h = current_h.copy()
        T *= alpha
    
    return (best_h, best_cost, n_points)
