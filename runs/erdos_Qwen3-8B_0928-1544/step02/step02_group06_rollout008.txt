import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        initial_guesses.append(np.full(n_points, n_points / 2) / n_points)
    
    def generate_diverse_guesses():
        num_guesses = 10
        for _ in range(num_guesses - 1):
            num_blocks = np.random.randint(2, 5)
            starts = np.random.uniform(0.0, 2.0 - 0.1, num_blocks)
            starts.sort()
            lengths = np.random.uniform(0.1, 0.3, num_blocks)
            lengths = lengths / np.sum(lengths)
            h = np.zeros(n_points)
            for i in range(num_blocks):
                start = starts[i]
                length = lengths[i]
                end = start + length
                s = int(np.round(start / dx))
                e = int(np.round(end / dx))
                h[s:e] = 1.0
            initial_guesses.append(h)
    
    generate_diverse_guesses()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        if result.success:
            current_h = result.x
        else:
            current_h = guess
        current_h = np.clip(current_h, 0.0, 1.0)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = guess
        curr_c5 = objective(current_h)
        if curr_c5 < best_c5:
            best_h = current_h
            best_c5 = curr_c5
    
    if best_h is not None:
        result_refine = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
        if result_refine.success:
            best_h = result_refine.x
            best_c5 = result_refine.fun
        else:
            best_h = best_h
            best_c5 = objective(best_h)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
