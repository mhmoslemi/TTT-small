from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        initial_guesses.append(np.full(n_points, 0.5))
        for _ in range(5):
            h = np.random.rand(n_points)
            h = (h - h.min()) / (h.max() - h.min())
            h *= (n_points / 2) / np.sum(h)
            initial_guesses.append(h)
        initial_guesses.append(np.zeros(n_points))
        initial_guesses.append(np.ones(n_points))

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')
    elapsed = 0

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False, 'eps': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess
            current_c5 = objective(current_h)
        current_h = np.clip(current_h, 0.0, 1.0)
        if abs(np.sum(current_h) - n_points / 2) > 1e-6:

            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    if best_h is not None:
