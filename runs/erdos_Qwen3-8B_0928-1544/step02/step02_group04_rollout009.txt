def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():

        n_points = len(initial_h)

        return h * (n_points / 2) / np.sum(h)

        sum_h = np.sum(h)
        return h * (n_points / 2) / sum_h if sum_h > 0 else np.zeros(n_points)

    def uniform(n_points):
        h = np.random.uniform(0, 1, n_points)
        return h * (n_points / 2) / np.sum(h)

        population.append(sine_wave(n_points))

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def fitness(h):

    start_time = time.time()
    while time.time() - start_time < budget_s:
