def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    # Check if initial_h_values is available
    n = 70
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values
    else:
        initial_guess = np.full(n, 0.5)
    
    dx = 2.0 / n
    
    # Objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraint: sum(h) = n/2
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 10000, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    current_h = result.x if result.success else initial_guess
    current_c5 = objective(current_h)
    
    # Ensure within budget
    if time.time() - start_time > budget_s:
