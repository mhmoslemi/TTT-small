import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Initial population setup
    pop_size = 50
    initial_h = initial_h_values.copy()
    pop = []
    for _ in range(pop_size):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    np.random.shuffle(pop)
    best_h = initial_h
    best_c5 = float('inf')

    # Parameters
    tournament_size = 3
    mutation_rate = 0.1
    crossover_rate = 0.8
    max_generations = 1500
    start_time = time.time()
    generation = 0

    while time.time() - start_time < budget_s and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in pop:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx]
        # Elitism
        new_pop = [pop[best_idx]]
        while len(new_pop) < pop_size:
            p1 = np.random.choice(len(pop), size=tournament_size, replace=False)
            p1_idx = p1[np.argmin(fitness[p1])]
            p2 = np.random.choice(len(pop), size=tournament_size, replace=False)
            p2_idx = p2[np.argmin(fitness[p2])]
            parent1 = pop[p1_idx]
            parent2 = pop[p2_idx]
            # Crossover
            if np.random.rand() < crossover_rate:
                child = (parent1 + parent2) / 2
            else:
                child = parent1.copy()
            # Mutation
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                child[indices] = np.random.uniform(0, 1, len(indices))
            # Feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_pop.append(child)
        pop = new_pop
        generation += 1

    return (best_h, best_c5, n_points)
