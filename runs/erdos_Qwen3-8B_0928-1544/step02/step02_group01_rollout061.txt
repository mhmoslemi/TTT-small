        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = initial_h_values + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        scale = (n_points / 2) / sum_perturbed if sum_perturbed != 0 else 1.0
        h_perturbed *= scale
        initial_population.append(h_perturbed)
    
    current_population = initial_population
    best_h = initial_h_values.copy()
    best_c5 = compute_C5(best_h)
    sigma = 0.1
    previous_best = best_c5
    start_time = time.time()
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        mutated_candidates = []
        for h in current_population:
            noise = np.random.normal(0, sigma, n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new = np.sum(new_h)
            scale = (n_points / 2) / sum_new if sum_new != 0 else 1.0
            new_h *= scale
            mutated_candidates.append(new_h)
        
        extended_population = current_population + mutated_candidates
        c5_values = [compute_C5(h) for h in extended_population]
        num_keep = int(len(extended_population) * 0.5)
        sorted_indices = np.argsort(c5_values)
        new_population = [extended_population[i] for i in sorted_indices[:num_keep]]
        
        for h in new_population:
            current_c5 = compute_C5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        if best_c5 < previous_best:
            sigma *= 1.05
        else:
            sigma *= 0.95
        previous_best = best_c5
        sigma = np.clip(sigma, 0.01, 1.0)
        current_population = new_population
    
    return best_h, best_c5, n_points
