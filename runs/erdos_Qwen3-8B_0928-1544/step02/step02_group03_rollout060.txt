import numpy as np
import random
import time
from scipy.stats import dirichlet

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    
    # Define constants
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    
    # Helper functions
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def create_random_individual():
        h = dirichlet(alpha=np.ones(n_points)).rvs(1)[0]
        h = h * 35.0
        h = np.clip(h, 0, 1)
        return h
    
    # Initialize population
    if initial_h_values is not None:
        population = [np.array(initial_h_values).copy()]
    else:
        population = []
    for _ in range(pop_size - 1 if initial_h_values is not None else pop_size):
        population.append(create_random_individual())
    
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Main genetic algorithm loop
    for gen in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        
        # Select parents via tournament selection
        parents = []
        for _ in range(pop_size // 2):
            tournament = random.sample(population, tournament_size)
            winner = min(tournament, key=compute_c5)
            parents.append(winner)
        
        # Crossover to create offspring
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1, parent2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutate offspring
        mutated = []
        for child in offspring:
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                child[idx] = np.random.rand()
                # Ensure sum constraint and bounds
                total = np.sum(child)
                child = np.clip(child, 0, 1)
                if total != 0:
                    child = child * (35.0 / total)
                child = np.clip(child, 0, 1)
            mutated.append(child)
        
        # Replace worst half with mutated offspring
        sorted_indices = np.argsort(fitness)
        for i in range(len(population) // 2):
            population[sorted_indices[i]] = mutated[i]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx]
        current_c5 = fitness[current_best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
    
    return best_h, best_c5, n_points
