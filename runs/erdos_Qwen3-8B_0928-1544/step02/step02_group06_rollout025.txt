import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        num_initial = 20
        for _ in range(num_initial):
            block1_len = np.random.randint(4, 124)
            block2_len = 128 - block1_len
            start1 = np.random.randint(0, 128 - block1_len)
            h = np.zeros(n_points)
            h[start1:start1 + block1_len] = 1.0
            h[start1 + block1_len:start1 + block1_len + block2_len] = 1.0
            initial_guesses.append(h)
        initial_guesses.append(np.full(n_points, n_points / 2) / n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        result = minimize(fun=objective, x0=guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        if result.success:
            current_h, current_c5 = result.x, result.fun
        else:
            current_h, current_c5 = guess, objective(guess)
        current_h = np.clip(current_h, 0, 1)
        if abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h, current_c5 = guess, objective(guess)
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
    
    if best_h is not None:
        result_refine = minimize(fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=cons,
                                 options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        if result_refine.success:
            best_h, best_c5 = result_refine.x, result_refine.fun
    
    best_h = np.clip(best_h, 0, 1)
    if abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h, best_c5 = initial_guesses[0], objective(initial_guesses[0])
    
    return (best_h, best_c5, n_points)
