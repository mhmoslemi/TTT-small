import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    pop_size = 100
    tournament_size = 5
    mut_rate = 0.1
    gen_limited = 50
    
    def fitness(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        sm = np.sum(h)
        return h * (n_points / 2) / sm if sm > 1e-12 else h
    
    pop = [initial_h.copy() for _ in range(pop_size)]
    for _ in range(pop_size - 1):
        h = initial_h + np.random.normal(0, 0.1, n_points)
        h = clip_rescale(h)
        pop.append(h)
    
    best_h = min(pop, key=fitness)
    best_c5 = fitness(best_h)
    
    start = time.time()
    for gen in range(gen_limited):
        if time.time() - start > budget_s:
            break
        parents = []
        for _ in range(pop_size):
            t = np.random.choice(pop, tournament_size)
            w = min(t, key=fitness)
            parents.append(w)
        
        offsprings = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < mut_rate:
                idx = np.random.choice(n_points)
                child[idx] = np.random.uniform(0, 1)
            child = clip_rescale(child)
            offsprings.append(child)
        
        combined = pop + offsprings
        f = [fitness(x) for x in combined]
        sorted_idx = np.argsort(f)
        new_pop = [combined[i] for i in sorted_idx[:pop_size]]
        pop = new_pop
        
        curr_best = min(pop, key=fitness)
        curr_c5 = fitness(curr_best)
        if curr_c5 < best_c5:
            best_h = curr_best
            best_c5 = curr_c5
    
    return (best_h, best_c5, n_points)
