from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    # Initial guess

    mutation_rate = 0.1
    max_iter = int(budget_s // 2)
    no_improvement = 0

    particles = []
    velocities = []
    for _ in range(num_particles):
        if np.random.rand() < 0.5:
            h = initial_guess.copy()
        else:
            h = np.random.uniform(0, 1, n)
        # Normalize sum
        h_sum = np.sum(h)
        h = h * (n / 2) / h_sum if h_sum != 0 else np.full(n, 0.5)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n))
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_pos = particles[0]
    best_val = evaluate(best_pos)
    personal_best = [p.copy() for p in particles]
    personal_best_val = [evaluate(p) for p in particles]
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break

            h += v
            
            # Clip and normalize

            h_sum = np.sum(h)

                h = h * (n / 2) / h_sum

                h = np.full(n, 0.5)

            current_val = evaluate(h)
            
            # Update personal best

            # Update global best
            if current_val < best_val:
