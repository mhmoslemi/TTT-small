import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n = 100
    dx = 2.0 / n
    
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        original_x = np.linspace(0, 2, 70)
        new_x = np.linspace(0, 2, n)
        initial_h = np.interp(new_x, original_x, initial_h_values)
    else:
        initial_h = np.full(n, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + 0.01 * np.sum(np.abs(h))
    
    def constraint(h):
        return np.sum(h) - n / 2
    
    bounds = np.array([[0.0, 1.0]] * n)
    
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = initial_h
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n
