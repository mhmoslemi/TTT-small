import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

        initial_population.append(initial_h_values)

        if base is not None:
            return [base]

        h /= np.sum(h) * (n_points / 2) / (n_points)

        return [h]
    
    while len(initial_population) < 20:
        initial_population += generate_initial_h(n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    constraints = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_population[0]
    best_c5 = objective(best_h)
    
    for guess in initial_population:
        result = minimize(

            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess

        best_c5 = objective(best_h)
