    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scale
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    start_time = time.time()
    alpha = 0.01
    perturbation_freq = 100
    perturb_amp = 0.01
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1-h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            if 0 <= m + k < n_points:
                grad[m] += 1 - h[m + k]
            if 0 <= m - k < n_points:
                grad[m] -= h[m - k]
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale
        if np.random.randint(0, perturbation_freq) == 0:
            h += np.random.normal(0, perturb_amp, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h *= scale
    return best_h, best_c5, n_points
