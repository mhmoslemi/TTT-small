from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n

    # Use initial_h_values if available, else start with 0.5
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(n, 0.5)

    # Objective function: maximize the overlap integral (we minimize it)
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: sum(h) == n / 2
    def constraint_sum(h):
        return np.sum(h) - n / 2

    # Bounds for h
    bounds = Bounds([0.0]*n, [1.0]*n)

    # Optimization parameters
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )

    # If optimization failed, return initial guess
    if not result.success:
        return h_initial, objective(h_initial), n

    # Check time budget
    if time.time() - start_time > budget_s:
        return result.x, result.fun, n

    return result.x, result.fun, n
