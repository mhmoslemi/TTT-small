import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    n_required = n_points // 2

    def evaluate_individual(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Initialize population using initial_h_values if available
    population = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        # Ensure sum is correct
        if np.abs(np.sum(h_initial) - n_required) > 1e-6:
            h_initial = np.zeros(n_points)
            indices = np.random.choice(n_points, size=n_required, replace=False)
            h_initial[indices] = 1.0
        population.append(h_initial)
    else:
        for _ in range(50):
            indices = np.random.choice(n_points, size=n_required, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            population.append(h)

    # GA parameters
    generations = 100
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8

    for gen in range(generations):
        fitness = [evaluate_individual(h) for h in population]
        sorted_indices = np.argsort(fitness)
        best_indices = sorted_indices[:int(population_size * 0.2)]
        best_individuals = [population[i] for i in best_indices]

        new_population = best_individuals[:]
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(best_individuals, size=2, replace=False)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_rate:
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    idx1, idx2 = np.random.choice(ones, 1), np.random.choice(zeros, 1)
                    child[idx1], child[idx2] = 0, 1
            new_population.append(child)

        population = new_population

    best_idx = np.argmin([evaluate_individual(h) for h in population])
    best_h = population[best_idx].copy()

    # Refinement with SLSQP
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return np.max(corr)

    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - n_required,
        'jac': lambda x: np.ones_like(x)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )

    if result.success:
        best_h = result.x
    else:
        best_h = best_h

    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_required) > 1e-6:
        best_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_required, replace=False)
        best_h[indices] = 1.0

    return (best_h, result.fun if result.success else evaluate_individual(best_h), n_points)
