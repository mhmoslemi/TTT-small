    result_higher = minimize(objective, new_h_clipped, method='SLSQP', bounds=bounds_higher, constraints=cons_higher, options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8})
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        current_c5_higher = result_higher.fun * (2.0 / n_points_higher)
        if current_c5_higher < best_c5:
            best_c5 = current_c5_higher
            best_h = result_higher.x
    
    return (best_h, best_c5, n_points_higher if n_points_higher != n_points_initial else n_points_initial)
