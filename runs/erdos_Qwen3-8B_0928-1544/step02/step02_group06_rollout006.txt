import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        num_initial = 5
        for _ in range(num_initial):
            h = np.random.rand(n_points)
            h = (h - np.min(h)) / (np.max(h) - np.min(h))
            h = h * (n_points / 2) / np.sum(h)
            initial_guesses.append(h)
        initial_guesses.append(np.full(n_points, n_points / 2) / n_points)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def objective(h):
        return compute_c5(h)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = np.copy(initial_guesses[0])
    best_c5 = compute_c5(initial_guesses[0])
    
    for guess in initial_guesses:
        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_h = res.x
        else:
            curr_h = guess
        curr_c5 = compute_c5(curr_h)
        if curr_c5 < best_c5:
            best_h = curr_h
            best_c5 = curr_c5
    
    refined_res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = refined_res.x
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
