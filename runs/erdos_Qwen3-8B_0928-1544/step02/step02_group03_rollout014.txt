import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    # Check if initial_h_values are provided
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Use uniform initial guess
        h = np.full(70, 0.5)
    dx = 2.0 / 70
    best_h = h.copy()
    best_c5 = objective(h)  # Initial objective value
    bounds = Bounds([0.0]*70, [1.0]*70)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35}

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Optimization loop
    for _ in range(20):  # Basin-hopping iterations
        # Random perturbation
        noise = np.random.normal(0, 0.05, size=70)
        h_perturb = h + noise
        h_perturb = np.clip(h_perturb, 0, 1)
        # Normalize sum to 35
        h_perturb = h_perturb * 35.0 / np.sum(h_perturb)
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturb,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h = result.x  # Update current solution
    return best_h, best_c5, 70
