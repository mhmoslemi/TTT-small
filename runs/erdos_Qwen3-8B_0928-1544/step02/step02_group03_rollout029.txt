import numpy as np
import scipy.stats
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    max_gen = int(budget_s / 5)
    mutation_rate = 0.1
    tournament_size = 10
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        population = [np.array(initial_h_values)]
        for _ in range(pop_size - 1):
            alpha = np.ones(n_points)
            dirichlet = scipy.stats.dirichlet.rvs(alpha, size=1)
            h = dirichlet * 35
            population.append(h)
    else:
        population = []
        for _ in range(pop_size):
            alpha = np.ones(n_points)
            dirichlet = scipy.stats.dirichlet.rvs(alpha, size=1)
            h = dirichlet * 35
            population.append(h)
    
    population = np.array(population)
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([compute_c5(h, dx) for h in population])
        
        parents = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(pop_size, tournament_size, replace=False)
            best_idx = idxs[np.argmin(fitness[idxs])]
            parents.append(population[best_idx].copy())
        
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            cross_point = np.random.randint(1, n_points)
            child1 = np.concatenate((p1[:cross_point], p2[cross_point:]))
            child2 = np.concatenate((p2[:cross_point], p1[cross_point:]))
            offspring.append(child1)
            offspring.append(child2)
        
        for child in offspring:
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                new_val = np.random.uniform(0, 1)
                delta = new_val - child[idx]
                new_sum = np.sum(child) + delta
                scale = 35.0 / new_sum
                child = child * scale
                child = np.clip(child, 0, 1)
                sum_clamped = np.sum(child)
                deficit = 35 - sum_clamped
                if deficit:
                    child += deficit / n_points
                    child = np.clip(child, 0, 1)
        
        new_population = np.vstack((population, offspring))
        fits = np.array([compute_c5(h, dx) for h in new_population])
        ranked = np.stack((fits, new_population), axis=1)
        ranked = ranked[ranked[:,0].argsort()]
        population = ranked[:pop_size, 1:]
        
        current_best = population[0]
        current_c5 = compute_c5(current_best, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    return best_h, best_c5, n_points
