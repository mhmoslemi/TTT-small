import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    best_h = initial_h_values.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)

    def project_feasible(h, n):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scaling = (n / 2) / sum_h if sum_h > 0 else 0
        return h * scaling

    # Initial multi-start optimization
    np.random.seed(seed)
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + noise
        perturbed = project_feasible(perturbed, n_points)
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
