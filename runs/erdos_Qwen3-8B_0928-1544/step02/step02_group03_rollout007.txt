import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    best_h = h.copy()
    n_points = 70
    dx = 2.0 / n_points
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = Bounds(0, 1)
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    for _ in range(20):
        perturbation = np.random.normal(0, 0.05, size=h.shape)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        if np.sum(h_perturbed) != 0:
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        else:
            h_perturbed = np.full(n_points, 0.5)
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < objective(best_h):
                best_h = result.x
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    
    if result.success:
        best_h = result.x
    
    return best_h, objective(best_h), n_points
