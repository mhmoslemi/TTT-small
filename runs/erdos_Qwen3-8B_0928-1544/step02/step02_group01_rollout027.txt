from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def feasi(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / s
    
    def update_velocities(vel, h, pbest, gbest, c1, c2, r1, r2):
        vel = w * vel + c1 * r1 * (pbest - h) + c2 * r2 * (gbest - h)
        return vel
    
    n_particles = 30
    particles = np.tile(h_initial, (n_particles, 1)) + np.random.normal(0, 0.1, (n_particles, n_points))
    particles = feasi(particles)
    velocities = np.zeros((n_particles, n_points))
    pbest = particles.copy()
    gbest = particles[0].copy()
    best_c5 = compute_C5(gbest)
    
    w = 0.7
    c1, c2 = 2.0, 2.0
    improvement = np.inf
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):

            velocities[i] = update_velocities(velocities[i], particles[i], pbest[i], gbest, c1, c2, r1, r2)
            particles[i] = particles[i] + velocities[i]
