import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    initial_population = []

    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        f = np.interp(np.linspace(0, 2, n_points), x_initial, initial_h_values)
        f_sum = np.sum(f)
        f *= (n_points / 2) / f_sum
        initial_population.append(f)
    else:
        initial_population.append(np.full(n_points, 0.5))

    x = np.linspace(0, 2, n_points)
    triangular = np.minimum(x, 2 - x)
    triangular_sum = np.sum(triangular) * dx
    triangular *= (n_points / 2) / triangular_sum
    initial_population.append(triangular)

    sine = np.sin(np.pi * x / 2)
    sine_sum = np.sum(sine) * dx
    sine *= (n_points / 2) / sine_sum
    initial_population.append(sine)

    for _ in range(47):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum if h_sum != 0 else 1
        initial_population.append(h)

    population = initial_population
    best_individual = population[0]
    best_score = np.max(np.correlate(population[0], 1 - population[0], mode='full') * dx)

    start_time = time.time()

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    while time.time() - start_time < budget_s:
        scores = [evaluate(ind) for ind in population]
        top_indices = np.argsort(scores)[:len(population)//2]
        top_population = [population[i] for i in top_indices]

        new_population = []
        for _ in range(len(population)):
            parent1 = np.random.choice(top_population)
            parent2 = np.random.choice(top_population)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:split], parent2[split:]])

            if np.random.rand() < 0.05:
                num_mutate = np.random.randint(1, 8)
                indices = np.random.choice(n_points, size=num_mutate, replace=False)
                for i in indices:
                    val = child[i]
                    delta = (np.random.rand() - 0.5) * 0.1
                    new_val = np.clip(val + delta, 0, 1)
                    child[i] = new_val

            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            child *= (n_points / 2) / child_sum if child_sum != 0 else 1
            new_population.append(child)

        population = new_population
        current_scores = [evaluate(ind) for ind in population]
        best_idx = np.argmin(current_scores)
        best_score = current_scores[best_idx]
        best_individual = population[best_idx]

    return best_individual, best_score, n_points
