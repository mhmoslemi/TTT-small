import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Initialize with parent solution if available
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        # Diverse initialization: blocks, random, and uniform
        initial_guesses.append(np.full(n_points, n_points / 2) / n_points)  # Uniform
        for _ in range(10):
            # Two blocks with varying spacing
            start1 = np.random.randint(0, n_points - 64)
            start2 = np.random.randint(start1 + 64, n_points - 64)
            h = np.zeros(n_points)
            h[start1:start1+64] = 1.0
            h[start2:start2+64] = 1.0
            initial_guesses.append(h)
        for _ in range(10):
            # Randomized sparse blocks
            positions = np.random.choice(n_points, size=128, replace=False)
            h = np.zeros(n_points)
            h[positions] = 1.0
            initial_guesses.append(h)
        for _ in range(5):
            # Single large block in different positions
            start = np.random.randint(0, n_points - 128)
            h = np.zeros(n_points)
            h[start:start+128] = 1.0
            initial_guesses.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed = 0.0
    
    for guess in initial_guesses:
        # Early stopping based on time
        if elapsed >= budget_s - 200:
            break
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess
            current_c5 = objective(current_h)
        current_h = np.clip(current_h, 0.0, 1.0)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = guess
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            elapsed += result.status * 10  # Placeholder for timing
        
    # Refinement phase
    if best_h is not None:
        result_refine = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )
        if result_refine.success:
            best_h = result_refine.x
            best_c5 = result_refine.fun
        else:
            best_h = best_h
            best_c5 = objective(best_h)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
