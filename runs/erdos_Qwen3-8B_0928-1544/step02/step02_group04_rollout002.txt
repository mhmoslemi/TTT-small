        initial_h_resampled = f(x_new)
        current_sum = np.sum(initial_h_resampled)
        scale_factor = (new_n_points / 2) / current_sum
        initial_h_resampled *= scale_factor
        initial_h = initial_h_resampled
    else:
        new_n_points = 512
        initial_h = np.ones(new_n_points) * (new_n_points / 2) / new_n_points

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]

    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,

    if result.success:
