import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    population = [initial_h.copy()]
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    mutation_step = 0.1
    pop_size = 30
    gen = 0
    start_time = time.time()

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def mutate(h, step):
        perturbed = h + np.random.normal(0, step, len(h))
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        scale = (n_points / 2) / total if total != 0 else 1
        return perturbed * scale

    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(pop_size):
            h = np.random.choice(population)
            new_h = mutate(h, mutation_step)
            new_pop.append(new_h)
        scores = [compute_C5(h) for h in new_pop]
        combined = sorted(zip(new_pop, scores), key=lambda x: x[1])
        new_pop = [h for h, _ in combined[:int(len(combined) * 0.25)]]
        population = new_pop
        for h, c in zip(new_pop, scores):
            if c < best_c5:
                best_h, best_c5 = h, c
        if np.random.rand() < 0.1:
            mutation_step *= 0.95
        gen += 1

    return best_h, best_c5, n_points
