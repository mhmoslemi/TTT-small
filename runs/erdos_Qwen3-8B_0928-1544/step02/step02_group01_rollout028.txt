        perturbation = np.random.normal(0, 0.1, n_points)
        candidate = h_initial + perturbation
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        scale_candidate = (n_points / 2) / sum_candidate if sum_candidate != 0 else 1.0
        candidate = candidate * scale_candidate
        population.append(candidate)

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    c5_values = [compute_C5(h) for h in population]
    best_index = np.argmin(c5_values)
    best_h = population[best_index].copy()
    best_c5 = c5_values[best_index]
    sigma = 0.1
    time_start = time.time()

    while time.time() - time_start < budget_s:
        new_population = []
        for h in population:
            noise = np.random.normal(0, sigma, n_points)
            mutated = h + noise
            mutated = np.clip(mutated, 0, 1)
            sum_mut = np.sum(mutated)
            scale_mut = (n_points / 2) / sum_mut if sum_mut != 0 else 1.0
            mutated *= scale_mut
            new_population.append(mutated)
        
        new_c5_values = [compute_C5(h) for h in new_population]
        best_new_idx = np.argmin(new_c5_values)
        new_population[best_new_idx] = best_h
        new_population.sort(key=lambda x: compute_C5(x))

        population = new_population[:30]
        if new_c5_values[best_new_idx] < best_c5:
            best_c5 = new_c5_values[best_new_idx]
            best_h = population[best_new_idx].copy()

        if new_c5_values[best_new_idx] < best_c5:
            sigma *= 1.05
        else:
            sigma *= 0.95

    return best_h, best_c5, n_points
