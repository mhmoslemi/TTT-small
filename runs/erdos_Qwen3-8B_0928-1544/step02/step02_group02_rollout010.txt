from scipy.optimize import minimize, Bounds
import time

    def get_initial_guess(n):

        return np.full(n, 0.5)
    
    # Define grid
    n = 70
    dx = 2.0 / n
    initial_guess = get_initial_guess(n)
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraint: sum(h) = n/2
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    # Bounds for h
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Optimization setup
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 1000, 'disp': False}
    )
    
    # Check time budget
