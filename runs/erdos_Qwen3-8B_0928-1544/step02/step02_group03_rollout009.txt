import numpy as np
from scipy.optimize import basinhopping, Bounds, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_init = initial_h_values.copy()
    else:
        h_init = np.full(n_points, 0.5)
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint}]
    bounds = Bounds(0, 1)
    
    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / total
        return h * scale
    
    def eval_func(x):
        return objective(project(x))
    
    # Basin Hopping with reduced iterations for time
    result_bh = basinhopping(eval_func, h_init, niter=50, T=1.0, stepsize=0.5,
                             minimizer_kwargs={'method': 'SLSQP', 'bounds': bounds,
                                               'constraints': constraints, 'tol': 1e-8})
    
    best_h = result_bh.x
    
    # SLSQP refinement with higher tolerance
    result_slsqp = minimize(objective, best_h, method='SLSQP', bounds=bounds,
                            constraints=constraints, tol=1e-8, options={'maxiter': 200})
    
    if result_slsqp.success:
        best_h = result_slsqp.x
        c5 = result_slsqp.fun
    else:
        c5 = objective(best_h)
    
    # Final check and time limit
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, c5, n_points
    return best_h, c5, n_points
