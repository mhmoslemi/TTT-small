def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    initial_pop_size = 50
    population = []
    start_time = time.time()

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:

        norm_h = np.clip(init_h * scale, 0, 1)
        population.append(norm_h)
        for _ in range(initial_pop_size - 1):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = norm_h + perturb
            new_h = np.clip(new_h, 0, 1)
            total = np.sum(new_h)
            scale = (n_points / 2) / total
            new_h = np.clip(new_h * scale, 0, 1)
            population.append(new_h)
    else:
        for _ in range(initial_pop_size):
            h = np.random.rand(n_points)
            total = np.sum(h)
            scale = (n_points / 2) / total
            h = np.clip(h * scale, 0, 1)
            population.append(h)
    
    best_h = population[0]
    best_c5 = float('inf')
    generations = 200
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [correlate(h, 1 - h, mode='full').max() * dx for h in population]
        curr_best = population[np.argmin(fitness)]
        curr_c5 = fitness[np.argmin(fitness)]
        if curr_c5 < best_c5:

            best_h = curr_best
