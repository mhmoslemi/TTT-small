import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    n = 100
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        x_initial = np.linspace(0, 2, 70)
        x_new = np.linspace(0, 2, n)
        initial_h = np.interp(x_new, x_initial, initial_h_values)
    else:
        initial_h = np.full(n, 0.5)
    def objective(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    def constraint_sum(h_val):
        return np.sum(h_val) - n / 2
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )
    return result.x, result.fun, n
