import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    best_c5 = float('inf')
    best_h = None
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def new_individual():
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * 35.0 / h_sum
        h = np.clip(h, 0, 1)
        return h
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
        initial_sum = np.sum(initial_h)
        if abs(initial_sum - 35) > 1e-6:
            initial_h = initial_h * 35.0 / initial_sum
        population = [initial_h]
        for _ in range(pop_size - 1):
            population.append(new_individual())
    else:
        population = [new_individual() for _ in range(pop_size)]
    
    for h in population:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    start_time = time.time()
    elapsed = 0
    while elapsed < budget_s:
        parents = []
        for _ in range(pop_size // 2):
            subset = random.sample(population, min(10, len(population)))
            parents.append(min(subset, key=evaluate))
        
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else parents[0]
            split = np.random.randint(1, n_points)
            child1 = np.concatenate((p1[:split], p2[split:]))
            child2 = np.concatenate((p2[:split], p1[split:]))
            offspring.append(child1)
            offspring.append(child2)
        
        for child in offspring:
            for j in range(n_points):
                if random.random() < 0.1:
                    child[j] = np.random.rand()
        
        for child in offspring:
            s = np.sum(child)
            child = np.clip(child * 35.0 / s, 0, 1)
        
        updated = population + offspring
        updated.sort(key=evaluate)
        for i in range(len(population)//2):
            population[i] = updated[len(population)//2 + i]
        
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        elapsed = time.time() - start_time
    
    return best_h, best_c5, n_points
