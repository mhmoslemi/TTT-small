                    s = np.random.randint(0, n//2)
                    l = np.random.randint(1, n//2)
                    g = np.zeros(n)
                    g[s:s+l] = 1.0
                    guesses.append(g)
        else:
            if best_h is None:
                g = np.full(n, n/2) / n
                guesses.append(g)
            else:
                prev = best_h
                h_up = upscale(prev, levels[i-1], n)
                for _ in range(5):
                    noise = np.random.randn(n) * 0.1
                    p = h_up + noise
                    p = np.clip(p, 0, 1)
                    t = np.sum(p)
                    if t != 0:
                        p = p * (n / 2) / t
                    guesses.append(p)
                guesses.append(h_up)

        start = time.time()
        best_val = float('inf')
        best_h_val = None
        for g in guesses:
            if not is_feasible(g, n):
                continue
            result = minimize(lambda x: objective(x, dx), g,
                              bounds=[(0,1)]*n,
                              constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n/2},
                              method='SLSQP',
                              options={'maxiter': 500, 'ftol': 1e-9})
            if result.success:
                h_opt = result.x
                c5 = result.fun
            else:
                h_opt = g
                c5 = objective(h_opt, dx)
            h_opt = np.clip(h_opt, 0, 1)
            if not is_feasible(h_opt, n):
                h_opt = g
                c5 = objective(h_opt, dx)
            if c5 < best_val:
                best_val = c5
                best_h_val = h_opt.copy()
        elapsed += time.time() - start
        if elapsed >= budget_s:
            break
        if best_h_val is not None and best_val < best_c5:
            best_h = best_h_val
            best_c5 = best_val

    return (best_h, best_c5, 256)
