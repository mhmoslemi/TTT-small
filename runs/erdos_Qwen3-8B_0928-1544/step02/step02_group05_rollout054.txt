from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Perturbations
    perturbed_starts = [
        initial_h + np.random.normal(0, 0.05, n_points) for _ in range(5)
    ]
    feasible_starts = []
    for p in perturbed_starts:
        p_clipped = np.clip(p, 0, 1)
        sum_p = np.sum(p_clipped)
        scaling = (n_points / 2) / sum_p
        feasible_starts.append(p_clipped * scaling)

    best_result = None
    start_time = time.time()
    for start in feasible_starts:
        result = minimize(
            objective,
            start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if best_result is None or result.fun < best_result.fun:
            best_result = result

    elapsed = time.time() - start_time

        sum_h = np.sum(new_h)

        new_h_scaled = new_h * scaling
        new_h_clipped = np.clip(new_h_scaled, 0, 1)
