import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [initial_h_values.copy()]
    for _ in range(population_size - 1):

        sum_rand = np.sum(h_rand)
        scale = (n_points / 2) / sum_rand if sum_rand != 0 else 1.0
        h_rand *= scale
        population.append(h_rand)

    best_h = initial_h_values.copy()
    best_c5 = compute_C5(best_h)
    mutation_step = 0.1
    start_time = time.time()

    while time.time() - start_time < budget_s:
        offspring = []
        for h in population:
            perturbation = np.random.normal(0, mutation_step, n_points)
            mutated = h + perturbation
            mutated = np.clip(mutated, 0, 1)
            sum_mut = np.sum(mutated)
            scale = (n_points / 2) / sum_mut if sum_mut != 0 else 1.0
            mutated *= scale
            offspring.append(mutated)

        evaluations = [compute_C5(h) for h in offspring]
        best_offspring_idx = np.argmin(evaluations)
        best_offspring = offspring[best_offspring_idx]
        current_c5 = evaluations[best_offspring_idx]

        if current_c5 < best_c5:

            best_c5 = current_c5

        new_population = [best_offspring]
