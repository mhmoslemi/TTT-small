import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = initial_h_values.copy()
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0, 1)  # Ensure bounds

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)

    result = minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-6,
        options={
            'maxiter': 10000,
            'ftol': 1e-9,
            'eps': 1e-8,
            'disp': False
        }
    )

    if time.time() - start_time > budget_s - 5:
        return h, objective(h), n

    if result.success:
        h = result.x
    return h, objective(h), n
