import numpy as np
from scipy.optimize import minimize, Bounds, basin_hopping

def run(seed=42, budget_s=1100, **kwargs):
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n, 0.5)
    
    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        s = np.sum(x_clipped)
        if s == 0:
            return float('inf')
        x_norm = x_clipped * (n / 2) / s
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx
    
    # Basin-Hopping global search
    res_bh = basin_hopping(
        fun=obj,
        x0=h,
        niter=100,
        T=1.0,
        seed=seed,
        method='L-BFGS-B'
    )
    best = res_bh.x
    
    # SLSQP refinement
    def con(x):
        return np.sum(x) - n / 2
    bounds = Bounds([0]*n, [1]*n)
    res_slsqp = minimize(
        fun=obj,
        x0=best,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': con},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    
    if res_slsqp.success:
        h_final = res_slsqp.x
        c5 = res_slsqp.fun
    else:
        h_final = h
        c5 = obj(h)
    
    return h_final, c5, n
