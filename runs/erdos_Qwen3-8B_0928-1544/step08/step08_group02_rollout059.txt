            particles[i] = np.random.rand(n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        sum_p = np.sum(particles[i])
        if sum_p == 0:
            particles[i] = np.full(n_points, ones_count / n_points)
        else:
            particles[i] = particles[i] * (ones_count / sum_p)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = particles.copy()
    pbest_c5 = np.array([objective(p) for p in particles])
    gbest = particles[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)
    
    omega = 0.7
    phi1 = 2.0
    phi2 = 2.0
    max_velocity = 0.1
    max_stagnation = 50
    stagnation_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = omega * velocities[i] + phi1 * r1 * (pbest[i] - particles[i]) + phi2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p == 0:
                particles[i] = np.full(n_points, ones_count / n_points)
            else:
                particles[i] = particles[i] * (ones_count / sum_p)
            
            current_c5 = objective(particles[i])
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
        
        if np.allclose(pbest_c5, pbest_c5.min(), atol=1e-8):
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                stagnation_count = 0
                for i in range(n_particles):
                    if i == 0:
                        particles[i] = initial_h.copy()
                    else:
                        particles[i] = np.random.rand(n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    sum_p = np.sum(particles[i])
                    if sum_p == 0:
                        particles[i] = np.full(n_points, ones_count / n_points)
                    else:
                        particles[i] = particles[i] * (ones_count / sum_p)
                pbest_c5 = np.array([objective(p) for p in particles])
                pbest = particles.copy()
                gbest = particles[np.argmin(pbest_c5)]
                gbest_c5 = np.min(pbest_c5)
        else:
            stagnation_count = 0
    
    return gbest, gbest_c5, n_points
