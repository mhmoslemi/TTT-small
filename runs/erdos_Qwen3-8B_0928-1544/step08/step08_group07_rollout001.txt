import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_phase1 = 35
    n_phase2 = 70
    pop_size = 50
    gen_phase1 = 100
    iter_phase2 = 200
    mut_std = 1.0
    mut_decay = 0.95

    def normalize(h, n):
        s = np.sum(h)
        if s == 0:
            return np.full(n, 0.5)
        return h * (n / 2) / s

    def calc_c5(h, n):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    def objective_phase2(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_phase2)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        phase1_initial = initial_h_values[::2]
    else:
        phase1_initial = np.random.rand(n_phase1)

    population = [normalize(phase1_initial + np.random.normal(0, 0.5, n_phase1), n_phase1) for _ in range(pop_size)]

    for _ in range(gen_phase1):
        if _ % 10 == 0:
            mut_std *= mut_decay
        idxs = np.random.choice(len(population), 5, replace=False)
        best = population[np.argmin([calc_c5(p, n_phase1) for p in population])]
        mut = np.random.normal(0, mut_std, n_phase1)

        population.append(new)
