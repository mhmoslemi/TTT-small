        new_conv = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_conv) * (2.0 / n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
        if new_c5 < best_c5:
            eta *= 1.1
        else:
            eta *= 0.9

    return best_h, best_c5, n_points
