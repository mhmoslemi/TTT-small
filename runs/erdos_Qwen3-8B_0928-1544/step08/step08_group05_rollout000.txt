import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    S = n_points / 2.0
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Adjust initial_h to meet the sum constraint
    sum_init = np.sum(initial_h)
    if abs(sum_init - S) > 1e-6:
        scale_factor = S / sum_init
        initial_h = initial_h * scale_factor
        initial_h = np.clip(initial_h, 0.0, 1.0)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()
    
    # Generate multiple perturbations and optimize
    num_perturbations = 5
    for _ in range(num_perturbations):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            scale = S / sum_perturbed
            perturbed *= scale
        perturbed = np.clip(perturbed, 0.0, 1.0)
        
        def objective(h):

            return np.max(corr) * (2.0 / n_points)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        res = minimize(

            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        )

            best_c5 = res.fun
