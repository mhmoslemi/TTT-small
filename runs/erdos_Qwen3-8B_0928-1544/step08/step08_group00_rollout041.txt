import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Global phase with enhanced DE
    pop_size = 100
    max_gen = 600
    F, CR = 0.8, 0.9
    pop = [np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1) * (target_sum / np.sum(np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1))) for _ in range(pop_size)]
    
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            trial = np.clip(trial, 0, 1) * (target_sum / trial_sum) if trial_sum != 0 else trial
            trial_obj = obj(trial)
            if trial_obj < obj(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    # Local refinement
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
