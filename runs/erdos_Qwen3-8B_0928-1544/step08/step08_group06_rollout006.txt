        h_k = best_h.copy()
        h_k_plus = h_k + 1e-6
        h_k_minus = h_k - 1e-6
        g_k = (np.correlate(h_k_plus, 1 - h_k_plus, mode='full')[k] - 
               np.correlate(h_k_minus, 1 - h_k_minus, mode='full')[k]) / (2e-6) * 2.0 / n_points
        
        h_k_grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = best_h.copy()
            h_perturbed[i] += 1e-6
            h_k_grad[i] = (np.correlate(h_perturbed, 1 - h_perturbed, mode='full')[k] - 
                           np.correlate(best_h, 1 - best_h, mode='full')[k]) * 2.0 / n_points / 1e-6
        
        best_h = best_h - alpha * h_k_grad
        best_h = np.clip(best_h, 0.0, 1.0)
        total = np.sum(best_h)
        if total > 0:
            best_h = best_h * (ones_count / total)
        
        if objective(best_h) < best_c5:
            best_c5 = objective(best_h)
            best_h = best_h.copy()
        
        if iterations % 100 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            best_h = np.clip(best_h + perturb, 0.0, 1.0)
            total_perturb = np.sum(best_h)
            if total_perturb > 0:
                best_h = best_h * (ones_count / total_perturb)
        alpha *= decay_rate
    
    return best_h, best_c5, n_points
