    initial_h = initial_h * (ones_count / np.sum(initial_h))
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * (2.0 / n_points)
    
    start_time = time.time()
    n = n_points
    max_iter = 1000
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_val = np.max(corr)
        k_max = np.argmax(corr)
        grad = np.zeros(n)
        for x in range(n):
            if x + k_max < n:
                term1 = 1.0 - best_h[x + k_max]
            else:
                term1 = 0.0
            if x - k_max >= 0:
                term2 = best_h[x - k_max]
            else:
                term2 = 0.0
            grad[x] = term1 - term2
        h_new = best_h - 0.01 * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h = np.sum(h_new)
        delta = (ones_count - sum_h) / n
        h_new = np.clip(h_new + delta, 0.0, 1.0)
        sum_h = np.sum(h_new)
        if abs(sum_h - ones_count) > 1e-6:
            scale = ones_count / sum_h
            h_new = np.clip(h_new * scale, 0.0, 1.0)
        current_c5 = max_val * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
    return best_h, best_c5, n_points
