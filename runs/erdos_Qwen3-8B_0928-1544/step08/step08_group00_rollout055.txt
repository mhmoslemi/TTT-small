import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Initial optimization
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 250, 'ftol': 1e-10, 'eps': 1e-8}
    )
    best_h = result.x
    best_c5 = result.fun
    
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Iterative refinement
    for _ in range(4):
        # Perturb and clamp
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        # Rescale
        if np.sum(perturbed) != 0:
            perturbed = perturbed * (target_sum / np.sum(perturbed))
        # Optimize
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
