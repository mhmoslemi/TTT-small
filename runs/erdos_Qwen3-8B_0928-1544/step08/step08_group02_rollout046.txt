import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Ensure sum is correct
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Generate 10 perturbed candidates
        candidates = []
        for _ in range(10):
            noise = np.random.normal(0, 0.1, n_points)
            h_perturbed = best_h + noise
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
            candidates.append(h_perturbed)
        
        # Evaluate and select best
        for candidate in candidates:
            current_c5 = compute_c5(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
