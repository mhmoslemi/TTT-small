from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0  # Alternating pattern

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    best_c5 = float('inf')
    best_h = h_initial.copy()
    start_time = time.time()

    # Generate initial guesses
    initial_guesses = [h_initial]
    for _ in range(50):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
        initial_guesses.append(h)

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return best_h, best_c5, n_points
