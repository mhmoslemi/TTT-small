        return np.clip(h + np.random.normal(0, scale, h.shape[0]), 0.0, 1.0)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Coarse grid optimization
    n_coarse = 34
    binary = np.zeros(n_coarse, dtype=np.float64)
    indices = np.random.choice(n_coarse, size=17, replace=False)
    binary[indices] = 1.0
    dx_coarse = 2.0 / n_coarse
    
    def coarse_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_coarse
    
    for _ in range(50):
        for i in range(n_coarse):
            h_new = binary.copy()
            h_new[i] = 1 - h_new[i]
            sum_new = np.sum(h_new)
            if sum_new != 17:
                for j in range(n_coarse):
                    if j == i:
                        continue
                    h_new[j] = 1 - h_new[j]
                    break
            if coarse_objective(h_new) < coarse_objective(binary):
                binary = h_new
    
    # Upsample and perturb
    upsampled = np.repeat(binary, 2)
    upsampled = np.append(upsampled, [1.0, 0.0])
    perturbed = perturb(upsampled)
    
    # Local refinement
    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=perturbed,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    final_h = result.x if result.success else perturbed
    final_c5 = result.fun if result.success else objective(perturbed)
    
    return final_h, final_c5, n_points
