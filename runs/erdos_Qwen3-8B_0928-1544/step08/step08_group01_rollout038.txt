        sorted_indices = np.argsort(scores)
        population = [population[i] for i in sorted_indices]
        num_parents = len(population) // 2
        parents = population[:num_parents]
        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                p1 = parents[i]
                p2 = parents[i+1]
                child1 = p1.copy()
                child2 = p2.copy()
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child1[j], child2[j] = p2[j], p1[j]
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        if np.random.rand() < 0.33:
                            child1[j] = 0.0
                        elif np.random.rand() < 0.66:
                            child1[j] = 1.0
                        else:
                            child1[j] = np.random.rand()
                sum_c1 = np.sum(child1)
                if sum_c1 != 0:
                    child1 /= (sum_c1 / target_sum)
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        if np.random.rand() < 0.33:
                            child2[j] = 0.0
                        elif np.random.rand() < 0.66:
                            child2[j] = 1.0
                        else:
                            child2[j] = np.random.rand()
                sum_c2 = np.sum(child2)
                if sum_c2 != 0:
                    child2 /= (sum_c2 / target_sum)
                offspring.append(child1)
                offspring.append(child2)
        population = population[:num_parents] + offspring
        for h in population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    scores = [compute_c5(h) for h in population]
    top_indices = np.argsort(scores)[:10]
    top_solutions = [population[i] for i in top_indices]
    for h in top_solutions:
        h = h * (target_sum / np.sum(h))
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points)
        def constraint(x):
            return np.sum(x) - target_sum
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    return best_h, best_c5, n_points
