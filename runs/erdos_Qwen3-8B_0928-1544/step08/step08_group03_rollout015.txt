import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):

        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Simulated Annealing parameters
    T = 1.5
    alpha = 0.998
    max_iterations = 1500
    no_improvement = 0
    max_no_improvement = 100

    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break

        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed *= (ones_count / np.sum(h_perturbed))

        if current_c5 < best_c5:

            no_improvement += 1
        T *= alpha

            break

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},

    if result.success:
