    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_c5 = float('inf')

    pop = [initial_h.copy() for _ in range(population_size)]
    
    for _ in range(500):
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break
        
        fitness = np.array([compute_c5(ind) for ind in pop])
        argsort = np.argsort(fitness)
        pop = np.array([pop[i] for i in argsort[:population_size//2]])
        
        new_pop = []
        for _ in range(population_size//2):
            p1, p2 = np.random.choice(population_size//2, 2, replace=False)

        pop = np.vstack([pop, new_pop])
        pop = pop[:population_size]

    for idx in top_indices:
        h = pop[idx].copy()
        result = minimize(
