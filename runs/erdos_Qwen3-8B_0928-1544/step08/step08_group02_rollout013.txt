import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    start_time = time.time()
    max_iter = 20
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        # Generate perturbed guess
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = initial_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def constraint_sum(h):
            return np.sum(h) - ones_count
        
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            initial_h = result.x
    
    return best_h, best_c5, n_points
