    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        # Evaluate population
        pop_fitness = [objective(ind) for ind in pop]
        # Select top 20%
        elite_indices = np.argsort(pop_fitness)[:int(0.2 * pop_size)]

        # Crossover
        offspring = []
