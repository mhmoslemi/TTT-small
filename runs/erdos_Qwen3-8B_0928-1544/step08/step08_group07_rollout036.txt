import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_coarse = 35
    n_fine = 70
    ones_count = n_coarse // 2
    if 'initial_h_values' in globals():
        h_coarse = initial_h_values.copy()
    else:
        h_coarse = np.full(n_coarse, 0.5)
    def compute_c5(h, n):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)
    def constraint_sum(h):
        return np.sum(h) - n_coarse / 2
    bounds = [(0.0, 1.0)] * n_coarse
    best_h = h_coarse.copy()
    best_c5 = compute_c5(best_h, n_coarse)
    start_time = time.time()
    for _ in range(200):
        h_perturbed = best_h + np.random.normal(0, 0.08, n_coarse)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        if np.sum(h_perturbed) > 0:
            h_perturbed *= (n_coarse / 2) / np.sum(h_perturbed)
        result = minimize(lambda x: compute_c5(x, n_coarse), 
                          x0=h_perturbed, 
                          method='SLSQP', 
                          bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint_sum},
                          options={'maxiter': 100, 'ftol': 1e-9})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    h_fine = np.repeat(best_h, 2)
    def constraint_fine(h):
        return np.sum(h) - n_fine / 2
    bounds_fine = [(0.0, 1.0)] * n_fine
