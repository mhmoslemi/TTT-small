import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    population = [initial_h]
    for _ in range(29):
        h = np.random.rand(n_points)
        h = h * (required_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
        population.append(h)
    population = np.array(population)
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time >= budget_s:
            break
        c5 = np.array([compute_c5(h) for h in population])
        top_idx = np.argsort(c5)[:int(0.1 * len(population))]
        top_h = population[top_idx]
        for h in top_h:
            def obj(x):
                return compute_c5(x)
            res = minimize(obj, h, method='SLSQP',
                           bounds=[(0,1)]*n_points,
                           constraints={'type':'eq', 'fun':lambda x: np.sum(x)-required_sum})
            if res.success:
                if res.fun < best_c5:
                    best_c5 = res.fun
                    best_h = res.x
        new_pop = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (required_sum / np.sum(mutant))
            new_pop.append(mutant)
        population = np.array(new_pop)
    return best_h, best_c5, n_points

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / 70)
