import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    pop_size = 50
    last_decay_time = 0.0
    sigma = 0.25
    decay_rate = 0.95

    if 'initial_h_values' in globals():

        mutants = [initial_h + np.random.normal(0, 0.25, n_points) for _ in range(10)]

        pop = mutants + [np.zeros(n_points) + (i % 2) for i in range(40)]
    else:
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0
        pop = [h_initial + np.random.normal(0, 0.25, n_points) for _ in range(pop_size)]
        pop = [np.clip(p, 0, 1) * (ones_count / np.sum(np.clip(p, 0, 1))) for p in pop]
    pop = np.array(pop)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0]
    best_c5 = objective(pop[0])
    start_time = time.time()

    while time.time() - start_time < budget_s:
        scores = np.array([objective(h) for h in pop])
        indices = np.argsort(scores)[:25]
        top_candidates = pop[indices]
        new_pop = []
        for h in top_candidates:
            for _ in range(2):
                perturbed = h + np.random.normal(0, sigma, n_points)
                perturbed = np.clip(perturbed, 0, 1)
                sum_p = np.sum(perturbed)
                if sum_p > 0:
                    perturbed = perturbed * (ones_count / sum_p)
                new_pop.append(perturbed)
        combined = np.vstack((pop, new_pop))
        scores_combined = np.array([objective(h) for h in combined])
        indices_combined = np.argsort(scores_combined)[:pop_size]
        pop = combined[indices_combined]
        current_scores = scores_combined[indices_combined]
        if np.min(current_scores) < best_c5:
            best_c5 = np.min(current_scores)
            best_h = combined[indices_combined[np.argmin(current_scores)]]
        elapsed = time.time() - start_time
        if elapsed - last_decay_time >= 10:

            last_decay_time = elapsed
    return best_h, best_c5, n_points
