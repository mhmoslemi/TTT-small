import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def project_h(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return h_clipped * scaling
    
    num_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        for _ in range(5):
            h_base = initial_h.copy()
            noise = np.random.normal(0, 0.1, n_points)
            perturbed = h_base + noise
            perturbed = project_h(perturbed)
            particles.append(perturbed)
            velocities.append(np.zeros(n_points))
        for _ in range(num_particles - 5):
            h = np.random.rand(n_points)
            h = project_h(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
    else:
        for _ in range(num_particles):
            h = np.random.rand(n_points)
            h = project_h(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
    
    pbest = [h.copy() for h in particles]
    pbest_c5 = []
    best_c5 = float('inf')
    best_h = None
    gbest = None
    
    for i in range(num_particles):
        h = particles[i]
