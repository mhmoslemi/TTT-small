    pop_coarse = np.random.uniform(0, 1, (population_size, n_points_initial))
    for i in range(population_size):
        pop_coarse[i] = repair(pop_coarse[i], n_points_initial)
    
    best_h_coarse = initial_coarse.copy()
    best_c5_coarse = evaluate_erdos_solution(best_h_coarse)
    
    t0 = time.time()
    for _ in range(500):
        if time.time() - t0 > budget_s:
            break
        new_pop = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = pop_coarse[a] + 0.8 * (pop_coarse[b] - pop_coarse[c])
            trial = pop_coarse[i].copy()
            for j in range(n_points_initial):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = repair(trial, n_points_initial)
            new_pop.append(trial)
        new_pop_c5 = [evaluate_erdos_solution(h) for h in new_pop]
        for i in range(population_size):
            if new_pop_c5[i] < evaluate_erdos_solution(pop_coarse[i]):
                pop_coarse[i] = new_pop[i]
            if new_pop_c5[i] < best_c5_coarse:
                best_c5_coarse = new_pop_c5[i]
                best_h_coarse = new_pop[i].copy()
    
    refined_h = interpolate_h(best_h_coarse, factor=2)
    
    population_size_refine = 75
    pop_refine = np.random.uniform(0, 1, (population_size_refine, n_points_refine))
    for i in range(population_size_refine):
        pop_refine[i] = repair(pop_refine[i], n_points_refine)
    
    best_h_refine = refined_h.copy()
    best_c5_refine = evaluate_erdos_solution(best_h_refine)
    
    t1 = time.time()
    for _ in range(500):
        if time.time() - t1 > budget_s:
            break
        new_pop_refine = []
        for i in range(population_size_refine):
            a, b, c = np.random.choice(population_size_refine, 3, replace=False)
            mutant = pop_refine[a] + 0.7 * (pop_refine[b] - pop_refine[c])
            trial = pop_refine[i].copy()
            for j in range(n_points_refine):
                if np.random.rand() < 0.85:
                    trial[j] = mutant[j]
            trial = repair(trial, n_points_refine)
            new_pop_refine.append(trial)
        new_pop_c5 = [evaluate_erdos_solution(h) for h in new_pop_refine]
        for i in range(population_size_refine):
            if new_pop_c5[i] < evaluate_erdos_solution(pop_refine[i]):
                pop_refine[i] = new_pop_refine[i]
            if new_pop_c5[i] < best_c5_refine:
                best_c5_refine = new_pop_c5[i]
                best_h_refine = new_pop_refine[i].copy()
    
    best_h = best_h_refine if best_c5_refine < best_c5_coarse else best_h_coarse
    best_c5 = best_c5_refine if best_c5_refine < best_c5_coarse else best_c5_coarse
    return best_h, best_c5, n_points_refine
