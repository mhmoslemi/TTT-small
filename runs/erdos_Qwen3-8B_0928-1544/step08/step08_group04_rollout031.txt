            recombination=0.9,
            seed=seed,
            polish=True,
            vectorized=True
        )
        best_coarse = result_coarse.x
    else:
        best_coarse = initial_coarse

    best_fine = np.repeat(best_coarse, 2)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        bounds_fine = [(0.0, 1.0)] * n_points_fine

        def obj_fine_perturbed(h):
            target = n_points_fine / 2
            penalty = max(0.0, np.sum(h) - target) * 1e4
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / n_points_fine) + penalty

        result_fine = differential_evolution(
            fun=obj_fine_perturbed,
            bounds=bounds_fine,
            popsize=75,
            maxiter=200,
            tol=1e-8,
            mutation=(0.5, 1.0),
            recombination=0.85,
            seed=seed,
            polish=True,
            vectorized=True,
            x0=best_fine
        )
        best_fine = result_fine.x

    corr = np.correlate(best_fine, 1 - best_fine, mode='full')
    c5_val = np.max(corr) * (2.0 / n_points_fine)
    return best_fine, c5_val, n_points_fine
