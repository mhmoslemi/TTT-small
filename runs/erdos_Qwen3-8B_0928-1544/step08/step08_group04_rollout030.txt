from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(70, 0.5)
    n_points = 70
    ones_count = n_points // 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    def evaluator(x):
        x = np.clip(x, 0, 1)
        x = x * (ones_count / np.sum(x))
        return objective(x)

    start_time = time.time()

    def optimize(n, initial_x):
        result = differential_evolution(
            evaluator,
            bounds,
            popsize=50,

            maxiter=200,
            tol=1e-7,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            disp=False
        )
        return result.x

    # Coarse grid
    n_coarse = 35
    if 'initial_h_values' in globals():
        init_coarse = np.repeat(initial_h[::2], 2)[:n_coarse]
    else:

    c_h = optimize(n_coarse, init_coarse)

    # Refine to fine grid
    n_fine = n_points

    init_fine = np.clip(init_fine, 0, 1)
    init_fine = init_fine * (ones_count / np.sum(init_fine))
    f_h = optimize(n_fine, init_fine)

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Final optimization in remaining time
        result = differential_evolution(
            evaluator,

            popsize=50,

            maxiter=200,
            tol=1e-7,
