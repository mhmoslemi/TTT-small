import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, n_points), 0, 1) * (target_sum / np.sum(np.clip(h + np.random.normal(0, scale, n_points), 0, 1)))
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h_pert = perturb(initial_h, 0.2)
        result = minimize(
            fun=objective,
            x0=h_pert,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'gtol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        initial_h = best_h
    
    def refine(h):
        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        return result.x if result.success else h
    
    best_h = refine(best_h)
    final_c5 = objective(best_h)
    
    return best_h, final_c5, n_points
