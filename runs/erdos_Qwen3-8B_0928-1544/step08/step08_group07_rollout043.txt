        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = initial_h * (required_sum / np.sum(initial_h))
    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    alpha_steps = np.linspace(0.0, 1.0, 11)
    
    for alpha in alpha_steps:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            sum_term = (np.sum(h) - required_sum) ** 2
            return c5 + alpha * sum_term
        
        bounds = [(0.0, 1.0)] * n_points
        h_perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (required_sum / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result.success:
            h_current = result.x
            h_current = np.clip(h_current, 0.0, 1.0)
            h_current = h_current * (required_sum / np.sum(h_current))
            current_c5 = objective(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                initial_h = h_current
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
