import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        n_points = len(h_initial)
    else:
        n_points = 70
        h_initial = np.full(n_points, 0.5)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Differential evolution for global search
    def objective_penalty(h):
        c = objective(h)
        return c + 100 * abs(np.sum(h) - n_points / 2)
    
    try:
        result = differential_evolution(
            objective_penalty, bounds, popsize=15, maxiter=50, tol=1e-5,
            mutation=(0.5, 1.0), recombination=0.8, seed=seed
        )
        best_h = result.x
        best_c5 = objective(best_h)
    except:
        pass
    
    # SLSQP refinement
    result = minimize(
        objective, best_h, method='SLSQP', bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Final fine-tuning
        for _ in range(3):
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
