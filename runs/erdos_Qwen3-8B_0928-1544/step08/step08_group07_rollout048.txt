        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )

    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_c5 = compute_c5(best_h, n_fine)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_fine / 2) / np.sum(best_h)
    return best_h, best_c5, n_fine
