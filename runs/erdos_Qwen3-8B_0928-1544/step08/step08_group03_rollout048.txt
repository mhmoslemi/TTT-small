import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (ones_count / total)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (ones_count / total)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    n_complexes = 5
    n_generations = 200
    mutation_scale = 0.2
    crossover_prob = 0.8
    patience = 50
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()

    # Initialize complexes
    complexes = []
    for _ in range(n_complexes):

        complex_pts = np.clip(complex_pts, 0, 1)
        complex_pts = complex_pts * (ones_count / np.sum(complex_pts))
        complexes.append(complex_pts)

    for gen in range(n_generations):
        if time.time() - start_time >= budget_s:
            break
        np.random.shuffle(complexes)
        for comp in complexes:
            values = [evaluate_c5(p) for p in comp]
            sorted_indices = np.argsort(values)
            sorted_pts = [comp[i] for i in sorted_indices]
            best_p = sorted_pts[0]
            new_pts = []
            for _ in range(len(sorted_pts)):
                if np.random.rand() < crossover_prob:
                    other_p = sorted_pts[np.random.randint(1, len(sorted_pts))]
                    new_p = best_p * 0.5 + other_p * 0.5
                else:
                    new_p = best_p + np.random.normal(0, mutation_scale, n_points)
                new_p = np.clip(new_p, 0, 1)
                total_new = np.sum(new_p)
                new_p = new_p * (ones_count / total_new)
                new_pts.append(new_p)
            new_values = [evaluate_c5(p) for p in new_pts]
            new_sorted = np.argsort(new_values)
            best_new = new_pts[new_sorted[0]]
            if evaluate_c5(best_new) < evaluate_c5(sorted_pts[-1]):
                sorted_pts[-1] = best_new
            comp[:] = sorted_pts

        current_c5 = evaluate_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif gen % patience == 0:
            curr = evaluate_c5(h)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
