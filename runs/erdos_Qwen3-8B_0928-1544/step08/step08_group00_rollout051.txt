from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points

    def binary_optimize(n_coarse=35):
        coarse_h = np.round(initial_h[:n_coarse]).copy()

        coarse_h = np.round(np.sum(coarse_h) / (n_coarse / 2) * coarse_h)
        coarse_h = np.clip(coarse_h, 0, 1)
        def bin_obj(h):
            return objective(h)
        res = minimize(bin_obj, x0=coarse_h, method='L-BFGS-B', bounds=[(0,1)]*n_coarse, tol=1e-6)
        return res.x

    coarse_h = binary_optimize()
    fine_h = np.zeros(n_points)
    fine_h[:n_points//2] = coarse_h[:n_points//2]
    fine_h[n_points//2:] = coarse_h[n_points//2:]

    fine_h = np.clip(fine_h, 0, 1)
    fine_h = fine_h * target_sum / np.sum(fine_h)

    best_h = fine_h.copy()
    best_c5 = objective(fine_h)

    start = time.time()
