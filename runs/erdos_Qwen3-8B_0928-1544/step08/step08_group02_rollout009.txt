import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 20
    F = 0.7
    CR = 0.9
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize population
    if 'initial_h_values' in globals():
        initial_population = [initial_h_values.copy()]
        for _ in range(pop_size - 1):
            h = np.random.uniform(0.0, 0.5, n_points)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full(n_points, 0.5)
            else:
                h = h * (ones_count / sum_h)
            initial_population.append(h)
    else:
        initial_population = [np.random.uniform(0.0, 0.5, n_points) for _ in range(pop_size)]
        for i in range(pop_size):
            h = initial_population[i]
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full(n_points, 0.5)
            else:
                h = h * (ones_count / sum_h)
            initial_population[i] = h

    population = np.array([np.array(h) for h in initial_population])

    for gen in range(1000):
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            mutant = a_vec + F * (b_vec - c_vec)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.full_like(trial, 0.0)
            else:
                scaling_factor = ones_count / trial_sum
                trial = trial * scaling_factor
