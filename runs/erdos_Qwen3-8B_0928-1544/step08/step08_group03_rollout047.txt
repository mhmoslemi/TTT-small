from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # SA parameters
    T = 1.0
    alpha = 0.998
    max_sa_iter = 300
    for _ in range(max_sa_iter):
        if time.time() - start_time >= budget_s:

        h_perturbed = best_h + np.random.normal(0, T, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        # Evaluate new solution
        new_c5 = objective(h_perturbed)
        # Acceptance probability
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):

    result = minimize(
