from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Enhanced SA with adaptive cooling
    temp = 1.0
    cooling_rate = 0.998
    max_sa_iter = 1500
    for _ in range(max_sa_iter):
        if time.time() - start_time >= budget_s:
            break
        h_perturbed = initial_h + np.random.normal(0, max(temp * 0.05, 0.01), n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
