import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_coarse = 35
    n_fine = 70
    ones_coarse = n_coarse // 2
    ones_fine = n_fine // 2
    
    # Initialize coarse grid
    if 'initial_h_values' in globals():
        h_coarse = initial_h_values[:n_coarse].copy()
    else:
        h_coarse = np.full(n_coarse, 0.5)
    h_coarse = h_coarse * (ones_coarse / np.sum(h_coarse))
    
    # Coarse phase: simulated annealing
    best_coarse = h_coarse.copy()
    best_c5 = evaluate_erdos_solution(best_coarse)
    T = 1.0
    start_time = time.time()
    
    for _ in range(1000):
        noise = np.random.normal(0, 0.2, n_coarse)
        h = h_coarse + noise
        h = np.clip(h, 0, 1)
        h = h * (ones_coarse / np.sum(h))
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5 or np.random.rand() < np.exp(-(c5 - best_c5) / T):
            best_c5 = c5
            best_coarse = h.copy()
        T *= 0.995
        if time.time() - start_time > budget_s:
            break
    
    # Upsample to fine grid
    h_fine = np.repeat(best_coarse, 2)
    h_fine = h_fine * (ones_fine / np.sum(h_fine))
    
    # Fine phase: simulated annealing
    best_fine = h_fine.copy()
    best_c5_fine = evaluate_erdos_solution(best_fine)
    T_fine = 1.0
    
    for _ in range(500):
        noise = np.random.normal(0, 0.1, n_fine)
        h = h_fine + noise
        h = np.clip(h, 0, 1)
        h = h * (ones_fine / np.sum(h))
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5_fine or np.random.rand() < np.exp(-(c5 - best_c5_fine) / T_fine):
            best_c5_fine = c5
            best_fine = h.copy()
        T_fine *= 0.995
        if time.time() - start_time > budget_s:
            break
    
    # Final SLSQP refinement
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_fine})
    result = minimize(evaluate_erdos_solution, best_fine, method='SLSQP',
                      bounds=[(0, 1)]*n_fine, constraints=cons,
                      options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8})
    if result.success:
        best_fine = result.x
        best_c5_fine = result.fun
    
    return best_fine, best_c5_fine, n_fine
