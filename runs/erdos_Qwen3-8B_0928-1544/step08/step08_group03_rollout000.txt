    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    mutation_scale = 0.2
    max_generations = 500
    generations_without_improvement = 0
    generations_patience = 50

    for _ in range(max_generations):
        if time.time() - start_time >= budget_s:
            break
        np.random.shuffle(complexes)
        for complex_points in complexes:
            c5_values = [evaluate_erdos_solution(h) for h in complex_points]
            best_idx = np.argmin(c5_values)
            worst_idx = np.argmax(c5_values)
            best_point = complex_points[best_idx].copy()
            mutation = np.random.normal(0, mutation_scale, n_points)
            new_point = best_point + mutation
            new_point = np.clip(new_point, 0.0, 1.0)
            new_sum = np.sum(new_point)
            if new_sum == 0:
                new_point[:] = 0
                new_point[0] = 1.0
                new_sum = 1.0
            new_point = new_point * (n_points / 2) / new_sum
            new_c5 = evaluate_erdos_solution(new_point)
            if new_c5 < c5_values[worst_idx]:
                complex_points[worst_idx] = new_point
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_point.copy()
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 == best_c5:
            generations_without_improvement += 1
            if generations_without_improvement >= generations_patience:
                break
        else:
            generations_without_improvement = 0
    return best_h, best_c5, n_points
