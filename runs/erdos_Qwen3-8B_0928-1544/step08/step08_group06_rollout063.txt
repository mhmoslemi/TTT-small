import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                initial_h[i] = 1.0
        initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    warmup = 30  # seconds
    
    if time.time() - start_time < warmup:
        perturbed = best_h.copy()
        perturbed += np.random.normal(0, 0.2, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (ones_count / np.sum(perturbed))
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 60, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    while time.time() - start_time < budget_s:
        sigma = 0.15 * (1 - (time.time() - start_time) / budget_s)
        perturbed = best_h + np.random.normal(0, sigma, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (ones_count / np.sum(perturbed))
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 60, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
