        result = minimize(obj_func, h, method='Nelder-Mead', bounds=bounds, options={'maxiter': 500, 'fatol': 1e-9})
        
        if result.success:
            current_h = result.x
        else:
            current_h = h
        
        current_c5 = compute_c5(current_h, points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, 70
