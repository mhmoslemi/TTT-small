            p1, p2 = parents[i], parents[(i + 1) % len(parents)]
            child = np.zeros_like(p1)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child[j] = p1[j]
                else:
                    child[j] = p2[j]
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    if np.random.rand() < 0.5:
                        child[j] = 0.0
                    elif np.random.rand() < 0.5:
                        child[j] = 1.0
                    else:
                        child[j] = np.random.rand()
            total = np.sum(child)
            if total > 0:
                child = child * (ones_count / total)
            offspring.append(child)
        
        new_population = parents + offspring
        evaluations_new = []
        for h in new_population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            evaluations_new.append(c5)
        sorted_new_population = [new_population[i] for i in np.argsort(evaluations_new)]
        population = sorted_new_population
        
        best_idx = np.argmin(evaluations)
        if evaluations[best_idx] < best_c5:
            best_c5 = evaluations[best_idx]
            best_h = population[best_idx].copy()
    
    evaluations = []
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        evaluations.append(c5)
    sorted_indices = np.argsort(evaluations)
    top_10 = [population[i] for i in sorted_indices[:10]]
    
    for h in top_10:
        def objective(x):
            h_current = x.copy()
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            return np.max(corr) * (2.0 / n_points)
        
        def constraint(x):
            return np.sum(x) - ones_count
        
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
