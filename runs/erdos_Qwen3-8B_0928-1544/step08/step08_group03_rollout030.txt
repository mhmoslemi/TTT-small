import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0  # Alternating 1s and 0s for better spread

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    max_sa = 200
    no_improvement = 0
    sigma = 0.05  # Initial perturbation scale

    for _ in range(max_sa):
        if time.time() - start_time >= budget_s:
            break
        noise = np.random.normal(0, sigma, n_points)
        h_perturbed = best_h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        current_c5 = objective(h_perturbed)
        delta = current_c5 - best_c5
        if delta < 0:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
            no_improvement = 0
        else:
            if np.random.rand() < np.exp(-delta / T):
                best_h = h_perturbed.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        T *= alpha
        if no_improvement >= 50:
            break

    # Gradient-based refinement
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    return best_h, best_c5, n_points
