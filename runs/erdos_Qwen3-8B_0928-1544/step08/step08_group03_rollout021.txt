import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy() * (ones_count / np.sum(initial_h))
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time >= budget_s:

        sigma = 0.05 * (1 - (time.time() - start_time) / budget_s)
        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)

        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
