from scipy.optimize import minimize

        return np.max(corr) * dx

    CR = 0.8

    max_generations = 200

        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        pop.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:

            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()

                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:

            trial_obj = objective(trial)
