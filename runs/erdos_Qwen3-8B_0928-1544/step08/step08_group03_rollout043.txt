        if time.time() - start_time >= budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = initial_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (ones_count / np.sum(perturbed_h))
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return best_h, best_c5, n_points
