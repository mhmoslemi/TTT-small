import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Function to scale h to satisfy the constraint
    def scale_h(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (ones_count / total)
    
    # Initialize with a better perturbation
    for _ in range(100):
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturbed = initial_h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = scale_h(h_perturbed)
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_h = result.x
    
    # Refine with smaller perturbations
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(150):
            sigma = 0.05 * (1 - (_ / 150))
            h_perturbed = initial_h + np.random.normal(0, sigma, n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = scale_h(h_perturbed)
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    return best_h, best_c5, n_points
