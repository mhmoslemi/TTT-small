            h_new = h_new * (n_points / 2) / h_new_sum
        h = h_new
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        step_size *= 1.1 if current_c5 < best_c5 else 0.9
    return best_h, best_c5, n_points
