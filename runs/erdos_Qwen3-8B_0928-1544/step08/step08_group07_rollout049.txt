import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    def clip_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 1.0 / n_points)
        return h * (ones_count / total)
    
    def evolve_population(pop, pop_size, mutation_scale, elite_frac):
        fitness = np.array([objective(ind) for ind in pop])
        elite_idx = np.argsort(fitness)[:int(elite_frac * pop_size)]
        elite = [pop[i] for i in elite_idx]
        new_pop = elite.copy()
        for _ in range(pop_size - len(elite)):
            p1, p2 = np.random.choice(pop_size, 2, replace=False)
            child = (pop[p1] + np.random.rand(n_points) * (pop[p2] - pop[p1])) / 2
            child = clip_rescale(child)
            new_pop.append(child)
        return np.array(new_pop)
    
    pop_size = 50
    mutation_scale = 0.1
    elite_frac = 0.2
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        if gen < 100:
            pop = evolve_population([initial_h] * pop_size, pop_size, mutation_scale, elite_frac)
            pop = [clip_rescale(h) for h in pop]
        else:
            pop = evolve_population([best_h] * pop_size, pop_size, mutation_scale * 0.5, elite_frac)
        for h in pop:
            h = clip_rescale(h)
            result = minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        if gen % 10 == 0:
            print(f"Generation {gen}, Best C5: {best_c5}")
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = clip_rescale(h_perturbed)
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},

                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
