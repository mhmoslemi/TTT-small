import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_pop = [initial_h_values.copy() for _ in range(100)]
    else:
        initial_pop = [np.full(n_points, 0.5) for _ in range(100)]
    
    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        h = h * (ones_count / s)
        return np.clip(h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_pop[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    population = initial_pop
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            base = population[best_idx]
            vec1, vec2, vec3 = population[a], population[b], population[c]
            mutant = base + 0.8 * (vec2 - vec3)
            mutant = repair(mutant)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < 0.5:

            trial = repair(trial)
            trial_fitness = objective(trial)
            if trial_fitness < fitness[i]:
                population[i] = trial
                fitness[i] = trial_fitness
    
    # SLSQP refinement
    def slsqp_func(x):
        return objective(x)
    
    bounds = [(0.0, 1.0)] * n_points

    result = minimize(slsqp_func, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300, 'ftol': 1e-9})
