import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = target_sum / total
        return h * scale

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    population_size = 100
    population = [initial_h]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = clip_rescale(h)
        population.append(h)

    best_h = initial_h
    best_c5 = evaluate(initial_h)

    generations = 50
    for gen in range(generations):
        current = np.array([evaluate(h) for h in population])
        indices = np.argsort(current)
        sorted_pop = np.array(population)[indices]
        parents = sorted_pop[:len(sorted_pop)//2]

            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)

                if np.random.rand() < 0.1:

            offspring.append(child)
