import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:ones_count] = 1.0
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    h = h * (ones_count / h_sum) if h_sum != 0 else h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def project_to_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        return h
    
    def mutate(h, scale):
        noise = np.random.normal(0, scale, n_points)
        new_h = h + noise
        return project_to_feasible(new_h)
    
    def crossover(h1, h2, prob):
        if np.random.random() < prob:
            alpha = np.random.rand()
            return alpha * h1 + (1 - alpha) * h2
        return h1
    
    n_complexes = 5
    population = [h.copy() for _ in range(n_complexes)]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    patience = 50
    no_improvement = 0

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        evaluated = []
        for point in population:
            c5 = compute_c5(point)
            evaluated.append((c5, point))
        evaluated.sort()
        current_best = evaluated[0][0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = evaluated[0][1].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= patience:
            break
        
        new_population = []
        for _ in range(n_complexes):
            if np.random.random() < 0.5:
                new_point = mutate(best_h, 0.2)
            else:
                idx1, idx2 = np.random.choice(n_complexes, 2, replace=False)
                new_point = crossover(population[idx1], population[idx2], 0.8)
            new_population.append(new_point)
        
        population = new_population
    
    return best_h, best_c5, n_points
