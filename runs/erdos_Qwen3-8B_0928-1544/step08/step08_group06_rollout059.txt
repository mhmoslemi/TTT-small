import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    best_h = np.zeros(n_points)
    best_c5 = 0.4
    pop_size = 50
    parent_pop = []

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        initial_h[::2] = 1.0
    parent_pop.append(initial_h)
    for _ in range(pop_size - 1):
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, ones_count, replace=False)
        h[idxs] = 1.0
        parent_pop.append(h)
    parent_pop = np.array(parent_pop)

    sigma = 0.25
    decay_rate = 0.95
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate and sort
        obj = [evaluate_erdos_solution(h) for h in parent_pop]
        order = np.argsort(obj)
        parent_pop = parent_pop[order[:pop_size]]
        obj = [evaluate_erdos_solution(h) for h in parent_pop]
        best_idx = np.argmin(obj)
        if obj[best_idx] < best_c5:
            best_c5 = obj[best_idx]
            best_h = parent_pop[best_idx].copy()

        # Mutate
        new_pop = []
        for i in range(pop_size):
            h = parent_pop[i].copy()
            noise = np.random.normal(0, sigma, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (ones_count / np.sum(h))
            new_pop.append(h)
        parent_pop = np.array(new_pop)
        sigma *= decay_rate if (time.time() - start_time) % 10 < 1 else 1.0

    return best_h, best_c5, n_points
