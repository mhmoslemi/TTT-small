import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    pop_size = 30
    gen = 0
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    
    while time.time() - start_time < budget_s:
        gen += 1
        evals = [evaluate(ind) for ind in population]
        elite_idx = np.argsort(evals)[:int(len(population)*0.1)]
        elites = [population[i] for i in elite_idx]
        
        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(elite_idx, 2, replace=False)
            child = np.concatenate([population[p1][:n_points//2], population[p2][n_points//2:]])
            child = np.clip(child, 0, 1)
            child = child * (ones_count / np.sum(child))
            new_pop.append(child)
        
        for _ in range(pop_size // 2):
            h = np.random.normal(0, 0.1, n_points)
            h = population[np.random.choice(elite_idx)][:]
            h += h * np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (ones_count / np.sum(h))
            new_pop.append(h)
        
        population = new_pop
        new_evals = [evaluate(ind) for ind in population]
        for i in range(len(population)):
            if new_evals[i] < best_c5:
                best_c5 = new_evals[i]
                best_h = population[i].copy()
        
        if gen % 5 == 0:
            for h in elites:
                result = minimize(
                    fun=evaluate,
                    x0=h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
                )
                if result.success:
                    if result.fun < best_c5:
                        best_c5 = result.fun
                        best_h = result.x.copy()
    
    return best_h, best_c5, n_points
