from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    max_generations = 100

            initial_pop[i] *= ones_count / np.sum(initial_pop[i])

    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * (2.0 / n_points)
        total_corr = np.sum(corr) * (2.0 / n_points)
        mean_corr = total_corr / (2 * n_points - 1)
        dist_diff = np.sum((corr - max_corr) ** 2) * (2.0 / n_points)
        f1 = max_corr
        f2 = total_corr

    def update_population(pop):
