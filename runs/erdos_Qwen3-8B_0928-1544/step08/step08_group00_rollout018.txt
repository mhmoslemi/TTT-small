import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Start with initial_h and refine
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    final_h = result.x if result.success else initial_h
    final_c5 = result.fun if result.success else objective(initial_h)
    
    # Perform local search with perturbations
    for _ in range(10):
        if time.time() - time.time() > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        trial = final_h + perturb
        trial = np.clip(trial, 0.0, 1.0)
        trial_sum = np.sum(trial)
        if trial_sum != 0:
            trial = trial * (target_sum / trial_sum)
        trial_obj = objective(trial)
        if trial_obj < final_c5:
            final_h = trial
            final_c5 = trial_obj
    
    return final_h, final_c5, n_points
