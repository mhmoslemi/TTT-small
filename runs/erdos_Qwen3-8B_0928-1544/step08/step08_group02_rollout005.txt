import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_norm = h * (target_sum / h_sum)
        corr = np.correlate(h_norm, 1.0 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='rand',
        popsize=20,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final normalization
    best_h_sum = np.sum(best_h)
    if best_h_sum > 0:
        best_h = best_h * (target_sum / best_h_sum)

    # Final evaluation
    final_c5 = objective(best_h)
    
    return best_h, final_c5, n_points
