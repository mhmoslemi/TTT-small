        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    n_particles = 50
    particles = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        particles.append(project_h(initial_h))
        for _ in range(n_particles - 1):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, size=35, replace=False)
            h[indices] = 1.0
            particles.append(project_h(h))
    else:
        for _ in range(n_particles):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, size=35, replace=False)
            h[indices] = 1.0
            particles.append(project_h(h))
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbests = [np.copy(p) for p in particles]
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        c5_values = [compute_C5(p) for p in particles]
        for i in range(n_particles):
            if c5_values[i] < compute_C5(pbests[i]):
                pbests[i] = np.copy(particles[i])
        gbest_index = np.argmin(c5_values)
        if c5_values[gbest_index] < best_c5:
            best_c5 = c5_values[gbest_index]
            best_h = np.copy(particles[gbest_index])
        
        omega = 0.9 - (0.5 * (time.time() - start_time) / budget_s)
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * omega + 2 * r1 * (pbests[i] - particles[i]) + 2 * r2 * (best_h - particles[i])
            vel = np.clip(vel, -0.1, 0.1)
            velocities[i] = vel
            particles[i] = project_h(particles[i] + vel)
    
    return best_h, best_c5, n_points
