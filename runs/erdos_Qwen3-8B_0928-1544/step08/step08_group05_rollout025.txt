import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        h = h.copy()
        s = np.sum(h)
        if abs(s - target_sum) < 1e-6:
            return calculate_c5(h)
        scale = target_sum / s
        h = h * scale
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target_sum) < 1e-6:
            return calculate_c5(h)
        return calculate_c5(h)
    
    def calculate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    
    try:
        result = differential_evolution(
            fun=objective,
            bounds=bounds,
            strategy='best1bin',
            popsize=150,
            mutation=(0.6, 1),
            recombination=0.7,
            maxiter=100,
            tol=1e-6,
            polish=True,
            vectorized=True,
            seed=seed
        )
        best_h = result.x
        best_c5 = calculate_c5(best_h)
    except:
        best_h = initial_h
        best_c5 = calculate_c5(best_h)
    
    if best_c5 > 0.38:
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
        result_slsqp = minimize(
            fun=objective,
            x0=best_h,

            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
        )
        best_h = result_slsqp.x
        best_c5 = calculate_c5(best_h)
    
    return best_h, best_c5, n_points
