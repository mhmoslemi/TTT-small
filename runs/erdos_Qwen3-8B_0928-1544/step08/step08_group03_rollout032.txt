    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial == 0:
        initial_h = np.full(n_points, 0.5)
    else:
        initial_h = initial_h * (n_points / 2) / sum_initial

    # Create initial population
    n_complexes = 5
    population = []
    for _ in range(n_complexes):
        perturb = np.random.normal(0, 0.1, n_points)
        p = np.clip(initial_h + perturb, 0, 1)
        p = p * (n_points / 2) / np.sum(p)
        population.append(p)

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = compute_C5(best_h)
    for p in population:
        c5 = compute_C5(p)
        if c5 < best_c5:
            best_c5 = c5
            best_h = p.copy()

    start_time = time.time()
    patience = 50
    no_improvement = 0
    generations = 0

    while time.time() - start_time < budget_s:
        generations += 1
        np.random.shuffle(population)
        current_c5s = [compute_C5(p) for p in population]
        best_idx = np.argmin(current_c5s)
        if current_c5s[best_idx] < best_c5:
            best_c5 = current_c5s[best_idx]
            best_h = population[best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= patience:
            break
        # Generate new population with mutations
        new_population = []
        for p in population:
            mutation = np.random.normal(0, 0.1, n_points)
            new_h = p + mutation
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_population.append(new_h)
        population = new_population

    return best_h, best_c5, n_points
