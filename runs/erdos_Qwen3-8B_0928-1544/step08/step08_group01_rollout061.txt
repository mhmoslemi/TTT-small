    positions = np.random.rand(num_particles, n_points)
    for i in range(num_particles):
        positions[i] = positions[i] * (ones_count / np.sum(positions[i]))
    # Set first particle to initial_h
    if 'initial_h_values' in globals():
        positions[0] = initial_h
    velocities = np.random.normal(0, 0.1, (num_particles, n_points))

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = compute_c5(positions[0])
    global_best = positions[0].copy()
    personal_best = positions.copy()

    start_time = time.time()
    w, c1, c2 = 0.7, 2.0, 2.0

    while time.time() - start_time < budget_s:
        r1 = np.random.rand(num_particles)
        r2 = np.random.rand(num_particles)
        for i in range(num_particles):
            velocities[i] = w * velocities[i] + c1 * r1[i] * (personal_best[i] - positions[i]) + c2 * r2[i] * (global_best - positions[i])
            positions[i] = positions[i] + velocities[i]
            h = np.clip(positions[i], 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros_like(h)
            else:
                h = h * (ones_count / sum_h)
            positions[i] = h
            current_c5 = compute_c5(positions[i])
            if current_c5 < compute_c5(personal_best[i]):
                personal_best[i] = positions[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = positions[i].copy()

    h_values = global_best
    return h_values, best_c5, n_points
