import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:

        initial_h[::2] = 1.0

        mutant += noise
        mutant = np.clip(mutant, 0, 1)
        mutant = mutant * (ones_count / np.sum(mutant))
        population[i] = mutant
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    sigma = 0.25
    while time.time() - start_time < budget_s:
        current_c5s = [evaluate_erdos_solution(h) for h in population]
        top_indices = np.argsort(current_c5s)[:25]
        top_candidates = [population[i] for i in top_indices]
        new_population = []
        for parent in top_candidates:
            for _ in range(2):

                noise = np.random.normal(0, sigma, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (ones_count / np.sum(child))
                new_population.append(child)

        population = new_population
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        sigma *= 0.95
    return best_h, best_c5, n_points
