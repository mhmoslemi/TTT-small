import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
    
    num_particles = 50
    particles = []
    velocities = []
    for _ in range(num_particles):
        h = initial_h.copy() if 'initial_h_values' in globals() else np.zeros(n_points)
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h *= (target_sum / np.sum(h))
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    def project_h(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) == 0:
            return np.zeros_like(h_clipped)
        scaling = target_sum / np.sum(h_clipped)
        return h_clipped * scaling
    
    pbest = [p.copy() for p in particles]
    pbest_fitness = []
    best_fitness = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_particles):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        pbest_fitness.append(c5)
        if c5 < best_fitness:
            best_fitness = c5
            best_h = particles[i].copy()
    
    max_velocity = 0.1
    c1 = 2.0
    c2 = 2.0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fraction = (time.time() - start_time) / budget_s
        inertia_weight = 0.9 - 0.5 * fraction
        
        for i in range(num_particles):
            rand1 = np.random.rand(n_points)
            rand2 = np.random.rand(n_points)
            velocity = velocities[i] * inertia_weight + c1 * rand1 * (pbest[i] - particles[i]) + c2 * rand2 * (best_h - particles[i])
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            velocities[i] = velocity
            particles[i] += velocity
            particles[i] = project_h(particles[i], target_sum)
        
        for i in range(num_particles):
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = c5
                if c5 < best_fitness:
                    best_fitness = c5
                    best_h = particles[i].copy()
    
    return best_h, best_fitness, n_points
