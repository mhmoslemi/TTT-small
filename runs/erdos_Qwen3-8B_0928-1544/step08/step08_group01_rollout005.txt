import time

def run(seed=42, budget_s=1100, **kwargs):

        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.0)
        return h * (ones_count / s)
    
    initial_h = rescale(initial_h)
    
    # PSO parameters
    swarm_size = 50
    w = 0.7

    personal_best = positions.copy()
    personal_best_values = np.full(swarm_size, np.inf)
    global_best = initial_h.copy()
    global_best_value = np.inf
    
    def calculate_objective(h):

        return np.max(corr) * (2.0 / n_points)
