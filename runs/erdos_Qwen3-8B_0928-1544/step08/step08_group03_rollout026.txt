    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.full(n_points, 0.5)
        else:
            h *= required_sum / h_sum
        population.append(h)
    population = np.array(population)
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()
    generation = 0
    mutation_rate = 0.15
    while time.time() - start_time < budget_s and generation < 100:
        evaluations = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            evaluations.append(c5)
        selected_parents = []
        for _ in range(pop_size):
            indices = np.random.choice(pop_size, 3, replace=False)
            candidates = population[indices]
            selected = candidates[np.argmin(evaluations[indices])]
            selected_parents.append(selected)
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected_parents[i]
            parent2 = selected_parents[i + 1] if i + 1 < pop_size else selected_parents[0]
            split = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            offspring.append(child1)
            offspring.append(child2)
        mutated_offspring = []
        for h in offspring:
            h_mut = h + np.random.normal(0, mutation_rate, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut_sum = np.sum(h_mut)
            if h_mut_sum == 0:
                h_mut = np.full(n_points, 0.5)
            else:
                h_mut *= required_sum / h_mut_sum
            mutated_offspring.append(h_mut)
        pool = np.vstack([population, mutated_offspring])
        pool_evaluations = []
        for h in pool:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            pool_evaluations.append(c5)
        indices = np.argsort(pool_evaluations)[:pop_size]
        population = pool[indices]
        top_indices = np.argsort(pool_evaluations)[:int(len(pool) * 0.05)]
        top_individuals = pool[top_indices]
        for h in top_individuals:
            h_perturb = h.copy() + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb_sum = np.sum(h_perturb)
            if h_perturb_sum == 0:
                h_perturb = np.full(n_points, 0.5)
            else:
                h_perturb *= required_sum / h_perturb_sum
            def objective(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n_points)
            def constraint(x):
                return np.sum(x) - required_sum
            bounds = [(0.0, 1.0)] * n_points
            result = minimize(
                fun=objective,
                x0=h_perturb,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        mutation_rate *= 0.99
        generation += 1
    return best_h, best_c5, n_points
