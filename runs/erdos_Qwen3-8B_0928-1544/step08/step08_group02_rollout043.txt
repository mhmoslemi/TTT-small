import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h *= (ones_count / np.sum(initial_h))
        particles = [initial_h]
        for _ in range(19):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h *= (ones_count / np.sum(h))
            particles.append(h)
    else:
        particles = []
        for _ in range(20):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h *= (ones_count / np.sum(h))
            particles.append(h)
    velocities = [np.zeros(n_points) for _ in range(len(particles))]
    pbest = particles.copy()
    gbest = min(pbest, key=lambda h: evaluate_erdos_solution(h))
    best_c5 = evaluate_erdos_solution(gbest)
    start_time = time.time()
    omega = 0.7
    c1, c2 = 2.0, 2.0
    max_stagnation = 50
    stagnation_count = 0
    
    while time.time() - start_time < budget_s:
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (ones_count / np.sum(particles[i]))
        for i in range(len(particles)):
            current_c5 = evaluate_erdos_solution(particles[i])
            if current_c5 < evaluate_erdos_solution(pbest[i]):
                pbest[i] = particles[i].copy()

                    particles[i] *= (ones_count / np.sum(particles[i]))

    return gbest, best_c5, n_points
