import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.repeat(0.5, n_points)
    
    population_size = 40
    population = [initial_h.copy() for _ in range(population_size)]
    
    def compute_fitness(h):
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        f2 = np.sum(corr) * (2.0 / n_points)
        return (c5, f2)
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = compute_fitness(best_h)[0]
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h) for h in population]
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx][0]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        
        # Generate new population
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            h1 = parent1 + np.random.normal(0, 0.1, n_points)
            h2 = parent2 + np.random.normal(0, 0.1, n_points)
            h1 = np.clip(h1, 0.0, 1.0) * (ones_count / np.sum(h1))
            h2 = np.clip(h2, 0.0, 1.0) * (ones_count / np.sum(h2))
            new_population.extend([h1, h2])
        
        population = new_population
    
    return best_h, best_c5, n_points
