                if np.random.rand() < 0.1:
                    if np.random.rand() < 0.33:
                        child[j] = 0.0
                    elif np.random.rand() < 0.66:
                        child[j] = 1.0
                    else:
                        child[j] = np.random.rand()
            # Rescale
            child = np.clip(child, 0, 1)
            total = np.sum(child)
            if total > 0:
                child = child * (ones_count / total)
            offspring.append(child)
        
        # Replace worst half
        new_pop = parents + offspring
        new_evals = [objective(h) for h in new_pop]
        new_sorted = sorted(zip(new_pop, new_evals), key=lambda x: x[1])
        population = [h for h, _ in new_sorted[:population_size]]
        
        # Update best
        for h in population:
            val = objective(h)
            if val < best_c5:
                best_c5 = val
                best_h = h.copy()
        
        gen += 1
        if gen > 50:
            break
    
    # Local optimization on top 10
    top_10 = sorted(zip(population, [objective(h) for h in population]), key=lambda x: x[1])[:10]
    for h, _ in top_10:
        bounds = [(0.0, 1.0)] * n_points
        result = minimize(objective, h, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, method='SLSQP', options={'maxiter': 100})
        if result.success:
            val = result.fun
            if val < best_c5:
                best_c5 = val
                best_h = result.x
    
    return best_h, best_c5, n_points
