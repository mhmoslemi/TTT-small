            return h

        idx = np.argmax(h)
        if delta > 0:
            max_inc = 1 - h[idx]
            if max_inc > 0:
                inc = min(delta, max_inc)
                h[idx] += inc
                delta -= inc
                if delta > 0:
                    other_idx = np.argmin(h)
                    inc = min(delta, 1 - h[other_idx])
                    h[other_idx] += inc
        else:
            max_dec = h[idx]
            if max_dec > 0:
                dec = -min(delta, max_dec)
                h[idx] -= dec
                delta += dec
                if delta < 0:
                    other_idx = np.argmax(h)
                    dec = -min(delta, h[other_idx])
                    h[other_idx] -= dec
        return h
    
    swarm_size = 50
