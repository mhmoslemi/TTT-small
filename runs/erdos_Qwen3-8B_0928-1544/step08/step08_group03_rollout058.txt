import time

    mutation_rate = 0.15
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def scale_to_sum(h):

        return h * (ones_count / s) if s != 0 else h

                noise = np.random.normal(0, 0.05, n_points)
                population[i] = np.clip(initial_h + noise, 0, 1)
        population = [scale_to_sum(h) for h in population]
        scores = [evaluate(h) for h in population]
        elite_idx = np.argsort(scores)[:5]
