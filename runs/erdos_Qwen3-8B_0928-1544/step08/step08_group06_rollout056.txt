import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        sigma = 0.2 * (1 - (time.time() - start_time) / budget_s)
        perturbed = best_h + np.random.normal(0, sigma, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (ones_count / np.sum(perturbed))
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        )
