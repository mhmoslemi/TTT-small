import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    h = np.full(n_points, 0.5)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    h = h * ones_count / np.sum(h)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(20):  # Increased number of iterations
        if time.time() - start_time > budget_s:
            break
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h = result.x

    return best_h, best_c5, n_points
