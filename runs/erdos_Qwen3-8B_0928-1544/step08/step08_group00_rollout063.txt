from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Start from initial values
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    res = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    if res.success:
        current_h = res.x
        current_c5 = res.fun
    else:
        current_h = h
        current_c5 = objective(h)
    if current_c5 < best_c5:
        best_h = current_h
        best_c5 = current_c5
    
    # Perturb and optimize
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = initial_h + 0.1 * np.random.randn(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        res = minimize(
            fun=objective,
            x0=h,

            bounds=bounds,

            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if res.success:
            current_h = res.x
            current_c5 = res.fun
        else:
            current_h = h
            current_c5 = objective(h)
