import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
        h = h * (ones_count / np.sum(h))
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iter = int(budget_s * 0.8)  # 80% of budget for optimization
    learning_rate = 0.1
    decay = 0.995
    tolerance = 1e-5
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                grad[i] -= h[i - k] * dx
        
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if np.abs(current_sum - ones_count) > tolerance:
            delta = (ones_count - current_sum) / n_points
            h += delta
            h = np.clip(h, 0.0, 1.0)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay
    
    return best_h, best_c5, n_points
