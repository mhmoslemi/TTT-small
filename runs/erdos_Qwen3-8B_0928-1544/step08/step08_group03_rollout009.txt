        for comp in complexes:
            obj_values = [compute_c5(h) for h in comp]
            sorted_indices = np.argsort(obj_values)
            worst_idx = sorted_indices[-1]
            worst_h = comp[worst_idx]
            new_h = worst_h + np.random.normal(0, mutation_scale, n_points)
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum == 0:
                new_h = np.random.rand(n_points)
                new_h = np.clip(new_h, 0, 1)
                new_sum = np.sum(new_h)
            scale_factor = ones_count / new_sum
            new_h = new_h * scale_factor
            comp[worst_idx] = new_h
            current_c5 = compute_c5(new_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
        np.random.shuffle(complexes)
        generations += 1
    return best_h, best_c5, n_points
