import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = []
    for _ in range(5):  # Perturb initial_h with noise
        guess = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        guess += noise
        guess = np.clip(guess, 0.0, 1.0)
        guess = guess * (ones_count / np.sum(guess))
        initial_guesses.append(guess)
    for _ in range(5):  # Add random guesses
        guess = np.random.rand(n_points)
        guess = guess * (ones_count / np.sum(guess))
        initial_guesses.append(guess)
    
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            # Ensure sum constraint is met
