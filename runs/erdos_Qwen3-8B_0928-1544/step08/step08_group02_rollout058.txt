import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (ones_count / np.sum(initial_h))
    n_particles = 20
    particles = [initial_h.copy()]
    velocities = [np.zeros(n_points)]
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h))
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    pbest = particles.copy()
    gbest = min(pbest, key=lambda x: objective(x))
    best_c5 = objective(gbest)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = velocities[i] + 0.7 * (r1 * (pbest[i] - particles[i]) + r2 * (gbest - particles[i]))
            v = np.clip(v, -0.1, 0.1)
            velocities[i] = v
            particles[i] += v
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                particles[i] *= (ones_count / sum_h)
        for i in range(n_particles):
            current_c5 = objective(particles[i])
            if current_c5 < objective(pbest[i]):
                pbest[i] = particles[i].copy()
            if current_c5 < objective(gbest):
                gbest = particles[i].copy()
                best_c5 = current_c5

    best_h = gbest
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (ones_count / np.sum(best_h))
    return best_h, best_c5, n_points
