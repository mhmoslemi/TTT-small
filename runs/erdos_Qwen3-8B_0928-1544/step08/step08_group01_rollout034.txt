    def frank_wolfe(h):
        h = h.copy()
        max_iter = 200
        for _ in range(max_iter):
            if elapsed > budget_s:
                return h
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            lag = max_idx - (n_points - 1)
            gradient = np.zeros(n_points)
            for j in range(n_points):
                j_plus_k = j + lag
                if 0 <= j_plus_k < n_points:
                    term1 = 1 - h[j_plus_k]
                else:
                    term1 = 0
                j_minus_k = j - lag
                if 0 <= j_minus_k < n_points:
                    term2 = h[j_minus_k]
                else:
                    term2 = 0
                gradient[j] = dx * (term1 - term2)
            grad_sort = np.argsort(gradient)[::-1]
            s = np.zeros(n_points)
            s[:ones_count] = 1
            s[grad_sort[:ones_count]] = 1
            s[grad_sort[ones_count:]] = 0
            s = s.astype(float)

            t = 0.5

                t = 0.2

                    h = new_h
