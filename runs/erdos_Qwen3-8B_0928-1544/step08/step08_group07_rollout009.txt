```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    # Initialize population
    population_size = 50
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    # Generate initial population with diversity
    def project(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros(n_points)
        factor = (ones_count) / total
        return np.clip(h * factor, 0, 1)
    population = [project(initial_h + np.random.normal(0, 0.1, n_points)) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def eval_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    for gen in range(200):
        # Evaluate fitness
        fitness = [eval_c5(h) for h in population]
        # Select tournament winners
        new_population = []
        for _ in range(population_size):
            a, b = np.random.randint(0, population_size, 2)
            parent_a, parent_b = population[a], population[b]
            # Blend parents with random alpha
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * parent_a + (1 - alpha) * parent_b
            # Mutate
            mut = np.random.normal(0, 0.1, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child = project(child)
            new_population.append(child)
        population = new_population
        # Update best
        for h in population:
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        # Early exit if time runs out
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    # Final local optimization
    def objective(h):
        return eval_c5(h)
    def constraint_sum(h):
        return np.sum(h) - ones_count
    bounds = [(0, 1)] * n_points
    result = minimize(
        objective, best_h, method='SLSQP',
        bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return best_h, best_c5, n_points
```
