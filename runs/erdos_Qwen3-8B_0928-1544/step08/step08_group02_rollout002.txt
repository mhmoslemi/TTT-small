            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h != 0:
                h *= (ones_count / sum_h)
            particles[reset_idx] = h.copy()
            current_c5 = compute_c5(particles[reset_idx])
            if current_c5 < pbest_c5[reset_idx]:
                pbest[reset_idx] = particles[reset_idx].copy()
                pbest_c5[reset_idx] = current_c5
                if current_c5 < gbest_c5:
                    gbest = particles[reset_idx].copy()
                    gbest_c5 = current_c5
                    no_improvement = 0
    
    return gbest, gbest_c5, n_points
