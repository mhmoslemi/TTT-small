        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - current_h[i + k_max]
            if i - k_max >= 0:
                grad[i] -= current_h[i - k_max]
        current_h -= learning_rate * grad
        current_h = np.clip(current_h, 0.0, 1.0)
        current_sum = np.sum(current_h)
        delta = target_sum - current_sum
        current_h += delta / n_points
        current_h = np.clip(current_h, 0.0, 1.0)
        current_sum = np.sum(current_h)
        delta = target_sum - current_sum
        current_h += delta / n_points
        current_h = np.clip(current_h, 0.0, 1.0)
        new_c5 = compute_c5(current_h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        learning_rate *= 0.9995
    return best_h, best_c5, n_points
