import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    num_particles = 50
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    personal_best = np.full((num_particles, n_points), np.nan)
    personal_best_obj = np.inf * np.ones(num_particles)
    global_best = initial_h.copy()
    global_best_obj = float('inf')
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for i in range(num_particles):
        h = np.random.rand(n_points)
        h = h * (35.0 / np.sum(h))
        particles[i] = h
        personal_best[i] = h
        personal_best_obj[i] = objective(h)
    
    global_best_obj = np.min(personal_best_obj)
    for i in range(num_particles):
        if personal_best_obj[i] < global_best_obj:
            global_best_obj = personal_best_obj[i]
            global_best = personal_best[i].copy()
    
    start_time = time.time()
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            new_pos = particles[i] + velocities[i]
            new_pos = np.clip(new_pos, 0.0, 1.0)
            S = np.sum(new_pos)
            if S == 0:
                scaled_pos = np.zeros_like(new_pos)
            else:
                scale = 35.0 / S
                scaled_pos = new_pos * scale
                scaled_pos = np.clip(scaled_pos, 0.0, 1.0)
            current_obj = objective(scaled_pos)
            if current_obj < personal_best_obj[i]:
                personal_best[i] = scaled_pos.copy()
                personal_best_obj[i] = current_obj
            if current_obj < global_best_obj:
                global_best_obj = current_obj
                global_best = scaled_pos.copy()
            particles[i] = scaled_pos.copy()
    
    return global_best, global_best_obj, n_points
