        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[best_idx].copy()

        new_population = []
        for h in elites:
            perturbation = np.random.normal(0, 0.1 / (max_generations - generations), 70)
            mutated = h + perturbation
            mutated = np.clip(mutated, 0, 1)
            mutated = rescale(mutated)
            new_population.append(mutated)

        for i in range(0, len(elites), 2):
            if i + 1 < len(elites):
                h1 = elites[i]
                h2 = elites[i+1]
                crossed = (h1 + h2) / 2
                crossed = np.clip(crossed, 0, 1)
                crossed = rescale(crossed)
                new_population.append(crossed)

        while len(new_population) < pop_size:
            perturbation = np.random.normal(0, 0.1, 70)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = rescale(new_h)
            new_population.append(new_h)

        top_5_indices = np.argsort(fitness)[:5]
        bounds = Bounds([0.0]*70, [1.0]*70)
        for idx in top_5_indices:
            h_candidate = population[idx].copy()
            result = minimize(evaluate_c5, h_candidate, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 35}, tol=1e-8, options={'maxiter': 150, 'ftol': 1e-9})
            if result.success:
                population[idx] = result.x

        population = new_population
        generations += 1

    return best_h, best_c5, 70
