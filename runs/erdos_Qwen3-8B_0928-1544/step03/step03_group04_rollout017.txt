from scipy.optimize import minimize, Bounds

import time

    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf
    start_time = time.time()

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    for _ in range(300):  # Reduced from 500 for efficiency
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
        )

            best_c5 = res.fun
