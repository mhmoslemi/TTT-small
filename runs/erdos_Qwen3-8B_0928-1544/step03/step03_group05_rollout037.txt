from scipy.signal import correlate

    best_c5 = np.inf
    dx = 2.0 / n_points

        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for gen in range(200):
        if gen < 100:
            n_iter = 100
        else:
            n_iter = 200
        
        candidates = []
        for h in h_pop:
            res = minimize(
                compute_c5,

                bounds=bounds,

                tol=1e-9,
                options={'maxiter': n_iter, 'ftol': 1e-12, 'eps': 1e-8}
            )
            if res.success:
                h_clipped = np.clip(res.x, 0, 1)

                mutated = mutated * (n_points / 2) / np.sum(mutated)
