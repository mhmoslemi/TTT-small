        return cand
    
    pop = [create_ind(h_initial) for _ in range(POP_SIZE)]

        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            child = (p1 + p2) / 2
            child = np.clip(child, 0, 1)
            s = np.sum(child)
            diff = (n_points / 2) - s
            child[-1] += diff
            child = np.clip(child, 0, 1)
            offspring.append(child)
        
        mutated = []
        for ind in offspring:
            if np.random.rand() < MUTATION_RATE:
                noise = np.random.normal(0, 0.05, n_points)
                mut = ind + noise
                mut = np.clip(mut, 0, 1)
                s = np.sum(mut)
                diff = (n_points / 2) - s
                mut[-1] += diff
                mut = np.clip(mut, 0, 1)
                mutated.append(mut)

                mutated.append(ind)
