            h = h * (n_points / 2) / total
    # Final clipping and normalization
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / total
    return h, best_c5, n_points
