```python
def initialize_structured(n_points=70, ones_count=35):
    h = np.zeros(n_points)
    interval = n_points / ones_count
    for i in range(ones_count):
        pos = round(i * interval)
        h[pos] = 1.0
    # Ensure sum is exactly ones_count by adjusting nearby points if necessary
    if np.sum(h) != ones_count:
        diff = ones_count - np.sum(h)
        # Distribute the difference to nearest zeros
        zeros = np.where(h == 0)[0]
        h[zeros[:diff]] = 1.0
    return h
```

```python
def mutate_h(h, generation, max_generations=100, sigma=0.1):
```

```python
def evaluate(h):
    n = len(h)
    max_overlap = 0.0
    for k in range(n):
        if k > n // 2:
            break
        cor = 0.0
        for i in range(n - k):
            cor += h[i] * (1 - h[i + k])
        cor *= 2.0 / n
        if cor > max_overlap:
            max_overlap = cor
    return max_overlap
```

```python
    initial_h = initialize_structured(n_points, ones_count) if 'initial_h_values' not in globals() else initial_h_values

    population_size = 20
    mutation_rate = 0.1
    generations = budget_s // 5  # Estimate based on 5s per generation

    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        mutant = mutate_h(initial_h, generation=0, max_generations=generations)
        population.append(mutant)

    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)

    for generation in range(generations):
        # Evaluate population
        c5_scores = [evaluate(h) for h in population]
        current_best_c5 = min(c5_scores)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[np.argmin(c5_scores)]

        # Selection
        sorted_indices = np.argsort(c5_scores)
        survivors = [population[i] for i in sorted_indices[:5]]  # Keep top 5

        # Mutation and Crossover
        new_population = []
        for h in survivors:
            for _ in range(4):  # Create 4 mutants per survivor
                mutant = mutate_h(h, generation, generations)
                new_population.append(mutant)
            # Perform crossover with another survivor
            partner = np.random.choice(survivors)
            offspring = crossover(h, partner)
            new_population.append(offspring)

        # SQP Refinement for top candidates
        top_indices = sorted_indices[:5]
        for idx in top_indices:
            h_top = population[idx]
            result = minimize(
                lambda x: evaluate(x),
                h_top,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count},
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

        # Maintain population size
        population = new_population[:population_size]

    return best_h, best_c5, n_points
```
