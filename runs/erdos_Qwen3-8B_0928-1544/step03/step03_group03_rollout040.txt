import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    initial_guess = np.full(n, 0.5) if 'initial_h_values' not in globals() else initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0.0, 1.0)
    initial_guess = initial_guess * (n / 2) / np.sum(initial_guess)
    
    population_size = 50
    num_generations = 100
    mutation_std = 0.03
    tournament_size = 5
    
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr * dx)
    
    def normalize(h):
        return h * (n / 2) / np.sum(h)
    
    population = [normalize(np.clip(initial_guess + np.random.normal(0, 0.05, n), 0, 1)) for _ in range(population_size - 1)]
    population.append(initial_guess)
    
    start_time = time.time()
    best_h = initial_guess
    best_c5 = -fitness(best_h)
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness_vals = [fitness(ind) for ind in population]
        
        # Tournament selection
        parents = []
        for _ in range(population_size // 2):
            contenders = np.random.choice(population_size, tournament_size, replace=False)
            winner = contenders[np.argmax(fitness_vals[contenders])]
            parents.append(population[winner])
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            cross_point = np.random.randint(1, n-1)
            child1 = np.concatenate([p1[:cross_point], p2[cross_point:]])
            child2 = np.concatenate([p2[:cross_point], p1[cross_point:]])
            offspring.append(normalize(np.clip(child1, 0, 1)))
            offspring.append(normalize(np.clip(child2, 0, 1)))
        
        # Mutation
        for ind in offspring:
            ind += np.random.normal(0, mutation_std, n)
            ind = np.clip(ind, 0, 1)
            ind = normalize(ind)
        
        # Replace population
        population = parents + offspring
        population = np.array([normalize(np.clip(ind, 0, 1)) for ind in population])
        population = np.vstack([population, np.array([initial_guess])]).astype(float)
        population = population[:population_size]
        
        # Update best
        for ind in population:
            current_c5 = -fitness(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind
    
    return best_h, best_c5, n
