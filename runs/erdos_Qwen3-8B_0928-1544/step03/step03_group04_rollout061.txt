        max_overlap = np.max(corr) * dx

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    h_flat = best_h.flatten()
    res = minimize(
        objective,
        h_flat,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
    )
    
    if res.success:
        best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total != 0:
        best_h = best_h * (n_points / 2) / total
    
    return best_h, best_c5, n_points
