def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    # Assume initial_h_values is provided
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum(h, target):
        s = np.sum(h)
        diff = target - s
        h[-1] += diff
        h = np.clip(h, 0, 1)
        return h

    # Initialize population
    pop_size = 50
    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        perturbed = h_initial + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = adjust_sum(perturbed, n_points / 2)
        population.append(perturbed)

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed = 0.0
    start = time.time()

    # Genetic algorithm loop
    gen = 0
    while elapsed < budget_s:
        if gen > 1000:
            break
        # Evaluate and select
        fitness = [compute_C5(ind) for ind in population]
        sel = np.argsort(fitness)[:pop_size // 2]
        selected = [population[i] for i in sel]

        # Crossover and mutation
        new_pop = []
        for i in range(pop_size // 2):
            p1 = selected[i]
            p2 = selected[(i+1)%len(selected)]
            child = (p1 + p2) / 2
            child = adjust_sum(child, n_points / 2)
            new_pop.append(child)
            mut = child + np.random.normal(0, 0.01, n_points)
            mut = np.clip(mut, 0, 1)
            mut = adjust_sum(mut, n_points / 2)
            new_pop.append(mut)
        population = new_pop

        # Simulated annealing
        temp = 1.0
        for _ in range(10):
            pid = np.argmin(fitness)
            best = population[pid]
            perturbed = best + np.random.normal(0, 0.01, n_points)
