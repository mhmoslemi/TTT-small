    n = 70
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = float('inf')
    num_perturbations = 10
    learning_rate = 0.1
    max_steps = 1000

    for _ in range(num_perturbations):
        if time.time() - start_time >= budget_s:
            break
        # Perturb and restart
        noise = np.random.normal(0, 0.05, n)
        h = np.clip(h + noise, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (35.0 / sum_h)
        # Optimization loop
        for _ in range(max_steps):
            if time.time() - start_time >= budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full') * dx
            max_corr = np.max(corr)
            if max_corr < best_c5:
                best_c5 = max_corr
                best_h = h.copy()
            k_max = np.argmax(corr)
            j1 = (np.arange(n) + k_max) % n
            j2 = (np.arange(n) - k_max) % n
            grad = (1 - h[j1]) - h[j2]
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * (35.0 / sum_h)
    return best_h, best_c5, n
