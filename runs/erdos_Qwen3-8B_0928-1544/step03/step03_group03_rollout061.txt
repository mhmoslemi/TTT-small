import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n

    # Initialize h with initial_h_values if available

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    iterations = 5000
