            c_minus = compute_c5(h_minus)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        h = rescale_h(h - learning_rate * grad)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break

    return best_h, best_c5, n_points
