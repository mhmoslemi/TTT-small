import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Predefined grid sizes
    n_coarse = 30
    n_fine = 70
    max_iter = 100
    
    # Initial guess
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_coarse] if len(h_initial) > n_coarse else h_initial
    else:
        h_initial = np.random.rand(n_coarse)
        h_initial = h_initial / np.sum(h_initial) * (n_coarse / 2)
    
    def optimize_grid(h, n_points):
        dx = 2.0 / n_points
        def obj(x):
            h_flat = x.reshape(n_points)
            corr = correlate(h_flat, 1.0 - h_flat, mode='full')
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def cons(x):
            return np.sum(x) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'maxiter': max_iter, 'ftol': 1e-9})
        return res.x, res.fun
    
    # Coarse optimization
    h_coarse, best_c5 = optimize_grid(h_initial, n_coarse)
    
    # Upsample to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    
    # Fine optimization
    h_fine, c5_fine = optimize_grid(h_fine, n_fine)
    best_c5 = min(best_c5, c5_fine)
    
    # Final check and return
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return h_fine, best_c5, n_fine
    
    # Ensure constraint satisfaction
    final_sum = np.sum(h_fine)
    if abs(final_sum - n_fine / 2) > 1e-6:
        h_fine = h_fine / final_sum * (n_fine / 2)
    
    return h_fine, best_c5, n_fine
