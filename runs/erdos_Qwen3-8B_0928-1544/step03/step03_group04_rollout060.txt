from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 500
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5

        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def eval_fitness(h):

        return np.max(corr) * dx

    for gen in range(max_generations):
        fitness = [eval_fitness(h) for h in population]
