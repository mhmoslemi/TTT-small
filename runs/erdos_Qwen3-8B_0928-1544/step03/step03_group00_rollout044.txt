import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)

        if np.sum(h_initial) != ones_count:
            diff = ones_count - int(np.sum(h_initial))
            indices = np.where(h_initial == 0)[0]
            h_initial[indices[:diff]] = 1.0
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h_initial.copy()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0] * n_points, [1.0] * n_points)

    for _ in range(100):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_initial + perturbation

        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
