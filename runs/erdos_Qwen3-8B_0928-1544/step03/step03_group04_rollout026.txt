import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 70
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    max_tries = 30
    
    def objective(x):
        h = x.copy()
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(max_tries):
        # Perturbate
        delta = np.random.normal(0, 0.03, size=n_points)
        h_perturbed = h + delta
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Optimize
        res = minimize(
            objective,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-8}
        )
        
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:

                best_h = res.x

    final_h = best_h.copy()

    final_sum = np.sum(final_h)
