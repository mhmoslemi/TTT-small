        corr_sub = correlate(best_h, 1 - best_h, mode='full')[shift_range]
        max_sub = np.max(corr_sub * dx)
        if max_sub < best_c5:
            best_c5 = max_sub
            best_h = best_h.copy()
        
        # Local optimization with L-BFGS-B for better convergence
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={
                'maxiter': 100,
                'ftol': 1e-12,
                'disp': False
            }
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        
        # Early exit if sufficient improvement is found
        if best_c5 < 0.38088 - 1e-6:
            break
    
    # Final normalization and constraint check
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
