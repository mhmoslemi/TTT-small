import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.ndimage import convolve1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values
    else:
        h = np.zeros(n_points)
        for i in range(ones_count):
            pos = i * (n_points // ones_count)
            h[pos] = 1.0
        # Ensure exact sum
        while np.sum(h) != ones_count:
            idx = np.where(h == 0)[0][0]
            h[idx] = 1.0
            if np.sum(h) > ones_count:
                idx = np.where(h == 1)[0][0]
                h[idx] = 0.0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    max_iter = int(budget_s / 20)

    def objective(h):
        # Precompute overlap for all k
        x = h
        y = 1 - x
        corr = convolve1d(x, y, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - ones_count

    bounds = Bounds([0.0] * n_points, [1.0] * n_points)

    for _ in range(max_iter):
        # Add perturbation
        perturb = np.random.normal(0, 0.15, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed *= ones_count / np.sum(h_perturbed)

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-10,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
