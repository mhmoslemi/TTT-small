import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 70
    dx = 2.0 / n

    # Initialize based on initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n, 0.5)

    population_size = 50
    population = []
    for _ in range(population_size):
        perturb = np.random.normal(0, 0.05, n)
        candidate = initial_guess + perturb
        candidate = np.clip(candidate, 0, 1)
        total = np.sum(candidate)
        candidate = candidate * (n / 2) / total
        population.append(candidate)
    population = np.array(population)

    best_h = initial_guess.copy()
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    max_generations = 100
    for _ in range(max_generations):
        # Fitness evaluation
        fitness = np.array([np.correlate(ind, 1 - ind, mode='full').max() * dx for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        # Parent selection via tournament
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, 5, replace=False)
            best_tournament = tournament[np.argmin(fitness[tournament])]
            parents.append(population[best_tournament])
        parents = np.array(parents)

        # Crossover and mutation
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[0]
            child1 = np.concatenate([p1[:n//2], p2[n//2:]])
            child2 = np.concatenate([p2[:n//2], p1[n//2:]])
            child1 = child1 * (n/2) / np.sum(child1)
            child2 = child2 * (n/2) / np.sum(child2)
            offspring.append(child1)
            offspring.append(child2)

        # Mutation
        mutated = []
        for child in offspring:
            perturb = np.random.normal(0, 0.03, n)
            mc = child + perturb
            mc = np.clip(mc, 0, 1)
            mc = mc * (n/2) / np.sum(mc)
            mutated.append(mc)

        # Population replacement
        sorted_indices = np.argsort(fitness)
        new_pop = population[sorted_indices[:int(0.6 * population_size)]]
        new_pop = np.vstack([new_pop, np.array(mutated)])
        np.random.shuffle(new_pop)
        population = new_pop[:population_size]

        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n
