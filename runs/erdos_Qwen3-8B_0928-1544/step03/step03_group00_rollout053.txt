    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    # Precompute all k for correlation

        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
