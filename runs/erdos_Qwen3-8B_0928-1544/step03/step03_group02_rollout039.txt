    from scipy.optimize import minimize
    import time

    dx = 2.0 / n_points

    # Objective function to minimize
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    # Optimization parameters
    num_restarts = 30  # Increase restarts for broader exploration
    max_iter_per_restart = 100
    np.random.seed(seed)

    # Perturbation and optimization settings
    perturbation_scale = 0.1

    for _ in range(num_restarts):
        # Perturb the initial guess slightly
        perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Adjust to meet sum constraint
        sum_perturbed = np.sum(perturbed)
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_perturbed
            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0.0, 1.0)

        # Optimize from perturbed guess
        result = minimize(
            compute_C5,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'disp': False}

                best_h = current_h

                print(f"New best C5: {best_c5}")
