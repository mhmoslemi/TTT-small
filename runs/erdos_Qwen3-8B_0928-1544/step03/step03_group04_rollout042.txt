import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 70
        h = np.zeros(n)
        h[:n//2] = 1.0
        h = h / np.sum(h) * (n / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for _ in range(150):
        delta = np.random.randn(n_points) * 0.3
        perturbed = np.clip(h + delta, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed == 0:
            perturbed = np.random.rand(n_points)
        else:
            scale = (n_points / 2) / sum_perturbed
            perturbed *= scale
        
        res = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            tol=1e-7,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        
        if best_c5 < 0.38085:
            break
        
        h = best_h
    
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
