    best_c5 = np.inf
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        max_overlap = 0.0

            shifted = np.roll(h, k)
            overlap = np.dot(h, 1.0 - shifted)
            max_overlap = max(max_overlap, overlap * dx)
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(500):
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
