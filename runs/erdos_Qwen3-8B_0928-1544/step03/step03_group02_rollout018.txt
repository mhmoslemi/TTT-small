import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def adjust_h(h):
        h = np.clip(h, 0.0, 1.0)
        diff = (n_points / 2) - np.sum(h)
        h[-1] += diff
        return h

    def mutate(h, mutation_rate=0.05):
        mask = np.random.rand(n_points) < mutation_rate
        h = h.copy()
        h[mask] += np.random.normal(0, 0.05, sum(mask))
        return adjust_h(h)

    def crossover(parent1, parent2):
        child = (parent1 + parent2) / 2
        return adjust_h(child)

    pop_size = 50
    temp = 1.0
    cooling_rate = 0.995
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    time_start = time.time()

    pop = [adjust_h(h_initial + np.random.normal(0, 0.1, n_points)) for _ in range(pop_size)]
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        curr_pop = []
        for h in pop:
            curr_pop.append(h)
        curr_pop.sort(key=compute_C5)
        parents = curr_pop[:int(pop_size / 2)]
        children = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                child1 = crossover(parents[i], parents[i+1])
                children.append(child1)
                child2 = mutate(parents[i])
                children.append(child2)
            else:
                children.append(mutate(parents[i]))
        children.sort(key=compute_C5)
        new_pop = parents[:int(pop_size / 2)] + children[:int(pop_size / 2)]
        pop = new_pop
        curr_pop = [adjust_h(h) for h in pop]
        curr_pop.sort(key=compute_C5)
        curr_c5 = compute_C5(curr_pop[0])
        if curr_c5 < best_c5:
            best_h = curr_pop[0].copy()
            best_c5 = curr_c5
        temp *= cooling_rate
        if _ % 10 == 0:
            print(f"Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
