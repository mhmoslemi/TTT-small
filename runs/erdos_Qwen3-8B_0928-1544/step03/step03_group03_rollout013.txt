        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = adjust_sum(h, n / 2)

    best_h = h
    best_c5 = compute_c5(best_h, dx)

    # Simulated Annealing Parameters
    temp_start = 1.0
    temp_end = 1e-3
    cooling_rate = 0.99
    num_iterations = 5000
    sigma = 0.05

    current_h = best_h.copy()
    current_c5 = best_c5
    temp = temp_start

    for _ in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        delta = np.random.normal(0, sigma, n)
        h_new = current_h + delta
        h_clipped = np.clip(h_new, 0, 1)
        h_clipped = adjust_sum(h_clipped, n / 2)
        new_c5 = compute_c5(h_clipped, dx)
        
        if new_c5 < current_c5:
            current_h = h_clipped
            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() < prob:
                current_h = h_clipped
                current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        temp *= cooling_rate

    # Local Optimization
    def objective(h):
        return compute_c5(h, dx)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-10,
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return best_h, best_c5, n

def adjust_sum(h, target_sum):
    current_sum = h.sum()
    if np.isclose(current_sum, target_sum):
        return h
    delta_needed = target_sum - current_sum
    if delta_needed == 0:
        return h
    indices = np.where(h < 1)[0]
    if not indices.size:
        return h
    amount_per = delta_needed / len(indices)
    for i in indices:
        h[i] += amount_per
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
