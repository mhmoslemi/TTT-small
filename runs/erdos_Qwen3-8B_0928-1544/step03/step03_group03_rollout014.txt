import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()

    best_h = initial_guess

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Multiple restarts with perturbations
    num_restarts = 50
    for _ in range(num_restarts):

        perturbed_guess = np.clip(initial_guess + perturb, 0.0, 1.0)
        
        # Use different solvers for diversification
        method = 'L-BFGS-B' if _ % 3 == 0 else 'SLSQP' if _ % 3 == 1 else 'COBYLA'
        
        result = minimize(
            fun=objective,

            tol=1e-10,
