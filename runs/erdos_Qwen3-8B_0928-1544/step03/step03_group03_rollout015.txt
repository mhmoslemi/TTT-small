from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):

    best_h = initial_guess
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)

    num_restarts = 30

        result = minimize(

            method='SLSQP',
            bounds=bounds,
