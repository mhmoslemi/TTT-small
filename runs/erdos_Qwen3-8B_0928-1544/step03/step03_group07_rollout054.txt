import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 160
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
    best_h = h
    best_c5 = np.inf
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(x):
        h_flat = x.reshape(n_points)
        corr = correlate(h_flat, 1 - h_flat, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for iteration in range(150):
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_overlap = np.max(corr * dx)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = best_h.copy()
        if best_c5 < 0.38085 - 1e-6:
            break
        
        top_shifts = np.argsort(corr * dx)[-10:]
        top_shifts = np.unique(top_shifts)
        worst_shifts = top_shifts[np.argsort(-corr[top_shifts] * dx)]
        
        for k in worst_shifts[:5]:
            h_k = best_h.copy()
            h_k[k] *= 0.9
            h_k[k+1] *= 1.1
            h_k[k-1] *= 1.1
            h_k = np.clip(h_k, 0, 1)
            res = minimize(objective, h_k, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-8)
            if res.success:
                if res.fun < best_c5:
                    best_c5 = res.fun
                    best_h = res.x
        
        if best_c5 < 0.38085:
            break
    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    return final_h, best_c5, n_points
