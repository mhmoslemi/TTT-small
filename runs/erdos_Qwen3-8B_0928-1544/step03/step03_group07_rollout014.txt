import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Initialize with provided initial_h_values or a random feasible starting point
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 100
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    elapsed = 0
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(500):
        # Compute current C5
        corr = correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        if best_c5 < 0.38085:
            break
        
        # Find worst shift
        shift_idx = np.argmax(corr * dx)
        k_max = shift_idx - (n_points - 1)
        
        # Refine around k_max
        mid = k_max
        if 0 <= mid < 2 * n_points:
            # Focus on the worst shift, optimize to reduce overlap
            res = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                tol=1e-8,
                options={
                    'maxiter': 30,
                    'ftol': 1e-10,
                    'disp': False
                }
            )
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        
        # Early exit if budget is nearly exhausted
        elapsed += 100
        if elapsed > budget_s:
            break
    
    # Final adjustments
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-5:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
