import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    
    population_size = 20
    pop = [h_initial.copy()]
    for _ in range(population_size - 1):
        m = h_initial + np.random.normal(0, 0.05, 70)
        m = np.clip(m, 0, 1)
        m = m * (35.0 / np.sum(m))
        pop.append(m)
    
    best_h = h_initial
    best_c5 = float('inf')
    gen = 0
    max_gen = 200
    start_time = time.time()
    
    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / 70)
    
    while gen < max_gen and time.time() - start_time < budget_s:
        scores = [calc_c5(h) for h in pop]
        idx = np.argmin(scores)
        if scores[idx] < best_c5:
            best_c5 = scores[idx]
            best_h = pop[idx]
        
        top_idx = np.argsort(scores)[:len(pop)//4]
        survivors = [pop[i] for i in top_idx]
        
        new_pop = []
        for h in survivors:
            for _ in range(4):
                m = h + np.random.normal(0, max(0.05 * (1 - gen/max_gen), 0.01), 70)
                m = np.clip(m, 0, 1)
                m = m * (35.0 / np.sum(m))
                new_pop.append(m)
        
        for _ in range(2):
            a, b = np.random.choice(survivors, 2, replace=False)
            c = (a + b) / 2
            c = np.clip(c, 0, 1)
            c = c * (35.0 / np.sum(c))
            new_pop.append(c)
        
        pop = new_pop[:population_size]
        gen += 1
    
    return best_h, best_c5, 70
