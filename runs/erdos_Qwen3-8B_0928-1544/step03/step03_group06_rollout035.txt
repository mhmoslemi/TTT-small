from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values

        h = np.random.rand(70)
        h = h / np.sum(h) * 70 / 2
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def rescale(h):
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1
        return np.clip(h * scale, 0, 1)
    
    for _ in range(300):
        h = rescale(h)
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
        h = h + np.random.normal(0, 0.03, n_points)
        h = rescale(h)
    
    best_h = rescale(best_h)
    return best_h, best_c5, n_points
