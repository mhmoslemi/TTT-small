from scipy.optimize import minimize, Bounds
from scipy.stats import truncnorm

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:

    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    
    def obj_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - 35
    
    bounds = Bounds([0.0]*70, [1.0]*70)
    
    max_iter = int(budget_s / 10)
    for _ in range(max_iter):
        np.random.seed(seed + _)  # Ensure reproducibility
        perturbation = truncnorm.rvs(-3, 3, loc=0, scale=0.07, size=70)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        
        result = minimize(
            fun=obj_func,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 150, 'ftol': 1e-9}
        )

            if current_c5 < best_c5:

                best_h = result.x
            h_initial = result.x
