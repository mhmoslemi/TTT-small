        h_initial = h_initial / np.sum(h_initial) * 70 / 2
        n_points = 70
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * (2.0 / n_points)
        return max_overlap
    
    def rescale_h(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        scale = (n_points / 2) / total if total != 0 else 1
        return h_clipped * scale
    
    best_h = h_initial.copy()
    best_c5 = np.inf
    max_iter = 500
    pop_size = 100
    mutation_scale = 0.1
    elite_ratio = 0.2
    crossover_rate = 0.8
    
    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, mutation_scale, n_points)
        candidate = h_initial + perturb
        candidate = rescale_h(candidate)
        population.append(candidate)
    
    for _ in range(max_iter):
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        if best_c5 < 0.38085:
            break
        
        elite_size = int(pop_size * elite_ratio)
        selected_indices = np.argsort(fitness)[:elite_size]
        selected = [population[i] for i in selected_indices]
        
        offspring = []
        for _ in range(pop_size - elite_size):
