import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    initial_h_values = None  # Assume this is defined externally

    # Initialize population
    population = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n:

            population.append(perturbed)
    else:
        for _ in range(50):
            h = np.random.rand(n)
            h = h * (n/2) / np.sum(h)
            population.append(h)

    def evaluate_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * dx

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()

    for gen in range(100):

        # Evaluate fitness
        fitness = []
        for h in population:

            fitness.append(-c5)

            tournament = np.random.choice(len(population), 5, replace=False)

            selected.append(population[best_idx])

        # Crossover
        offspring = []
        for i in range(0, len(selected), 2):
            p1, p2 = selected[i], selected[i+1] if i+1 < len(selected) else selected[i]

            offspring.append(child1)
            offspring.append(child2)

        # Mutation
        mutated = []
        for child in offspring:
            noise = np.random.normal(0, 0.03, n)
            mut = child + noise
            mut = np.clip(mut, 0, 1)

            mutated.append(mut)

        # Replacement
        sorted_indices = np.argsort(fitness)

        new_population = new_population[:50]
        population = new_population

        # Update best
        for h in population:
            c5 = evaluate_c5(h)

                best_c5 = c5
                best_h = h

    return best_h, best_c5, n
