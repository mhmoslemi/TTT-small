import time

from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    n_pop = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 200
    best_h = None
    best_c5 = np.inf

            initial_population.append(h)
    else:
        initial_population = [np.random.rand(n_points) for _ in range(n_pop)]

            initial_population[i] = initial_population[i] / np.sum(initial_population[i]) * target_sum

    def fitness(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
