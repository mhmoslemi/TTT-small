        # Update h
        h_new = h_initial - learning_rate * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        h_sum = np.sum(h_new)
        if h_sum != 0:

        # Compute current C5
        current_c5 = current_corr * dx

        if iteration % 50 == 0:
