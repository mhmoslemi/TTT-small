from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):

        h_initial *= scale
    else:
        h_initial = np.random.rand(70)
        h_initial = h_initial / np.sum(h_initial) * 70 / 2
        n_points = 70

    population_size = 80
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.03, n_points)
        candidate = h_initial + perturb
        candidate = np.clip(candidate, 0, 1)
        total = np.sum(candidate)
        scale = (n_points / 2) / total if total != 0 else 1
        candidate *= scale

    best_h = h_initial.copy()

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * (2.0 / n_points) if len(corr) > 0 else 0
        return max_overlap

    def rescale_h(h):

        total = np.sum(h_clipped)
        scale = (n_points / 2) / total if total != 0 else 1

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        if best_c5 < 0.38085:
            break
        selected = [population[i] for i in np.argsort(fitness)[:population_size // 2]]
        offspring = []
