from numpy import roll, argmax, sum, arange, mod

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    n_points = 70
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.1
    decay_rate = 0.995
    max_iter = 2000
    for iteration in range(max_iter):
        overlaps = []
        for k in range(n_points):
            shifted = roll(h, shift=k)
            overlap = sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        current_max = max(overlaps)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        k_max = argmax(overlaps)
        indices_plus = (arange(n_points) + k_max) % n_points
        indices_minus = (arange(n_points) - k_max) % n_points
        grad = (1 - h[indices_plus]) - h[indices_minus]
        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h * (35.0 / sum(h))
        learning_rate *= decay_rate
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            h = h * (35.0 / sum(h))
    return best_h, best_c5, n_points
