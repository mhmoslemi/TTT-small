    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(

            method='SLSQP',
            bounds=bounds,

            tol=1e-8,
