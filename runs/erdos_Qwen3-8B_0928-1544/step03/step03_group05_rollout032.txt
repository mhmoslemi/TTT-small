from scipy.optimize import minimize
from scipy.signal import correlate

    def objective(h_flat):

        corr = correlate(h, 1 - h, mode='full')

        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
