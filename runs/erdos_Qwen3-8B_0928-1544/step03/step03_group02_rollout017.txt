        initial_h_values = globals()['initial_h_values']
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum(arr):
        sum_total = np.sum(arr)
        diff = target_sum - sum_total
        arr[-1] = np.clip(arr[-1] + diff, 0, 1)
        arr[-1] = target_sum - np.sum(arr[:-1])

    population = [initial_h_values.copy() for _ in range(50)]
    for i in range(50):
        perturb = np.random.normal(0, 0.05, n_points)
        population[i] += perturb
        population[i] = np.clip(population[i], 0, 1)
        adjust_sum(population[i])

    best_h = population[0]
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_C5(h) for h in population]
        parent_indices = np.argsort(fitness)[:25]
        parents = [population[i] for i in parent_indices]
        offspring = []
        for _ in range(50):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.1:
                perturb = np.random.normal(0, 0.05, n_points)
                child += perturb
                child = np.clip(child, 0, 1)
                adjust_sum(child)
            offspring.append(child)
        population = offspring
        curr_best = min(population, key=compute_C5)
        if compute_C5(curr_best) < best_c5:
            best_c5 = compute_C5(curr_best)
            best_h = curr_best

    return best_h, best_c5, n_points
