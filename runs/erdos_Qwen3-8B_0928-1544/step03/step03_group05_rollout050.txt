import numpy as np
from scipy.optimize import minimize, basinhopping
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 30
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf

    def objective_coarse(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    res = minimize(objective_coarse, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-9, options={'maxiter': 100})
    best_h = res.x
    best_c5 = res.fun
    elapsed = time.time() - start_time

    if elapsed > budget_s:
        return best_h, best_c5, n_points

    n_fine = 70
    x_coarse = np.linspace(0, 2, n_points)
    interp_func = interp1d(x_coarse, best_h, kind='linear', fill_value='extrapolate')
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = interp_func(x_fine)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)

    def objective_fine(h_flat):
        h = h_flat.reshape(n_fine)
        corr = correlate(h, 1 - h, mode='full')
        dx_fine = 2.0 / n_fine
        max_overlap = np.max(corr) * dx_fine
        sum_h = np.sum(h)
        penalty = 1e3 * max(0, abs(sum_h - n_fine / 2))
