        h = h / np.sum(h) * n / 2

    dx = 2.0 / 70
    n = 70

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)

        sum_h = np.sum(h_clip)
        if sum_h == 0:

        scale = (n / 2) / sum_h
        return h_clip * scale

    best_c5 = float('inf')

    current_h = project_h(initial_h)
