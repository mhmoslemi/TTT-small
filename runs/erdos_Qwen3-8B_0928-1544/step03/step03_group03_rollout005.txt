        h_perturbed = current_h + delta
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum == 0:
            h_perturbed = np.zeros(n)
        else:
            scale = (n / 2) / h_perturbed_sum
            h_perturbed *= scale

        new_energy = compute_c5(h_perturbed)

        if new_energy < current_energy:
            current_h = h_perturbed.copy()
            current_energy = new_energy
        else:
            prob = np.exp(-(new_energy - current_energy) / T)
            if np.random.rand() < prob:
                current_h = h_perturbed.copy()
                current_energy = new_energy

        if current_energy < best_c5:
            best_c5 = current_energy

        T *= cooling_rate
        iteration_count += 1

    # Ensure final h has correct sum
    h_sum = np.sum(current_h)
    if h_sum != n / 2:
        scale = (n / 2) / h_sum
        current_h *= scale

    return current_h, best_c5, n
