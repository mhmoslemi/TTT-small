import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    np.random.seed(seed)
    num_restarts = 50
    max_iter_per_restart = 200
    perturbation_scale = 0.05

    for _ in range(num_restarts):
        perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_perturbed
            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0.0, 1.0)

        result = minimize(
            compute_C5,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'eps': 1e-8}
        )

        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
            print(f"New best C5: {best_c5}")

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
