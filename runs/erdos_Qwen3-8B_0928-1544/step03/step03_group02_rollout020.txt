import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution
    import time
    import sys
    sys.setrecursionlimit(10000)

    # Assume initial_h_values is defined globally
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-6:
            diff = target_sum - current_sum
            h[-1] = np.clip(h[-1] + diff, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess from initial_h_values
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    best_c5 = min(best_c5, compute_C5(best_h))

    # Optimization parameters
    max_time = 0.9 * budget_s
    start_time = time.time()

    # Run differential_evolution
    result = differential_evolution(
        compute_C5,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        maxiter=1000,
        seed=seed,
        polish=True,
        dispatch=(None, None),
        disp=False
    )

    best_h = result.x
    best_c5 = compute_C5(best_h)
    best_c5 = min(best_c5, compute_C5(best_h))

    # Early termination if time expired
    elapsed_time = time.time() - start_time
    if elapsed_time > max_time:
        print("Time expired, returning best known solution")
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
