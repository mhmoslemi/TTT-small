import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(corr) * dx
        penalty = 1e4 * (np.sum(h) - n_points / 2) ** 2
        return max_overlap_val + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True
    )
    
    best_h = result.x
    best_c5 = result.fun - 1e4 * (np.sum(best_h) - n_points / 2) ** 2
    
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2) > 1e-6:
        best_h = best_h / final_sum * (n_points / 2)
    
    return best_h, best_c5, n_points
