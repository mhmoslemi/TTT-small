import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h):
        return np.sum(h) - 35
    
    bounds = Bounds([0.0]*70, [1.0]*70)
    
    for _ in range(150):  # Increased number of iterations
        perturbation = np.random.normal(0, 0.05, 70)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h_initial = result.x
