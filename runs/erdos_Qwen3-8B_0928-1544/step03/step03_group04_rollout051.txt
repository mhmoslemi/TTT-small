    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(trials):
        pert = np.random.normal(0, 0.25, n_points)
        h_pert = np.clip(h + pert, 0, 1)
        h_pert = h_pert / np.sum(h_pert) * (n_points / 2)
        
        res = minimize(
            objective,
            h_pert,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
