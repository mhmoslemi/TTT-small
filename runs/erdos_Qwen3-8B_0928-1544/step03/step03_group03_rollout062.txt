import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    import random
    random.seed(seed)
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n, 0.5)
    
    best_h = initial_guess
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    
    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx
    
    # Constraint: sum(h) == n/2
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Multiple restarts with perturbations
    num_restarts = 40
    for _ in range(num_restarts):
        perturb = np.random.normal(0, 0.07, n)
        perturbed_guess = np.clip(initial_guess + perturb, 0.0, 1.0)
        
        result = minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(perturbed_guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_guess
        
        # Respect time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n
