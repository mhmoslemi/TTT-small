from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    dx = 2.0 / n_points

    def initial_guess():
        if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
            return initial_h_values.copy()
        h = np.zeros(n_points)
        pos = np.round(np.linspace(0, n_points-1, ones_count)).astype(int)
        h[pos] = 1.0
        if np.sum(h) != ones_count:
            diff = ones_count - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
        return h

    h_initial = initial_guess()
    best_h = h_initial
    best_c5 = float('inf')

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    for _ in range(100):
        perturbation = np.random.normal(0, 0.08, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 150, 'ftol': 1e-9}
        )

                best_h = result.x
