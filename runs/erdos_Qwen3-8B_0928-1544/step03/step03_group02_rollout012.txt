        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_perturbed
            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0.0, 1.0)

        result = minimize(compute_C5, perturbed, method='COBYLA', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': max_iter, 'ftol': 1e-8})
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

        elapsed = time.time() - start
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
