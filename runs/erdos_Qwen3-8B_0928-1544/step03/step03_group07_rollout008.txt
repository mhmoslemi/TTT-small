import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Initialize with provided initial_h_values
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:

    result = differential_evolution(objective, bounds, strategy='best1bin',

                                   mutation=(0.5, 1), recombination=0.9,
                                   disp=False)

    c5_bound = np.max(corr * dx)
