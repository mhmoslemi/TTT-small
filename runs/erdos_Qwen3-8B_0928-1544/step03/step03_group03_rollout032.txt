import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n = 70
    dx = 2.0 / n
    start_time = time.time()
    
    # Initialize population with initial_h_values or random
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.random.rand(n)
        initial_guess = (initial_guess / initial_guess.sum()) * 35
        initial_guess = np.clip(initial_guess, 0, 1)
    
    population = [initial_guess]
    for _ in range(49):
        perturb = np.random.normal(0, 0.05, n)
        new_h = initial_guess + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = (new_h / new_h.sum()) * 35
        population.append(new_h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, 'full')
        return -corr.max() * dx
    
    best_h = population[0]
    best_val = evaluate(best_h)
    
    def select_parents(pop):
        parents = []
        for _ in range(len(pop)):
            idxs = np.random.choice(len(pop), 5, replace=False)
            best = pop[idxs[0]]
            for i in idxs[1:]:
                if evaluate(pop[i]) > evaluate(best):
                    best = pop[i]
            parents.append(best)
        return parents
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        parents = select_parents(population)
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            split = np.random.randint(1, n-1)
            c1 = np.concatenate([p1[:split], p2[split:]])
            c2 = np.concatenate([p2[:split], p1[split:]])
            c1 = (c1 / c1.sum()) * 35
            c1 = np.clip(c1, 0, 1)
            c2 = (c2 / c2.sum()) * 35
            c2 = np.clip(c2, 0, 1)
            new_pop.append(c1)
            new_pop.append(c2)
        for i in range(len(new_pop)):
            h = new_pop[i]
            mu = np.random.normal(0, 0.03, n)
            h_mut = h + mu
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = (h_mut / h_mut.sum()) * 35
            new_pop[i] = h_mut
        population = new_pop
        for h in population:
            val = evaluate(h)
            if val > best_val:
                best_val = val
                best_h = h
    
    return best_h, -best_val, n
