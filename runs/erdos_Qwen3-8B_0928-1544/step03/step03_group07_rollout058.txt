        corr = correlate(h, 1 - h, mode='full')

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

        max_overlap = np.max(corr * dx)

            best_h = best_h.copy()
        if best_c5 < 0.38085 - 1e-6:
            break
        shift_idx = np.argmax(corr * dx)
        k_max = shift_idx - (n_points - 1)

        if best_c5 < 0.38085:

        res = minimize(
