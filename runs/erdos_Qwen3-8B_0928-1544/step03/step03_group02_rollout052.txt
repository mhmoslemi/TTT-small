import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import cvxpy as cp

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    S = n_points / 2.0

    h_binary = (h_initial > 0.5).astype(int)
    sum_h = np.sum(h_binary)
    diff = S - sum_h
    if abs(diff) > 1e-6:
        h_binary[-1] += np.round(diff)
        h_binary = np.clip(h_binary, 0, 1)

    h = cp.Variable(n_points, boolean=True)
    o = []
    for k in range(n_points):
        o_k = 0.0
        for i in range(n_points - k):
            o_k += h[i] * h[i + k]
        o.append(o_k)

    M = cp.Variable()
    constraints = [S - o[k] <= M for k in range(n_points)]
    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        prob.solve(solver=cp.SCIP, time_limit=budget_s)
    except:
        pass  # Fall back to last known solution if solver fails

    best_h = h.value.astype(int)
    best_c5 = M.value * dx if prob.status == 'optimal' else np.inf
    return best_h, best_c5, n_points
