import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 100
        h = np.zeros(n)
        h[:n//3] = 1
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 1
        h = h / np.sum(h) * (n/2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for _ in range(20):
        perturb = np.random.uniform(-0.2, 0.2, size=n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        res = minimize(
            objective,
            h_perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10, 'eps': 1e-8}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    final_sum = np.sum(best_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        best_h = best_h / final_sum * (n_points / 2)
    
    return best_h, best_c5, n_points
