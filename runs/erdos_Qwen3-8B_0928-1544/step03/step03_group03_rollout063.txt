from scipy.optimize import minimize, Bounds
import time

    best_h = initial_guess
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    
    num_restarts = 50
    for _ in range(num_restarts):
        perturb = np.random.normal(0, 0.05, n)
        perturbed_guess = np.clip(initial_guess + perturb, 0.0, 1.0)
        
        result = minimize(
            fun=objective,
            x0=perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(perturbed_guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_guess
        
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n
