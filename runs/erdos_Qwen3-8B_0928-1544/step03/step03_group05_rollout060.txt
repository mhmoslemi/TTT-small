import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
        n_points = len(h)
        dx = 2.0 / n_points
    else:
        n_points = 70
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    best_h = h
    best_c5 = np.inf

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    population_size = 50
    population = [np.random.rand(n_points) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] / np.sum(population[i]) * (n_points / 2)

    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        scores = [compute_c5(h) for h in population]
        top_pop = sorted(zip(population, scores), key=lambda x: x[1])[:int(len(population) * 0.2)]
        new_pop = []
        while len(new_pop) < population_size:
            if len(top_pop) < 2:
                break
            p1, p2 = np.random.choice([p for p, s in top_pop], 2, replace=False)
            child = (p1 + p2) / 2
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            total = np.sum(child)
            child = child * (n_points / 2) / total if total > 0 else child
            new_pop.append(child)
            if len(new_pop) < population_size:
                rand_h = np.random.rand(n_points)
                rand_h = rand_h / np.sum(rand_h) * (n_points / 2)
                new_pop.append(rand_h)
        population = new_pop
        if any(compute_c5(h) < best_c5 for h in population):
            best_c5 = min(compute_c5(h) for h in population)
            best_h = min(population, key=compute_c5)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    res = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-10,
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
    )
    final_h = res.x
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
