import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    needed_sum = n_points / 2.0

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum_and_clip(h, needed):
        curr = np.sum(h)
        delta = needed - curr
        h[-1] += delta
        h = np.clip(h, 0, 1)
        curr = np.sum(h)
        delta = needed - curr
        if abs(delta) > 1e-10:
            h[-1] += delta
            h = np.clip(h, 0, 1)
        return h

    population_size = 50
    population = [h_initial.copy() for _ in range(population_size)]
    for i in range(population_size):
        noise = np.random.normal(0, 0.05, n_points)
        population[i] += noise
        population[i] = np.clip(population[i], 0, 1)
        population[i] = adjust_sum_and_clip(population[i], needed_sum)

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    max_generations = 500
    generations = 0
    mutation_rate = 0.1
    temp = 1.0
    temp_decay = 0.99

    while time.time() - start_time < budget_s and generations < max_generations:
        fitness = np.array([compute_C5(h) for h in population])
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:population_size//2]]
        new_pop = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2.0
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            child = adjust_sum_and_clip(child, needed_sum)
            new_pop.append(child)
        population = new_pop

        for h in population:
            c5 = compute_C5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        temp *= temp_decay
        generations += 1

    return best_h, best_c5, n_points
