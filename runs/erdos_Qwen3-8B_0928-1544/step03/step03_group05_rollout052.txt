import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()

    best_h = h

    n_pop = 50
    n_gen = 100
    time_start = time.time()
    
    def score(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        val = np.max(corr) * dx
        return val
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(n_gen):
        if time.time() - time_start > budget_s:

        best_c5 = min(best_c5, c_vals[best_idx])
        if best_c5 < 0.38085:

    def obj(h_flat):

        corr = correlate(h, 1 - h, mode='full')
        val = np.max(corr) * dx
        return val
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 200})
    final_h = res.x
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
