from scipy.optimize import minimize, Bounds
from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    n = 70
    dx = 2.0 / n
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.astype(np.float64)

    best_h = initial_guess
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    best_c5 = float(best_c5)
    best_h = best_h.astype(np.float64)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return float(corr.max()) * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Fourier-based initialization
    def fourier_init(h):
        H = fft(h)
        F = fft(1 - h)
        C = H * np.conj(F)
        max_C = np.abs(C).max()
        H_opt = np.sqrt(1.0 / max_C) * H
        h_opt = np.real(ifft(H_opt))
        return np.clip(h_opt, 0.0, 1.0)
    
    initial_guess = fourier_init(initial_guess)
    
    best_h = initial_guess
    best_c5 = compute_c5(best_h)
    
    # Multi-start optimization with different strategies

            perturbed_guess = np.clip(initial_guess + perturb, 0.0, 1.0)

        result = minimize(
