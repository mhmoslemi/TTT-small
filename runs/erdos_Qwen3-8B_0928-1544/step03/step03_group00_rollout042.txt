import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    n_points = 70
    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = float('inf')
    pop_size = 20
    gen_limit = 100
    elapsed = 0.0
    np.random.seed(seed)
    
    def eval_c5(h):
        return np.max(correlate(h, 1 - h, mode='full')) * dx
    
    def mutate(h, gen):
        noise = np.random.normal(0, 0.1 * (1 - gen / gen_limit), n_points)
        h_mut = np.clip(h + noise, 0, 1)
        h_mut *= 35.0 / np.sum(h_mut)
        return h_mut
    
    def cross(parent1, parent2):
        child = (parent1 + parent2) / 2
        child = np.clip(child, 0, 1)
        child *= 35.0 / np.sum(child)
        return child
    
    def refine(h):
        def obj(x):
            return eval_c5(x)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35}
        return minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8).x
    
    pop = [h_initial.copy() for _ in range(pop_size)]
    start_time = time.time()
    
    for gen in range(gen_limit):
        c5s = [eval_c5(h) for h in pop]
        best_idx = np.argmin(c5s)
        if c5s[best_idx] < best_c5:
            best_c5 = c5s[best_idx]
            best_h = pop[best_idx].copy()
        
        top5 = np.argsort(c5s)[:5]
        for idx in top5:
            pop[idx] = refine(pop[idx])
        
        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(len(pop), 2, replace=False)
            new_pop.append(mutate(pop[p1], gen))
            new_pop.append(cross(pop[p1], pop[p2]))
        
        pop = new_pop
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
