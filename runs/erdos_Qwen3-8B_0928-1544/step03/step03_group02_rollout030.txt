import numpy as np
from scipy.optimize import basinhopping, minimize

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        return compute_C5(h)

    result = basinhopping(

        h_initial,
        niter=150,
        T=1.0,
        stepsize=0.5,
        minimize_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': [{'type': 'eq', 'fun': constraint_sum}],
            'options': {'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        }
    )

    best_h = result.x
    best_c5 = compute_C5(best_h)
