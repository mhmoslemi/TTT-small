import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n = 70
    dx = 2.0 / n

    # Initialize h with initial_h_values if available or use uniform 0.5
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = initial_h_values.copy()
    else:
        h = np.full(n, 0.5)

    best_h = h.copy()

    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return corr.max() * dx

    best_c5 = compute_c5(best_h)

    # Simulated Annealing parameters
    temp = 0.5
    cooling_rate = 0.99
    max_iter = 5000
    iterations = 0

    while time.time() - start_time < budget_s and iterations < max_iter:
        perturbation = np.random.normal(0, 0.05, n)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_sum = np.sum(h_perturbed)
        h_norm = h_perturbed * (n / 2.0) / h_sum

        new_c5 = compute_c5(h_norm)
        if new_c5 < best_c5:
            h = h_norm.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = h_norm.copy()
        temp *= cooling_rate
        iterations += 1

    # Final local optimization
    def objective(x):
        return compute_c5(x)

    def constraint_sum(x):
        return np.sum(x) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-10,
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return best_h, best_c5, n
