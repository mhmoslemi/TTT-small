import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    max_iterations = 5000
    early_stop = 1000
    no_improvement = 0
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        # Compute gradient for this k
        j_plus_k = (np.arange(n_points) + k) % n_points
        j_minus_k = (np.arange(n_points) - k) % n_points
        gradient = (1 - h[j_plus_k]) - h[j_minus_k]
        gradient *= dx  # scale by dx

        # Update velocity and h
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity

        # Clip and normalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            scale = (n_points / 2) / current_sum
            h *= scale
        else:
            h = np.zeros(n_points)

        # Compute current C5
        curr_c5 = max_val * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= early_stop:
                break

        # Adaptive learning rate
        if _ % 500 == 0 and learning_rate > 1e-5:
            learning_rate *= 0.95

        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

    # Final check
    h = np.clip(best_h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h *= scale
    else:
        h = np.zeros(n_points)

    return h, best_c5, n_points
