import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    pop_size = 50
    max_gen = 300
    mut_rate = 0.1
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def adjust_sum(h):
        s = np.sum(h)
        diff = (n_points / 2) - s
        h[-1] += diff
        h = np.clip(h, 0, 1)
        return h

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [adjust_sum(h_initial + np.random.normal(0, 0.05, n_points)) for _ in range(pop_size)]

    start_time = time.time()
    for gen in range(max_gen):
        fitness = [compute_C5(h) for h in population]
        indices = np.argsort(fitness)
        parents = [population[i] for i in indices[:int(len(population)/2)]]
        offspring = []
        for i in range(0, len(parents), 2):
            a, b = parents[i], parents[i+1]
            child = np.random.rand() * a + (1 - np.random.rand()) * b
            child = adjust_sum(child)
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.01, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = adjust_sum(child)
            offspring.append(child)
        population = parents + offspring
        if compute_C5(population[0]) < best_c5:
            best_h = population[0].copy()
            best_c5 = compute_C5(best_h)
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
