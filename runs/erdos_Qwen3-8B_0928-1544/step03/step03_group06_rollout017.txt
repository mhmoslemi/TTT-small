from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

        scale = (n_points / 2) / total if total != 0 else 1
        h_initial *= scale
    else:
        h_initial = np.full(180, 0.5, dtype=np.float32)
        n_points = 180

    population = [h_initial]
    best_h = h_initial.copy()
    best_c5 = np.inf

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * (2.0 / n_points)
        return max_overlap

    def rescale_h(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        scale = (n_points / 2) / total if total != 0 else 1
        return h_clipped * scale

    for _ in range(300):
        population = [rescale_h(p) for p in population]
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        if best_c5 < 0.38085:

        selected_indices = np.argsort(fitness)[:len(population)//2]
        selected = [population[i] for i in selected_indices]
