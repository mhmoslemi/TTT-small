import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 100
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for iteration in range(150):
        corr = correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        if best_c5 < 0.38085:
            break
        
        shift_idx = np.argmax(corr * dx)
        k_max = shift_idx - (n_points - 1)
        
        shift_range = np.arange(max(0, k_max - 40), min(n_points*2, k_max + 40))
        corr_sub = correlate(best_h, 1 - best_h, mode='full')[shift_range]
        max_sub = np.max(corr_sub * dx)
        if max_sub < best_c5:
            best_c5 = max_sub
            best_h = best_h.copy()
        
        if best_c5 < 0.38085:
            break
        
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-12, 'disp': False}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
