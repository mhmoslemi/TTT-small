from scipy.optimize import minimize

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Adjust initial guess to meet sum constraint
    avg_sum = n_points / 2
    if not np.isclose(np.sum(h_initial), avg_sum):

        h_initial[-1] += diff
        h_initial = np.clip(h_initial, 0, 1)

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed = 0.0
    start_time = time.time()

    # Optimize using SLSQP with enhanced exploration
    num_restarts = 30
    max_iter = 500
    pert_scale = 0.05

    for _ in range(num_restarts):
        np.random.seed(seed + _)  # Ensure different restarts
        perturbed = best_h + np.random.uniform(-pert_scale, pert_scale, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum constraint
        s = np.sum(perturbed)
        if not np.isclose(s, avg_sum):
            diff = avg_sum - s
            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0, 1)
        # Optimize from perturbed guess
        result = minimize(
            compute_C5,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
