        m = np.argmax(corr)
        lag = m - (n_points - 1)
        k = lag
        indices = (np.arange(n_points) + k) % n_points
        grad = (1 - h[indices]) * dx
        
        velocity = alpha * velocity - eta * grad
        h = h + velocity
        
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            h = h / sum_h * (n_points / 2)
        
        current_c5 = max_val * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improve_count = 0
        else:
            improve_count += 1
        
        if improve_count >= early_stop:
            break
        
        eta *= decay
    
    final_h = np.clip(best_h, 0, 1)
    sum_final = np.sum(final_h)
    if np.abs(sum_final - n_points / 2) > 1e-6:
        final_h = final_h / sum_final * (n_points / 2)
    
    return final_h, best_c5, n_points
