import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_half = n_points / 2
    dx = 2.0 / n_points
    P = 100
    elite_ratio = 0.2
    tournament_size = 5
    max_gen = 200
    time_limit = budget_s
    T = time.time()

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        h_initial = np.clip(h_initial * (n_half / np.sum(h_initial)), 0, 1)
        pop = [h_initial]
        for _ in range(P - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (n_half / h_sum)
            pop.append(h)
    else:
        pop = [np.random.rand(n_points) * 0.5 * (n_half / np.sum(np.random.rand(n_points) * 0.5)) for _ in range(P)]

    def compute_c5(h_val):
        return np.max(np.correlate(h_val, 1 - h_val, mode='full')) * dx

    best_h = pop[0]
    best_c5 = compute_c5(best_h)

    for gen in range(max_gen):
        if time.time() - T > budget_s:
            break
        fit = [compute_c5(h) for h in pop]
        idx = np.argsort(fit)
        sorted_pop = [pop[i] for i in idx]
        sorted_fit = [fit[i] for i in idx]
        elite_size = int(P * elite_ratio)
        elites = sorted_pop[:elite_size]
        new_pop = elites.copy()
        while len(new_pop) < P:
            t = np.random.choice(range(len(sorted_pop) - elite_size), size=tournament_size)
            p1, p2 = sorted_pop[elite_size + t[0]], sorted_pop[elite_size + t[1]]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
