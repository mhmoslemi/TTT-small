        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()

        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
