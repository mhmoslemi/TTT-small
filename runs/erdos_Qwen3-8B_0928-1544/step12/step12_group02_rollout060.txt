    best_c5 = evaluate_erdos_solution(global_best)

    for p in particles:
        c5 = evaluate_erdos_solution(p)
        if c5 < best_c5:
            best_c5 = c5
            global_best = p.copy()

    start_time = time.time()
    w = w_start
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            vel = np.clip(vel, -0.1, 0.1)
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            particle_sum = np.sum(particles[i])
            if np.isclose(particle_sum, 0):
                particles[i] = np.random.rand(n_points)
                particle_sum = np.sum(particles[i])
            particles[i] = particles[i] * (ones_count / particle_sum)
            current_c5 = evaluate_erdos_solution(particles[i])
            if current_c5 < evaluate_erdos_solution(personal_bests[i]):
                personal_bests[i] = np.copy(particles[i])
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = np.copy(particles[i])
        w -= (w_start - w_end) / 1000
        if w < w_end:
            w = w_end

    return global_best, best_c5, n_points
