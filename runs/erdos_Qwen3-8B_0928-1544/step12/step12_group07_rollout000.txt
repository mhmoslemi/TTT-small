    h = np.clip(h, 0.0, 1.0)
    if np.sum(h) != target_sum:
        h = h * (target_sum / np.sum(h))

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, jac=gradient, options={'maxiter': 1000})

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
