    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(199):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(200):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    def objective_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h_de = pop[0]
    best_c5_de = objective_de(best_h_de)
    start_time = time.time()

    F = 0.6
    CR = 0.6
    stagnation = 0
    iter_de = 0
    max_iter_de = int(0.7 * budget_s)

    while iter_de < max_iter_de and time.time() - start_time < budget_s:
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i] * (1 - CR) + mutant * CR
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial_scaled = trial * (ones_count / trial_sum)
            curr_c5 = objective_de(pop[i])
            new_c5 = objective_de(trial_scaled)
            if new_c5 < curr_c5:
                pop[i] = trial_scaled.copy()
                if new_c5 < best_c5_de:
                    best_c5_de = new_c5
                    best_h_de = trial_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 200:
                F = max(F + 0.01, 0.9)
                CR = min(CR + 0.01, 0.95)
                stagnation = 0
        iter_de += 1

    # SQP phase
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def grad_func(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (n_points - 1)
            grad = np.zeros(n_points)
            for j in range(n_points):
                if 0 <= j + k < n_points:
                    grad[j] += (1 - h[j + k]) * dx
                if 0 <= j - k < n_points:
                    grad[j] -= h[j - k] * dx
            return grad

        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        start_sqp = time.time()
        try:
            result = minimize(objective, x0=best_h_de, method='SLSQP', jac=grad_func, constraints=cons, bounds=bounds, tol=1e-6)
            best_h_sqp = result.x
            best_c5_sqp = objective(best_h_sqp)
        except:
            best_h_sqp = best_h_de
            best_c5_sqp = best_c5_de

        best_c5 = min(best_c5_de, best_c5_sqp)
        best_h = best_h_sqp if best_c5_sqp < best_c5_de else best_h_de
    else:
        best_h = best_h_de
        best_c5 = best_c5_de

    return best_h, best_c5, n_points
