        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k_max = m - (n_points - 1)
        i = np.arange(n_points)
        j = i + k_max
        valid = (j >= 0) & (j < n_points)
        grad = np.zeros(n_points)
        grad[valid] = 1.0 - h[j[valid]]
        ones = np.ones(n_points)
        grad_sum = np.dot(grad, ones)
        proj_grad = grad - (grad_sum / n_points) * ones
        alpha = 1.0
        while True:
            h_test = h + alpha * proj_grad
            h_test_clipped = np.clip(h_test, 0.0, 1.0)
            h_test_sum = np.sum(h_test_clipped)
            if h_test_sum == 0:
                alpha *= 0.5
                if alpha < 1e-6:
                    break
                continue
            h_test_scaled = h_test_clipped * (ones_count / h_test_sum)
            corr_test = np.correlate(h_test_scaled, 1 - h_test_scaled, mode='full')
            new_max_corr = np.max(corr_test)
            if new_max_corr < np.max(corr):
                break
            else:
                alpha *= 0.5
                if alpha < 1e-6:
                    break
        h = h + alpha * proj_grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h = h * (ones_count / h_sum)
        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.380876:
            break
    return best_h, best_c5, n_points
