            change_in_R = perturbed_R - original_R
            grad[i] = (change_in_c5 + λ * change_in_R) / 1e-4

        if np.abs(np.max(grad)) < 1e-5:
            break

        gradient_norm = np.sqrt(np.sum(grad**2))
        step_size = 0.3 / (1 + gradient_norm)
        neg_grad = -grad
        idx = np.argmax(neg_grad)
        new_h = h.copy()
        new_h[idx] += step_size
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (n_half / new_h_sum)
        c_new = compute_c5(new_h)
        if c_new < best_c5:
            h = new_h
            best_c5 = c_new
            iter_count = 0
        else:
            iter_count += 1
            if iter_count > 300:
                break

        if _ % 100 == 0:
            h = h + 0.01 * (2 * np.random.rand(n_points) - 1)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
                h = h * (n_half / np.sum(h))
            else:
                h = h * (n_half / h_sum)
            c_new = compute_c5(h)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()

        if time.time() - T > budget_s * 0.9:
            break

    return h, best_c5, n_points
