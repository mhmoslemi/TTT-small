import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    c1, c2 = 2.0, 2.0
    inertia_start, inertia_end = 0.9, 0.4
    vel_clip = 0.1

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0]
    best_c5 = objective(pop[0])
    start_time = time.time()
    inertia = inertia_start
    stagnation = 0

    for _ in range(10000):  # ~1000 iterations
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            vel = pop[i] * np.random.rand(n_points) * 0.5
            vel += c1 * r1 * (np.random.rand(n_points) - 0.5) * 0.5
            vel += c2 * r2 * (np.random.rand(n_points) - 0.5) * 0.5
            vel = np.clip(vel, -vel_clip, vel_clip)
            trial = pop[i] + vel
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial_scaled = trial * (ones_count / trial_sum)
            curr_c5 = objective(pop[i])
            new_c5 = objective(trial_scaled)
            if new_c5 < curr_c5:
                pop[i] = trial_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = trial_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 200:
                inertia = max(inertia_start, inertia - 0.002)
                stagnation = 0

    return best_h, best_c5, n_points
