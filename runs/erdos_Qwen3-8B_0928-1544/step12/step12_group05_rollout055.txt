        if sum_initial > 0:
            initial_h = initial_h * (target_sum / sum_initial)
        else:
            initial_h = np.random.rand(n_points) * 0.5
            sum_initial = np.sum(initial_h)
            initial_h = initial_h * (target_sum / sum_initial)
    else:
        initial_h = np.random.rand(n_points) * 0.5
        sum_initial = np.sum(initial_h)
        if sum_initial > 0:
            initial_h = initial_h * (target_sum / sum_initial)
        else:
            initial_h = np.random.rand(n_points) * 0.5
    
    # Setup convex problem
    s = cp.Variable(n_points + 1)
    C5 = cp.Variable()
    constraints = [
        s[0] == 0,
        s[n_points] == 1.0
    ]
    for i in range(1, n_points + 1):
        constraints += [s[i] >= s[i-1], s[i] <= s[i-1] + dx]
    # Add cross-correlation constraints
    for k in range(1, n_points):
        for i in range(0, n_points - k + 1):
            constraints.append(s[i + k] - s[i] <= C5)
    
    prob = cp.Problem(cp.Minimize(C5), constraints)
    
    # Solve convex problem with fallback
    best_h = initial_h.copy()
    best_c5 = float('inf')
    try:
        prob.solve(solver='MOSEK', eps=1e-6, verbose=False)
        if prob.status == 'OPTIMAL':
            s_opt = s.value
            h_opt = (s_opt[1:] - s_opt[:-1]) / dx
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_opt = np.sum(h_opt)
            if sum_opt > 0:
                h_opt = h_opt * (target_sum / sum_opt)
            best_h, best_c5 = h_opt, C5.value
        else:
            # Fallback to initial guess
            best_h, best_c5 = initial_h, float('inf')
    except:
        # Fallback to initial guess
        best_h, best_c5 = initial_h, float('inf')
    
    # Optimize using Nelder-Mead
    def objective(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx
    
    def clip_and_scale(h_candidate):
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_h = np.sum(h_candidate)
        if sum_h == 0:
            return np.inf
        return h_candidate * (target_sum / sum_h)
    
    # Initial guess for Nelder-Mead
    h_initial = clip_and_scale(best_h)
    
    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='Nelder-Mead',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-6}
    )
    elapsed_time = time.time() - start_time
    
    # Determine best solution based on time remaining
    if elapsed_time < budget_s - 100:
        final_h = clip_and_scale(result.x)
        final_c5 = result.fun
    else:
        final_h = clip_and_scale(best_h)
        final_c5 = best_c5
    
    return final_h, final_c5, n_points
