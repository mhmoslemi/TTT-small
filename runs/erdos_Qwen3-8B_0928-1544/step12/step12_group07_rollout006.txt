import time

    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Compute gradient for the current max k
        gradient = np.zeros(n_points)
        for x in range(n_points):
            x_shifted = x + k
            if 0 <= x_shifted < n_points:
                gradient[x] = 1 - h[x_shifted]
        
        # Compute direction vector
        direction = gradient - h
        
        # Line search for optimal alpha
        possible_alphas = [0.05, 0.1, 0.15, 0.2, 0.25]
        best_new_c5 = float('inf')
        best_alpha = 0.15
        for alpha in possible_alphas:
            new_h = h + alpha * direction
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h_sum = np.sum(new_h)
            if new_h_sum == 0:
                continue
            new_h = new_h * (ones_count / new_h_sum)
            new_c5 = compute_c5(new_h, n_points)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
        
        # Apply best alpha
        new_h = h + best_alpha * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (ones_count / new_h_sum)
        new_c5 = compute_c5(new_h, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        h = new_h

    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(corr) * (2.0 / n_points)
    return max_correlation

    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(corr) * (2.0 / n_points)
    return max_correlation
