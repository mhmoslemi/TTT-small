import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    T = time.time()
    max_iter = int(budget_s / 3)

    iter_count = 0

    while time.time() - T < budget_s:
