                h_try = h.copy()
            else:
                h_try = h_try * (required_sum / sum_h)
            obj_try = compute_obj(h_try)
            if obj_try < best_val or np.random.rand() < np.exp((best_val - obj_try) / T):
                h = h_try.copy()
                best_val = obj_try
                best_h = h.copy()
        T *= cooling_rate
    return best_h, best_val, n_points
