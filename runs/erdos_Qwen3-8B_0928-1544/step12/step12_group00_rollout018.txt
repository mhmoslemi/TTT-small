import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    max_velocity = 0.1
    inertia_weight = 0.7
    cognitive = 2.0
    social = 2.0
    max_iterations = 500
    stagnation_threshold = 50
    initial_velocity = 0.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.5
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (ones_count / initial_sum)

    particles = [initial_h.copy() for _ in range(population_size)]
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in particles]
    global_best = initial_h.copy()
    best_c5 = float('inf')

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    stagnation = 0

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia_weight * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            # Update position
            particles[i] += velocities[i]
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_p = np.sum(particles[i])
            if sum_p == 0:
                continue
            particles[i] = particles[i] * (ones_count / sum_p)
            # Evaluate
            current_c5 = objective(particles[i])
            if current_c5 < objective(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = particles[i].copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > stagnation_threshold:
            max_velocity = np.clip(max_velocity + 0.02, 0.0, 0.2)
            stagnation = 0

    return global_best, best_c5, n_points
