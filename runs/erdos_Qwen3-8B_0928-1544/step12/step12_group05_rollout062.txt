        s_initial[0] = 0.0
        for i in range(n_points):
            s_initial[i+1] = s_initial[i] + initial_h[i] * dx
    else:
        s_initial = np.linspace(0.0, 1.0, n_points + 1)

    # Set up convex problem
    s = cp.Variable(n_points + 1)
    C = cp.Variable()
    constraints = [
        s[0] == 0.0,
        s[n_points] == 1.0
    ]
    for i in range(1, n_points + 1):
        constraints.append(s[i] - s[i-1] <= 1.0)
        constraints.append(s[i] - s[i-1] >= 0.0)

    for k in range(1, n_points):
        for i in range(0, n_points - k):
            constraints.append(s[i + k] - s[i] <= C)

    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)

    # Solve convex problem with fallback
    s_solution = s_initial.copy()
    try:
        problem.solve(solver=cp.MOSEK, verbose=False)
        if problem.status == 'OPTIMAL':
            s_solution = s.value
    except:
        try:
            problem.solve(solver=cp.GUROBI, verbose=False)
            if problem.status == 'OPTIMAL':
                s_solution = s.value
        except:
            pass

    # Recover h from convex solution
    h_solution = np.zeros(n_points)
    for i in range(n_points):
        h_solution[i] = (s_solution[i+1] - s_solution[i]) / dx

    # Nelder-Mead refinement
    def objective(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return float('inf')
        h_scaled = h_clipped * (n_points / 2) / total
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Initial guess from convex solution
    result = minimize(objective, h_solution, method='Nelder-Mead',
                      options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 1000})

    # Compare with convex solution
    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return float('inf')
        h_scaled = h_clipped * (n_points / 2) / total
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    c5_convex = compute_c5(h_solution)
    best_h = result.x
    best_c5 = result.fun

    if c5_convex < best_c5:
        best_h = h_solution
        best_c5 = c5_convex

    return best_h, best_c5, n_points
