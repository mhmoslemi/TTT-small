import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h * 0.7 + 0.15, 0, 1)
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (target_sum / initial_sum)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    h_population = [initial_h]
    for _ in range(199):
        h = np.random.rand(n_points) * 0.3 + 0.2
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.3 + 0.2
        h = h * (target_sum / np.sum(h))
        h_population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    F = 0.8
    CR = 0.9
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        for _ in range(len(h_population)):
            a, b, c = np.random.choice(len(h_population), 3, replace=False)
            a_vec, b_vec, c_vec = h_population[a], h_population[b], h_population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(h_population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u = u * (target_sum / u_sum)
            new_c5 = objective(u)
            if new_c5 < best_c5:
                best_h = u.copy()
                best_c5 = new_c5
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
                F = max(0.6, F - 0.01)
                CR = min(0.98, CR + 0.01)
                stagnation = 0
    
    return best_h, best_c5, n_points
