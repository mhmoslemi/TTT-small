import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):

    best_h = h.copy()

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Compute accurate gradient for k
        gradient = np.zeros(n_points)
        for y in range(n_points):
            y_k = (y + k) % n_points
            y_km = (y - k) % n_points
            gradient[y] = (1 - h[y_k]) - h[y_km]
        
        direction = -gradient  # negative gradient for descent
        alpha = 0.5

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(corr) * (2.0 / n_points)
    return max_correlation
