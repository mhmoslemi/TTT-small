import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2  # 70
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = initial_h * (required_sum / np.sum(initial_h))
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (required_sum / np.sum(initial_h))
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    current_h = initial_h.copy()
    current_best = obj(current_h)
    best_h = current_h.copy()
    best_obj = current_best
    
    T = 0.5
    T_final = 1e-4
    cooling_rate = 0.995
    iterations_per_temp = 100
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for _ in range(iterations_per_temp):
            perturbation = T * np.random.randn(n_points)
            h_try = current_h + perturbation
            h_try = np.clip(h_try, 0.0, 1.0)
            sum_try = np.sum(h_try)
            if sum_try == 0.0:
                h_try = current_h.copy()
            else:
                h_try = h_try * (required_sum / sum_try)
            obj_try = obj(h_try)
            if obj_try < current_best or np.random.rand() < np.exp((current_best - obj_try) / T):
                current_h = h_try.copy()
                current_best = obj_try
                if current_best < best_obj:
                    best_h = current_h.copy()
                    best_obj = current_best
        T *= cooling_rate
        if T < T_final:
            break
    
    return best_h, best_obj, n_points
