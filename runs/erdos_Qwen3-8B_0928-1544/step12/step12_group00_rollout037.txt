        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = j + lag
            if 0 <= idx < n_points:
                grad[j] = 1.0 - h[idx]
        # Project gradient to exclude sum constraint
        sum_grad = np.sum(grad)
        proj = (sum_grad / n_points) * np.ones(n_points)
        adjusted_grad = grad - proj
        # Line search
        alpha = 0.1
        for _ in range(10):
            new_h = h - alpha * adjusted_grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h_sum = np.sum(new_h)
            if new_h_sum == 0:
                alpha *= 0.5
                continue
            new_h_rescaled = new_h * (ones_count / new_h_sum)
            new_corr = np.correlate(new_h_rescaled, 1 - new_h_rescaled, mode='full')
            new_c5 = np.max(new_corr) * dx
            if new_c5 < current_c5:
                h = new_h_rescaled
                current_c5 = new_c5
                alpha = 0.1  # Reset step size
                break
            alpha *= 0.5
    return best_h, best_c5, n_points
