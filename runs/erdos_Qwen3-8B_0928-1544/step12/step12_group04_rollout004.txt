    pop_size = 200
    tournament_size = 10
    mutation_rate = 0.2
    crossover_rate = 0.8

            pop.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
