            p1, p2 = np.random.choice(selected, 2, replace=False)
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:split], p2[split:]])
            noise = np.random.normal(0, 0.01, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            child = child * (ones_count / child_sum) if child_sum > 0 else child
            offspring.append(child)
        population = offspring
        for h in population:
            c5 = compute_c5(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(corr) * (2.0 / n_points)
    return max_correlation
