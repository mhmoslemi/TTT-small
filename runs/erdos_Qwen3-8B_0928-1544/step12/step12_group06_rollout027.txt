                    delta = np.random.uniform(-0.1, 0.1)
                    mutated[k] += delta
                    mutated[k] = np.clip(mutated[k], 0.0, 1.0)
            mut_sum = np.sum(mutated)
            if mut_sum == 0:
                mutated = np.random.rand(n_points) * 0.5
                mut_sum = np.sum(mutated)
                if mut_sum == 0:
                    mutated = np.random.rand(n_points) * 0.5
                mutated = mutated * (n_points / 2) / mut_sum
            else:
                mutated = mutated * (n_points / 2) / mut_sum
            mutated = np.clip(mutated, 0.0, 1.0)
            offspring[j] = mutated

        next_gen += offspring[:remaining]
        population = next_gen

        current_best_idx = np.argmin(sorted_fitness)
        current_best = sorted_population[current_best_idx]
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

    return best_h, best_c5, n_points
