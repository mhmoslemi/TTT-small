                    gbest_value = current_value
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > 50:
            w = max(w - 0.01, 0.4)
            sorted_indices = np.argsort(pbest_value)
            for j in range(int(population_size * 0.1)):
                idx = sorted_indices[j]
                h = np.random.rand(n_points) * 0.5
                h_sum = np.sum(h)
                if h_sum == 0:
                    h = np.random.rand(n_points) * 0.5
                population[idx] = h * (target_sum / h_sum)
            stagnation = 0
    
    gbest = np.clip(gbest, 0.0, 1.0)
    current_sum = np.sum(gbest)
    if current_sum != target_sum:
        gbest *= target_sum / current_sum
    return gbest, gbest_value, n_points
