    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for x in range(n_points):
            x_shifted = x + k
            if 0 <= x_shifted < n_points:
                gradient[x] = (1 - h[x_shifted]) * 2 / n_points
        
        direction = -gradient
        new_h = h + alpha * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (ones_count / new_h_sum)
        
        new_c5 = compute_c5(new_h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        h = new_h

    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(corr) * (2.0 / n_points)
    return max_correlation
