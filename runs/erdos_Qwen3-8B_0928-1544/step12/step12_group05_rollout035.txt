import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        sum_h = np.sum(initial_h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        initial_h = np.clip(initial_h * scale, 0.0, 1.0)
        s_initial = np.zeros(n_points + 1)
        for i in range(n_points):
            s_initial[i + 1] = s_initial[i] + initial_h[i] * dx
        s_initial[-1] = 1.0
    else:
        initial_h = np.random.rand(n_points)
        sum_h = np.sum(initial_h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        initial_h = np.clip(initial_h * scale, 0.0, 1.0)
        s_initial = np.zeros(n_points + 1)
        for i in range(n_points):
            s_initial[i + 1] = s_initial[i] + initial_h[i] * dx
        s_initial[-1] = 1.0

    s = cp.Variable(n_points + 1)
    C5 = cp.Variable()
    constraints = [
        s[0] == 0,
        s[n_points] == 1.0,
        [s[i] >= s[i - 1] for i in range(1, n_points + 1)]
    ]

    for i in range(n_points):
        constraints.append(s[i + 1] - s[i] <= dx)
        constraints.append(s[i + 1] - s[i] >= 0.0)

    for i in range(n_points):
        for k in range(1, n_points - i + 1):
            constraints.append(s[i + k] - s[i] <= C5)

    problem = cp.Problem(cp.Minimize(C5), constraints)
    problem.solve(solver=cp.MOSEK, verbose=False)

    h_values = np.zeros(n_points)
    for i in range(n_points):
        h_values[i] = (s[i + 1].value - s[i].value) / dx
    c5_bound = C5.value
    return h_values, c5_bound, n_points
