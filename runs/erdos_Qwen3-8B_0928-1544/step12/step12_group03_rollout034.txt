import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    initial_guesses = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        else:
            initial_h_scaled = np.random.rand(n_points) * 0.5
        initial_guesses.append(initial_h_scaled)
    # Uniform guess
    uniform_h = np.full(n_points, 0.5)
    initial_guesses.append(uniform_h)
    # 8 random guesses
    for _ in range(8):
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        initial_guesses.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = None
    best_c5 = float('inf')
    time_start = time.time()

    for h_initial in initial_guesses:
        h = np.clip(h_initial, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
        else:
            h = h * (ones_count / h_sum)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, h, method='L-BFGS-B', jac='2-point', bounds=bounds, tol=1e-6, options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        elapsed = time.time() - time_start
        if elapsed >= budget_s:
            break

    return best_h, best_c5, n_points
