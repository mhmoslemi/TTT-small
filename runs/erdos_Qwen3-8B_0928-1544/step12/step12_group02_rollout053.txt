import time

            h = np.random.rand(n_points) * 0.2 + 0.4
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.2 + 0.4
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize velocities and track bests

    personal_best = [h.copy() for h in pop]
    global_best = min(pop, key=objective)
    best_c5 = objective(global_best)
    start_time = time.time()
    inertia = inertia_start
    stagnation = 0

    while time.time() - start_time < budget_s:

            velocities[i] = (inertia * velocities[i] +

                             c2 * r2 * (global_best - pop[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            # Update position
            pop[i] += velocities[i]
            # Clip and scale

            h_sum = np.sum(pop[i])
