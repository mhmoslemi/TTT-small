import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    dx = 2.0 / n_points
    pop_size = 300
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h0 = initial * (required_sum / np.sum(initial))
        pop = [h0]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            pop.append(h * (required_sum / h_sum))
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            pop.append(h * (required_sum / h_sum))
    
    velocities = np.zeros((pop_size, n_points))
    
    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    pbest = [h.copy() for h in pop]
    pbest_vals = [obj(h) for h in pop]
    gbest = pop[0].copy()

    w = 0.7
    phi_p = 2.0
    phi_g = 2.0
    max_vel = 0.2

    while time.time() - start < budget_s:
