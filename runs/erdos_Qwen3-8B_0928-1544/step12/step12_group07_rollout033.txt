        return np.clip(h + np.random.uniform(-scale, scale, size=h.shape), 0, 1) * (ones_count / np.sum(h + np.random.uniform(-scale, scale, size=h.shape)))
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    for _ in range(3):
        h_perturb = perturb(h)
        result = minimize(objective, h_perturb, bounds=bounds, constraints=constraints, method='SLSQP', tol=1e-6)
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
