import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.15

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    # Initialize velocities and personal bests
    velocities = [np.random.rand(n_points) * max_velocity for _ in range(pop_size)]
    personal_best = [np.copy(p) for p in pop]
    global_best = np.copy(pop[0])
    best_c5 = float('inf')
    best_h = global_best

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    inertia = inertia_start

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocity = velocities[i] + c1 * r1 * (personal_best[i] - pop[i]) + c2 * r2 * (global_best - pop[i])
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            velocities[i] = velocity

            # Update position
            pop[i] += velocity
            pop[i] = np.clip(pop[i], 0, 1)

            # Scale to meet sum constraint
            h_sum = np.sum(pop[i])
            if h_sum == 0:
                continue
            pop[i] *= (ones_count / h_sum)

            # Evaluate
            current_c5 = objective(pop[i])

            # Update personal best
            if current_c5 < objective(personal_best[i]):
                personal_best[i] = np.copy(pop[i])

            # Update global best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.copy(pop[i])

        # Decrease inertia linearly
        remaining_time = budget_s - (time.time() - start_time)
        inertia = max(inertia_end, inertia_start - (inertia_start - inertia_end) * (remaining_time / budget_s))

    return best_h, best_c5, n_points
