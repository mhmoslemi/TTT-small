import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / np.sum(h))
    
    max_iterations = 1000
    learning_rate = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        j = np.argmax(corr)
        lag = j - (n_points - 1)
        
        gradient = np.zeros_like(h)
        if lag == 0:
            gradient = 1 - 2 * h
        elif lag > 0:
            start_i = 0
            end_i = n_points - lag - 1
            for i in range(start_i, end_i + 1):
                gradient[i] += (1 - h[i + lag])
                gradient[i + lag] -= h[i]
        elif lag < 0:
            start_i = -lag
            end_i = n_points - 1
            for i in range(start_i, end_i + 1):
                gradient[i] += (1 - h[i + lag])
                gradient[i + lag] -= h[i]
        
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        h_sum = np.sum(h_new)
        if h_sum == 0:
            h_new = h.copy()
        else:
            h_new = h_new * (ones_count / h_sum)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * (2.0 / n_points)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9
        
        h = h_new
    
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr_final) * (2.0 / n_points)
    return best_h, c5_final, n_points
