import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / np.sum(h))
    def logit(x):
        return np.log(x / (1 - x))
    a = logit(h)
    temp = 0.5
    best_c5 = float('inf')
    best_h = h.copy()
    def corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1 * temp, size=n_points)
        a_perturbed = a + noise
        h_new = 1 / (1 + np.exp(-a_perturbed))
        h_new = np.clip(h_new, 0, 1)
        summed = np.sum(h_new)
        if summed == 0:
            continue
        h_new = h_new * (target_sum / summed)
        c = corr(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        temp *= 0.999
    return best_h, best_c5, n_points
