    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=lambda h: _grad(h, n_points, dx), options={'ftol': 1e-6, 'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points

def _grad(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n_points - 1)
    grad = np.zeros(n_points)
    for i in range(n_points):
        i_plus_shift = (i + shift) % n_points
        i_minus_shift = (i - shift) % n_points
        grad[i] = (1 - h[i_plus_shift]) - h[i_minus_shift]
    return grad
