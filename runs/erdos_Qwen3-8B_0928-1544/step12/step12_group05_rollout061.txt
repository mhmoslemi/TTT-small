    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        initial_pop = [initial_h_values.copy()]
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / np.sum(h))
        initial_pop[0] = h
    else:
        initial_pop = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / h_sum)
            initial_pop.append(h)

    population = initial_pop
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0

    while time.time() - start_time < budget_s * 0.7:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u = u * (ones_count / u_sum)
            current_c5 = evaluate_erdos_solution(population[i], n_points, dx)
            new_c5 = evaluate_erdos_solution(u, n_points, dx)
            if new_c5 < current_c5:
                population[i] = u
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > stagnation_threshold:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

    def obj_nelder_mead(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h = h * (ones_count / h_sum)
        return evaluate_erdos_solution(h, n_points, dx)

    h_initial = best_h.copy()
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial_sum = np.sum(h_initial)
    h_initial = h_initial * (ones_count / h_initial_sum) if h_initial_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / np.sum(np.random.rand(n_points) * 0.5))

    res = minimize(obj_nelder_mead, h_initial, method='Nelder-Mead', options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 100})
    best_h = res.x
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    return best_h, best_c5, n_points
