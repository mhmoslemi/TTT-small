    def project(h):

        sum_clipped = np.sum(h_clipped)
        if np.isclose(sum_clipped, ones_count):
            return h_clipped
        if sum_clipped < ones_count:
            idxs = np.where(h_clipped < 1.0)[0]
            if len(idxs) == 0:
                return h_clipped
            add = np.zeros_like(h_clipped)
            add[idxs] = (ones_count - sum_clipped) / len(idxs)
            return h_clipped + add
        else:
            idxs = np.where(h_clipped > 0.0)[0]
            if len(idxs) == 0:
                return h_clipped
            sub = np.zeros_like(h_clipped)
            sub[idxs] = (sum_clipped - ones_count) / len(idxs)
            return h_clipped - sub

    def compute_gradient(h, k):

                g[i] += 1.0 - h[i + k]
            if 0 <= i - k < len(h):
