def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    max_iter = int(budget_s * 0.8)
    T = time.time()

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:

        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2 / h_sum)

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    iter_count = 0

    while iter_count < max_iter:
        grad = np.zeros(n_points)
