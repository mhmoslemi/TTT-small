        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0]
    best_c5 = objective(pop[0])
    start_time = time.time()

    F, CR = 0.6, 0.6
    stagnation = 0

    while time.time() - start_time < 0.7 * budget_s:
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i] * (1 - CR) + mutant * CR
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial_scaled = trial * (ones_count / trial_sum)
            curr_c5 = objective(pop[i])
            new_c5 = objective(trial_scaled)
            if new_c5 < curr_c5:
                pop[i] = trial_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = trial_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 200:
                F = max(0.6, F - 0.01)
                CR = min(0.95, CR + 0.01)
                stagnation = 0

    # SQP refinement
    def grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_indices = np.where(corr == np.max(corr))[0]
        k_index = max_indices[0]
        k = k_index - (n_points - 1)
        grad_vals = np.zeros_like(h)
        dx = 2.0 / n_points
        for j in range(n_points):
            term1 = 1.0
            if 0 <= j + k < n_points:
                term1 = 1.0 - h[j + k]
            else:
                term1 = 0.0
            term2 = 0.0
            if 0 <= j - k < n_points:
                term2 = h[j - k]
            grad_vals[j] = (term1 - term2) * dx
        return grad_vals

    cons = [LinearConstraint(np.ones(n_points), [ones_count], [ones_count])]
    bounds = Bounds(0, 1)
    res = minimize(objective, best_h, method='SLSQP', jac=grad, constraints=cons, bounds=bounds, options={'maxiter': 100})
    best_h = res.x
    best_c5 = res.fun
    return best_h, best_c5, n_points
