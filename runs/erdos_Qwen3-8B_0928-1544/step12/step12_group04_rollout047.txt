                h_new = h_var.value
                c5_new = c5_var.value
                h_current = h_new
                best_c5 = min(best_c5, c5_new)
                best_h = h_current.copy()
            else:
                break
        except:
            break

        total_time = time.time() - time.time()
        time.sleep(0.1)
        total_time = time.time() - time.time()

    return best_h, best_c5, n_points
