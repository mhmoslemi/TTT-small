            vel = np.clip(vel, -velocity_clip, velocity_clip)
            velocities[i] = vel
            pop[i] += vel
            pop[i] = np.clip(pop[i], 0, 1)
            sum_pop = np.sum(pop[i])
            if sum_pop == 0:
                pop[i] = np.random.rand(n_points)
            else:
                pop[i] = pop[i] * (ones_count / sum_pop)
            current_c5 = objective(pop[i])
            if current_c5 < objective(pbest[i]):
                pbest[i] = pop[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = pop[i].copy()

        for i in range(pop_size):
            if objective(pbest[i]) < objective(best_h):
                best_h = pbest[i].copy()
                best_c5 = objective(best_h)

        w = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
        stagnation += 1
        if stagnation > 200:
            w = max(w_start, w - 0.01)
            stagnation = 0

    return best_h, best_c5, n_points
