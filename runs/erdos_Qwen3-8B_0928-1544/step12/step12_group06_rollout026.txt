import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    P = 100
    mutation_rate = 0.1
    max_generations = 250
    stagnation_threshold = 50
    elitism_ratio = 0.2

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_pop = [initial_h_values]
        for _ in range(P - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / h_sum
            initial_pop.append(h)
    else:
        initial_pop = []
        for _ in range(P):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / h_sum
            initial_pop.append(h)
    
    population = np.array(initial_pop)
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([compute_c5(ind) for ind in population])
        sorted_indices = np.argsort(fitness)
        sorted_population = population[sorted_indices]
        sorted_fitness = fitness[sorted_indices]
        
        elite_size = int(elitism_ratio * P)

            best_idx = tournament[np.argmin(sorted_fitness[tournament])]
            parents.append(sorted_population[best_idx])
