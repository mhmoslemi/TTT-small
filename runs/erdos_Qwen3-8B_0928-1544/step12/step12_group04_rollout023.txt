        return np.max(corr * dx)

        scale = required_sum / total
        h_scaled = h_vec * scale
        return compute_overlap(h_scaled)

    class PSO:
        def __init__(self, n_particles, n_points, budget_s, initial_h):

            self.velocities = np.zeros((n_particles, n_points))
            self.best_positions = self.particles.copy()
            self.best_fitness = np.array([objective_func(p) for p in self.particles])
            self.global_best = self.particles[np.argmin(self.best_fitness)]
