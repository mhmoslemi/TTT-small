import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    best_c5 = current_c5

    initial_temp = 0.1
    cooling_rate = 0.999
    mutation_scale = 0.1
    variance_falloff = 0.99
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate perturbation
        perturbation = np.random.randn(n_points) * (mutation_scale * temp)
        candidate = h + perturbation
        candidate = np.clip(candidate, 0, 1)

        candidate_sum = candidate.sum()

            candidate *= (n_points / 2) / candidate_sum
        else:
            candidate = h.copy()
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate, n_points)
        
        # Acceptance criterion
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):

            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
