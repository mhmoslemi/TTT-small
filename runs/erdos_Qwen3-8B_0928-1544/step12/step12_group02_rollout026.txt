import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        swarm_size = 200
        particles = [initial_h_scaled + np.random.rand(n_points) * 0.05 for _ in range(swarm_size)]
        velocities = [np.random.rand(n_points) * 0.05 for _ in range(swarm_size)]
    else:
        swarm_size = 200
        particles = [np.random.rand(n_points) for _ in range(swarm_size)]
        velocities = [np.random.rand(n_points) * 0.05 for _ in range(swarm_size)]
        for i in range(swarm_size):
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
    
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1
    personal_best = [p.copy() for p in particles]
    global_best = particles[0].copy()
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(corr) * dx
            corr_pbest = np.correlate(personal_best[i], 1 - personal_best[i], mode='full')
            pbest_c5 = np.max(corr_pbest) * dx
            if current_c5 < pbest_c5:
                personal_best[i] = particles[i].copy()
                if current_c5 < best_c5:
                    global_best = particles[i].copy()
                    best_c5 = current_c5
        w = max(0.4, w - 0.001)
    
    return global_best, best_c5, n_points
