        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        swarm = [initial_h_scaled.copy()]
        for _ in range(swarm_size - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            swarm.append(h_scaled)
    else:
        swarm = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            swarm.append(h_scaled)
    
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(swarm_size)]
    pbest = [np.copy(s) for s in swarm]
    gbest = np.copy(swarm[0])
    best_c5 = np.max(np.correlate(gbest, 1 - gbest, mode='full')) * (2.0 / n_points)
    
    w = w_start
    
    while time.time() - time_start < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = w * velocities[i] + c1 * r1 * (pbest[i] - swarm[i]) + c2 * r2 * (gbest - swarm[i])
            vel = np.clip(vel, -0.1, 0.1)
            velocities[i] = vel
            swarm[i] += vel
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm_sum = np.sum(swarm[i])
            if swarm_sum == 0:
                swarm[i] = np.random.rand(n_points)
                swarm_sum = np.sum(swarm[i])
                swarm[i] = swarm[i] * (ones_count / swarm_sum)
            else:
                swarm[i] = swarm[i] * (ones_count / swarm_sum)
            corr = np.correlate(swarm[i], 1 - swarm[i], mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < np.max(np.correlate(pbest[i], 1 - pbest[i], mode='full')) * (2.0 / n_points):
                pbest[i] = np.copy(swarm[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                gbest = np.copy(swarm[i])
        w = w_start - (w_start - w_end) * (time.time() - time_start) / budget_s
        if time.time() - time_start >= budget_s:
            break
    
    return gbest, best_c5, n_points
