def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    population_size = 100
    mutation_rate = 0.1

            population.append(h)
    else:
        population = []

            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0].copy()

    start_time = time.time()
