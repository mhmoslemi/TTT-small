        if improvement_count > 100:
            break
        if np.max(np.abs(grad)) < early_stop_threshold:
            break
        grad_norm = np.linalg.norm(grad)
        directions = np.zeros(n_points)
        directions[np.abs(grad) > 0] = -grad[np.abs(grad) > 0] / grad_norm
        for _ in range(population_size):
            perturbed = best_h + np.random.randn(n_points) * mutation_scale
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            perturbed = perturbed * (target_sum / perturbed_sum)
            c5_pert = compute_c5_and_gradient(perturbed)[0]
            if c5_pert < c5:
                best_h = perturbed
                c5 = c5_pert
        direction = directions
        step = learning_rate * direction
        new_h = best_h + step
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (target_sum / new_h_sum)
        new_c5 = compute_c5_and_gradient(new_h)[0]
        if new_c5 < c5:
            best_h = new_h
            best_c5 = new_c5
            learning_rate *= learning_rate_decay
        else:
            learning_rate *= 0.9

    return best_h, best_c5, n_points
