                h[i] += delta
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum == 0:
                    continue
                h = h * (n_half / h_sum)
            c_mut = compute_c5(h)
            if c_mut < best_c5:
                best_c5 = c_mut
                best_h = h.copy()

    return best_h, best_c5, n_points
