import time

    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0

    if 'initial_h_values' in globals():

        vel = [np.random.rand(n_points) * vel_limit for _ in range(swarm_size)]
        p_best = pos[:]
        g_best = min(p_best, key=lambda h: objective(h))
    else:
        pos = [np.random.rand(n_points) for _ in range(swarm_size)]
        vel = [np.random.rand(n_points) * vel_limit for _ in range(swarm_size)]
        p_best = pos[:]
        g_best = min(p_best, key=lambda h: objective(h))
        for i in range(swarm_size):
            pos[i] *= (ones_count / np.sum(pos[i]))
            p_best[i] = pos[i].copy()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = objective(g_best)
    best_h = g_best.copy()
    start_time = time.time()
    w = w_start

            pos[i] = pos[i] + vel[i]
            pos[i] = np.clip(pos[i], 0, 1)
            pos_sum = np.sum(pos[i])
