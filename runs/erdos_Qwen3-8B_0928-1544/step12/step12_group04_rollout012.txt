import numpy as np
import time
from numpy.random import rand, randn, choice, uniform

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    pop_size = 200
    if 'initial_h_values' in globals():

        fitness = np.array([obj(h) for h in pop])
        idxs = np.argsort(fitness)
