                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > stagnation_threshold:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

    # Validate best solution
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h_sum = np.sum(final_h)
    if final_h_sum == 0:
        final_h = np.random.rand(n_points) * 0.5
        final_h = final_h * (ones_count / np.sum(final_h))
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h_sum = np.sum(final_h)
    if final_h_sum != 0:
        final_h = final_h * (ones_count / final_h_sum)
    final_c5 = objective(final_h)
    return final_h, final_c5, n_points
