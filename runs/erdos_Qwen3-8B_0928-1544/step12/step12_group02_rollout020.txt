import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    velocity_limit = 0.1
    time_start = time.time()
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        particles = [initial_h_scaled.copy()]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            particles.append(h_scaled.copy())
    else:
        particles = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            particles.append(h_scaled.copy())
    
    velocities = [np.random.rand(n_points) * velocity_limit for _ in range(pop_size)]
    personal_best = [p.copy() for p in particles]
    global_best = particles[0].copy()
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * (2.0 / n_points)
    
    while time.time() - time_start < budget_s:
        w = w_start - (w_start - w_end) * (time.time() - time_start) / budget_s
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            vel = np.clip(vel, -velocity_limit, velocity_limit)
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            if current_sum == 0:
                continue
            particles[i] = particles[i] * (ones_count / current_sum)
            curr_c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * (2.0 / n_points)
            pbest_c5 = np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full')) * (2.0 / n_points)
            if curr_c5 < pbest_c5:
                personal_best[i] = particles[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    global_best = particles[i].copy()
    
    return global_best, best_c5, n_points
