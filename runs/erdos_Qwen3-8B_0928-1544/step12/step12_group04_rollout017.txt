            new_val = obj(new_pos)
            if new_val < pbest_vals[i]:
                pbest[i] = new_pos.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest = new_pos.copy()
                    gbest_val = new_val
                    best_h = new_pos.copy()
            pop[i] = new_pos
        if time.time() - start > budget_s - 1:
            break
    return best_h, gbest_val, n_points
