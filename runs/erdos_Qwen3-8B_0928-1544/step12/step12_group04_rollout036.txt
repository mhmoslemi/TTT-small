def compute_c5_fft(h):
    # Compute the cross-correlation using FFT
    n = len(h)
    # pad the sequences to length 2n -1
    h_fft = np.fft.fft(h, 2*n -1)
    one_minus_h = 1 - h
    one_minus_h_fft = np.fft.fft(one_minus_h, 2*n -1)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    # the maximum value in the cross-correlation
