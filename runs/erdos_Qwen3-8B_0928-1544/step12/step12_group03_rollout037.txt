import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        original_len = len(initial_h)
        original_x = np.linspace(0, 2, original_len)
        new_x = np.linspace(0, 2, n_points)
        h_upsampled = np.interp(new_x, original_x, initial_h)
        sum_h = np.sum(h_upsampled)
        h_upsampled = h_upsampled * (ones_count / sum_h)
        population = [h_upsampled]
        for _ in range(49):
            h = np.random.rand(n_points) * 0.5
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            else:
                h = h * (ones_count / sum_h)
            population.append(h)
    else:
        population = []
        for _ in range(50):
            h = np.random.rand(n_points) * 0.5
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            else:
                h = h * (ones_count / sum_h)
            population.append(h)
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    velocities = np.random.rand(len(population), n_points) * 0.1
    pbest = [h.copy() for h in population]
    pbest_c5 = [objective(h) for h in population]
    gbest = best_h.copy()
    gbest_c5 = best_c5
    w = 0.9
    stagnation = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + 2.0 * r1 * (pbest[i] - population[i]) + 2.0 * r2 * (gbest - population[i])
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0.0, 1.0)
            sum_h = np.sum(population[i])
            if sum_h == 0:
                population[i] = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            else:
                population[i] *= (ones_count / sum_h)
            new_c5 = objective(population[i])
            if new_c5 < pbest_c5[i]:
                pbest[i] = population[i].copy()
                pbest_c5[i] = new_c5
                if new_c5 < gbest_c5:
                    gbest = population[i].copy()
                    gbest_c5 = new_c5
                    best_h = gbest.copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > 100:
            w = max(w - 0.005, 0.3)
            stagnation = 0
        if time.time() - start_time >= budget_s:
            break
    return best_h, gbest_c5, n_points
