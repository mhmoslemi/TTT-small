import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    velocity_max = 0.5
    c1 = 2.0
    c2 = 2.0
    inertia_start = 0.9
    inertia_end = 0.4
    obj_func = lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled.copy()]
        for _ in range(pop_size - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.05
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)
    
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(pop_size)]
    personal_bests = [h.copy() for h in pop]
    global_best = pop[0].copy()
    best_c5 = obj_func(pop[0])
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        current_time = time.time()
        inertia = inertia_start - (inertia_start - inertia_end) * (current_time - start_time) / budget_s
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] + c1 * r1 * (personal_bests[i] - pop[i]) + c2 * r2 * (global_best - pop[i])
            v = np.clip(v, -velocity_max, velocity_max)
            new_pop = pop[i] + v
            new_pop = np.clip(new_pop, 0.0, 1.0)
            new_pop_sum = np.sum(new_pop)
            if new_pop_sum == 0:
                continue
            new_pop = new_pop * (ones_count / new_pop_sum)
            curr_c5 = obj_func(pop[i])
            new_c5 = obj_func(new_pop)
            if new_c5 < curr_c5:
                pop[i] = new_pop
                velocities[i] = v
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    global_best = pop[i].copy()
    
    return global_best, best_c5, n_points
