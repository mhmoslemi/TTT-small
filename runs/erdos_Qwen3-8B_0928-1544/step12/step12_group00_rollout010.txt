                stagnation += 1
        if stagnation > 50:
            w = max(0.5, w - 0.01)
            max_v += 0.01
            for i in range(population_size // 10):
                population[i] = np.random.rand(n_points) * 0.5
                h_sum = np.sum(population[i])
                if h_sum == 0:
                    population[i] = np.random.rand(n_points) * 0.5
                population[i] = population[i] * (T / np.sum(population[i]))
                corr = np.correlate(population[i], 1 - population[i], mode='full')
                current_c5 = np.max(corr) * (2.0 / n_points)
                if current_c5 < best_c5:
                    global_best = population[i].copy()
                    best_c5 = current_c5
            stagnation = 0
    global_best = np.clip(global_best, 0, 1)
    global_best_sum = np.sum(global_best)
    if global_best_sum != 0:
        global_best = global_best * (T / global_best_sum)
    return global_best, best_c5, n_points
