import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    pop_size = 200
    if 'initial_h_values' in globals():
        initial = initial_h_values
        initial_sum = np.sum(initial)

        h_initial = initial * scale
        pop = [h_initial]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    best_h = pop[0].copy()
    best_val = obj(best_h)
    start = time.time()
    
    while time.time() - start < budget_s:
        fitness = [obj(h) for h in pop]
        sorted_indices = np.argsort(fitness)
        best_idx = sorted_indices[0]
        new_pop = [pop[best_idx].copy()]
        for _ in range(pop_size - 1):
            t1 = np.random.choice(len(pop), 10, replace=False)

                m = np.random.normal(0, 0.1, n_points)

        current_best = np.argmin(new_fitness)
