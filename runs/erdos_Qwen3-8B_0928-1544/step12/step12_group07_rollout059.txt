import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        h_initial = h_initial * (ones_count / np.sum(h_initial))
    else:
        h_initial = np.random.rand(n_points) * 0.5
        h_initial_sum = np.sum(h_initial)
        if h_initial_sum == 0:
            h_initial = np.random.rand(n_points) * 0.5
        h_initial = h_initial * (ones_count / h_initial_sum)
    
    population_size = 60
    population = [h_initial.copy() for _ in range(population_size)]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_correlation = np.max(corr) * (2.0 / n_points)
        return max_correlation
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(h_initial)
    
    for _ in range(population_size):
        noise = np.random.normal(0, 0.05, n_points)
        individual = h_initial + noise
        individual = np.clip(individual, 0.0, 1.0)
        individual_sum = np.sum(individual)
        if individual_sum == 0:
            individual = np.random.rand(n_points) * 0.5
        individual = individual * (ones_count / individual_sum)
        population[_] = individual
    
    for h in population:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    start_time = time.time()
    max_generations = 1500
    
    while time.time() - start_time < budget_s and max_generations > 0:
        fitness = [compute_c5(h) for h in population]

        selected = [population[i] for i in sorted_indices[:population_size//2]]
        new_population = []
        
        for _ in range(population_size//2):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            crossover_point = np.random.randint(1, n_points)

            child = np.clip(child, 0.0, 1.0)
            child_sum = np.sum(child)
            if child_sum == 0:

            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        max_generations -= 1
