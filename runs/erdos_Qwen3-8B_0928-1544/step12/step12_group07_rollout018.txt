import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    target_sum = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = initial_h * (target_sum / np.sum(initial_h))
        population = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(60)]
        population = [np.clip(p, 0, 1) * (target_sum / np.sum(np.clip(p, 0, 1))) for p in population]
    else:
        population = [np.random.dirichlet([1.0]*n_points) * target_sum for _ in range(60)]

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_correlation = np.max(corr)
        return max_correlation * dx

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    gen = 0

    while (time.time() - start_time) < budget_s:
        fitness = [compute_c5(p) for p in population]
        idx = np.argsort(fitness)
        selected = [population[i] for i in idx[:int(len(population)*0.5)]]

        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 >= len(selected):
                break
            a, b = selected[i], selected[i+1]
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((a[:split], b[split:]))
            child2 = np.concatenate((b[:split], a[split:]))
            offspring.append(child1)
            offspring.append(child2)

        mutated = []
        for h in offspring:
            h_mut = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.02:
                    h_mut[i] += np.random.uniform(-0.1, 0.1)
            h_mut = np.clip(h_mut, 0, 1)
            current_sum = np.sum(h_mut)
            if current_sum != 0:
                h_mut *= target_sum / current_sum
            h_mut = np.clip(h_mut, 0, 1)
            mutated.append(h_mut)

        new_pop = selected[:int(len(selected)*0.5)] + mutated
