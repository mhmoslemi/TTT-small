import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    velocities = [np.random.rand(n_points) * 0.1 for _ in pop]
    pbest = [h.copy() for h in pop]
    gbest = pbest[0].copy()
    global_best_c5 = float('inf')
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    for h in pbest:
        current_c5 = objective(h)
        if current_c5 < global_best_c5:
            gbest = h.copy()
            global_best_c5 = current_c5
    best_h = gbest.copy()
    start_time = time.time()
    c1, c2 = 2.0, 2.0
    w_start, w_end = 0.9, 0.4
    max_velocity = 0.1
    while time.time() - start_time < budget_s:
        inertia_weight = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
        for i in range(pop_size):
            h = pop[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            vel_new = inertia_weight * vel + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            vel_new = np.clip(vel_new, -max_velocity, max_velocity)
            h_new = h + vel_new
            h_new = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_new)
            if h_sum == 0:
                continue
            h_scaled = h_new * (ones_count / h_sum)
            new_c5 = objective(h_scaled)
            if new_c5 < objective(pbest[i]):
                pop[i] = h_scaled.copy()
                velocities[i] = vel_new.copy()
                pbest[i] = h_scaled.copy()
                if new_c5 < global_best_c5:
                    gbest = h_scaled.copy()
                    global_best_c5 = new_c5
                    best_h = h_scaled.copy()
    return best_h, global_best_c5, n_points
