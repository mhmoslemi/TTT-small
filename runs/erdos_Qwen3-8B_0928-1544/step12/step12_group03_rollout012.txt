import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    ones_count = n_points // 2
    max_iter = 500
    swarm_size = 100
    w_start, w_end = 0.9, 0.4
    phi_p, phi_g = 2.0, 2.0
    vel_limit = 0.1
    stagnation_threshold = 50

    # Initialize population
    if 'initial_h_values' in globals():
        m = len(initial_h_values)
        x_initial = np.linspace(0, 2, m)
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h_values)
        population = [h_initial.copy()]
        for _ in range(swarm_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    velocities = np.random.rand(swarm_size, n_points) * vel_limit
    pbest = [h.copy() for h in population]
    pbest_c5 = [objective(h) for h in population]
    gbest = best_h.copy()
    gbest_c5 = best_c5
    w = w_start
    stagnation = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and max_iter > 0:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + phi_p * r1 * (pbest[i] - population[i]) + phi_g * r2 * (gbest - population[i])
            velocities[i] = np.clip(velocities[i], -vel_limit, vel_limit)
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0.0, 1.0)
            sum_pop = np.sum(population[i])
            if sum_pop == 0:
                population[i] = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            else:
                population[i] *= (ones_count / sum_pop)
            current_c5 = pbest_c5[i]
            new_c5 = objective(population[i])
            if new_c5 < current_c5:
                pbest[i] = population[i].copy()
                pbest_c5[i] = new_c5
                if new_c5 < gbest_c5:
                    gbest = population[i].copy()
                    gbest_c5 = new_c5
                    best_h = gbest.copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > stagnation_threshold:
            w = max(w_start, w - 0.01)
            stagnation = 0
        max_iter -= 1
    return best_h, gbest_c5, n_points
