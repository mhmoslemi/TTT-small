import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    n_particles = 200
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    vel_clip = 0.5
    max_iter = 1000

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(n_particles - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(n_particles):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    vel = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    pbest = pop.copy()
    gbest = min(pop, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * (2.0 / n_points))
    best_c5 = np.max(np.correlate(gbest, 1 - gbest, mode='full')) * (2.0 / n_points)

    start_time = time.time()
    inertia = inertia_start

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

            vel[i] = np.clip(vel[i], -vel_clip, vel_clip)
            pop[i] = pop[i] + vel[i]
            pop[i] = np.clip(pop[i], 0, 1)
            pop_sum = np.sum(pop[i])
