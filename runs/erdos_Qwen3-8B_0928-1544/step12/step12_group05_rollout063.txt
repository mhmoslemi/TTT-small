            C5_val = 1.0
        else:
            s_val = s.value
            C5_val = C5.value
    except:
        s_val = s_initial
        C5_val = 1.0

    h = (s_val[1:] - s_val[:-1]) / dx
    h_clamped = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h_clamped)
    if h_sum == 0:
        h_scaled = np.zeros(n_points)
    else:
        h_scaled = h_clamped * (ones_count / h_sum)

    def objective(x):
        h_clamped = np.clip(x, 0.0, 1.0)
        h_sum = np.sum(h_clamped)
        if h_sum == 0:
            return float('inf')
        h_scaled = h_clamped * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    result = minimize(objective, h_scaled, method='Nelder-Mead', tol=1e-6, options={'maxiter': 100})
    best_h = np.clip(result.x, 0.0, 1.0)
    h_sum = np.sum(best_h)
    if h_sum == 0:
        best_h_scaled = np.zeros(n_points)
    else:
        best_h_scaled = best_h * (ones_count / h_sum)
    corr = np.correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)
    return best_h_scaled, c5_bound, n_points
