            candidate_h_clipped = np.clip(candidate_h, 0.0, 1.0)
            sum_clipped = np.sum(candidate_h_clipped)
            if sum_clipped == 0:
                candidate_h_scaled = candidate_h_clipped.copy()
            else:
                candidate_h_scaled = candidate_h_clipped * (50.0 / sum_clipped)
            current_c5 = compute_c5(candidate_h_scaled)
            if current_c5 < best_c5:
                best_h = candidate_h_scaled.copy()
                best_c5 = current_c5
            stagnation = 0

    return best_h, best_c5, n_points
