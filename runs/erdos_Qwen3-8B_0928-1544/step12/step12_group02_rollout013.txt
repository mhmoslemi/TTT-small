import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    c1, c2 = 2.0, 2.0
    velocity_limit = 0.1
    inertia_start = 0.9
    inertia_end = 0.4
    reduction_rate = 0.001

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)
    else:
        pop = [np.random.rand(n_points) for _ in range(pop_size)]
        for i in range(pop_size):
            h = pop[i]
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop[i] = h_scaled

    velocities = [np.random.rand(n_points) * 0.1 for _ in range(pop_size)]
    personal_best = [h.copy() for h in pop]
    global_best = min(pop, key=objective)
    best_c5 = objective(global_best)

    start_time = time.time()
    inertia = inertia_start

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * inertia + c1 * r1 * (personal_best[i] - pop[i]) + c2 * r2 * (global_best - pop[i])
            vel = np.clip(vel, -velocity_limit, velocity_limit)
            velocities[i] = vel
            pop[i] = pop[i] + vel
            pop[i] = np.clip(pop[i], 0, 1)
            pop_sum = np.sum(pop[i])
            if pop_sum == 0:
                pop[i] = np.random.rand(n_points)
