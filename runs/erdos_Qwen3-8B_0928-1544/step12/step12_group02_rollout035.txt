import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    w_start, w_end = 0.8, 0.4
    c1, c2 = 2.0, 2.0
    vel_clip = 0.1

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)
    
    velocities = np.random.rand(pop_size, n_points) * vel_clip
    pbest = [h.copy() for h in pop]
    best_c5 = np.max(np.correlate(pbest[0], 1 - pbest[0], mode='full')) * (2.0 / n_points)
    best_h = pbest[0].copy()
    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            r1, r2 = np.random.rand(), np.random.rand()
            w = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - pop[i]) + c2 * r2 * (best_h - pop[i])
            velocities[i] = np.clip(velocities[i], -vel_clip, vel_clip)
            pop[i] += velocities[i]
            pop[i] = np.clip(pop[i], 0, 1)
            sum_h = np.sum(pop[i])
            if sum_h == 0:
                continue
            pop[i] *= ones_count / sum_h
            new_corr = np.correlate(pop[i], 1 - pop[i], mode='full')
            new_c5 = np.max(new_corr) * (2.0 / n_points)
            if new_c5 < np.max(np.correlate(pbest[i], 1 - pbest[i], mode='full')) * (2.0 / n_points):
                pbest[i] = pop[i].copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = pop[i].copy()
                stagnation = 0
            else:
                stagnation += 1
        if stagnation > 500:
            break

    return best_h, best_c5, n_points
