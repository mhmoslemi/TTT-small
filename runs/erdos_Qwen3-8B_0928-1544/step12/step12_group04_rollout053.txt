import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    pop_size = 200
    elite_ratio = 0.2  # Keep 20% of population as elites
    tournament_size = 10
    crossover_prob = 0.8
    mutation_prob = 0.2

    # Initialize population
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h_initial = initial * (required_sum / np.sum(initial))
        pop = [h_initial]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def evaluate_population(pop):
        return [obj(h) for h in pop]

    best_h = pop[0].copy()
    best_val = obj(pop[0])
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fit = evaluate_population(pop)
        indices = np.argsort(fit)
        elite_count = int(pop_size * elite_ratio)
        elites = [pop[i].copy() for i in indices[:elite_count]]

        new_pop = []
        for _ in range(pop_size - elite_count):
            # Tournament selection
            tournament = np.random.choice(pop, tournament_size)
            winners = [tournament[i] for i in np.argsort([obj(ind) for ind in tournament])]
            parent1, parent2 = winners[0], winners[1]

            # Crossover
            if np.random.rand() < crossover_prob:
                cross_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            else:
                child = parent1.copy()

            # Mutation
            if np.random.rand() < mutation_prob:
                sigma = 0.1
                mutation = np.random.normal(0, sigma, n_points)
                child = np.clip(child + mutation, 0.0, 1.0)
                child_sum = np.sum(child)
                if child_sum != 0:
                    child = child * (required_sum / child_sum)
                else:
                    child = parent1.copy()

            new_pop.append(child)

        # Combine elites and new_pop
        combined = elites + new_pop
        np.random.shuffle(combined)
        pop = combined[:pop_size]

        # Update best solution
        current_best = min(fit)
        if current_best < best_val:
            best_val = current_best
            best_h = pop[np.argmin(fit)]

        # Early exit if no improvement
        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_val, n_points
