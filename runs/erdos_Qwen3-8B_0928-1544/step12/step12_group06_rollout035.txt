import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    budget = budget_s
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    if 'initial_h_values' in globals():

    for i in range(len(h_pop)):
        h_pop[i] = h_pop[i] * (n_points / 2) / np.sum(h_pop[i])

        return np.max(corr) * dx

    best_idx = np.argmin(fitness)
