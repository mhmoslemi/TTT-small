import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    population_size = 50
    inertia = 0.7
    c1, c2 = 1.5, 1.5
    max_velocity = 0.1
    min_velocity = -max_velocity

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum)
    population = [initial_h_scaled.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        h = initial_h_scaled + perturbation
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled.copy())

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_bests = [p.copy() for p in population]
    global_best = best_h
    stagnation = 0
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] + inertia * velocities[i] + c1 * r1 * (personal_bests[i] - population[i]) + c2 * r2 * (global_best - population[i])
            vel = np.clip(vel, min_velocity, max_velocity)
            velocities[i] = vel.copy()
            population[i] += vel
            population[i] = np.clip(population[i], 0, 1)
            h_sum = np.sum(population[i])
            if h_sum == 0:
                population[i] = np.random.rand(n_points) * 0.5
                h_sum = np.sum(population[i])
            population[i] *= (ones_count / h_sum)
            curr_c5 = objective(population[i])
            if curr_c5 < objective(personal_bests[i]):
                personal_bests[i] = population[i].copy()
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = population[i].copy()
                global_best = best_h
            if curr_c5 >= best_c5:
                stagnation += 1
            else:
                stagnation = 0
            if stagnation > 100:
                inertia = max(0.4, inertia - 0.01)
                stagnation = 0
    return best_h, best_c5, n_points
