            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0]
    best_c5 = objective(pop[0])
    start_time = time.time()
    F, CR = F_start, CR_start
    stagnation = 0

    # DE Phase
    while time.time() - start_time < 0.7 * budget_s:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i] * (1 - CR) + mutant * CR
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial_scaled = trial * (ones_count / trial_sum)
            curr_c5 = objective(pop[i])
            new_c5 = objective(trial_scaled)
            if new_c5 < curr_c5:
                pop[i] = trial_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = trial_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 200:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

    # SQP Phase
    def objective_sqpeq(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def gradient_sqpeq(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        k_idx = max_corr_index - (n_points - 1)
        grad = np.zeros_like(h)
        dx = 2.0 / n_points
        for j in range(n_points):
            if 0 <= j + k_idx < n_points:
                grad[j] += (1 - h[j + k_idx]) * dx
            x = j - k_idx
            if 0 <= x < n_points:
                grad[j] -= h[x] * dx
        return grad

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}

    result = minimize(objective_sqpeq, best_h, method='SLSQP', jac=gradient_sqpeq, bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h = result.x if result.success else best_h
    best_c5 = result.fun if result.success else best_c5
    return best_h, best_c5, n_points
