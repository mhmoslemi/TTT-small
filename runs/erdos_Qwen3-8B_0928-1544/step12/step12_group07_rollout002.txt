                child[mutate_indices] += np.random.uniform(-0.1, 0.1, size=len(mutate_indices))
                child = np.clip(child, 0.0, 1.0)
                child = repair_h(child)
            offspring.append(child)
        population = selected + offspring
        for h in population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    return best_h, best_c5, n_points
