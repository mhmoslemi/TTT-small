def project(h, n_points, c):
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0.0, 1.0)
    total = np.sum(h_clipped)
    if total == 0:
        return np.zeros_like(h)
    if total < c:
        deficit = c - total
        # Identify indices where we can increase values
        non_one = np.where(h_clipped < 1.0)[0]
        if len(non_one) == 0:
            return h_clipped  # all are 1, can't increase
        # Compute how much can be added
        max_add = np.sum(1.0 - h_clipped[non_one])
        if max_add <= deficit:
            # fill to 1
            h_clipped[non_one] = 1.0
            deficit -= max_add
        else:
            # distribute deficit proportionally
            weights = 1.0 - h_clipped[non_one]
            delta = (deficit * weights) / (max_add)
            h_clipped[non_one] += delta
    elif total > c:
        surplus = total - c
        non_zero = np.where(h_clipped > 0.0)[0]
        if len(non_zero) == 0:
            return h_clipped
        max_reduce = np.sum(h_clipped[non_zero])
        if max_reduce <= surplus:
            h_clipped[non_zero] = 0.0
        else:
            weights = h_clipped[non_zero]
            delta = (surplus * weights) / (max_reduce)
            h_clipped[non_zero] -= delta
    return h_clipped
