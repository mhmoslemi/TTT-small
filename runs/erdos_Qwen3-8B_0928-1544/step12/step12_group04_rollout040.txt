import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    pop_size = 300
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h0 = initial * (required_sum / np.sum(initial))
        pop = [h0]
        for _ in range(pop_size - 1):
            h = initial + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = initial.copy()
            else:
                h *= required_sum / h_sum
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h *= required_sum / h_sum
            pop.append(h)
    velocities = np.zeros((pop_size, n_points))
    pbest = [h.copy() for h in pop]
    pbest_vals = [np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points) for h in pop]
    gbest = pop[0].copy()
    gbest_val = pbest_vals[0]
    start = time.time()
    w, phi_p, phi_g = 0.7, 2.0, 2.0
    max_vel = 0.2
    while time.time() - start < budget_s - 1:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = phi_p * r1 * (pbest[i] - pop[i]) + phi_g * r2 * (gbest - pop[i])
            velocities[i] = w * velocities[i] + vel_update
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            new_pos = pop[i] + velocities[i]
            for j in range(n_points):
                if new_pos[j] < 0:
                    velocities[i][j] *= -1
                    new_pos[j] = 0
                elif new_pos[j] > 1:
                    velocities[i][j] *= -1
                    new_pos[j] = 1
            new_sum = np.sum(new_pos)
            if new_sum == 0:
                new_pos = pop[i].copy()
            else:
                new_pos *= required_sum / new_sum
            new_val = np.max(np.correlate(new_pos, 1 - new_pos, mode='full')) * (2.0 / n_points)
            if new_val < pbest_vals[i]:
                pbest[i] = new_pos.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest = new_pos.copy()
                    gbest_val = new_val
            pop[i] = new_pos
    return gbest, gbest_val, n_points
