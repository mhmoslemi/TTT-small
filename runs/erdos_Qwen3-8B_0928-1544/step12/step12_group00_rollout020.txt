        for i in range(n_points):
            if i + lag < n_points:
                grad[i] += 1 - h[i + lag]
            if i - lag >= 0:
                grad[i] -= h[i - lag]
        ones = np.ones(n_points)
        alpha = np.dot(grad, ones) / np.dot(ones, ones)
        grad_proj = grad - alpha * ones
        step = 0.01
        h_test = h - step * grad_proj
        h_test = np.clip(h_test, 0.0, 1.0)
        h_test_sum = np.sum(h_test)
        if h_test_sum == 0:
            h_test = h.copy()
        else:
            h_test = h_test * (ones_count / h_test_sum)
        new_c5 = np.max(np.correlate(h_test, 1 - h_test, mode='full')) * (2.0 / n_points)
        if new_c5 < best_c5:
            h = h_test.copy()
            best_c5 = new_c5
            best_h = h.copy()
        elapsed = time.time() - start_time
    
    return best_h, best_c5, n_points
