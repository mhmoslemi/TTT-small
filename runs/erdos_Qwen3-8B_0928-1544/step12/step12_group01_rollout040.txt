import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

        h_sum = h.sum()
        if h_sum != n_points / 2:

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    temp = 0.1
    cooling_rate = 0.999
    mutation_scale = 0.1
    variance_falloff = 0.9
    start_time = time.time()
    iteration = 0
    max_iterations = 100000
    while iteration < max_iterations and time.time() - start_time < budget_s:
        current_c5 = evaluate_erdos_solution(h, n_points)
        candidate = h + np.random.randn(n_points) * (mutation_scale * temp)
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = candidate.sum()
        if candidate_sum > 0:
            candidate *= (n_points / 2) / candidate_sum
        else:
            h_rand = np.random.rand(n_points)
            h_rand_sum = h_rand.sum()
            h_rand *= (n_points / 2) / h_rand_sum
            candidate = h_rand
            candidate = np.clip(candidate, 0, 1)
        candidate_c5 = evaluate_erdos_solution(candidate, n_points)
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
        temp *= cooling_rate
        mutation_scale *= variance_falloff
        iteration += 1
