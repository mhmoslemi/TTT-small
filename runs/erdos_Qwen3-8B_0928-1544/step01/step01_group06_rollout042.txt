        cross_corr = np.correlate(h_values, 1 - h_values, mode='full')
        max_corr_idx = np.argmax(cross_corr)
        s = max_corr_idx - (n_points - 1)

        grad = np.zeros_like(h_values)
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                grad[i] = 1 - h_values[j]

        h_values -= learning_rate * grad
        h_values = np.clip(h_values, 0, 1)
        current_sum = np.sum(h_values)
        if current_sum != 0:
            scale = target_sum / current_sum
            h_values *= scale

        current_c5 = compute_C5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 500:
                learning_rate *= 0.5
                no_improvement = 0

        if steps % 500 == 0:
            noise = np.random.normal(0, 1e-3, n_points)
            h_values += noise
            h_values = np.clip(h_values, 0, 1)
            current_sum = np.sum(h_values)
            scale = target_sum / current_sum
            h_values *= scale
        steps += 1

    return (best_h, best_c5, n_points)
