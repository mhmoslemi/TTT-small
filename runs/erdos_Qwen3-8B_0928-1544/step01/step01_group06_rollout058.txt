    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h *= scale

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.1
    steps = 0
    no_improvement = 0
    start_time = time.time()

    while steps < 10000 and time.time() - start_time < budget_s:
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        index = np.argmax(cross_corr)
        k0 = index - (n_points - 1)

        valid_indices = k0 + np.arange(n_points)
        valid_mask = (valid_indices >= 0) & (valid_indices < n_points)
        grad = np.zeros_like(h)
        grad[valid_mask] = 1.0 - h[valid_indices[valid_mask]]

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
        h *= scale

        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 500:
                learning_rate *= 0.5
                no_improvement = 0

        if steps % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
            h *= scale

        steps += 1

    return (best_h, best_c5, n_points)
