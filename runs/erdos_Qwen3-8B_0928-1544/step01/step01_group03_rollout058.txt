        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Fallback to initial guess if time is up
        from evaluate_erdos_solution import initial_h_values
        h_initial = initial_h_values
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        c5_initial = np.max(corr) * dx
        return (h_initial, c5_initial, n_points)
    
    h_optimized = result.x
    c5_bound = result.fun
    
    return (h_optimized, c5_bound, n_points)
