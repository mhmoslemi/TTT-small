    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        h_initial = np.clip(h_initial, 0, 1)
        h_sum = np.sum(h_initial)
        h_initial = h_initial * (n_points / 2) / h_sum
        population = [h_initial]
    else:
        population = [np.clip(np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points)), 0, 1) for _ in range(pop_size)]
    
    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    best_c5 = float('inf')
    best_h = population[0]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        scores = [(compute_c5(h), h) for h in population]
        scores.sort()
        num_parents = int(len(population) * 0.2)
        parents = [h for c5, h in scores[:num_parents]]
        
        new_population = []
        best_c5_val, best_h_candidate = scores[0]
        if best_c5_val < best_c5:
            best_c5 = best_c5_val
            best_h = best_h_candidate
        
        for _ in range(pop_size - 1):
            p1, p2 = parents[np.random.randint(len(parents))], parents[np.random.randint(len(parents))]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            mutate = np.random.rand(n_points) < 0.1
            child[mutate] += np.random.uniform(-0.1, 0.1)
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            child = child * (n_points / 2) / child_sum
            new_population.append(child)
        new_population.append(best_h_candidate)
        population = new_population[:pop_size]
    
    return (best_h, best_c5, n_points)
