import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    assert np.isclose(sum_h, n_points / 2), "Initial h must have sum = n_points / 2"
    
    def perturb(h_in, p_amt=0.1, n_pert=10):
        h_pert = h_in.copy()
        idxs = np.random.choice(len(h_pert), size=n_pert, replace=False)
        for i in idxs:
            h_pert[i] += np.random.uniform(-p_amt, p_amt)
            h_pert[i] = np.clip(h_pert[i], 0., 1.)
        scale = (n_points / 2) / np.sum(h_pert)
        h_pert *= scale
        return h_pert
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr * dx)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_c5 = 1.0
    best_h = h.copy()
    start = time.time()
    
    for _ in range(20):
        pert = perturb(h)
        result = minimize(objective, pert, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 250})
        if result.success:
            curr_c5 = result.fun
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = result.x.copy()
        if time.time() - start > budget_s - 150:
            break
    
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    c5_val = np.max(corr * dx)
    return best_h, c5_val, n_points
