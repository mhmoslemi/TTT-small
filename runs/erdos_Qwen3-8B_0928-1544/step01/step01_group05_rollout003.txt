import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h) + 0.5
        scale = (n_points / 2) / total
        return h * scale

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 30
    max_gen = 200
    F = 0.7
    CR = 0.8
    stagnation_gen = 50

    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair(h)
        pop.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    for gen in range(max_gen):
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[c] + F * (pop[a] - pop[b])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = repair(trial)
            trial_c5 = compute_c5(trial)
            if trial_c5 < compute_c5(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        current_best = min(pop, key=compute_c5)
        if compute_c5(current_best) < best_c5:
            best_h = current_best
            best_c5 = compute_c5(best_h)
        if gen >= stagnation_gen and compute_c5(best_h) == best_c5:
            break

    def objective(h):
        return compute_c5(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8})
    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
