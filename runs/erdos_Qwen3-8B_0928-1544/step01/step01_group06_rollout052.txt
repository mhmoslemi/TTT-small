import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    def tournament_selection(population, fitness, tournament_size):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[np.argmin(fitness[indices])]
        return population[best_idx]

    if 'initial_h_values' in globals():

        n_points = len(initial_population[0])
    else:

        initial_population = []
        for _ in range(50):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            initial_population.append(h)

    population = initial_population
    best_h = None
    best_c5 = float('inf')
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.05
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            cross_corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(cross_corr)
            c5 = max_corr * (2.0 / n_points)
            fitness.append(c5)
        
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        new_population = []
        for _ in range(len(population) // 2):
            p1 = tournament_selection(population, fitness, tournament_size)
            p2 = tournament_selection(population, fitness, tournament_size)
            child = (p1 + p2) / 2 if np.random.rand() < crossover_rate else p1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 1e-4, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                child = child * (n_points / 2) / sum_child if sum_child != 0 else child
            new_population.append(child)

        sorted_indices = np.argsort(fitness)
        for i in range(len(new_population)):
            population[sorted_indices[-(i+1)]] = new_population[i]

    return (best_h, best_c5, n_points)
