import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h_initial)
    assert np.isclose(sum_h, n_points / 2), "Initial h must have sum = n_points / 2"
    
    def perturb(h, seed, p=0.1):
        np.random.seed(seed)
        n = len(h)
        idx = np.random.choice(n, int(n * p), replace=False)
        h_perturbed = h.copy()
        for i in idx:
            h_perturbed[i] += np.random.uniform(-0.1, 0.1)
            h_perturbed[i] = np.clip(h_perturbed[i], 0., 1.)
        delta_sum = (n / 2 - np.sum(h_perturbed)) / n
        h_perturbed += delta_sum
        h_perturbed = np.clip(h_perturbed, 0., 1.)
        return h_perturbed
    
    num_starts = 10
    best_result = None
    best_c5 = float('inf')
    start_seeds = [seed + i for i in range(num_starts)]
    start_time = time.time()
    
    for i, seed in enumerate(start_seeds):
        h_perturbed = perturb(h_initial, seed)
        bounds = [(0., 1.) for _ in range(n_points)]
        def obj(x):
            h = x.reshape(n_points)
            c = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(c)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 600})
            if res.success:
                if res.fun < best_c5:
                    best_c5 = res.fun
                    best_result = res.x
        except:
            pass
    
    if best_result is None:
        h = h_initial
        c = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(c)
        return h, c5, n_points
    
    h = best_result
    c = np.correlate(h, 1 - h, mode='full') * dx
    c5 = np.max(c)
    return h, c5, n_points
