        h_clamped = np.clip(h_scaled, 0, 1)
        current_clamped_sum = np.sum(h_clamped)
        delta = target_sum - current_clamped_sum
        h_clamped[-1] += delta
        h_clamped[-1] = np.clip(h_clamped[-1], 0, 1)
        return h_clamped

    population = [initial_h.copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = evaluate_overlap(initial_h)

    start_time = time.time()

    for gen in range(100):
        fitness = [evaluate_overlap(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        if time.time() - start_time > budget_s:
            break

        parents = []
        for _ in range(len(population)//2):
            tournament = random.sample(population, 3)
            winner = min(tournament, key=lambda x: evaluate_overlap(x))
            parents.append(winner)

        new_population = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else parents[i]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            child = repair(child)
            new_population.append(child)
        population = new_population

    return best_h, best_c5, n_points
