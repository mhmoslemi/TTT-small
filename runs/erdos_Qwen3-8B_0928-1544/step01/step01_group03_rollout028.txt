import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        perturbed = np.clip(h_initial + noise, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        scaling = (n_points / 2.0) / sum_perturbed if sum_perturbed != 0 else 1.0
        perturbed *= scaling
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',

            constraints=[{'type': 'eq', 'fun': constraint_sum}],
