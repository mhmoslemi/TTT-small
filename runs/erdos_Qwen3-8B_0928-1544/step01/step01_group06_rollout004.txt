def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 45 if initial_h is None else len(initial_h)
    if initial_h is None:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population = [h.copy() for _ in range(50)]
    else:
        population = [initial_h.copy() for _ in range(50)]
        for i in range(50):
            noise = np.random.normal(0, 1e-3, n_points)
            population[i] += noise
            population[i] = np.clip(population[i], 0, 1)
            population[i] *= (n_points / 2) / np.sum(population[i])
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    def normalize(h):
        return h * (target_sum / np.sum(h))

    best_h = population[0]
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = np.array([compute_C5(h) for h in population])
        idxs = np.argsort(fitness)
        for i in range(50):
            p1, p2 = np.random.choice(50, 2, replace=False)
            child = np.concatenate([population[idxs[p1]][:-5], population[idxs[p2]][-5:]])
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 1e-3, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = normalize(child)
            else:
                child = normalize(child)
            population[i] = child
        for i in range(50):
            if compute_C5(population[i]) < best_c5:
                best_c5 = compute_C5(population[i])
                best_h = population[i]
    return (best_h, best_c5, n_points)
