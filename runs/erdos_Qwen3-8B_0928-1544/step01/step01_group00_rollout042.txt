                child = np.full(n_points, 0.5)
            else:
                child = child * (n_points / 2) / child_sum
            new_population.append(child)
        population = np.vstack([elites, new_population])
        if evaluate_erdos_solution(population[0]) < best_c5:
            best_c5 = evaluate_erdos_solution(population[0])
            best_h = population[0]
    return (best_h, best_c5, n_points)
