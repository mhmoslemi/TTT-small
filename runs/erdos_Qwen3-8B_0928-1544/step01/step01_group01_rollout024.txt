import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values)
    
    def generate_block_guess():
        block_size = n_points // 2
        start = np.random.randint(0, n_points - block_size + 1)
        h = np.zeros(n_points)
        h[start:start+block_size] = 1.0
        return h

    for _ in range(10):
        initial_guesses.append(generate_block_guess())

    best_h = None
    best_c5 = float('inf')

    for guess in initial_guesses:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(corr)

            'type': 'eq',
            'fun': lambda h: np.sum(h) - n_points / 2,
            'jac': lambda h: np.ones_like(h)
        }]
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        current_h = result.x if result.success else guess
        current_c5 = result.fun if result.success else objective(guess)
