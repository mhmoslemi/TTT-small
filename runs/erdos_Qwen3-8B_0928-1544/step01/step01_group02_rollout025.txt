def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if not hasattr(initial_h_values, '__len__') or len(initial_h_values) == 0:
        return np.array([]), 1.0, 0

    def perturb(h, seed):

            h[i] += delta
        total = np.sum(h)
        diff = (n_points / 2) - total
        h[-1] += diff
        h = np.clip(h, 0, 1)
        return h

    num_starts = 10
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    best_n = n_points
    start_time = time.time()
    for start_id in range(num_starts):
        np.random.seed(seed + start_id)
        if start_id == 0:
            h_guess = np.array(initial_h_values, dtype=np.float64)
        else:
            h_guess = perturb(initial_h_values, seed + start_id)
        # Ensure sum constraint
        current_sum = np.sum(h_guess)
        if not np.isclose(current_sum, n_points / 2, atol=1e-6):
            h_guess[-1] += (n_points / 2 - current_sum)
            h_guess = np.clip(h_guess, 0, 1)
        # Define objective
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cross_corr = np.correlate(h, 1 - h, mode='full')
            return np.max(cross_corr * dx)
        # Constraint
        def constraint_sum(h_flat):
            return np.sum(h_flat) - n_points / 2
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        # Run optimization
        try:
            result = minimize(objective, h_guess, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 500})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except:

        elapsed = time.time() - start_time
        if elapsed >= budget_s:

    # Final check with initial if no improvement
    if best_c5 == float('inf'):

    return best_h, best_c5, n_points
