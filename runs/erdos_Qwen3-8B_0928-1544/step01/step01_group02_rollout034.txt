import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0
    dx = 2.0 / n_points
    initial_guesses = [initial_h_values]
    for i in range(9):
        h = initial_h_values.copy()
        np.random.seed(seed + i)
        idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for j in idxs:
            h[j] += np.random.uniform(-0.1, 0.1)
            h[j] = np.clip(h[j], 0.0, 1.0)
        s = np.sum(h)
        h *= (n_points / 2) / s
        initial_guesses.append(h)
    
    best_h = initial_h_values
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        guess = np.array(guess, dtype=np.float64)
        def obj(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr * dx)
        def cons(x):
            return np.sum(x) - n_points / 2
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'ftol': 1e-9, 'maxiter': 500})

            h_opt = res.x
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        if time.time() - start_time > budget_s - 100:
