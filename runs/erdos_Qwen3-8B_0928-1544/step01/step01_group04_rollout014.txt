    import numpy as np
    from scipy.optimize import minimize
    import time

    # Predefined initial solution
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Objective function to minimize
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    # Generate extended initial vector from seed
    def generate_initial(n_points, seed):
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)

    # Optimization parameters
    num_restarts = 30
    max_iter_per_restart = 100

    # Generate several diverse initial guesses
    guesses = []
    for i in range(num_restarts):
        seed_val = seed + i
        guess = generate_initial(n_points, seed_val)
        guesses.append(guess)

    # Run optimization from each guess
    for idx, guess in enumerate(guesses):
        # Scale the initial guess to meet sum constraint
        guess_sum = np.sum(guess)
        if guess_sum == 0:
            scale = 1.0
        else:
            scale = (n_points / 2) / guess_sum
        guess_scaled = guess * scale

        # Perform optimization
        result = minimize(
            compute_C5,
            guess_scaled,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'eps': 1e-8}

                best_h = current_h

                print(f"New best C5: {best_c5}")
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
