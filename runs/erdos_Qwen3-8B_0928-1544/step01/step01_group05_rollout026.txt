def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def fitness(h):
        # Compute C₅
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        # Penalty for sum constraint
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return c5 + penalty

    # Bounds: each h[i] is between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result_de = differential_evolution(
        fitness,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=200,
        seed=seed
    )

    # Best solution from DE
    best_h_de = result_de.x

    # Now run SLSQP on best_h_de
    def objective_SLSQP(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Constraints: sum(h) = target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Bounds for SLSQP
    bounds_SLSQP = [(0.0, 1.0) for _ in range(n_points)]

    result_slsqp = minimize(
        objective_SLSQP,
        best_h_de,
        method='SLSQP',
        bounds=bounds_SLSQP,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8}
    )

    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun * dx

    # Check if the sum is correct
    assert np.isclose(np.sum(best_h), target_sum), "Sum constraint not met"

    return (best_h, best_c5, n_points)

import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def fitness(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(
        fitness,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=200,
        seed=seed
    )
    best_h_de = result_de.x

    def objective_SLSQP(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds_SLSQP = [(0.0, 1.0) for _ in range(n_points)]

    result_slsqp = minimize(
        objective_SLSQP,
        best_h_de,
        method='SLSQP',
        bounds=bounds_SLSQP,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8}
    )

    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun * dx

    assert np.isclose(np.sum(best_h), target_sum), "Sum constraint not met"
    return (best_h, best_c5, n_points)
