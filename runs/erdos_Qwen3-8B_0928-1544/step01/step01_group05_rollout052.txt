import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    # Generate random initial guess
    np.random.seed(seed)
    random_h = np.random.uniform(0, 1, n_points)
    total = np.sum(random_h)
    scaling_factor = (n_points / 2) / total
    random_h = random_h * scaling_factor
    random_h[0] = (n_points / 2) - np.sum(random_h[1:] if n_points > 1 else 0)

    # Define initial points
    h1 = initial_h_values.copy()
    h2 = random_h

    # Optimization parameters
    method1 = 'L-BFGS-B'
    method2 = 'SLSQP'
    maxiter = 150
    ftol = 1e-8

    # Run optimizations
    result1 = minimize(
        objective,
        h1,
        method=method1,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-8}
    )
    result2 = minimize(
        objective,
        h2,
        method=method2,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-8}
    )

    best_result = result1 if result1.fun < result2.fun else result2
    best_h = best_result.x
    best_c5 = best_result.fun * dx

    return (best_h, best_c5, n_points)
