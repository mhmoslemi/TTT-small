                break
        
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])
        
        new_population = []
        for i in range(population_size // 2):
            p1, p2 = parents[2*i], parents[2*i+1]
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([p1[:point], p2[point:]])
                child2 = np.concatenate([p2[:point], p1[point:]])
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            for child in [child1, child2]:
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.normal(0, 0.05)
                        child[j] = np.clip(child[j], 0, 1)
                child = child * sum_required / np.sum(child)
            
            new_population.append(child1)
            new_population.append(child2)
        
        combined = np.vstack([population, new_population])
        fitness_combined = np.array([np.max(np.correlate(h, 1 - h, mode='full') * dx) for h in combined])
        indices = np.argsort(fitness_combined)
        population = combined[indices[:population_size]]
    
    best_h = population[np.argmin([np.max(np.correlate(h, 1 - h, mode='full') * dx) for h in population])]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * sum_required / np.sum(best_h)
    
    return best_h, best_c5, n_points
