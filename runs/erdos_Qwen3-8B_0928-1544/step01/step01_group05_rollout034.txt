        if current_best < best_fitness:
            best_fitness = current_best
            best_h = population[best_idx]
        if gen > 50 and current_best == best_fitness:
            break
        new_population = []
        for _ in range(population_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + mutation_factor * (b - c)
            target = np.random.choice(population)
            offspring = np.copy(mutant)
            for i in range(n_points):
                if np.random.rand() < crossover_prob:
                    offspring[i] = mutant[i]
                else:
                    offspring[i] = target[i]
            # Feasibility repair
            offspring = np.clip(offspring, 0, 1)
            current_sum = np.sum(offspring)
            if current_sum != 0:
                scale = (n_points / 2) / current_sum
                offspring *= scale
            offspring = np.clip(offspring, 0, 1)
            new_population.append(offspring)
        population = new_population

    # Refine with SLSQP
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8}
    )

    final_h = result.x
    final_c5 = result.fun * dx
    return (final_h, final_c5, n_points)
