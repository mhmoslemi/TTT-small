        best_c5 = min(best_c5, fitness[best_idx])
        best_h = h_list[best_idx]

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Selection
        num_parents = int(0.2 * pop_size)
        parents = [population[i] for i in np.argsort(fitness)[:num_parents]]

        # Generate new population
        new_population = []
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = (parent1 + parent2) / 2
            for i in range(len(child)):
                if np.random.rand() < mut_rate:
                    delta = np.random.normal(0, mut_std)
                    if i < M:
                        child[i] = np.clip(child[i], 0, 0.5)
                    elif i < 2*M:
                        child[i] = np.clip(child[i], 1, max_f)
                    else:
                        child[i] = np.clip(child[i], 0, 2*np.pi)
                    child[i] += delta
            new_population.append(child)
        population = new_population

    return (best_h, best_c5, n_points)
