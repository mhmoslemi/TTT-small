def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)

        return np.array([]), 1.0, 0

    sum_h_target = n_points / 2.0
    
    # Objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr * dx)
    
    # Constraint function
    def constraint_sum(h_flat):
        return np.sum(h_flat) - sum_h_target
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate multiple initial guesses
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Use a random seed for perturbations
    np.random.seed(seed)
    
    for perturb_seed in range(10):
        # Generate a perturbed guess
        h_perturbed = initial_h_values.copy()
        np.random.seed(perturb_seed)
        # Perturb 10% of the elements
        num_perturb = int(n_points * 0.1)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_perturbed[i] += delta
            h_perturbed[i] = np.clip(h_perturbed[i], 0.0, 1.0)
        # Adjust the sum
        current_sum = np.sum(h_perturbed)
        delta_sum = current_sum - sum_h_target

        if abs(delta_sum) > 1e-6:
            h_perturbed[-1] += delta_sum
            h_perturbed[-1] = np.clip(h_perturbed[-1], 0.0, 1.0)
        
        # Run optimization
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,

import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0
    dx = 2.0 / n_points
    sum_h_target = n_points / 2.0

        return np.max(cross_corr * dx)

        return np.sum(h_flat) - sum_h_target
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    np.random.seed(seed)
