    velocities = np.random.uniform(-max_velocity, max_velocity, (num_particles, n_points))
    start_time = time.time()
    
    for _ in range(num_iterations):
        r1, r2 = np.random.rand(num_particles, n_points), np.random.rand(num_particles, n_points)
        velocities = w * velocities + c1 * r1 * (p_best - particles) + c2 * r2 * (g_best - particles)
        velocities = np.clip(velocities, -max_velocity, max_velocity)
        
        particles += velocities
        particles = np.clip(particles, 0, 1)
        sum_p = np.sum(particles, axis=1)
        scale_factors = (n_points / 2) / sum_p
        particles = particles * scale_factors[:, np.newaxis]
        
        current_c5s = np.array([compute_C5(p) for p in particles])
        
        for i in range(num_particles):
            if current_c5s[i] < compute_C5(p_best[i]):
                p_best[i] = particles[i].copy()
        
        best_idx = np.argmin(current_c5s)
        if current_c5s[best_idx] < best_c5:
            best_c5 = current_c5s[best_idx]
            g_best = particles[best_idx].copy()
        
        if time.time() - start_time > budget_s:
            break
    
    return g_best, best_c5, n_points
