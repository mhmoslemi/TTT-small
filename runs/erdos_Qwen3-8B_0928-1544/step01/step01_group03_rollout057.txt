import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 95
    dx = 2.0 / n_points
    initial_h = initial_h_values
    
    n_starts = 5
    scale = 0.1 * np.sqrt(1.0 / n_points)
    
    def objective(h_vector):
        corr = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    for i in range(n_starts):
        np.random.seed(seed + i)
        noise = np.random.randn(n_points) * scale
        h_perturbed = np.clip(initial_h + noise, 0, 1)
        scaling = (n_points / 2) / np.sum(h_perturbed)
        h_perturbed = h_perturbed * scaling
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 800, 'rhobeg': 0.1, 'rhoend': 1e-5}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
