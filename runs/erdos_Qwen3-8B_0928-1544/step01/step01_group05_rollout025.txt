import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    current_h = initial_h_values.copy()
    n_points = len(current_h)
    best_c5 = float('inf')
    best_h = current_h

    for step in range(3):
        new_n = n_points * 2
        if new_n > 280:
            break
        initial_guess = np.repeat(current_h, 2)
        dx = 2.0 / new_n
        bounds = [(0.0, 1.0) for _ in range(new_n)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - new_n / 2})

        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8}
        )

        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
            n_points = new_n
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break

    return (best_h, best_c5, len(best_h))
