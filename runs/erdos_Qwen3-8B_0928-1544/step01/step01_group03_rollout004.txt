import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 95
    dx = 2.0 / n_points
    h_initial = initial_h_values

        return np.max(corr) * dx

            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],

                'disp': False
