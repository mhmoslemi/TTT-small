            if np.random.rand() < crossover_rate:
                parent1, parent2 = new_population[i], new_population[i+1]
                alpha = np.random.rand()
                child1 = parent1 * alpha + parent2 * (1 - alpha)
                child2 = parent2 * alpha + parent1 * (1 - alpha)
                child1 = np.clip(repair_h(child1, n_points / 2), 0, 1)
                child2 = np.clip(repair_h(child2, n_points / 2), 0, 1)
                new_population[i], new_population[i+1] = child1, child2
        for i in range(pop_size):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                new_population[i] += noise
                new_population[i] = np.clip(repair_h(new_population[i], n_points / 2), 0, 1)
        population = new_population
        for h in population:
            current_fitness = evaluate_erdos_solution(h)
            if current_fitness < best_fitness:
                best_fitness = current_fitness
                best_h = h.copy()
    return (best_h, best_fitness, n_points)
