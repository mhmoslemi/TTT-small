        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty * 1e4
    
    for _ in range(3):
        if time_remaining <= 0:
            break
        dx = 2.0 / n_points
        if n_points == len(initial_h_values):
            h_init = initial_h_values.copy()
        else:
            h_init = np.random.uniform(0, 1, n_points)
            scale = (n_points / 2) / np.sum(h_init)
            h_init *= scale
        
        bounds = [(0.0, 1.0) * np.ones(n_points)]
        start_de = time.time()
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=100, popsize=30,
            tol=1e-6, mutation=(0.5, 1), recombination=0.8
        )
        de_time = time.time() - start_de
        time_remaining -= de_time
        
        start_slsqp = time.time()
        result_slsqp = minimize(
            objective, result_de.x, method='SLSQP',
            bounds=bounds, constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-8}
        )
        slsqp_time = time.time() - start_slsqp
        time_remaining -= slsqp_time
        
        current_c5 = result_slsqp.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
        
        n_points *= 2
    
    return (best_h, best_c5, len(best_h))
