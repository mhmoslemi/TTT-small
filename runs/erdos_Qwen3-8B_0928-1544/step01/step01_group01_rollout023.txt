from scipy.optimize import minimize
from scipy.signal import correlate

    n_points = len(h0)
    dx = 2.0 / n_points

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr * dx)
        return max_corr

    def smooth_max(val, temp=10.0):
        return np.log(np.sum(np.exp(val / temp))) * temp

    def admm_step(h, u, rho, temp):
        corr = correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        smoothed = smooth_max(np.log(corr / max_corr + 1e-12), temp)
        grad = np.zeros_like(h)
        
        for k in range(len(corr) - 1):
            w = np.zeros(n_points)
            if k < n_points:
                w[k:n_points] = h[:n_points - k] * dx
            if k > 0:
                w[:k] = h[k:] * dx
            grad += (1 / (1 + np.exp(-smoothed))) * (w - 1 + u / rho)
        return grad

    h = h0.copy()
    u = np.zeros(n_points)

    tol = 1e-5
