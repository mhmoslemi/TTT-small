            trial_c5 = compute_c5(trial)
            if trial_c5 < current_c5:
                new_population.append(trial)
            else:
                new_population.append(target)
        population = new_population
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cross_corr = np.correlate(h, 1 - h, mode='full')
            return np.max(cross_corr * dx)
        
        def constraint_sum(h_flat):
            return np.sum(h_flat) - n_points / 2
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
