import numpy as np
from scipy.optimize import differential_evolution, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if not initial_h_values.size:
        return np.array([]), 1.0, 0
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    assert np.isclose(sum_h, n_points / 2), "Initial h sum mismatch"
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr * dx)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    pop_size = 20
    mutation = 0.8
    crossover = 0.9
    max_gens = int(budget_s / 50)
    
    def de_objective(x):
        return evaluate(x)
    
    result = differential_evolution(
        de_objective, bounds, strategy='best1bin', 
        popsize=pop_size, mutation=mutation, 
        recombination=crossover, maxiter=max_gens, 
        tol=1e-5, disp=False, polish=False, 
        mutation_tuple=(mutation, 0.5), 
        constraints=[constraint_sum]
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
