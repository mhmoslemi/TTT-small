        return np.max(corr) * dx

    def scale(h):
        return h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

        return compute_C5(h)

    for gen in range(200):
        if elapsed > budget_s - 10:

        probs = fit_scaled / np.sum(fit_scaled)
        parents = np.random.choice(len(pop), size=pop_size, p=probs)
        offspring = []
        for i in range(pop_size // 2):

            p = np.random.randint(1, n_points)
