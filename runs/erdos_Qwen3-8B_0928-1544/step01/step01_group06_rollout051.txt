    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        valid_shifts = cross_corr[n_points - 1: 2 * n_points - 1]
        max_corr = np.max(valid_shifts)
        return max_corr * dx

    def compute_gradient(h, k):
        grad = np.zeros(len(h))
        for x in range(len(h)):
            if x + k < len(h):
                grad[x] = 1 - h[x + k]
        return grad

    best_h = h_values.copy()
    best_c5 = compute_C5(best_h)
    learning_rate = 0.1
    steps = 0
    no_improvement = 0
    start_time = time.time()
    while steps < 10000 and time.time() - start_time < budget_s:
        cross_corr = np.correlate(h_values, 1 - h_values, mode='full')
        valid_shifts = cross_corr[n_points - 1: 2 * n_points - 1]
        max_index = np.argmax(valid_shifts)
        k = max_index
        grad = compute_gradient(h_values, k)
        h_values -= learning_rate * grad
        h_values = np.clip(h_values, 0, 1)
        current_sum = np.sum(h_values)
        scale = target_sum / current_sum
        h_values *= scale

        current_c5 = compute_C5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 500:
                learning_rate *= 0.5
                no_improvement = 0

        if steps % 250 == 0:
            noise = np.random.normal(0, 1e-3, n_points)
            h_values += noise
            h_values = np.clip(h_values, 0, 1)
            current_sum = np.sum(h_values)
            scale = target_sum / current_sum
            h_values *= scale
        steps += 1
    return (best_h, best_c5, n_points)
