def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy().astype(float)

    def objective(h_flat):
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    # Initial optimization
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (result.x, result.fun * dx, n_points)

    # Perturbation and second optimization
    perturbed = h + 0.05 * np.random.normal(0, 1, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    norm = np.sum(perturbed)
    if norm > 0:
        perturbed *= (n_points / 2) / norm
    result2 = minimize(
        objective,
        perturbed,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    final_h = result.x if result.fun * dx < result2.fun * dx else result2.x
    final_c5 = result.fun * dx if result.fun * dx < result2.fun * dx else result2.fun * dx
    return (final_h, final_c5, n_points)
