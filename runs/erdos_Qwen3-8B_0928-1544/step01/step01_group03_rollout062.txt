import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h.copy()
    best_c5 = np.inf
    
    np.random.seed(seed)
    start_time = time.time()
    for _ in range(20):
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = initial_h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_h = np.sum(h_perturbed)
        h_perturbed *= (n_points / 2.0) / sum_h
        
        res = minimize(objective, h_perturbed, method='COBYLA',
                      bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      options={'maxiter': 500, 'rhobeg': 0.1, 'rhoend': 1e-5})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h, np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx, n_points)
    return (best_h, best_c5, n_points)
