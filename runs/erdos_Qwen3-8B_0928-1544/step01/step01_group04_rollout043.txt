import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h = np.array(initial_h_values).copy()
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    s = np.max(corr) * dx
    lambda_k = np.zeros_like(corr)
    rho = 1.0
    max_iter = 100
    for _ in range(max_iter):
        grad = np.zeros(n)
        for k in range(len(corr)):
            for i in range(n):
                if k <= n - 1:
                    j = i + k
                    if j < n:
                        grad[i] += lambda_k[k] * (1 - h[j]) * (corr[k] - s / dx)
                else:
                    j = i - (k - n + 1)
                    if j >= 0:
                        grad[i] += lambda_k[k] * (1 - h[j]) * (corr[k] - s / dx)
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h *= (n / 2.0) / (sum_h if sum_h != 0 else 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        s = np.max(corr) * dx
        lambda_k += rho * (corr - s / dx)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return h, c5, n
