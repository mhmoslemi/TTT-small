def run(seed=42, budget_s=1100, **kwargs):
    # Initialize with the provided initial H values or a default
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(70)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(20):  # Limit iterations to stay within budget
        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                h = current_h
            # Check if we should increase n_points
            if (abs(current_c5 - best_c5) < 1e-6):
                # No improvement; increase n_points
                n_points += 10
                h = np.zeros(n_points)
                dx = 2.0 / n_points
                for i in range(n_points):
                    h[i] = 0.5 if i % 2 else 0.5
                h *= (n_points / 2) / np.sum(h)
            else:
                h = current_h
        else:
            break
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
