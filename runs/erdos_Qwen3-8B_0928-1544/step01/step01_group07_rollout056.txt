        bounds = Bounds([0.0]*new_n, [1.0]*new_n)
        def constraint_sum(h):
            return np.sum(h) - new_n / 2
        result = minimize(
            fun=lambda h: objective(h, dx),
            x0=new_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        elapsed_time = time.time() - start_time
        time_used += elapsed_time
        if time_used >= budget_s:
            break
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x.copy()
        current_h = result.x.copy()
        n_points = new_n

    return best_h, best_c5, n_points
