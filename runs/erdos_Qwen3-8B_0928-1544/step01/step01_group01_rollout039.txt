import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 256
        h = np.full(n_points, n_points / 2 / n_points)
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 1000

    def objective(h):
        corr = correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': np.ones_like
    }]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9})

    if result.success:
        h_best = result.x
        c5 = result.fun
    else:
        h_best = h
        c5 = objective(h_best)

    h_best = np.clip(h_best, 0, 1)
    if np.abs(np.sum(h_best) - n_points / 2) > 1e-6:
        h_best = h
        c5 = objective(h_best)

    return h_best, c5, n_points
