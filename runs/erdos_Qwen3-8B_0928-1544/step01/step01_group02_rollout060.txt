        new_pop = []
        for i in range(M):
            a, b, c = np.random.choice(M, 3, replace=False)
            h_a = population[a]
            h_b = population[b]
            h_c = population[c]
            mutant = h_a + 0.8 * (h_b - h_c)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial = adjust(trial)
            cross_corr = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(cross_corr * dx)
            cross_corr_curr = np.correlate(population[i], 1 - population[i], mode='full')
            c5_curr = np.max(cross_corr_curr * dx)
            if c5_trial < c5_curr:
                new_pop.append(trial)
            else:
                new_pop.append(population[i])
        population = new_pop
    
    best_h = population[0]
    min_c5 = np.inf
    for h in population:
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr * dx)
        if c5 < min_c5:
            min_c5 = c5
            best_h = h
    return best_h, min_c5, n_points
