import numpy as np
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    def create_initial_h(n_points):
        if 'initial_h_values' in globals():
            return np.array(initial_h_values[:n_points])
        return np.full(n_points, 0.5)
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        corr = correlate(h, 1.0 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def optimize_level(n_points, prev_h=None):
        h = create_initial_h(n_points)
        if prev_h is not None:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(prev_h)), prev_h)
        bounds = Bounds(0.0, 1.0)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = minimize(
            fun=lambda x: compute_c5(x, n_points),
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, result.fun
    
    best_h = None
    best_c5 = float('inf')

    # Coarsest level
