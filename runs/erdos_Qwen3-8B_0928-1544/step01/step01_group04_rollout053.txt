import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return h
        return h * (n_points / 2) / s

    # Initialize population
    pop_size = 50
    pop = [np.copy(h_initial)]
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = scale_h(h)
        pop.append(h)

    best_h = np.copy(h_initial)
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break

        # Evaluate fitness and select parents
        fitness_scores = np.array([-compute_C5(h) for h in pop])
        total_fitness = np.sum(fitness_scores)
        if total_fitness == 0:
            break
        selection_probs = fitness_scores / total_fitness

        def select_parent():
            return np.random.choice(pop, p=selection_probs)

        new_pop = []
        for _ in range(pop_size):
            p1 = select_parent()
            p2 = select_parent()
            split = np.random.randint(n_points)
            child = np.concatenate([p1[:split], p2[split:]])
            child = scale_h(child)

            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = scale_h(child)

            new_pop.append(child)

        pop = new_pop

        for h in pop:
            curr_c5 = compute_C5(h)
            if curr_c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = curr_c5
                print(f"Gen {gen}: C5={best_c5}")

    final_h = scale_h(best_h)
    final_c5 = compute_C5(final_h)
    return final_h, final_c5, n_points
