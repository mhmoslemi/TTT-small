import numpy as np
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    def fitness(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    pop_size = 30
    pop = [initial_h.copy() for _ in range(pop_size)]
    for i in range(1, pop_size):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop[i] = h
    
    max_gen = 200
    mut = 0.7
    cross = 0.8
    best_h = initial_h.copy()
    best_val = fitness(best_h)
    
    for _ in range(max_gen):
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mut * (pop[b] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < cross:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            s = np.sum(trial)
            trial *= (n_points / 2) / s
            f = fitness(trial)
            if f < fitness(pop[i]):
                pop[i] = trial
            if f < best_val:
                best_val = f
                best_h = trial
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    res = minimize(fitness, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8})
