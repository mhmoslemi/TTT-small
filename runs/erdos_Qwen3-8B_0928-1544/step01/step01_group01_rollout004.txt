import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values
    else:
        initial_guess = None

    n_points = 256
    dx = 2.0 / n_points
    population_size = 100
    max_iter = 500
    w, c1, c2 = 0.7, 2.0, 2.0
    start_time = time.time()

    # Generate population
    np.random.seed(seed)
    pos = []
    vel = []
    pbest = []
    for _ in range(population_size):
        if initial_guess is not None and np.random.rand() < 0.5:
            h = initial_guess + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
        else:
            h = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        pos.append(h)
        vel.append(np.zeros(n_points))
        pbest.append(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

    global_best = np.min(pbest)
    best_pos = np.array(pos[np.argmin(pbest)])

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel[i] = w * vel[i] + c1 * r1 * (pos[i] - pos[i]) + c2 * r2 * (best_pos - pos[i])
            new_pos = pos[i] + vel[i]
            new_pos = np.clip(new_pos, 0, 1)
            sum_h = np.sum(new_pos)
            if sum_h != 0:
                new_pos = new_pos * (n_points / 2) / sum_h
            else:
                new_pos = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
            pos[i] = new_pos
            corr = np.correlate(pos[i], 1 - pos[i], mode="full") * dx
            curr = np.max(corr)
            if curr < pbest[i]:
                pbest[i] = curr
                pos[i].copyto(pos[i])
            if curr < global_best:
                global_best = curr
                best_pos = pos[i].copy()

    h_values = best_pos
    c5_bound = global_best
    return h_values, c5_bound, n_points
