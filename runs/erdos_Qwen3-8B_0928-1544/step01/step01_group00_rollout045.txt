import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:

    def compute_c5(h_vals):

        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    T_start = 0.1
    alpha = 0.99
    T_min = 1e-5
    max_iter = 10000
    T = T_start
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Perturb h
        perturb_size = int(n_points * 0.1)
        indices = np.random.choice(n_points, size=perturb_size, replace=False)
        h_perturb = h.copy()
        for i in indices:
            # To make the perturbation more effective, maybe use a Gaussian
            delta = np.random.normal(0, 0.1)
            h_perturb[i] += delta
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb_sum = np.sum(h_perturb)
