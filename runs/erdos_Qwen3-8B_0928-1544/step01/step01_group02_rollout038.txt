from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
