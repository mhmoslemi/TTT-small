from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(200):
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        
        # Early exit if budget is nearing
        if _ > 10 and best_c5 < 0.38085:
            break
        
        # Update h for next iteration
        h = res.x
    
    # Ensure final solution meets constraints
    final_h = best_h

    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:

    return final_h, best_c5, n_points
