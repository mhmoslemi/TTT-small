        max_pos = np.argmax(cross_corr[n_points-1:]) + (n_points-1)
        k_max = max_pos - (n_points-1)
        grad = np.zeros_like(h)
        if k_max < len(h):
            for i in range(len(h) - k_max):
                grad[i] = -(1 - h[i + k_max])
        return grad

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    learning_rate = 0.1

    best_h = h_values.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        t += 1
        grad = compute_gradient(h_values)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_values -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h_values = np.clip(h_values, 0, 1)
        current_sum = np.sum(h_values)
        scale = target_sum / current_sum
        h_values *= scale
        if t % 250 == 0:
            noise = np.random.normal(0, 1e-3, n_points)
            h_values += noise
            h_values = np.clip(h_values, 0, 1)
            current_sum = np.sum(h_values)
            scale = target_sum / current_sum
            h_values *= scale
        current_c5 = compute_C5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        if t % 1000 == 0:
            learning_rate *= 0.95
    return (best_h, best_c5, n_points)
