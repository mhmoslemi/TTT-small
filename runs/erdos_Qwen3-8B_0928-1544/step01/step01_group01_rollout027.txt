                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_c = np.sum(child)
                if sum_c > 1e-12:
                    child *= (target_sum / sum_c)
                    child = np.clip(child, 0, 1)
            offspring.append(child)

        population = sorted(population, key=fitness)[:population_size // 2]
        population.extend(offspring)
        population = population[:population_size]

    h_values = np.clip(best_individual, 0, 1)
    c5_bound = best_fitness
    return h_values, c5_bound, n_points
