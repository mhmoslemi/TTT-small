def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h_values)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def compute_gradient(h):
        n = len(h)
        grad = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            c5_plus = compute_C5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            c5_minus = compute_C5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    best_h = h_values.copy()

        grad = compute_gradient(h_values)

        current_c5 = compute_C5(h_values)
