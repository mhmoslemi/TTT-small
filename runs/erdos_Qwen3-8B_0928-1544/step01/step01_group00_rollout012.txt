        h = np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    T = 0.1
    alpha = 0.99
    max_iter = 20000

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        candidate = h.copy()
        num_perturb = max(1, int(n_points * 0.05))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.1 * T, 0.1 * T)
            candidate[idx] += delta
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if candidate_sum == 0:
            candidate = candidate * (n_points / 2) / 1e-6
        else:
            candidate = candidate * (n_points / 2) / candidate_sum
        new_c5 = compute_c5(candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate.copy()
        else:
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                h = candidate.copy()
        T *= alpha

    return (best_h, best_c5, n_points)
