import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    # Assume initial_h_values is already defined and accessible globally
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Parameters for PSO
    num_particles = 30
    max_iter = 100
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_limit = 0.5

    # Precompute dx once
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize particles and velocities
    particles = []
    velocities = []
    p_best = []
    p_best_c5 = []

    np.random.seed(seed)
    for _ in range(num_particles):
        h = h_initial.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale
        particles.append(h)
        velocities.append(np.zeros(n_points))
        current_c5 = compute_C5(h)
        p_best.append(h)
        p_best_c5.append(current_c5)

    # Initialize global best
    g_best = particles[0]
    g_best_c5 = p_best_c5[0]
    for i in range(num_particles):
        if p_best_c5[i] < g_best_c5:
            g_best = particles[i]
            g_best_c5 = p_best_c5[i]

    start_time = time.time()

    # PSO loop
    for iter in range(max_iter):
        w = w_start - (w_start - w_end) * iter / max_iter
        for i in range(num_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = velocities[i] * w + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            v = np.clip(v, -velocity_limit, velocity_limit)
            velocities[i] = v

            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            scale = (n_points / 2) / sum_new if sum_new != 0 else 1.0
