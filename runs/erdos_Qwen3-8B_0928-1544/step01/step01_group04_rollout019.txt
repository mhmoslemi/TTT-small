    from scipy.optimize import minimize
    import time

    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    constraint_sum = lambda h: np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    num_restarts = 5
    max_iter_per_restart = 200

    for _ in range(num_restarts):
        perturbation = np.random.normal(0, 0.01, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale = (n_points / 2) / sum_perturbed if sum_perturbed != 0 else 1.0
        h_perturbed *= scale

        result = minimize(
            compute_C5,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart}
        )

        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
