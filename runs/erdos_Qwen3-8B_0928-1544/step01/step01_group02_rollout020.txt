def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:

    # Initialize population
    M = 20
    population = []

    initial_h = np.array(initial_h_values, dtype=np.float64)
    sum_initial = np.sum(initial_h)
    if not np.isclose(sum_initial, target_sum, atol=1e-6):
        factor = target_sum / sum_initial
        initial_h = initial_h * factor
        initial_h = np.clip(initial_h, 0, 1)
    population.append(initial_h.copy())
    # Generate other candidates by perturbing initial_h
    for _ in range(M - 1):
        candidate = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        candidate += noise
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        if abs(current_sum - target_sum) < 1e-6:
            population.append(candidate)
        else:
            factor = target_sum / current_sum
            adjusted = candidate * factor
            adjusted = np.clip(adjusted, 0, 1)
            population.append(adjusted)
    # DE loop
    max_gens = int(budget_s // 50)  # 50s per generation
    max_gens = min(max_gens, 200)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full') * dx)
    start_time = time.time()
    for gen in range(max_gens):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            h_a, h_b, h_c = population[a], population[b], population[c]
            mutant = h_a + 0.8 * (h_b - h_c)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            current_sum = np.sum(trial)
            if not np.isclose(current_sum, target_sum, atol=1e-6):
                factor = target_sum / current_sum
                trial = np.clip(trial * factor, 0, 1)
            cross_corr = np.correlate(trial, 1 - trial, mode='full')
            c5 = np.max(cross_corr * dx)
            current_c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full') * dx)
            if c5 < current_c5:
                population[i] = trial
                if c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = c5
    return best_h, best_c5, n_points
