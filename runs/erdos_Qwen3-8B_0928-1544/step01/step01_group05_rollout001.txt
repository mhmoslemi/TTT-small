import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def penalty_func(h):
        return objective(h) + 1e3 * abs(np.sum(h) - n_points / 2)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    popsize = int(np.sqrt(n_points)) + 1
    maxiter_de = max(100, int(budget_s // 300))
    maxiter_slsqp = int(budget_s // 200)

    # Differential evolution with adaptive penalty
    result_de = differential_evolution(
        penalty_func,
        bounds,
        strategy='best1bin',
        maxiter=maxiter_de,
        popsize=popsize,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        callback=lambda x: None
    )
    best_h_de = result_de.x
    best_c5_de = result_de.fun

    # SLSQP refinement
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    result_slsqp = minimize(
        objective,
        best_h_de,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        options={'maxiter': maxiter_slsqp, 'ftol': 1e-8, 'eps': 1e-8}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    return (best_h, best_c5, n_points)
