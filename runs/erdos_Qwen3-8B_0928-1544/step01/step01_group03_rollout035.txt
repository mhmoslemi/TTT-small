            p2 = parents[i+1] if i+1 < len(parents) else p1
            child1 = []
            child2 = []
            for j in range(len(p1)):
                if np.random.rand() < crossover_rate:
                    child1.append(0.5 * (p1[j] + p2[j]))
                    child2.append(0.5 * (p1[j] + p2[j]))
                else:
                    child1.append(p1[j])
                    child2.append(p2[j])
            # Mutation
            for j in range(len(child1)):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.1)
                    child1[j] = np.clip(child1[j], 0, 5)
                    child2[j] += np.random.normal(0, 0.1)
                    child2[j] = np.clip(child2[j], 0, 5)
            new_pop.append(child1)
            new_pop.append(child2)

        population = new_pop
        pop_perf = [evaluate(p) for p in population]
        current_best = np.min(pop_perf)
        if current_best < best_c5:
            best_c5 = current_best
            best_params = population[np.argmin(pop_perf)]

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    final_h = 0.5
    for i in range(M):
        f = best_params[3*i]
        A = best_params[3*i + 1]
        phi = best_params[3*i + 2]
        final_h += A * np.sin(2 * np.pi * f * x + phi)
    final_h = np.clip(final_h, 0, 1)
    best_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx

    return (final_h, best_c5, n_points)
