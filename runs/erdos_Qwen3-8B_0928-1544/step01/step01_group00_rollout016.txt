import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():

    h = h * (n_points / 2) / h_sum

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    def compute_gradient(h_vals):
        n_points = len(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i + shift) % n_points
            m = (i - shift) % n_points
            gradients[i] = (1 - h_vals[j]) - h_vals[m]
        return gradients

    best_c5 = compute_c5(h)
    start_time = time.time()
    learning_rate = 0.01
    max_iter = 100000
