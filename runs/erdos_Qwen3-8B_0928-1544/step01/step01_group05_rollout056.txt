import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import math

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)

        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-8, 'eps': 1e-9}
    )

    best_h = result.x
    best_c5 = result.fun * dx
    return (best_h, best_c5, n_points)
