import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values, dtype=float)]
        for _ in range(49):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        population = [np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points)) for _ in range(50)]

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        c5 = np.array([compute_c5(h) for h in population])
        idx = np.argsort(c5)
        parents = population[idx[:int(len(population)*0.2)]]
        new_pop = []
        for _ in range(50):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = population[np.random.randint(len(population))]
                mutate_idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                for i in mutate_idx:
                    child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            new_pop.append(child)
        population = new_pop
        current_c5 = c5[np.argmin(c5)]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[np.argmin(c5)]
        if time.time() - start_time > budget_s - 10:
            break

    return (np.array(best_h), best_c5, n_points)
