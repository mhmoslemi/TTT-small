            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            current = calc_c5(particles[i])
            if current < personal_best[i]:
                personal_best[i] = current
                personal_positions[i] = particles[i].copy()
            if current < best_fitness:
                best_fitness = current
                best_particle = particles[i].copy()
        if iter >= 100 and best_fitness < 1e-6:
            break

    h_values = best_particle
    c5_bound = best_fitness
    return h_values, c5_bound, n_points
