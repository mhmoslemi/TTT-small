import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    max_generations = 200

    population = [h_initial.copy()]
    for _ in range(pop_size - 1):

        sum_rand = np.sum(h_rand)
        scale = (n_points / 2) / sum_rand if sum_rand != 0 else 1.0
        h_rand *= scale
        population.append(h_rand)

    def evaluate_population(pop):
        return [-compute_C5(h) for h in pop]

    def select_parents(pop, fitnesses):
        total_fitness = np.sum(fitnesses)
        if total_fitness == 0:
            return [np.random.choice(pop) for _ in range(2)]
        probs = fitnesses / total_fitness
        return np.random.choice(pop, size=2, p=probs)

    def crossover(parent1, parent2):
        point = np.random.randint(0, n_points)
        child = np.copy(parent1)
        child[point:] = parent2[point:]
        sum_child = np.sum(child)
        scale = (n_points / 2) / sum_child if sum_child != 0 else 1.0
        return child * scale

    def mutate(h):
        noise = np.random.normal(0, 0.1, n_points)
        h_mut = h + noise
        h_mut = np.clip(h_mut, 0, 1)
        sum_mut = np.sum(h_mut)
        scale = (n_points / 2) / sum_mut if sum_mut != 0 else 1.0
        return h_mut * scale

    best_h = h_initial.copy()
