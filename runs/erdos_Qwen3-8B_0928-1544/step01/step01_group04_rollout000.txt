import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    # Assume initial_h_values is pre-defined globally
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    n_particles = 30
    max_iter = 100
    w = 0.7
    c1, c2 = 1.5, 1.5
    max_velocity = 0.5

    # Initialize particles and velocities
    particles = np.random.normal(0, 0.1, (n_particles, n_points))
    particles += h_initial
    particles = np.clip(particles, 0, 1)
    sum_particles = np.sum(particles, axis=1)
    scale_factors = (n_points / 2) / sum_particles
    particles = particles * scale_factors[:, np.newaxis]

    velocities = np.random.uniform(-max_velocity, max_velocity, (n_particles, n_points))
    p_best = particles.copy()
    p_best_vals = np.array([compute_C5(p) for p in particles])
    g_best = p_best[np.argmin(p_best_vals)]
    g_best_val = np.min(p_best_vals)

    start_time = time.time()

    for _ in range(max_iter):
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                particles[i] *= (n_points / 2) / sum_h
            current_val = compute_C5(particles[i])
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
        current_min = np.min(p_best_vals)
        if current_min < g_best_val:
            g_best_val = current_min
            g_best = p_best[np.argmin(p_best_vals)]
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return g_best, g_best_val, n_points
