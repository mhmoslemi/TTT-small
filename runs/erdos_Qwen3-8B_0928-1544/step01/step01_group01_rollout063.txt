        # Update h
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)

        # Enforce sum constraint
        sum_h = np.sum(h)
        if np.abs(sum_h - c) > 1e-6:
            delta = (c - sum_h) / n_points
            h += delta

        # Update dual variables
        dual += rho * (sum_h - c)

        # Check convergence
        if np.abs(sum_h - c) < tol:
            break

    # Final check and compute C5
    integrals = np.correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(integrals)
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - c) > 1e-6:
        h = initial_h_values
        integrals = np.correlate(h, 1 - h, mode='full') * dx
        c5_bound = np.max(integrals)

    return h, c5_bound, n_points
