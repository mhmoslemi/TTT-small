import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 45 if initial_h is None else len(initial_h)
    if initial_h is None:
        h_values = np.zeros(n_points)
        h_values[:n_points // 2] = 1
        h_values = h_values * (n_points / 2) / n_points
    else:
        h_values = initial_h.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h_values)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def compute_gradient(h, eps=1e-4):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            c5_plus = compute_C5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            c5_minus = compute_C5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    best_h = h_values.copy()
    best_c5 = compute_C5(best_h)
    learning_rate = 0.5
    steps = 0
    no_improvement = 0
    start_time = time.time()

    while steps < 20000 and time.time() - start_time < budget_s:
        grad = compute_gradient(h_values)
        h_values += -learning_rate * grad
        h_values = np.clip(h_values, 0, 1)
        current_sum = np.sum(h_values)
        scale = target_sum / current_sum
        h_values *= scale

        current_c5 = compute_C5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 1000:
                learning_rate *= 0.9
                no_improvement = 0

        if steps % 500 == 0:
            noise = np.random.normal(0, 1e-2, n_points)
            h_values += noise
            h_values = np.clip(h_values, 0, 1)
            current_sum = np.sum(h_values)
            scale = target_sum / current_sum
            h_values *= scale
        steps += 1
    return (best_h, best_c5, n_points)
