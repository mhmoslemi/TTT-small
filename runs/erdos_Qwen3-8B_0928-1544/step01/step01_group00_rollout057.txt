import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
    else:

    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        sum_expr = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            sum_expr += h[i] * (1 - h[j])
        constraints.append(sum_expr <= t * dx)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'optimal':
            best_h = h.value
            best_c5 = t.value
        else:
            best_h = h_initial
            best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full') * dx)
    except:
        best_h = h_initial
        best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full') * dx)
    if np.any(np.isnan(best_h)):

        best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full') * dx)
