def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    # DE parameters
    pop_size = 30
    mutation_factor = 0.7
    crossover_prob = 0.8
    max_gen = 200
    stagnation = 50
    initial_pop = [initial_h.copy()]
    
    # Generate random individuals with correct sum
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_pop.append(h)
    
    pop = np.array(initial_pop)
    best_c5 = float('inf')
    best_h = initial_h
    
    start = time.time()
    
    # DE optimization
    for gen in range(max_gen):
        new_pop = []
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            mutant = pop[a] + mutation_factor * (pop[b] - pop[c])

                    trial[j] = mutant[j]

            ts = np.sum(trial)
            if ts != 0:

            corr = correlate(trial, 1 - trial, mode='full')

            curr = np.max(correlate(pop[i], 1 - pop[i], mode='full')) * dx
            if trial_c5 < curr:
                new_pop.append(trial)

                new_pop.append(pop[i])
        pop = np.array(new_pop)

    return (res.x, res.fun, n_points)
