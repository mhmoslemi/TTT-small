    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0

    def compute_C5(h):

        max_corr = np.max(cross_corr)

    def compute_gradient(h, cross_corr):

        k_max = np.argmax(cross_corr)
        l = k_max - (n - 1)

            if 0 <= x + l < n:
                grad[x] += (1 - h[x + l])
                grad[x + l] -= h[x]
        grad *= dx
