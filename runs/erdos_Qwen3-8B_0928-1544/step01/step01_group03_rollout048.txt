import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 95
    dx = 2.0 / n_points
    initial_h = initial_h_values
    n_starts = 5
    perturbation_scale = 0.1 * np.sqrt(1.0 / n_points)
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    np.random.seed(seed)
    start_time = time.time()
    
    def objective(h_vector):
        corr = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for _ in range(n_starts):
        h_perturbed = initial_h + np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum == 0:
            continue
        scale_factor = (n_points / 2) / current_sum
        h_perturbed *= scale_factor
        h_perturbed = np.clip(h_perturbed, 0, 1)
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 800, 'rhobeg': 0.1, 'rhoend': 1e-5, 'disp': False}
        )
        if result.success:
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return (best_h, best_c5, n_points)
