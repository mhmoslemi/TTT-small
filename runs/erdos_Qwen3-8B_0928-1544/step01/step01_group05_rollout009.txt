import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    best_c5 = 1.0
    best_h = None
    n_points = 70
    n_points_list = [70, 140, 280]
    elapsed_total = 0.0
    initial_guess_70 = initial_h_values.copy()
    
    for i, n_points in enumerate(n_points_list):
        dx = 2.0 / n_points
        if i == 0:
            initial_guess = initial_guess_70
        else:
            if best_h is None:
                continue
            # Interpolate previous solution
            prev_h = best_h
            if len(prev_h) < n_points:
                new_h = np.repeat(prev_h, 2)
                if len(new_h) < n_points:
                    new_h = np.concatenate([new_h, [0.5] * (n_points - len(new_h))])
                initial_guess = new_h[:n_points]
            else:
                initial_guess = np.repeat(prev_h, 2)[:n_points]
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        start_time = time.time()
        try:
            result = differential_evolution(
                objective, bounds,
                strategy='best1bin',
                popsize=30,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                maxiter=100,
                seed=seed
            )
        except:
            result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}])
        
        elapsed = time.time() - start_time
        elapsed_total += elapsed
        if elapsed_total > budget_s:
            break
        
        h_candidate = result.x
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()
    
    return (best_h, best_c5, len(best_h))
