    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 1.0
    cooling_rate = 0.995
    max_iter = 100000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()
        new_h[i] += delta
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        scaling = (n_points / 2) / new_h_sum
        new_h = new_h * scaling
        new_c5 = compute_c5(new_h)
        current_c5 = compute_c5(h)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = new_h
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
        T *= cooling_rate

    return (best_h, best_c5, n_points)
