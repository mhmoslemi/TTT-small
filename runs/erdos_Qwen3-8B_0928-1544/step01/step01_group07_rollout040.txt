from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    # Determine the grid steps
    N_steps = [14, 28, 56, 70]
    best_c5 = float('inf')
    best_h = None
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else None
    for i, n in enumerate(N_steps):
        if i == 0 and initial_h is not None and len(initial_h) == n:
            h = initial_h.copy()
        else:
            h = np.full(n, 0.5)
        # Time check for budget
        if time.time() - start_time > budget_s - 500:
            break
        dx = 2.0 / n
        # Objective function: max overlap
        def objective(h_flat):
            corr = np.correlate(h_flat, 1 - h_flat, mode='full')
            return np.max(corr) * dx
        # Constraint: sum(h) = n/2
        def constraint_sum(h_flat):
            return np.sum(h_flat) - n / 2
        # Bounds
        bounds = Bounds([0.0]*n, [1.0]*n)
        # Optimization
        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        if time.time() - start_time > budget_s - 100:
            break
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
        # Upscale if not last step
        if i < len(N_steps) - 1:
            if n == N_steps[i + 1] // 2:
                if best_h is not None:
                    h = np.repeat(best_h, 2)
                else:
                    h = np.full(N_steps[i + 1], 0.5)
            else:
                h = np.full(N_steps[i + 1], 0.5)
        else:
            h = result.x
    # Final check with initial_h_values if available
    if initial_h is not None and len(initial_h) == 70:
        dx = 2.0 / 70
        corr = np.correlate(initial_h, 1 - initial_h, mode='full')
        current_c5_initial = np.max(corr) * dx
        if current_c5_initial < best_c5:

            best_h = initial_h
    return best_h, best_c5, len(best_h)
