    assert np.isclose(sum_h, n_points / 2), "Initial h must have sum = n_points / 2"
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr * dx)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use differential evolution with a balance between exploration and exploitation
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        maxiter=150, 
        popsize=20, 
        tol=1e-6, 
        mutation=(0.8, 1.0), 
        recombination=0.7, 
        polish=True, 
        disp=False
    )
    
    h_opt = result.x
    cross_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_value = np.max(cross_corr * dx)
    
    return h_opt, c5_value, n_points
