import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    required_sum = n_points / 2

    population_size = 100
    population = []

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        if len(initial_h) == n_points:
            population.append(initial_h)
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * required_sum
        population.append(h)

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    generations = 500
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    sigma = 0.05
    early_stopping_gen = 100
    best_fitness = float('inf')
    best_h = None
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [fitness(h) for h in population]
        best_idx = np.argmin(fitnesses)
        best_fitness_history = [fitnesses[best_idx]]

        if gen >= early_stopping_gen:
            if np.abs(best_fitness_history[-1] - best_fitness_history[-early_stopping_gen]) < 1e-7:
                break
        best_fitness = best_fitness_history[-1]

        parents = [population[random.choice(range(population_size))] for _ in range(population_size)]
        offspring = []

        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                p1, p2 = parents[i], parents[i+1]
                if random.random() < crossover_rate:
                    split = random.randint(1, n_points-1)
                    c1 = np.concatenate((p1[:split], p2[split:]))
                    c2 = np.concatenate((p2[:split], p1[split:]))
                else:
                    c1, c2 = p1.copy(), p2.copy()
                c1 = c1 * (required_sum / np.sum(c1))
                c2 = c2 * (required_sum / np.sum(c2))
            else:
                c1 = parents[i].copy()
                c2 = parents[i].copy()
            offspring += [c1, c2]

        for i in range(len(offspring)):
            h = offspring[i]
            if random.random() < mutation_rate:
                noise = np.random.normal(0, sigma, size=n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * (required_sum / np.sum(h))
            offspring[i] = h

        sorted_indices = np.argsort(fitnesses)
        replace_count = population_size // 2
        for i in range(replace_count):
            population[sorted_indices[-(i+1)]] = offspring[i]

        if fitnesses[best_idx] < best_fitness:
            best_fitness = fitnesses[best_idx]
            best_h = population[best_idx]

    best_h = np.clip(best_h, 0, 1)
    if not np.isclose(np.sum(best_h), required_sum, atol=1e-6):
        if 'initial_h_values' in globals():
            best_h = initial_h_values
        else:
            best_h = population[0]
        best_fitness = fitness(best_h)

    return best_h, best_fitness, n_points
