import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 20
    F = 0.8
    CR = 0.9
    max_generations = int(budget_s / 50)
    if max_generations < 1:
        max_generations = 10

    # Initialize population with initial_h_values and perturbations
    population = [np.array(initial_h_values, dtype=np.float64)]
    for _ in range(population_size - 1):
        perturb = np.random.rand(n_points) * 0.1 - 0.05
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        if current_sum == 0:
            scale = 1.0
        else:
            scale = target_sum / current_sum
        candidate *= scale
        population.append(candidate)

    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr * dx)

    best_h = np.array(initial_h_values)
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for gen in range(max_generations):
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + F * (b - c)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR or j == np.random.choice(range(n_points)):
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            current_sum = np.sum(trial)
            if current_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial *= target_sum / current_sum
            new_population.append(trial)
        # Evaluate and update best
        for trial in new_population:
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
        population = new_population
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
