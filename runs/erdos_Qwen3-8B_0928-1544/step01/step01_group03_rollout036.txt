```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 95
    dx = 2.0 / n_points
    h_initial = initial_h_values
    
    def objective(h_vector):
        corr = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(corr) * dx
    
    initial_c5 = objective(h_initial)
    best_c5 = initial_c5
    best_h = h_initial.copy()
    
    n_starts = 5
    start_time = time.time()
    
    for _ in range(n_starts):
        noise = np.random.normal(0, 0.1 * np.sqrt(1 / n_points), n_points)
        perturbed = h_initial + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if np.isclose(current_sum, 0):
            perturbed = h_initial.copy()
        else:
            perturbed = perturbed * (n_points / 2) / current_sum
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            options={'maxiter': 800, 'disp': False}
        )
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
