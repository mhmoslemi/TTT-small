def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():

        initial_h = np.zeros(70)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_target = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        excess = np.sum(h) - sum_target
        if np.abs(excess) < 1e-6:
            return h
        scale = sum_target / np.sum(h) if np.sum(h) != 0 else 1.0
        return h * scale

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    n_pop = 50
    particles = [initial_h.copy() for _ in range(n_pop)]
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(n_pop)]
    pbest = particles.copy()
    pbest_vals = [objective(p) for p in pbest]
    gbest = min(pbest, key=objective)
    gbest_val = objective(gbest)
    time_used = 0

    for iter in range(10000):
        if time_used >= budget_s:
            break
        start = time_used
        for i in range(n_pop):
            h = particles[i]
            v = velocities[i]
            h_new = h + v
            h_new = repair(h_new)
            particles[i] = h_new
            curr_val = objective(h_new)
            if curr_val < pbest_vals[i]:
                pbest[i] = h_new
                pbest_vals[i] = curr_val
