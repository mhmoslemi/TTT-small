import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    def objective(h_flat):
        h_arr = h_flat.reshape(n_points)
        corr = correlate(h_arr, 1.0 - h_arr, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2.0})
    
    # Initial optimization with L-BFGS-B for better convergence
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-6, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
