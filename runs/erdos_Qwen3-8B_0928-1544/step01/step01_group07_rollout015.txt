from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import COBYLA

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Start with initial guess, optimize with COBYLA and SLSQP
    def optimize(h0):
        result = minimize(

            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-5}
        )
        if result.success:
            return result.x, result.fun
        return h0, float('inf')
