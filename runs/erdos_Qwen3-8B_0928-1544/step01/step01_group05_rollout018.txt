import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_h(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scaling = (n_points / 2) / sum_h
        h = h * scaling
        return h

    population_size = 30
    max_generations = 200
    mutation_factor = 0.7
    crossover_prob = 0.8
    stagnation_gen = 50

    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h
        h = h * scaling
        population.append(h)

    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    improved = True

    for gen in range(max_generations):
        improved = False
        for i in range(len(population)):
            a, b, c = population[np.random.choice(len(population), 3, replace=False)]
            mutant = a + mutation_factor * (b - c)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            trial = repair_h(trial)
            if compute_c5(trial) < compute_c5(population[i]):
                population[i] = trial.copy()

        objective_SLSQP,
