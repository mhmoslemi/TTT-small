import numpy as np
from numpy.random import choice, normal, randint

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 100
    generations = 500
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    early_stop = 100

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
        if np.abs(np.sum(initial_h) - target_sum) > 1e-6:
            initial_h = initial_h / np.sum(initial_h) * target_sum
        population = [initial_h]
    else:
        population = []

    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * target_sum
        population.append(h)

    population = np.array(population)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for _ in range(generations):
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            fitness.append(c5)
        fitness = np.array(fitness)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]

        if _ >= early_stop and np.min(fitness) == best_c5:
            break

        new_pop = []
        for _ in range(pop_size):
            a = choice(pop_size, tournament_size, replace=False)
            a = a[np.argmin(fitness[a])]
            parent_a = population[a]
            if np.random.rand() < crossover_rate:
                b = choice(pop_size, replace=False)
                parent_b = population[b]
                cross = randint(1, n_points - 1)
                child = np.concatenate([parent_a[:cross], parent_b[cross:]])
