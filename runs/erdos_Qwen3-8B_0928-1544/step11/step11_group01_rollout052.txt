        new_c5 = evaluate_overlap(h)
        if new_c5 < current_c5:
            learning_rate = min(0.1, learning_rate * 1.2)
        else:
            h = best_h.copy()
            learning_rate = max(min_lr, learning_rate * 0.5)

        if iteration % 500 == 0:
            perturb = np.random.randn(n_points) * learning_rate * 0.1
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            total = np.sum(h)
            if not np.isclose(total, 0.0):
                scaling = (n_points / 2.0) / total
                h *= scaling
            perturb_c5 = evaluate_overlap(h)
            if perturb_c5 < best_c5:
                best_c5 = perturb_c5
                best_h = h.copy()

        iteration += 1

    return best_h, best_c5, n_points
