            h_minus[i] -= eps
            obj_plus = objective(h_plus)
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    best_h = h.copy()
    best_c5 = objective(h)
    time_start = time.time()
    for _ in range(500):
        if time.time() - time_start > budget_s:
            break
        grad = compute_gradient(h)
        best_j = np.argmax(grad * 0.01 + np.random.rand(n_points) * 0.01)
        basis = np.exp(-((np.arange(n_points) - best_j) * dx)**2 / 0.04)
        step_size = 0.3
        new_h = h + step_size * (basis - h)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (ones_count / new_h_sum)
        current_c5 = objective(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        h = new_h

    return best_h, best_c5, n_points
