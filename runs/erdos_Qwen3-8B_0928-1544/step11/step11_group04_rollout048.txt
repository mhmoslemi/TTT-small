import numpy as np

from scipy.optimize import minimize
import warnings

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    best_h = population[0].copy()
    best_c5 = objective(best_h)

            mutant = population[a] + F * (population[b] - population[c])

                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
