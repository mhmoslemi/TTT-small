import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h_sum = np.sum(initial_h)
        initial_h = initial_h * (ones_count / initial_h_sum)
    initial_h = np.clip(initial_h, 0, 1)
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    temp = 1.5
    cooling_rate = 0.997
    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_h = best_h.copy()
        H = np.fft.fft(current_h)
        pert_magnitude = 0.05 * (np.random.rand(len(H)) - 0.5)
        pert_phase = 0.1 * np.pi * (np.random.rand(len(H)) - 0.5)
        H_perturbed = H * (1 + pert_magnitude) * np.exp(1j * pert_phase)
        new_h = np.fft.ifft(H_perturbed).real
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            new_h = np.random.rand(n_points) * 0.5
            new_h_sum = np.sum(new_h)
            if new_h_sum == 0:
                new_h = np.random.rand(n_points) * 0.5
            new_h = new_h * (ones_count / new_h_sum)
        else:
            new_h = new_h * (ones_count / new_h_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        temp *= cooling_rate
    return best_h, best_c5, n_points
