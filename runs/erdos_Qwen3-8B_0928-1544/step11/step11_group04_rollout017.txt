import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint(h):
        return np.sum(h) - ones_count
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = differential_evolution(
        objective,
        bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        strategy='best1bin',
        popsize=50,
        mutation=(0.8, 1.0),
        recombination=0.9,
        tol=1e-10,
        maxiter=1000,
        disp=False,
        polish=True,
        seed=seed,
        x0=initial_h
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
