import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    swarm_size = 100
    inertia_start = 0.8
    inertia_end = 0.2
    phi_p = 2.0
    phi_g = 2.0
    max_velocity = 0.1
    time_start = time.time()

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        current_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / current_sum)
        particles = [initial_h_scaled]
        for _ in range(swarm_size - 1):
            h = np.random.rand(n_points) * 0.5
            current_sum = np.sum(h)
            if current_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / current_sum)
            particles.append(h_scaled)
    else:
        particles = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points) * 0.5
            current_sum = np.sum(h)
            if current_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / current_sum)
            particles.append(h_scaled)

    velocities = np.random.rand(swarm_size, n_points) * max_velocity - max_velocity / 2
    pbest = [p.copy() for p in particles]
    gbest = particles[0].copy()
    best_c5 = np.inf

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    for i in range(swarm_size):
        if objective(particles[i]) < best_c5:
            best_c5 = objective(particles[i])
            gbest = particles[i].copy()

    def update_velocity(vel, pbest, gbest, current, inertia, phi_p, phi_g, r1, r2):
        vel = inertia * vel + phi_p * r1 * (pbest - current) + phi_g * r2 * (gbest - current)
        return np.clip(vel, -max_velocity, max_velocity)

    while time.time() - time_start < budget_s:
        inertia = inertia_start + (inertia_end - inertia_start) * (time.time() - time_start) / budget_s
        stagnation = 0
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = update_velocity(velocities[i], pbest[i], gbest, particles[i], inertia, phi_p, phi_g, r1, r2)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            current_sum = np.sum(particles[i])
            if current_sum == 0:
                continue
            particles[i] *= (ones_count / current_sum)
            current_c5 = objective(particles[i])
            if current_c5 < objective(pbest[i]):
                pbest[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = particles[i].copy()
                stagnation = 0
            else:
                stagnation += 1
        if stagnation > 50:
            inertia = max(inertia_start, inertia - 0.01)
    return gbest, best_c5, n_points
