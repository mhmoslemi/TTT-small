import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    pop_size = 50
    population = [initial_h.copy()]

    for _ in range(pop_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def repair_solution(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (ones_count / h_sum)
        return h

    def nm_objective(h):
        try:
            repaired = repair_solution(h)
            return objective(repaired)
        except:
            return np.inf

    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    F = 0.7
    CR = 0.9

    de_phase_end = start_time + budget_s * 0.6

    while time.time() < de_phase_end and time.time() - start_time < budget_s:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = repair_solution(trial)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()

    simplex_size = 10
    initial_simplex = []
    for _ in range(simplex_size):
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
