import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_sum = np.sum(initial_h)
    h = initial_h * (target_sum / initial_sum)
    h = np.clip(h, 0.0, 1.0)
    
    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(obj_func, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x.reshape(n_points)
    best_c5 = result.fun
    
    start_time = time.time()
    F = 0.8
    CR = 0.9
    population_size = 100
    population = [best_h.copy()]
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (target_sum / h_sum)
        population.append(h_scaled)
    
    best_c5 = float('inf')
    best_h = population[0]
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:

            u_scaled = np.clip(u_scaled, 0.0, 1.0)
            new_c5 = obj_func(u_scaled)

                best_h = u_scaled.copy()
