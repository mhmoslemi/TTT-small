import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 50
    population = [initial_h.copy()]
    
    for _ in range(pop_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.7
    CR = 0.9
    
    while time.time() - start_time < budget_s * 0.8:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
    
    # Nelder-Mead refinement
    def obj_func(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros_like(h)
        else:
            h = h * (n_points / 2) / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        result = minimize(obj_func, best_h, method='Nelder-Mead', 
                          options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 100, 'maxfev': 100})
        best_h = result.x
        best_c5 = obj_func(best_h)
    
    return best_h, best_c5, n_points
