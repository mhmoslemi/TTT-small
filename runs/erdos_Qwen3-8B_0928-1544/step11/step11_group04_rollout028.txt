import time

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2

    population = [initial_h.copy()]
    for _ in range(49):
        h = initial_h.copy()

        h += noise
        h = np.clip(h, 0, 1)

        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.8
    CR = 0.95
    pop_size = 100
    # Use initial selection to improve early diversity
    for _ in range(pop_size - len(population)):
        h = initial_h.copy()

        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())
    
    while time.time() - start_time < budget_s:
        for i in range(pop_size):

            mutant = population[a] + F * (population[b] - population[c])

                    trial[j] = population[i][j]
