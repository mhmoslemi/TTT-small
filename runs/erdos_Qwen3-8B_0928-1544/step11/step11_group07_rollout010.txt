import time

            h = np.random.rand(n_points) * 0.5

            population.append(h)

    best_h = population[0]
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)

    def evaluate(h):
        return max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    while time.time() - start_time < budget_s:
