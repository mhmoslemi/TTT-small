import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points // 2
    population_size = 100
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        population = [initial_h]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h = h * (target_sum / np.sum(h))
            population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h = h * (target_sum / np.sum(h))
            population.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    
    start_time = time.time()
    inertia = 0.9
    inertia_decay = (0.9 - 0.1) / budget_s
    c1, c2 = 2.0, 2.0
    max_velocity = 0.5
    velocities = np.random.uniform(-max_velocity, max_velocity, (population_size, n_points))
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] += c1 * r1 * (population[i] - np.random.choice(population, 1, replace=False)[0]) + c2 * r2 * (best_h - population[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            population[i] += velocities[i]
            
            population[i] = np.clip(population[i], 0, 1)
            sum_pop = np.sum(population[i])
            if sum_pop != 0:
                population[i] = population[i] * (target_sum / sum_pop)
            
            curr_c5 = objective(population[i])
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = population[i].copy()
        
        inertia -= inertia_decay
        if inertia < 0.1:
            inertia = 0.1
    
    return best_h, best_c5, n_points
