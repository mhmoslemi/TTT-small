def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2
    population_size = 50
    mutation_rate = 0.15
    mutation_scale = 0.1
    cooling_factor = 0.995
    tabu_size = 20
    threshold = 0.02
    max_iter = int(budget_s / 10)
    start_time = time.time()

    if 'initial_h_values' in globals():

        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    for h in population:
        c5 = evaluate_erdos_solution(h) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    tabu_list = []

    while time.time() - start_time < budget_s:
        if len(population) == 0:
            break
        fitness = [evaluate_erdos_solution(h) * (2.0 / n_points) for h in population]
        inverse_fitness = [1 / (f + 1e-10) for f in fitness]
        total_inverse = sum(inverse_fitness)
        probabilities = [f / total_inverse for f in inverse_fitness]
        parent_indices = np.random.choice(range(population_size), size=population_size, p=probabilities)
        parents = [population[i] for i in parent_indices]
        children = []

        for p1, p2 in zip(parents[::2], parents[1::2]):
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum == 0:

            children.append(child)

        for child in children:
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_scale, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)

                    child = np.random.rand(n_points) * 0.5
                child = child * (ones_count / child_sum)

            for t in tabu_list:

        candidates = [p for p in population]
        for c in children:

                worst_idx = np.argmin([evaluate_erdos_solution(p) * (2.0 / n_points) for p in candidates])
                if evaluate_erdos_solution(c) * (2.0 / n_points) < evaluate_erdos_solution(candidates[worst_idx]) * (2.0 / n_points):

        population = candidates.copy()
        tabu_list.append(population[0])
        if len(tabu_list) > tabu_size:
