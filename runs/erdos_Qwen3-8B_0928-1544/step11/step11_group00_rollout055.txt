import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    iterations = 0
    max_iter = int(budget_s / 10)  # 100 iterations per second

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5 * (ones_count / (n_points / 2))
        return h * (ones_count / h_sum)

    def compute_gradient(h):
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = clip_and_scale(h_p)
            c_p = np.max(np.correlate(h_p, 1 - h_p, mode='full')) * dx

            h_m = h.copy()
            h_m[i] -= eps
            h_m = clip_and_scale(h_m)
            c_m = np.max(np.correlate(h_m, 1 - h_m, mode='full')) * dx

            grad[i] = (c_p - c_m) / (2.0 * eps)
        return grad

    while time.time() - start_time < budget_s and iterations < max_iter:
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        grad = compute_gradient(h)
        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            continue
        h_new = h_new * (ones_count / h_new_sum)
        h = h_new
        iterations += 1

    return best_h, best_c5, n_points
