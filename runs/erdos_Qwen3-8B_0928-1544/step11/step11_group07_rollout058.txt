import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = h * (ones_count / np.sum(h))
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    temp = 1.0
    cooling = 0.995
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h_copy = h.copy()
        idxs = np.random.choice(n_points, min(5, n_points//2), replace=False)
        for i in idxs:
            h_copy[i] += np.random.uniform(-0.05, 0.05)
        h_copy = np.clip(h_copy, 0, 1)
        h_sum = np.sum(h_copy)
        if h_sum == 0:
            h_copy = np.random.rand(n_points) * 0.5
            h_copy = h_copy * (ones_count / np.sum(h_copy))
        else:
            h_copy = h_copy * (ones_count / h_sum)
        new_c5 = compute_c5(h_copy, n_points)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5)/temp):
            best_c5 = new_c5
            best_h = h_copy.copy()
        temp *= cooling
        h = h_copy
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n_points)
