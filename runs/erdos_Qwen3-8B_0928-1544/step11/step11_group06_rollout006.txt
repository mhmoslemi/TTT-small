    best_h = h.copy()
    best_C5 = compute_C5(h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        def sub_obj(x):
            h_opt = x
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            return corr_opt[k]
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        res = minimize(sub_obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        
        if res.success:
            new_C5 = compute_C5(res.x)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = res.x.copy()
                h = res.x.copy()
    
    return best_h, best_C5, n_points
