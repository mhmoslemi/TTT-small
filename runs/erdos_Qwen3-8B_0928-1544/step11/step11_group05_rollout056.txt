            best_c5 = objectives[best_idx]
        
        # Generate candidate from best
        perturbation = np.random.normal(0, sigma, n_points)
        candidate = population[best_idx] + perturbation
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if candidate_sum == 0:
            candidate = np.random.rand(n_points) * 0.5
            candidate_sum = np.sum(candidate)
            candidate = candidate * (target_sum / candidate_sum)
        else:
            candidate = candidate * (target_sum / candidate_sum)
        
        candidate_c5 = compute_c5(candidate)
        
        # Replace worst
        worst_idx = np.argmax(objectives)
        if candidate_c5 < objectives[worst_idx]:
            population[worst_idx] = candidate.copy()
        
        # Adaptive sigma
        if candidate_c5 < objectives[best_idx]:
            sigma *= sigma_increase
        else:
            sigma *= sigma_decay
    
    return best_h, best_c5, n_points
