import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points // 2
    pop_size = 100
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = (initial_h * (target_sum / np.sum(initial_h))).clip(0, 1)
        population = [initial_h]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points) * 0.3
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = (h * (target_sum / h_sum)).clip(0, 1)
            population.append(h)
    else:
        population = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = (h * (target_sum / h_sum)).clip(0, 1)
            population.append(h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    inertia = 0.9
    max_vel = 0.5
    c1, c2 = 2.0, 2.0
    velocities = np.random.uniform(-max_vel, max_vel, (pop_size, n_points))

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            vel = velocities[i] + c1 * r1 * (population[i] - best_h) + c2 * r2 * (np.array([np.mean(population[i])]*n_points) - best_h)
            vel = np.clip(vel, -max_vel, max_vel)
            velocities[i] = vel
            new_h = population[i] + vel
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum == 0:
                continue
            new_h = (new_h * (target_sum / new_h_sum)).clip(0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            population[i] = new_h.copy()

        inertia -= (0.9 - 0.1) / (budget_s * 2)
        if inertia < 0.1:
            inertia = 0.1

    return best_h, best_c5, n_points
