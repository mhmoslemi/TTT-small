import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initialize with given or random starting point
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        total = np.sum(h)
        scaling = target_sum / total
        h = h * scaling
        h = np.clip(h, 0.0, 1.0)
    else:
        h = np.random.rand(n_points)
        total = np.sum(h)
        scaling = target_sum / total
        h = h * scaling
        h = np.clip(h, 0.0, 1.0)

    # Objective: minimize max correlation
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Constraint to fix the sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    # Optimization options
    options = {
        'maxiter': 300,
        'ftol': 1e-8,
        'disp': False,
        'gtol': 1e-8
    }

    # Run the optimization
    result = minimize(
        objective, h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    # Final check
    best_h = result.x
    best_c5 = objective(best_h)
    assert np.isclose(np.sum(best_h), target_sum), f"Sum constraint violated: {np.sum(best_h)}"
    assert np.all(0 <= best_h <= 1), "Bounds violated"

    return best_h, best_c5, n_points
