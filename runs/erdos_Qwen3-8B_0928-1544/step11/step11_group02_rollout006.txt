        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h_clipped)
        scale = target_sum / sum_clipped
        return h_clipped * scale
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)
    tabu_list = []
    tabu_size = 20
    temp = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        h_new = h_current.copy()
        num_perturb = np.random.randint(5, 15)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-0.1, 0.1)
        h_new = project(h_new)
        new_c5 = compute_c5(h_new)
        
        in_tabu = False
        rounded_new = np.round(h_new, 5)
        for t in tabu_list:
            if np.allclose(t, rounded_new, atol=1e-5):
                in_tabu = True
                break
        if not in_tabu:
            delta = new_c5 - best_c5
            if delta < 0:
                h_current = h_new.copy()
                best_c5 = new_c5
            else:
                prob = np.exp(-delta / temp)
                if np.random.rand() < prob:
                    h_current = h_new.copy()
                    best_c5 = new_c5
            tabu_list.append(h_current.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        temp *= cooling_rate
        if temp < 0.1:
            temp = 0.1
        if time.time() - start_time > budget_s - 500:
            cooling_rate = 0.99
    
    return best_h, best_c5, n_points
