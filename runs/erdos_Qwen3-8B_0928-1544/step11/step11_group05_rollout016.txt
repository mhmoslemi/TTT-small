import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    best_h = None
    best_c5 = float('inf')
    sigma = 0.2
    max_iter = 1000

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_sum = np.sum(initial_h)
        scale_factor = target_sum / initial_sum
        initial_h_scaled = np.clip(initial_h * scale_factor, 0, 1)
        best_h = initial_h_scaled.copy()
        best_c5 = objective(best_h)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        scale_factor = target_sum / h_sum
        initial_h_scaled = np.clip(h * scale_factor, 0, 1)
        best_h = initial_h_scaled.copy()
        best_c5 = objective(best_h)

    start_time = time.time()
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:
        noise = np.random.normal(0, sigma, n_points)
        new_h = best_h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            iter_count += 1
            continue
        scale_factor = target_sum / new_h_sum
        new_h_scaled = new_h * scale_factor
        new_h_scaled = np.clip(new_h_scaled, 0, 1)
        new_c5 = objective(new_h_scaled)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h_scaled.copy()
            sigma *= 1.05
        else:
            sigma *= 0.95
        sigma = max(0.01, min(sigma, 1.0))
        iter_count += 1

    return best_h, best_c5, n_points
