            h_new = np.clip(h_new, 0.0, 1.0)
            sum_h_new = np.sum(h_new)
            if sum_h_new == 0:
                continue
            scale = ones_count / sum_h_new
            h = h_new * scale
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
