import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h_values = np.random.rand(n_points) * 0.5
    initial_h_values *= (ones_count / np.sum(initial_h_values))
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    iterations = 0
    max_iterations = 100  # Balancing exploration and computational time
    while time.time() - start_time < budget_s and iterations < max_iterations:
        h = cp.Variable(n_points)
        tau = cp.Variable()
        # Ensure constraints on h
        constraints = [
            cp.sum(h) == ones_count,
            0 <= h <= 1,
            cp.norm(np.fft.fft(h), 'inf') <= tau
        ]
        # Objective is to minimize the maximum Fourier magnitude
        prob = cp.Problem(cp.Minimize(tau), constraints)
        try:
            prob.solve()  # Use default solver, may vary based on environment
            if prob.status == 'optimal':
                current_h = h.value
                # Clip and rescale
                current_h_clipped = np.clip(current_h, 0, 1)
                sum_clipped = np.sum(current_h_clipped)
                scaling_factor = (ones_count / sum_clipped) if sum_clipped != 0 else n_points / 2
                current_h = current_h_clipped * scaling_factor
                # Evaluate C5
                current_c5 = evaluate_erdos_solution(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        except:
            # Fallback in case of solver failure
            pass
        iterations += 1
    return best_h, best_c5, n_points
