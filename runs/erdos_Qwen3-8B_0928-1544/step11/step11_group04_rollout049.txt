    best_h = np.array(best_h)
    result = minimize(nm_objective, best_h, method='Nelder-Mead', options={'xatol': 1e-5, 'fatol': 1e-5})
    best_h = result.x
    
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
