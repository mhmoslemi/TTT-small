import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    h_initial = np.random.rand(n_points) * 0.5
    h_initial_sum = np.sum(h_initial)
    h_initial = h_initial * (ones_count / h_initial_sum)
    h_initial = np.clip(h_initial, 0.0, 1.0)

    def objective(z):
        h = 1.0 / (1.0 + np.exp(-z))
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        scaling = ones_count / h_sum
        h = h * scaling
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def grad_objective(z, epsilon=1e-6):
        n = len(z)
        grad = np.zeros(n)
        for i in range(n):
            z_plus = z.copy()
            z_plus[i] += epsilon
            obj_plus = objective(z_plus)
            z_minus = z.copy()
            z_minus[i] -= epsilon
            obj_minus = objective(z_minus)
            grad[i] = (obj_plus - obj_minus) / (2.0 * epsilon)
        return grad

    z_initial = np.log(h_initial / (1.0 - h_initial))
    z_initial = np.clip(z_initial, -6.0, 6.0)

    result_l_bfgs = minimize(
        objective,
        z_initial,
        method='L-BFGS-B',
        bounds=[(-6.0, 6.0) for _ in range(n_points)],
        jac=grad_objective,
        options={'maxiter': 50, 'ftol': 1e-6}
    )

    z_best_l_bfgs = result_l_bfgs.x
    h_best_l_bfgs = 1.0 / (1.0 + np.exp(-z_best_l_bfgs))
    h_best_l_bfgs = np.clip(h_best_l_bfgs, 0.0, 1.0)
    h_best_l_bfgs_sum = np.sum(h_best_l_bfgs)
    scaling = ones_count / h_best_l_bfgs_sum
    h_best_l_bfgs = h_best_l_bfgs * scaling
    h_best_l_bfgs = np.clip(h_best_l_bfgs, 0.0, 1.0)
    best_c5_l_bfgs = objective(z_best_l_bfgs)

    result_nelder = minimize(
        objective,
        z_best_l_bfgs,
        method='Nelder-Mead',
        options={'xatol': 1e-6, 'disp': False}
    )

    z_best_nelder = result_nelder.x
    h_best_nelder = 1.0 / (1.0 + np.exp(-z_best_nelder))
    h_best_nelder = np.clip(h_best_nelder, 0.0, 1.0)
    h_best_nelder_sum = np.sum(h_best_nelder)
    scaling = ones_count / h_best_nelder_sum
    h_best_nelder = h_best_nelder * scaling
    h_best_nelder = np.clip(h_best_nelder, 0.0, 1.0)
    best_c5_nelder = objective(z_best_nelder)

    if best_c5_nelder < best_c5_l_bfgs:
        h_best = h_best_nelder
        best_c5 = best_c5_nelder
    else:
        h_best = h_best_l_bfgs
        best_c5 = best_c5_l_bfgs

    return h_best, best_c5, n_points
