import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    start_time = time.time()

    # Load initial_h_values
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
        h = np.clip(h, 0.0, 1.0)

    def objective(z):
        h_ = 1 / (1 + np.exp(-z))
        h_sum = np.sum(h_)
        if h_sum == 0:
            return np.inf
        scaling = ones_count / h_sum
        h_scaled = h_ * scaling
        h_clamped = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_clamped, 1 - h_clamped, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    def clip_z(z):
        return np.clip(z, -6, 6)

    z_initial = np.log(np.clip(h, 1e-6, 1 - 1e-6) / (1 - np.clip(h, 1e-6, 1 - 1e-6)))
    z_initial = clip_z(z_initial)

    best_z = z_initial
    best_c5 = objective(best_z)

    # L-BFGS optimization
    while time.time() - start_time < budget_s * 0.7:
        res = minimize(objective, best_z, method='L-BFGS-B', bounds=[(-6, 6)]*n_points, tol=1e-6)
        if res.success:
            best_z = res.x
            best_c5 = objective(best_z)
        else:
            break

    # Nelder-Mead optimization
    h_clamped = 1 / (1 + np.exp(-best_z))
    h_clamped_sum = np.sum(h_clamped)
    h_clamped = h_clamped * (ones_count / h_clamped_sum)
    h_clamped = np.clip(h_clamped, 0.0, 1.0)
    z_nelder = np.log(np.clip(h_clamped, 1e-6, 1 - 1e-6) / (1 - np.clip(h_clamped, 1e-6, 1 - 1e-6)))
    z_nelder = clip_z(z_nelder)

    res = minimize(objective, z_nelder, method='Nelder-Mead', bounds=[(-6, 6)]*n_points, tol=1e-4, options={'maxiter': 100})
    best_z = res.x
    best_c5 = objective(best_z)

    h_final = 1 / (1 + np.exp(-best_z))
    h_final_sum = np.sum(h_final)
    h_final = h_final * (ones_count / h_final_sum)
    h_final = np.clip(h_final, 0.0, 1.0)

    return h_final, best_c5, n_points
