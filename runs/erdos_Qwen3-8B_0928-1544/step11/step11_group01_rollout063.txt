import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    learning_rate = 0.1
    decay_rate = 0.995
    max_iterations = 10000
    initial_h_values = kwargs.get('initial_h_values', None)

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
    h_sum = np.sum(h)
    h = h * (ones_count / h_sum)
    h = np.clip(h, 0.0, 1.0)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] += 1 - h[int(i + lag)]
            if 0 <= i - lag < n_points:
                grad[i] -= h[int(i - lag)]

            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5, n_points
