import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    pop_size = 50
    sigma = 0.15
    max_iter = 1000
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        scale = target_sum / np.sum(h)
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
    else:
        h = np.random.rand(n_points) * 0.5
        scale = target_sum / np.sum(h)
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        pop = []
        for _ in range(pop_size):
            delta = np.random.normal(0, sigma, n_points)
            h_trial = best_h + delta
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_sum = np.sum(h_trial)
            if h_sum == 0:
                continue
            h_scaled = h_trial * (target_sum / h_sum)
            h_scaled = np.clip(h_scaled, 0.0, 1.0)
            pop.append(h_scaled)
        
        scores = [evaluate_erdos_solution(h) for h in pop]
        best_idx = np.argmin(scores)
        current_c5 = scores[best_idx]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[best_idx].copy()
            sigma *= 0.95
        else:
            sigma *= 1.05
        
        if np.abs(best_c5 - current_c5) < 1e-6:
            break
    
    return best_h, best_c5, n_points
