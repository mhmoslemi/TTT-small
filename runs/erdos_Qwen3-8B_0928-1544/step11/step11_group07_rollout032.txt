                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    population = np.array(population, dtype=np.float64)
    best_h = population[0]
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.random.rand(n_points) * 0.5
                trial_sum = np.sum(trial)
                if trial_sum == 0:
                    trial = np.random.rand(n_points) * 0.5
                trial = trial * (ones_count / trial_sum)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = max(np.correlate(trial, 1 - trial, mode='full')) * (2.0 / n_points)
            if current_c5 < max(np.correlate(population[i], 1 - population[i], mode='full')) * (2.0 / n_points):
                population[i] = trial
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = trial
    return best_h, best_c5, n_points
