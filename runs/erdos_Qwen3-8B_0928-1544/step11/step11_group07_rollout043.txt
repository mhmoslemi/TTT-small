import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    mutation_rate = 0.15
    mutation_scale = 0.1
    cooling_factor = 0.995
    tabu_size = 20

    if 'initial_h_values' in globals():

        for _ in range(population_size - 1):

            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            init_pop.append(h_scaled)
    else:
        init_pop = [np.random.rand(n_points) * 0.5 for _ in range(population_size)]
        for i in range(population_size):
            h = init_pop[i]
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            init_pop[i] = h * (ones_count / h_sum)

    population = [np.clip(h, 0, 1) for h in init_pop]
    personal_best = population.copy()
    best_h = population[0]
    best_c5 = compute_c5(best_h, n_points)

    tabu = set()
    start_time = time.time()
    max_iter = int(budget_s / 10)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(population_size):
            # Tournament selection
            idxs = np.random.choice(population_size, 5, replace=False)
            best_idx = idxs[np.argmin([compute_c5(population[j], n_points) for j in idxs])]
            parent = population[best_idx].copy()
            
            # Crossover

                jdx = np.random.choice(population_size, replace=False)
                cross = np.random.rand(n_points)
                child = parent * cross + population[jdx] * (1 - cross)
            else:
                child = parent.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_scale, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Renormalize
            child_sum = np.sum(child)
            if child_sum != 0:
                child *= (ones_count / child_sum)
            else:
                child = np.random.rand(n_points) * 0.5
                child *= (ones_count / np.sum(child))
            
            # Tabu check
            if len(tabu) >= tabu_size:
                tabu.discard(min(tabu, key=lambda x: np.linalg.norm(x - child)))
            if np.any(np.abs(child - tabu) < 0.02):
                continue
            tabu.add(tuple(child))
            new_population.append(child)
        
        # Elitism
        new_population += [best_h]

def compute_c5(h, n_points):
    correlate = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(correlate)
    return max_corr * (2.0 / n_points)
