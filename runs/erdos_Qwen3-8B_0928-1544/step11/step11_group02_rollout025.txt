import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    population_size = 100
    F = 0.5
    CR = 0.8
    best_c5 = float('inf')
    best_h = None

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled.copy())

    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s * 0.7:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0, 1)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = u_scaled.copy()
            if new_c5 < objective(population[i]):
                population[i] = u_scaled.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining < 500:
            F = 0.8
            CR = 0.95
        else:
            F = 0.5
            CR = 0.8

    def obj_with_penalty(h_vec):
        h = h_vec.copy()
        c5 = objective(h)
        penalty = 1e6 * (np.sum(h) - ones_count)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(obj_with_penalty, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
    best_h = result.x.copy()
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
