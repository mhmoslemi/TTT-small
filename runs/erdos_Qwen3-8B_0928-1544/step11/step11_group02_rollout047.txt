import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 150
    target_sum = n_points // 2  # 75
    
    # Initialize h
    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        h = np.interp(np.linspace(0, 2, n_points), x_initial, initial_h_values)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    tabu_list = set()
    T = 0.5
    alpha = 0.99
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Perturb h
        h_candidate = h.copy()
        num_flips = np.random.randint(5, 15)
        flip_indices = np.random.choice(n_points, num_flips, replace=False)
        h_candidate[flip_indices] = 1 - h_candidate[flip_indices]
        
        # Clip and scale
        h_clipped = np.clip(h_candidate, 0, 1)
        h_sum_clipped = np.sum(h_clipped)
        if h_sum_clipped == 0:
            continue
        h_scaled = h_clipped * (target_sum / h_sum_clipped)
        
        # Tabu check using hash
        h_hash = tuple(h_scaled.round(6))
        if h_hash in tabu_list:
            continue
        tabu_list.add(h_hash)
        if len(tabu_list) > 20:
            tabu_list.remove(min(tabu_list))
        
        new_c5 = evaluate_c5(h_scaled)
        if new_c5 < best_c5 or np.random.rand() < np.exp((new_c5 - best_c5) / T):
            best_h = h_scaled.copy()
            best_c5 = new_c5
        
        # Local search with small perturbation
        if np.random.rand() < 0.1:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_local = h_scaled + perturbation
            h_local = np.clip(h_local, 0, 1)
            h_sum_loc = np.sum(h_local)
            h_local_scaled = h_local * (target_sum / h_sum_loc)
            local_c5 = evaluate_c5(h_local_scaled)
            if local_c5 < best_c5:
                best_h = h_local_scaled.copy()
                best_c5 = local_c5
        
        T *= alpha
    
    return best_h, best_c5, n_points
