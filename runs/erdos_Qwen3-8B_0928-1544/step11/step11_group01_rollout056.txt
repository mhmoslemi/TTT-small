import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 200
    F = 0.9
    CR = 0.95
    mutation_ratio = 0.3
    selection_pressure = 0.95
    
    def initial_h_values():
        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            return h_scaled
        else:
            h = np.zeros(n_points)
            for i in range(n_points):
                if i < n_points // 2:
                    h[i] = i / (n_points // 2)
                else:
                    h[i] = 1 - (i - n_points // 2) / (n_points // 2)
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            return h_scaled
    
    initial_h = initial_h_values()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[i]):
                new_population.append(u_scaled)

                new_population.append(population[i])
        population = new_population[:population_size]
        for ind in population:
            if objective(ind) < best_c5:
                best_c5 = objective(ind)
                best_h = ind.copy()
    
    return best_h, best_c5, n_points
