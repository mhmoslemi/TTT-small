    return np.max(corr) * (2.0 / n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_sum = np.sum(initial_h)
        if initial_sum == 0:
            initial_h = np.random.rand(n_points) * 0.5
        initial_h = initial_h * (target_sum / initial_sum)
        initial_h = np.clip(initial_h, 0.0, 1.0)
        best_h = initial_h.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / h_sum)
        h = np.clip(h, 0.0, 1.0)
        best_h = h.copy()
    
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    sigma = 0.1
    
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, sigma, n_points)
        candidate = best_h + noise
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate_sum = np.sum(candidate)
        if candidate_sum == 0:
            continue
        candidate_scaled = candidate * (target_sum / candidate_sum)
        candidate_scaled = np.clip(candidate_scaled, 0.0, 1.0)
        new_c5 = compute_c5(candidate_scaled, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate_scaled.copy()
            sigma *= 1.05
        else:
            sigma *= 0.95
    
    return best_h, best_c5, n_points
