import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    def project_and_rescale(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clipped)
        if np.isclose(total, 0.0):
            return np.ones(n_points) * (n_points / (2.0 * n_points))
        scaling = (n_points / 2.0) / total
        return h_clipped * scaling

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = project_and_rescale(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    learning_rate = 0.1
    min_lr = 1e-5
    iteration = 0
    max_iterations = int(budget_s * 20)

    while iteration < max_iterations and time.time() - start_time < budget_s:
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        overlap = np.correlate(h, 1 - h, mode='full')
        max_overlap = overlap.max()
        max_lags = np.where(overlap == max_overlap)[0]

        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < 0:
                continue
            for j in range(n_points):
                if j + lag < n_points:
                    gradient[j] -= (1 - h[j + lag])
                if j - lag >= 0:
                    gradient[j] -= h[j - lag]

        h -= learning_rate * gradient
        h = project_and_rescale(h)

        if np.random.rand() < 0.05:
            perturb = np.random.randn(n_points) * learning_rate * 0.1
            h += perturb
            h = project_and_rescale(h)

        if np.abs(current_c5 - best_c5) < 1e-7:
            learning_rate *= 0.5
        else:
            learning_rate = min(0.1, learning_rate * 1.1)
        iteration += 1
    return best_h, best_c5, n_points
