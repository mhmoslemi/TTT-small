import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    HM_size = 50
    HM = [initial_h.copy()]
    for _ in range(HM_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        HM.append(h.copy())

    def get_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = HM[0].copy()
    best_c5 = get_c5(best_h)
    start_time = time.time()
    BW = 0.2
    BW_decay = 0.995

    while time.time() - start_time < budget_s:
        idx = np.random.randint(HM_size)
        trial = HM[idx].copy()
        perturbation = np.random.uniform(-BW, BW, n_points)
        trial += perturbation
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * (ones_count / trial_sum)
        current_c5 = get_c5(trial)
        worst_idx = np.argmax([get_c5(h) for h in HM])
        if current_c5 < get_c5(HM[worst_idx]):
            HM[worst_idx] = trial.copy()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = trial.copy()
        BW *= BW_decay

    return best_h, best_c5, n_points
