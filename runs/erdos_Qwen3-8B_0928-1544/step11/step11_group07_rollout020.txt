def compute_gradient_k(h, k, n_points):
    gradient = np.zeros(n_points)
    for j in range(n_points):
        term1 = 0.0

        term2 = 0.0
        if 0 <= j - k < n_points:
            term2 = h[j - k]

    return gradient
