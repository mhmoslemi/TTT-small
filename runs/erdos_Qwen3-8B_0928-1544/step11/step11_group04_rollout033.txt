            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
    
    # Apply Nelder-Mead on best_h
    dx = 2.0 / n_points
    def objective_nm(h_vec):
        h = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (ones_count / h_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective_nm, best_h, method='Nelder-Mead', bounds=bounds, options={'xatol': 1e-4, 'fatol': 1e-4, 'maxfun': 500})
    
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
