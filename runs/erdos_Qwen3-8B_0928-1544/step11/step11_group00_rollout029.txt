            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points) * 0.5

                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):

        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()

    velocities = np.random.rand(swarm_size, n_points) * max_velocity

    global_best = best_h.copy()
    inertia = inertia_start
