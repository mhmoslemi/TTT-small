                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
    
    # SLSQP phase
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        def objective_SLSQP(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / n_points)
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        x0 = best_h.copy()
        result = minimize(objective_SLSQP, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-6, 'maxiter': 300})
        if result.success:
            best_h = result.x.copy()
            best_c5 = objective_SLSQP(best_h)
    
    return best_h, best_c5, n_points
