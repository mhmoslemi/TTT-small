            h = np.zeros_like(h)
        
        if abs(current_c5 - best_c5) < improvement_threshold:
            improvement_count += 1
            if improvement_count >= no_improvement_patience:
                noise = np.random.normal(0, 0.05, n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (ones_count / h_sum)
                improvement_count = 0
        else:
            improvement_count = 0
        
        learning_rate *= decay_rate
    
    return best_h, best_c5, n_points
