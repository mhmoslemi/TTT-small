    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (target_sum / initial_sum)
        best_h = initial_h_scaled
    else:
        best_h = np.random.rand(n_points) * 0.5
        best_h_sum = np.sum(best_h)
        best_h = best_h * (target_sum / best_h_sum)
        best_h = np.clip(best_h, 0.0, 1.0)
    
    best_c5 = evaluate_erdos_solution(best_h)
    sigma = 0.15
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, sigma, n_points)
        candidate = best_h + noise
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate_sum = np.sum(candidate)
        if candidate_sum == 0:
            continue
        candidate_scaled = candidate * (target_sum / candidate_sum)
        candidate_scaled = np.clip(candidate_scaled, 0.0, 1.0)
        new_c5 = evaluate_erdos_solution(candidate_scaled)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate_scaled.copy()
            sigma = np.clip(sigma * 1.1, 0.01, 1.0)
        else:
            sigma = np.clip(sigma * 0.9, 0.01, 1.0)
    
    return best_h, best_c5, n_points
