        new_c5 = evaluate_overlap(h)
        if new_c5 < best_c5:
            learning_rate = min(0.1, learning_rate * 1.2)
        else:
            learning_rate = max(min_lr, learning_rate * 0.5)

        if iteration % 500 == 0:
            perturb = np.random.randn(n_points) * learning_rate * 0.1
            h += perturb
            h = project_and_rescale(h)
            if evaluate_overlap(h) < best_c5:
                best_h = h.copy()
                best_c5 = evaluate_overlap(h)

        iteration += 1

    return best_h, best_c5, n_points
