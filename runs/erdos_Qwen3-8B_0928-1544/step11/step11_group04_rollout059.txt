import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        HM.append(h)

        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()

    BW = 0.2
    start_time = time.time()
    
    while time.time() - start_time < budget_s:

        for j in range(n_points):

        current_c5 = objective(new_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        
        worst_index = np.argmax([objective(h) for h in HM])
