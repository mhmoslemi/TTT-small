        h -= alpha * grad
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Scale to sum to 1
        h_sum = np.sum(h)
        if h_sum == 0:
            break
        h = h * (target_sum / h_sum)
    return h

import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    population_size = 50

    # Initialize population using initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        # Scale to meet sum constraint
        initial_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (target_sum / initial_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])

    start_time = time.time()

    # Refinement function
    def refine_candidate(h, n_points, dx, alpha=0.01, steps=3):
        h_refined = h.copy()
        for _ in range(steps):
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (n_points - 1)
            grad = np.zeros_like(h_refined)
            for j in range(n_points):
                j_plus_k = (j + k) % n_points
                j_minus_k = (j - k) % n_points
                grad[j] = dx * (1 - h_refined[j_plus_k] - h_refined[j_minus_k])
            h_refined -= alpha * grad
            h_refined = np.clip(h_refined, 0.0, 1.0)
            h_sum = np.sum(h_refined)
            if h_sum == 0:
                break
            h_refined = h_refined * (target_sum / h_sum)
        return h_refined

    F = 0.8
    CR = 0.95

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]

            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)

            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]

            # Clip to [0,1]
            u = np.clip(u, 0.0, 1.0)

            # Scale to meet sum constraint
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (target_sum / u_sum)
            u_scaled = np.clip(u_scaled, 0.0, 1.0)

            # Refine the candidate
            dx = 2.0 / n_points
            refined_u = refine_candidate(u_scaled, n_points, dx, alpha=0.01, steps=3)

            # Evaluate
            new_c5 = objective(refined_u)

            # Check if improvement
            if new_c5 < objective(population[i]):
                population[i] = refined_u
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = refined_u.copy()

    return best_h, best_c5, n_points
