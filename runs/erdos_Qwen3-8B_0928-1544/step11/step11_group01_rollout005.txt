import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    # Load initial values if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = np.clip(h, 0.0, 1.0)
        if np.isclose(np.sum(h), n_points / 2):
            pass
        else:
            h = h * (n_points / 2.0) / np.sum(h)
    else:
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iter = int(budget_s * 20)
    learning_rate = 0.1
    min_lr = 1e-5
    iter_counter = 0

    while iter_counter < max_iter and time.time() - start_time < budget_s:
        # Compute overlap and find maximum lags
        overlap = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(overlap)
        max_lags = np.where(overlap == max_overlap)[0]

        # Compute gradient
        gradient = np.zeros(n_points)
        for lag in max_lags:
            for j in range(n_points):
                if 0 <= j + lag < n_points:
                    gradient[j] -= (1 - h[j + lag])

        # Update h
        h -= learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            h = np.full(n_points, 0.5)
        else:
            h = h * (n_points / 2.0) / total

        # Evaluate new c5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        else:
            h = best_h.copy()

        # Adapt learning rate
        if current_c5 < best_c5:
            learning_rate = min(0.1, learning_rate * 1.1)
        else:
            learning_rate = max(min_lr, learning_rate * 0.9)

        # Random perturbation
        if iter_counter % 50 == 0:
            perturb = np.random.randn(n_points) * learning_rate * 0.1
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            total = np.sum(h)
            if total == 0:
                h = np.full(n_points, 0.5)
            else:
                h = h * (n_points / 2.0) / total
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5

        iter_counter += 1

    return best_h, best_c5, n_points
