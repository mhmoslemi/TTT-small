import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_sum = np.sum(initial_h)
        if initial_sum == 0:
            initial_h = np.random.rand(n_points) * 0.5
        else:
            initial_h = initial_h * (target_sum / initial_sum)
    else:
        initial_h = np.random.rand(n_points) * 0.5
        initial_sum = np.sum(initial_h)
        initial_h = initial_h * (target_sum / initial_sum)
    initial_h = np.clip(initial_h, 0, 1)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    time_start = time.time()
    population = [initial_h.copy()]
    mut_strength = 0.3
    cr = 0.8
    pop_size = 20
    generations = 50
    
    for gen in range(generations):
        if time.time() - time_start > budget_s - 5:
            break
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + mut_strength * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            if mutant_sum == 0:
                continue
            mutant = mutant * (target_sum / mutant_sum)
            mutant = np.clip(mutant, 0, 1)
            new_pop.append(mutant)
        population = new_pop
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Local refinement
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
    if res.success:
        best_h = res.x
    
    return best_h, best_c5, n_points
