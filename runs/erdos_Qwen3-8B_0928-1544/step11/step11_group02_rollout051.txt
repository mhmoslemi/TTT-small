import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2

        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            perturb = np.random.rand(n_points) * 0.1
            h_scaled = np.clip(h_scaled + perturb, 0, 1)
            h_scaled = h_scaled * (ones_count / np.sum(h_scaled))
            population.append(h_scaled)
    else:
