def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    population_size = 100
    F = 0.5
    CR = 0.8
    best_c5 = float('inf')
    best_h = None

        initial_h_sum = np.sum(initial_h)
        scale = ones_count / initial_h_sum
        initial_h_scaled = initial_h * scale
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            if np.sum(h) == 0:
                h = np.random.rand(n_points) * 0.5
            scale = ones_count / np.sum(h)
            population.append(h * scale)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            if np.sum(h) == 0:
                h = np.random.rand(n_points) * 0.5
            scale = ones_count / np.sum(h)
            population.append(h * scale)
    
    def obj_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    start = time.time()
    while time.time() - start < budget_s * 0.7:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
