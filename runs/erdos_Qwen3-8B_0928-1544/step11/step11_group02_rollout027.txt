                h_scaled = np.random.rand(n_points) * 0.6
                h_scaled = h_scaled * (50.0 / np.sum(h_scaled))
            else:
                h_scaled = h_clipped * (50.0 / h_sum)
            population[i] = h_scaled
            current_c5 = objective(population[i])
            if current_c5 < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = current_c5
                if current_c5 < global_best_value:
                    global_best = population[i].copy()
                    global_best_value = current_c5
                    best_c5 = current_c5

    return global_best, best_c5, n_points

def objective(h):
    corr = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / 100
    return np.max(corr) * dx
