        current_fft = np.fft.fft(best_h)
        delta_real = np.random.normal(0, 0.02, len(current_fft))
        delta_imag = np.random.normal(0, 0.02, len(current_fft))
        perturbed_fft = current_fft + delta_real + 1j * delta_imag
        h_perturbed = np.fft.ifft(perturbed_fft).real
        h_clipped = np.clip(h_perturbed, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            continue
        scaling_factor = sum_h / sum_clipped
        h_scaled = h_clipped * scaling_factor
        new_c5 = compute_c5(h_scaled)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_scaled.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                best_c5 = new_c5
                best_h = h_scaled.copy()
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
