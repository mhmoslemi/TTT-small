        scaling = (n_points / 2) / total
        h = h * scaling

    best_h = h.copy()
    best_c5 = compute_overlap(h)
    start_time = time.time()
    iteration = 0
    max_iterations = int(budget_s * 20)
    learning_rate = 0.1
    min_lr = 1e-5

    while iteration < max_iterations and time.time() - start_time < budget_s:
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        overlap = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(overlap)
        max_lags = np.where(overlap == max_overlap)[0]
        gradient = np.zeros(n_points)
        for lag in max_lags:
            for j in range(n_points):
                part1 = 1 - h[j + lag] if (j + lag < n_points) else 0.0
                part2 = h[j - lag] if (j - lag >= 0) else 0.0
                gradient[j] += part1 - part2
        h -= learning_rate * gradient
        h = project_and_rescale(h)
        new_c5 = compute_overlap(h)
        if new_c5 < best_c5:
            learning_rate = min(0.1, learning_rate * 1.2)
        else:
            h = best_h.copy()
            learning_rate = max(min_lr, learning_rate * 0.5)
        iteration += 1
        if iteration % 500 == 0:
            perturb = np.random.randn(n_points) * learning_rate * 0.1
            h += perturb
            h = project_and_rescale(h)
            if compute_overlap(h) < best_c5:
                best_h = h.copy()
                best_c5 = compute_overlap(h)

    return best_h, best_c5, n_points
