import time

def run(seed=42, budget_s=1100, **kwargs):

        h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    learning_rate = 0.05
    max_iter = int(budget_s / 50)
    eps = 1e-5

    for _ in range(max_iter):
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = max(corr) * dx
        k_max = np.argmax(corr)

        # Gradient approximation for k_max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k_max] * dx

            h_minus = h.copy()
            h_minus[i] -= eps
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k_max] * dx

            gradient[i] = (val_plus - val_minus) / (2 * eps)

        # Update h
        h -= learning_rate * gradient

        # Normalize sum
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / np.sum(h))

        # Clip to [0, 1]
        h = np.clip(h, 0, 1)

        # Evaluate new C5
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = max(corr_new) * dx
        if new_c5 < best_c5:
