import time

def run(seed=42, budget_s=1100, **kwargs):

            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()

    velocities = np.random.rand(swarm_size, n_points) * vel_limit

    pbest_c5 = [objective(h) for h in population]

    w = w_start
    stagnation = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):

            velocities[i] = w * velocities[i] + phi_p * r1 * (pbest[i] - population[i]) + phi_g * r2 * (gbest - population[i])
