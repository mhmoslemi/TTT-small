import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    pop_size = 100
    F = 0.5
    CR = 0.8
    best_c5 = float('inf')
    best_h = None

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.zeros_like(h)
        return h_clipped * (ones_count / h_sum)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        pop = [np.random.rand(n_points) * 0.5 for _ in range(pop_size - 1)]
        pop.append(initial_h)
    else:
        pop = [np.random.rand(n_points) * 0.5 for _ in range(pop_size)]
        for i in range(pop_size):
            h = pop[i]
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            pop[i] = h * (ones_count / h_sum)

    for h in pop:
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()

    start = time.time()
    while time.time() - start < budget_s - 300:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            u = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = mutant[j]
            u_proj = project(u)
            curr_c5 = compute_c5(u_proj)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = u_proj.copy()
            if curr_c5 < compute_c5(pop[i]):
                pop[i] = u_proj.copy()
        F = 0.8 if time.time() - start > 700 else 0.5
        CR = 0.95 if time.time() - start > 700 else 0.8

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        penalty = 1e6 * (np.sum(h) - ones_count)**2
        return c5 + penalty

    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    h_final = res.x
    h_final = project(h_final)
    return h_final, compute_c5(h_final), n_points
