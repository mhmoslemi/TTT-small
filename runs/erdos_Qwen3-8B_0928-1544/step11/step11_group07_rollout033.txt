    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = initial_h * (ones_count / np.sum(initial_h))
        population = [initial_h]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for step in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        velocities = [np.zeros(n_points) for _ in range(population_size)]
        for i in range(population_size):
            w = w_initial - (w_initial - w_final) * (time.time() - start_time) / budget_s
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + phis[0] * r1 * (population[i] - np.mean(population)) + phis[1] * r2 * (best_h - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            h_sum = np.sum(population[i])
            if h_sum == 0:
                population[i] = np.random.rand(n_points) * 0.5
            population[i] *= (ones_count / h_sum)
            if np.random.rand() < p_perturb:
                perturb = np.random.uniform(-0.05, 0.05, n_points)
                population[i] += perturb
                population[i] = np.clip(population[i], 0, 1)
                h_sum = np.sum(population[i])
                if h_sum == 0:
                    population[i] = np.random.rand(n_points) * 0.5
                population[i] *= (ones_count / h_sum)
        for i in range(population_size):
            current_c5 = evaluate(population[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.copy(population[i])
    return best_h, best_c5, n_points
