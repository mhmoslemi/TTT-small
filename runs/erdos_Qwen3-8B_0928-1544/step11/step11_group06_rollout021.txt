import time
from scipy.optimize import minimize

    else:
        particles = [np.random.uniform(0.4, 0.6, n_points) * (ones_count / np.sum(np.random.uniform(0.4, 0.6, n_points))) for _ in range(swarm_size)]
    
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    pbest = particles.copy()
    pbest_vals = np.array([objective(p) for p in particles])
    gbest = particles[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)
    best_h = gbest
    best_c5 = gbest_val
    iteration = 0
    
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')

            particles[i] += velocities[i]

            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] *= (ones_count / np.sum(particles[i]))
            
            # Evaluate

            # Update pbest

        # Local optimization
