import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    max_iter = 1000  # Adjust based on budget

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize h
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))

    # Constraints and bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    # Optimization
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': max_iter}
    )

    # Adjust to meet constraints
    best_h = result.x
    if np.sum(best_h) != 0:
        best_h = best_h * (target_sum / np.sum(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)

    # Final objective
    c5_bound = objective(best_h)

    return best_h, c5_bound, n_points
