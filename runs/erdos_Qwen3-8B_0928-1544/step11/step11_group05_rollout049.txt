            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = 0.6  # Mutation factor

    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[indices[0]], population[indices[1]], population[indices[2]]
            v = a_vec + F * (b_vec - c_vec)
            v = np.clip(v, 0.0, 1.0)
            v_sum = np.sum(v)
            if v_sum == 0:
                continue
            v_scaled = v * (target_sum / v_sum)
            v_scaled = np.clip(v_scaled, 0.0, 1.0)
            new_c5 = objective(v_scaled)
            worst_idx = np.argmin([objective(sol) for sol in population])
            if new_c5 < objective(population[worst_idx]):
                population[worst_idx] = v_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = v_scaled.copy()
    return best_h, best_c5, n_points
