import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2
    pop_size = 200
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h0 = initial * (required_sum / np.sum(initial))
        pop = [h0]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            pop.append(h * (required_sum / h_sum))
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            pop.append(h * (required_sum / h_sum))

    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    best_h = pop[0].copy()
    best_val = obj(pop[0])
    pbest = [h.copy() for h in pop]
    pbest_vals = [obj(h) for h in pop]
    gbest = best_h.copy()
    gbest_val = best_val
    velocities = np.zeros((pop_size, n_points))
    w, phi_p, phi_g = 0.8, 2.0, 2.0
    max_vel = 0.1
    start = time.time()

    while time.time() - start < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = phi_p * r1 * (pbest[i] - pop[i]) + phi_g * r2 * (gbest - pop[i])
            velocities[i] = w * velocities[i] + vel_update
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            new_pos = pop[i] + velocities[i]
            new_pos = np.clip(new_pos, 0.0, 1.0)
            new_sum = np.sum(new_pos)
            if new_sum == 0:
                new_pos = pop[i].copy()
            else:
                new_pos *= required_sum / new_sum
            new_val = obj(new_pos)
            if new_val < pbest_vals[i]:
                pbest[i] = new_pos.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest = new_pos.copy()
                    gbest_val = new_val
                    best_h = new_pos.copy()
            pop[i] = new_pos
        if time.time() - start > budget_s - 1:
            break
    return best_h, gbest_val, n_points
