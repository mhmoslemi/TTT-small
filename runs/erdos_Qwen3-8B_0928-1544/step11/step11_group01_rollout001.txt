import numpy as np
import time

import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = np.array(initial_h_values)

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.zeros_like(h)
        return h_clipped * (ones_count / h_sum)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    h = clip_and_rescale(initial_h)

    # Initial guess
    best_c5 = objective(h)
    best_h = h.copy()

    # Gradient-based optimization with constraints
    def optimize_h(h_start):
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == ones_count,
            0 <= h_opt,
            h_opt <= 1
        ]
        # Objective: minimize max correlation
        def f(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points)
        problem = cp.Problem(cp.Minimize(f(h_opt)), constraints)
        try:
            result = problem.solve(solver=cp.ECOS, maxiter=1000, eps=1e-6)
            if result is not None:
                return h_opt.value
        except:
            pass
        return h_start

    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = optimize_h(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Penalize unchanged iterations
        if np.allclose(h, best_h, atol=1e-6):
            break

    return best_h, best_c5, n_points
