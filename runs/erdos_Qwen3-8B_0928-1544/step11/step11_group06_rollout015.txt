import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    else:
        h = np.random.uniform(0.4, 0.6, n_points)

        h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    time_remaining = budget_s
    start_time = time.time()

    def objective(x):
        h_opt = x
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        max_index = np.argmax(corr)
        return corr[max_index] * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]

    while time_remaining > 0:

        max_index = np.argmax(corr)
        current_c5 = corr[max_index] * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
