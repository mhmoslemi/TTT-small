import numpy as np
import time
import numpy.fft as fft

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 100
    ones_count = n // 2
    dx = 2.0 / n
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    # Apply low-pass filter
    H = fft.fft(h)
    cutoff = int(0.3 * n / 2)
    H[:cutoff] = 0
    H[n - cutoff:] = 0
    h_smoothed = fft.ifft(H).real
    h_smoothed = np.clip(h_smoothed, 0, 1)
    h_smoothed_sum = np.sum(h_smoothed)
    h_smoothed = h_smoothed * (ones_count / h_smoothed_sum)
    h = h_smoothed.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    eta = 0.1
    eta_decay = 0.997
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate current C5
        max_c = -np.inf
        best_k = 0
        for k in range(n):
            shifted = np.roll(h, k)
            overlap = h * (1 - shifted)
            c = np.sum(overlap) * dx
            if c > max_c:
                max_c = c
                best_k = k
        if max_c < best_c5:
            best_c5 = max_c
            best_h = h.copy()
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            j1 = (i + best_k) % n
            j2 = (i - best_k) % n
            grad[i] = (1 - h[j1]) * dx - h[j2] * dx
        
        # Update h
        h_new = h - eta * grad
        h_clipped = np.clip(h_new, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:

        scale = ones_count / sum_clipped
        h = h_clipped * scale
