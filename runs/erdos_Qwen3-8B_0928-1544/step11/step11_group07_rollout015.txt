import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)

        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
    
    best_h = h.copy()
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / n_points)

    while time.time() - start_time < budget_s:

        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum == 0:
            perturbed = np.random.rand(n_points)
            perturbed_sum = np.sum(perturbed)
            perturbed = perturbed * (target_sum / perturbed_sum)
        else:
            perturbed = perturbed * (target_sum / perturbed_sum)
        new_c5 = max(np.correlate(perturbed, 1 - perturbed, mode="full")) * (2.0 / n_points)
        if new_c5 < best_c5:
