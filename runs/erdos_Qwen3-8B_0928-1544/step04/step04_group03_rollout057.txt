import numpy as np
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n)
        ones = 35
        for i in range(ones):
            pos = int(i * 2)
            h[pos] = 1.0
        if np.sum(h) != ones:
            diff = ones - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
    def objective(h_flat):
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx
    bounds = Bounds([0.0]*n, [1.0]*n)
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}
    T = 1.0
    alpha = 0.999
    max_iter = 1000
    t0 = time.time()
    h_best = h.copy()
    c5_best = objective(h_best)
    for _ in range(max_iter):
        if time.time() - t0 > budget_s - 500:
            break
        noise = np.random.normal(0, T * 0.05, n)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed > 0:
            h_perturbed = h_perturbed * (35.0 / sum_perturbed)
        c5_new = objective(h_perturbed)
        c5_old = objective(h)
        if c5_new < c5_old:
            h = h_perturbed
            c5_best = c5_new
        else:
            prob = np.exp((c5_old - c5_new) / T)
            if np.random.rand() < prob:
                h = h_perturbed
                c5_best = c5_new
        T *= alpha
    top_candidates = [h_best]
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n)
        h_candidate = h_best + noise
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate > 0:
            h_candidate = h_candidate * (35.0 / sum_candidate)
        top_candidates.append(h_candidate)
    for candidate in top_candidates:
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < c5_best:
                c5_best = current_c5
                h_best = result.x
    return h_best, c5_best, n
