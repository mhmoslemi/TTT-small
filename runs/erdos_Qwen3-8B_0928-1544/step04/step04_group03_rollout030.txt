import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        n = 70
        ones = 35
        interval = n / ones
        for i in range(ones):
            pos = round(i * interval)
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)

    swarm_size = 50
    particles = []
    for _ in range(swarm_size):
        if _ == 0:
            h = h_initial.copy()
        else:
            h = h_initial.copy() + np.random.normal(0, 0.1, 70)
        h = np.clip(h, 0, 1)
        h = h * (35.0 / np.sum(h))
        particles.append(h)
    particles = np.array(particles)
    pbest = particles.copy()
    gbest = particles[0].copy()
    pbest_values = np.array([objective(p) for p in particles])
    gbest_val = np.min(pbest_values)

    omega = 0.9
    c1, c2 = 2.0, 2.0
    vlimit = 0.2
    velocities = np.random.uniform(-vlimit, vlimit, (swarm_size, 70))

    for _ in range(1000):
        if time.time() - t0 > budget_s - 200:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -vlimit, vlimit)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (35.0 / np.sum(particles[i]))
            curr_val = objective(particles[i])
            if curr_val < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = curr_val
                if curr_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = curr_val
        omega *= 0.999

    top_candidates = pbest
    top_values = pbest_values
    top_candidates = top_candidates[top_values < np.inf]
    top_values = top_values[top_values < np.inf]
    top_candidates = top_candidates[:10]
    top_values = top_values[:10]

    best_h = top_candidates[0]
    best_c5 = np.inf
    for i in range(len(top_candidates)):
        if top_values[i] < best_c5:
            best_c5 = top_values[i]
            best_h = top_candidates[i]

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-10,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        best_h = result.x

    return best_h, best_c5, 70
