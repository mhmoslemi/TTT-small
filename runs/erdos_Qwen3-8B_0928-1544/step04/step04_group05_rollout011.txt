        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1 = parents[i]
                p2 = parents[i+1]
                c1, c2 = crossover(p1, p2)
                offspring.append(c1)
                offspring.append(c2)
        
        mutated_offspring = [mutate(ind, mutation_rate, mutation_std, n_points) for ind in offspring]
        
        new_population = parents + mutated_offspring
        new_fitness = np.array([evaluate_fitness(ind, n_points) for ind in new_population])
        new_fitness = np.where(new_fitness == 0, float('inf'), new_fitness)
        
        combined_pop = initial_pop + new_population
        combined_fit = np.concatenate([fitness, new_fitness])
        
        sorted_indices = np.argsort(combined_fit)
        top_idx = sorted_indices[:pop_size]
        initial_pop = [combined_pop[i] for i in top_idx]
        fitness = combined_fit[top_idx]
        
        current_best_idx = np.argmax(fitness)
        current_best_c5 = 1 / fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = initial_pop[current_best_idx]
    
    return best_h, best_c5, n_points
