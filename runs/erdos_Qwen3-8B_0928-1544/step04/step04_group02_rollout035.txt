    def evaluate(h_vec):
        nonlocal dx
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Optimizer parameters
    n_pop = 30
    elite_ratio = 0.3
    mutation_rate = 0.1
    max_gen = 1000
    warmup = 200
    cooldown = 0.5

    # Initial population
    population = [h]
    for _ in range(n_pop - 1):
        perturb = np.random.normal(0, mutation_rate, 70)
        h_mut = np.clip(h + perturb, 0, 1)
        h_mut = h_mut * (35.0 / np.sum(h_mut))
        population.append(h_mut)

    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all individuals
        c5_values = np.array([evaluate(h) for h in population])
        indices = np.argsort(c5_values)
        
        # Select elites
        num_elites = int(n_pop * elite_ratio)
        elites = population[indices[:num_elites]]
        
        # Create new population
        new_pop = elites.copy()
        while len(new_pop) < n_pop:
            idx1, idx2 = np.random.choice(len(elites), 2, replace=False)
            parent1 = elites[idx1]
            parent2 = elites[idx2]
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = child * (35.0 / np.sum(child))
            noise = np.random.normal(0, mutation_rate * (1 - gen / max_gen), 70)
            child_mut = child + noise
            child_mut = np.clip(child_mut, 0, 1)
            child_mut = child_mut * (35.0 / np.sum(child_mut))
            new_pop.append(child_mut)
        
        population = new_pop
        
        # Local optimization on top performers
        for h_candidate in population[indices[:5]]:
            result = minimize(
                lambda x: evaluate(x),
                h_candidate,
