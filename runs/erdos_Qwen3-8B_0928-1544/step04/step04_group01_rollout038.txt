import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:

    decay_rate = 0.99

    improve_threshold = 1e-4
    prev_c5 = float('inf')
    
    for _ in range(max_iterations):

        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        
        # Early termination if no improvement
        if abs(c5 - prev_c5) < improve_threshold:

        prev_c5 = c5

        k_max = np.argmax(corr)
