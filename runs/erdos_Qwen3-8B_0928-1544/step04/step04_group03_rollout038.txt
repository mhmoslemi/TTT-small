import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        ones = 35
        interval = 70 / ones
        for i in range(ones):
            pos = int(round(i * interval))
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()
    n_points = 70
    n_swarm = 50
    n_iter = 500
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    v_limit = 0.2
    swarm = [h_initial.copy() for _ in range(n_swarm)]
    velocities = [np.zeros(n_points) for _ in range(n_swarm)]
    p_best = [h.copy() for h in swarm]
    p_best_fitness = [float('inf') for _ in range(n_swarm)]
    g_best = h_initial.copy()
    g_best_fitness = float('inf')
    constraint_sum = lambda h: np.sum(h) - 35.0
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    for iter in range(n_iter):
        if time.time() - t0 > budget_s - 500:
            break
        for i in range(n_swarm):
            h = swarm[i]
            curr_fitness = objective(h)
            if curr_fitness < p_best_fitness[i]:
                p_best[i] = h.copy()
                p_best_fitness[i] = curr_fitness
            if curr_fitness < g_best_fitness:
                g_best = h.copy()
                g_best_fitness = curr_fitness
        for i in range(n_swarm):
            h = swarm[i]
            v = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = w * v + c1 * r1 * (p_best[i] - h) + c2 * r2 * (g_best - h)
            np.clip(v, -v_limit, v_limit, out=v)
            h += v
            h = np.clip(h, 0.0, 1.0)
            if np.sum(h) != 35.0:
                h = h * (35.0 / np.sum(h))
            swarm[i] = h.copy()
            velocities[i] = v.copy()
        w *= 0.999
    top_particles = sorted([ (objective(h), h) for h in swarm ], key=lambda x: x[0])[:10]
    refined = []
    for f, h in top_particles:
        res = minimize(
            fun=lambda x: objective(x),
            x0=h,
            method='SLSQP',
            bounds=Bounds([0.0]*n_points, [1.0]*n_points),
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-10,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if res.success:
            refined.append((res.fun, res.x))
    if refined:
        best_refined = min(refined, key=lambda x: x[0])
        best_c5, best_h = best_refined
    else:
        best_c5 = g_best_fitness
        best_h = g_best
    return best_h, best_c5, n_points
