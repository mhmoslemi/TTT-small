            mid = np.random.randint(1, n_points)
            child1 = np.concatenate([p1[:mid], p2[mid:]])
            child2 = np.concatenate([p2[:mid], p1[mid:]])
            offspring.append(child1)
            offspring.append(child2)
        for i in range(len(offspring)):
            h = offspring[i].copy()
            mutate_idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[mutate_idx] += np.random.normal(0, 0.05, size=len(mutate_idx))
            h = project_h(h)
            offspring[i] = h
        combined = population + offspring
        combined_fitness = [1.0 / compute_c5(h) for h in combined]
        sorted_indices = np.argsort(combined_fitness)[::-1]
        population = [combined[i] for i in sorted_indices[:pop_size]]
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    return best_h, best_c5, n_points
