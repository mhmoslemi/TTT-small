from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n = 70

        swarm = [initial_h_values.copy()]
    else:
        h_uniform = np.zeros(n)
        ones = 35
        step = n // ones
        for i in range(ones):
            h_uniform[i * step] = 1.0
        if np.sum(h_uniform) != ones:
            diff = ones - np.sum(h_uniform)
            zeros = np.where(h_uniform == 0)[0]
            h_uniform[zeros[:diff]] = 1.0
        swarm = [h_uniform]
    swarm += [np.random.uniform(0, 1, n) for _ in range(49)]
    particles = np.array(swarm)
    velocities = np.random.uniform(-0.2, 0.2, (len(swarm), n))
    pbest = particles.copy()
    pbest_fitness = np.full(len(swarm), np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    ω = 0.9
    c1, c2 = 2.0, 2.0
    vlimit = 0.2
    max_iter = 1000
    t0 = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)
    
    for iter in range(max_iter):
        if time.time() - t0 > budget_s - 100:
            break
        for i in range(len(swarm)):
            h = particles[i]
            fitness = objective(h)
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = h.copy()
                gbest_fitness = fitness
        for i in range(len(swarm)):
            h = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            v = ω * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            v = np.clip(v, -vlimit, vlimit)
            velocities[i] = v
            particles[i] = np.clip(h + v, 0.0, 1.0)
            if np.sum(particles[i]) < 34.9 or np.sum(particles[i]) > 35.1:
                particles[i] *= 35.0 / np.sum(particles[i])
        ω *= 0.999
    
    top = np.argsort(pbest_fitness)[:10]

    best_c5 = gbest_fitness
    for cand in best_candidates:
        res = minimize(objective, cand, method='SLSQP', bounds=[(0,1)]*n, constraints={'type':'eq','fun':lambda x: np.sum(x)-35.0}, tol=1e-10)
