import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    h_ones = np.ones(n_points)
    h_ones_minus = 1.0 - h_ones

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        best_h = initial_h_values.copy()
    else:
        best_h = np.random.rand(n_points)
        best_h /= best_h.sum()
        best_h *= 35.0
    best_c5 = float('inf')

    def evaluate(h):
        corr = np.correlate(h, h_ones_minus, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - 35.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def mutate(h):
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h *= 35.0 / total
        return h

    population = [np.random.rand(n_points) for _ in range(50)]
    for i in range(50):
        population[i] /= population[i].sum()
        population[i] *= 35.0

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        population[0] = initial_h_values.copy()
        population[0] /= population[0].sum()
        population[0] *= 35.0

    time_start = time.time()
    for _ in range(500):
        if time.time() - time_start > budget_s - 300:
            break
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        parents = []
        for _ in range(25):
            idxs = np.random.choice(50, 5, replace=False)
            winner = idxs[np.argmin(fitness[idxs])]
            parents.append(population[winner])

        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            child = (p1 + p2) / 2
            offspring.append(mutate(child))

        combined = population + offspring
        combined_fitness = np.array([evaluate(h) for h in combined])
        order = np.argsort(combined_fitness)
        population = [combined[i] for i in order[:50]]

    result = minimize(evaluate, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8, options={'maxiter': 200})
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total != 35.0:
        best_h *= 35.0 / total
    return best_h, best_c5, n_points
