    max_iterations = 500
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        # Frequency adjustment step
        current_fft = fft(h)
        for i in range(1, n_points):
            current_fft[i] *= 1.0 / (i + 1)
        h_candidate = ifft(current_fft).real
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate != 0:
            h_candidate = h_candidate * (sum_h / sum_candidate)
        else:
            h_candidate = np.zeros_like(h_candidate)
        
        # Local optimization step
        def obj_func(x):
            h_vec = x.copy()
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
        
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h}
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = minimize(obj_func, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            sum_opt = np.sum(h_opt)
            if sum_opt != 0:
                h_opt = h_opt * (sum_h / sum_opt)
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            current_c5 = np.max(corr_opt) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
        else:
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            current_c5 = np.max(corr_candidate) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate.copy()
        
        # Update h with the candidate or optimized version
        h = h_opt if result.success else h_candidate
        iterations += 1
    
    return best_h, best_c5, n_points
