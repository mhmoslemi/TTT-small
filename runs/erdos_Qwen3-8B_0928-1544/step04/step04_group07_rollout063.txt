        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    n = 70
    dx = 2.0 / n

    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale

    def objective(h):
        projected_h = project_h(h)
        corr = np.correlate(projected_h, 1 - projected_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n)]

    # Differential evolution parameters
    strategy = 'best1bin'
    popsize = 15
    tol = 1e-6
    mutation = (0.5, 1)
    recombination = 0.8
    maxiter = 300
    polish = True

    # Run differential evolution
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy=strategy,
        popsize=popsize,
        tol=tol,
        mutation=mutation,
        recombination=recombination,
        maxiter=maxiter,
        polish=polish,
        disp=False,
        x0=initial_h
    )

    best_h = result.x
    best_c5 = result.fun

    # Run SLSQP refinement
    def constraint_sum(h):
        return np.sum(h) - n / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    result_slsqp = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x

    return best_h, best_c5, n
