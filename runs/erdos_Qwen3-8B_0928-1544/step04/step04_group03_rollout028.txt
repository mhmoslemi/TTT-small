    best_c5 = float('inf')
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s - 500:
            break

        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5s.append(np.max(corr) * dx)

        best_idx = np.argmin(current_c5s)
        if current_c5s[best_idx] < best_c5:
            best_c5 = current_c5s[best_idx]
            best_h = population[best_idx].copy()

        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = (p1 + p2) / 2
            child = child * (target_sum / np.sum(child))
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.01, n_points)
                child += noise
                child = np.clip(child, 0.0, 1.0)
                child = child * (target_sum / np.sum(child))
            new_pop.append(child)

        population = new_pop

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0.1, 0.9) for _ in range(n_points)]
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-6,
        options={'maxiter': 200, 'ftol': 1e-10}
    )

    if result.success:
        best_h = result.x
        best_c5 = result.fun

    best_h = best_h * (target_sum / np.sum(best_h))
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    return best_h, best_c5, n_points
