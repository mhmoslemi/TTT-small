from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    dx = 2.0 / n_points
    target_sum = 35.0

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_swarm = [initial_h_values.copy() for _ in range(50)]
    else:
        h_swarm = []
        base = np.zeros(n_points)
        ones = 35
        interval = n_points / ones
        for i in range(ones):
            pos = int(round(i * interval))
            base[pos] = 1.0
        if np.sum(base) != ones:
            diff = ones - np.sum(base)
            zeros = np.where(base == 0)[0]
            base[zeros[:diff]] = 1.0
        for _ in range(50):
            h = base.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            h_swarm.append(h)

    best_c5 = float('inf')

    best_swarm = h_swarm.copy()
    best_personal = [np.inf] * len(h_swarm)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    t0 = time.time()

    for iter in range(1000):

            best_global = global_best
