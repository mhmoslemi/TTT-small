import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    best_h = h_initial
    best_c5 = float('inf')
    max_iterations = 300
    start_time = time.time()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Generate multiple perturbations
        scales = [0.05 * (1 - iteration / max_iterations), 0.025 * (1 - iteration / max_iterations)]
        best_perturbed = None
        best_result = None
        for scale in scales:
            perturbation = np.random.normal(0, scale, n_points)
            h_perturbed = h_initial + perturbation
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                tol=1e-8,
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
                best_perturbed = result.x
        if best_perturbed is not None:
            h_initial = best_perturbed
    
    return best_h, best_c5, n_points
