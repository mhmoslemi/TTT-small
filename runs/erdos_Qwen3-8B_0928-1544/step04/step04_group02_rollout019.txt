from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:

    else:
        h_initial = np.full(70, 0.5)
    h_initial = h_initial * (35.0 / np.sum(h_initial))
    dx = 2.0 / 70
    
    n_points = 70
    best_h = h_initial.copy()
    best_c5 = float('inf')
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate(h):
        nonlocal best_c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def optimize(h_start):
        result = minimize(evaluate, h_start, method='L-BFGS-B', bounds=Bounds(0, 1), tol=1e-8, options={'maxiter': 200})
        if result.success:
            return result.x
        return h_start
    
    for _ in range(20):
        h_opt = optimize(h_initial)
        if compute_c5(h_opt) < best_c5:
            best_h = h_opt.copy()
            best_c5 = compute_c5(h_opt)
        h_initial = np.clip(h_opt + 0.01 * np.random.randn(70), 0, 1)
        h_initial = h_initial * (35.0 / np.sum(h_initial))
        if time.time() > budget_s:
            break
    
    return best_h, best_c5, n_points
