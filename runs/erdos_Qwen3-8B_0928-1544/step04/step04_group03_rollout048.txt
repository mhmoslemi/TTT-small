import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.zeros(70)
        n = 70
        ones = 35
        interval = n // ones
        for i in range(ones):
            pos = i * interval
            h[pos] = 1.0
        if np.sum(h) != ones:
            diff = ones - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
    n_points = 70
    dx = 2.0 / n_points
    best_h = h
    best_c5 = float('inf')
    t0 = time.time()
    sa_iter = 1000
    sa_temp = 1.0
    sa_alpha = 0.999
    top_candidates = 10
    sa_results = []

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    for _ in range(sa_iter):
        if time.time() - t0 > budget_s - 100:
            break
        noise = np.random.normal(0, sa_temp, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        curr_c5 = objective(h)
        new_c5 = objective(h_perturbed)
        if np.random.rand() < np.exp((curr_c5 - new_c5) / sa_temp):
            h = h_perturbed
        sa_temp *= sa_alpha
        sa_results.append((new_c5, h.copy()))

    sa_results.sort()
    top_h = sa_results[:top_candidates][1]
    top_c5 = sa_results[:top_candidates][0]

    for h_refine in top_h:
        result = minimize(
            fun=objective,
            x0=h_refine,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
