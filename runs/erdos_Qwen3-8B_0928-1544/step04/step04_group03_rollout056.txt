import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        ones = 35

                h[pos] = 1.0

            diff = ones - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    # Simulated Annealing
    temperature = 1.0
    alpha = 0.998
    sa_iterations = 1000
    for _ in range(sa_iterations):
        if time.time() - t0 > budget_s - 500:
            break
        noise = np.random.normal(0, temperature, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed == 0:
            h_perturbed = h.copy()

        e_new = compute_c5(h_perturbed)
        e_old = best_c5
        if e_new < e_old or np.random.rand() < np.exp((e_old - e_new) / temperature):

            best_c5 = e_new

    def constraint(x):
        return np.sum(x) - 35.0

    result = minimize(

        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-9,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
