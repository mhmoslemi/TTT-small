import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    current_pop_values = [objective(h) for h in population]
    best_idx = np.argmin(current_pop_values)
    best_h = population[best_idx]
    best_c5 = current_pop_values[best_idx]

    time_used = 0
    max_iter = int(budget_s // 10)
    T = 1.0
    sigma = 0.2
    alpha = 0.995

    for _ in range(max_iter):
        top_idx = np.argsort(current_pop_values)[:int(0.3 * population_size)]
        top_candidates = [population[i] for i in top_idx]
        new_candidates = []

        for h in top_candidates:
            noise = np.random.normal(0, sigma * T, n_points)
            h_perturbed = h + noise
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            sum_perturbed = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (ones_count / sum_perturbed) if sum_perturbed != 0 else np.full(n_points, 0.5)
            new_candidates.append(h_perturbed)

        for h in new_candidates:
            res = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if res.success:
                new_h = res.x
                new_h = np.clip(new_h, 0.0, 1.0)
                sum_new = np.sum(new_h)
                new_h = new_h * (ones_count / sum_new) if sum_new != 0 else np.full(n_points, 0.5)
                new_candidates.append(new_h)

        population += new_candidates
        population.sort(key=lambda x: objective(x))
        population = population[:population_size]
        current_pop_values = [objective(h) for h in population]

        best_idx = np.argmin(current_pop_values)
        best_h = population[best_idx]
        best_c5 = current_pop_values[best_idx]

        T *= alpha
        sigma *= alpha
        time_used += 10
        if time_used >= budget_s:
            break

    return best_h, best_c5, n_points
