from scipy.optimize import minimize, Bounds

    def objective(h):

        return np.max(corr) * dx

        return np.sum(h) - n / 2

    def project_h(h):

        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
