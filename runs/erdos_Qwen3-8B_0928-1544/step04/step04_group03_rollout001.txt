import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        ones = 35
        interval = n_points / ones
        for i in range(ones):
            pos = round(i * interval)
            h[pos] = 1.0
        if np.sum(h) != ones:
            diff = ones - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def cons(h):
        return np.sum(h) - 35.0
    
    t0 = time.time()
    T, alpha = 1.0, 0.999
    sa_itrs = 2000
    
    for _ in range(sa_itrs):
        if time.time() - t0 > budget_s - 500:
            break
        noise = np.random.normal(0, T, n_points)
        h_pert = h + noise
        h_pert = np.clip(h_pert, 0.0, 1.0)
        curr_sum = np.sum(h_pert)
        if curr_sum == 0:
            h_pert = np.full(n_points, 35.0 / n_points)
        else:
            h_pert = h_pert * (35.0 / curr_sum)
        E_new = obj(h_pert)
        E_old = obj(h)
        if E_new < E_old:
            accept = 1.0
        else:
            accept = np.exp((E_old - E_new) / T)
        if np.random.rand() < accept:
            h = h_pert.copy()
            if E_new < best_c5:
                best_c5 = E_new
                best_h = h.copy()
        T *= alpha
    
    if time.time() - t0 < budget_s - 500:
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8, options={'maxiter': 200})
        if res.success:
            best_c5 = res.fun
            best_h = res.x
    
    return best_h, best_c5, n_points
