from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_needed = 35
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        # Generate random initial configuration
        h_initial = np.zeros(n_points)
        positions = np.random.rand(ones_needed) * 2
        positions.sort()
        for pos in positions:
            idx = int(np.round(pos / dx))
            idx = np.clip(idx, 0, n_points - 1)
            h_initial[idx] = 1.0
        # Adjust to ensure exactly 35 ones
        ones_count = np.sum(h_initial)
        if ones_count > ones_needed:
            # Remove excess
            ones_to_remove = ones_count - ones_needed
            idxs = np.where(h_initial == 1.0)[0]
            for i in range(ones_to_remove):
                h_initial[idxs[i]] = 0.0
        elif ones_count < ones_needed:
            # Add missing
            zeros_idx = np.where(h_initial == 0.0)[0]
            for i in range(ones_needed - ones_count):
                h_initial[zeros_idx[i]] = 1.0
    
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    constraint_sum = lambda h: np.sum(h) - ones_needed
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    T = 1.0

        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Rescale to maintain sum
        h_perturbed = h_perturbed * (ones_needed / np.sum(h_perturbed))
        # Compute objective
        current_c5 = objective(h_perturbed)
        # Acceptance probability
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
            h_initial = h_perturbed.copy()
        else:
            # Use Metropolis criterion
            delta = current_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_c5 = current_c5
                best_h = h_perturbed.copy()
                h_initial = h_perturbed.copy()
        T *= alpha
    
    # Refinement with SLSQP
    for _ in range(5):
        if time.time() - t0 > budget_s - 500:

        result = minimize(
