import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(70)
        h = (h / np.sum(h) * 35.0)
    
    n = 70
    dx = 2.0 / n
    
    def project_h(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    h = project_h(h)
    
    def objective(x):
        h_opt = x
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return np.sum(x) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    def local_optimize(x0):
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                      tol=1e-8, options={'maxiter': 150, 'ftol': 1e-9})
        if res.success:
            return res.x, res.fun
        return x0, float('inf')
    
    best_h, best_c5 = h.copy(), objective(h)
    
    n_candidates = 30
    for _ in range(n_candidates):
        noise = np.random.normal(0, 0.1, n)
        candidate = h + noise
        candidate = project_h(candidate)
        opt_h, opt_c5 = local_optimize(candidate)
        if opt_c5 < best_c5:
            best_h, best_c5 = opt_h, opt_c5
    
    for _ in range(10):
        noise = np.random.normal(0, 0.05, n)
        candidate = best_h + noise
        candidate = project_h(candidate)
        opt_h, opt_c5 = local_optimize(candidate)
        if opt_c5 < best_c5:
