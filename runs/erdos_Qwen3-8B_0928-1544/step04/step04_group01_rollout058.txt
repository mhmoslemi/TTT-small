from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))

    np.random.seed(seed)

        nonlocal inertia, c1, c2
