import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    h = h * (35.0 / np.sum(h))
    dx = 2.0 / 70
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    def obj_func(x):
        return evaluate_erdos_solution(x)

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break

        # Frequency domain adjustment
        H = np.fft.fft(h)
        for k in range(1, len(H)):
            H[k] *= 1.0 / (k + 1)
        h_new = np.fft.ifft(H).real
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h != 0:
            h_new = h_new * (35.0 / sum_h)
        else:
            h_new = np.zeros(70)

        # Time domain optimization
        result = minimize(
            fun=obj_func,
            x0=h_new,
            method='SLSQP',
            bounds=Bounds([0]*70, [1]*70),

        )

            h_new = h_new * (35.0 / sum_h)

        # Update best solution

        if current_c5 < best_c5:
