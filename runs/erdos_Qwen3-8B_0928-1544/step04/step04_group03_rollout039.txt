import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        ones = 35
        interval = 70 / ones
        for i in range(ones):
            pos = int(np.round(i * interval))
            h_initial[pos] = 1.0
        diff = ones - np.sum(h_initial)
        zeros = np.where(h_initial == 0)[0]
        h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)

    n_particles = 50
    swarm = [h_initial + np.random.normal(0, 0.1, 70) for _ in range(n_particles - 1)]
    swarm = [np.clip(s, 0, 1) * (35.0 / np.sum(np.clip(s, 0, 1))) for s in swarm]
    swarm = [np.concatenate([h_initial, s]) for s in swarm]
    swarm = [np.clip(s, 0, 1) * (35.0 / np.sum(np.clip(s, 0, 1))) for s in swarm]
    personal_best = [s.copy() for s in swarm]
    personal_best_val = [objective(s) for s in swarm]
    global_best = min(swarm, key=objective)
    global_best_val = objective(global_best)

    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    vel_limit = 0.2
    velocities = [np.zeros(70) for _ in range(n_particles)]

    iter_count = 0
    max_iter = 1000
    while iter_count < max_iter and time.time() - t0 < budget_s - 100:
        for i in range(n_particles):
            r1 = np.random.rand(70)
            r2 = np.random.rand(70)
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - swarm[i]) + c2 * r2 * (global_best - swarm[i])
            velocities[i] = np.clip(velocities[i], -vel_limit, vel_limit)
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (35.0 / np.sum(swarm[i]))
            current_c5 = objective(swarm[i])
            if current_c5 < personal_best_val[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_val[i] = current_c5
        current_global = min(swarm, key=objective)
        current_global_val = objective(current_global)
        if current_global_val < global_best_val:
            global_best = current_global.copy()
            global_best_val = current_global_val
        omega *= 0.999
        iter_count += 1

    top_candidates = sorted(swarm, key=objective)[:10]
    for h in top_candidates:
        result = minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-10,

        if result.success:
            current_c5 = result.fun

                best_h = result.x

    return best_h, best_c5, 70
