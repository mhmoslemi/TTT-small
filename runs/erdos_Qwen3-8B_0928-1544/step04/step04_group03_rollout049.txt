        h_perturb = h + perturbation
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        sum_perturb = np.sum(h_perturb)
        if sum_perturb != 0:
            h_perturb = h_perturb * (35.0 / sum_perturb)
        e_new = evaluate_C5(h_perturb)
        e_old = evaluate_C5(h)
        
        if e_new < best_sa_c5:
            best_sa_candidate = h_perturb.copy()
            best_sa_c5 = e_new
        
        if e_new < e_old:
            h = h_perturb
        else:
            p = np.exp((e_old - e_new) / T)
            if np.random.rand() < p:
                h = h_perturb
        
        T *= alpha
        if time.time() - t0 > budget_s - 500:
            break
    
    def objective(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_candidate):
        return np.sum(h_candidate) - 35.0
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    result = minimize(
        fun=objective,
        x0=best_sa_candidate,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = best_sa_candidate
        best_c5 = best_sa_c5
    
    return best_h, best_c5, n_points
