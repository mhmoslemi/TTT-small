from scipy.optimize import minimize

    scale = (n / 2) / total if total != 0 else 1.0
    initial_h *= scale
    np.random.seed(seed)
    np.seterr(all='ignore')
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    particles = [initial_h.copy() for _ in range(100)]
    velocities = [np.random.uniform(-0.1, 0.1, n) for _ in range(100)]
    pbest = [p.copy() for p in particles]
    gbest = initial_h.copy()
    best_fitness = objective(gbest)
    
    w = 0.7
    c1, c2 = 2.0, 2.0

            h_new = project_h(h_new)
