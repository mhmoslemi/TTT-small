    dx = 2.0 / n
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = pop[0]
    best_c5 = fitness(best_h)
    
    for gen in range(300):

                child = a.copy()
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.03, n)
                child += noise
