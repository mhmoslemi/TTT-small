import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(70, 0.5, dtype=np.float64)
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    alpha = 0.1
    decay_rate = 0.995
    iterations = 0
    max_update = 0.1
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(corr)
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        m = np.argmax(corr)
        k = m - (n_points - 1)
        gradient = np.zeros(n_points, dtype=np.float64)
        for j in range(n_points):
            j_plus_k = j + k
            term1 = 0.0
            if j_plus_k < n_points:
                term1 = (1 - h[j_plus_k]) * dx
            j_minus_k = j - k
            term2 = 0.0
            if j_minus_k >= 0:
                term2 = -h[j_minus_k] * dx
            gradient[j] = term1 + term2
        delta = alpha * gradient
        h += delta
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= (n_points / 2) / sum_h
        if np.max(np.abs(delta)) < max_update:
            alpha *= decay_rate
        iterations += 1
        if iterations > 300:
            break
    return best_h, best_c5, n_points
