import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    h_initial = h_initial * (35.0 / np.sum(h_initial))
    dx = 2.0 / 70
    POP_SIZE = 20
    population = [h_initial]
    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()
    sigma_start = 0.1
    sigma_end = 0.01
    max_gen = 500
    for _ in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            current_c5s.append(c5)
        sorted_indices = np.argsort(current_c5s)
        num_elites = int(POP_SIZE * 0.1)
        elites = [population[i] for i in sorted_indices[:num_elites]]
        new_pop = elites.copy()
        for _ in range(POP_SIZE - num_elites):
            idx1, idx2 = np.random.choice(len(elites), 2, replace=False)
            parent1 = elites[idx1]
            parent2 = elites[idx2]
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            if sum_c == 0:
                child = np.zeros(70)
            else:
                child = child * (35.0 / sum_c)
            sigma = sigma_start - (sigma_start - sigma_end) * (_ / max_gen)
            perturb = np.random.normal(0, sigma, 70)
            child_mut = child + perturb
            child_mut = np.clip(child_mut, 0, 1)
            sum_mut = np.sum(child_mut)
            if sum_mut != 0:
                child_mut = child_mut * (35.0 / sum_mut)
            new_pop.append(child_mut)
        population = new_pop
        top_idx = sorted_indices[:5]
        for i in top_idx:
            h = population[i]
            def obj_func(x):
                h_vec = x.copy()
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                return np.max(corr) * dx
            def cons(x):
                return np.sum(x) - 35.0
            result = minimize(obj_func, h, method='SLSQP', bounds=Bounds([0]*70, [1]*70), constraints={'type': 'eq', 'fun': cons}, tol=1e-8, options={'maxiter': 150})
