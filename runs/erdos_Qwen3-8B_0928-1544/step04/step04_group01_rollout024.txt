import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values.copy())
    # Add binary pattern
    binary = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    binary[indices] = 1
    initial_guesses.append(binary)
    # Add shifted version
    shifted = np.roll(initial_guesses[0], 14)
    shifted[shifted < 0] = 0
    shifted = shifted * (35.0 / np.sum(shifted))
    initial_guesses.append(shifted)
    # Add sinusoidal pattern
    sinusoidal = (np.sin(np.linspace(0, 2*np.pi, n_points)) + 1) / 2
    sinusoidal = sinusoidal * (35.0 / np.sum(sinusoidal))
    initial_guesses.append(sinusoidal)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
    
    for guess in initial_guesses:
        h_initial = guess.copy()
        for iteration in range(200):
            perturbation = np.random.normal(0, 0.05 * (1 - iteration / 200), n_points)
            h_perturbed = h_initial + perturbation
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
            
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                tol=1e-8,
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                h_initial = result.x
            if time.time() - start_time > budget_s:
                break
    
    return best_h, best_c5, n_points
