import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones = n_points // 2

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        interval = n_points / ones
        for i in range(ones):
            pos = round(i * interval)
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    # Simulated Annealing
    T = 1.0
    alpha = 0.996
    n_iterations = 1000
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    t0 = time.time()

    for _ in range(n_iterations):
        if time.time() - t0 > budget_s - 500:
            break
        perturbation = np.random.normal(0, T, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed != 0:
            h_perturbed = h_perturbed * (ones / sum_perturbed)
        new_c5 = objective(h_perturbed)
        current_c5 = objective(h_initial)
        if new_c5 < current_c5:
            h_initial = h_perturbed
            best_c5 = new_c5
            best_h = h_initial.copy()
        else:
            prob = np.exp((current_c5 - new_c5) / T)
            if np.random.rand() < prob:
                h_initial = h_perturbed
                best_c5 = new_c5
                best_h = h_initial.copy()
        T *= alpha

    # SLSQP refinement
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        final_h = result.x
        final_c5 = result.fun
    else:
        final_h = best_h
        final_c5 = best_c5

    return final_h, final_c5, n_points
