    # Fill the population with feasible solutions
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_half) / sum_h
        return h_clip * scale
    
    # Project all initial individuals
    population = [project_h(ind) for ind in initial_population]

        h = project_h(h)
        population.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_half
    
    # GA parameters
    tournament_size = 3
    mutation_rate = 0.05
    crossover_rate = 0.7

    for gen in range(generations):
        if time.time() - time_start > budget_s:

        # Evaluate all
        fitness = [objective(ind) for ind in population]
        # Find the best fitness
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        
        # Selection via tournament
        parents = []
        for _ in range(pop_size):
            tournament = np.random.choice(range(pop_size), tournament_size, replace=False)

        for i in range(0, pop_size, 2):

            p1 = parents[i]

            child1, child2 = np.copy(p1), np.copy(p2)
            for j in range(n_points):

                    child2[j] = p1[j]

        population = offspring
