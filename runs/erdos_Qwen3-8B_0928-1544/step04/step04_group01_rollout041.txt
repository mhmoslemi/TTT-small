import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    best_h = h_initial
    best_c5 = float('inf')
    max_iterations = 300
    start_time = time.time()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}]
    
    for iteration in range(max_iterations):
        perturbation_scale = 0.15 * (1 - iteration / max_iterations)
        # Generate 3 perturbations for diverse exploration
        for _ in range(3):
            Perturb = np.random.normal(0, perturbation_scale, n_points)
            h_perturbed = h_initial + Perturb
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
            
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                tol=1e-8,
                options={'maxiter': 150, 'ftol': 1e-9}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            if time.time() - start_time > budget_s:
                break
        h_initial = best_h
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
