import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.full(70, 0.5)
    
    n = 70
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    particles = np.array([initial_h + np.random.normal(0, 0.05, n) for _ in range(100)])
    velocities = np.random.uniform(-0.1, 0.1, size=(100, n))
    p_best = particles.copy()
    global_best_h = p_best[0].copy()
    global_best_fitness = objective(global_best_h)
    
    best_h = initial_h.copy()
    best_fitness = objective(best_h)
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        
        r1, r2 = np.random.rand(100, n), np.random.rand(100, n)
        velocities = 0.7 * velocities + 2 * r1 * (p_best - particles) + 2 * r2 * (global_best_h - particles)
        particles += velocities
        
        for i in range(100):
            h = particles[i]
            h_clip = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h_clip)
            if sum_h == 0:
                particles[i] = np.zeros(n)
            else:
                particles[i] = h_clip * (n / 2) / sum_h
        
        for i in range(100):
            h_candidate = particles[i]
            fit = objective(h_candidate)
            if fit < objective(p_best[i]):
                p_best[i] = h_candidate.copy()
            if fit < global_best_fitness:
                global_best_h = h_candidate.copy()
                global_best_fitness = fit
        
        if global_best_fitness < best_fitness:
            best_h = global_best_h.copy()
            best_fitness = global_best_fitness
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    
    return result.x, result.fun, n
