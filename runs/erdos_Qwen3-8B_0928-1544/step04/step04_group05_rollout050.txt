import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n, 0.5)
    
    pop_size = 50
    max_gen = 100
    to_size = 10
    mut_rate = 0.1

    # Initialize population
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        sm = np.sum(h)
        if sm == 0:
            h = np.zeros(n)
        else:
            h = h * (n/2) / sm
        pop.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        
        # Fitness and tournament selection
        fit = [compute_c5(h, dx) for h in pop]
        parents = []
        for _ in range(pop_size):
            tour = random.sample(range(pop_size), to_size)
            tour_fit = [fit[i] for i in tour]
            best_idx = tour[np.argmin(tour_fit)]
            parents.append(pop[best_idx])
        
        # Crossover
        off = []
        for i in range(0, pop_size, 2):
            p1, p2 = parents[i], parents[i+1]
            mid = np.random.randint(1, n-1)
            c1 = np.concatenate([p1[:mid], p2[mid:]])
            c2 = np.concatenate([p2[:mid], p1[mid:]])
            off.append(c1)
            off.append(c2)
        
        # Mutation
        for i in range(len(off)):
            if random.random() < mut_rate:
                mut = off[i] + np.random.normal(0, 0.05, n)
                mut = np.clip(mut, 0, 1)
                sm = np.sum(mut)
                if sm == 0:
                    mut = np.zeros(n)
                else:
                    mut = mut * (n/2) / sm
                off[i] = mut
        
        pop = off

        # Update best
        for h in pop:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    return (best_h, best_c5, n)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
