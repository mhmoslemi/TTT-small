import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        ones = 35
        for i in range(ones):
            h_initial[i*2] = 1.0
        if np.sum(h_initial) != ones:
            for i in range(70):
                if h_initial[i] == 0:
                    h_initial[i] = 1.0
                    ones -= 1
                    if ones == 0:
                        break
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)

    # Simulated Annealing
    T = 1.0
    alpha = 0.9996
    sa_iters = 5000
    sa_early_stop = 200
    best_sa = h_initial
    sa_time = 0
    for _ in range(sa_iters):
        if time.time() - t0 > budget_s - 500:
            break
        perturb = np.random.normal(0, T, 70)
        new_h = h_initial + perturb
        new_h = np.clip(new_h, 0, 1)
        s = np.sum(new_h)
        if s != 0:
            new_h *= 35.0 / s
        E_old = objective(h_initial)
        E_new = objective(new_h)
        if E_new < E_old:
            accept = 1.0
        else:
            accept = np.exp((E_old - E_new) / T)
        if np.random.rand() < accept:
            h_initial = new_h
            if E_new < best_c5:
                best_c5 = E_new
                best_sa = new_h
        T *= alpha
        sa_time += 1
        if sa_time % sa_early_stop == 0 and best_c5 < 0.38:
            break

    # SLSQP Refinement
    result = minimize(
        fun=objective,
        x0=best_sa,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = best_sa

    return best_h, best_c5, 70
