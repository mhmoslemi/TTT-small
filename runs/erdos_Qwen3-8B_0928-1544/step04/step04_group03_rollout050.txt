            h_initial = h_perturbed
            best_c5 = current_c5
        else:
            if np.random.rand() < np.exp((best_c5 - current_c5) / T):
                h_initial = h_perturbed
                best_c5 = current_c5
        T *= cooling_rate
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    def constraint_sum(h):
        return np.sum(h) - 35.0
    bounds = Bounds([0.0]*70, [1.0]*70)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
    return best_h, best_c5, 70
