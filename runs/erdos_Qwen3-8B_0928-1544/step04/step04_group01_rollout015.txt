    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_iterations = 500
    learning_rate = 0.1
    decay_rate = 0.999
    epsilon = 1e-4
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        m = np.argmax(corr)
        k = m - (n_points - 1)
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_perturb = h.copy()
            h_perturb[j] += epsilon
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb * (n_points / 2.0) / np.sum(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full') * dx
            new_c5 = np.max(corr_perturb)
            grad[j] = (new_c5 - current_c5) / epsilon
        
        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        learning_rate *= decay_rate
    
    return best_h, best_c5, n_points
