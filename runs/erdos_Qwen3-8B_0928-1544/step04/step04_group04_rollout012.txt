import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(70)
        h = (h / np.sum(h) * 35.0)
    n = 70
    dx = 2.0 / n

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35

    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = 35.0 / sum_h
        return h_clip * scale

    population_size = 50
    mutation_rate = 0.05
    crossover_rate = 0.7
    tournament_size = 3
    generations = 100
    best_h = h.copy()
    best_c5 = objective(best_h)

    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n)
        h_rand = (h_rand / np.sum(h_rand) * 35.0)
        population.append(h_rand)

    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
        fitness = [objective(ind) for ind in population]
        parents = []
        for _ in range(population_size):
            tour = np.random.choice(population, tournament_size)
            parents.append(min(tour, key=lambda x: objective(x)))
        new_pop = []
        for i in range(0, population_size, 2):
            p1, p2 = parents[i], parents[i + 1] if i + 1 < len(parents) else parents[i]
            if np.random.rand() < crossover_rate:
                mask = np.random.rand(n) < 0.5
                c1 = np.where(mask, p1, p2)
                c2 = np.where(~mask, p1, p2)
            else:
                c1, c2 = p1.copy(), p2.copy()
            c1 = project_h(c1 + np.random.normal(0, 0.05, n))
            c2 = project_h(c2 + np.random.normal(0, 0.05, n))
            new_pop.extend([c1, c2])
        population = new_pop
        curr_best = min(population, key=objective)
        curr_c5 = objective(curr_best)
        if curr_c5 < best_c5:

            cand,
            method='SLSQP',
            bounds=Bounds([0.0]*n, [1.0]*n),
