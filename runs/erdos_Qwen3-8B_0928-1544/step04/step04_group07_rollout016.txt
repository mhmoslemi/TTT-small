import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    
    def objective(h_flat):
        h = h_flat.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h_flat):
        h_clip = np.clip(h_flat, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    temp = 0.1
    cooling_rate = 0.995
    max_iter = 100
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, temp, size=n)
        perturbed = np.clip(h + noise, 0, 1)
        perturbed = project_h(perturbed)
        res = minimize(fun=objective, x0=perturbed, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        h = res.x
        temp *= cooling_rate
    
    res_final = minimize(fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    if res_final.success and res_final.fun < best_c5:
        best_c5 = res_final.fun
        best_h = res_final.x
    
    return best_h, best_c5, n
