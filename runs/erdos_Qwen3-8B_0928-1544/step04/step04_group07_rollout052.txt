        return h_clip * scale
    
    if 'initial_h_values' in globals():
        particles = np.random.rand(50, n) * 0.1 + initial_h
    else:
        particles = np.random.rand(50, n) * 0.5
    
    velocities = np.random.rand(50, n) * 0.05
    personal_bests = particles.copy()
    global_best = particles[0].copy()
    global_best_score = float('inf')
    personal_best_scores = np.full(50, float('inf'))
    
    for iter in range(200):
        for i in range(50):
            r1, r2 = np.random.rand(2)
            velocities[i] *= 0.8 - 0.0005 * iter
            velocities[i] += c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])
        
        for i in range(50):
            score = objective(particles[i])
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_bests[i] = particles[i].copy()
        
        current_best = personal_best_scores.argmin()
        if personal_best_scores[current_best] < global_best_score:
            global_best_score = personal_best_scores[current_best]
            global_best = personal_bests[current_best].copy()
    
    result = minimize(
        fun=objective,
        x0=global_best,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n)],
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    
    best_h = result.x if result.success and result.fun < global_best_score else global_best
    best_c5 = result.fun if result.success else global_best_score
    
    return best_h, best_c5, n
