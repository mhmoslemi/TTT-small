import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_h = np.full(n_points, 0.5)
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def generate_starting_points():
        if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
            start = initial_h_values.copy()
        else:
            start = np.full(n_points, 0.5)
        start_norm = start * target_sum / np.sum(start)
        yield start_norm
        # Random perturbation
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        yield h_rand
        # Block pattern
        h_block = np.zeros(n_points)
        h_block[::2] = 1.0
        h_block = h_block * target_sum / np.sum(h_block)
        yield h_block
        # Alternating pattern
        h_alt = np.zeros(n_points)
        h_alt[::3] = 1.0
        h_alt[1::3] = 0.5
        h_alt = h_alt * target_sum / np.sum(h_alt)
        yield h_alt

    starting_points = list(generate_starting_points())
    
    for h_start in starting_points:
        if time.time() - start_time > budget_s:
            break
        try:
            result = minimize(
                fun=objective,
                x0=h_start,
                method='SLSQP',
                bounds=Bounds([0.0]*n_points, [1.0]*n_points),
                constraints=constraints,
                tol=1e-8,
                options={'maxiter': 150, 'ftol': 1e-9}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
