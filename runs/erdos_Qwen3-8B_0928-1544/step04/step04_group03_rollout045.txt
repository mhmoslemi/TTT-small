            h = h * 35.0 / np.sum(h)
            h = np.clip(h, 0.0, 1.0)
            initial_population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * 35.0 / np.sum(h)
            h = np.clip(h, 0.0, 1.0)
            initial_population.append(h)
    
    # GA loop
    max_generations = 500
    num_parents = int(population_size * 0.2)
    
    for generation in range(max_generations):
        # Evaluate population
        fitness = []
        for h in initial_population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Select parents
        sorted_indices = np.argsort(fitness)
        parents = [initial_population[i] for i in sorted_indices[:num_parents]]
        
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            if len(parents) >= 2:
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                child = (p1 + p2) / 2
                child_sum = np.sum(child)
                if child_sum == 0:
                    continue
                child = child * 35.0 / child_sum
                child = np.clip(child, 0.0, 1.0)
                new_population.append(child)
            else:
                h = np.random.rand(n_points)
                h = h * 35.0 / np.sum(h)
                h = np.clip(h, 0.0, 1.0)
                new_population.append(h)
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                mutated = new_population[i] + noise
                mutated = np.clip(mutated, 0.0, 1.0)
                mutated_sum = np.sum(mutated)
                if mutated_sum == 0:
                    continue
                mutated = mutated * 35.0 / mutated_sum
                new_population[i] = mutated
        
        initial_population = new_population
        
        # Check time
        if time.time() - time_start > budget_s - 500:
            break
    
    # Select the best individual
    fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in initial_population]
    best_index = np.argmin(fitness)
    best_h = initial_population[best_index]
    
    # SLSQP refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.sum(best_h) != 35.0:
        norm = 35.0 / np.sum(best_h)
        best_h = best_h * norm
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return best_h, best_c5, n_points
