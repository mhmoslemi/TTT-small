import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    h_initial = h_initial * (35.0 / np.sum(h_initial))
    dx = 2.0 / 70
    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()
    n_iter = 200
    n_points = 70

    for _ in range(n_iter):
        # Frequency domain adjustment
        H = np.fft.fft(h_initial)
        magnitudes = np.abs(H)
        freq_indices = np.arange(70) + 1
        freq_indices[0] = 1e-10  # avoid division by zero
        target_mag = magnitudes / freq_indices
        target_energy = np.sum(magnitudes ** 2)
        adjusted_target = target_mag * np.sqrt(target_energy / np.sum(target_mag ** 2))
        adjusted_H = adjusted_target * np.exp(1j * np.angle(H))
        h_freq = np.fft.ifft(adjusted_H).real

        # Clip and rescale
        h_freq = np.clip(h_freq, 0, 1)
        h_freq = h_freq * (35.0 / np.sum(h_freq))

        # Time domain optimization
        def obj_func(x):
            h_vec = x.copy()
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx

        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}
        result = minimize(obj_func, h_freq, method='L-BFGS-B',
                          bounds=[(0, 1)] * 70, constraints=cons, tol=1e-8,
                          options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
        if result.success:
            h_initial = result.x

        # Compute current C5
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_initial.copy()

        # Check elapsed time
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
