import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        # Generate initial guess with triangular basis
        N = 10
        p = np.linspace(0.2, 1.8, N)
        w = np.full(N, 0.3)
        a = np.full(N, 0.3)
        b = 0.275
        x = np.linspace(0, 2, n_points)
        h_values = np.zeros(n_points)
        for i in range(N):
            p_i = p[i]
            w_i = w[i]
            a_i = a[i]
            dx_vals = x - p_i
            tri = np.clip(1 - np.abs(dx_vals) / w_i, 0, 1)
            h_values += a_i * tri
        h_values += b
        # Ensure sum constraint
        sum_h = np.sum(h_values)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h_values *= scale
        initial_h = h_values
    
    # Function to evaluate overlap
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / sum_h
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    # Optimization loop
    for _ in range(200):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = current_h + perturbation
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 300, 'ftol': 1e-9}
        )
        
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:
