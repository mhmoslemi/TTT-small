    gbest_c5, gbest_coarse = np.min(pbest_c5), pbest_coarse[np.argmin(pbest_c5)]

    while time.time() - start_time < time_limit:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            v_coarse[i] = omega * v_coarse[i] + c1 * r1 * (pbest_coarse[i] - h_coarse[i]) + c2 * r2 * (gbest_coarse - h_coarse[i])
            h_coarse[i] += v_coarse[i]
            h_coarse[i] = np.clip(h_coarse[i], 0, 1)
            h_sum = np.sum(h_coarse[i])
            if h_sum != 0:
                h_coarse[i] = h_coarse[i] * (n_coarse / 2) / h_sum
            current_c5 = compute_c5(h_coarse[i], dx_coarse)
            if current_c5 < pbest_c5[i]:
                pbest_c5[i] = current_c5
                pbest_coarse[i] = h_coarse[i].copy()
        gbest_idx = np.argmin(pbest_c5)
        if pbest_c5[gbest_idx] < gbest_c5:
            gbest_c5, gbest_coarse = pbest_c5[gbest_idx], pbest_coarse[gbest_idx]
        omega = max(0.4, omega - 0.001)
        c1 = max(1.0, c1 - 0.01)
        c2 = min(2.5, c2 + 0.01)

    remained_time = budget_s - (time.time() - start_time)
    if remained_time < 0:
        return (gbest_coarse, gbest_c5, n_coarse)

    # Upscale and refine
    h_fine = np.repeat(gbest_coarse, 2)
    h_fine = np.reshape(h_fine, (1, n_fine))
    swarm_size = 30
    omega, c1, c2 = 0.9, 2.0, 2.0
    start_time_fine = time.time()

    h_fine_swarm = np.full((swarm_size, n_fine), h_fine[0])
    v_fine = np.random.uniform(-0.05, 0.05, (swarm_size, n_fine))
    pbest_fine = h_fine_swarm.copy()
    pbest_c5_fine = np.array([compute_c5(h, dx_fine) for h in h_fine_swarm])
    gbest_c5_fine, gbest_fine = np.min(pbest_c5_fine), pbest_fine[np.argmin(pbest_c5_fine)]

    while time.time() - start_time_fine < remained_time:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            v_fine[i] = omega * v_fine[i] + c1 * r1 * (pbest_fine[i] - h_fine_swarm[i]) + c2 * r2 * (gbest_fine - h_fine_swarm[i])
            h_fine_swarm[i] += v_fine[i]
            h_fine_swarm[i] = np.clip(h_fine_swarm[i], 0, 1)
            h_sum = np.sum(h_fine_swarm[i])
            if h_sum != 0:
                h_fine_swarm[i] = h_fine_swarm[i] * (n_fine / 2) / h_sum
            current_c5 = compute_c5(h_fine_swarm[i], dx_fine)
            if current_c5 < pbest_c5_fine[i]:
                pbest_c5_fine[i] = current_c5
                pbest_fine[i] = h_fine_swarm[i].copy()
        gbest_idx = np.argmin(pbest_c5_fine)
        if pbest_c5_fine[gbest_idx] < gbest_c5_fine:
            gbest_c5_fine, gbest_fine = pbest_c5_fine[gbest_idx], pbest_fine[gbest_idx]
        omega = max(0.4, omega - 0.001)
        c1 = max(1.0, c1 - 0.01)
        c2 = min(2.5, c2 + 0.01)

    return (gbest_fine, gbest_c5_fine, n_fine)
