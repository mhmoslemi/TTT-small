from scipy.optimize import minimize, Bounds
import time

        h_mut = np.clip(h_mut, 0, 1)
        sum_mut = np.sum(h_mut)
        if sum_mut != 0:
            h_mut = h_mut * (35.0 / sum_mut)
        population.append(h_mut)
    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()
    sigma_start = 0.08
    sigma_end = 0.01
    max_gen = 500
    while time.time() - start_time < budget_s:
        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            current_c5s.append(c5)
        sorted_indices = np.argsort(current_c5s)
        num_elites = int(POP_SIZE * 0.2)
        elites = [population[i] for i in sorted_indices[:num_elites]]

        for _ in range(POP_SIZE - num_elites):
            idx1, idx2 = np.random.choice(len(elites), 2, replace=False)
            parent1 = elites[idx1]
            parent2 = elites[idx2]
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            if sum_c == 0:
                child = np.zeros(70)
            else:
                child = child * (35.0 / sum_c)
            sigma = sigma_start - (sigma_start - sigma_end) * (len(sorted_indices) / max_gen)
            perturb = np.random.normal(0, sigma, 70)

        for i in top_idx:
            h = population[i]

                return np.max(corr) * dx
            def cons(x):
                return np.sum(x) - 35.0
