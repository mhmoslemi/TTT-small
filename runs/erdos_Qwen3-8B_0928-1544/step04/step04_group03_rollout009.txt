import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.zeros(70)
        n = 70
        ones = 35
        interval = n / ones
        for i in range(ones):
            pos = round(i * interval)
            h[pos] = 1.0
        if np.sum(h) != ones:
            diff = ones - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h
    best_c5 = float('inf')
    t0 = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)

    # Simulated Annealing Phase
    T = 1.0
    alpha = 0.999
    sa_iter = 0
    max_sa_iter = 1000
    sa_candidates = [h.copy()]
    while sa_iter < max_sa_iter and time.time() - t0 < budget_s - 500:
        sa_iter += 1
        perturb = np.random.normal(0, T, 70)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb * (35.0 / np.sum(h_perturb))
        curr_c5 = objective(h)
        new_c5 = objective(h_perturb)
        if np.random.rand() < np.exp((curr_c5 - new_c5)/T):
            h = h_perturb
        T *= alpha
        if sa_iter % 100 == 0:
            sa_candidates.append(h.copy())
            if len(sa_candidates) > 10:
                sa_candidates.pop(0)
    
    # Final Refinement
    top_candidates = sa_candidates
    for cand in top_candidates:
        result = minimize(
            fun=objective,
            x0=cand,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, 70
