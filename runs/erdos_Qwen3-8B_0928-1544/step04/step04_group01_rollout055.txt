import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)

    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_iterations = 300
    alpha = 0.1
    decay_factor = 0.95
    tolerance = 1e-5
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=tolerance,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h_initial = result.x
        
        # Adaptive perturbation to avoid local minima
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.1, n_points)
            h_perturbed = h_initial + perturbation

            perturbation = h_perturbed - h_initial
            h_initial += 0.05 * perturbation
        
    return best_h, best_c5, n_points
