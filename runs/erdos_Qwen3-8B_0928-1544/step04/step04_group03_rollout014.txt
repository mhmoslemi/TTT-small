import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        ones = 35
        interval = 70 / ones
        for i in range(ones):
            pos = int(round(i * interval))
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()

    # PSO parameters
    swarm_size = 50
    inertia_start = 0.9
    inertia_decay = 0.999
    c1, c2 = 2.0, 2.0
    vel_limit = 0.2
    max_iter = 1000
    refine_count = 10

    # Initialize swarm
    particles = [h_initial.copy() for _ in range(swarm_size)]
    velocities = [np.zeros(70) for _ in range(swarm_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_val = [float('inf')] * swarm_size
    global_best = h_initial.copy()
    global_best_val = float('inf')

    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (35.0 / s)

    def clamp_and_rescale(h):
        h_clamped = np.clip(h, 0.0, 1.0)
        return rescale(h_clamped)

    def update_velocity(v, p, g, c1, c2, r1, r2):
        return inertia_start * v + c1 * r1 * (p - v) + c2 * r2 * (g - v)

    for iter in range(max_iter):
        if time.time() - t0 > budget_s - 500:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = update_velocity(velocities[i], personal_best[i], global_best, c1, c2, r1, r2)
            velocities[i] = np.clip(velocities[i], -vel_limit, vel_limit)
            particles[i] += velocities[i]
            particles[i] = clamp_and_rescale(particles[i])
            current_c5 = calc_c5(particles[i])
            if current_c5 < personal_best_val[i]:
                personal_best[i] = particles[i].copy()
                personal_best_val[i] = current_c5
        current_global = np.argmin(personal_best_val)
        if personal_best_val[current_global] < global_best_val:
            global_best = personal_best[current_global].copy()
            global_best_val = personal_best_val[current_global]
        inertia_start *= inertia_decay

    # Refinement
    refine_h = [global_best.copy() for _ in range(refine_count)]
    for h_ref in refine_h:
        h_ref[:], _ = minimize(
            fun=lambda x: calc_c5(x),
            x0=h_ref,
            method='SLSQP',
            bounds=Bounds([0.0]*70, [1.0]*70),
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0},
            tol=1e-10,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        ).x
    best_h = min(refine_h, key=calc_c5)
    best_c5 = calc_c5(best_h)
    return best_h, best_c5, 70
