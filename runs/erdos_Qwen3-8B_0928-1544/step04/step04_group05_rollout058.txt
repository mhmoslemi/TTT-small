def run(seed=42, budget_s=1100, **kwargs):

        return h_clip * s

    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        population.append(project_h(h))

    def evaluate(h):
        return evaluate_erdos_solution(h)

    best_h = initial_h
    best_c5 = evaluate(initial_h)

    for gen in range(max_gen):
        fitness = [evaluate(ind) for ind in population]
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)

        parents = []

            tournament = np.random.choice(population, tournament_size, replace=False)
            f = [evaluate(ind) for ind in tournament]
            best_idx = np.argmin(f)
            parents.append(tournament[best_idx])

        offspring = []
        for i in range(0, pop_size, 2):
            p1, p2 = parents[i], parents[i+1]
            mid = np.random.randint(1, n_points - 1)

            offspring.append(child1)
