import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n)]
    
    result = differential_evolution(
        objective,

        strategy='rand',
        maxiter=200,
        popsize=20,
        tol=1e-8,
        mutation=(0.5, 1),

        polish=True,
        vectorized=True,
