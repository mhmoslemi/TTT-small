import numpy as np
from scipy.optimize import minimize, Bounds
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        np.random.seed(seed)
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    
    n_points = 70
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / sum_h
        return h_clip * scale
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    population_size = 50
    population = [project_h(np.random.rand(n_points)) for _ in range(population_size)]
    population[0] = current_h
    
    for gen in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_pop = population[best_idx]
        
        parents = np.random.choice(population_size, size=population_size, p=fitness / fitness.sum())
        new_population = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.1:
                perturb = np.random.normal(0, 0.05, n_points)
                child = np.clip(child + perturb, 0, 1)
            new_population.append(project_h(child))
        
        population = new_population
        current_h = best_pop
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 150, 'ftol': 1e-9}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
