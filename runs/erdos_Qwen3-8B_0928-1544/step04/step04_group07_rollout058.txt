from scipy.optimize import minimize, Bounds

    np.random.seed(seed)
    n = 70
    dx = 2 / n
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vec) - 70 / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h_vec):
        h_clip = np.clip(h_vec, 0, 1)
        total = np.sum(h_clip)
        if total == 0:
            return np.zeros(n)
        scale = (70 / 2) / total
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = np.copy(h)
    best_h = project_h(best_h)
    current_h = best_h.copy()
    current_c5 = objective(current_h)
    best_c5 = min(best_c5, current_c5)
