import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:

        scale = (n / 2) / s
        return h_clip * scale
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    pop_size = 100
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n)
        population.append(project_h(h))
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    generations = 200
    tournament_size = 5
    mutation_rate = 0.05
    
    for _ in range(generations):
        fitness = [compute_c5(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        new_population = []

            t = np.random.choice(population, tournament_size)
            t_fit = [compute_c5(ind) for ind in t]
            p1 = t[np.argmin(t_fit)]
            t2 = np.random.choice(population, tournament_size)
            t_fit2 = [compute_c5(ind) for ind in t2]
            p2 = t2[np.argmin(t_fit2)]
            
            if np.random.rand() < 0.25:
                pt = np.random.randint(1, n)
                c1 = np.concatenate((p1[:pt], p2[pt:]))

            else:

                c2 = project_h(c2)
