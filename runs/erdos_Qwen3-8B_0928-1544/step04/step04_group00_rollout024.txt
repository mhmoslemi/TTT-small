import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        s = np.sum(h)
        if s == 0:
            return h
        return h * (n_points / 2) / s
    
    population_size = 30
    if 'initial_h_values' in globals():
        initial_population = [initial_h + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
        initial_population = [np.clip(p, 0, 1) for p in initial_population]
    else:
        t = np.linspace(0, 2, n_points)
        initial_population = [0.5 * (1 + np.sin(2 * np.pi * t + np.random.uniform(0, 2 * np.pi))) for _ in range(population_size)]
    
    initial_population = [normalize_sum(h) for h in initial_population]
    best_h = initial_population[0]
    best_c5 = evaluate(best_h)
    for h in initial_population:
        c5 = evaluate(h)
        if c5 < best_c5:

            best_h = h

            pert = np.random.normal(0, 0.1, n_points)
            new_h = h + pert

            new_population.append(new_h)
