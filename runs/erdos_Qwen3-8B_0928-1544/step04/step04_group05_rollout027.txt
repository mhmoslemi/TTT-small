import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):

    else:
        initial_h = np.full(70, 0.5)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Process initial_h if available
    if initial_h is not None:
        h_sum = np.sum(initial_h)
        if h_sum == 0:
            h_scaled = np.full(n_points, 0.5)
        else:
            h_scaled = initial_h * (n_points / 2) / h_sum
            h_scaled = np.clip(h_scaled, 0, 1)
        population.append(h_scaled)
    
    # Fill the rest with random samples
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h_scaled = np.full(n_points, 0.5)
        else:
            h_scaled = h * (n_points / 2) / h_sum
            h_scaled = np.clip(h_scaled, 0, 1)
        population.append(h_scaled)
    
    best_h = None
    best_c5 = float('inf')
    
    # Parameters
    tournament_size = 10
    mutation_rate = 0.1
    max_generations = 100
    
    start_time = time.time()
    
    for gen in range(max_generations):
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Update best solution
        min_c5 = min(fitness)
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = population[fitness.index(min_c5)]
        
        # Selection: tournament selection
        parents = []
        for _ in range(pop_size):
            tournament = random.sample(population, tournament_size)
            best_tournament = min(tournament, key=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx)
            parents.append(best_tournament)
        
        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else parents[0]
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
            offspring.append(child1)
            offspring.append(child2)
        
        # Mutation
        mutated_offspring = []
        for h in offspring:
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
                h_mut = h.copy()
                h_mut[indices] += np.random.normal(0, 0.03, size=len(indices))

                    h_mut = h_mut * (n_points / 2) / h_sum
                mutated_offspring.append(h_mut)
            else:
                mutated_offspring.append(h)
        
        # Replace population with mutated offspring
        population = mutated_offspring
        
        # Check if time is up
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
    
    # Ensure best solution is set
    if best_h is None:
        best_h = population[0]
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return best_h, best_c5, n_points
