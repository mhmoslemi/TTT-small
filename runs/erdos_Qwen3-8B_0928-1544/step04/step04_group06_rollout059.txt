import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    n = 70
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    particle_count = 100
    particles = []
    velocities = []
    for _ in range(particle_count):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.05, size=n)
        h += noise
        h = project_h(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, size=n))
    
    best_c5 = objective(initial_h)
    best_h = initial_h.copy()
    
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    for _ in range(500):
        for i in range(particle_count):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (best_h - particles[i]) + c2 * r2 * (best_h - particles[i])
            vel = np.clip(vel, -v_max, v_max)
            velocities[i] = vel
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])
            current_c5 = objective(particles[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = particles[i].copy()
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n
