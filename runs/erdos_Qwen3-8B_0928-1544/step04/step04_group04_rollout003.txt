import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:

        return h_clip * scale

        proj_h = project_h(h)

        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n)]

        objective,
