import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Generate diverse starting points
    start_points = [h_initial]
    # Add a uniform starting point
    start_points.append(np.full(n_points, 0.5))
    # Add perturbed versions
    for _ in range(2):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_initial + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        start_points.append(h_perturbed)
    # Add random binary patterns
    for _ in range(2):
        h_binary = np.random.rand(n_points) < 0.2
        h_binary = h_binary * (35.0 / np.sum(h_binary))
        start_points.append(h_binary + 0.5)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
    
    for h_start in start_points:
        result = minimize(
            fun=objective,
            x0=h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
