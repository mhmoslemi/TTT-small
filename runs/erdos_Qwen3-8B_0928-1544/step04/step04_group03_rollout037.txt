from scipy.optimize import minimize
import time

    n_points = 70
    dx = 2.0 / n_points
    required_sum = 35.0
    population_size = 50
    max_generations = 200
    tournament_size = 5
    mutation_scale = 0.05
    mutation_rate = 0.1

    # Initialize population
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial = initial_h_values.copy()

        population = [initial]

            h = np.random.rand(n_points)
            h = (h / np.sum(h)) * required_sum
            population.append(h)
    else:
        population = [np.random.rand(n_points) * required_sum / np.sum(np.random.rand(n_points)) for _ in range(population_size)]

    # Fitness function
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Mutation function
    def mutate(h):
        noise = np.random.normal(0, mutation_scale, h.shape)
        h = np.clip(h + noise, 0, 1)
        return (h / np.sum(h)) * required_sum
