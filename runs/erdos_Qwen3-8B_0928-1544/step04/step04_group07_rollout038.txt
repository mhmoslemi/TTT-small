import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    np.random.seed(seed)
    
    n = 70
    dx = 2.0 / n
    
    def project_h(h_candidate):
        h_clip = np.clip(h_candidate, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    def objective(h_candidate):
        h = project_h(h_candidate)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    start_time = time.time()
    
    # Evolutionary optimization for global search
    result_de = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        xtol=1e-8
    )
    elapsed = time.time() - start_time
    
    # Local refinement with SLSQP if time remains
    if elapsed < budget_s * 0.7:
        def constraint_sum(h):
            return np.sum(h) - n / 2
        
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds_slsqp = Bounds([0.0]*n, [1.0]*n)
        
        result_slsqp = minimize(
            fun=objective,
            x0=result_de.x,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            best_c5 = min(result_de.fun, result_slsqp.fun)
            best_h = result_slsqp.x if result_slsqp.fun < result_de.fun else result_de.x
        else:
            best_c5, best_h = result_de.fun, result_de.x
    else:
        best_c5, best_h = result_de.fun, result_de.x
    
    return best_h, best_c5, n
