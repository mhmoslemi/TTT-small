import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    sum_required = 35.0

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(35):
            pos = int(round(i * (n_points / 35)))
            h[pos] = 1.0
        if np.sum(h) != sum_required:
            diff = sum_required - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    T = 1.0
    alpha = 0.999
    sa_max_iter = 500

    for sa_iter in range(sa_max_iter):
        noise = np.random.normal(0, T, size=n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (sum_required / sum_perturbed)

        E_current = compute_c5(h)
        E_perturbed = compute_c5(h_perturbed)

        if E_perturbed < E_current:
            h = h_perturbed
        else:
            delta = E_perturbed - E_current
            prob = np.exp((E_current - E_perturbed) / T)
            if np.random.rand() < prob:
                h = h_perturbed

        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()

        T *= alpha
        if time.time() - t0 > budget_s - 500:
            break

    def objective(x):
        return compute_c5(x)

    def constraint_sum(x):
        return np.sum(x) - sum_required

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    result = minimize(fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-8, options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})

        best_c5 = result.fun
