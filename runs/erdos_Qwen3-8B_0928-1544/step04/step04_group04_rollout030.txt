import numpy as np
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_iter = 100
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros_like(h_clip)
        scale = (n_points / 2) / sum_h
        return h_clip * scale
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def evaluate_fitness(h):
        return compute_c5(h)
    
    def update_velocity(inertia, cognitive, social, r1, r2, vel, pbest, gbest, h):
        vel = inertia * vel + cognitive * r1 * (pbest - h) + social * r2 * (gbest - h)
        return np.clip(vel, -0.5, 0.5)
    
    particles = np.random.rand(population_size, n_points)
    particles = (particles / np.sum(particles, axis=1, keepdims=True) * 35.0)
    particles = np.array([project_h(p) for p in particles], dtype=np.float64)
    
    pbest = particles.copy()
    pbest_fitness = np.array([evaluate_fitness(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    
    velocities = np.random.uniform(-0.5, 0.5, size=(population_size, n_points))
    
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    for iteration in range(max_iter):
        for i in range(population_size):
            vel = update_velocity(inertia, cognitive, social, np.random.rand(), np.random.rand(), velocities[i], pbest[i], gbest, particles[i])
            new_h = particles[i] + vel
            new_h = project_h(new_h)
            new_fitness = evaluate_fitness(new_h)
            if new_fitness < pbest_fitness[i]:
                pbest[i] = new_h
                pbest_fitness[i] = new_fitness
            if new_fitness < gbest_fitness:
                gbest = new_h
                gbest_fitness = new_fitness
        inertia *= 0.995
    
    return gbest, gbest_fitness, n_points
