from scipy.optimize import minimize, Bounds
import time

        h = np.full(70, 0.5)
    h = h * (35.0 / np.sum(h))
    n_points = 70
    dx = 2.0 / n_points
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    max_iterations = 500
    iterations = 0
    progress = 0
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        # Compute cross-correlation and find max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Find shift with maximum overlap
        k_max = np.argmax(corr) - n_points + 1
        if k_max < 0 or k_max >= n_points:

            for i in range(n_points):
