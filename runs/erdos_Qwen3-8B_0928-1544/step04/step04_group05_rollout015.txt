        h = np.full(70, 0.5)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = h.copy()
    current_h = project_h(h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    for _ in range(150):
        perturbation = np.random.normal(0, 0.02, size=n)
        perturbed_h = current_h + perturbation
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 300, 'ftol': 1e-9}
        )
        
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x
                current_h = result.x
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    return best_h, best_c5, n
