import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(35):
            pos = int(i * n_points / 35)
            h[pos] = 1.0
        if np.sum(h) != 35:
            diff = 35 - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    T = 1.0
    cooling_rate = 0.999
    steps = 0
    max_steps = int(budget_s * 0.6)
    while steps < max_steps:
        if time.time() - t0 > budget_s - 100:
            break
        noise = np.random.normal(0, T, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) > 0:
            perturbed *= 35.0 / np.sum(perturbed)
        new_c5 = compute_c5(perturbed)
        if new_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = new_c5
        else:
            prob = np.exp((best_c5 - new_c5) / T)
            if np.random.rand() < prob:
                best_h = perturbed.copy()
                best_c5 = new_c5
        T *= cooling_rate
        steps += 1
    # Final refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    result = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons,

    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return best_h, best_c5, n_points

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
