import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h)
        if sum_h == 0:
            initial_h = np.zeros(70)
        else:
            initial_h = initial_h * (35.0 / sum_h)
    
    n = 70
    dx = 2.0 / n

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = 35.0 / sum_h
        return h_clip * scale

    population_size = 50
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_vel = 0.5
    particles = [initial_h.copy() for _ in range(population_size)]
    vel = [np.zeros(n) for _ in range(population_size)]

    pbest = particles.copy()
    pbest_fitness = [objective(p) for p in pbest]
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)

    start = time.time()

    for _ in range(1000):
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel[i] = inertia * vel[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            vel[i] = np.clip(vel[i], -max_vel, max_vel)
            particles[i] += vel[i]
            particles[i] = project(particles[i])
        for i in range(population_size):
            curr = objective(particles[i])
            if curr < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = curr
        curr_gbest = np.argmin(pbest_fitness)
        if pbest_fitness[curr_gbest] < gbest_fitness:
            gbest = pbest[curr_gbest].copy()
            gbest_fitness = pbest_fitness[curr_gbest]
        if time.time() - start > budget_s:
            break

    return gbest, gbest_fitness, n
