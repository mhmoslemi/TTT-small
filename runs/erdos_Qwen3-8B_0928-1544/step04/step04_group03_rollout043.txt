import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        n = 70
        ones = 35
        interval = n / ones
        for i in range(ones):
            pos = round(i * interval)
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial.copy()
    best_c5 = float('inf')
    t0 = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)

    T, alpha = 1.0, 0.9985
    max_iter = 200
    best_h = h_initial.copy()
    best_c5 = float('inf')

    for step in range(max_iter):
        if time.time() - t0 > budget_s - 100:
            break
        noise_std = T * 0.1
        perturbation = np.random.normal(0, noise_std, 70)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed != 0:
            h_perturbed = h_perturbed * (35.0 / sum_perturbed)
        else:
            h_perturbed = np.zeros_like(h_perturbed)
        E_new = objective(h_perturbed)
        E_old = objective(h_initial)
        if E_new < E_old:
            accept = True
        else:
            prob = np.exp((E_old - E_new) / T)
            accept = np.random.rand() < prob
        if accept:
            h_initial = h_perturbed
        if E_new < best_c5:
            best_c5 = E_new
            best_h = h_initial.copy()
        T *= alpha

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    return best_h, best_c5, 70
