    h = h * (35.0 / np.sum(h))
    n_points = 70
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_iter = 500
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx_max = np.argmax(corr)
        k_shift = idx_max - n_points + 1
        max_overlap = max_corr * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        def obj_func(x):
            h_vec = x.copy()
            total = 0.0
            for i in range(n_points):
                j = (i + k_shift) % n_points
                total += h_vec[i] * (1 - h_vec[j])
            return total * dx
        
        def cons(x):
            return np.sum(x) - 35.0
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8, options={'maxiter': 100})
        
        if result.success:
            h = result.x
    
    return best_h, best_c5, n_points
