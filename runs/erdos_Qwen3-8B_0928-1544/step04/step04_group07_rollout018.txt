import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project(x):
        x_clip = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x_clip)
        if sum_x == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_x
        return x_clip * scale
    
    best_c5 = float('inf')
    current = project(h)
    best_h = current
    best_c5 = objective(current)
