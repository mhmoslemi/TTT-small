import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = n_points / 2
    initial_population = []
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_population.append(initial_h)
    
    # Generate sinusoidal or random initial variants
    for _ in range(29):
        if 'initial_h_values' in globals():
            base = initial_h_values.copy()
        else:
            x = np.linspace(0, 2, n_points)
            base = 0.5 * (1 + np.sin(np.pi * x))
        noise = np.random.normal(0, 0.1, n_points)

        h = np.clip(h, 0, 1)
        h = h * sum_h / np.sum(h)
        initial_population.append(h)
    
    population = np.array(initial_population)
    best_c5 = float('inf')
    best_h = None
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for idx in range(len(population)):
        c = evaluate(population[idx])
        if c < best_c5:
            best_c5 = c
            best_h = population[idx].copy()
    
    start_time = time.time()
    max_iter = int(budget_s // 10)
    
    for _ in range(max_iter):
        c5s = [evaluate(h) for h in population]
        indices = np.argsort(c5s)
        best = indices[:5]
        worst = indices[-2:]
        
        new = []
        for i in best:
            perturb = np.random.normal(0, 0.15, n_points)
            h = population[i] + perturb
            h = np.clip(h, 0, 1)
            h = h * sum_h / np.sum(h)

            best_c5 = evaluate(population[0])
            best_h = population[0].copy()
            
        if time.time() - start_time > budget_s:
