    n_points = 70
    dx = 2.0 / n_points
    required_sum = n_points / 2

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        interval = n_points / ones

            pos = round(i * interval)
            h_initial[pos] = 1.0

    for _ in range(swarm_size - 1):
        perturbation = np.random.normal(0, 0.05, n_points)

    c2 = 2.0
