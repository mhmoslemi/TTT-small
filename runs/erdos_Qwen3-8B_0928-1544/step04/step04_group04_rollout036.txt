import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    dx = 2.0 / 70
    n = 70

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    def project_h(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale

    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        new_h = np.random.rand(n)
        new_h = (new_h / np.sum(new_h) * 35.0)
        population.append(new_h.copy())

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for gen in range(50):
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        parents = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, 3, replace=False)
            best_t = tournament[np.argmin(fitness[tournament])]
            parents.append(population[best_t])

        children = []
        for i in range(0, population_size, 2):
            if np.random.rand() < 0.7:
                p1, p2 = parents[i], parents[i+1]
                child1 = np.where(np.random.rand(n) < 0.5, p1, p2)
                child2 = np.where(np.random.rand(n) < 0.5, p2, p1)
                children.extend([child1, child2])
            else:
                children.extend([parents[i], parents[i+1]])

        for i in range(len(children)):
            if np.random.rand() < 0.05:
                children[i] += np.random.normal(0, 0.05, n)
            children[i] = project_h(children[i])

        population = children

        if time.time() - start_time > budget_s:
            break

    top_candidates = sorted(population, key=lambda x: objective(x))[:5]
    for candidate in top_candidates:
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 150, 'ftol': 1e-9}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

    return best_h, best_c5, n
