        scale = (n / 2) / sum_h
        return h_clip * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def tournament_selection(population, k=5):

        return min(indices, key=lambda i: objective(population[i]))
    
    population = [initial_h.copy()]

    best_c5 = objective(best_h)
