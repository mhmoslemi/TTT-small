    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    def find_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        return max_index - (n - 1)
    
    for _ in range(300):
        k_max = find_k_max(current_h)
        def grad_func(h):
            grad = np.zeros(len(h))
            for i in range(len(h)):
                if i + k_max < len(h):
                    grad[i] += (1 - h[i + k_max]) * dx
                if i - k_max >= 0:
                    grad[i] -= h[i - k_max] * dx
            return grad
        
        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            jac=grad_func,
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            new_h = project_h(result.x)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
            current_h = new_h
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        jac=None,
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    return best_h, best_c5, n
