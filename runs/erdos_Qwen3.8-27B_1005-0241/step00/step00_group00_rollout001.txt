import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(a, target_sum):
    """Project a onto {h : 0 <= h[i] <= 1, sum(h) = target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(a + mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(a + (lo + hi) / 2.0, 0.0, 1.0)

def c5_and_grad(h, beta):
    """Compute log-sum-exp approximation of C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = corr * dx
    
    # Log-sum-exp with temperature 1/beta
    max_val = np.max(scaled)
    exp_vals = np.exp(beta * (scaled - max_val))
    w = exp_vals / np.sum(exp_vals)
    lse = max_val + np.log(np.sum(exp_vals)) / beta
    
    # Gradient
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        l = lags[k]
        # First factor: (1-h[j-l]) for valid j
        j_lo = max(0, l)
        j_hi = min(n-1, l + n - 1)
        if j_lo <= j_hi:
            j_idx = np.arange(j_lo, j_hi + 1)
            i_idx = j_idx - l
            grad[j_lo:j_hi+1] += w[k] * (1 - h[i_idx])
        # Second factor: -h[j+l] for valid j
        j_lo2 = max(0, -l)
        j_hi2 = min(n-1, n - 1 - l)
        if j_lo2 <= j_hi2:
            j_idx = np.arange(j_lo2, j_hi2 + 1)
            i_idx = j_idx + l
            grad[j_lo2:j_hi2+1] -= w[k] * h[i_idx]
    
    return lse, dx * grad

    if 'initial_h_values' in dir() or True:

    h = project_feasible(h, target_sum)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()

    beta = 10.0
    lr = 0.01
    no_improve = 0
    
    while time.time() - start_time < budget_s - 5:

        h_new = h - lr * grad

        h_new = project_feasible(h_new, target_sum)
        
        # Check improvement
        c5_new = compute_c5(h_new)

                    beta *= 1.5
                    if beta > 1000:

        best_h, best_c5 = h1, c5_1

    time_left = budget_s - (time.time() - t_start) - 10
    if time_left < 30:
        return best_h, best_c5, len(best_h)

    # Phase 2: try different beta/lr settings on same n
    phase2_time = min(time_left * 0.2, 120)
    h2, c5_2 = optimize_h(h1, beta_start=50.0, lr_start=0.02, max_time=phase2_time)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    time_left = budget_s - (time.time() - t_start) - 10
    if time_left < 30:
        return best_h, best_c5, len(best_h)

    # Phase 3: upsample to 2n
    phase3_time = min(time_left * 0.3, 200)
    h_up = upsample(best_h, 2)
    h3, c5_3 = optimize_h(h_up, beta_start=15.0, lr_start=0.08, max_time=phase3_time)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    time_left = budget_s - (time.time() - t_start) - 10
    if time_left < 30:
        return best_h, best_c5, len(best_h)

    # Phase 4: further optimize the 2n solution
    phase4_time = min(time_left * 0.4, 250)
    h4, c5_4 = optimize_h(h3, beta_start=40.0, lr_start=0.04, max_time=phase4_time)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

    time_left = budget_s - (time.time() - t_start) - 10
    if time_left < 20:
        return best_h, best_c5, len(best_h)

    # Phase 5: upsample to 3n from original
    phase5_time = min(time_left * 0.5, 200)
    h_up3 = upsample(best_h if len(best_h) == n else h1, 3 if len(best_h) == n else 1)
    if len(h_up3) > 300:
        # downsample if too large
        n_target = 135
        x_src = np.linspace(0, 2, len(h_up3), endpoint=False)
        x_dst = np.linspace(0, 2, n_target, endpoint=False)
        h_up3 = np.interp(x_dst, x_src, np.concatenate([h_up3, h_up3[:1]]))
        h_up3 = project_feasible(h_up3, n_target / 2.0)
    h5, c5_5 = optimize_h(h_up3, beta_start=20.0, lr_start=0.06, max_time=phase5_time)
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5

    # Final verification
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
