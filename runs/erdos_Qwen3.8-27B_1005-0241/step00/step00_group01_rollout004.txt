import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        adj = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = adj.sum()
        if cnt == 0:
            adj = h < 1.0 if diff > 0 else h > 0.0
            cnt = adj.sum()
        if cnt == 0:
            break
        step = diff / cnt
        if diff > 0:
            room = (1.0 - h[adj]).min()
            if step > room:
                step = room * 0.5
        else:
            room = h[adj].min()
            if -step > room:
                step = -room * 0.5
        h[adj] += step
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    scaled = beta * (corr - mc)
    ex = np.exp(scaled)
    ws = ex.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[corr.argmax()] = 1.0
    else:
        w = ex / ws
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        wi = w[lag + n - 1]
        if wi < 1e-12:
            continue
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:-lag])
            grad[:-lag] -= wi * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:n - L]
    return grad * dx

def optimize(h0, deadline, lr0=0.04, beta0=10.0, mom=0.0):
    h = project(h0.copy(), len(h0) / 2.0)
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = lr0
    beta = beta0
    stall = 0
    vel = np.zeros(n)
    while time.time() < deadline:
        g = compute_grad(h, beta)
        vel = mom * vel - lr * g
        h_new = h + vel
        h_new = project(h_new, n / 2.0)
        val = compute_c5(h_new)
        if val < best_val - 1e-10:
            best_val = val
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.005, 0.3)
        else:
            stall += 1
            lr *= 0.98
            if stall > 200:
                h = best_h + 0.03 * np.random.randn(n)
                h = project(h, n / 2.0)
                vel = np.zeros(n)
                lr = lr0
                stall = 0
        h = h_new
        beta = min(beta * 1.002, 80.0)
    return best_h, best_val

def make_block_init(n, rng):
    h0 = np.zeros(n)
    n_ones = n // 2
    if n % 2 == 1:
        n_ones = (n + 1) // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h0[idx] = 1.0
    return project(h0, n / 2.0)

def make_sin_init(n, freq):
    i = np.arange(n)
    h0 = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * i / n)
    return project(h0, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def _update(h, c5):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, len(h)

    # Phase 1: optimize initial_h_values (40% budget)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        c5_init = compute_c5(h_init)
        _update(h_init, c5_init)
        d1 = t0 + (deadline - t0) * 0.35
        if time.time() < d1:
            h1, c5_1 = optimize(h_init, d1, lr0=0.03, beta0=8.0, mom=0.3)
            _update(h1, c5_1)

    # Phase 2: second pass on initial with different params (20%)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        d2 = t0 + (deadline - t0) * 0.55
        if time.time() < d2:
            h2 = np.array(initial_h_values, dtype=float)
            h2 += 0.02 * rng.randn(len(h2))
            h2 = project(h2, len(h2) / 2.0)
            h2o, c5_2 = optimize(h2, d2, lr0=0.02, beta0=15.0, mom=0.5)
            _update(h2o, c5_2)

    # Phase 3: try other n values (25%)
    d3 = t0 + (deadline - t0) * 0.80
    for n_try in [47, 63, 96]:
        if time.time() > d3:
            break
        tries = 3
        for t in range(tries):
            if time.time() > d3:
                break
            if t == 0:
                h0 = make_block_init(n_try, rng)
            elif t == 1:
                h0 = make_sin_init(n_try, freq=3 + t)
            else:
                h0 = make_block_init(n_try, rng)
            te = min(d3, time.time() + (d3 - time.time()) / (tries - t))
            ho, cv = optimize(h0, te, lr0=0.02, beta0=12.0, mom=0.4)
            _update(ho, cv)

    # Phase 4: final polish on best (15%)
    if best_h is not None and time.time() < deadline - 5:
        hp, cvp = optimize(best_h, deadline - 3, lr0=0.005, beta0=30.0, mom=0.6)
        _update(hp, cvp)

    # Final check: ensure initial is considered
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_i = np.array(initial_h_values, dtype=float)
        c5_i = compute_c5(h_i)
        _update(h_i, c5_i)

    if best_h is None:
        n = 47
        best_h = project(np.ones(n) * 0.5, n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
