```python
import numpy as np

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h[i]<=1, sum(h)=target_sum}"""
    # Use bisection on lambda
    # h[i] = clip(x[i] - lambda, 0, 1)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0, 1)
    # Adjust for exact sum (numerical)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among interior points
        interior = (h > 1e-10) & (h < 1 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_overlap_shifts(h, dx):
    """Compute all shift overlaps"""
    n = len(h)
    g = 1 - h
    corr = np.correlate(h, g, mode="full") * dx
    return corr

def lse_approx(overlaps, beta):
    """Log-sum-exp approximation of max"""
    max_ov = np.max(overlaps)
    shifted = beta * (overlaps - max_ov)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    return max_ov + np.log(np.sum(np.exp(shifted))) / beta

def lse_grad_weights(overlaps, beta):
    """Softmax weights for log-sum-exp gradient"""
    max_ov = np.max(overlaps)
    shifted = beta * (overlaps - max_ov)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def gradient_objective(h, beta, target_sum, mu=0.0):
    """Compute gradient of log-sum-exp(overlaps) + mu * penalty"""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1-h, mode="full") * dx
    weights = lse_grad_weights(overlaps, beta)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if abs(w) < 1e-15:
            continue
        s = k - (n-1)  # actual shift
        for j in range(n):
            if 0 <= j + s < n:
                grad[j] += w * (1 - h[j + s])
            if 0 <= j - s < n:
                grad[j] -= w * h[j - s]
    
    if mu > 0:
        # Penalty for constraint violation
        sum_h = h.sum()
        grad += 2 * mu * (sum_h - target_sum)
    
    return grad

def optimize_h(h0, target_sum, n, max_iters=5000, beta_start=10, beta_end=1000, 
               lr_start=0.01, lr_end=0.001, time_limit=1000):
    """Projected gradient descent with log-sum-exp"""
    import time
    start_time = time.time()
    h = h0.copy()
    h = project_feasible(h, target_sum)
    
    best_c5 = np.inf
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        
        # Compute gradient
        grad = gradient_objective(h, beta, target_sum)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project
        h_new = project_feasible(h_new, target_sum)
        
        # Evaluate
        c5 = compute_overlap_shifts(h_new, 2.0/n).max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5
```

```python
    _, weights = lse_and_weights(overlaps, beta)

        if abs(w) < 1e-18:

        s = k - (n - 1)
```

```python
            grad[s:] += w * one_minus_h[:n-s]

            grad[:n-sp] += w * one_minus_h[sp:]
            # h[j-sp] valid for j in [sp, n-1]
            grad[sp:] -= w * h[:n-sp]
    
    return grad
```

```python
def project_feasible(x, target_sum):
```
```python
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            np.clip(h, 0.0, 1.0, out=h)
```

```python
import numpy as np
import time

def project_feasible(x, target_sum):
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def lse_weights(ov, beta):
    m = ov.max()
    sh = np.clip(beta * (ov - m), -500.0, 500.0)
    e = np.exp(sh)
    return e / e.sum()

def grad_lse(h, beta):
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full") * dx
    w = lse_weights(ov, beta)
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += wk * omh[s:]
            g[s:] -= wk * h[:n - s]
        else:
            sp = -s
            g[sp:] += wk * omh[:n - sp]
            g[:n - sp] -= wk * h[sp:]
    return g

def opt_pgd(h0, tlimit, b0, b1, lr0, lr1, sseed):
    np.random.seed(sseed)
    n = len(h0)
    ts = n / 2.0
    h = project_feasible(h0.astype(float).copy(), ts)
    bc5 = compute_c5(h)
    bh = h.copy()
    t0 = time.time()
    pat = 0
    while time.time() - t0 < tlimit:
        frac = min((time.time() - t0) / max(tlimit, 1.0), 1.0)
        beta = b0 * (b1 / b0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        g = grad_lse(h, beta)
        h_new = project_feasible(h - lr * g, ts)
        c5 = compute_c5(h_new)
        if c5 < bc5 - 1e-13:
            bc5 = c5
            bh = h_new.copy()
            pat = 0
        else:
            pat += 1
        h = h_new
        if pat > 150:
            h = bh + 0.01 * np.random.randn(n)
            h = project_feasible(h, ts)
            pat = 0
    return bh, bc5

def up2(h):
    n = len(h)
    out = np.empty(2 * n)
    out[0::2] = h
    out[1::2] = h
    return out

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    T0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n0

    # Phase 1: Optimize from initial construction
    rem = budget_s - (time.time() - T0)
    if rem > 60:
        t1 = min(rem * 0.35, 350)
        h1, c1 = opt_pgd(h0, t1, 5.0, 500.0, 0.05, 0.001, seed)
        if c1 < best_c5:
            best_h, best_c5, best_n = h1, c1, n0

    # Phase 2: Upscale to 2x and optimize
    rem = budget_s - (time.time() - T0)
    if rem > 60:
        hu = up2(best_h)
        hu = project_feasible(hu, len(hu) / 2.0)
        t2 = min(rem * 0.3, 300)
        h2, c2 = opt_pgd(hu, t2, 5.0, 500.0, 0.05, 0.001, seed + 1)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, len(hu)

    # Phase 3: High-beta refinement
    rem = budget_s - (time.time() - T0)
    if rem > 30:
        t3 = min(rem * 0.3, 200)
        h3, c3 = opt_pgd(best_h, t3, 100.0, 2000.0, 0.01, 0.0003, seed + 2)
        if c3 < best_c5:
            best_h, best_c5, best_n = h3, c3, best_n

    # Phase 4: Random restarts
    attempts = 0
    while budget_s - (time.time() - T0) > 50 and attempts < 4:
        hr = np.random.randint(0, 2, best_n).astype(float)
        hr = project_feasible(hr, best_n / 2.0)
        t4 = min((budget_s - (time.time() - T0)) / (4 - attempts + 1), 100)
        h4, c4 = opt_pgd(hr, t4, 10.0, 400.0, 0.03, 0.001, seed + 10 + attempts)
        if c4 < best_c5:
            best_h, best_c5, best_n = h4, c4, best_n
        attempts += 1

    # Phase 5: Final polish on best solution
    rem = budget_s - (time.time() - T0)
    if rem > 20:
        t5 = min(rem * 0.7, 100)
        h5, c5_5 = opt_pgd(best_h, t5, 200.0, 5000.0, 0.005, 0.0001, seed + 100)
        if c5_5 < best_c5:
            best_h, best_c5, best_n = h5, c5_5, best_n

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
```
