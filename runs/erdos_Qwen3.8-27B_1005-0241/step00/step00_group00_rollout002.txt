The sum constraint complicates things a bit. If we change h[j], we need to adjust another variable to maintain the sum. This makes simple coordinate descent not directly applicable.

One approach: use a pair swap. Pick two indices i and j, and transfer mass from one to the other (h[i] += δ, h[j] -= δ). This maintains the sum constraint. Then optimize δ for this pair.

For a pair (i, j) and transfer δ:
h[i] += δ, h[j] -= δ, all others fixed.
The objective becomes max_k (A_k * (h[i]+δ) + (-A_k') * (h[j]-δ) + B_k)
= max_k ((A_k - A_k') * δ + C_k)

where A_k is the coefficient of h[i] in c_k, A_k' is the coefficient of h[j] in c_k, and C_k is the rest.

This is a max of linear functions in δ, which is convex. We can minimize it by finding the δ where the lower envelope of the max is minimized. This is a 1D convex optimization that can be solved exactly or with bisection.

import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    n = len(h)
    h = h.copy()
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

def lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = beta * corr
    v -= v.max()
    e = np.exp(v)
    se = e.sum()
    obj = (v.max() + np.log(se)) / (beta * dx)
    w = e / se
    g = np.zeros(n)
    half = n - 1
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-12:
            continue
        k = ki - half
        lo = max(0, -k)
        hi = min(n, n - k)
        xv = np.arange(lo, hi)
        g[xv + k] += wi * (1.0 - h[xv])
        g[xv] -= wi * h[xv + k]
    return obj, g / dx

def pg_step(h, grad, ts, lr):
    n = len(h)
    def s(lam):
        return np.clip(h - lr * (grad + lam), 0, 1).sum()
    lo, hi = -100.0, 100.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if s(mid) > ts:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(h - lr * (grad + lam), 0, 1)
    if abs(result.sum() - ts) > 1e-6:
        result = project_feasible(result, ts)
    return result

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_init = 45
        h0 = np.zeros(n_init)
        h0[::2] = 1.0
        h0 = project_feasible(h0, n_init / 2.0)

    n = len(h0)
    ts = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    starts = [best_h.copy()]
    for std in [0.03, 0.08, 0.15]:
        starts.append(project_feasible(best_h + rng.normal(0, std, n), ts))
    starts.append(project_feasible(rng.uniform(0, 1, n), ts))
    h_bin = (rng.uniform(0, 1, n) > 0.5).astype(float)
    starts.append(project_feasible(h_bin, ts))

    for h_start in starts:
        if rem() < 30:
            break
        h = h_start.copy()
        beta = 2.0
        lr = 5e-4
        for stage in range(25):
            if rem() < 10:
                break
            stall = 0
            obj_prev = lse_obj_grad(h, beta)[0]
            for it in range(3000):
                if rem() < 10:
                    break
                obj, g = lse_obj_grad(h, beta)
                h_new = pg_step(h, g, ts, lr)
                obj_new, _ = lse_obj_grad(h_new, beta)
                if obj_new <= obj_prev + 1e-12:
                    h = h_new
                    obj_prev = obj_new
                    stall = 0
                else:
                    lr *= 0.6
                    stall += 1
                    if lr < 1e-8:
                        break

                if it % 50 == 49:
                    c = compute_c5(h)
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
            beta *= 1.35
            if beta > 300:
                beta = 300
            lr = max(lr * 1.5, 2e-4)

        if rem() < 15:
            break

        # Subgradient polish
        h = best_h.copy()
        no_imp = 0
        for it in range(8000):
            if rem() < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            ki = int(corr.argmax())
            k = ki - (n - 1)
            lo = max(0, -k)
            hi = min(n, n - k)
            xv = np.arange(lo, hi)
            sg = np.zeros(n)
            sg[xv + k] += 1.0 - h[xv]
            sg[xv] -= h[xv + k]
            c_cur = compute_c5(h)
            improved = False
            for step in [0.01, 0.03, 0.1, 0.3, 1.0]:
                h_try = pg_step(h, sg, ts, step)

                    best_h = h.copy()

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
