    if c_max * beta > 500:

        return c_max

import numpy as np
import time

def _project_simplex(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _compute_c5_full(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return corr.max(), corr

def _compute_gradient(h, p, dx, n):
    g = 1.0 - h
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n)[None, :]
    idx1 = j_idx + (n - 1) - m_idx
    idx2 = m_idx - j_idx + (n - 1)
    first_term = p[idx1] @ g
    second_term = p[idx2] @ h
    return dx * (first_term - second_term)

def _get_softmax_weights(corr, beta):
    c_max = corr.max()
    if beta * c_max > 500:
        mask = corr >= c_max - 1e-12
        p = np.zeros_like(corr)
        p[mask] = 1.0 / mask.sum()
        return p
    shifted = beta * (corr - c_max)
    exp_s = np.exp(shifted)
    return exp_s / exp_s.sum()

def _gradient_optimize(h, dx, target_sum, time_limit, start_time):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.02
    beta = 5.0
    best_h = h.copy()
    best_c5, _ = _compute_c5_full(h, dx)
    no_improve = 0

    while time.time() - start_time < time_limit:
        t += 1
        _, corr = _compute_c5_full(h, dx)
        c5_val = corr.max()

        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            lr *= 0.7
            beta = min(beta * 1.4, 800.0)
            no_improve = 0
            if lr < 1e-9:
                break

        p = _get_softmax_weights(corr, beta)
        grad = _compute_gradient(h, p, dx, n)

        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = _project_simplex(h_new, target_sum)

    return best_h, best_c5

def _continuous_local_search(h, dx, target_sum, time_limit, start_time):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = _project_simplex(h, target_sum)
    best_c5, _ = _compute_c5_full(h, dx)
    best_h = h.copy()
    tolerance = 1e-10

    while time.time() - start_time < time_limit:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                # Try swapping values
                h[i], h[j] = h[j], h[i]
                c5_v, _ = _compute_c5_full(h, dx)
                if c5_v < best_c5 - tolerance:
                    best_c5 = c5_v
                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]

                # Try pushing i up, j down (maintain sum)
                delta = min(h[i], 1.0 - h[j], 0.1)
                if delta > 1e-6:
                    h[i] += delta
                    h[j] -= delta
                    h = _project_simplex(h, target_sum)
                    c5_v, _ = _compute_c5_full(h, dx)
                    if c5_v < best_c5 - tolerance:
                        best_c5 = c5_v
                        best_h = h.copy()
                        improved = True
                    else:
                        h = best_h.copy()

                # Try pushing i down, j up
                delta = min(1.0 - h[i], h[j], 0.1)
                if delta > 1e-6:
                    h = best_h.copy()
                    h[i] -= delta
                    h[j] += delta
                    h = _project_simplex(h, target_sum)
                    c5_v, _ = _compute_c5_full(h, dx)
                    if c5_v < best_c5 - tolerance:
                        best_c5 = c5_v
                        best_h = h.copy()
                        improved = True
                    else:
                        h = best_h.copy()
                else:
                    h = best_h.copy()

        if not improved:
            break

    return best_h, best_c5

def _make_spread_binary(n, seed):
    n_ones = n // 2
    h = np.zeros(n)
    if n_ones == 0:
        return h
    positions = np.round(np.arange(n_ones) * n / float(n_ones)).astype(int)
    positions = positions % n
    seen = set()
    unique_pos = []
    for p in positions:
        if p not in seen:
            seen.add(p)
            unique_pos.append(p)
    if len(unique_pos) < n_ones:
        remaining = [i for i in range(n) if i not in seen]
        rng = np.random.RandomState(seed)
        rng.shuffle(remaining)
        for p in remaining:
            if len(unique_pos) >= n_ones:
                break
            unique_pos.append(p)
    h[unique_pos[:n_ones]] = 1.0
    return h

def _make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    n_ones = n // 2
    pos = np.sort(rng.choice(n, n_ones, replace=False))
    h = np.zeros(n)
    h[pos] = 1.0
    return h

def _binary_swap_search(h, dx, target_sum, time_limit, start_time):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = _project_simplex(h, target_sum)
    best_c5, _ = _compute_c5_full(h, dx)
    best_h = h.copy()
    tol = 1e-10

    while time.time() - start_time < time_limit:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in zeros:
            if len(ones) == 0:
                break
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c5_v, _ = _compute_c5_full(h, dx)
                if c5_v < best_c5 - tol:
                    best_c5 = c5_v
                    best_h = h.copy()
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    improved = True
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
            if improved:
                break
        if not improved:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = _make_spread_binary(45, seed)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_simplex(h0, target_sum)

    best_h = h0.copy()
    best_c5, _ = _compute_c5_full(h0, dx)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Gradient optimization from initial
    if remaining() > 30:
        t_phase = min(remaining() * 0.35, 400)
        h1, c5_1 = _gradient_optimize(h0.copy(), dx, target_sum, t_phase, time.time())
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Binary swap search from initial
    if remaining() > 15:
        t_phase = min(remaining() * 0.2, 120)
        h2, c5_2 = _binary_swap_search(h0, dx, target_sum, t_phase, time.time())
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Binary swap search from gradient result
    if remaining() > 15:
        h_r = np.round(best_h)
        h_r = _project_simplex(h_r, target_sum)
        t_phase = min(remaining() * 0.15, 90)
        h3, c5_3 = _binary_swap_search(h_r, dx, target_sum, t_phase, time.time())
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Continuous local search
    if remaining() > 20:
        t_phase = min(remaining() * 0.15, 90)
        h4, c5_4 = _continuous_local_search(best_h, dx, target_sum, t_phase, time.time())
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Multi-start binary swap search
    if remaining() > 30:
        for trial in range(15):
            if remaining() < 10:
                break
            h_rand = _make_random_binary(n, seed + 5000 + trial)
            h_rand = _project_simplex(h_rand, target_sum)
            t_phase = min(remaining() * 0.1, 40)
            h5, c5_5 = _binary_swap_search(h_rand, dx, target_sum, t_phase, time.time())
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()

    # Phase 6: Try even n values (allow pure binary)
    if remaining() > 30:
        for n_trial in [44, 46, 48, 50, 52, 56]:
            if remaining() < 20:
                break
            dx_t = 2.0 / n_trial
            ts_t = n_trial / 2.0
            h_t = _make_spread_binary(n_trial, seed + n_trial)
            t_phase = min(remaining() * 0.15, 60)
            h_ot, c5_ot = _gradient_optimize(h_t, dx_t, ts_t, t_phase * 0.5, time.time())
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()
            if remaining() > 10:
                h_bt, c5_bt = _binary_swap_search(h_ot, dx_t, ts_t, min(remaining() * 0.1, 30), time.time())
                if c5_bt < best_c5:
                    best_c5 = c5_bt
                    best_h = h_bt.copy()

    # Phase 7: More gradient refinement from best
    if remaining() > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        t_phase = min(remaining() * 0.3, 200)
        h7, c5_7 = _gradient_optimize(best_h.copy(), dx_f, ts_f, t_phase, time.time())
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final local search
    if remaining() > 10:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        t_phase = min(remaining() * 0.4, 60)
        h8, c5_8 = _continuous_local_search(best_h, dx_f, ts_f, t_phase, time.time())
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = _project_simplex(best_h, n_final / 2.0)
    final_c5, _ = _compute_c5_full(best_h, dx_final)

    return best_h, final_c5, n_final
