- When n+k = j, i.e., n = j-k: term is h[j] * (1-h[j-k])
- When n = j: term is h[j+k] * (1-h[j])

So:
∂c_k / ∂h[j] = (1-h[j-k]) - h[j+k]  (when both indices are valid)
= (1-h[j-k]) - h[j+k] + corrections for boundary

More precisely:
∂c_k / ∂h[j] = [j-k is valid ? (1-h[j-k]) : 0] - [j+k is valid ? h[j+k] : 0]

So the new c_k after changing h[j] by delta:
c_k' = c_k + delta * ([j-k valid ? (1-h[j-k]) : 0] - [j+k valid ? h[j+k] : 0])

But wait, we also need to update h[j] itself for the next computation. And if we're changing h[j] to a specific value (not just by delta), we need to be careful.

Actually, for our use case: we want to change h[j] from its current value to a new value v. So delta = v - h[j].

The update is:
for each k (represented by output index i = k + n - 1):
    c_k += delta * (coeff)  where coeff = (1-h[j-k] if 0<=j-k<n else 0) - (h[j+k] if 0<=j+k<n else 0)

Then set h[j] = v.

This is O(n) per coordinate change, which is great!

1. For each pair (j, m), try changing h[j] to 0 or 1, and h[m] to compensate
2. Use the O(n) update to quickly evaluate the change

                corr = corr - d * coefs[j] + d * coefs[m]
                h[j] = 0.0
                h[m] += d
            best_c5 = float(np.max(corr) * dx)
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Perturb from best
                h = best_h.copy()
                corr = np.correlate(h, 1.0 - h, mode="full")
                found = False
                for _ in range(300):
                    idx = np.random.choice(n, size=2, replace=False)
                    d = np.random.uniform(0.05, 0.7)
                    if np.random.rand() < 0.5:
                        d = -d
                    h[idx[0]] += d
                    h[idx[1]] -= d
                    h = np.clip(h, 0, 1)
                    diff = target_sum - np.sum(h)
                    if abs(diff) > 1e-8:
                        for p in range(n):
                            if abs(diff) < 1e-10:
                                break
                            if diff > 0 and h[p] < 1:
                                a = min(diff, 1 - h[p])
                                h[p] += a
                                diff -= a
                            elif diff < 0 and h[p] > 0:
                                a = min(-diff, h[p])
                                h[p] -= a
                                diff += a
                    if abs(target_sum - np.sum(h)) < 1e-8:
                        c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                        if c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = c5
                            found = True
                            break
                stall = 0
                if found:
                    continue

    # Continuous refinement with scipy
    if elapsed() < budget_s - 30:
        try:
            from scipy.optimize import minimize

            def smooth_obj(x, t=0.03):
                c = np.correlate(x, 1.0 - x, mode="full")
                M = c.max()
                return M + t * np.log(np.sum(np.exp((c - M) / t)))

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bnds = [(0.0, 1.0)] * n

            for t in [0.05, 0.02, 0.01]:
                def obj_t(x, tt=t):
                    c = np.correlate(x, 1.0 - x, mode="full")
                    M = c.max()
                    return M + tt * np.log(np.sum(np.exp((c - M) / tt)))

                res = minimize(obj_t, best_h, method='SLSQP', bounds=bnds,
                               constraints=cons, options={'maxiter': 300, 'ftol': 1e-15})
                if res.x is not None:
                    rh = np.clip(res.x, 0, 1)
                    diff = target_sum - rh.sum()
                    if abs(diff) > 1e-8:
                        for p in range(n):
                            if abs(diff) < 1e-10:
                                break
                            if diff > 0 and rh[p] < 1:
                                a = min(diff, 1 - rh[p])
                                rh[p] += a
                                diff -= a
                            elif diff < 0 and rh[p] > 0:
                                a = min(-diff, rh[p])
                                rh[p] -= a
                                diff += a
                    if abs(target_sum - rh.sum()) < 1e-8:
                        c5r = float(np.max(np.correlate(rh, 1.0 - rh, mode="full")) * dx)
                        if c5r < best_c5:
                            best_h = rh.copy()
                            best_c5 = c5r

            # Further local search after scipy
            for _ in range(50):
                corr2 = np.correlate(best_h, 1.0 - best_h, mode="full")
                coefs2 = compute_all_coeffs(best_h)
                improved2 = False
                for j in range(n):
                    if best_h[j] < 1.0 - 1e-10:
                        d = 1.0 - best_h[j]
                        base2 = corr2 + d * coefs2[j]
                        nm = (base2[None, :] - d * coefs2).max(axis=1)
                        valid = (np.arange(n) != j) & (best_h >= d - 1e-10)
                        if valid.any():
                            nm[~valid] = np.inf
                            mb = int(nm.argmin())
                            if nm[mb] * dx < best_c5 - 1e-14:
                                corr2 = base2 - d * coefs2[mb]
                                best_h[j] = 1.0
                                best_h[mb] -= d
                                best_c5 = float(np.max(corr2) * dx)
                                improved2 = True
                    if best_h[j] > 1e-10:
                        d = best_h[j]
                        base2 = corr2 - d * coefs2[j]
                        nm = (base2[None, :] + d * coefs2).max(axis=1)
                        valid = (np.arange(n) != j) & (best_h + d <= 1.0 + 1e-10)
                        if valid.any():
                            nm[~valid] = np.inf
                            mb = int(nm.argmin())
                            if nm[mb] * dx < best_c5 - 1e-14:
                                corr2 = base2 + d * coefs2[mb]
                                best_h[j] = 0.0
                                best_h[mb] += d
                                best_c5 = float(np.max(corr2) * dx)
                                improved2 = True
                if not improved2:
                    break
        except Exception:
            pass

    # Final feasibility fix
    best_h = np.clip(best_h, 0, 1)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                adj = min(diff, 1.0 - best_h[i])
                best_h[i] += adj
                diff -= adj
            else:
                adj = min(-diff, best_h[i])
                best_h[i] -= adj
                diff += adj

    final_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx)
    return best_h, final_c5, n
