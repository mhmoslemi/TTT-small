        else:
            hi = mid

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode="full") * dx
    return c.max()

def project_onto_simplex(h, target_sum):
    n = len(h)
    lo, hi = -h.max() - 1.0, 1.0 - h.min()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1 - h, mode="full")
    bc = beta * c * dx
    bc -= bc.max()
    e = np.exp(bc)
    w = e / e.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)
        wj = w[j] * dx
        if wj < 1e-14:
            continue
        if k >= 0:
            if n > k:
                grad[:n-k] += wj * (1.0 - h[k:])
                grad[k:] -= wj * h[:n-k]
        else:
            nk = -k
            if n > nk:
                grad[nk:] += wj * (1.0 - h[:n-nk])
                grad[:nk] -= wj * h[nk:]
    return grad

def gd_step(h, dx, target_sum, beta, lr):
    grad = lse_gradient(h, dx, beta)
    h_new = project_onto_simplex(h - lr * grad, target_sum)
    return h_new

def grad_descent(h, dx, target_sum, beta, lr, max_iters, deadline):
    best_c5 = compute_c5(h, dx)

    stuck = 0

        if time.time() > deadline:

        h_new = gd_step(best_h, dx, target_sum, beta, lr)
        c5_new = compute_c5(h_new, dx)

            best_h = h_new

                lr = min(lr * 1.05, 0.3)
        else:
            stuck += 1
            lr *= 0.85
            if stuck > 200 or lr < 1e-7:
                break

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode="full") * dx
    return c.max()

def project_onto_simplex(h, target_sum):
    n = len(h)
    lo, hi = -h.max() - 1.0, 1.0 - h.min()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1 - h, mode="full")
    bc = beta * c * dx
    bc -= bc.max()
    e = np.exp(bc)
    w = e / e.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)
        wj = w[j] * dx
        if wj < 1e-14:
            continue
        if k >= 0:
            if n > k:
                grad[:n - k] += wj * (1.0 - h[k:])
                grad[k:] -= wj * h[:n - k]
        else:
            nk = -k
            if n > nk:
                grad[nk:] += wj * (1.0 - h[:n - nk])
                grad[:nk] -= wj * h[nk:]
    return grad

def grad_descent(h, dx, target_sum, beta, lr, max_iters, deadline, rng=None):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = lse_gradient(best_h, dx, beta)
        h_new = project_onto_simplex(best_h - lr * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new
            stuck = 0
            if it > 0 and it % 1000 == 0:
                lr = min(lr * 1.05, 0.3)
        else:
            stuck += 1
            lr *= 0.85
            if stuck > 50 and rng is not None:
                pert = rng.randn(*best_h.shape) * 0.1
                h_try = project_onto_simplex(best_h + pert, target_sum)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    stuck = 0
                    lr = max(lr, 0.05)
            if stuck > 200 or lr < 1e-7:
                break
    return best_h, best_c5

def swap_search(h, dx, deadline, max_rounds=5):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, best_c5

    rng = np.random.RandomState(seed)

    h[idx] = 1.0
    if n % 2 == 1:
        rem = np.where(h == 0.0)[0]
        h[rem[0]] = 0.5
    return h
