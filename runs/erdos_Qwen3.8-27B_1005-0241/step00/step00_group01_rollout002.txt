def compute_corr_components(h):
    """Return all correlation components (without dx factor)"""
    n = len(h)
    one_minus_h = 1.0 - h
    return np.correlate(h, one_minus_h, mode='full')

    dx = 2.0 / n
    corr = compute_corr_components(h)
    return np.max(corr) * dx

def analytical_gradient(h, alpha):

    n = len(h)
    corr = compute_corr_components(h)

    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)

        w = weights[k]

    m = corr.max()
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        d = k - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                grad[d:] += wk * (1.0 - h[:L])
                grad[:L] -= wk * h[d:]
        else:
            e2 = -d
            L = n - e2
            if L > 0:
                grad[:L] += wk * (1.0 - h[e2:])
                grad[e2:] -= wk * h[:L]
    return grad

def project(h, n):
    t = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        cnt = int(m.sum())
        if cnt == 0:
            break
        h[m] = np.clip(h[m] + d / cnt, 0.0, 1.0)
    return h

def optimize(h0, deadline, alpha0=15.0, alpha_rate=1.015):
    n = len(h0)
    h = project(h0.copy(), n)
    best = h.copy()
    best_val = c5_value(h)
    alpha = alpha0
    step = 0.08
    momentum = 0.0
    while time.time() < deadline:
        g = gradient_c5(h, alpha)
        momentum = 0.85 * momentum + 0.15 * g
        h_new = project(h - step * momentum, n)
        val = c5_value(h_new)
        if val < best_val:
            best = h_new.copy()
            best_val = val
        h = h_new
        if val <= best_val + 1e-10:
            step = min(step * 1.04, 0.8)
        else:
            step *= 0.72
        alpha = min(alpha * alpha_rate, 800.0)
        if step < 1e-11:
            break
    return best, best_val

def make_starts(n, rng, n_starts):
    starts = []
    starts.append(np.full(n, 0.5))
    starts.append(rng.uniform(0.0, 1.0, n))
    x = np.linspace(0, 2, n, endpoint=False)
    for f in [1, 2, 3, 4, 6]:
        starts.append(0.5 + 0.35 * np.cos(np.pi * f * x + rng.uniform(0, 2*np.pi)))
    starts.append(rng.uniform(0.1, 0.9, n))
    binary = (rng.rand(n) < 0.5).astype(float)
    starts.append(binary)
    starts = [project(s, n) for s in starts[:n_starts]]
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = float('inf')
    best_n = 0

    # Phase 1: optimize initial_h_values if available
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project(h0, n0)
        dl = t0 + budget_s * 0.30
        h, v = optimize(h0, dl, alpha0=30.0)
        if v < best_val:
            best_val, best_h, best_n = v, h, n0
    except Exception:
        pass

    # Phase 2: search over n values and starts
    n_list = [46, 47, 48, 50, 54, 60, 64, 72, 80, 90, 96, 110, 128, 150, 200]
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 20
    per_n = max(remaining / len(n_list), 4.0)

    for n in n_list:
        if time.time() - t0 > budget_s - 20:
            break
        if n == best_n and best_h is not None:
            per_n_this = per_n * 0.4
        else:
            per_n_this = per_n
        n_starts = max(2, int(per_n_this / 6.0))
        per_start = per_n_this / n_starts
        starts = make_starts(n, rng, n_starts)

        for s in range(n_starts):
            if time.time() - t0 > budget_s - 20:
                break
            tlim = min(per_start, budget_s - (time.time() - t0) - 10)
            if tlim < 3.0:
                break
            h, v = optimize(starts[s], t0 + tlim, alpha0=10.0)
            if v < best_val:
                best_val, best_h, best_n = v, h, n

    # Phase 3: polish best with high alpha
    if best_h is not None and time.time() - t0 < budget_s - 30:
        dl = time.time() + (budget_s - (time.time() - t0) - 5)
        h, v = optimize(best_h.copy(), dl, alpha0=100.0, alpha_rate=1.03)
        if v < best_val:
            best_val, best_h = v, h

    # Fallback
    if best_h is None:
        n = 47
        best_h = np.full(n, 0.5)
        best_val = c5_value(best_h)
        best_n = n

    return best_h, best_val, best_n
