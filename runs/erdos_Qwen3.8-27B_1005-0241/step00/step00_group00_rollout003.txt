import numpy as np

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, beta_smooth):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    n_corr = len(corr)
    
    shifted = beta_smooth * corr
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    ones = 1 - h
    for i in range(n):
        # Term 1: sum_m w[m] * (1-h[i-m]) for valid i-m in [0,n)
        # i-m in [0,n) means m in [i-n+1, i], intersect with [0, n_corr-1]
        lo = max(0, i - (n-1))
        hi = min(n_corr - 1, i)
        if lo <= hi:
            js = i - np.arange(lo, hi+1)
            grad[i] += np.dot(w[lo:hi+1], ones[js])
        
        # Term 2: -sum_m w[m] * h[i+m] for valid i+m in [0,n)
        # i+m in [0,n) means m in [-i, n-1-i], intersect with [0, n_corr-1]
        lo = max(0, -i)
        hi = min(n_corr - 1, n - 1 - i)
        if lo <= hi:
            js = i + np.arange(lo, hi+1)
            grad[i] += -np.dot(w[lo:hi+1], h[js])
    
    grad *= dx
    return grad
