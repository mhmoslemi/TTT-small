def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = beta * corr * dx
    max_v = np.max(vals)
    exp_vals = np.exp(vals - max_v)
    weights = exp_vals / np.sum(exp_vals)
    obj = max_v / beta + np.log(np.sum(exp_vals)) / beta

    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        for j in range(n):
            term = 0.0
            jm = j - k
            jp = j + k
            if 0 <= jm < n:
                term += (1.0 - h[jm])
            if 0 <= jp < n:
                term -= h[jp]
            if term != 0.0:
                grad[j] += w * dx * term
