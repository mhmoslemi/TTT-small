import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_single_lag(h, L):
    """Gradient of f(L) where L is the absolute lag. Returns (g_plus, g_minus) for positive/negative lag."""
    n = len(h)
    if L == 0:
        g = np.zeros(n)
        # f(0) = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # df/dh[i] = 1 - 2*h[i]
        g[:] = 1.0 - 2.0 * h
        return g
    if L < n:
        # For positive lag L: f = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
        g = np.zeros(n)
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
        return g
    return np.zeros(len(h))

def full_gradient_soft(h, beta=20.0):
    """Compute soft-max weighted gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    f_max = np.max(vals)
    weights = np.exp(beta * (vals - f_max))
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for k_pos in range(len(weights)):
        w = weights[k_pos]
        if w < 1e-8:
            continue
        L = k_pos - (n - 1)
        g += w * gradient_single_lag(h, abs(L)) * dx
    return g, f_max

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_single_lag(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    if L < n:
        g = np.zeros(n)
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
        return g
    return np.zeros(n)

def soft_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    fmax = np.max(vals)
    w = np.exp(beta * (vals - fmax))
    w /= np.sum(w)
    g = np.zeros(n)

        if w[kp] < 1e-10:
            continue
        L = kp - (n - 1)
        g += w[kp] * grad_single_lag(h, abs(L))
    return g * dx, fmax
