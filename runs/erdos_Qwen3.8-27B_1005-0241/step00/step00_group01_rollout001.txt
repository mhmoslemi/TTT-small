            i, j = np.random.randint(n, 2)
            if i == j:
                continue
            oi, oj = h[i], h[j]
            h[i], h[j] = oj, oi
            new_c5 = eval_c5(h, dx)
            if new_c5 < best_c5:
                best_h, best_c5 = h.copy(), new_c5
            elif np.random.rand() >= np.exp(-(new_c5 - best_c5) / max(T, 1e-15)):
                h[i], h[j] = oi, oj
        else:
            # Pair adjustment
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo >= hi:
                continue
            delta = np.random.uniform(lo, hi)
            oi, oj = h[i], h[j]
            h[i], h[j] = oi + delta, oj - delta
            new_c5 = eval_c5(h, dx)
            if new_c5 < best_c5:
                best_h, best_c5 = h.copy(), new_c5
            elif np.random.rand() >= np.exp(-(new_c5 - best_c5) / max(T, 1e-15)):
                h[i], h[j] = oi, oj
        T = max(T_min, T * cool_rate)
    return best_h, best_c5

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    candidates = []

    # Approach 1: Optimize from initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None:
        n = len(h0)
        dx = 2.0 / n
        h = np.clip(h0, 0.0, 1.0)
        h = fix_sum(h, n / 2.0)
        sa_time = min(budget_s * 0.35, 250.0)
        h, c5 = simulated_annealing(h, dx, n, 0.004, 1e-9, 0.9997, sa_time, start_time)
        remaining = budget_s * 0.55 - (time.time() - start_time)
        if remaining > 5:
            h, c5 = local_search(h, dx, n, 150, remaining, start_time)
        candidates.append((h, c5, n))

    # Approach 2: Random binary with even n
    for n_try in [46, 94]:
        if time.time() - start_time > budget_s * 0.65:
            break
        dx = 2.0 / n_try
        for s in [seed, seed + 7, seed + 42, seed + 99]:
            if time.time() - start_time > budget_s * 0.8:
                break
            h = make_random_binary(n_try, s)
            rem = min(budget_s * 0.08, 40.0)
            h, c5 = local_search(h, dx, n_try, 40, rem, start_time)
            candidates.append((h, c5, n_try))

    # Approach 3: Alternating pattern
    if time.time() - start_time < budget_s * 0.85:
        for n_try in [46, 94]:
            dx = 2.0 / n_try
            for off in [0, 1]:
                h = np.zeros(n_try)
                h[off::2] = 1.0
                if abs(h.sum() - n_try / 2.0) > 1e-10:
                    continue
                rem = min(budget_s * 0.05, 25.0)
                h, c5 = local_search(h, dx, n_try, 30, rem, start_time)
                candidates.append((h, c5, n_try))

    # Pick best candidate
    best_h, best_c5, best_n = min(candidates, key=lambda x: x[1])

    # Final cleanup and verification
    best_h = np.clip(best_h, 0.0, 1.0)
    # Re-fix sum after clipping
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        best_h = fix_sum(best_h, best_n / 2.0)
    final_c5 = eval_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
