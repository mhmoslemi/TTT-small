import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lse(h, beta):
    n, dx = len(h), 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        L = k - n + 1
        il = max(0, -L)
        ih = min(n - 1, n - 1 - L)
        if il > ih:
            continue
        i = np.arange(il, ih + 1)
        g[i + L] += w[k] * (1.0 - h[i])
        g[i] -= w[k] * h[i + L]
    return g * dx

def pgd(h0, target, deadline, betas=None):
    if betas is None:
        betas = [2, 5, 10, 20, 50, 100, 200, 500, 1000]
    h = h0.copy()
    best = h.copy()
    bv = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        lr = 0.8 / (1.0 + beta)
        stall = 0
        for it in range(5000):
            if time.time() > deadline:
                break
            g = grad_lse(h, beta)
            h = project(h - lr * g, target)
            v = compute_c5(h)
            if v < bv:
                bv = v
                best = h.copy()
                stall = 0
            else:
                stall += 1
                if stall > 800:
                    lr *= 0.5
                    stall = 0
    return best, bv

    h = np.zeros(n)
    m = int(n / 2)
    frac = (n / 2.0) - m
    positions = rng.permutation(n)
    h[positions[:m]] = 1.0
    if frac > 0:
        h[positions[m]] = frac
    return h
