def round_and_adjust(h, n):
    """Try to round h values to 0 or 1, adjusting free coordinates to maintain sum."""
    target_sum = n / 2.0
    h_rounded = (h > 0.5).astype(float)
    # This gives sum = number of 1s, which might not be 23.5
    # Need to adjust
    current_sum = np.sum(h_rounded)
    diff = target_sum - current_sum  # e.g., if 24 ones, diff = -0.5
    
    # Find coordinates to adjust
    # If diff > 0: need to increase sum by diff → find a 0 to set to diff
    # If diff < 0: need to decrease sum by |diff| → find a 1 to set to 1+diff
    if abs(diff) < 1e-10:

        zeros = np.where(h_rounded == 0)[0]
        if len(zeros) > 0:

    else:

        ones = np.where(h_rounded == 1)[0]
        if len(ones) > 0:
