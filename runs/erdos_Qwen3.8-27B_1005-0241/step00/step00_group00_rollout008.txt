import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def binarize(h, target_sum):
    n = len(h)
    k = int(np.floor(target_sum))
    frac = target_sum - k
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:k]] = 1.0
    if frac > 1e-9:
        h_bin[sorted_idx[k]] = frac
    return h_bin

def greedy_construct(n, seed=42):
    """Greedy construction: add ones one at a time"""
    rng = np.random.RandomState(seed)
    k = n // 2
    h = np.zeros(n)
    for _ in range(k):
        best_pos = -1
        best_c5 = np.inf
        for pos in range(n):
            if h[pos] == 0:
                h[pos] = 1
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_pos = pos
                h[pos] = 0
        h[best_pos] = 1
    return h

def local_search(h, target_sum, max_iter=5000, rng=None):
    """Local search with swaps"""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for _ in range(max_iter):
        improved = False
        # Get indices of 1s and 0s
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break

        pairs = [(i, j) for i in ones for j in zeros]

            best[j] = 1

def simulated_annealing(h, target_sum, rng, start_temp=1.0, n_iter=50000):
    """Simulated annealing on binary solutions"""
    n = len(h)
    current = h.copy()
    current_c5 = compute_c5(current)
    best = current.copy()
    best_c5 = current_c5
    
    ones = np.where(current > 0.5)[0]
    zeros = np.where(current < 0.5)[0]
    
    for it in range(n_iter):
        temp = start_temp * (1 - it / n_iter)
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        current[i] = 0
        current[j] = 1
        new_c5 = compute_c5(current)
        
        if new_c5 < current_c5 or rng.rand() < np.exp(-(new_c5 - current_c5) / max(temp, 1e-10)):
            current_c5 = new_c5
            ones = np.where(current > 0.5)[0]

                best_c5 = new_c5

def gradient_descent(h, target_sum, budget_s=300, rng=None):
    """Gradient descent with LSE smoothing"""
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    beta = 20.0
    lr = 0.005
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    
    while budget_s > 5:
        t += 1
        # Compute LSE gradient (numerically for simplicity)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
        max_corr = np.max(corr)
        weights = np.exp(corr - max_corr)
        weights /= np.sum(weights)
        
        # Gradient: for each j, compute d/dh[j] of weighted sum of correlations
        # Use finite differences
        grad = np.zeros(n)
        eps = 1e-5
        for j in range(n):
            h_p = h.copy()
            h_p[j] = min(1.0, h[j] + eps)
            h_p = project_h(h_p, target_sum)
            c_p = np.correlate(h_p, 1 - h_p, mode="full") * (2.0 / n)
            # LSE value
            fp = np.max(c_p) + np.log(np.sum(np.exp(c_p - np.max(c_p))))
            
            h_m = h.copy()
            h_m[j] = max(0.0, h[j] - eps)
            h_m = project_h(h_m, target_sum)
            c_m = np.correlate(h_m, 1 - h_m, mode="full") * (2.0 / n)
            fm = np.max(c_m) + np.log(np.sum(np.exp(c_m - np.max(c_m))))
            
            grad[j] = (fp - fm) / (2 * eps)
        
        # Adam update
        beta1, beta2 = 0.8, 0.999
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h = project_h(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve % 200 == 0 and no_improve > 0:
            lr *= 0.7
            beta *= 1.5
            if lr < 1e-7:
                break
    
    return best, best_c5

def compute_c5(h):

def local_search(h, max_iter=5000, time_limit=60.0):
    """Efficient local search for binary solutions"""
    n = len(h)
    best = h.copy().astype(float)
    corr = np.correlate(best, 1.0 - best, mode="full").astype(float)
    best_max = float(np.max(corr))
    
    d = np.arange(-(n-1), n)  # lag values
    ones = list(np.where(best > 0.5)[0])
    zeros = list(np.where(best < 0.5)[0])
    
    start = time.time()
    
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in ones:
            for j in zeros:
                # Compute delta vector (vectorized)
                # i: 1->0, j: 0->1
                dm = np.zeros(2*n-1)
                # i as first factor (was 1, now 0): contribution was 1*(1-h[i+d]), now 0
                # Change: -(1 - h[i+d]) = h[i+d] - 1, for valid i+d
                m1 = (i + d >= 0) & (i + d < n)
                dm[m1] += best[i + d[m1]] - 1.0
                # i as second factor (was 1, now 0): contribution was h[i-d]*0, now h[i-d]*1
                # Change: +h[i-d], for valid i-d
                m2 = (i - d >= 0) & (i - d < n)
                dm[m2] += best[i - d[m2]]
                # j as first factor (was 0, now 1): contribution was 0, now 1*(1-h[j+d])
                # Change: +(1 - h[j+d]), for valid j+d
                m3 = (j + d >= 0) & (j + d < n)
                dm[m3] += 1.0 - best[j + d[m3]]
                # j as second factor (was 0, now 1): contribution was h[j-d]*1, now h[j-d]*0
                # Change: -h[j-d], for valid j-d
                m4 = (j - d >= 0) & (j - d < n)
                dm[m4] += -best[j - d[m4]]
                
                new_corr = corr + dm
                new_max = float(np.max(new_corr))
                
                if new_max < best_max - 1e-12:
                    best[i] = 0.0
                    best[j] = 1.0
                    corr = new_corr
                    best_max = new_max
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return best, best_max * (2.0 / n)
