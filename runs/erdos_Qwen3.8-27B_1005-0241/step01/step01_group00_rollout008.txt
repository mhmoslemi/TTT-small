import time

def get_corr(h):
    return np.correlate(h, 1-h, mode="full")

def eval_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * 2.0/n

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

    grad = np.zeros(n)
    for k in range(len(p)):
        w = p[k]
        if w < 1e-10:
            continue
        d = k - (n-1)

        hi1 = min(n, n+d)
        if lo1 < hi1:
            grad[lo1:hi1] += w*(1.0-h[lo1-d:hi1-d])
        lo2 = max(0, -d)
        hi2 = min(n, n-d)
        if lo2 < hi2:

    return grad

    c_max = corr.max()
    return c_max + T*np.log(np.exp((corr - c_max)/T).sum())

def grad_soft_max(corr, T):
    """Returns weights for gradient of soft-max"""
    exp_vals = np.exp((corr - corr.max())/T)
    return exp_vals / exp_vals.sum()

def optimize_softmax(h, target, time_limit, T_start=0.05, T_end=5.0, n_phases=10):

    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)

Let me also add one more idea: **coordinate descent**. Instead of updating all h[i] simultaneously, update one coordinate at a time. For each coordinate h[i], the optimal value (given all other coordinates fixed) can be found by 1D optimization.

For a given h[i] = x, the objective is:
C₅(x) = max_d f_d(x)

where f_d(x) is linear in x (since only terms involving h[i] change, and they're linear in h[i]).

So C₅(x) is the max of linear functions = convex in x. The minimum over x ∈ [0,1] can be found by checking the intersection points of the linear functions, or simply by evaluating at x=0, x=1, and the points where two of the active linear functions intersect.

This is a very efficient approach! For each coordinate, we can find the optimal value in O(n) time (just evaluate the correlation at a few candidate values).

Let me refine the coordinate descent:

Instead of ternary search (which requires 2 evaluations per step), I can use the fact that C₅(x) is piecewise linear (max of linear functions). The minimum of a piecewise linear convex function occurs at a breakpoint (where two linear functions intersect) or at a boundary (x=0 or x=1).

The breakpoints occur where two shifts have equal correlation values. For each pair of shifts (d1, d2), the breakpoint is where f_{d1}(x) = f_{d2}(x), which is:
a1*x + b1 = a2*x + b2
x = (b2 - b1) / (a1 - a2) (if a1 ≠ a2)

where a_d and b_d are the slope and intercept of f_d as a function of x.

I can compute all breakpoints, filter for those in [0,1], evaluate C₅ at each, plus at x=0 and x=1, and take the minimum.

The number of breakpoints is O(n²) (one for each pair of shifts), but many will be outside [0,1]. In practice, there might be O(n) relevant breakpoints.

This gives an exact solution for each coordinate in O(n²) time, which is much faster than ternary search.

import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1-h, mode='full')

def eval_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * 2.0 / n

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    else:
        m = h > 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    return np.clip(h, 0.0, 1.0)

def compute_grad_weighted(h, p, n):
    grad = np.zeros(n)
    for k in range(len(p)):
        w = p[k]
        if w < 1e-10:
            continue
        d = k - (n - 1)
        lo1 = max(0, d)
        hi1 = min(n, n + d)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-d:hi1-d])
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2+d:hi2+d])
    return grad

def coord_descent(h, target, time_limit):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best_c5 = eval_c5(h)
    best_h = h.copy()
    grid = np.linspace(0, 1, 21)
    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start >= time_limit:
                return best_h, best_c5
            old = h[i]
            best_x = old
            best_v = eval_c5(h)
            for x in grid:
                h[i] = x
                v = eval_c5(h)
                if v < best_v:
                    best_v = v
                    best_x = x
            h[i] = best_x
            if best_x != old:
                improved = True
        h_proj = project(h, target)
        c5 = eval_c5(h_proj)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_proj.copy()
            h = h_proj
        elif not improved:
            break
    return best_h, best_c5

def grad_optimize(h, target, time_limit, T=0.1, alpha_init=0.03):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    alpha = alpha_init
    no_improve = 0
    while time.time() - start < time_limit:
        corr = get_corr(h)
        w = np.exp((corr - corr.max()) / T)
        w /= w.sum()
        grad = compute_grad_weighted(h, w, n)
        step = alpha
        found = False
        for _ in range(20):
            h_new = project(h - step * grad, target)
            c5 = eval_c5(h_new)
            if c5 < eval_c5(h) - 1e-13:
                h = h_new
                if c5 < best_c5 - 1e-10:
                    best_h = h.copy()
                    best_c5 = c5
                no_improve = 0
                found = True
                break
            step *= 0.5
        if not found:
            no_improve += 1
            alpha = max(alpha * 0.8, 1e-6)
            if no_improve >= 40:
                break
    return best_h, best_c5

def swap_search(h, target, time_limit):
    n = len(h)
    start = time.time()
    h = h.copy()
    best_c5 = eval_c5(h)
    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - start >= time_limit:
                    return h, best_c5
                if abs(h[i] - h[j]) < 1e-8:
                    continue
                h[i], h[j] = h[j], h[i]
                c5 = eval_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, best_c5

def transfer_search(h, target, time_limit):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best_c5 = eval_c5(h)
    best_h = h.copy()
    deltas = np.array([0.05, 0.1, 0.15, 0.2, 0.3])
    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if time.time() - start >= time_limit:
                    return best_h, best_c5
                for d in deltas:
                    if h[i] < d or h[j] > 1.0 - d:
                        continue
                    h[i] -= d
                    h[j] += d
                    c5 = eval_c5(h)
                    if c5 < best_c5 - 1e-12:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[j] -= d
                        h[i] += d
        if not improved:
            break
    return best_h, best_c5

def binarize(h, target):
    h = h.copy()
    h[h < 0.25] = 0.0
    h[h > 0.75] = 1.0
    return project(h, target)

def make_initial(n, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.0
    h = project(h, n / 2.0)
    return h

def optimize_one(h0, target, time_limit, rng):

    best_c5 = eval_c5(best_h)
    start = time.time()

    def rem():

    for phase in range(30):
        if rem() < 10:
            break
        # Coordinate descent
        t = min(rem() * 0.35, 90)
        if t > 5:
            ho, c5o = coord_descent(best_h, target, t)
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o
        # Gradient
        t = min(rem() * 0.25, 60)
        if t > 5:
            ho, c5o = grad_optimize(best_h, target, t, T=0.08, alpha_init=0.02)
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o
        # Binarize + coord
        if rem() > 30:
            h_bin = binarize(best_h, target)
            ho, c5o = coord_descent(h_bin, target, min(20, rem() - 5))
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o
        # Swap
        t = min(rem() * 0.15, 40)
        if t > 3:
            ho, c5o = swap_search(best_h, target, t)
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o
        # Transfer
        t = min(rem() * 0.1, 30)
        if t > 5:
            ho, c5o = transfer_search(best_h, target, t)
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o
        # Perturb + restart
        if rem() > 50:
            noise = 0.02 + 0.05 * rng.rand()
            hp = project(best_h + rng.normal(0, noise, len(best_h)), target)
            ho, c5o = coord_descent(hp, target, min(30, rem() - 10))
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o
            ho, c5o = swap_search(best_h, target, min(15, rem() - 5))
            if c5o < best_c5:
                best_h, best_c5 = ho, c5o

    # Final coordinate polish
    if rem() > 5:
        ho, c5o = coord_descent(best_h, target, min(rem() - 2, 30))
        if c5o < best_c5:
            best_h, best_c5 = ho, c5o
    return best_h, best_c5
