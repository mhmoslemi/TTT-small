import numpy as np
import time

def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * 2.0 / n

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
        if mask.sum() > 0:
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0, 1)
    else:
        mask = h > 1e-10
        if mask.sum() > 0:
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0, 1)
    return h

def grad_smooth_max(h, T, n):
    """Compute gradient of smooth max of correlations"""
    corr = np.correlate(h, 1 - h, mode="full")
    p = np.exp((corr - corr.max()) / T)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-12:
            continue
        d = k - (n - 1)
        grad += p[k] * grad_for_shift(h, d, n)
    return grad, corr.max() * 2.0 / n

def grad_for_shift(h, d, n):
    grad = np.zeros(n)
    if d == 0:
        return 1 - 2 * h
    elif d > 0:
        grad[:n-d] += (1 - h[d:])
        grad[d:] += -h[:-d]
    else:
        d2 = -d
        grad[d2:] += (1 - h[:-d2])
        grad[:n-d2] += -h[d2:]
    return grad

def adam_step(h, grad, target, m, v, t, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad**2
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = project_sum(h_new, target)
    return h_new, m, v

def optimize_adam(h_init, target, time_limit, lr=0.01, T_init=1.0, T_end=0.1):
    n = len(h_init)
    start = time.time()
    h = project_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    while time.time() - start < time_limit:
        t += 1
        # Anneal temperature
        frac = min(1.0, (time.time() - start) / time_limit)
        T = T_init * (T_end / T_init) ** frac
        
        grad, c5 = grad_smooth_max(h, T, n)
        h_new, m, v = adam_step(h, grad, target, m, v, t, lr=lr)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)
    else:
        idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)
    return h

def gs(h, d, n):
    g = np.zeros(n)
    if d == 0:
        return 1.0 - 2.0 * h
    if d > 0:
        g[:n-d] += 1.0 - h[d:]
        g[d:] += -h[:-d]
    else:
        d2 = -d
        g[d2:] += 1.0 - h[:-d2]
        g[:n-d2] += -h[d2:]
    return g

def smooth_grad(h, T, n, topk):
    corr = np.correlate(h, 1 - h, mode="full")
    if topk is not None and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
        c_use = corr[idx]
        p = np.exp((c_use - c_use.max()) / T)
        p /= p.sum()
        g = np.zeros(n)
        for i in range(topk):
            d = idx[i] - (n - 1)
            g += p[i] * gs(h, d, n)
        return g
    p = np.exp((corr - corr.max()) / T)
    p /= p.sum()
    g = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-10:
            continue
        d = k - (n - 1)
        g += p[k] * gs(h, d, n)
    return g

def optimize_mom(h0, target, tlim, lr, T0, T1, mom, topk, kick_interval=50, kick_scale=0.02):
    n = len(h0)
    t0 = time.time()
    h = project_sum(h0.copy(), target)

    stale = 0

        elapsed = time.time() - t0
        frac = min(1.0, elapsed / max(tlim, 1e-6))
        T = T0 * (T1 / T0) ** frac
        g = smooth_grad(h, T, n, topk)
        v = mom * v - lr_cur * g

        c5_cur = eval_c5(h)

            h_best = h.copy()

                stale = 0
            elif stale % 40 == 0:

                v *= 0.3
                if lr_cur < 1e-6:

    return h_best, c5_best
