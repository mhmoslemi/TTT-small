            h[i] += adj
            diff -= adj
        else:

            h[i] -= adj

    best_h = h.copy()

        return time.time() - start
    
    def optimize_coord(h, i, corr):
        """Find optimal h[i] given others fixed. Returns new value."""
        n = len(h)
        # Compute a and b for each lag (excluding k=0 which is special)
        a = np.zeros(2*n - 1)
        b = np.zeros(2*n - 1)
        m0 = n - 1  # index of k=0 in full correlation

            if m == m0:
                continue
            k = m - (n - 1)
            ik = i - k
            ip = i + k
            dk = 1.0 if (0 <= ik < n) else 0.0
            dp = 1.0 if (0 <= ip < n) else 0.0
            a[m] = dk * (1.0 - h[ik]) - dp * h[ip]
            b[m] = corr[m] - h[i] * (1.0 - h[ik]) * dk + h[i] * h[ip] * dp

        c0_base = corr[m0] + hi * hi  # c0(t) = c0_base - t^2 = corr[m0] + hi^2 - t^2

            vals = b + t * a
            vals[m0] = c0_base - t * t
            return np.max(vals)

        return res.x

            break

            M = c.max()
            return M + (1.0/alpha) * np.log(np.sum(np.exp((c - M) * alpha)))
        
        def smooth_grad(x, alpha):

            M = c.max()
            p = np.exp((c - M) * alpha)

    grad = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k == 0:
            grad[m] = 1.0 - 2.0*h
        else:
            # (1-h[i-k]) for valid i-k
            ik = np.arange(n) - k

            g2 = np.zeros(n)
            g2[valid_p] = -h[ip[valid_p]]
            grad[m] = g1 + g2

Let me think about what the optimal structure might look like. The Erdős problem is about finding a function that minimizes the maximum overlap at any shift. Intuitively, we want h to be "spread out" so that at any shift, the overlap between h and 1-h is minimized.

For a binary function (h ∈ {0,1}), the overlap at shift k counts the number of positions where h=1 and h(x+k)=0. This is like counting "1→0" transitions at distance k.

For a balanced binary sequence (n/2 ones, n/2 zeros), the minimum possible maximum overlap is related to the "discrepancy" of the sequence.

Actually, let me think about this from a different angle. The sum of all overlap values (over all shifts k) is:

    best_h, best_c5 = coord_descent(h, max_sweeps=300, tlimit=budget_s - 300)
    
    # Phase 2: Binary local search
    if elapsed() < budget_s - 150:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_sum(h_bin)
        if abs(h_bin.sum() - target_sum) < 1e-10:
            c5_b = eval_c5(h_bin)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_bin.copy()
            
            # Greedy swap search
            for it in range(100):
                if elapsed() > budget_s - 30:
                    break
                improved = False
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                best_sw_c5 = c5_b
                best_oi, best_zi = -1, -1
                for oi in ones:
                    for zi in zeros:
                        h_bin[oi] = 0
                        h_bin[zi] = 1
                        c5_try = float(np.max(get_corr(h_bin)) * dx)
                        h_bin[oi] = 1
                        h_bin[zi] = 0
                        if c5_try < best_sw_c5 - 1e-12:
                            best_sw_c5 = c5_try
                            best_oi, best_zi = oi, zi
                if best_oi >= 0:
                    h_bin[best_oi] = 0
                    h_bin[best_zi] = 1
                    c5_b = best_sw_c5
                    best_c5 = c5_b
                    best_h = h_bin.copy()
                    improved = True
                if not improved:
                    break
    
    # Phase 3: Multi-start coordinate descent
    if elapsed() < budget_s - 60:
        for Rs in range(30):
            if elapsed() > budget_s - 30:
                break
            h_try = best_h + np.random.normal(0, 0.12, n)
            h_try = fix_sum(h_try)
            if abs(h_try.sum() - target_sum) < 1e-8:
                h_r, c5_r = coord_descent(h_try, max_sweeps=80, tlimit=budget_s - 30)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r
    
    # Phase 4: Binary local search from improved continuous solution
    if elapsed() < budget_s - 30:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = fix_sum(h_bin2)
        if abs(h_bin2.sum() - target_sum) < 1e-10:
            c5_b2 = eval_c5(h_bin2)
            for it in range(80):
                if elapsed() > budget_s - 10:
                    break
                improved = False
                ones = np.where(h_bin2 == 1)[0]
                zeros = np.where(h_bin2 == 0)[0]
                best_sw_c5 = c5_b2
                best_oi, best_zi = -1, -1
                for oi in ones:
                    for zi in zeros:
                        h_bin2[oi] = 0
                        h_bin2[zi] = 1
                        c5_try = float(np.max(get_corr(h_bin2)) * dx)
                        h_bin2[oi] = 1
                        h_bin2[zi] = 0
                        if c5_try < best_sw_c5 - 1e-12:
                            best_sw_c5 = c5_try
                            best_oi, best_zi = oi, zi
                if best_oi >= 0:
                    h_bin2[best_oi] = 0
                    h_bin2[best_zi] = 1
                    c5_b2 = best_sw_c5
                    if c5_b2 < best_c5:
                        best_c5 = c5_b2
                        best_h = h_bin2.copy()
                    improved = True
                if not improved:
                    break
    
    # Phase 5: Final polish with coordinate descent
    if elapsed() < budget_s - 10:
        h_final, c5_final = coord_descent(best_h, max_sweeps=200, tlimit=budget_s - 5)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    d = target_sum - best_h.sum()
    if abs(d) > 1e-10:
        for i in range(n):
            if abs(d) < 1e-12:
                break
            if d > 0:
                adj = min(d, 1.0 - best_h[i])
                best_h[i] += adj
                d -= adj
            else:
                adj = min(-d, best_h[i])
                best_h[i] -= adj
                d += adj
    
    final_c5 = float(np.max(get_corr(best_h)) * dx)
    return best_h, final_c5, n
