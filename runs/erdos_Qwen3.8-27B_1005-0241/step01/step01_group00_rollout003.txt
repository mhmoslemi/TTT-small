    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-12

            h[m] += diff / m.sum()

        m = h > 1e-12

def update_best(best_h, best_c5, h_cand):
    n = len(h_cand)
    h_cand = project(h_cand.copy(), n / 2.0)
    c5 = eval_c5(h_cand)
    if c5 < best_c5:
        return h_cand, c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - t_start

    def rem():
        return max(0.0, budget_s - elapsed() - 5.0)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    # Phase 1: Coordinate descent on initial values (most important)
    if h_init is not None:
        n = len(h_init)
        target = n / 2.0
        h_init = project(h_init.copy(), target)
        best_h, best_c5 = update_best(best_h, best_c5, h_init)
        t1 = rem() * 0.35
        if t1 > 5:
            h_opt, c5_opt = coord_descent(h_init, target, t1)
            best_h, best_c5 = update_best(best_h, best_c5, h_opt)

    # Phase 2: Try multiple n values with different initializations
    n_list = [49, 51, 47, 53, 45, 43, 55, 41]
    if h_init is not None:
        n_list = [x for x in n_list if x != len(h_init)]
        n_list.insert(0, len(h_init))  # retry with more time

    for n in n_list:
        if rem() < 30:
            break
        target = n / 2.0
        t_each = max(min(rem() * 0.06, 50.0), 10.0)
        for att in range(3):
            if rem() < 10:
                break
            if att == 0:
                h_r = make_binary_init(n, rng)
            elif att == 1:
                h_r = make_smooth_init(n, rng)
            else:
                h_r = make_fourier_init(n, rng, target)
            best_h, best_c5 = update_best(best_h, best_c5, h_r)
            h_opt, c5_opt = coord_descent(h_r, target, t_each)
            best_h, best_c5 = update_best(best_h, best_c5, h_opt)

    # Phase 3: Pair descent on best solution
    if best_h is not None and rem() > 20:
        n = len(best_h)
        target = n / 2.0
        t_pair = min(rem() * 0.2, 60.0)
        h_opt, c5_opt = pair_descent(best_h, target, t_pair)
        best_h, best_c5 = update_best(best_h, best_c5, h_opt)

    # Phase 4: Perturbation + coordinate descent restarts
    if best_h is not None:
        n = len(best_h)
        target = n / 2.0
        remaining = rem()
        n_restart = max(1, int(remaining / 15))
        t_each = max(remaining / max(n_restart, 1) * 0.85, 5.0)
        for _ in range(n_restart):
            if rem() < 5:
                break
            ns = 0.02 + 0.06 * rng.rand()
            h_pert = project(best_h + rng.normal(0, ns, n), target)
            h_opt, c5_opt = coord_descent(h_pert, target, t_each)
            best_h, best_c5 = update_best(best_h, best_c5, h_opt)

    # Phase 5: Final smooth polish + coordinate descent
    if best_h is not None and rem() > 15:
        n = len(best_h)
        target = n / 2.0
        t_s = min(rem() * 0.4, 40.0)
        h_opt, c5_opt = smooth_optimize(best_h, target, t_s, T=0.3, alpha_init=0.008)
        best_h, best_c5 = update_best(best_h, best_c5, h_opt)
        if rem() > 10:
            t_cd = min(rem() - 5, 30.0)
            h_opt, c5_opt = coord_descent(h_opt, target, t_cd)
            best_h, best_c5 = update_best(best_h, best_c5, h_opt)

    # Final: coordinate descent polish
    if best_h is not None and rem() > 8:
        n = len(best_h)
        target = n / 2.0
        h_opt, c5_opt = coord_descent(best_h, target, rem() - 3)
        best_h, best_c5 = update_best(best_h, best_c5, h_opt)

    if best_h is None:
        n = 47
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if i % 2 == 0 else 0.0
        h[0] = 0.5
        best_h = project(h, n / 2.0)
        best_c5 = eval_c5(best_h)

    n = len(best_h)
    best_h = project(best_h, n / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, n
