            h[k::block] = 1.0
    return project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - start) - 5.0)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def update(bh, bc5, bn):
        nonlocal best_h, best_c5, best_n
        if bc5 < best_c5:
            best_h, best_c5, best_n = bh, bc5, bn

    # Phase 1: Coordinate descent on initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        n = len(h_init)
        target = n / 2.0
        h = project(h_init.copy(), target)
        t1 = min(rem() * 0.35, 350)
        h_opt, c5_opt = coord_descent(h, target, n, t1, rng)
        update(h_opt, c5_opt, n)
        # Pair refinement
        if rem() > 30:
            t2 = min(rem() * 0.1, 100)
            h_opt2, c5_opt2 = coord_descent(h_opt, target, n, t2, rng, use_pairs=True)
            update(h_opt2, c5_opt2, n)

    # Phase 2: Try different n values
    n_candidates = [46, 48, 50, 52, 56, 64]
    if h_init is not None:
        n_candidates = [x for x in n_candidates if x != len(h_init)]

    for n in n_candidates:
        if rem() < 40:
            break
        target = n / 2.0
        n_tries = 3
        for s in range(n_tries):
            if rem() < 15:
                break
            if s == 0:
                h = make_binary_start(n, rng)
            elif s == 1:
                h = make_block_start(n, rng)
            else:
                h = project(rng.uniform(0.2, 0.8, n), target)
            t_lim = min(rem() * 0.08, 80)
            h_opt, c5_opt = coord_descent(h, target, n, t_lim, rng)
            update(h_opt, c5_opt, n)
            if c5_opt < best_c5 + 0.001 and rem() > 20:
                t_lim2 = min(rem() * 0.05, 50)
                h_opt2, c5_opt2 = coord_descent(h_opt, target, n, t_lim2, rng, use_pairs=True)
                update(h_opt2, c5_opt2, n)

    # Phase 3: More coordinate descent on best
    if best_h is not None and rem() > 30:
        n = best_n
        target = n / 2.0
        t3 = min(rem() * 0.3, 200)
        h_opt, c5_opt = coord_descent(best_h, target, n, t3, rng)
        update(h_opt, c5_opt, n)

    # Phase 4: Random restarts from best
    if best_h is not None and rem() > 20:
        n = best_n
        target = n / 2.0
        n_restarts = int(rem() // 15)
        for r in range(min(n_restarts, 30)):
            if rem() < 8:
                break
            scale = 0.03 + 0.04 * rng.rand()
            h_pert = project(best_h + rng.normal(0, scale, n), target)
            t_r = min(rem() * 0.08, 40)
            h_opt, c5_opt = coord_descent(h_pert, target, n, t_r, rng)
            update(h_opt, c5_opt, n)

    # Final pair polish
    if best_h is not None and rem() > 15:
        n = best_n
        target = n / 2.0
        h_opt, c5_opt = coord_descent(best_h, target, n, min(rem() - 3, 60), rng, use_pairs=True)
        update(h_opt, c5_opt, n)

    return best_h, best_c5, best_n
