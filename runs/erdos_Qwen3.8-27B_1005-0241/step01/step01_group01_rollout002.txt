```python
    """Project v onto {h ∈ [0,1]^n : sum(h) = target_sum}"""
    n = len(v)
    # Binary search for lambda such that sum(clamp(v - s + lambda)) = target_sum
    lo, hi = v.min() - target_sum - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v + mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
```

```python
def best_delta(corr, d, delta_min, delta_max, dx):
    """Find delta in [delta_min, delta_max] that minimizes max(corr + delta*d) * dx."""
    if delta_max <= delta_min:
        return delta_min
    # The function f(delta) = max_k (corr[k] + delta*d[k]) is convex in delta.
    # Use golden section search.
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a = delta_min
    b = delta_max
    c = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = (corr + c * d).max()
    fd = (corr + d_ * d).max()
    for _ in range(60):
        if fc < fd:
            b = d_
            d_ = c
            fd = fc
            c = b - gr * (b - a)
            fc = (corr + c * d).max()
        else:
            a = c
            c = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = (corr + d_ * d).max()
    delta_opt = (a + b) / 2
    return delta_opt
```

This is clean and efficient. Each evaluation is O(n) (computing the max of a vector), and we do 60 iterations, so it's O(60n) per (i,j) pair.

For all pairs: n(n-1) × 60n ≈ 47×46×60×47 ≈ 6M operations. That's very fast!

But wait, I also need to compute the coefficient vectors coef_j and coef_m for each j. That's O(n·(2n-1)) per j, or O(n²·(2n-1)) total = O(n³) ≈ 47³ ≈ 100K. Fine.

Actually, I can precompute all coefficient vectors at once:
coef[j, k_idx] = (1-h[j+k])·1[0≤j+k<n] - h[j-k]·1[0≤j-k<n]

This is exactly the gradient of g_k with respect to h[j], summed over k with weight 1 (not weighted). So coef[j, k_idx] = ∂g_{k_idx}/∂h[j].

I can compute this as a matrix of shape (n, 2n-1).

Let me put it all together in a clean implementation:

```python
def compute_coefs_fast(h, n):

    # For each k, compute the gradient vector ∇g_k
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                coefs[:n-k, k_idx] += (1.0 - h[k:])
                coefs[k:, k_idx] -= h[:n-k]
        else:
            m = -k
            if n - m > 0:
                coefs[m:, k_idx] += (1.0 - h[:n-m])
                coefs[:n-m, k_idx] -= h[m:]
    return coefs
```

        else:

def smooth_obj_val(h, t):
    corr = get_corr(h)
    M = corr.max()
    return M + t * np.log(np.sum(np.exp((corr - M) / t)))

def smooth_obj_grad(h, t, n):
    corr = get_corr(h)
    M = corr.max()
    w = np.exp((corr - M) / t)
    w /= w.sum()
    # Weighted sum of gradients
    coefs = compute_coefs_fast(h, n)
    grad = coefs.T @ w  # shape (n,)
    return grad, corr

def c5_of(h, dx):
    return float(get_corr(h).max() * dx)

def golden_delta(corr, d, dmin, dmax):
    """Minimize max(corr + delta*d) over delta in [dmin, dmax]."""
    if dmax <= dmin:
        return dmin

            b = dd; dd = c; fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + c * d)
        else:
            a = c; c = dd; fc = fd
            dd = a + gr * (b - a)
            fd = np.max(corr + dd * d)
