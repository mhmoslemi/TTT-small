import time

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def eval_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * 2.0 / n

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    else:
        m = h > 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    return np.clip(h, 0.0, 1.0)

def coord_optimize_sweep(h, target, order=None):
    """One sweep of coordinate descent"""
    n = len(h)
    h = h.copy()
    if order is None:
        order = range(n)
    for j in order:
        h[j] = 0.0
        A = np.correlate(h, 1 - h, mode="full")
        B = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            d = k - (n - 1)
            if 0 <= j + d < n:
                B[k] += (1.0 - h[j + d])
            if 0 <= j - d < n:
                B[k] -= h[j - d]
        # Find optimal x
        # Vectorized pairwise intersections
        Bi = B[:, None]
        Aj = A[None, :]
        Ai = A[:, None]
        Bj = B[None, :]
        denom = Bi - Bj
        mask = np.abs(denom) > 1e-12
        x_cands = np.zeros(denom.shape)
        x_cands[mask] = (Aj - Ai)[mask] / denom[mask]
        valid = (x_cands > 0) & (x_cands < 1)
        cands = np.concatenate([[0.0, 1.0], x_cands[valid]])
        # Deduplicate
        cands = np.unique(cands)
        vals = A[None, :] + B[None, :] * cands[:, None]
        f = vals.max(axis=1)
        h[j] = cands[np.argmin(f)]
    return h

def coord_descent(h, target, time_limit, n_sweeps_max=1000):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)

import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def eval_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * 2.0 / n

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    else:
        m = h > 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, target, order):
    n = len(h)
    h = h.copy()
    for j in order:
        h[j] = 0.0
        A = np.correlate(h, 1.0 - h, mode="full")
        B = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            d = k - (n - 1)
            if 0 <= j + d < n:
                B[k] += (1.0 - h[j + d])
            if 0 <= j - d < n:
                B[k] -= h[j - d]
        # Find best x in [0,1]
        # Vectorized: compute all pairwise intersections
        denom = B[:, None] - B[None, :]
        num = A[None, :] - A[:, None]
        mask = np.abs(denom) > 1e-12
        x_mat = np.zeros_like(denom)
        x_mat[mask] = num[mask] / denom[mask]
        valid = (x_mat > 0) & (x_mat < 1)
        cands = np.unique(np.concatenate([[0.0, 1.0], x_mat[valid]]))
        vals = A[None, :] + B[None, :] * cands[:, None]
        h[j] = cands[np.argmin(vals.max(axis=1))]
    return h

def run_coord_descent(h, target, time_limit, rng, n_iters=500):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    for it in range(n_iters):
        if time.time() - start >= time_limit:
            break
        order = rng.permutation(n)
        h_new = coord_sweep(h, target, order)
        h_new = project(h_new, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best = h_new.copy()
            best_c5 = c5
            h = h_new
        else:
            h = h_new
            # Stagnation: try a random perturbation
            if it > 30:
                n_pert = rng.randint(2, 6)
                idx = rng.choice(n, size=n_pert, replace=False)
                h[idx] = rng.uniform(0, 1, n_pert)
                h = project(h, target)
    return best, best_c5

import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def eval_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * 2.0 / n

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    else:
        m = h > 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, target, order):
    n = len(h)
    h = h.copy()
    d_vals = np.arange(2 * n - 1) - (n - 1)
    for j in order:
        h[j] = 0.0
        A = np.correlate(h, 1.0 - h, mode="full")
        B = np.zeros(2 * n - 1)
        jd = j + d_vals
        valid1 = (jd >= 0) & (jd < n)
        B[valid1] += (1.0 - h[jd[valid1]])
        jdim = j - d_vals
        valid2 = (jdim >= 0) & (jdim < n)
        B[valid2] -= h[jdim[valid2]]
        # Find optimal x
        denom = B[:, None] - B[None, :]
        num = A[None, :] - A[:, None]
        mask = np.abs(denom) > 1e-12
        x_mat = np.zeros_like(denom)
        x_mat[mask] = num[mask] / denom[mask]
        valid = (x_mat > 0) & (x_mat < 1)
        cands = np.unique(np.concatenate([[0.0, 1.0], x_mat[valid]]))
        vals = A[None, :] + B[None, :] * cands[:, None]
        h[j] = cands[np.argmin(vals.max(axis=1))]
    return h

def run_cd(h, target, time_limit, rng, max_sweeps=2000):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    stagnant = 0
    for _ in range(max_sweeps):
        if time.time() - start >= time_limit:
            break
        order = rng.permutation(n)
        h_new = coord_sweep(h, target, order)
        h_new = project(h_new, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best = h_new.copy()
            best_c5 = c5
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 20:
                # Perturb
                k = rng.randint(1, 8)
                idx = rng.choice(n, size=k, replace=False)
                h_new[idx] = rng.uniform(0, 1, k)
                h_new = project(h_new, target)
                stagnant = 0
        h = h_new
    return best, best_c5

def smooth_refine(h, target, time_limit, T, alpha_init, rng):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    alpha = alpha_init
    no_imp = 0
    while time.time() - start < time_limit:
        corr = get_corr(h)
        mc = corr.max()
        p = np.exp((corr - mc) / T)
        p /= p.sum()
        # Gradient of weighted sum
        grad = np.zeros(n)
        for k in range(len(p)):
            w = p[k]
            if w < 1e-10:
                continue
            d = k - (n - 1)
            jd = np.arange(n) + d
            m1 = (jd >= 0) & (jd < n)
            grad[m1] += w * (1.0 - h[jd[m1]])
            jdim = np.arange(n) - d
            m2 = (jdim >= 0) & (jdim < n)
            grad[m2] += w * (-h[jdim[m2]])
        h_new = project(h - alpha * grad, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-10:
            best = h_new.copy()
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                alpha *= 0.75
                if alpha < 1e-6:
                    break
        h = h_new
    return best, best_c5

- k = 0: sum_n h[n]*(1-h[n]) = sum h[n] - sum h[n]²
- k > 0: sum h[n+k]*(1-h[n]) for valid n
- k < 0: sum h[n+k]*(1-h[n]) = sum h[m]*(1-h[m-k]) where m = n+k

Let me define d = k. Then:
For k = d ≥ 0: corr[d] = sum_{n=0}^{n-1-d} h[n+d]*(1-h[n])
For k = -d: corr[-d] = sum_{n=d}^{n-1} h[n-d]*(1-h[n])

Let me re-index: for shift d (positive):
overlap(d) = sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])

This is the number/weight of "0→1" transitions at distance d (reading left to right).

And for the problem, C₅ = max_k corr[k] * dx = max_k corr[k] * 2/n

So the corr array has indices 0 to 2n-2, corresponding to shifts -(n-1) to (n-1).

Now, for the gradient of corr[k] with respect to h[j]:
corr[k] = sum_n h[n+k] * (1-h[n]) (with the appropriate range)

Terms involving h[j]:
1. When n+k = j, i.e., n = j-k: contribution h[j] * (1-h[j-k]) if j-k is valid (0 ≤ j-k < n)
2. When n = j: contribution h[j+k] * (1-h[j]) if j+k is valid (0 ≤ j+k < n)

So:
- ∂corr[k]/∂h[j] = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)

Now for coordinate descent on h[j]:
Set h[j] = x. Then:
corr[k] = A_k + B_k * x

where:
- B_k = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)
  (with h[j] = 0, so h[j-k] and h[j+k] are the fixed values)
- A_k = corr[k] when h[j] = 0

Let me verify: when h[j] = x:
The term 1 (n = j-k): h[j] * (1-h[j-k]) = x * (1-h[j-k])
The term 2 (n = j): h[j+k] * (1-x) = h[j+k] - h[j+k]*x

So corr[k] with h[j]=x = [corr[k] without these terms] + x*(1-h[j-k]) + h[j+k] - h[j+k]*x
= [corr[k] without these terms + h[j+k]] + x*[(1-h[j-k]) - h[j+k]]

So:
A_k = corr[k] when h[j]=0 (which includes the h[j+k]*(1-0) = h[j+k] term)
B_k = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)

This matches what I had before! Good.

Now let me also verify with the problem's formulation. The problem says:
C₅ = max_k ∫ h(x)(1-h(x+k)) dx

With the discretization h[i] at x_i = i*dx, this becomes:
∫ h(x)(1-h(x+k)) dx ≈ sum_i h[i]*(1-h[i+k])*dx  (for integer shifts k)

But the evaluate function uses np.correlate(h, 1-h, mode="full") * dx.

Let me check: np.correlate(h, 1-h, mode="full") at index k gives:
sum_n h[n+k] * (1-h[n])  (as I derived above)

The problem wants: sum_i h[i] * (1-h[i+k])

These are related: sum_i h[i]*(1-h[i+k]) = sum_n h[n]*(1-h[n+k])

While the correlate gives: sum_n h[n+k]*(1-h[n])

These are different! Let me check...

sum_i h[i]*(1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]
sum_n h[n+k]*(1-h[n]) = sum_n h[n+k] - sum_n h[n+k]*h[n]

The first sum: sum_i h[i] (over valid i) vs sum_n h[n+k] (over valid n). These are different sums!

For k > 0:
- Problem: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])  [i+k must be < n]
  = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]

- Correlate at index k: sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
  = sum_{n=0}^{n-1-k} h[n+k] - sum_{n=0}^{n-1-k} h[n+k]*h[n]

So: problem_k = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]
    correlate_k = sum_{n=0}^{n-1-k} h[n+k] - sum_{n=0}^{n-1-k} h[n+k]*h[n]

The second terms are the same (just different dummy variable). The first terms are different:
- problem: sum of h[0] to h[n-1-k]
- correlate: sum of h[k] to h[n-1]

So they're NOT the same in general!

But wait, the problem says "C₅ = max(np.correlate(h, 1-h, mode='full') * dx)". So the actual evaluation uses the correlate output directly. Let me just work with what the code actually computes.

So the objective is: max_k correlate(h, 1-h, "full")[k] * 2/n

And correlate(h, 1-h, "full")[k] = sum_n h[n+k] * (1-h[n]) (as I derived).

For the coordinate descent, I need to work with this exact formula. Let me re-derive:

corr[k] = sum_{n: 0≤n<n, 0≤n+k<n} h[n+k] * (1-h[n])

Terms involving h[j]:
1. n+k = j, so n = j-k: valid if 0≤j-k<n and 0≤j<n
   Contribution: h[j] * (1-h[j-k]) = x * (1-h[j-k])
   
2. n = j: valid if 0≤j<n and 0≤j+k<n
   Contribution: h[j+k] * (1-x) = h[j+k] - h[j+k]*x

So:
B_k = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)
A_k = corr[k] with h[j]=0

This is exactly what I had! Good, so my CD is correct for the actual objective.

Now, one more thought: the problem statement says the integral is ∫ h(x)(1-h(x+k)) dx. But the actual evaluation uses np.correlate(h, 1-h, "full") which computes sum h[n+k]*(1-h[n]). These are actually the same up to a shift in k!

sum_n h[n+k]*(1-h[n]) = sum_m h[m]*(1-h[m-k]) [substituting m=n+k]

So correlate[h, 1-h](k) = the "cross-correlation" at shift -k in the problem's notation. Since we're taking the max over all k, and the set of all k is symmetric (from -(n-1) to (n-1)), the max is the same either way!

One more optimization idea: instead of computing all pairwise intersections (O(n²) candidates), I can be smarter. The minimum of a convex PL function on [0,1] is where the slope changes from negative to positive. I can find this by:

1. Sort the lines (A_k + B_k * x) by their value at x=0 (i.e., by A_k descending)
2. Start with the top line. The function's slope at x=0 is B_top.
3. If B_top ≥ 0, the minimum is at x=0.
4. Otherwise, find the next line that crosses the current top as x increases.
5. At the crossing point, the active set changes. Continue.
6. Stop when the slope becomes ≥ 0 (or we reach x=1).

This is the "upper envelope" algorithm and is O(n log n) per coordinate. But for n=47, the O(n²) approach is already fast enough.

import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def eval_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * 2.0 / n

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    else:
        m = h > 1e-9
        if m.sum() > 0:
            h[m] += diff / m.sum()
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, target, order):
    n = len(h)
    h = h.copy()
    d_vals = np.arange(2 * n - 1) - (n - 1)
    for j in order:
        h[j] = 0.0
        A = np.correlate(h, 1.0 - h, mode="full")
        B = np.zeros(2 * n - 1)
        jd = j + d_vals
        v1 = (jd >= 0) & (jd < n)
        B[v1] += (1.0 - h[jd[v1]])
        jd2 = j - d_vals
        v2 = (jd2 >= 0) & (jd2 < n)
        B[v2] -= h[jd2[v2]]
        denom = B[:, None] - B[None, :]
        num = A[None, :] - A[:, None]
        mask = np.abs(denom) > 1e-12
        x_mat = np.zeros_like(denom)
        x_mat[mask] = num[mask] / denom[mask]
        valid = (x_mat > 0) & (x_mat < 1)
        cands = np.unique(np.concatenate([[0.0, 1.0], x_mat[valid]]))
        vals = A[None, :] + B[None, :] * cands[:, None]
        h[j] = cands[np.argmin(vals.max(axis=1))]
    return h

def run_cd(h, target, time_limit, rng, max_sweeps=5000):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    stag = 0
    for _ in range(max_sweeps):
        if time.time() - start >= time_limit:
            break
        order = rng.permutation(n)
        h_new = coord_sweep(h, target, order)
        h_new = project(h_new, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best = h_new.copy()
            best_c5 = c5
            stag = 0
        else:
            stag += 1
            if stag > 15:
                k = rng.randint(2, 7)
                idx = rng.choice(n, size=k, replace=False)
                h_new[idx] = rng.uniform(0, 1, k)
                h_new = project(h_new, target)
                stag = 0
        h = h_new
    return best, best_c5

def cd_with_binary_push(h, target, time_limit, rng):
    """CD that also tries to push values toward 0 or 1"""
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    stag = 0
    for it in range(5000):
        if time.time() - start >= time_limit:
            break
        order = rng.permutation(n)
        h_new = coord_sweep(h, target, order)
        h_new = project(h_new, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best = h_new.copy()
            best_c5 = c5
            stag = 0
        else:
            stag += 1
            if stag > 10:
                # Try binary push: move one value to 0 or 1
                j = rng.randint(n)
                if h[j] > 0.5:
                    h_new[j] = 1.0
                else:
                    h_new[j] = 0.0
                h_new = project(h_new, target)
                stag = 0
        h = h_new
    return best, best_c5

def smooth_refine(h, target, time_limit, T, alpha_init):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    alpha = alpha_init
    no_imp = 0
    idx = np.arange(n)
    while time.time() - start < time_limit:
        corr = get_corr(h)
        mc = corr.max()
        p = np.exp((corr - mc) / T)
        s = p.sum()
        if s > 0:
            p /= s
        grad = np.zeros(n)
        for k in range(len(p)):
            w = p[k]
            if w < 1e-10:
                continue
            d = k - (n - 1)
            jd = idx + d
            m1 = (jd >= 0) & (jd < n)
            grad[m1] += w * (1.0 - h[jd[m1]])
            jd2 = idx - d
            m2 = (jd2 >= 0) & (jd2 < n)
            grad[m2] += w * (-h[jd2[m2]])
        h_new = project(h - alpha * grad, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-11:
            best = h_new.copy()
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 25 == 0:
                alpha *= 0.72
                if alpha < 1e-7:
                    break
        h = h_new
    return best, best_c5

def topk_refine(h, target, time_limit, k, alpha_init):
    n = len(h)
    start = time.time()
    h = project(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    alpha = alpha_init
    no_imp = 0
    idx = np.arange(n)
    while time.time() - start < time_limit:
        corr = get_corr(h)
        top = np.argsort(corr)[-k:]
        p = np.zeros(len(corr))
        p[top] = 1.0 / k
        grad = np.zeros(n)
        for ki in top:
            d = ki - (n - 1)
            jd = idx + d
            m1 = (jd >= 0) & (jd < n)
            grad[m1] += (1.0 - h[jd[m1]]) / k
            jd2 = idx - d
            m2 = (jd2 >= 0) & (jd2 < n)
            grad[m2] += (-h[jd2[m2]]) / k
        h_new = project(h - alpha * grad, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-11:
            best = h_new.copy()
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                alpha *= 0.7
                if alpha < 1e-7:
                    break
        h = h_new
    return best, best_c5

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def make_binary_start(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    if n % 2 == 0:
        positions = rng.choice(n, size=n_ones, replace=False)
        h[positions] = 1.0
    else:
        positions = rng.choice(n, size=n_ones, replace=False)
        h[positions] = 1.0
        h[rng.randint(n)] = 0.5
        positions2 = rng.choice(n, size=n_ones, replace=False)
        h = np.zeros(n)
        h[positions2] = 1.0
        h[n // 2] = 0.5
        # Fix: need sum = n/2 = (n-1)/2 + 0.5
        # So (n-1)/2 ones and one 0.5
        h = np.zeros(n)
        pos = list(range(n))
        pos.remove(n // 2)
        chosen = rng.choice(pos, size=(n - 1) // 2, replace=False)
        h[chosen] = 1.0
        h[n // 2] = 0.5
    return project(h, n / 2.0)

    def rem():

    # Get initial
