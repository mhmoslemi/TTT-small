    return float(np.max(corr) * dx)

def compute_gradient_max(h):
    """Compute gradient of the maximizing overlap function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr)) - (n - 1)
    
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + k_star < n:
            g += (1.0 - h[j + k_star])
        if 0 <= j - k_star < n:
            g -= h[j - k_star]
        grad[j] = g * dx
    return grad, k_star

        return h

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
    else:
        order = np.argsort(-h)
    for i in order:
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
        else:
            a = min(-diff, h[i])
            h[i] -= a
            diff += a
    return h

def grad_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ks = int(np.argmax(corr))
    s = ks - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + s < n:
            g += 1.0 - h[j + s]
        if 0 <= j - s < n:
            g -= h[j - s]
        grad[j] = g * dx
    return grad

def all_contribs(h):
    n = len(h)
    C = np.zeros((n, 2 * n - 1))
    for j in range(n):
        for k in range(2 * n - 1):
            s = k - (n - 1)
            c = 0.0
            if 0 <= j + s < n:
                c += 1.0 - h[j + s]
            if 0 <= j - s < n:
                c -= h[j - s]
            C[j, k] = c
    return C

def subgradient_phase(h, target, budget_deadline, iters=4000):
    n = len(h)
    best = h.copy()
    best_c5 = eval_c5(best)
    alpha = 0.08
    for it in range(iters):
        if time.time() > budget_deadline:
            break
        g = grad_max(best)
        hn = project(best - alpha * g, target)
        c = eval_c5(hn)
        if c < best_c5:
            best, best_c5 = hn, c
            alpha = max(alpha * 0.998, 0.001)
        else:
            alpha *= 0.93
            if alpha < 0.003:
                hp = best + 0.15 * np.random.randn(n)
                hp = project(hp, target)
                if eval_c5(hp) < best_c5 + 0.004:
                    best = hp
                    best_c5 = eval_c5(best)
                alpha = 0.04
    return best, best_c5

    best = h.copy()
    best_c5 = eval_c5(best)
    dx = 2.0 / n
    stall = 0

        cm = all_contribs(best)
        improved = False
        for j in range(n):
            if time.time() > budget_deadline:
                break
            if best[j] < 1.0 - 1e-9:
                d = 1.0 - best[j]
                base = corr + d * cm[j]
                nms = (base[None, :] - d * cm).max(axis=1)
                v = np.ones(n, dtype=bool); v[j] = False
                v[best < d - 1e-9] = False
                if v.any():
                    nms[~v] = 999.0
                    mb = int(np.argmin(nms))
                    if nms[mb] * dx < best_c5 - 1e-11:
                        corr = base - d * cm[mb]
                        best[j] = 1.0

def simulated_annealing(h, target, budget_deadline):

    best = h.copy()
    best_c5 = eval_c5(best)
    T = 0.05
    T_min = 0.001
    cooling = 0.9995
    for it in range(50000):
        if time.time() > budget_deadline:
            break
        j = np.random.randint(n)
        m = np.random.randint(n - 1)
        if m >= j:
            m += 1
        delta = np.random.uniform(-0.4, 0.4)
        h_trial = best.copy()
        h_trial[j] += delta
        h_trial[m] -= delta
        h_trial = np.clip(h_trial, 0.0, 1.0)
        diff = target - h_trial.sum()
        if abs(diff) > 1e-8:
            h_trial = project(h_trial, target)
        c_trial = eval_c5(h_trial)

        if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
            best = h_trial
            if c_trial < best_c5:
                best_c5 = c_trial
        T = max(T * cooling, T_min)
    return best, best_c5
