import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def local_search_binary(h, n, dx, max_iters=10000, time_limit=10):
    """Local search for binary (or near-binary) sequences."""
    start = time.time()
    h = h.copy()
    c5 = evaluate_c5(h, dx)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        improved = False
        # Try random pair swaps
        for _ in range(200):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_c5 = evaluate_c5(h, dx)
            if new_c5 < c5 - 1e-12:
                c5 = new_c5
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, c5

def subgradient_descent(h, n, dx, alpha=0.01, max_iters=500, time_limit=10):
    """Subgradient descent for the max correlation."""
    start = time.time()
    h = h.copy()
    c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        # Subgradient: average of gradients at max indices
        max_indices = np.where(corr == max_val)[0]
        grad = np.zeros(n)
        for k_idx in max_indices:
            s = k_idx - (n - 1)
            for j in range(n):
                # d O_s / d h[j]
                if 0 <= j - s < n:
                    grad[j] += (1.0 - h[j - s])
                if 0 <= j + s < n:
                    grad[j] += (-h[j + s])
        grad /= len(max_indices)
        
        # Update with projection
        alpha_step = alpha / (1 + 0.01 * it)
        h_new = h - alpha_step * grad
        h_new = np.clip(h_new, 0, 1)
        # Project onto sum constraint
        diff = n / 2.0 - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        h = h_new
    
    return best_h, best_c5

    """Compute coefs[j, k] = dO_k/dh[j] for all j, k."""
