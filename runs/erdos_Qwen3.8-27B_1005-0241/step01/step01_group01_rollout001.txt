        if w[k_idx] < 1e-15:
            continue
        k = k_idx - (n - 1)  # actual lag
        # For each m, dc[k']/dh[m] = (1-h[m-k]) * [0<=m-k<n] - h[m+k] * [0<=m+k<n]
        # (1-h[m-k]) for m-k in [0,n-1], i.e., m in [k, k+n-1] intersect [0,n-1]
        m_start = max(0, k)
        m_end = min(n, k + n)
        if m_start < m_end:
            indices = np.arange(m_start, m_end)
            grad[indices] += w[k_idx] * (1.0 - h[indices - k])
        
        # -h[m+k] for m+k in [0,n-1], i.e., m in [-k, n-1-k] intersect [0,n-1]
        m_start2 = max(0, -k)
        m_end2 = min(n, n - k)
        if m_start2 < m_end2:
            indices = np.arange(m_start2, m_end2)
            grad[indices] += w[k_idx] * (-h[indices + k])
    
    return grad

def local_search_fractional(h, n, max_iters=5000, tol=1e-8):
    """Fractional local search: try small transfers between pairs"""
    best_h = h.copy()
    best_c5 = get_c5(h)
    target = n / 2.0
    corr = get_corr(h)
    
    for it in range(max_iters):
        improved = False
        # Try all pairs with small transfers
        for i in range(n):
            for j in range(i+1, n):
                # Try moving from i to j
                for d in [0.05, 0.1, 0.2, 0.05, 0.1, 0.2]:
                    if h[i] < d:
                        continue
                    h[i] -= d
                    h[j] += d
                    if h[j] > 1:
                        h[i] += d
                        h[j] = 1
                        continue
                    c5 = float(np.max(get_corr(h)) * (2.0/n))
                    if c5 < best_c5 - 1e-10:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
        if not improved:
            break
    
    return best_h, best_c5
