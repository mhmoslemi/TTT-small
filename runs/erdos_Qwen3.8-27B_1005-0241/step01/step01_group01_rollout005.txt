import numpy as np
import time
from scipy.optimize import minimize

def _fix_sum(h, target_sum):
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            adj = min(diff, 1.0 - h[i])
            h[i] += adj
            diff -= adj
        else:
            adj = min(-diff, h[i])
            h[i] -= adj
            diff += adj

def _eval_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_obj(x, dx, t):
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = corr / t
    sm = s.max()
    return t * (sm + np.log(np.exp(s - sm).sum()))

def _smooth_grad(x, dx, t):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = corr / t
    sm = s.max()
    exps = np.exp(s - sm)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for j in range(n):
        gc = 0.0
        for s_val in range(-(n-1), n):
            k = s_val + n - 1
            wk = w[k]
            i1 = j + s_val
            i2 = j - s_val
            if 0 <= i1 < n:
                gc += wk * (1.0 - x[i1])
            if 0 <= i2 < n:
                gc -= wk * x[i2]
        grad[j] = gc * dx
    return grad

def _smooth_opt(h, n, dx, target_sum, t, maxiter=300):
    def obj(x):
        return _smooth_obj(x, dx, t)
    def grad(x):
        return _smooth_grad(x, dx, t)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if res.x is not None:
        x = np.clip(res.x, 0.0, 1.0)
        _fix_sum(x, target_sum)
        return x
    return None
