```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def enforce_feasibility(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the correction
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            adj = min(diff, 1.0 - h[i])
            h[i] += adj
            diff -= adj
        else:
            adj = min(-diff, h[i])
            h[i] -= adj
            diff += adj
    return h

def compute_coeffs(h):
    """Compute coeff[j][k_idx] for all j and k_idx."""
    n = len(h)
    # coeff[j][k_idx] where k_idx = k + (n-1)
    # coeff[j][k_idx] = (1-h[j-k])_valid - h[j+k]_valid
    # where k = k_idx - (n-1)
    coefs = np.zeros((n, 2*n-1))
    for j in range(n):
        for k_idx in range(2*n-1):
            k = k_idx - (n-1)
            c = 0.0
            # (1-h[j-k])_valid: j-k in [0, n-1]
            idx1 = j - k
            if 0 <= idx1 < n:
                c += (1.0 - h[idx1])
            # -h[j+k]_valid: j+k in [0, n-1]
            idx2 = j + k
            if 0 <= idx2 < n:
                c -= h[idx2]
            coefs[j, k_idx] = c
    return coefs

def local_search_binary(h, target_sum, dx, max_sweeps=200, time_limit=None, start_time=None):
    """Binary local search pushing coordinates to 0 or 1."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    
    for sweep in range(max_sweeps):
        if time_limit and (time.time() - start_time) > time_limit:
            break
        coefs = compute_coeffs(h)
        improved = False
        for j in range(n):
            if time_limit and (time.time() - start_time) > time_limit:
                break
            # Push j to 1
            if h[j] < 1.0 - 1e-10:
                d = 1.0 - h[j]
                base = corr + d * coefs[j]
                new_maxs = (base[None, :] - d * coefs).max(axis=1)
                valid = (np.arange(n) != j) & (h >= d - 1e-10)
                if valid.any():
                    new_maxs_safe = np.where(valid, new_maxs, np.inf)
                    m_best = int(np.argmin(new_maxs_safe))
                    new_c5 = new_maxs_safe[m_best] * dx
                    if new_c5 < best_c5 - 1e-14:
                        corr = base - d * coefs[m_best]
                        h[j] = 1.0
                        h[m_best] -= d
                        best_c5 = new_c5
                        improved = True
            # Push j to 0
            if h[j] > 1e-10:
                d = h[j]
                base = corr - d * coefs[j]
                new_maxs = (base[None, :] + d * coefs).max(axis=1)
                valid = (np.arange(n) != j) & (h + d <= 1.0 + 1e-10)
                if valid.any():

                    m_best = int(np.argmin(new_maxs_safe))

            M = c.max()
            # Gradient of log-sum-exp
            weights = np.exp((c - M) / t) / np.sum(np.exp((c - M) / t))
            # dc/dx = gradient of correlation
            # This is complex, let's use numerical gradient or just SLSQP without gradient
            return None

def random_restart(n, target_sum, rng):
    """Generate a random feasible h."""
    h = rng.random(n)
    # Make it binary-ish
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Adjust to meet sum constraint
    current_sum = h.sum()
    if current_sum > target_sum:
        # Randomly flip some 1s to 0s
        idx1 = np.where(h == 1.0)[0]
        rng.shuffle(idx1)
        for idx in idx1:
            if h.sum() <= target_sum:
                break
            h[idx] = 0.0
    elif current_sum < target_sum:
        idx0 = np.where(h == 0.0)[0]
        rng.shuffle(idx0)
        for idx in idx0:
            if h.sum() >= target_sum:
                break
            h[idx] = 1.0
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_g_target_sum = 0.0
_g_t = 0.01

def _smooth_obj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    M = c.max()
    return float(M + _g_t * np.log(np.sum(np.exp((c - M) / _g_t))))

def _cons_fun(x):
    return np.sum(x) - _g_target_sum

def _cons_jac(x):
    return np.ones(len(x))

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def enforce_feasibility(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            adj = min(diff, 1.0 - h[i])
            h[i] += adj
            diff -= adj
        else:
            adj = min(-diff, h[i])
            h[i] -= adj
            diff += adj
    return h

def compute_coeffs(h):
    n = len(h)
    coefs = np.zeros((n, 2 * n - 1))
    for j in range(n):
        for k_idx in range(2 * n - 1):
            k = k_idx - (n - 1)
            c = 0.0
            idx1 = j - k
            if 0 <= idx1 < n:
                c += (1.0 - h[idx1])
            idx2 = j + k
            if 0 <= idx2 < n:
                c -= h[idx2]
            coefs[j, k_idx] = c
    return coefs

def binary_local_search(h, target_sum, dx, max_sweeps=200, time_limit=None, start_time=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        coefs = compute_coeffs(h)
        improved = False
        for j in range(n):
            if time_limit is not None and (time.time() - start_time) > time_limit:
                break
            if h[j] < 1.0 - 1e-10:
                d = 1.0 - h[j]
                base = corr + d * coefs[j]
                new_maxs = (base[None, :] - d * coefs).max(axis=1)
                valid = (np.arange(n) != j) & (h >= d - 1e-10)
                if valid.any():
                    nm = np.where(valid, new_maxs, np.inf)
                    m_best = int(np.argmin(nm))
                    new_c5 = nm[m_best] * dx
                    if new_c5 < best_c5 - 1e-15:
                        corr = base - d * coefs[m_best]
                        h[j] = 1.0
                        h[m_best] -= d
                        best_c5 = new_c5
                        improved = True
            if h[j] > 1e-10:
                d = h[j]
                base = corr - d * coefs[j]
                new_maxs = (base[None, :] + d * coefs).max(axis=1)
                valid = (np.arange(n) != j) & (h + d <= 1.0 + 1e-10)
                if valid.any():
                    nm = np.where(valid, new_maxs, np.inf)
                    m_best = int(np.argmin(nm))
                    new_c5 = nm[m_best] * dx
                    if new_c5 < best_c5 - 1e-15:
                        corr = base + d * coefs[m_best]
                        h[j] = 0.0
                        h[m_best] += d
                        best_c5 = new_c5
                        improved = True
        if not improved:
            break
    return h, best_c5

def smooth_optimize(h, target_sum, dx, time_limit=None, start_time=None):
    global _g_target_sum, _g_t
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    _g_target_sum = target_sum
    for t in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
        if time_limit is not None and (time.time() - start_time) > time_limit - 15:
            break
        _g_t = t
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(_smooth_obj, best_h, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-14})
            if res.x is not None:
                rh = enforce_feasibility(res.x.copy(), target_sum)
                c5r = compute_c5(rh, dx)
                if c5r < best_c5:
                    best_h = rh
                    best_c5 = c5r
        except Exception:
            pass
    return best_h, best_c5

def annealing_search(h, target_sum, dx, n_iter=30000, T0=0.001, T_end=0.00002,
                     time_limit=None, start_time=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    coefs = compute_coeffs(h)
    c5_cur = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        T = T0 * (T_end / T0) ** (it / max(n_iter, 1))
        zeros = np.where(h < 0.01)[0]
        ones = np.where(h > 0.99)[0]
        if len(zeros) == 0 or len(ones) == 0:
            h = enforce_feasibility(h, target_sum)
            zeros = np.where(h < 0.5)[0]
            ones = np.where(h > 0.5)[0]
            if len(zeros) == 0 or len(ones) == 0:
                continue
            j = zeros[rng.randint(len(zeros))]
            m = ones[rng.randint(len(ones))]
            d = min(h[m], 1.0 - h[j])
            if d < 1e-6:
                continue
            h[j] += d
            h[m] -= d
            corr = np.correlate(h, 1.0 - h, mode="full")
            coefs = compute_coeffs(h)
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < c5_cur or rng.random() < np.exp((c5_cur - new_c5) / T):
                c5_cur = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[j] -= d
                h[m] += d
                corr = np.correlate(h, 1.0 - h, mode="full")
                coefs = compute_coeffs(h)
                c5_cur = float(np.max(corr)) * dx
            continue
        j = zeros[rng.randint(len(zeros))]
        m = ones[rng.randint(len(ones))]
        new_corr = corr + coefs[j] - coefs[m]
        new_c5 = float(np.max(new_corr)) * dx
        if new_c5 < c5_cur or rng.random() < np.exp((c5_cur - new_c5) / T):
            h[j] = 1.0
            h[m] = 0.0
            corr = new_corr
            coefs = compute_coeffs(h)
            c5_cur = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5

def iterated_local_search(h, target_sum, dx, n_kicks=50, kick_size=3,
                          time_limit=None, start_time=None, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for kick in range(n_kicks):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        trial = best_h.copy()
        for _ in range(kick_size):
            j = rng.randint(n)
            d = rng.uniform(0.3, 0.9) * (1 if rng.random() < 0.5 else -1)
            trial[j] = np.clip(trial[j] + d, 0, 1)
        trial = enforce_feasibility(trial, target_sum)
        trial, c5_trial = binary_local_search(trial, target_sum, dx,
                                              max_sweeps=80, time_limit=min(5, remaining_time(start_time, time_limit)),
                                              start_time=time.time())
        if c5_trial < best_c5:
            best_h = trial
            best_c5 = c5_trial
    return best_h, best_c5

def remaining_time(start_time, time_limit):
    if time_limit is None:
        return 9999
    return max(0.1, time_limit - (time.time() - start_time))

def structured_init(n, rng, ptype):
    target_sum = n / 2.0
    h = np.zeros(n)
    if ptype == 0:
        h = rng.randint(0, 2, n).astype(float)
    elif ptype == 1:
        h = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        for _ in range(n // 5):
            h[rng.randint(n)] = 1.0 - h[rng.randint(n)]
    elif ptype == 2:
        blk = max(1, n // 5)
        h[:blk] = 1.0
        h[n // 2:n // 2 + blk] = 1.0
    else:
        for i in range(n):
            if (i * i + i + 1) % n < n // 2:
                h[i] = 1.0
    return enforce_feasibility(h, target_sum)
```
