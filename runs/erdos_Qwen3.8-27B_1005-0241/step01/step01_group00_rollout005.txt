    return np.max(get_corr(h)) * 2.0 / n

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        m = h < 1.0 - 1e-10
        if m.sum() > 0:
            h[m] += diff / m.sum()
    else:
        m = h > 1e-10
        if m.sum() > 0:
            h[m] += diff / m.sum()
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, d, n):
    grad = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n, n + d)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - d:hi1 - d])
    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    if lo2 < hi2:
        grad[lo2:hi2] += (-h[lo2 + d:hi2 + d])
    return grad

def grad_weighted(h, p, n):
    grad = np.zeros(n)
    for k in range(len(p)):
        w = p[k]
        if w < 1e-12:
            continue
        d = k - (n - 1)
        lo1 = max(0, d)
        hi1 = min(n, n + d)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - d:hi1 - d])
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2 + d:hi2 + d])
    return grad

def optimize_softmax(h0, target, time_limit, T_init=2.0, T_final=0.03, alpha_init=0.04, beta=0.85):
    n = len(h0)
    t0 = time.time()
    h = project(h0.copy(), target)
    v = np.zeros(n)
    alpha = alpha_init
    best = h.copy()
    best_c5 = eval_c5(h)
    count = 0
    total_est = 400
    while time.time() - t0 < time_limit:
        count += 1
        frac = min(1.0, count / total_est)
        T = T_init * (T_final / T_init) ** frac
        T = max(T, T_final)
        corr = get_corr(h)
        mc = corr.max()
        p = np.exp((corr - mc) / T)
        s = p.sum()
        if s > 1e-15:
            p /= s
        else:
            p = np.ones(len(corr)) / len(corr)
        grad = grad_weighted(h, p, n)
        beta_c = beta * (1.0 - 0.3 * frac)
        v = beta_c * v + (1.0 - beta_c) * grad
        h_new = project(h - alpha * v, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best = h_new.copy()
            best_c5 = c5
            h = h_new
        else:
            if c5 > best_c5 + 0.005:
                alpha *= 0.6
                v *= 0.3
                h = best.copy()
            else:
                h = h_new
        alpha = max(alpha * 0.9995, 0.002)
    return best, best_c5

def optimize_flatten(h0, target, time_limit, alpha_init=0.05):
    n = len(h0)
    t0 = time.time()
    h = project(h0.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    alpha = alpha_init
    stall = 0
    while time.time() - t0 < time_limit:
        corr = get_corr(h)
        w = corr.copy()
        ws = w.sum()
        if ws > 1e-10:
            w /= ws
        else:
            w = np.ones(len(corr)) / len(corr)
        grad = grad_weighted(h, w, n)
        h_new = project(h - alpha * grad, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best = h_new.copy()
            best_c5 = c5
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                alpha *= 0.7
            if stall > 300:
                break
    return best, best_c5

def local_pair_search(h0, target, time_limit):
    n = len(h0)
    t0 = time.time()
    h = project(h0.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    while time.time() - t0 < time_limit:
        corr = get_corr(h)
        idx = np.argsort(corr)[::-1]
        improved = False
        for ki in range(min(3, len(idx))):
            k = idx[ki]
            d = k - (n - 1)
            gd = grad_shift(h, d, n)
            order = np.argsort(gd)
            found = False
            for ii in range(min(15, n)):
                i = order[n - 1 - ii]
                if h[i] >= 1.0 - 1e-8:
                    continue
                for jj in range(min(15, n)):
                    j = order[jj]
                    if h[j] <= 1e-8 or i == j:
                        continue
                    eps = min(0.3, (1.0 - h[i]) * 0.4, h[j] * 0.4)
                    if eps < 1e-5:
                        continue
                    h[i] += eps
                    h[j] -= eps
                    c5 = eval_c5(h)
                    if c5 < best_c5 - 1e-12:
                        best = h.copy()
                        best_c5 = c5
                        improved = True
                        found = True
                        break
                    else:
                        h[i] -= eps
                        h[j] += eps
                if found:
                    break
            if found:
                break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    if h_init is not None:
        n = len(h_init)
        target = n / 2.0
        best_h = project(h_init.copy(), target)
    else:
        n = 47
        target = n / 2.0
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_h[0] = 0.5
        best_h = project(best_h, target)
    best_c5 = eval_c5(best_h)

    def rem():
        return max(0.0, budget_s - (time.time() - t_start) - 3.0)

    # Phase 1: Flatten (smooth warm-up)
    t = rem() * 0.15
    if t > 5:
        h_o, c5_o = optimize_flatten(best_h, target, t, alpha_init=0.05)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 2: Softmax gradient with momentum
    t = rem() * 0.30
    if t > 5:
        h_o, c5_o = optimize_softmax(best_h, target, t, T_init=2.0, T_final=0.05, alpha_init=0.04)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 3: Local pair search
    t = rem() * 0.20
    if t > 3:
        h_o, c5_o = local_pair_search(best_h, target, t)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 4: Tighter softmax
    t = rem() * 0.15
    if t > 5:
        h_o, c5_o = optimize_softmax(best_h, target, t, T_init=0.3, T_final=0.01, alpha_init=0.015)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 5: More local search
    t = rem() * 0.10
    if t > 3:
        h_o, c5_o = local_pair_search(best_h, target, t)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 6: Perturbation + local search
    t = rem() * 0.50
    if t > 10:
        n_rounds = int(t / 4)
        for _ in range(n_rounds):
            if rem() < 5:
                break
            noise = 0.005 + 0.02 * rng.rand()
            h_p = project(best_h + rng.normal(0, noise, n), target)
            t_sub = min(4.0, rem() - 2.0)
            if t_sub < 1.5:
                break
            h_o, c5_o = local_pair_search(h_p, target, t_sub)
            if c5_o < best_c5:
                best_h, best_c5 = h_o, c5_o

    # Final polish
    if rem() > 5:
        h_o, c5_o = local_pair_search(best_h, target, min(rem() - 2, 15))
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    return best_h, best_c5, n
