    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if mask.sum() > 0:
            step = diff / mask.sum()
            # Need to ensure we don't exceed 1
            # Find how much we can add
            room = 1.0 - h[mask]
            if room.sum() < diff:
                # Can't add that much, set as many to 1 as possible

                for i in idx:

def optimize_smooth_ls(h, target, time_limit, T_init=2.0, T_min=0.1, 
                        alpha_init=0.1, n_anneal_steps=50):

    n = len(h)
    start = time.time()
    h = project_sum(h.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    
    total_steps = 200  # approximate
    for step in range(total_steps):
        if time.time() - start > time_limit - 2:
            break
        # Anneal temperature
        frac = step / max(1, n_anneal_steps)
        T = T_init * (T_min / T_init) ** min(1.0, frac * 3)
        T = max(T, T_min)
        
        corr = np.correlate(h, 1 - h, mode="full") * 2.0 / n
        mc = corr.max()
        logits = T * (corr - mc)
        p = np.exp(logits - logits.max())
        p /= p.sum()
        
        grad = compute_weighted_grad(h, p, n)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        # Backtracking line search
        alpha = alpha_init
        c5_curr = best_c5
        accepted = False
        for ls_iter in range(20):
            h_new = project_sum(h - alpha * gnorm * grad, target)
            c5_new = eval_c5(h_new)
            if c5_new < c5_curr - 1e-10 * alpha:
                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break

            h = h_new
            if c5_new < best_c5:
                best = h_new.copy()
                best_c5 = c5_new
            alpha_init = min(alpha * 1.5, 0.3)

            if c5_new < best_c5 - 1e-12:
                h = h_new
                best = h_new.copy()
                best_c5 = c5_new
            alpha_init *= 0.8
    
    return best, best_c5

    # Phase 2: Top-k focused
    t = rem() * 0.22
    if t > 5:
        h_o, c5_o = optimize_topk(best_h, target, t, k=3, alpha_init=0.05)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 3: Worst-shift refinement
    t = rem() * 0.18
    if t > 5:
        h_o, c5_o = optimize_worst(best_h, target, t, alpha_init=0.04)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 4: Momentum with low T
    t = rem() * 0.15
    if t > 5:
        h_o, c5_o = optimize_momentum(best_h, target, t, T_init=2.0, T_min=0.1, alpha_init=0.04)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 5: Try different n
    for n_try in [50, 49, 55, 51, 53, 57]:
        if rem() < 25:
            break
        if n_try == len(best_h):
            continue
        tgt = n_try / 2.0
        h_try = make_initial(n_try, 'block3')
        h_try = project_sum(h_try + 0.2 * rng.rand(n_try) - 0.1, tgt)
        t_sub = min(rem() * 0.12, 45)
        if t_sub < 5:
            t_sub = min(rem() * 0.25, 20)
        if t_sub < 3:
            break
        h_o, c5_o = optimize_momentum(h_try, tgt, t_sub, T_init=5.0, T_min=0.3)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 6: Random restarts with subgradient
    n_restarts = 40
    t_each = max(rem() / max(1, n_restarts), 2.0)
    for _ in range(n_restarts):
        if rem() < 8:
            break
        scale = 0.03 + 0.04 * rng.rand()
        h_p = project_sum(best_h + rng.normal(0, scale, len(best_h)), target)
        t_sub = min(t_each, rem() - 3)
        if t_sub < 2:
            break
        h_o, c5_o = optimize_worst(h_p, target, t_sub, alpha_init=0.03)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Final polish
    if rem() > 12:
        h_o, c5_o = optimize_worst(best_h, target, min(rem() - 3, 30), alpha_init=0.02)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
    if rem() > 10:
        h_o, c5_o = optimize_momentum(best_h, target, min(rem() - 3, 25), T_init=1.0, T_min=0.05, alpha_init=0.02)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    return best_h, best_c5, len(best_h)
