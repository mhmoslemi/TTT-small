from scipy.optimize import minimize
from numba import njit
import math

@njit
def distance_squared(a, b):
    return (a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2

@njit
def objective(x, n, num_circles):

    radii = x[2 * n:].copy()

    for i in range(n):
        total += radii[i]
    return -total  # minimize negative of total

@njit
def non_overlapping_constraints(x, n, num_circles):
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:].copy()
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist_squared = distance_squared(centers[i], centers[j])
            if dist_squared < (radii[i] + radii[j]) ** 2 - 1e-8:
                # Overlap detected
                constraints.append(0)
            else:
                constraints.append(1)
    return constraints
