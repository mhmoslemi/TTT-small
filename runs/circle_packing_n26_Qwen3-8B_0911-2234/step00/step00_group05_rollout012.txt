import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles
    n = 26

    # Step 1: Place 19 interior circles in a hexagonal grid
    grid_rows = 5
    grid_cols = 4
    interior_circles = []

    # Hexagonal grid: vertical spacing between rows is sqrt(3)/2 times the horizontal spacing
    grid_spacing_x = 0.25
    grid_spacing_y = np.sqrt(3) / 2 * grid_spacing_x

    for row in range(grid_rows):
        for col in range(grid_cols):
            x = col * grid_spacing_x + 0.5
            y = row * grid_spacing_y + 0.5
            interior_circles.append([x, y])

    # Step 2: Add 7 boundary circles near the corners and edges
    boundary_circles = [
        [0.1, 0.1],  # Top-left corner
        [0.1, 0.9],  # Bottom-left corner
        [0.9, 0.1],  # Top-right corner
        [0.9, 0.9],  # Bottom-right corner
        [0.5, 0.1],  # Top edge
        [0.5, 0.9],  # Bottom edge
        [0.1, 0.5],  # Left edge
    ]

    # Combine interior and boundary circles
    centers = np.array(interior_circles + boundary_circles)

    # Step 3: Initial radii based on local packing density
    # Interior circles: radius depends on grid spacing
    delta = grid_spacing_x / 2
    r_interior = delta * 0.8
    # Boundary circles: radius is smaller to fit in corners
    r_boundary = delta * 0.4

    # Step 4: Optimize the radii using constrained optimization
    # Objective: maximize sum of radii
    # Constraints: no overlap, boundary constraints

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_no_overlap(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = centers[i]
                xj, yj = centers[j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(n):
            xi, yi = centers[i]
            if xi - radii[i] < -1e-8 or xi + radii[i] > 1 + 1e-8:
                return -1
            if yi - radii[i] < -1e-8 or yi + radii[i] > 1 + 1e-8:
                return -1
        return 0

    # Initial guess for radii
    radii_initial = np.array([r_interior] * len(interior_circles) + [r_boundary] * len(boundary_circles))

    # Setup optimization problem
    bounds = [(0, 1)] * n

    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_no_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Run optimization
    result = optimize.minimize(
        objective,
        radii_initial,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-6,
        options={'maxiter': 1000, 'eps': 1e-6}
    )

    if not result.success:
        raise ValueError("Optimization failed to converge.")

    # Extract final result
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate the packing
    success, msg = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Validation failed: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
