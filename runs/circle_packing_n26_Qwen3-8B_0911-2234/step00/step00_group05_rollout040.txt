    num_cols = 7
    grid_spacing = 0.2
    x_offset = 0.5
    y_offset = 0.5

    centers = np.zeros((num_rows * num_cols, 2))
    radii = np.zeros(num_rows * num_cols)

        for col in range(num_cols):
            x = x_offset + col * grid_spacing
            y = y_offset + row * grid_spacing

                x += grid_spacing / 2
            centers[row * num_cols + col] = [x, y]
    
    # Scale the grid to fit within [0,1]x[0,1]
    centers = (centers - np.min(centers)) / np.ptp(centers)
    radii = np.full_like(centers, 0.05)  # Initial guess for radii
    
    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # We minimize the negative sum
    
    # Define constraints for non-overlap and boundaries
    def non_overlap_constraint(radii):
        # Ensure circles don't overlap and stay within [0,1]^2
        n = len(radii)

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
