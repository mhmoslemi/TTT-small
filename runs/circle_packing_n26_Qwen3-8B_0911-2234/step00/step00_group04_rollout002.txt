from scipy.optimize import Bounds

    # This is a simple hexagonal grid packed into a square.
    # We will start with 4 rows to fit 26 circles.
    num_circles = 26
    max_rows = 4
    cols = (num_circles + max_rows - 1) // max_rows
    cols = max(1, cols)

    # Generate initial guess - attempt grid packing
    centers = []
    for row in range(max_rows):

            centers.append((x, y))

        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints_func(radii):

        constraints = []
        # Boundary constraints

            if r > 1e-12:
                constraints.append(lambda r, i=i: r - (1 - centers[i, 0]) - 1e-12)
                constraints.append(lambda r, i=i: r - (centers[i, 0]) - 1e-12)
                constraints.append(lambda r, i=i: r - (1 - centers[i, 1]) - 1e-12)
                constraints.append(lambda r, i=i: r - (centers[i, 1]) - 1e-12)
        # Pairwise non-overlap constraints

                    dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    return r[i] + r[j] - dist - 1e-12

        return constraints

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
