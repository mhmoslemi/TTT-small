from scipy.optimize import minimize

def run_packing():
    # Parameters for the hexagonal packing
    num_layers = 5
    num_circles_per_layer = [1] + [6 * i for i in range(1, num_layers)]

    # Start with center circle at (0.5, 0.5)
    centers = []
    radii = []

    for layer_idx in range(num_layers):
        if layer_idx == 0:
            # First layer: only the center circle
            centers.append([0.5, 0.5])
            radii.append(0.3)  # Initial radius (this will be optimized)
        else:
            # For higher layers, build outward
            # The angle between adjacent circles in a hexagonal packing is 60 degrees
            num_circles = num_circles_per_layer[layer_idx]
            angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
            radius = 0.1  # Seed radius for this layer
            for theta in angles:
                x = 0.5 + radius * np.cos(theta)
                y = 0.5 + radius * np.sin(theta)
                centers.append([x, y])
                radii.append(radius)

    centers = np.array(centers)
    radii = np.array(radii)

    # Sizing optimization
    def objective(radii_flat):
        return -np.sum(radii_flat)

    def constraints(radii_flat):
        # Ensure all circles are inside square [0,1]^2
        for i in range(len(radii_flat)):
            x, y = centers[i]
            if x - radii_flat[i] < 0 or x + radii_flat[i] > 1:
                return False
            if y - radii_flat[i] < 0 or y + radii_flat[i] > 1:
                return False

        # Ensure no overlapping circles
        for i in range(len(radii_flat)):
            for j in range(i + 1, len(radii_flat)):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return False

        return True

    # Bounds for the radii: must be non-negative
    bounds = [(0, 1) for _ in range(len(radii))]

    # Convert radii to flat list for optimization
    radii_flat = radii.copy()

    # Create the optimization problem
    def constraint_func(x):
        return constraints(x)

    result = minimize(
        objective,
        radii_flat,
        bounds=bounds,
        constraints={"type": "success", "fun": constraint_func},
        method="SLSQP"

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
