import cvxpy as cp

    # We'll start with a 5x5 grid (25 circles) and add one more to fill the last spot
    # Let's define the centers

            x = x / 3.0

            centers[i * 5 + j] = [x, y]

    centers[-1] = [0.9, 0.9]
    radii[-1] = 0.05

    # Now, use a convex optimization approach to refine the packing
    # We'll define variables for radii and centers
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Objective: Maximize total radius sum
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(n):

        constraints += [x[i, 0] + r[i] <= 1]

    # Solve the optimization problem with CVXPY
    prob = cp.Problem(objective, constraints)
