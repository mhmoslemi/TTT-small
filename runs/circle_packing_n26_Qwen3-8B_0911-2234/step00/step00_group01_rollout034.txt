import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Base configuration: Hexagonal packing with 5 rows (total 26 circles)
    # The pattern is row-wise with 5 columns, 5, 6, 5, 1 columns
    # Starting from the bottom, adjusting vertically and horizontally for tight packing
    
    # Initial positions (row-wise around center)
    circles = []
    for row in range(5):
        for col in range(5 + (row == 1) + (row == 3) + (row == 4)):
            x = col / 4.5  # Scaled for tight horizontal packing
            y = row / 4.0  # Scaled for vertical packing
            circles.append((x, y))
    
    # Convert to numpy arrays
    centers = np.array(circles)
    radii = np.full(len(centers), 0.05)  # Initial guess for radius
    
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize
    
    # Define constraints: non-overlapping and inside square
    constraints = []
    
    # Non-overlapping constraint: for each pair (i, j)
    def non_overlapping_constraint(radii):
        return np.array([np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j]) for i in range(len(centers)) for j in range(i + 1, len(centers))])
    
    constraints.append({'type': 'ineq', 'fun': lambda r: non_overlapping_constraint(r)})
    
    # Boundary constraint: ensure circles are fully inside square
    def boundary_constraint(radii):
        return np.array([np.min(np.array([centers[i, 0] - radii[i], centers[i, 1] - radii[i], 1 - centers[i, 0] - radii[i], 1 - centers[i, 1] - radii[i]])) for i in range(len(centers))])
    
    constraints.append({'type': 'ineq', 'fun': boundary_constraint})
    
    # Optimization problem
    result = minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(len(centers))]
    )
    
    if not result.success:
        raise ValueError("Optimization failed: " + result.message)
    
    centers = np.array(circles)
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
