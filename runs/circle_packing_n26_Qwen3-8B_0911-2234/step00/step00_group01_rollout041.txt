import numpy as np

def run_packing():

    # Radii and centers for 26 circles
    # First, place 6 rows: row 0 has 3 circles, row 1 has 4 circles, row 2 has 5 circles, row 3 has 4, row 4 has 3, row 5 has 2
    # These are manually placed to use the area efficiently.

    # Define a grid-like approach with slight spacing for non-overlapping
    rows = 6
    cols = 5
    spacing = 0.25  # spacing between circles
    shift = 0.1  # shift for staggered layout

    centers = []
    radii = []

    for r in range(rows):
        for c in range(cols):
            if r % 2 == 0:
                x = c * spacing
                y = r * spacing
            else:
                x = c * spacing + shift
                y = r * spacing + shift

            # Convert coordinates into unit square by scaling
            x = x / (rows * spacing)
            y = y / (rows * spacing)

            # Define radius based on position, with larger radii at the center
            if 0 < r < rows - 1 and 0 < c < cols - 1:
                r_val = 0.1
            elif r == 0 or r == rows - 1 or c == 0 or c == cols - 1:
                r_val = 0.08
            else:
                r_val = 0.12

            centers.append([x, y])
            radii.append(r_val)

    # Ensure there are exactly 26 circles

        raise ValueError("Number of circles does not match 26")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
