import scipy.optimize as optimize

def run_packing():

    # This is a rough estimate with 10 circles per row and 3 rows

    centers = []
    radii = []

    # Initial radius approximation based on spacing
    spacing = np.sqrt(3) / 2  # Hexagonal spacing
    initial_radius = spacing / 2

    # Start at the bottom of the square
    y_start = 0.5 * spacing
    y_step = spacing

    # First row (bottom)
    for i in range(10):
        x = i * spacing
        y = y_start
        centers.append([x, y])
        radii.append(initial_radius)

    # Second row (offset by half spacing)
    for i in range(9):
        x = i * spacing + spacing / 2
        y = y_start + y_step
        centers.append([x, y])
        radii.append(initial_radius)

    # Third row (top)
    for i in range(8):
        x = i * spacing
        y = y_start + 2 * y_step
        centers.append([x, y])
        radii.append(initial_radius)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    def objective_function(radii):
        return -np.sum(radii)  # Maximize total radii

    def constraint_circle_boundary(radii, centers):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return -1
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -1
        return 0

    def constraint_circles_overlap(radii, centers):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Initial guess for radii
    initial_guess = np.array([initial_radius] * n)

    # Optimization settings
    bounds = [(0, 1) for _ in range(n)]  # Radii must be non-negative and <= 1

    return (centers, optimal_radii, sum_radii)
