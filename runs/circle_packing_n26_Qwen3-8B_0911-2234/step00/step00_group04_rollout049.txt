import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    """
    This function attempts to pack 26 circles in a unit square [0,1]×[0,1] to maximize the sum of radii.
    The approach involves:
    1. Starting with an initial guess using a hexagonal packing in a 5x5 grid (5 rows of 5 circles each).
    2. Using a greedy optimization to incrementally adjust the radii of the circles, trying to increase the total sum.
    3. In each step, the circle with the smallest radius is selected, and its radius is increased as much as possible while maintaining non-overlapping and boundary constraints.
    4. The process is repeated until no further improvements can be made within a certain number of iterations.
    """

    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    def energy_function(radii, centers):
        energy = 0.0
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            energy += max(0, (x - r) + 1e-12) + max(0, (1 - x) - r + 1e-12) + max(0, (y - r) + 1e-12) + max(0, (1 - y) - r + 1e-12)
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    energy += (radii[i] + radii[j] - dist) ** 2
        return energy

    def optimize_increase(centers, initial_radii, max_iterations=1000, tolerance=1e-6):
        radii = initial_radii.copy()
        for _ in range(max_iterations):
            best_circle = -1
            best_increase = 0.0
            for i in range(len(radii)):
                x, y = centers[i]
                r = radii[i]
                max_possible_increase = 0.0
                # Try to increase radius by moving as far as possible while staying within bounds
                max_x = min(1 - r, 1 - 1e-12)
                max_y = min(1 - r, 1 - 1e-12)
                min_x = max(r, 1e-12)
                min_y = max(r, 1e-12)
                if max_x > x and max_y > y:
                    max_possible_increase = 1e-6
                elif max_x > x:
                    max_possible_increase = max(1e-6, max_x - x)
                elif max_y > y:
                    max_possible_increase = max(1e-6, max_y - y)
                elif min_x < x:
                    max_possible_increase = max(1e-6, x - min_x)
                elif min_y < y:
                    max_possible_increase = max(1e-6, y - min_y)
                else:
                    max_possible_increase = 0.0

                if max_possible_increase > best_increase:
                    best_increase = max_possible_increase
                    best_circle = i

            if best_increase < tolerance:
                break

            # Increase the radius of the best_circle
            radii[best_circle] += best_increase

            # Check if the new radii violate constraints
            valid, message = validate_packing(centers, radii)
            if not valid:
                radii[best_circle] -= best_increase
                break

        return radii

    # Initial guess: 5x5 grid with spacing adjustment
    num_circles = 26
    num_rows = 5
    num_cols = 5
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    for i in range(num_rows):
        for j in range(num_cols):
            idx = i * num_cols + j
            x = j / (num_cols - 1) if num_cols > 1 else 0.5
            y = i / (num_rows - 1) if num_rows > 1 else 0.5
            centers[idx] = [x, y]
            radii[idx] = 0.025  # Small initial radius

    # First pass: heuristically increase radii to maximize sum
    max_iterations = 100
    for _ in range(max_iterations):
        radii = optimize_increase(centers, radii)
        print(f"Total sum after iteration {_ + 1}: {np.sum(radii)}")

    # Final validation
    valid, message = validate_packing(centers, radii)
    if not valid:
        print(f"Validation failed: {message}")
        return np.array([[0, 0]] * 26), np.zeros(26), 0.0

    print(f"Final total sum of radii: {np.sum(radii)}")
    return centers, radii, np.sum(radii)
