import random
import math

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]

            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"

def compute_hexagonal_grid(n_circles):
    # Generate a hexagonal packing in a grid
    grid = []
    radius = 1.0 / math.sqrt(3)  # Assume radius to be fixed for now
    max_rows = int(np.sqrt(n_circles))
    if max_rows * max_rows < n_circles:
        max_rows += 1

        for j in range(max_rows):
            if i * max_rows + j < n_circles:
                x = j + 0.5 * i
                y = i * (np.sqrt(3) / 2)
                grid.append((x, y))
    return np.array(grid), np.full(n_circles, radius)

def perturb_centers(centers, perturbation_limit=0.1):
    perturbed_centers = centers.copy()
    for i in range(centers.shape[0]):
        dx = random.uniform(-perturbation_limit, perturbation_limit)
        dy = random.uniform(-perturbation_limit, perturbation_limit)
        perturbed_centers[i] += [dx, dy]
    return perturbed_centers

    scale = min(scale_x, scale_y)
    scaled_centers = centers * scale
    return scaled_centers
