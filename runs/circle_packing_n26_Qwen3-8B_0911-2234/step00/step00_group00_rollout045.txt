import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initialize centers in a 5x5 grid with some padding
    grid_points = np.linspace(0.1, 0.9, 5)
    centers = np.array([[x, y] for x in grid_points for y in grid_points])
    n = len(centers)
    radii = np.zeros(n)
    
    # Define the objective function to maximize sum of radii
    def objective(radii):
        # Compute the effective radius for each circle
        for i in range(n):
            r = radii[i]
            # Compute distance to boundaries
            dist_left = centers[i, 0] - r
            dist_right = 1.0 - centers[i, 0] - r
            dist_bottom = centers[i, 1] - r
            dist_top = 1.0 - centers[i, 1] - r
            r_boundary = min(dist_left, dist_right, dist_bottom, dist_top)
            # Compute distance to adjacent circles
            r_adjacent = np.inf
            for j in range(n):
                if i == j:
                    continue
                dist_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist_ij < radii[j] + radii[i] + 1e-12:
                    continue
                r_adjacent = min(r_adjacent, dist_ij - radii[j] - radii[i])
            radii[i] = min(r_boundary, r_adjacent)
        return -np.sum(radii)  # Minimize negative of sum of radii

    # Use optimization to maximize the sum of radii
    result = optimize.minimize(
        objective,
        x0=radii,
        bounds=[(0, 1.0) for _ in range(n)],
        method='L-BFGS-B'
    )
    
    # Extract results
    centers = np.array([[x, y] for x in grid_points for y in grid_points])
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Validate packing to ensure no overlaps and boundaries respected
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise Exception(f"Packing validation failed: {message}")
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
