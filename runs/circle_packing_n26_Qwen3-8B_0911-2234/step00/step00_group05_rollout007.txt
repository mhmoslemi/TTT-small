import numpy as np
import cvxpy as cp

def run_packing():

    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Non-overlapping constraint: distance between centers >= sum of radii
    for i in range(n):

            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS)

    # Check for success
    if not problem.success:
