from scipy.optimize import minimize

def run_packing():

    # The grid spans from (0, 0) to (1, 1)
    # We'll generate a triangular grid with up to 6 columns and 5 rows to fit 26 circles

    centers = []

    def constraints(radii):

        for i in range(n):
