def run_packing():

    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    grid = []

        for j in range(cols):

                x = j + 0.5
            else:
                x = j + 1
            y = i + 0.5

        for i in range(len(radii)):

            r = radii[i]
            boundaries[i] = 1 - (np.max([x - r, y - r, 1 - x - r, 1 - y - r])) ** 2
