def run_packing():

    # Start with a grid pattern and then improve with optimization

    # Precompute grid points
    grid = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    # Initial guess for radii: all radius = 0.05 (small)
    radii = np.full(grid.shape[0], 0.05)

    # Objective function: maximize sum of radii while minimizing energy
    def objective(params):
        centers = params[:grid.shape[0] * 2].reshape(-1, 2)
        radii = params[grid.shape[0] * 2:]
        energy = 0.0

        # Enforce boundaries
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                energy += 1e6 * r  # Penalize circles going outside

        # Enforce no overlaps
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    energy += 1e5 * (radii[i] + radii[j] - dist)  # Penalize overlaps

        return -np.sum(radii) + energy  # Minimize energy, maximize sum of radii

    # Prepare initial parameters: centers and radii
    initial_params = grid.flatten()
    initial_params = np.concatenate((initial_params, radii))

    # Optimize
    result = optimize.minimize(
        objective,
        initial_params,
        method='L-BFGS-B',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
