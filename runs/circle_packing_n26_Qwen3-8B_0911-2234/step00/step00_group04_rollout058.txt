from scipy.optimize import minimize

def run_packing():

    n = 26
    rows = int(np.sqrt(n)) + 1
    cols = int(np.sqrt(n)) + 1
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        centers[i] = [x, y]

    # Initial radius (based on grid spacing)
    dx = 1.0 / (cols - 1)
    dy = 1.0 / (rows - 1)

    radii = np.full(n, initial_radius)

    def objective(x):
        # x is a flat array of (centers, radii)
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        return -np.sum(radii_flat)  # negate for minimization

    # Define constraints
    def constraint_overlap(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:

        return 1.0  # no overlap
