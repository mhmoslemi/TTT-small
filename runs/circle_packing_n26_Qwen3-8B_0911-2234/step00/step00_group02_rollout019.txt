    def objective(radii):
        return -np.sum(radii)  # Use negative for minimization

    def constraint_center_bounds(center, radius):
        return np.array([
            center[0] - radius >= 0,
            center[0] + radius <= 1,
            center[1] - radius >= 0,
            center[1] + radius <= 1
        ])

    def constraint_overlap(center_i, radius_i, center_j, radius_j):

        return dist - radius_i - radius_j
