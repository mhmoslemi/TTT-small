import random
import math
from scipy.optimize import minimize

            y = row + 0.5
            positions.append((x, y))
        return positions

        return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

        n = len(config)
        total_sum = 0.0
        # Try to adjust radii based on neighbor distances
        max_radius = 0.0
        for i in range(n):
            r = max(0.01, 0.5 / n) * (1.0 + 0.5 * np.sin(config[i][0] * 10))
            config[i] = (config[i][0], config[i][1], r)
            max_radius = max(max_radius, r)
        return max_radius

    def energy_with_constraints(config):
        n = len(config)
        total_sum = 0.0
        for i in range(n):
            x, y, r = config[i]
            if r < 0:
                return -float('inf')
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -float('inf')
            total_sum += r
        for i in range(n):
            x1, y1, r1 = config[i]
            for j in range(i + 1, n):
                x2, y2, r2 = config[j]
                if distance((x1, y1), (x2, y2)) < r1 + r2 - 1e-8:
                    return -float('inf')
        return total_sum

    def simulated_annealing(n=26):

        cooling_rate = 0.995

        best_config = config.copy()
