import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles
    n = 26

    # Define base grid for initial positions in a hexagonal-like arrangement
    # We use a shifted grid to simulate hexagonal packing
    positions = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        positions.append([x, y])

    # Convert positions to a NumPy array
    centers = np.array(positions)

    # Initial guess for radii: uniform
    radii = np.full(n, 0.05)

    # Optimization: Maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        # Define constraints: no overlap, within bounds
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    # Use non-linear optimization with constraints
    res = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': constraints}],
        bounds=[(0, 1) for _ in range(n)]
    )

    # Extract the optimal radii and centers
    radii = res.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
