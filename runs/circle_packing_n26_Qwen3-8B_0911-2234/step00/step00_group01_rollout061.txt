from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize.constraints import Constraint

def run_packing():
    # Define initial grid layout: 5 rows and 5 columns (25 circles)
    num_circles = 26

        for i in range(num_circles):

            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
