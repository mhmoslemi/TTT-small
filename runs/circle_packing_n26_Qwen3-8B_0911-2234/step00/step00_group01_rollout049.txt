import cvxpy as cp

def run_packing():

    n = centers.shape[0]
    radii = cp.Variable(n)
    obj = cp.Maximize(cp.sum(radii))

        constraints.append(y_i + radii[i] <= 1 + 1e-12)
    
    # 2. No overlaps between circles

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append(radii[i] + radii[j] <= dist + 1e-12)
