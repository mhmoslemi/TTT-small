import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial hexagonal grid arrangement: 26 circles arranged in a roughly hexagonal grid
    # rows: 5 (approximate)
    # Each row has 6, 6, 5, 5, 5 circles (row-wise, adjusted to meet 26 total)

    # Step 1: Generate hexagonal grid positions
    rows = 5
    cols = 5
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5
            centers.append([x, y])
    centers = np.array(centers)

    # Step 2: Initial guess for radii: all equal, small value for scaling
    # Use a small initial radius to avoid overlap
    radii_initial = np.full(len(centers), 0.05)

    # Step 3: Define the optimization variables
    # We'll use a vector x where the first 2*len(centers) elements are the centers,
    # and the next len(centers) elements are the radii
    x0 = np.concatenate([centers.flatten(), radii_initial])

    # Constraints
    def constraint_func(x):
        # Extract centers and radii from x
        centers_concat = x[:2 * len(centers)].reshape(len(centers), 2)
        radii_concat = x[2 * len(centers):]

        # All circles must lie within the square
        for i in range(len(centers_concat)):
            x, y = centers_concat[i]
            r = radii_concat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        # No overlaps between circles
        for i in range(len(centers_concat)):
            for j in range(i + 1, len(centers_concat)):
                dx = centers_concat[i][0] - centers_concat[j][0]
                dy = centers_concat[i][1] - centers_concat[j][1]
                if np.sqrt(dx**2 + dy**2) < radii_concat[i] + radii_concat[j] - 1e-12:
                    return -1
        return 0

    # Objective: maximize the sum of radii
    def objective(x):
        centers_concat = x[:2 * len(centers)].reshape(len(centers), 2)
        radii_concat = x[2 * len(centers):]
        return -np.sum(radii_concat)  # Minimize negative sum to maximize the real sum

    # Bounds for x: [0, 1] for x-coordinates, [0, 1] for y-coordinates, and [0, 1] for radii
    bounds = []

    for i in range(len(centers)):
        bounds.append((0, 1))  # x_i
        bounds.append((0, 1))  # y_i
        bounds.append((0, 1))  # r_i

    # Define equality constraint
    def constraint(x):
        return constraint_func(x)

    # Constraint is equality (must be >= 0), so we use '>= 0'
    cons = ({'type': 'ineq', 'fun': constraint_func})

    # Run optimization
    result = opt.minimize(
        objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'max_iter': 500}
    )

    # Extract final centers and radii
    x_opt = result.x
    centers_opt = x_opt[:2 * len(centers)].reshape(len(centers), 2)
    radii_opt = x_opt[2 * len(centers):]

    # Compute sum of radii
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
