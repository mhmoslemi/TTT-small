def compute_voronoi_min_distance(centers):
    vor = Voronoi(centers)
    distances = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            if vor.vertices.shape[0] == 0:
                distances.append(0)

                        min_dist = distance
                distances.append(min_dist)
    return min(np.array(distances))
