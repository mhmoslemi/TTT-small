from scipy.optimize import minimize
from itertools import combinations

def run_packing():

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(r <= x)
            constraints.append(r <= 1.0 - x)

        return constraints

        return [(0, 1.0) for _ in radii]

        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP"
