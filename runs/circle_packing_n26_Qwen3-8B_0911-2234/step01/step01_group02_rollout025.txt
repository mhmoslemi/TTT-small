import numpy as np
from scipy.optimize import minimize

def run_packing():

    n = 26

    # Create a hexagonal packing with 5 rows and 5 columns
    # Start with 25 circles arranged in a hexagonal grid (base configuration)
    centers = []
    radii = []

    # Define hexagonal grid with 5 rows and 5 columns
    cols = 5
    rows = 5
    dx = 1.0 / cols
    dy = np.sqrt(3.0) / (2.0 * rows)

    for i in range(rows):
        for j in range(cols):
            x = j * dx
            y = i * dy
            if i % 2 == 1:
                x += dx / 2
            centers.append([x, y])

    # Add the 26th circle to the upper right corner as an initial point
    centers.append([0.95, 0.95])

    # Convert to NumPy arrays
    centers = np.array(centers)

    # Set an initial guess for radii: place each circle to fit the nearest boundary
    def initial_radii(centers):
        radii = []
        for x, y in centers:
            min_dist = np.inf
            min_dist = min(min_dist, x)
            min_dist = min(min_dist, 1.0 - x)
            min_dist = min(min_dist, y)
            min_dist = min(min_dist, 1.0 - y)
            radii.append(min_dist / 2)
        return np.array(radii)

    radii_initial = initial_radii(centers)

    # Optimization function to maximize the total area
    def objective(radii):
        return -np.sum(radii)  # Use negative for maximization

    # Constraints:
    # 1. Distance between any two circles must be ≥ r_i + r_j
    # 2. Each circle must stay inside the square: r_i ≤ distance to boundary
    # 3. Radii ≥ 0

    def constraint_distances(radii):
        # Ensure that circles do not overlap
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=-1))
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) >= -1e-8

    def constraint_boundary(radii):
        # Ensure that each circle is within the square
        dists_to_boundary = np.minimum(
            np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
            np.minimum(centers[:, 1], 1.0 - centers[:, 1])
        )
        return dists_to_boundary - radii >= -1e-8

    cons = [

        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Bounds for radii: non-negative
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Run the optimization
    result = minimize(
        fun=objective,
        x0=radii_initial,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )

    # Extract the result
    radii = result.x
    centers = centers[:n]

    # Return the result
    return (centers, radii, np.sum(radii))
