import scipy.optimize as opt

def run_packing():

    centers = []

        if i % 2 == 0:
            x = i / 2
            y = 0.5

            y = 0.5
        centers.append([x, y])

    def objective(radii):
        sum_radii = np.sum(radii)
        return -sum_radii  # Convert to minimization problem

        dist = np.zeros(len(centers))

                dist[i] = np.hypot(dx, dy)

        return radii - (dist - 1e-12)
    
    bounds = [(0, 1) for _ in range(n)]
    
    result = opt.minimize(
        objective,
        x0=np.full(n, 0.1),
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': constraint_min_dist}],
        bounds=bounds
    )
