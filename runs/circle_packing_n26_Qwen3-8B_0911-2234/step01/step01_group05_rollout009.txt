import numpy as np

def run_packing():

    # Define a hexagonal grid within the unit square [0, 1] x [0, 1]
    # This will be created in a staggered pattern with rows and columns

    centers = []
    # We will construct a hexagonal arrangement with rows and columns, and slightly shift
    # the positions on odd rows to create the staggered effect
    rows = 5
    cols = 5
    dx = 1.0 / (cols - 1)
    dy = np.sqrt(3.0) / rows / 2  # Spacing based on equilateral triangle side

    # Generate staggered grid
    for i in range(rows):
        for j in range(cols):
            x = j * dx
            y = i * dy
            if i % 2 == 1:  # Staggered rows
                x += dx / 2
            centers.append([x, y])

    # If we have not enough circles, add one more in the top-right to maximize density
    if len(centers) < n:
        # Example: Add a circle near the top-right corner of the square
        centers.append([0.95, 0.95])

    # Now compute radii based on the minimum distance to boundaries and other circles
    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        min_dist = np.inf

        # Check distance to boundaries
        min_dist = min(min_dist, x)
        min_dist = min(min_dist, 1.0 - x)
        min_dist = min(min_dist, y)
        min_dist = min(min_dist, 1.0 - y)

        # Check distance to other circles
        for j in range(len(centers)):
            if i == j:

            dist = np.sqrt((x - xj)**2 + (y - yj)**2)
            min_dist = min(min_dist, dist)

        # Assign radius as half the minimum distance
        r = min_dist / 2
        radii.append(r)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Calculate the sum of radii
    sum_radii = np.sum(radii)
