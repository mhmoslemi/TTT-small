def run_packing():

    n = 26

    # We'll use a hexagonal grid with 5 rows and 6 columns, allowing for dynamic placement
    
    # Set grid parameters
    rows = 5
    cols = 6
    dx = 1.0 / (cols - 1)
    dy = np.sqrt(3.0) / (2.0 * (rows - 1))
    
    centers = []
    
    # Generate the hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * dx
            y = i * dy
            if i % 2 == 1:
                x += dx / 2
            if i % 2 == 1 and j % 2 == 1:
                x -= dx / 2
            centers.append([x, y])
    
    # Filter and select the 26 most optimal positions to maximize radii

    def compute_radius(x, y, centers, dx, dy):

        # Check boundaries
        min_dist = min(min_dist, x)
        min_dist = min(min_dist, 1.0 - x)
        min_dist = min(min_dist, y)
        min_dist = min(min_dist, 1.0 - y)
        
        # Check other circles
        for other_x, other_y in centers:
            if [x, y] == [other_x, other_y]:
                continue
            dist = np.hypot(x - other_x, y - other_y)
            min_dist = min(min_dist, dist)
        
        return min_dist / 2

    # Compute radii and filter the 26 most optimal positions
    temp_centers = []
