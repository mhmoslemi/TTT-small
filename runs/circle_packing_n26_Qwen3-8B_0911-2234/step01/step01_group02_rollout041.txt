from scipy.optimize import minimize

def run_packing():

    centers = []

        angle = 2 * np.pi * i / n

        x = 0.5 + radius * np.cos(angle)
        y = 0.5 + radius * np.sin(angle)
        centers.append([x, y])

    # Define objective function
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint_min_dist(x):
        centers_np = np.array(x).reshape((n, 2))
        radii = np.array([x[i * 2 + 1] for i in range(n)])
        distances = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i != j:
                    distances[i, j] = np.hypot(centers_np[i, 0] - centers_np[j, 0],
                                             centers_np[i, 1] - centers_np[j, 1])
        min_dist = np.min(distances, axis=1)
        min_dist = np.min(np.vstack((min_dist, np.array(radii) + np.array(radii))), axis=0)
        return min_dist - 1e-6

    # Flatten centers and radii
    flat_centers = np.array(centers).flatten()
    flat_radii = np.array([0.05 for _ in range(n)])

    def boundary_constraint(x):

        bottom = y_coords - radii

        return np.concatenate((left, right, bottom, top))

    result = minimize(
