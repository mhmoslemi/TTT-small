import numpy as np
from scipy.optimize import minimize

def run_packing():

    # Parameters for grid
    rows = 5
    cols = 6
    dx = 1.0 / cols
    dy = np.sqrt(3.0) / (2.0 * rows)

    # Initialize centers in a hexagonal grid
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * dx
            y = i * dy
            if i % 2 == 1:
                x += dx / 2
            centers.append([x, y])

    # Add extra circle as needed (this is adjusted in the optimization)
    centers.append([0.5, 0.98])  # start with a heuristic placement

    # Initial guess for radii: based on boundary distance
    radii = []
    for x, y in centers:
        min_dist = min(x, 1.0 - x, y, 1.0 - y)
        radii.append(min_dist / 2)

    # Define the objective function for optimization
    def objective(radii):
        return -np.sum(radii)  # maximize sum of radii

    # Define constraints: circle must not overlap with another
    def constraint_overlap(i, j, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.hypot(x1 - x2, y1 - y2)
        return dist - radii[i] - radii[j]

    # Build optimization problem
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            def cons(vars):
                idx = 0
                radii = vars[idx:]
                return constraint_overlap(i, j, radii)
            constraints.append(cons)

    # Solve the optimization problem
    bounds = [(0.0, 1.0) for _ in range(len(centers))]  # all radii nonnegative
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    if not result.success:
        raise ValueError("Optimization failed with message: " + result.message)

    radii = result.x
    centers = np.array(centers)
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
