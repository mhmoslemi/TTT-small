import numpy as np
import scipy.optimize as opt
import math

def run_packing():

    # Hexagonal grid with 5 rows: 6, 5, 6, 5, 6
    num_rows = 5
    max_cols = 6
    centers = []
    radii = []

        if row % 2 == 0:
            cols = max_cols
        else:
            cols = max_cols - 1
        for col in range(cols):
            x = col + 0.5 * (row % 2)
            y = row * math.sqrt(3) / 2
            centers.append([x, y])

    # Convert to numpy arrays
    centers = np.array(centers)

    def initial_radius(x, y):
        dist_to_edge = min(x, 1 - x, y, 1 - y)
        return dist_to_edge

    radii = np.array([initial_radius(x, y) for x, y in centers])

    # Define optimization variables

    r = radii

    # Objective function: maximize sum of radii
    def objective(vars):

        new_r = vars[centers.shape[0] * 2:]
        return -np.sum(new_r)  # Negative for minimization

    # Constraints: circles within the unit square

        new_r = vars[centers.shape[0] * 2:]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
