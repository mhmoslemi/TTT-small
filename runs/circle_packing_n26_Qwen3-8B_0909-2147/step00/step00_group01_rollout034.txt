import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess: grid of circles in a 5x6 grid
    num_circles = 26
    grid = np.linspace(0.1, 0.9, 5)
    x_coords = np.repeat(grid, 6)
    y_coords = np.tile(grid, 6)
    centers = np.column_stack((x_coords, y_coords))
    radii = np.full(num_circles, 0.05)

    # Define the cost function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints
    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0  # No overlap

    def constraint_boundary(radii, centers):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return x - radii[i] + 1e-8
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e8:
                return y - radii[i] + 1e-8
        return 0  # All within square

    # Define bounds: radii must be nonnegative
    bounds = [(0, 1) for _ in range(num_circles)]

    # Initial guess for radii
    initial_guess = np.full(num_circles, 0.05)

    # Define the constraints in a form suitable for scipy
    def constraints(radii):
        overlap = constraint_overlap(radii, centers)
        boundary = constraint_boundary(radii, centers)
        return [overlap, boundary]

    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": constraint_overlap, "args": (centers,)},
                     {"type": "ineq", "fun": constraint_boundary, "args": (centers,)}]
    )

    # Extract results
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
