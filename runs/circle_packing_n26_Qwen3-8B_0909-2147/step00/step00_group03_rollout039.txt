import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():

    MAX_ITERATIONS = 100
    INITIAL_RADIUS = 0.05

    def create_initial_hexagonal_grid():
        """Create an initial aligned hexagonal grid in the unit square."""
        # We'll use a diagonal layout with staggered rows
        centers = []
        for row in range(4):
            for col in range(6):
                x = col / 6.0
                y = (row + (col % 2)) / 4.0
                centers.append([x, y])
        return np.array(centers)

    def optimize_radii(centers):
        """Optimize radii to maximize sum under non-overlapping and unit square constraints."""
        num_circles = len(centers)

        x0 = np.full(num_circles, INITIAL_RADIUS)

            return -np.sum(radii)  # Minimize negative sum to maximize sum

        def constraint_overlap(i, j, radii):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            dist = np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
            return dist - radii[i] - radii[j]

        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': lambda radii, i=i, j=j: constraint_overlap(i, j, radii)})

        for i in range(num_circles):
            def constraint_boundary(i, radii):
                x, y = centers[i]
                return x - radii[i] + 1e-12  # Ensure not to go out of bounds
            constraints.append({'type': 'ineq', 'fun': lambda radii, i=i: constraint_boundary(i, radii)})

                x, y = centers[i]
                return x + radii[i] - 1e-12
            constraints.append({'type': 'ineq', 'fun': lambda radii, i=i: constraint_boundary_right(i, radii)})

                x, y = centers[i]
                return y - radii[i] + 1e-12
            constraints.append({'type': 'ineq', 'fun': lambda radii, i=i: constraint_boundary_top(i, radii)})

                return y + radii[i] - 1e-12

            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': MAX_ITERATIONS, 'ftol': 1e-12, 'eps': 1e-12}

            raise RuntimeError("Optimization failed")

                return False, f"Circle {i} has negative radius {radii[i]}"

            if (x - r < -1e-12 or x + r > 1 + 1e-12
