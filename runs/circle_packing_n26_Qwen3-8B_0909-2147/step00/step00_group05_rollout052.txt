import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    def total_radius(centers, radii):
        return np.sum(radii)

    def is_valid(centers, radii):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Initial arrangement: hexagonal + grid-based
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Start with a hexagonal-like grid structure
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col / 5.0 + 0.1
        y = row / 5.0 + 0.1
        centers[i, 0] = x
        centers[i, 1] = y

    # Run SPSA to optimize centers and radii
    def objective(x):
        # x has shape (52,)
        centers_flat = x[:52].reshape((26, 2))
        radii = x[52:]
        return -total_radius(centers_flat, radii)

    def constraint(x):
        centers_flat = x[:52].reshape((26, 2))
        radii = x[52:]
        return 1 if is_valid(centers_flat, radii) else -np.inf

    # Bounds for centers (between 0 and 1)
    center_bounds = [(0, 1) for _ in range(52)]
    # Bounds for radii (non-negative)
    radius_bounds = [(0, 1) for _ in range(26)]

    # Combine bounds
    all_bounds = center_bounds + radius_bounds

    # Optimization
    x0 = np.concatenate((centers.flatten(), np.full(26, 0.1)))
    result = minimize(
        objective,
        x0,
        bounds=all_bounds,
        constraints={"type": "ineq", "fun": constraint}
    )

    centers = result.x[:52].reshape((26, 2))
    radii = result.x[52:]

    sum_radii = total_radius(centers, radii)
    return centers, radii, sum_radii
