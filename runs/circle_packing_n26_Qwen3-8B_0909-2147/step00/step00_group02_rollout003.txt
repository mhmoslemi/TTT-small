from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    n_rows = 5

    radii = np.zeros(n_rows * n_cols)
    
    for i in range(n_rows):

            x = offset + i * spacing
            y = offset + j * spacing
            centers[i * n_cols + j] = [x, y]
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative to maximize
    
    # Constraints: no overlapping between circles
    def constraint_func(radii):

                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-8

    result = minimize(

        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_func},
