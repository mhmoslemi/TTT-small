            total += radii[i] ** 2 * np.pi
        return -total  # Minimize negative => Maximize

    def constraint_circles(params):
        params = params.reshape(-1, 2)
        centers = params[:, :-1]
        radii = params[:, -1]
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
