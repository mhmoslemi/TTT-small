import numpy as np

def run_packing():
    # Define the hexagonal packing grid
    num_rows = 6
    num_cols = [5, 6, 6, 6, 6, 5]
    spacing = 0.2
    row_height = np.sqrt(3) * spacing
    
    centers = []
    radii = []

    for row in range(num_rows):
        for col in range(num_cols[row]):
            x = col * spacing
            y = row * row_height
            # Ensure circles are within the square and adjust spacing to avoid overflow
            # Calculate radii based on minimal spacing
            dist_to_left = x
            dist_to_right = 1 - x
            dist_to_top = y
            dist_to_bottom = 1 - y
            radius = min(dist_to_left, dist_to_right, dist_to_top, dist_to_bottom)
            # Adjust inner circles to tamely expand spacing
            if row == 0 or row == num_rows - 1:
                radius = min(radius, spacing / 2)
            else:
                radius = min(radius, spacing)
            centers.append([x, y])
            radii.append(radius)

    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
