import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt

def run_packing():
    # Initial grid-based configuration of 25 circles with one extra circle at center
    def initial_grid(n):
        grid = []
        for i in range(n):
            for j in range(n):
                x = i / (n - 1)
                y = j / (n - 1)
                grid.append((x, y))
        return np.array(grid)
    
    # Helper function to compute the minimal distance between two circles
    def distance(c1, c2):
        return np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2)
    
    # Objective function to maximize sum of radii under constraints
    def objective(radii, centers):
        return -np.sum(radii)  # We minimize negative sum to maximize the actual sum
    
    # Constraints for non-overlapping and boundary conditions
    def constraints(radii, centers):
        constraints = []
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
            for j in range(i+1, n):
                dist = distance(centers[i], centers[j])
                if dist < radii[i] + radii[j] - 1e-6:
                    return False
        return True
    
    # Define centers as a grid of 25 points
    centers = initial_grid(5)
    # Add 1 extra circle at the center
    centers = np.vstack((centers, [0.5, 0.5]))

    radii = np.full(n, 0.05)
    
    # Optimization setup
    bounds = [(0, 0.5) for _ in range(n)]
