import numpy as np

def run_packing():
    # Grid spacing
    grid_spacing = 0.15
    grid_size = int(1 / grid_spacing)
    
    # Generate grid points
    grid_points = np.array([[x * grid_spacing, y * grid_spacing] for y in range(grid_size) for x in range(grid_size)])
    
    # Initialize centers and radii
    centers = []
    radii = []
    used = np.zeros(grid_size * grid_size, dtype=bool)
    
    # Greedy algorithm to place circles
    for i, point in enumerate(grid_points):
        x, y = point
        # Check if this point is usable
        if used[i]:
            continue
        
        # Try to find the largest possible radius for this circle
        max_radius = 0
        for j in range(i + 1, len(grid_points)):
            dx = grid_points[j, 0] - x
            dy = grid_points[j, 1] - y
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < 1e-10:
                continue
            max_radius = max(max_radius, dist / 2)
        
        # Check for boundary constraints
        max_edge_dist = min(x, 1 - x, y, 1 - y)
        max_radius = min(max_radius, max_edge_dist)
        
        # Add this circle
        centers.append([x, y])
        radii.append(max_radius)
        used[i] = True
    
    # Cut off to only 26 circles
    centers = centers[:26]
    radii = radii[:26]
    
    # Calculate sum of radii
    sum_radii = sum(radii)
    
    return np.array(centers), np.array(radii), sum_radii
