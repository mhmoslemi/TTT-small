import numpy as np
from scipy.optimize import minimize

def run_packing():

    # We will adjust the radii and positions using optimization.

    # Initial grid-based location for each circle
    rows = 5
    cols = 5
    points = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5
            y = i + 0.5
            points.append((x, y))
    centers = np.array(points)

    # Initial guess for radii: small values to ensure valid packing
    radii = np.full(26, 0.05)

    # Define cost function to maximize total sum of radii (negative for minimization)
    def cost(radii):
        return -np.sum(radii)

    # Define constraints
    constraints = []

    # Constraint: all circles are within the unit square
    def in_square(radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    # Constraint: no overlapping circles
    def no_overlap(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Use radius constraints (non-negativity)
    bounds = [(0, 1) for _ in range(26)]

    # Combine and optimize
    result = minimize(
        fun=cost,
        x0=radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: in_square(r)},
            {'type': 'ineq', 'fun': lambda r: no_overlap(r)},
        ]
    )
