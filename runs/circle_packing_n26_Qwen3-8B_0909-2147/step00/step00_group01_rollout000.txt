from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult

def run_packing():

    def hex_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        x = np.linspace(0.5, 0.5, cols)
        y = np.linspace(0.5, 0.5, rows)
        grid = np.empty((n, 2))
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if r % 2 == 0:
                    grid[idx] = [x[c], y[r]]
                else:
                    grid[idx] = [x[c - 1], y[r]]
                idx += 1
        return grid

    centers = hex_grid(26)
    radii = np.full(26, 0.1)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints

    # Constraint 1: Circle within bounds
    def bound_constraint(x):
        x, r = x[:2], x[2:]
        for i in range(26):
            if x[i * 2] - r[i] < -1e-12 or x[i * 2] + r[i] > 1 + 1e-12:
                return -1
            if x[i * 2 + 1] - r[i] < -1e-12 or x[i * 2 + 1] + r[i] > 1 + 1e-12:
                return -1
        return 0

    # Constraint 2: Circle i and j not overlapping
    def non_overlap_constraints(x):
        x = x.reshape(26, 2)
        r = x[:, 2:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)

                    return -1
        return 0

    # Combine constraints
    def all_constraints(x):
        if bound_constraint(x) < 0:
            return -1
        if non_overlap_constraints(x) < 0:
            return -1
        return 0

    bounds = []

    result = minimize(
