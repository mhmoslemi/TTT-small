import math
from scipy.optimize import minimize

def run_packing():

    centers = []
    radii = []

    # Define spacing parameters
    h = 1 / (math.sqrt(3) + 0.1)
    v = 1 / (math.sqrt(3) + 0.1)

    for i in range(5):
        for j in range(5 + (i % 2)):
            x = 0.5 + (j - 2.5) * h
            y = 0.5 + (i - 2) * v
            centers.append([x, y])
            radii.append(0.05)  # Initial guess for radii

    centers = np.array(centers)
    radii = np.array(radii)

        if (center[0] - r < -1e-12 or center[0] + r > 1 + 1e-12 or
            center[1] - r < -1e-12 or center[1] + r > 1 + 1e-12):
            return -1.0
        return 0.0

    # Constraints: Ensure no overlaps between circles
    def overlap_constraint(radii, i, j):
        r1 = radii[i]
        r2 = radii[j]
        dx = centers[i, 0] - centers[j, 0]
