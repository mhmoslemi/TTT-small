import numpy as np
from scipy.optimize import minimize

def run_packing():

    def hexagonal_grid(n_rows, n_cols, spacing):
        centers = []
        for i in range(n_rows):
            for j in range(n_cols):
                x = j * spacing
                y = i * spacing * np.sqrt(3) / 2
                centers.append((x, y))
        return np.array(centers)

    # Start with a 4x4 hexagonal grid (16 circles)
    grid = hexagonal_grid(4, 4, 0.2)
    num_large_circles = grid.shape[0]
    
    # Additional circles to fill gaps
    extra_circles = 10
    
    # Initial guess for the radii
    initial_radii = np.full(num_large_circles + extra_circles, 0.05)

    def generate_initial_positions(num, bounds=(0, 1)):
        return np.random.uniform(bounds[0], bounds[1], (num, 2))
    
    extra_positions = generate_initial_positions(extra_circles)
    
    centers = np.vstack([grid, extra_positions])
    radii = initial_radii
    
    # Define the optimization objective: maximize total radii
    def objective(x, centers, radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define the constraints: non-overlapping and within square
    def constraint(x, centers, radii):
        # x is a flat array of x and y values for each constraint
        # Flatten centers and radii
        num_circles = centers.shape[0]
        x_flat = x.reshape(-1, 2)
        r_flat = radii

        # Distance between all pairs
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = x_flat[i, 0] - x_flat[j, 0]
                dy = x_flat[i, 1] - x_flat[j, 1]

                if dist < r_flat[i] + r_flat[j] - 1e-8:
                    return -1  # Constraint violated
        return 0  # Constraint satisfied

    # Define bounds for all positions
    bounds = [(0, 1) for _ in range(2 * (num_large_circles + extra_circles))]

    # Optimization: adjust positions to maximize radii
    result = minimize(
        objective,
        x0=np.concatenate([centers.flatten(), radii]),
        bounds=bounds,
        constraints=constraint,
        method="SLSQP",
