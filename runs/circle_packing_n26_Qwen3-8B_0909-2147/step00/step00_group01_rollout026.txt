import scipy.optimize as optimize

def run_packing():

    n = 26
    max_rows = 5  # Number of rows in hexagonal packing
    num_cols = (n + max_rows - 1) // max_rows  # Number of columns per row

    # Base hexagonal packing grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):

        col = i % num_cols

        row_y = row * (np.sqrt(3) / 2) * 0.5
        col_x = col * 0.5
        centers[i, 0] = col_x
        centers[i, 1] = row_y
        # Initially estimate radii as small, then optimize
        radii[i] = 0.1  # Initial guess

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

        # Ensure radii are non-negative
        return radii >= 0

    def constraint2(radii, centers):

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    result = optimize.minimize(
