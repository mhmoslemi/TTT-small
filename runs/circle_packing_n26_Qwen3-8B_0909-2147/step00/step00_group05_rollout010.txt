import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
import random

def run_packing():
    # Initial hexagonal packing with grid spacing
    def hexagonal_grid(n, max_radius):
        grid = []
        for i in range(n):
            row = []
            for j in range(n):
                x = i / n
                y = j / n
                r = np.sqrt((x - 0.5)**2 + (y - 0.5)**2) * max_radius
                row.append((x, y, r))
            grid.extend(row)
        return grid

    # Generate initial configuration
    n = 26
    max_initial_radius = 0.1
    grid = hexagonal_grid(n, max_initial_radius)
    
    # Extract initial positions and radii
    centers = np.array([p[0:2] for p in grid])
    radii = np.array([p[2] for p in grid])
    
    # Constraint function: returns (constraint values, Jacobian)
    def constraint_func(x):
        # x is a flat array: [x1, y1, r1, x2, y2, r2, ...]
        n = len(radii)
        x_flat = x.reshape(-1, 2)
        r_flat = x[-n:]
        
        # Check boundaries
        for i in range(n):
            xi, yi = x_flat[i]
            ri = r_flat[i]
            if xi - ri < -1e-12 or xi + ri > 1 + 1e-12 or yi - ri < -1e-12 or yi + ri > 1 + 1e-12:
                return np.array([1.0])
        
        # Check circle-circle overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_flat[i, 0] - x_flat[j, 0]
                dy = x_flat[i, 1] - x_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_flat[i] + r_flat[j] - 1e-12:
                    return np.array([1.0])
        
        return np.array([])

    # Objective function to maximize sum of radii
    def objective(x):
        x_flat = x.reshape(-1, 2)
        r_flat = x[-n:]
        return -np.sum(r_flat)  # Minimize negative sum to maximize sum

    # Initial guess: x_flat + r_flat
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Bounds: x must be in [0,1], y must be in [0,1], radii must be >= 0
    bounds = []

        bounds.extend([(0, 1), (0, 1)])  # x and y for each circle

    # Set up optimization problem
    cons = {'type': 'ineq', 'fun': constraint_func}
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)

    # Extract the optimal solution
    x_opt = result.x

    radii_opt = x_opt[2*n:]

    # Validate the packing and return
    return centers_opt, radii_opt, np.sum(radii_opt)
