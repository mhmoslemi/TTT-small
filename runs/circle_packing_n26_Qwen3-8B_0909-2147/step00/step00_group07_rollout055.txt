import scipy.optimize as opt

def run_packing():

    row_count = 5
    col_count = 5

    radii = np.zeros(n_circles)
    
    # Define a hexagonal packing in a grid
    for i in range(row_count):
        for j in range(col_count):
            idx = i * col_count + j

                x = j / (col_count - 1)

                centers[idx] = [x, y]

    # Define the objective function
    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum
    
    # Define constraints: distance between centers must be >= sum of radii
    def distance_constraints(centers, radii):
        n = centers.shape[0]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                def constraint(radii):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx**2 + dy**2)
