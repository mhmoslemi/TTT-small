    centers = np.array([

    ])

    # Initial guess for radii
    radii = np.ones(26) * 0.1  # Start with small radii

    # Define objective function: total sum of radii
    def objective(r):
        return -np.sum(r)  # Minimize negative sum to maximize total sum

    # Define constraints:
    def constraint_func(r, centers):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)

        return constraint_func(r, centers)

    result = minimize(

        method='SLSQP',
