import numpy as np
import scipy.optimize as optimize
from itertools import combinations

def run_packing():
    # Initial placement: Hexagonal grid with 5 rows and 5 columns (but adapt to square)
    # We will use 5 rows and 6 columns (total 30 circles), but we will select the best 26 circles
    # to maximize the sum of radii.

    # Let's attempt to create a hexagonal grid with 5 rows and 6 columns.
    rows = 5
    cols = 6
    centers = np.zeros((rows * cols, 2))
    radii = np.zeros(rows * cols)

    for i in range(rows):
        for j in range(cols):
            x = j * (1.0 / (cols - 1)) if cols > 1 else 0.5
            y = i * (1.0 / (rows - 1)) if rows > 1 else 0.5
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.1  # Initial guess

    # Define objective function to maximize total radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints: non-overlapping and within bounds
    def constraint_non_overlap(radii, centers):
        n = len(radii)
        for i, j in combinations(range(n), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                return dist - (radii[i] + radii[j] - 1e-8)
        return 0

    # Define bounds for radii (non-negative)
    bounds = [(0, 1.0) for _ in range(len(radii))]

    # Optimize
    res = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': constraint_non_overlap,
            'args': (centers,)
        }
    )

    # After optimization, select the best 26 circles based on the optimized radii
    # Sort circles by descending radius and select top 26
    to_keep = np.argsort(res['x'])[::-1][:26]
    centers = centers[to_keep]
    radii = res['x'][to_keep]
    radii_sum = np.sum(radii)

    return centers, radii, radii_sum
