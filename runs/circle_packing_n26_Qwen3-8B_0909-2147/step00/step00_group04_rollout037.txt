from scipy.optimize import minimize
from scipy.optimize import Bounds

def distance(x, y):
    """Compute Euclidean distance between two points."""
    return np.sqrt((x[0] - y[0])**2 + (x[1] - y[1])**2)

def compute_max_radii(centers, boundaries, num_circles):
    """Estimate the maximum radius for each circle based on distance to other circles and boundaries."""
    radii = np.full(num_circles, 0.0)
    for i in range(num_circles):

        radii[i] = min(min_dist / 2, dist_to_boundary / 2)
    return radii

def compute_total_area(radii):

    return np.sum(np.pi * radii ** 2)
