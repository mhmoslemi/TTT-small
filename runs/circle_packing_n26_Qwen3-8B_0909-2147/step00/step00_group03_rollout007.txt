import scipy.optimize as opt

    n = 26

    # To maximize density, we place the most crowded rows in the center

    # Define positions in a hexagonal layout
    centers = []
    radii = []

    # Define spacing between circle centers
    spacing = 0.5

    # First 2 rows (top and bottom)
    for i in range(2):
        for j in range(3):
            x = i * spacing + 0.5
            y = j * spacing + 0.5
            centers.append([x, y])
            radii.append(0.1)  # initial radius for all

    # Next 6 rows (middle)
    for i in range(6):
        for j in range(4):
            x = i * spacing + 0.5
            y = j * spacing + 0.5
            centers.append([x, y])
            radii.append(0.1)

    # Last 2 rows (bottom)
    for i in range(2):
        for j in range(3):
            x = i * spacing + 0.5
            y = j * spacing + 0.5
            centers.append([x, y])
            radii.append(0.1)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define radius optimization using scipy.optimize
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii, so we minimize the negative

    def constraint_circle_bounds(radii):

        for i in range(len(radii)):

            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or

    def constraint_non_overlap(radii):
