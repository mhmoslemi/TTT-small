import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define a hexagonal grid of 26 circles in a 5x6 grid (with some adjusted for corners)
    # Initial grid in a coordinate system that fits into [0, 1] x [0, 1]
    num_rows = 5
    num_cols = 6
    grid = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j / (num_cols - 1)
            y = i / (num_rows - 1)
            grid.append([x, y])

    centers = np.array(grid)
    radii = np.ones(26) * 0.05  # initial guess for radius

    def objective(radii):
        return -np.sum(radii)  # maximize sum of radii

    def constraint_func(radii):
        n = len(radii)
        # Check if any radius is negative
        if np.any(radii < 0):
            return -np.inf

        # Check boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf

        # Check pairwise non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return np.sum(radii)

    # Perform optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=[(0, 1)] * len(radii),
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': constraint_func}
        ]
    )

    # Extract results
    optimal_radii = result.x
    optimal_centers = centers

    # Verify and return
    assert np.all(optimal_radii >= 0), "Some radii are negative"

    assert np.all(np.abs(optimal_centers - 1) <= 1e-12), "Some centers not within [0,1]^2"

    total_sum = np.sum(optimal_radii)
    return (optimal_centers, optimal_radii, total_sum)
