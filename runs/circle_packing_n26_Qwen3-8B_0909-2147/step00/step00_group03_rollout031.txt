import numpy as np
import scipy.optimize as optimize

def run_packing():

    # We will use a hexagonal close packing pattern in a unit square

    # Generate 26 circles in a hexagonal packing in a unit square
    # We can approximate the hexagonal packing with rows
    num_rows = 6
    max_per_row = 5  # First row has 5 circles, then alternating offset
    row_height = np.sqrt(3) / 3  # Height of each row in hexagonal packing

    centers = []
    radii = []

    # Generate the rows
    for row_idx in range(num_rows):
        row_y = row_idx * row_height
        if row_idx % 2 == 0:
            # Even rows: left-aligned
            for i in range(max_per_row):
                x = i / (max_per_row - 1) if max_per_row > 1 else 0.5
                centers.append([x, row_y])
                radii.append(0.1)
        else:
            # Odd rows: right-aligned with offset
            for i in range(max_per_row):
                x = (i + 0.5) / (max_per_row - 1) if max_per_row > 1 else 0.5
                centers.append([x, row_y])
                radii.append(0.1)

    # Convert to arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Step 2: Define a function to maximize the sum of radii
    # with constraints: circles inside square, no overlaps

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint1(x):
        # All circles must be inside the unit square
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = x[len(centers):][i] if len(x) > len(centers) else 0
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf
        return 0

    def constraint2(x):
        # No two circles can overlap
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = x[len(centers):][i] if len(x) > len(centers) else 0
                r_j = x[len(centers):][j] if len(x) > len(centers) else 0
                if dist < r_i + r_j - 1e-10:
                    return -np.inf
        return 0

    # We flatten the array: first n values are centers, then radii
    n = len(centers)
    lb = np.concatenate([np.zeros(n * 2), np.zeros(n)])  # Lower bounds
    ub = np.concatenate([np.ones(n * 2), np.ones(n)])    # Upper bounds

    # Initial guess: centers are fixed, radii start at 0.1
    x0 = np.concatenate([centers.flatten(), np.full(n, 0.1)])

    # Set up the optimization
    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]

    # Use Sequential Quadratic Programming (SQP)
    result = optimize.minimize(

        x0,
        bounds=zip(lb, ub),
        constraints=cons,
        method='SLSQP',
        tol=1e-8,
        options={'maxiter': 1000}
    )

    # Extract centers and radii from the result
    centers_result = result.x[:n * 2].reshape(n, 2)
    radii_result = result.x[n * 2:]

    # Return the result
    return centers_result, radii_result, np.sum(radii_result)
