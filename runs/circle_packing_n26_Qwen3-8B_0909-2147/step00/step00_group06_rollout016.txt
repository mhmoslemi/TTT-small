from scipy.optimize import minimize

    grid = np.zeros((n_circles, 2))
    for i in range(n_circles):
        row = i // 5
        col = i % 5
        grid[i, 0] = col / 5.0
        grid[i, 1] = row / 5.0

    def in_square(centers, radii):

        return True
