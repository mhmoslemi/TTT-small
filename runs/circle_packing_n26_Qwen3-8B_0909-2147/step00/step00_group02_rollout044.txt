import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Basic setup: we'll try to optimize a packing with 26 circles
    n = 26
    centers = np.random.rand(n, 2)  # Initial random positions
    radii = np.random.rand(n)  # Initial random radii (for optimization)
    radii = np.clip(radii, a_min=0.0, a_max=1.0)  # Clip to [0, 1]

    # Optimization bounds
    bounds = Bounds(
        lb=np.concatenate((np.zeros(n * 2), np.zeros(n))),
        ub=np.concatenate((np.ones(n * 2), np.ones(n)))
    )

    # Objective: maximize sum of radii
    def objective(x):
        # x has shape (n*2 + n) = 3n
        # First n*2 entries are centers, next n entries are radii
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)

        total_sum = np.sum(radii_flat)
        return -total_sum  # Minimize negative sum to maximize

    # Constraints: circle stays within the unit square
    def constraints(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        # Ensure circle is within square
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return False
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
        return True

    # Define constraints for overlap between circles
    def overlap_constraints(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = centers_flat[i]
                x_j, y_j = centers_flat[j]
                r_i = radii_flat[i]
                r_j = radii_flat[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if dist < r_i + r_j - 1e-12:
                    return False
        return True

    # Optimization problem
    x0 = np.concatenate((centers.flatten(), radii))

    # We need to solve:
    # - minimize -sum(radii)
    # - s.t. circles are within [0,1]^2
    # - and no two circles overlap

    # Convert constraints to a list of functions for scipy.optimize
    cons = [
        {'type': 'ineq', 'fun': lambda x: constraints(x)},
        {'type': 'ineq', 'fun': lambda x: overlap_constraints(x)}
    ]

    # Run the optimization
    result = opt.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    # Extract the result

    centers_result = x_result[:n * 2].reshape(n, 2)

    valid, msg = validate_packing(centers_result, radii_result)
    if not valid:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
