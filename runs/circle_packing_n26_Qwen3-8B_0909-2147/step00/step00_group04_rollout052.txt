import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Constants
    n = 26
    max_iter = 100
    tol = 1e-5

    # Define initial hexagonal grid positions
    def hexagonal_grid(n):
        row = int(np.sqrt(n))
        col = n - row
        x = np.linspace(0.5, 0.5, row)
        y = np.linspace(0.5, 0.5, col)
        grid = np.column_stack((x.repeat(col), y.repeat(row)))
        return grid

    # Initial grid (reduced for 26 circles)
    centers = hexagonal_grid(n)
    radii = np.full(n, 0.01)  # Initial small radius

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(i, j):
        def cons(x):
            x_i = centers[i]
            x_j = centers[j]
            r_i = x[n + i]
            r_j = x[n + j]
            return np.sqrt(np.sum((x_i - x_j) ** 2)) - (r_i + r_j)
        return cons

    def constraint_boundary(i):
        def cons(x):
            x_i = centers[i]
            r_i = x[n + i]
            return np.sqrt(np.sum((x_i - np.array([0.5, 0.5])) ** 2)) - r_i
        return cons

    # Set up optimization problem
    x0 = np.append(centers.flatten(), radii)
    bounds = [(-1, 1) for _ in range(2 * n)]  # Coordinates are in [0,1]
    bounds += [(0, 1) for _ in range(n)]  # Radii must be nonnegative

    # Constraints: no overlaps, no boundary violations
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Run optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': tol}
    )

    # Extract final centers and radii
    final_centers = result.x[:2 * n].reshape(n, 2)
    final_radii = result.x[2 * n:]

    # Validate packing
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(msg)

    return final_centers, final_radii, np.sum(final_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
