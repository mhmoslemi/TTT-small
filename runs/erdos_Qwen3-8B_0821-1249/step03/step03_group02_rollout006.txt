import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Generate multiple initial configurations
    n_points = 100
    initial_h_list = []
    for _ in range(5):
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=50, replace=False)
        h[idx] = 1.0
        initial_h_list.append(h)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    best_n_points = n_points
    
    start_time = time.time()
    
    for initial_h in initial_h_list:
        # Normalize to satisfy integral constraint
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
        
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            overlap = correlate(h, 1 - h, mode='full') * (2.0 / n_points)
            return np.max(overlap)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = opt.minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        current_h = result.x
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
