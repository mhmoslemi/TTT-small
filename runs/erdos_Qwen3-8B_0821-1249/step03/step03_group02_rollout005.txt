import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # L-BFGS-B optimization
    def lbfgs_optimize(start_h):
        start_time = time.time()
        result = opt.minimize(
            objective,
            start_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        elapsed_time = time.time() - start_time
        return result.x, objective(result.x), elapsed_time
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    def eval_func(individual):
        h = np.clip(individual, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return (np.max(overlap),)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxSBX, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    def ga_optimize(start_h):
        start_time = time.time()
        pop = toolbox.population(n=50)
        for ind in pop:
            ind[:] = start_h[:]
        fitnesses = toolbox.map(toolbox.evaluate, pop)
        for ind, fit in zip(pop, fitnesses):
            ind.fitness.values = fit
        for _ in range(100):
            offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
            fits = toolbox.map(toolbox.evaluate, offspring)
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            pop = toolbox.select(pop + offspring, len(pop))
            if time.time() - start_time > budget_s - 2:
                break
        best_ind = min(pop, key=lambda x: x.fitness.values[0])
        return np.array(best_ind), best_ind.fitness.values[0], time.time() - start_time
    
    # Run L-BFGS-B optimization
    best_h, best_c5, elapsed_time = lbfgs_optimize(initial_h)
    
    # Use remaining time for genetic algorithm optimization
    while elapsed_time < budget_s - 1:
        best_h, best_c5, elapsed_time = ga_optimize(best_h)
    
    return (best_h, best_c5, n_points)
