import time
import random

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

        return np.sum(h) - required_sum

    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm with tournament selection
    population_size = 20
    generations = 50
    mutation_rate = 0.2
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Create new population
        population = [best_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            # Apply mutation
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                population[i][idx] = 1 - population[i][idx]
            # Normalize
            population[i] = population[i] / np.sum(population[i]) * required_sum
            
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select best individuals
        elite_indices = np.argsort(fitness)[:population_size//2]
        best_population = [population[i] for i in elite_indices]
        
        # Tournament selection
        new_population = []
        for _ in range(population_size//2):
            a, b = np.random.choice(elite_indices, 2, replace=False)
            if fitness[a] < fitness[b]:
                new_population.append(population[a])
            else:
                new_population.append(population[b])
        
        # Add some random perturbations
        for i in range(len(new_population)):
