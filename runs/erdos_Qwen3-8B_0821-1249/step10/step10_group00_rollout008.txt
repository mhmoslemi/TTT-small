import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 15
    mutation_rate = 0.15

    population = [initial_h.copy() for _ in range(population_size)]
    population = [h / np.sum(h) * required_sum for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break

        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        selected = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(selected[i])
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = 1 - offspring[i][j]
        
        # Normalize offspring
        offspring = [h / np.sum(h) * required_sum for h in offspring]
        
        # Replace population
        population = offspring
    
    # Refinement with local optimization
    while time.time() - start_time < budget_s - 1:

        h = best_h.copy()

            method='SLSQP',
