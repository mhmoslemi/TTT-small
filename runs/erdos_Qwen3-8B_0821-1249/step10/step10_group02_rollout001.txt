import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Mutation-based optimization with guided exploration
    for _ in range(30):
        if time.time() - start_time > budget_s - 1:
            break
        h = best_h.copy()
        # Guided perturbation: more significant changes in high overlap regions
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        high_overlap_indices = np.argsort(overlap)[-int(n_points / 4):]
        perturbation = np.zeros(n_points)
        perturbation[high_overlap_indices] = np.random.normal(0, 0.05, len(high_overlap_indices))
        h += perturbation
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Local refinement with directional mutations
    for _ in range(15):
        if time.time() - start_time > budget_s - 1:
            break
        idx = np.random.randint(n_points)
        h = best_h.copy()
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    # Final optimization step
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 50}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
