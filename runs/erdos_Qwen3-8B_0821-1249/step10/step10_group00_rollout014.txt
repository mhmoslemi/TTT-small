import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()

    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 10
    mutation_rate = 0.1
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    for indiv in population:
        indiv = indiv / np.sum(indiv) * required_sum
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = [objective(indiv) for indiv in population]
        # Select top performers
        top_indices = np.argsort(fitness)[:pop_size//2]
        top_individuals = [population[i] for i in top_indices]
        # Crossover
        new_population = []
        for _ in range(pop_size//2):
            parent1 = random.choice(top_individuals)
            parent2 = random.choice(top_individuals)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            new_population.append(child)
        # Mutation
        for i in range(len(new_population)):

                    new_population[i][j] = np.random.uniform(0, 1)
            new_population[i] = new_population[i] / np.sum(new_population[i]) * required_sum

    for indiv in population:
        c5 = objective(indiv)
        if c5 < best_c5:
            best_h = indiv
            best_c5 = c5

    while time.time() - start_time < budget_s - 1:

        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]

        )

    return (best_h, best_c5, n_points)
