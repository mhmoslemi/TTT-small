import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Define initial configurations
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    # Stage 1: Multi-start with random perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Create a new random starting point
        h = np.random.rand(n_points)
        h = h / np.sum(h) * required_sum
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 150}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Stage 2: Simulated annealing for global exploration
    temp = 1.0
    cooling_rate = 0.995
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current solution
        perturb = np.random.normal(0, 0.1, n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        
        # Accept the new solution with probability based on temperature
        delta = current_c5 - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            best_h = h
            best_c5 = current_c5
        
        temp *= cooling_rate
    
    # Stage 3: Local refinement with gradient descent
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
