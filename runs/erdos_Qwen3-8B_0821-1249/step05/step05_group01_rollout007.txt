import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Create genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    def eval_func(individual):
        h = np.array(individual)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap),
    
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=1/n_points)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start optimization with genetic algorithm
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        pop = toolbox.population(n=50)
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        pop, logbook = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=20)
        best_ind = tools.selBest(pop, 1)[0]
        current_c5 = best_ind.fitness.values[0]
        if current_c5 < best_c5:
            best_h = np.array(best_ind)
            best_c5 = current_c5
    
    # Refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
