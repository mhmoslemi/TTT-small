import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    random.seed(seed)

    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    
    for gen in range(generations):

            break

        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * required_sum for h in population]
        population.append(best_h)  # Add best solution to population
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Apply mutation
        for i in range(population_size):

        for h in population:
