import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    generations = 50

    population = [np.random.rand(n_points) for _ in range(pop_size)]
    population = [h / np.sum(h) * required_sum for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break

        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        for _ in range(pop_size // 2):
            # Select parents
            parent1 = population[np.random.choice(pop_size)]
            parent2 = population[np.random.choice(pop_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
            
            # Mutation

                if np.random.rand() < mutation_rate:

                    child2[i] = np.random.rand()
            
            # Normalize and enforce constraints

            child2 = child2 / np.sum(child2) * required_sum
            
            new_population.append(child1)
            new_population.append(child2)
        
        # Replace old population
        population = new_population
    
    # Refinement with local optimization
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
