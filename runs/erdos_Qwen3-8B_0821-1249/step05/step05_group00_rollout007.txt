import scipy.optimize as opt
from scipy.signal import correlate
import time

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    elite_size = 2
    start_time = time.time()
    
    # Initial population
    def create_individual():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * required_sum
        return h
    
    population = [create_individual() for _ in range(population_size)]
    
    # Evaluate population
    def evaluate_population(pop):
        return np.array([objective(ind) for ind in pop])
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size - elite_size):
            contestants = np.random.choice(range(population_size), size=5, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parents.append(population[winner])
        
        # Crossover
        new_population = elites.copy()
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i+1]
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
                child2 = np.concatenate((p2[:crossover_point], p1[crossover_point:]))
                new_population.append(child1)
                new_population.append(child2)
            else:
                new_population.append(p1)
                new_population.append(p2)
        
        # Mutation
        for i in range(len(new_population)):
            ind = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    ind[j] = np.random.rand()
                    ind = ind / np.sum(ind) * required_sum
            new_population[i] = ind
        
        # Replace population
        population = new_population
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    # Refinement with local optimization
    while time.time() - start_time < budget_s - 1:

        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        if current_c5 < best_c5:

            best_c5 = current_c5

            objective,
            best_h,
            method='SLSQP',

            options={'ftol': 1e-8, 'maxiter': 80}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
