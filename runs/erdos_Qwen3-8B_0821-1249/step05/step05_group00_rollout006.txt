import scipy.optimize as opt
from scipy.signal import correlate
import time

    # Start with a higher resolution for better initial approximation
    n_points = 120
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # Initial guess from provided initial_h_values scaled to new resolution
    initial_h = np.array(initial_h_values)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)

    initial_h = initial_h / np.sum(initial_h) * required_sum
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start optimization with diverse perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Random perturbation with varying amplitude
        perturb = np.random.normal(0, 0.25, n_points)
        h = initial_h + perturb
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Adaptive refinement with greedy and local steps
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
