import scipy.optimize as opt
from scipy.signal import correlate
import time

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def evaluate_individual(h):
        return objective(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    
    # Initialize population
    population = np.random.rand(population_size, n_points)
    population = np.clip(population, 0, 1)
    population = population / np.sum(population, axis=1, keepdims=True) * required_sum
    
    best_h = initial_h.copy()
    best_c5 = evaluate_individual(best_h)
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate_individual(ind) for ind in population])
        
        # Select top performers
        elite_indices = np.argsort(fitness)[:population_size//2]
        elite_population = population[elite_indices]
        
        # Create new population
        new_population = []
        
        # Add elites
        new_population.extend(elite_population)
        
        # Generate new individuals through crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elite_population, size=2, replace=False)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

                mutation = np.random.normal(0, 0.1, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize

            new_population.append(child)
        
        population = np.array(new_population)
        
        # Update best solution
        current_best = population[np.argmin(fitness)]
        current_c5 = evaluate_individual(current_best)

        method='SLSQP',

    )
