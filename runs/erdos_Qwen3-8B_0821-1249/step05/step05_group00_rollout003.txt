        def __init__(self, n_particles, n_dim, bounds, objective, max_iter):
            self.n_particles = n_particles
            self.n_dim = n_dim
            self.bounds = bounds
            self.objective = objective
            self.max_iter = max_iter
            self.particles = np.random.uniform(self.bounds[:, 0], self.bounds[:, 1], (n_particles, n_dim))
            self.velocities = np.random.uniform(-1, 1, (n_particles, n_dim))
            self.pbest = self.particles.copy()
            self.pbest_obj = np.array([self.objective(p) for p in self.particles])
            self.gbest = self.pbest[np.argmin(self.pbest_obj)]
            self.gbest_obj = np.min(self.pbest_obj)
        
        def update(self):
            for i in range(self.n_particles):
                r1, r2 = np.random.rand(self.n_dim), np.random.rand(self.n_dim)

                    self.pbest_obj[i] = current_obj
            self.gbest = self.pbest[np.argmin(self.pbest_obj)]
            self.gbest_obj = np.min(self.pbest_obj)
        
        def run(self):
            for _ in range(self.max_iter):
                self.update()
            return self.gbest, self.gbest_obj
