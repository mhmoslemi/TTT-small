        population = [p / np.sum(p) * required_sum for p in population]
        fitness = [objective(p) for p in population]
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)
            winner = contestants[np.argmin([fitness[i] for i in contestants])]
            selected.append(population[winner])
        
        # Crossover
        crossed = []
        for i in range(0, population_size, 2):

                crossed.append(child1)
                crossed.append(child2)
            else:
                crossed.append(selected[i])

            h_mut = h_mut / np.sum(h_mut) * required_sum

        # Evaluate and select

        best_idx = np.argmin(fitness_mut)
        best_h = mutated[best_idx]
        best_c5 = fitness_mut[best_idx]
