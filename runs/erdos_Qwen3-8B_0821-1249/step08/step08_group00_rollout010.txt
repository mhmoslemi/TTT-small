import scipy.optimize as opt
import time
import random

    random.seed(seed)

    n_points = len(initial_h)

    def objective(h):

        h = h / np.sum(h) * required_sum
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Adaptive gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    grad_momentum = np.zeros(n_points)
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        h = best_h.copy()
        h = h / np.sum(h) * required_sum
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlap)
        
        # Compute gradient of max_overlap with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += 1e-5
            h_perturbed = h_perturbed / np.sum(h_perturbed) * required_sum
            overlap_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full') * dx
            grad[i] = (np.max(overlap_perturbed) - max_overlap) / 1e-5
        
        # Update with momentum
        grad_momentum = momentum * grad_momentum + learning_rate * grad
        h_update = -grad_momentum
        h_new = h + h_update
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * required_sum
        
        # Accept update if it improves
        if objective(h_new) < max_overlap:
            best_h = h_new
            best_c5 = objective(best_h)
    
    # Refinement with local optimization
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        
        # Continue optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
