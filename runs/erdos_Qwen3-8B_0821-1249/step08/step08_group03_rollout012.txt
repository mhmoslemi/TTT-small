import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    random.seed(seed)

    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 100
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Tournament selection
        new_population = []
        for _ in range(population_size):
            # Select two random individuals
            a, b = np.random.choice(population_size, 2, replace=False)
            if fitness[a] < fitness[b]:
                parent = population[a]
            else:
                parent = population[b]
            
            # Mutate
            mutated = parent.copy()
            for i in range(n_points):
                if random.random() < mutation_rate:
                    mutated[i] = 1 - mutated[i]
            
            # Normalize
            mutated = mutated / np.sum(mutated) * required_sum
            new_population.append(mutated)
        
        population = new_population
    
    return (best_h, best_c5, n_points)
