import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with random perturbations and SLSQP refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate random perturbations
        perturbations = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_h = best_h + perturbations
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * required_sum
        
        # Optimize from perturbed point
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Local refinement with greedy perturbations and SLSQP
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
