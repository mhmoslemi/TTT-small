import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    random.seed(seed)

    initial_h = np.array(initial_h_values)

    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initialize population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Tournament selection
        new_population = []
        for _ in range(population_size):
            # Select two random individuals
            a, b = np.random.choice(population_size, 2, replace=False)
            if fitness[a] < fitness[b]:
                selected = population[a]
            else:
                selected = population[b]
            
            # Apply crossover
            if np.random.rand() < crossover_rate:
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([selected[:idx], population[best_idx][idx:]])
            else:
                child = selected.copy()
            
            # Apply mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = 1 - child[i]
            
            # Normalize
            child = child / np.sum(child) * required_sum
            new_population.append(child)
        
        population = new_population
    
    # Local refinement
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        overlap = correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(overlap)
        idx = max_idx % n_points
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h

        result = opt.minimize(

            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )

        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
