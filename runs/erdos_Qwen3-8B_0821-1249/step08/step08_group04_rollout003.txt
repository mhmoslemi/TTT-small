import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 20
    mutation_rate = 0.15
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Create population
        population = [best_h + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p / np.sum(p) * required_sum for p in population]
        
        # Evaluate fitness
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Tournament selection
        tournament_size = 5
        for _ in range(population_size // 2):
            candidates = np.random.choice(range(population_size), tournament_size, replace=False)
            best_candidate = np.argmin([fitness[i] for i in candidates])
            parent1 = population[candidates[best_candidate]]
            
            # Mutate
            mutation = np.random.normal(0, mutation_rate, n_points)
            child = parent1 + mutation
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * required_sum
            
            # Evaluate and replace
            child_c5 = objective(child)
            if child_c5 < best_c5:
                best_h = child
                best_c5 = child_c5
        
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
