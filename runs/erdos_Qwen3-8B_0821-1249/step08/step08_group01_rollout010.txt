import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    initial_h = np.array(initial_h_values)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start with diverse perturbations
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break

        h = initial_h.copy()
        h += np.random.uniform(-0.2, 0.2, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Genetic algorithm for global search
    def genetic_algorithm(population_size=20, generations=10, mutation_rate=0.1):
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * required_sum for h in population]
        for _ in range(generations):
            if time.time() - start_time > budget_s - 1:
                break
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            # Select parents
            parents = np.argsort(fitness)[:population_size//2]
            parents = [population[i] for i in parents]
            # Crossover
            offspring = []
            for _ in range(population_size - len(parents)):
                p1, p2 = random.sample(parents, 2)
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * required_sum
                offspring.append(child)
            population = parents + offspring
            # Mutation
            for i in range(len(population)):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    population[i] += mutation
                    population[i] = np.clip(population[i], 0, 1)
                    population[i] = population[i] / np.sum(population[i]) * required_sum
            # Update best
            current_best = min(population, key=objective)
            current_c5 = objective(current_best)
            if current_c5 < best_c5:
                best_h = current_best
                best_c5 = current_c5
    
    genetic_algorithm()
    
    # Refinement with guided local optimization
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum

            best_c5 = current_c5

            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
