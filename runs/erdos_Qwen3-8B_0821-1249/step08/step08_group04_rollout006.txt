import scipy.optimize as opt
from scipy.signal import correlate

import random

        return np.max(overlap)

    best_c5 = objective(best_h)

    population_size = 50
    generations = 15
    mutation_rate = 0.1
    crossover_rate = 0.8

        population = [np.clip(p, 0, 1) for p in population]

        # Evaluate fitness
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Crossover and mutation
        new_population = []

            if np.random.rand() < crossover_rate:

                parent2 = population[np.random.randint(population_size)]
                child = np.zeros(n_points)

                        child[j] = parent1[j]

                new_population[i] = new_population[i] / np.sum(new_population[i]) * required_sum
