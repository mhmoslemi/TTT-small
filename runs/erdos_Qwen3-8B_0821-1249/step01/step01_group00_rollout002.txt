import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
from genetic_algorithm import GeneticAlgorithm

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # Define objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    # Define constraint
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Genetic algorithm parameters
    ga_params = {
        'population_size': 50,
        'mutation_rate': 0.1,
        'crossover_rate': 0.8,
        'generations': 100,
        'tolerance': 1e-4
    }
    
    # Create genetic algorithm instance
    ga = GeneticAlgorithm(n_points, required_sum, objective, constraint_sum, ga_params)
    
    start_time = time.time()
    
    # Run genetic algorithm
    best_h, best_c5 = ga.run()
    
    # If time allows, perform local optimization
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
