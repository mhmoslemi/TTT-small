def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initialize with a smooth transition
    h = np.concatenate([np.linspace(0, 1, 32), np.ones(64), np.linspace(1, 0, 32)])
    h = np.clip(h, 0, 1)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        # Add penalty for rapid changes
        penalty = np.sum(np.abs(np.diff(h)))
        return c5_val + 1e-3 * penalty
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
