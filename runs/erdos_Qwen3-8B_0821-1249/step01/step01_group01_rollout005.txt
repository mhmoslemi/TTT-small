def run(seed=42, budget_s=10, **kwargs):

        initial_h = initial_h * (n_points / 2) / sum_h

        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        h = np.pad(h, (0, 1), mode='constant', constant_values=0)
        overlap = np.correlate(h, 1 - h, mode='full')

        return c5

        return np.sum(h) - n_points / 2

    cx_prob = 0.8
    mut_prob = 0.2
    n_gen = 50
    n_elite = 2

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

    # Initialize toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    def eval_func(individual):
        h = np.array(individual)
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        h = np.pad(h, (0, 1), mode='constant', constant_values=0)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5,

    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    start_time = time.time()
    population = toolbox.population(n=pop_size)

            ind.fitness.values = fit
        population = toolbox.select(offspring, k=len(population))
        if gen % 5 == 0:
            # Keep elite individuals
            elite = sorted(population, key=lambda x: x.fitness.values)[0:n_elite]
            population = elite + toolbox.population(n=pop_size - n_elite)
        if time.time() - start_time > budget_s:
            break

    # Refine best individual with gradient descent
    best_h = np.array([ind for ind in population if ind.fitness.values[0] == min([ind.fitness.values[0] for ind in population])][0])
    best_h = best_h.copy()

    best_c5 = objective(best_h)

        h = np.clip(h, 0, 1)

        h = np.pad(h, (0, 1), mode='constant', constant_values=0)
        overlap = np.correlate(h, 1 - h, mode='full')

        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus.reshape(-1)

            h_minus = h.copy()
            h_minus[i] -= 1e-5

            overlap_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(overlap_minus) * dx
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)

        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},

        jac=True,
        options={'ftol': 1e-9, 'maxiter': 50}
    )
    
    final_h = np.clip(result.x, 0, 1)
    final_c5 = objective(final_h)
    return (final_h, final_c5, n_points)
