        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    
    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            # Select parents
            parents = np.array(population)[np.argsort(fitness)[:population_size//2]]
            # Crossover
            offspring = []
            for _ in range(population_size//2):
                parent1 = random.choice(parents)
                parent2 = random.choice(parents)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                offspring.append(child)
            # Mutate
            for i in range(len(offspring)):
                for j in range(n_points):
                    if random.random() < mutation_rate:
                        offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)
            # Replace population
            population = np.array(offspring)
        best = min(population, key=objective)
        return best

    best_h = genetic_algorithm()

    start_time = time.time()

        method='SLSQP',
