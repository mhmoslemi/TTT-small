    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial optimization with SLSQP
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 50, 'ftol': 1e-6, 'disp': False}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm to explore the solution space
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 1
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] = np.clip(mutated[i] + np.random.normal(0, 0.1), 0, 1)
        return mutated
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for _ in range(generations):
        fitness = [objective(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        population = elite
        
        next_generation = []
        for _ in range(population_size - elite_size):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = crossover(parent1, parent2)
            child = mutate(child)

        population += next_generation

    best_c5_genetic = objective(best_genetic)
