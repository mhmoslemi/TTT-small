def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = objective(h)
    time_used = 0
    
    for _ in range(10):
        # Apply random perturbations
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0.0, 1.0)
        
        # Optimize
        result = optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-6,
            options={'maxiter': 50, 'ftol': 1e-6, 'disp': False}
        )
        
        # Update best solution
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Check time budget
        time_used += 5  # approximate time for each iteration
        if time_used >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
