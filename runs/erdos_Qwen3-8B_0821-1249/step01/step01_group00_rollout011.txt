import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Start with a random initial guess
    n_points = 100
    h_initial = np.random.uniform(0, 1, n_points)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    h_initial = h_initial / np.sum(h_initial) * required_sum
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = h_initial
    best_c5 = objective(best_h)
    
    for _ in range(10):
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
