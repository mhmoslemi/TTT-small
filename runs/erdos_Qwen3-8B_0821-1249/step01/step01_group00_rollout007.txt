import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

        return np.max(overlap)

        population_size = 30
        generations = 50
        mutation_rate = 0.1
        elite_size = 5
        
        def fitness(individual):
            return objective(individual)
        
        def crossover(parent1, parent2):
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            return child
        
        def mutate(individual):
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    individual[i] = np.random.uniform(0, 1)
            return individual
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [ind / np.sum(ind) * required_sum for ind in population]
        
        for generation in range(generations):
            scores = [fitness(ind) for ind in population]
            best_idx = np.argmin(scores)
            best_h = population[best_idx]
            
            if generation % 10 == 0:

            elite_indices = np.argsort(scores)[:elite_size]
            elites = [population[i] for i in elite_indices]
            
            # Generate new population
            new_population = elites[:]
            while len(new_population) < population_size:
                parent1, parent2 = random.choices(elites, k=2)
                child = crossover(parent1, parent2)

    start_time = time.time()
