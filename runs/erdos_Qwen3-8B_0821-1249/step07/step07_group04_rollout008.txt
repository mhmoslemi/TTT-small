import numpy as np
import scipy.signal as signal
import time
import random

    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess from provided initial_h_values

    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def fitness(h):
        return 1.0 / (1.0 + objective(h))
    
    # Initialize population
    population_size = 50
    population = [np.clip(np.random.rand(n_points), 0, 1) for _ in range(population_size)]
    population = [h / np.sum(h) * required_sum for h in population]
    
    best_h = population[0]
    best_fitness = fitness(best_h)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness_values = [fitness(h) for h in population]
        
        # Select parents
        parents = random.choices(population, weights=fitness_values, k=population_size)
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i + 1]
                child1 = np.zeros_like(parent1)
                child2 = np.zeros_like(parent2)
                for j in range(n_points):
                    if random.random() < 0.5:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                new_population.extend([child1, child2])
            else:
                new_population.append(parents[i])
        
        # Mutation
        for i in range(len(new_population)):
            h = new_population[i]
            for j in range(n_points):
                if random.random() < 0.1:
                    h[j] = np.clip(h[j] + np.random.uniform(-0.2, 0.2), 0, 1)
            new_population[i] = h / np.sum(h) * required_sum
        
        # Evaluate new population
        new_fitness_values = [fitness(h) for h in new_population]
        population = [h for _, h in sorted(zip(new_fitness_values, new_population), key=lambda x: x[0], reverse=True)[:population_size]]
        
        # Update best solution
        for h in population:
            current_fitness = fitness(h)
            if current_fitness > best_fitness:
                best_fitness = current_fitness
                best_h = h
                best_c5 = objective(h)
    
    return (best_h, best_c5, n_points)
