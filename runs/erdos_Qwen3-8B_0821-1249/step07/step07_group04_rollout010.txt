from scipy.signal import correlate

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * required_sum for h in population]
        
        for gen in range(generations):
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Select parents
            parents = np.array(population)[np.argsort(fitness)[:int(population_size/2)]]
            
            # Crossover
            offspring = []
            for _ in range(population_size - len(parents)):
                parent1, parent2 = random.sample(parents, 2)
                child = np.copy(parent1)
                for i in range(n_points):
                    if random.random() < crossover_rate:
                        child[i] = parent2[i]
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if random.random() < mutation_rate:
                        offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)
                offspring[i] = offspring[i] / np.sum(offspring[i]) * required_sum
            
            population = np.array(parents + offspring)

    best_h, best_c5 = genetic_algorithm()
