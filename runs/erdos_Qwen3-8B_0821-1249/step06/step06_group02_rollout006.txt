import scipy.optimize as opt
from scipy.signal import correlate
import time

def run(seed=42, budget_s=10, **kwargs):

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Genetic algorithm parameters
    population_size = 30
    generations = 150
    mutation_rate = 0.15
    elite_size = 5
    
    # Initialize population
    population = [np.clip(np.random.uniform(0, 1, n_points), 0, 1) for _ in range(population_size)]
    population = [h / np.sum(h) * required_sum for h in population]
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Randomly select parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            parent1 = population[parent1]
            parent2 = population[parent2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                child[mutation_indices] = np.random.uniform(0, 1, len(mutation_indices))
            
            # Normalize
            child = child / np.sum(child) * required_sum
            new_population.append(child)
        
        population = new_population
    
    # Refinement with local optimization
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        idx = np.random.randint(n_points)
        h[idx] = 1 - h[idx]
        h = h / np.sum(h) * required_sum
        current_c5 = objective(h)

            best_h = h
            best_c5 = current_c5
        
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 80}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
