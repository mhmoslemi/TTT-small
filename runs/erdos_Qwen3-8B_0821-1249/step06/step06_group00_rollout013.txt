import scipy.optimize as opt
from scipy.signal import correlate

import random

    cons = {'type': 'eq', 'fun': constraint_sum}

    pop_size = 20
    generations = 50

    crossover_rate = 0.8
    elite_size = 2
    
    start_time = time.time()

    best_c5 = objective(best_h)
    
    # Initialize population
    population = np.array([np.random.rand(n_points) for _ in range(pop_size)])
    population = np.array([h / np.sum(h) * required_sum for h in population])
    
    # Genetic algorithm
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(h) for h in population])
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = population[elite_indices]
        
        # Select parents
        parents = population[np.argsort(fitness)[:int(pop_size * crossover_rate)]]
        
        # Crossover
        offspring = []
        for _ in range(pop_size - elite_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.rand()
        
        # Combine population
        population = np.vstack((elite, offspring))
        population = np.array([h / np.sum(h) * required_sum for h in population])
        
        # Update best solution

        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    # Refinement with local optimization
    while time.time() - start_time < budget_s - 1:
