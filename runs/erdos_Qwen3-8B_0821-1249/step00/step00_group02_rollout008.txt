    def objective(h_values):
        h = h_values.copy()
        h = np.concatenate((h, h[:n_points//2]))  # Extend for full correlation

        h_initial,
        method='SLSQP',

        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    # Return best solution found within the time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_initial, objective(h_initial), n_points)

        return (result.x, objective(result.x), n_points)
