import numpy as np
import random
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0  # Initial simple step function
    h = h * (n_points / 2)  # Ensure integral is 1
    
    def objective(x):
        h_values = x.reshape(n_points)
        h_values = np.clip(h_values, 0, 1)
        overlap = correlate(h_values, 1 - h_values, mode='full') * dx
        return np.max(overlap)
    
    def perturb(x):
        new_x = x.copy()
        idx = np.random.randint(n_points)
        new_x[idx] = np.random.uniform(0, 1)
        return new_x
    
    best_h = h
    best_c5 = objective(best_h)
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        perturbed = perturb(best_h)
        current_c5 = objective(perturbed)
        if current_c5 < best_c5:
            best_h = perturbed
            best_c5 = current_c5
            print(f"New best C5: {best_c5}")
    
    return best_h, best_c5, n_points
