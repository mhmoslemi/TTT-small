import time

    target_c5 = 0.380876

    start_time = time.time()

    def evaluate(h):
        h = np.array(h)
        shifted_h = np.roll(h, shift=1, axis=0)
        overlap = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child

    population_size = 50
    population = [mutate(initial_h) for _ in range(population_size)]

    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        if best_c5 < target_c5:
            best_h = population[best_idx]
        population = [crossover(population[best_idx], np.random.choice(population)) for _ in range(population_size)]
