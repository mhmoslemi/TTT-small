import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
    
    def compute_c5(h_values):
        h_array = np.array(h_values)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()
    
    for _ in range(100):
        if time.time() - time_start > budget_s:
            break
        # Optimization step
        obj = cp.Minimize(cp.Max([cp.sum(h[i] * (1 - h[(i + k) % n_points]) for k in range(1, n_points)])))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            if prob.status == "optimal":
                h_values = h.value
                c5 = compute_c5(h_values)
                if c5 < best_c5:
                    best_h = h_values
                    best_c5 = c5
        except:
            pass
    
    return (best_h, best_c5, n_points)
