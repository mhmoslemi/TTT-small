import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial guess: uniform distribution
    h[::n_points//4] = 1.0
    h[::n_points//2] = 0.5
    h[::n_points//4] = 0.0
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    h = h * (1 / total) * 2.0  # Ensure integral is 1
    
    def compute_c5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = h * (1 / total) * 2.0  # Ensure integral is 1
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        total = np.sum(x) * dx
        x = x * (1 / total) * 2.0
        corr = np.correlate(x, 1 - x, mode="full")
        return np.max(corr * dx)
    
    # Optimization
    x0 = h.copy()
    x0 = x0 * (n_points / 2.0)
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        method='SLSQP',
        options={'ftol': 1e-6, 'maxiter': 100},
        tol=1e-6
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, compute_c5(h), n_points)
    
    best_h = result.x
    best_h = best_h.reshape(n_points)
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * dx
    best_h = best_h * (1 / total) * 2.0
    
    return (best_h, compute_c5(best_h), n_points)
