import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraints},
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
