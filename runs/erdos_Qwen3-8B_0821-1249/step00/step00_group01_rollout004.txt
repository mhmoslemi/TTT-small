import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_values = initial_h_values.copy()
    
    def objective(h_values):
        h = h_values.reshape(-1)
        h = np.clip(h, 0, 1)
        h = np.vstack((h, np.roll(h, 1))).T  # Add shifted h for correlation
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[1:-1]  # Remove overlapping edge cases
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_values, np.inf, n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
