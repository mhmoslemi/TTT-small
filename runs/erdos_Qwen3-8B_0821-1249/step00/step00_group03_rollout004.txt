import scipy.optimize as optimize

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    target_sum = n_points / 2
    if abs(sum_h - target_sum) > 1e-6:
        initial_h = (initial_h / sum_h) * target_sum

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) > 1e-6:
            return float('inf')
        h = h.reshape(-1, 1)
        ones_minus_h = 1 - h
        corr = np.correlate(h.flatten(), ones_minus_h.flatten(), mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-6}
    )

    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
