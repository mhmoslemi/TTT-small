import scipy.optimize as opt
import time

def run(seed=42, budget_s=10, **kwargs):

    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    options = {'maxiter': 100, 'ftol': 1e-6, 'disp': False}

    result = opt.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options=options
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h, np.max(np.correlate(h, 1 - h, mode="full")) * dx, n_points)
    
    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
