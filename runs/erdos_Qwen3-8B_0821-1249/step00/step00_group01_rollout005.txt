import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    target_sum = n_points / 2
    if not np.isclose(sum_h, target_sum):
        initial_h = initial_h * target_sum / sum_h

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, target_sum):
            return float('inf')
        shift_correlations = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(shift_correlations) * dx
        return max_overlap

    def optimize_h(h0):
        result = optimize.minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )
        return result.x, result.fun

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s - 0.5:
            break
        h_start = np.random.uniform(0, 1, n_points)
        h_start = h_start * target_sum / np.sum(h_start)
        h_opt, c5 = optimize_h(h_start)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    return best_h, best_c5, n_points
