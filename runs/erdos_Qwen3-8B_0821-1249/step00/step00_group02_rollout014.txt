import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

initial_h_values = np.load('initial_h.npy')  # Assuming initial construction is saved
n_points = len(initial_h_values)

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Use a small number of samples for speed
    n_points = 100  # Optimal balance between speed and accuracy
    h = np.random.rand(n_points) * 2 - 1  # Initial guess
    h = np.clip((h + 1) / 2, 0, 1)  # Scale to [0,1]
    total_area = np.sum(h) * (2.0 / n_points)
    h = h * (1.0 / (total_area))  # Normalize to satisfy the integral constraint
    
    def objective(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        total_area = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / (total_area))  # Normalize
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Use a local optimizer to refine the solution
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    # Use a genetic algorithm for global optimization
    from deap import base, creator, tools, algorithms
    
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("idx", np.random.randint, 0, n_points)
    toolbox.register("val", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.val, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    def evaluate(individual):
        h = np.array(individual)
        total_area = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / (total_area))
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5,
    
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxTwoPoint)

    stats.register("min", np.min)

    best_individual = tools.selBest(population, k=1)[0]
    best_h = np.array(best_individual)
    best_c5 = best_individual.fitness.values[0]
    
    return best_h, best_c5, n_points
