import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Initial guess with low values
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        one_minus_h = 1 - h
        # Compute the correlation
        corr = np.correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2

    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    # Optimization
    result = optimize.minimize(
        objective,
        h.flatten(),
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="L-BFGS-B",
        options={"maxiter": 100}
    )

    h_opt = result.x.reshape(n_points)
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
