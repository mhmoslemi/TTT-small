import numpy as np
import scipy.signal as signal
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = (initial_h * n_points / 2) / sum_h
    h_values = initial_h.copy()
    best_c5 = np.inf
    best_h = h_values.copy()
    start_time = time.time()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            h = (h * n_points / 2) / sum_h
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def crossover(parent1, parent2):
        if np.random.rand() < 0.9:
            idx = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:idx], parent2[idx:]])
        else:
            child = np.random.rand(n_points)
        child = (child * n_points / 2) / np.sum(child)
        return child
    
    def mutate(h):
        idx = np.random.randint(n_points)
        h[idx] = np.random.rand()
        h = (h * n_points / 2) / np.sum(h)
        return h
    
    def tournament_selection(population):
        candidates = np.random.choice(population, size=5, replace=False)
        return candidates[np.argmin([evaluate(ind) for ind in candidates])]
    
    population = [h_values.copy() for _ in range(10)]
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(len(population) // 2):
            p1 = tournament_selection(population)
            p2 = tournament_selection(population)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        current_c5 = evaluate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
    
    return best_h, best_c5, n_points
