import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    random.seed(seed)

    initial_h = np.array(initial_h_values)

    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 20
    mutation_rate = 0.1
    elite_size = 2
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i].copy() for i in elite_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            
            # Normalize
            child = child / np.sum(child) * required_sum
            offspring.append(child)
        
        # Create new population
        population = elites + offspring
        
        # Local refinement of best solution
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
