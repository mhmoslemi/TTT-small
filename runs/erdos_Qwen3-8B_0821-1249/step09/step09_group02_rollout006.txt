import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    
    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population / np.sum(population, axis=1, keepdims=True) * required_sum
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break

        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        selected = population[np.argsort(fitness)[:population_size//2]]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            child = np.minimum(parent1, parent2)
            child += np.random.uniform(-0.1, 0.1, n_points)
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * required_sum
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                offspring[i][idx] = 1 - offspring[i][idx]
                offspring[i] = offspring[i] / np.sum(offspring[i]) * required_sum
        
        # Combine and replace
        population = np.vstack([selected, offspring])
        population = np.unique(population, axis=0)
        population = population[:population_size]
    
    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()

        h[idx] = 1 - h[idx]

            method='SLSQP',
            bounds=bounds,
            constraints=cons,
