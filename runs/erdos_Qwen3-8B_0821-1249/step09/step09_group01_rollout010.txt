import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial construction
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 3
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * required_sum for h in population]
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_population = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite_population.copy()
        
        # Crossover and mutation
        for _ in range(population_size - elite_size):
            # Select parents
            parent1, parent2 = np.random.choice(elite_population, 2, replace=False)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            
            # Normalize
            child = child / np.sum(child) * required_sum
            new_population.append(child)
        
        population = new_population
    
    # Evaluate final population
    fitness = [objective(h) for h in population]
    best_h = population[np.argmin(fitness)]
    best_c5 = np.min(fitness)
    
    # Refine with local optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 50}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
