import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the provided initial construction
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(elite)
            parent2 = random.choice(elite)
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, int(n_points * 0.05), replace=False)
                child[mutation_indices] = np.random.uniform(0, 1, len(mutation_indices))
            
            # Normalize
            child = child / np.sum(child) * required_sum
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        population = new_population
    
    # Find the best solution
    fitness = [objective(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    return (best_h, best_c5, n_points)
