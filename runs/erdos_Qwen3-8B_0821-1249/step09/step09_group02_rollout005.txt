import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start optimization with random perturbations and genetic algorithm
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate random initial population with perturbations
        perturbations = np.random.uniform(-0.1, 0.1, (10, n_points))
        population = initial_h.copy() + perturbations
        population = np.clip(population, 0, 1)
        population = population / np.sum(population, axis=1, keepdims=True) * required_sum
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_pso_h = population[best_idx]
        best_pso_c5 = fitness[best_idx]
        
        # Refine with SLSQP
        result = opt.minimize(
            objective,
            best_pso_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Local refinement with genetic algorithm and local optimization
    while time.time() - start_time < budget_s - 1:
        # Genetic algorithm step
        population = np.random.uniform(0, 1, (10, n_points))
        population = population / np.sum(population, axis=1, keepdims=True) * required_sum
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_ga_h = population[best_idx]
        best_ga_c5 = fitness[best_idx]
        
        # Refine with SLSQP
        result = opt.minimize(
            objective,
            best_ga_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
