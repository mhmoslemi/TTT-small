import numpy as np
import scipy.optimize as opt
import time
from scipy.signal import correlate
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the provided initial construction
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm with local refinement
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate offspring through mutation and crossover
        parents = [best_h.copy() for _ in range(5)]
        offspring = []
        for p1 in parents:
            for p2 in parents:
                child = np.copy(p1)
                # Crossover
                idx = np.random.choice(n_points, size=n_points//2, replace=False)
                child[idx] = p2[idx]
                # Mutation
                for i in idx:
                    if np.random.rand() < 0.15:
                        child[i] = 1 - child[i]
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * required_sum
                offspring.append(child)
        
        # Evaluate offspring
        for h in offspring:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
