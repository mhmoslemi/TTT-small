import numpy as np
import scipy.optimize as opt
import time
from scipy.signal import correlate

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Multi-start optimization with random mutations
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate random mutations
        mutations = np.random.uniform(-0.3, 0.3, (10, n_points))
        population = initial_h.copy() + mutations
        population = np.clip(population, 0, 1)
        population = population / np.sum(population, axis=1, keepdims=True) * required_sum
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_pso_h = population[best_idx]
        best_pso_c5 = fitness[best_idx]
        
        # Refine with SLSQP
        result = opt.minimize(
            objective,
            best_pso_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Local refinement with simulated annealing
    temp = 1.0
    while temp > 1e-5 and time.time() - start_time < budget_s - 1:
        for _ in range(10):
            h = best_h.copy()
            idx = np.random.randint(n_points)
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * required_sum
            new_c5 = objective(h)
            if np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                best_h = h
                best_c5 = new_c5
        temp *= 0.95
    
    return (best_h, best_c5, n_points)
