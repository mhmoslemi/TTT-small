    for _ in range(generations):

        fitness = np.array([objective(p) for p in population])

        tournament_size = 5
        parents = []
        for _ in range(population_size):
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)

        offspring = []

            child = np.clip(child, 0, 1)

            offspring.append(child)
        
        # Combine population and select elites
        new_population = population + offspring
        fitness_new = np.array([objective(p) for p in new_population])
