import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial construction
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * required_sum for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parents.append(population[winner])
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            alpha = np.random.rand(n_points)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = alpha * parent2 + (1 - alpha) * parent1
            offspring.append(child1)
            offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                perturbation = np.random.normal(0, 0.05, n_points)
                offspring[i] += perturbation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] / np.sum(offspring[i]) * required_sum
        
        # Replace population
        population = offspring
        
        # Keep best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    # Final local optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
