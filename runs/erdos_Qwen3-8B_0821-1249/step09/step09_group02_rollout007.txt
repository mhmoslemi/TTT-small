import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

    random.seed(seed)

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initial population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population / np.sum(population, axis=1, keepdims=True) * required_sum
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        selected = population[np.argsort(fitness)][:int(population_size/2)]
        
        # Crossover
        offspring = []
        for _ in range(population_size - len(selected)):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < crossover_rate:
                child = np.copy(parent1)
                mask = np.random.rand(n_points) < 0.5
                child[mask] = parent2[mask]
            else:
                child = np.random.uniform(0, 1, n_points)
            child = child / np.sum(child) * required_sum
            offspring.append(child)
        population = np.vstack((selected, offspring))
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                population[i] = np.clip(population[i] + mutation, 0, 1)
                population[i] = population[i] / np.sum(population[i]) * required_sum
    
    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        # Greedy perturbation
        h = best_h.copy()
        idx = np.random.randint(n_points)

        result = opt.minimize(

            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
