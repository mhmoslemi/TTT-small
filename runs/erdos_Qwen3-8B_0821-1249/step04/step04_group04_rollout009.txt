import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a finer grid than the initial solution
    n_points = 120
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # Construct a better initial guess using a sinusoidal pattern
    def initial_guess(x):
        return 0.5 + 0.3 * np.sin(2 * np.pi * x / 2 - np.pi / 2)
    
    initial_h = np.array([initial_guess(i * dx) for i in range(n_points)])
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.sum(initial_h) * required_sum
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Multi-start optimization with guided perturbations
    for _ in range(6):
        if time.time() - start_time > budget_s - 2:
            break
        # Create a perturbed version of the current best solution
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * required_sum
        
        # Run optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        if objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
    
    # Additional refinement
    for _ in range(4):
        if time.time() - start_time > budget_s - 1:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        if objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
