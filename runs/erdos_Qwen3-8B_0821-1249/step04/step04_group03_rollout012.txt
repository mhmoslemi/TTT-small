import random

        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

        return np.sum(h) - required_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def fitness(h):
        return objective(h)
    
    def evolve(population, n_parents, n_offsprings):
        # Sort by fitness
        ranked = sorted(enumerate(population), key=lambda x: fitness(x[1]))
        parents = [x[1] for x in ranked[:n_parents]]
        
        # Crossover
        offspring = []
        for _ in range(n_offsprings):
            p1, p2 = random.sample(parents, 2)
            child = np.copy(p1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = p2[i]
            offspring.append(child)

                    child[i] = 1 - child[i]

        return offspring
