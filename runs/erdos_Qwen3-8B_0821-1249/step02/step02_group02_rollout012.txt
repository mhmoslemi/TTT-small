def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    def normalize(h_flat):
        h = h_flat.copy()
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (n_points / 2) / total
    
    def mutate(h_flat):
        h = h_flat.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [normalize(p) for p in population]
    
    best_h = initial_h_values
    best_c5 = evaluate(initial_h_values)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate(p) for p in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = normalize(child)
            new_population.append(child)
        population = new_population
        # Update best solution
        for h in population:
            c5_val = evaluate(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h
    
    return (best_h, best_c5, n_points)
