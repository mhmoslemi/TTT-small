import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    pop_size = 20
    gen_size = 50
    cxpb = 0.5
    mutpb = 0.1
    n_elites = 2
    
    # Setup DEAP
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
    
    def eval_func(individual):
        h = np.clip(individual, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap),
    
    def mut_gaussian(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] += np.random.normal(0, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual,
    
    def cx_uniform(ind1, ind2):
        ind1 = np.copy(ind1)
        ind2 = np.copy(ind2)
        for i in range(len(ind1)):
            if np.random.random() < 0.5:
                ind1[i], ind2[i] = ind2[i], ind1[i]
        return ind1, ind2
    
    # Initialize population
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.rand)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", cx_uniform)
    toolbox.register("mutate", mut_gaussian, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize population
    pop = toolbox.population(n=pop_size)
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    for gen in range(gen_size):
        offspring = algorithms.varAnd(pop, toolbox, cxpb, mutpb)
        fits = toolbox.map(toolbox.evaluate, offspring)
        offspring = list(map(lambda ind, fit: ind.fitness.values, offspring, fits))
        pop = toolbox.select(offspring, k=len(pop))
        elites = sorted(pop, key=lambda x: x.fitness.values)[0:n_elites]
        pop = elites + toolbox.population(n=len(pop) - n_elites)
        
        # Check best
        current_best = min(pop, key=lambda x: x.fitness.values)
        current_c5 = current_best.fitness.values[0]
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
    
    # Local optimization
    def objective_local(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * required_sum
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - required_sum}
    
    result = opt.minimize(
        objective_local,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    best_h = result.x
    best_c5 = objective_local(best_h)
    
    return (best_h, best_c5, n_points)
