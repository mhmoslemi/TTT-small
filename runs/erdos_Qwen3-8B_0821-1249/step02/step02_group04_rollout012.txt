def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 64
    dx = 2.0 / n_points
    h = initial_h_values[:n_points].copy()
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    return (best_h, best_c5, n_points)
