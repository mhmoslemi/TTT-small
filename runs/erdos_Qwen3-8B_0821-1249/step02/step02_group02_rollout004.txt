    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")

        return c5_val

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h
    best_c5 = objective(h)
    
    for _ in range(5):  # 5 different starting points
        # Perturb the initial guess

            best_c5 = current_c5
