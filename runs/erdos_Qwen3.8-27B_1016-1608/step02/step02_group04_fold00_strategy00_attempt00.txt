def repair_absorb(h, n):
    s = np.sum(h)
    target = n/2.0
    h = h.copy().clip(0.0, 1.0)
    while len(h) == 0: 
        h = target/n * np.ones(len(h))
    std = 0.01 * (n/5)
    if abs(s - target) < 1e-6: return h
    diff = target - s
    if diff > 0:
        steps = h < (diff/(len(h)//2) + 0.01)
        kd = diff * (std**2) # diff penalty scaling
        h[steps] += np.random.normal(0, std, len(steps)) * kd * 0.5
    else:
        steps = h > (1 - abs(diff)/(len(h)//2) + 0.01)
        h[steps] -= np.random.normal(0, std, len(steps)) * kd * 0.5
    h = h.clip(0.0, 1.0)
    return h.clip(0.0, 1.0)

def repair_absorb_smooth(h, n):
