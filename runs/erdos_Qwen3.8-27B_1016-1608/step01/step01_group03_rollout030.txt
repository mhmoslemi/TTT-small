    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair(h, target, max_iter=500):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_constr(x):
    return np.sum(x) - _SUM_TARGET

def slsqp_optimize(h, n, tgt, dx, maxiter=200, ftol=1e-13):
    global _SUM_TARGET
    _SUM_TARGET = tgt
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constr}]
    
    def obj(x):
        h_clip = np.clip(x, 0.0, 1.0)
        return compute_c5(h_clip, dx)
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        h_new = repair(r.x, tgt)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h, compute_c5(h, dx)

def smooth_max_grad(h, T, n):
    """Compute gradient of smooth max of correlations."""
    corr = get_corr(h)
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_vals = np.exp(T * shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return np.zeros(n)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    v = 1.0 - h
    half_n = n - 1
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        s = k_idx - half_n
        if s >= 0:
            idx = n - s
            if idx > 0:
                grad[0:idx] += w * v[s:n]
                grad[s:n] -= w * h[0:idx]
        else:
            idx = n + s
            if idx > 0:
                grad[-s:n] += w * v[0:idx]
                grad[0:idx] -= w * h[-s:n]
    return T * grad

    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

import numpy as np
from scipy.optimize import minimize
import time

_SUM_TARGET = 0.0
_OBJ_DX = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair(h, target, max_iter=500):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_constr(x):
    return np.sum(x) - _SUM_TARGET

def obj_func(x):
    h_clip = np.clip(x, 0.0, 1.0)
    return compute_c5(h_clip, _OBJ_DX)

def slsqp_optimize(h, n, tgt, dx, maxiter=200, ftol=1e-13):
    global _SUM_TARGET, _OBJ_DX
    _SUM_TARGET = tgt
    _OBJ_DX = dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constr}]
    try:
        r = minimize(obj_func, h, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        h_new = repair(r.x, tgt)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h, compute_c5(h, dx)

def smooth_max_grad(h, T, n):
    corr = get_corr(h)
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_vals = np.exp(T * shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return np.zeros(n)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    v = 1.0 - h
    half_n = n - 1
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        s = k_idx - half_n
        if s >= 0:
            idx = n - s
            if idx > 0:
                grad[0:idx] += w * v[s:n]
                grad[s:n] -= w * h[0:idx]
        else:
            idx = n + s
            if idx > 0:
                grad[-s:n] += w * v[0:idx]
                grad[0:idx] -= w * h[-s:n]
    return T * grad

    return np.interp(x_new, x_old, h)
