def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    m = np.max(v)
    if alpha < 1e6:
        return m + np.log(np.sum(np.exp(np.clip(alpha*(v-m), -60, 60)))) / alpha
    return m

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    m = np.max(v)
    e = np.exp(np.clip(alpha * (v - m), -60, 60))

            grad[:n+s] -= ws * h[-s:]

        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)

    xn = np.arange(n_new)
    h_new = np.interp(xn, xo, h)
