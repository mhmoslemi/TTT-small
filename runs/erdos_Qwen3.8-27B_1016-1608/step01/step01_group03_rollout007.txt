def subgrad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    # Contribution from h[m] as h[j+k]: j = m-k, valid if 0<=m-k<n
    if k >= 0:
        # m-k >= 0 => m >= k; m-k < n => m < n+k (always true since m<n)
