        c = compute_c5(h_t, dx)
        if c < best_c5:
            best_c5, best_h, best_n = c, h_t.copy(), n
        h_t2 = slsqp_polish(h_t, n, dx, tgt, maxiter=300, ftol=1e-15)
        c = compute_c5(h_t2, dx)
        if c < best_c5:
            best_c5, best_h, best_n = c, h_t2.copy(), n

    # Random restarts around best
    n = best_n
    dx = 2.0 / n
    tgt = n / 2.0
    _dx_G = dx
    _tgt_G = tgt
    n_restarts = 20
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 40:
            break
        sc = 0.003 + 0.025 * (1.0 - i / float(n_restarts))
        hp = repair(best_h + rng.randn(n) * sc, tgt)
        try:
            r = minimize(max_obj, hp, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': sum_constr}],
                         options={'maxiter': 250, 'ftol': 1e-15})
            h_new = repair(r.x, tgt)
            c = compute_c5(h_new, dx)
            if c < best_c5:
                best_c5, best_h = c, h_new
        except Exception:
            pass

    # Second wave: L-BFGS-B polish from best with very low T
    if time.time() - t0 < budget_s - 30:
        h_l2 = lbb_optimize(best_h, n, dx, tgt, t0 + budget_s - 35)
        c = compute_c5(h_l2, dx)
        if c < best_c5:
            best_c5, best_h = c, h_l2

    # Final SLSQP polish (multiple passes for convergence)
    if time.time() - t0 < budget_s - 10:
        for _ in range(4):
            if time.time() - t0 > budget_s - 10:
                break
            h_f = slsqp_polish(best_h, n, dx, tgt, maxiter=400, ftol=1e-17)
            c = compute_c5(h_f, dx)
            if c < best_c5:
                best_c5, best_h = c, h_f

    # Targeted perturbation on final best
    if time.time() - t0 < budget_s - 5:
        h_tp = target_perturb(best_h, n, dx, tgt, rng)
        c = compute_c5(h_tp, dx)
        if c < best_c5:
            best_c5, best_h = c, h_tp
        h_tf = slsqp_polish(best_h, n, dx, tgt, maxiter=300, ftol=1e-16)
        c = compute_c5(h_tf, dx)
        if c < best_c5:
            best_c5, best_h = c, h_tf

    # Final validation
    best_n = len(best_h)
    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
