from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(np.sum(hp))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def softmax_gradient_descent(h, n, max_iters=2000, beta=50.0, lr=0.1):
    """Optimize using softmax-smoothed gradient of max_k f_k(h)."""
    h = h.copy()
    target = n / 2.0
    h = project_to_set(h, target)
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        # Compute all f_k values
        f_all = compute_c5_all(h, n)  # shape (2n-1,)
        # Softmax weights
        f_shifted = f_all - np.max(f_all)
        weights = np.exp(beta * f_shifted)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        n_corr = 2 * n - 1
        for k in range(n_corr):
            if weights[k] < 1e-10:
                continue
            s = k - (n - 1)  # actual shift
            # d f_k / d h[j] = dx * [(1-h[j+s]) - h[j-s]] (with validity)
            j = np.arange(n)
            mask1 = (j + s >= 0) & (j + s < n)
            mask2 = (j - s >= 0) & (j - s < n)
            g = np.zeros(n)
            g[mask1] += (1.0 - h[j[mask1] + s])
            g[mask2] -= h[j[mask2] - s]
            grad += weights[k] * dx * g
        
        # Gradient step with decay
        lr_cur = lr / (1.0 + 0.001 * it)
        h_new = h - lr_cur * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_set(h_new, target)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def _time_left():
    return max(0.0, _budget - (time.time() - _start_t))

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def softmax_grad_descent(h, n, max_iters=3000, beta=30.0, lr=0.05):
    global _budget, _start_t
    h = h.copy()
    target = n / 2.0
    h = project_to_set(h, target)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if _time_left() < 30.0:
            break
        f_all = compute_c5_all(h, n)
        f_max = np.max(f_all)
        f_shifted = f_all - f_max
        weights = np.exp(beta * f_shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.ones(n_corr) / n_corr
        else:
            weights /= wsum
        
        grad = np.zeros(n)
        j_arr = np.arange(n)
        for k in range(n_corr):
            w = weights[k]
            if w < 1e-10:
                continue
            s = k - (n - 1)
            m1 = (j_arr + s >= 0) & (j_arr + s < n)
            m2 = (j_arr - s >= 0) & (j_arr - s < n)
            if np.any(m1):
                grad[m1] += w * (1.0 - h[j_arr[m1] + s])
            if np.any(m2):
                grad[m2] -= w * h[j_arr[m2] - s]
        grad *= dx
        
        lr_cur = lr / (1.0 + 0.005 * it)
        h_new = np.clip(h - lr_cur * grad, 0.0, 1.0)
        h_new = project_to_set(h_new, target)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def coordinate_descent(h, n, n_sweeps=20, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    
    for sweep in range(n_sweeps):
        if _time_left() < 30.0:
            break
        improved = False
        for j in range(n):
            old = h[j]
            cands = np.array([0.0, 1.0, max(0.0, old-0.1), min(1.0, old+0.1),
                            max(0.0, old-0.05), min(1.0, old+0.05)])
            bc, bv = old, best_c
            for c in cands:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv, bc = v, c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best_c = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best_c

def simulated_annealing(h, n, rng, n_steps=3000, t0=0.05):
    h = h.copy()
    target = n / 2.0
    # Ensure approximately binary
    h = (h > 0.5).astype(float)
    h = project_to_set(h, target)
    # Fix to exact binary
    n_ones = int(round(target))
    idx = np.argsort(-h)
    h = np.zeros(n)
    h[idx[:n_ones]] = 1.0
    target = n_ones  # for project_to_set to maintain
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    
    for it in range(n_steps):
        if _time_left() < 30.0:
            break
        T = t0 * 0.995 ** it
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h, n)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def make_triangle(n):
    x = np.linspace(0.0, 2.0, n)
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return project_to_set(h, n / 2.0)

def make_burst(n, n_bursts, rng):
    h = np.zeros(n)
    burst_len = max(2, (n // 2) // n_bursts)
    positions = rng.choice(n, size=n_bursts, replace=False)
    for p in positions:
        for d in range(burst_len):
            h[(p + d) % n] = 1.0
    return project_to_set(h, n / 2.0)

def make_smooth_random(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5
    for _ in range(rng.randint(2, 6)):
        f = rng.randint(1, 10)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.1, 0.5)
        h = h + a * np.sin(f * np.pi * x + p)
    return project_to_set(h, n / 2.0)

def slsqp_optimize(h0, n, maxiter=300):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    # Use a simple constraint
    def _con(x):
        return float(np.sum(x) - target)
    cons = [{'type': 'eq', 'fun': _con}]
    
    def _obj(x):
        return compute_c5(x, n)
    
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        h_out = project_to_set(res.x.copy(), target)
        c = compute_c5(h_out, n)
        return h_out, c
    except Exception:
        return h0, compute_c5(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
