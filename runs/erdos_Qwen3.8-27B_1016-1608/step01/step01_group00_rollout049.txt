        grad[m:] -= h[:n-m]
    else:
        p = -m
        grad[p:] += (1.0 - h[:n-p])
        grad[:n-p] -= h[p:]
    return grad

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_adam(h0, n, maxiter, top_k=1, lr=0.01):
    h = h0.copy()
    v = np.zeros(n)
    s = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for t in range(maxiter):
        corr = compute_corr(h)
        # Find top_k active lags
        top_idx = np.argsort(corr)[-top_k:]
        # Average gradient of active lags
        grad = np.zeros(n)
        for idx in top_idx:
            m = idx - (n - 1)
            grad += grad_corr(h, m)
        grad /= top_k
        
        # Adam update
        v = beta1 * v + (1 - beta1) * grad
        s = beta2 * s + (1 - beta2) * grad**2
        v_hat = v / (1 - beta1**(t+1))
        s_hat = s / (1 - beta2**(t+1))
        h_new = h - lr * v_hat / (np.sqrt(s_hat) + eps)
        h_new = repair(h_new, n)
        
        c = compute_c5(h_new)
        if c < best_c:
            best_h = h_new.copy()
            best_c = c
        h = h_new
    
    return best_h, best_c

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    target = n / 2.0

        diff = target - np.sum(h)

def grad_corr(h, m):
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        g[:n-m] += (1.0 - h[m:])
