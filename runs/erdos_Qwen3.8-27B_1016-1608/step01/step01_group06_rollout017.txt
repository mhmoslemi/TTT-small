    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode='full')  # length 2n-1

    exp_vals = np.exp(tau * (c_scaled - max_c))
    p = exp_vals / np.sum(exp_vals)  # length 2n-1
    
    # For each i, grad[i] = dx * sum_k p[k] * (dc_k/ dh_i)
    # dc_k/dh_i where k is index in full correlation (0 to 2n-2)

def soft_max_obj(h, tau):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    return max_c + np.log(np.sum(np.exp(tau * (c - max_c)))) / tau

def soft_max_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp(tau * (c - max_c))
    p = exp_c / np.sum(exp_c)
    obj = max_c + np.log(np.sum(exp_c)) / tau
    
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = (lags >= -i) & (lags <= n - 1 - i)
        idx2 = (lags >= i - (n-1)) & (lags <= i)
        s1 = np.dot(p[idx1], 1.0 - h[i + lags[idx1]])
        s2 = np.dot(p[idx2], h[i - lags[idx2]])
        grad[i] = dx * (s1 - s2)
    return obj, grad

def projected_gradient_descent(h, n, tau, n_iter=5000, lr_init=0.01, momentum=0.9):
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_obj = compute_score(h)
    
    for it in range(n_iter):
        obj, grad = soft_max_obj_grad(h, tau)
        v = momentum * v + (1 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        
        # Check if we improved
        obj_check = soft_max_obj(h_new, tau)
        if obj_check > obj + 0.01:
            lr *= 0.5
            if lr < 1e-8:
                break
            v *= 0  # reset momentum on failed step
        else:
            h = h_new
            lr = min(lr * 1.05, 0.1)
        
        s = compute_score(h)
        if s < best_obj:
            best_obj = s
            best_h = h.copy()
    
    return best_h, best_obj

def slsqp_optimize(h, n, tau=None, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0
    
    if tau is not None:
        def obj(x):
            return soft_max_obj(x, tau)
    else:
        def obj(x):
            return compute_score(x)
    
    def cons_eq(x):
        return np.sum(x) - target_sum
    
    constraints = [{'type': 'eq', 'fun': cons_eq}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_feasible(res['x'], n)
        return h_opt
    except Exception:
        return h

        s = compute_score(h_w)
        if s < best_s:
            best_s, best_h = s, h_w.copy()

    # Phase 2: Multi-start
    if time.time() - t0 < budget_s * 0.55:
        starts = [best_h.copy()]
        for f in [1, 2, 3, 4, 5, 7]:
            for a in [0.25, 0.4, 0.5]:
                for ph in [0.0, np.pi / 3, np.pi / 2, 2.0 * np.pi / 3]:
                    starts.append(make_sine(n0, f, a, ph))
        for nb in [2, 3, 4, 5, 7]:
            for r in [0.3, 0.4, 0.5, 0.6]:
                starts.append(make_block(n0, nb, r))
        for _ in range(15):
            h_r = rng.uniform(0, 1, n0)
            starts.append(project_feasible(h_r, n0))
        for _ in range(8):
            starts.append(project_feasible(best_h + rng.randn(n0) * 0.12, n0))

        for hs in starts:
            if time.time() - t0 > budget_s * 0.55:
                break
            h_c, s_c = pgd(hs, n0, 150, 800, 0.005, 0.9)
            h_c = slsqp_smooth(h_c, n0, 800, 100)
            h_c = slsqp_refine(h_c, n0, 100)
            s_c = compute_score(h_c)
            if s_c < best_s:
                best_s, best_h = s_c, h_c.copy()

    # Phase 3: Try other resolutions
    for n_try in [40, 50, 64, 80, 100]:
        if n_try == n0 and time.time() - t0 > budget_s * 0.5:
            continue
        if time.time() - t0 > budget_s * 0.72:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_c, s_c = pgd(h_i, n_try, 200, 2000, 0.005, 0.88)
        h_c = slsqp_refine(h_c, n_try, 150)
        s_c = compute_score(h_c)
        if s_c < best_s:
            best_s, best_h = s_c, h_c.copy()

    # Phase 4: Intensive refinement with subgradient + PDG
    nb = len(best_h)
    h_w = best_h.copy()
    if time.time() - t0 < budget_s * 0.82:
        h_w, s = subgrad_optimize(h_w, nb, 3000, 0.02, 0.005)
        if s < best_s:
            best_s, best_h = s, h_w.copy()
    if time.time() - t0 < budget_s * 0.9:
        for tau in [500.0, 2000.0, 8000.0]:
            if time.time() - t0 > budget_s * 0.9:
                break
            h_w, s = pgd(h_w, nb, tau, 2000, 0.003, 0.88)
            if s < best_s:
                best_s, best_h = s, h_w.copy()
    if time.time() - t0 < budget_s * 0.93:
        h_w = slsqp_refine(best_h, nb, 400)
        s = compute_score(h_w)
        if s < best_s:
            best_s, best_h = s, h_w.copy()

    # Phase 5: Coordinate descent polish
    if time.time() - t0 < budget_s * 0.97:
        remaining = (budget_s * 0.98) - (time.time() - t0)
        h_w, s = coord_descent(best_h, nb, 50, remaining)
        if s < best_s:
            best_s, best_h = s, h_w.copy()

    return best_h, best_s, len(best_h)
