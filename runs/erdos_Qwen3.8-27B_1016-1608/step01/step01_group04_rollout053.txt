from scipy.optimize import minimize
import time

_opt_n = 80
_opt_eps = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        # Spread adjustment to non-bound coordinates
        mask = (h > 0.001) & (h < 0.999)
        if not np.any(mask):
            mask = np.ones(n, dtype=bool)
        h[mask] += d / np.sum(mask.astype(float))
    return np.clip(h, 0.0, 1.0)

def obj_fn(x):
    n = _opt_n
    eps = _opt_eps
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-13:
        return m
    s = np.clip((c - m) / eps, -40.0, 40.0)
    return m + eps * np.log(np.sum(np.exp(s)))

def cons_fun(x):
    return np.sum(x) - _opt_n / 2.0

def cons_jac(x):
    return np.ones(_opt_n)

def slsqp(h0, n, eps, maxiter=300):
    global _opt_n, _opt_eps
    _opt_n = n
    _opt_eps = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj_fn, h0, method='SLSQP', bounds=bds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return repair(res.x, n)

def coord_descent(h, n, n_sweeps=50, tol=1e-10):
    """Coordinate descent with exact C5"""
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(n_sweeps):
        improved = False
        for i in np.random.permutation(n):
            # Try several values for coordinate i
            candidates = [best[i]]
            # Add perturbation
            for delta in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                candidates.append(best[i] + delta)
            # Try 0 and 1
            candidates.extend([0.0, 1.0])
            best_val = best_c5

                test[i] = v

                if c5 < best_val - tol:

                    best_new = v
                    test_best = test.copy()
