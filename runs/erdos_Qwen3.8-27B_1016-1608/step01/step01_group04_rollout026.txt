def subgradient_step(h, n, step_size, eps_active=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # Active set: correlations within eps_active of max
    active = np.where(corr >= m - eps_active)[0]
    
    # Subgradient: average gradient over active set
    grad = np.zeros(n)
    # For each active correlation index 'a' in the full correlation:
    # The shift is a - (n-1) (since full correlation has 2n-1 entries)
    for a in active:
        k = a - (n - 1)  # shift in units of grid points
        # ∂c_k/∂h[i] = 1 - h[i+k] - h[i-k] (with boundary)
        for i in range(n):
            g = 1.0
            if 0 <= i + k < n:
                g -= h[i + k]
            if 0 <= i - k < n:
                g -= h[i - k]
            grad[i] += g
    grad /= len(active)
    
    # Update
    h_new = h - step_size * grad
    # Project
    h_new = project_constraint(h_new, n)
    return h_new, m

def project_constraint(h, n):

        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Distribute uniformly
        h += diff / n
    return np.clip(h, 0, 1)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    if eps < 1e-13:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def grad_smooth(h, n, eps):
    """Gradient of smoothed objective"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    grad = np.zeros(n)
    if eps < 1e-13:
        active = np.where(c >= m - 1e-10)[0]
        for a in active:
            k = a - (n - 1)
            for i in range(n):
                g = 1.0
                if 0 <= i + k < n: g -= h[i + k]
                if 0 <= i - k < n: g -= h[i - k]
                grad[i] += g
        grad /= len(active)
        grad *= dx
        return grad
    else:
        s = np.clip((c - m) / eps, -50.0, 50.0)
        exp_s = np.exp(s)
        w = exp_s / np.sum(exp_s)
        for j in range(2 * n - 1):
            if w[j] < 1e-12:
                continue
            k = j - (n - 1)
            for i in range(n):
                g = 1.0
                if 0 <= i + k < n: g -= h[i + k]
                if 0 <= i - k < n: g -= h[i - k]
                grad[i] += w[j] * dx * g
        return grad

def slsqp_opt(h, n, eps, maxiter=300, ftol=1e-12):
    """SLSQP with analytic gradient"""
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        return objective_smooth(x, n, eps)
    
    def obj_grad(x):
        return grad_smooth(x, n, eps)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    r = minimize(obj, h, method='SLSQP', jac=obj_grad, bounds=bds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(r.x, n)

def subgradient_polish(h, n, steps=2000, base_lr=0.01):
    """Projected subgradient method for the max objective"""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for t in range(1, steps + 1):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(c)
        if m < best_c:
            best_c = m
            best_h = h.copy()
        
        # Subgradient: average over active set
        active_thresh = 0.005  # relative to max
        active = np.where(c >= m - active_thresh)[0]
        grad = np.zeros(n)
        for a in active:
            k = a - (n - 1)
            for i in range(n):
                g = 1.0
                if 0 <= i + k < n: g -= h[i + k]
                if 0 <= i - k < n: g -= h[i - k]
                grad[i] += g
        grad /= len(active)
        grad *= dx
        
        # Diminishing step size
        lr = base_lr / np.sqrt(t)
        h = h - lr * grad
        h = project_h(h, n)
    
    return best_h, best_c

def coord_descent(h, n, iters=10):
    """Simple coordinate descent: for each coord, try 0, current, 1, and small perturbations"""
    h = h.copy()
    target = n / 2.0
    for _ in range(iters):
        improved = False
        for i in range(n):
            old = h[i]
            # Candidate values: try to improve max corr
            best_v = old
            best_c = compute_c5(h)
            for v in [0.0, min(1.0, old + 0.05), max(0.0, old - 0.05)]:
                if abs(v - old) < 1e-10:
                    continue
                h[i] = v
                h = project_h(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v
                h[i] = old
            if best_v != old:
                h[i] = best_v
                h = project_h(h, n)
                improved = True
        if not improved:
            break
    return h

def subgrad_polish(h, n, steps=2000, lr0=0.02, thresh=0.003):

import numpy as np
from scipy.optimize import minimize
import time

_gn = 0
_geps = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _obj_impl(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-13:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _grad_impl(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)

    return _obj_impl(x, _gn, _geps)

def _obj_grad(x):
    return _grad_impl(x, _gn, _geps)

    return np.sum(x) - _gn / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, n, eps, maxiter=300, ftol=1e-12):
    global _gn, _geps
    _gn = n
    _geps = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    r = minimize(_obj, h, method='SLSQP', jac=_obj_grad,
                 bounds=bds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(r.x, n)

def subgrad_polish(h, n, steps=2000, lr0=0.02, thresh=0.003):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, steps + 1):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = float(np.max(c))
        if m < best_c:
            best_c = m
            best_h = h.copy()
        active = np.where(c >= m - thresh)[0]
        grad = np.zeros(n)
        for a in active:
            k = a - (n - 1)

                grad[i] += g

    h = ((h > 0.5).astype(float)).copy()
    h = project_h(h, n)
    best_c = compute_c5(h)
    best_h = h.copy()
    for _ in range(max_flips):
        i = np.random.randint(n)
        if h[i] > 0.5:
            j_candidates = np.where(h < 0.5)[0]
        else:
            j_candidates = np.where(h > 0.5)[0]
        if len(j_candidates) == 0:
            break
        j = j_candidates[np.random.randint(len(j_candidates))]
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 80
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * np.arange(n) / n)
    n = len(h0)
    h0 = project_h(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    def _has(rem):
        return time.time() - t0 < budget_s - rem

    # Phase 1: Multi-scale SLSQP with correct gradients
    for eps in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if not _has(180):
            break
        hn = slsqp_opt(best_h, n, eps, 500, 1e-13)
        c = compute_c5(hn)
        if c < best_c:
            best_c, best_h = c, hn
        if _has(160):
            hn2, c2 = subgrad_polish(best_h, n, 800, 0.015, 0.002)
            if c2 < best_c:
                best_c, best_h = c2, hn2

    # Phase 2: Structured restarts
    for p in [4, 5, 7, 8, 10, 13, 16, 20]:
        if not _has(120):
            break
        hp = np.array([1.0 if (i % p) < (p + 1) // 2 else 0.0 for i in range(n)])
        hp = project_h(hp, n)
        hn = slsqp_opt(hp, n, 0.005, 250, 1e-12)
        c = compute_c5(hn)
        if c < best_c:
            best_c, best_h = c, hn

    # Phase 3: Sine-wave restarts
    for k in [2, 3, 4, 5, 7]:
        if not _has(100):
            break
        hp = 0.5 + 0.4 * np.sin(2.0 * np.pi * k * np.arange(n) / n)
        hp = project_h(hp, n)
        hn = slsqp_opt(hp, n, 0.005, 200, 1e-12)
        c = compute_c5(hn)
        if c < best_c:
            best_c, best_h = c, hn

    # Phase 4: Random restarts with diverse strategies
    rc = 0
    while _has(60) and rc < 100:
        r = rc % 6
        if r == 0:
            hp = best_h + np.random.randn(n) * 0.04
        elif r == 1:
            hp = best_h + np.random.randn(n) * 0.10
        elif r == 2:
            hp = best_h.copy()
            s = np.random.randint(0, max(1, n - 5))
            l = np.random.randint(2, min(10, n - s))
            hp[s:s + l] = 1.0 - hp[s:s + l]
        elif r == 3:
            pw = np.random.randint(3, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            hp = best_h + 0.05 * np.sin(2.0 * np.pi * pw * np.arange(n) / n + ph)
        elif r == 4:
            hp = best_h.copy()
            idx = np.random.choice(n, size=max(1, n // 3), replace=False)
            hp[idx] = np.random.uniform(0, 1, size=len(idx))
        else:
            hp = best_h.copy()
            for _ in range(np.random.randint(2, 6)):
                idx = np.random.randint(n)
                hp[idx] = 1.0 - hp[idx]
        hp = project_h(hp, n)
        hn = slsqp_opt(hp, n, 0.003, 180, 1e-12)
        c = compute_c5(hn)
        if c < best_c:
            best_c, best_h = c, hn
            if _has(50):
                hn2, c2 = subgrad_polish(best_h, n, 400, 0.01, 0.002)
                if c2 < best_c:
                    best_c, best_h = c2, hn2
        rc += 1

    # Phase 5: Binary local search on best solution
    if _has(40):
        hb, cb = binary_local_search(best_h, n, 8000)
        if cb < best_c:
            best_c, best_h = cb, hb
            # Refine the binary solution with SLSQP
            if _has(30):
                hn = slsqp_opt(best_h, n, 0.003, 300, 1e-12)
                c = compute_c5(hn)
                if c < best_c:
                    best_c, best_h = c, hn

    # Phase 6: Final SLSQP polish
    if _has(25):
        hn = slsqp_opt(best_h, n, 0.0001, 600, 1e-14)
        c = compute_c5(hn)
        if c < best_c:
            best_c, best_h = c, hn

    # Phase 7: Subgradient fine polish
    if _has(15):
        hn, cn = subgrad_polish(best_h, n, 3000, 0.008, 0.001)
        if cn < best_c:
            best_c, best_h = cn, hn

    # Phase 8: Tiny perturbations
    if _has(5):
        for _ in range(3000):
            if time.time() - t0 > budget_s - 2:
                break
            hp = best_h + np.random.randn(n) * 0.002
            hp = project_h(hp, n)
            c = compute_c5(hp)
            if c < best_c:
                best_c, best_h = c, hp

    best_h = project_h(best_h, n)
    fc = compute_c5(best_h)
    if fc < best_c:
        best_c = fc
    return (best_h.tolist(), float(best_c), n)
