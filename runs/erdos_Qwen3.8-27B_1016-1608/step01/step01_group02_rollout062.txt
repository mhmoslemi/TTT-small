    x = np.linspace(0, 2, n)
    starts = []
    starts.append(project_feasible(rng.random(n), n / 2.0))
    starts.append(gen_binary(n, rng))
    if ref_h is not None and len(ref_h) == n:
        starts.append(project_feasible(ref_h + rng.randn(n) * 0.15, n / 2.0))
    for f in [1, 3, 5, 7, 11]:
        ph = rng.uniform(0, 2.0 * np.pi)
        starts.append(project_feasible(0.5 + 0.35 * np.cos(np.pi * f * x + ph), n / 2.0))
    h2 = 0.5 + 0.25 * np.cos(np.pi * rng.uniform(1, 4) * x) + 0.25 * np.cos(np.pi * rng.uniform(4, 10) * x)
    starts.append(project_feasible(h2, n / 2.0))
    return starts

def cyclic_cd(h, n, dx, target, n_rounds, step0):
    step = step0
    c5 = compute_c5(h, dx)
    for r in range(n_rounds):
        for idx in range(n):
            for d in [step, -step]:
                ht = h.copy()
                ht[idx] = np.clip(ht[idx] + d, 0.0, 1.0)
                ht = project_feasible(ht, target)
                c5t = compute_c5(ht, dx)
                if c5t < c5 - 1e-14:
                    h, c5 = ht, c5t
                    break
        step = max(0.003, step * 0.88)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):

        h = smooth_optimize(best_h, best_n, best_dx, best_n/2.0, [20, 80, 300, 1000, 3000], 400)
        h = project_feasible(h, best_n / 2.0)
        c5t = compute_c5(h, best_dx)
        if c5t < best_c5:
            best_h, best_c5 = h, c5t
