                pert[j] += (1.0 - h[i]) * h[j]

                                'maxls': 50})
        h = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h) < compute_c5(h0) + 1e-14:
            return h
    except Exception:
        pass
    return h0

def nm_polish(h0, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='Nelder-Mead', bounds=bnds,
                       options={'maxiter': 15000, 'xatol': 1e-14, 'fatol': 1e-16,
                                'adaptive': True})
        h = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h) < compute_c5(h0) + 1e-14:
            return h
    except Exception:
        pass
    return h0

def try_resample(h_best, n, n2, maxiter=800):
    if n2 == n:
        return h_best
    xp = np.linspace(0.0, 1.0, n)
    xn = np.linspace(0.0, 1.0, n2)
    h2 = np.interp(xn, xp, h_best)
    h2 = repair(h2, n2)
    h2 = slsqp_opt(h2, n2, maxiter=maxiter)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Long SLSQP from initial (most critical for tight local opt)
    if time.time() - t0 < budget_s * 0.15:
        h = slsqp_opt(h_best, n, maxiter=8000, ftol=1e-16)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: Active-lag direction-aware perturbations + SLSQP
    if time.time() - t0 < budget_s * 0.45:
        for it in range(60):
            if time.time() - t0 > budget_s * 0.45:
                break
            # Mix of active-lag and random perturbation
            if it % 3 == 0:
                pert = active_lag_perturb(h_best, n, scale=0.02 + 0.01 * np.random.rand())
            else:
                scale = 0.005 * np.exp(3.0 * np.random.rand())
                pert = np.random.randn(n) * scale
            h_p = repair(h_best + pert, n)
            h = slsqp_opt(h_p, n, maxiter=1500, ftol=1e-15)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 3: L-BFGS-B with soft-max for broader exploration
    if time.time() - t0 < budget_s * 0.60:
        for it in range(20):
            if time.time() - t0 > budget_s * 0.60:
                break
            scale = 0.01 + 0.05 * np.random.rand()
            h_p = repair(h_best + np.random.randn(n) * scale, n)
            h = lbfgs_soft(h_p, n, maxiter=1500)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c
            # Follow up with SLSQP
            h = slsqp_opt(h, n, maxiter=2000, ftol=1e-15)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 4: More aggressive random restarts
    if time.time() - t0 < budget_s * 0.78:
        scales = [0.002, 0.004, 0.007, 0.012, 0.02, 0.035, 0.06, 0.1, 0.18, 0.3]
        it = 0
        while time.time() - t0 < budget_s * 0.78:
            scale = scales[it % len(scales)]
            h_p = repair(h_best + np.random.randn(n) * scale, n)
            h = slsqp_opt(h_p, n, maxiter=2000, ftol=1e-15)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c
            it += 1

    # Phase 5: Nelder-Mead polish
    if time.time() - t0 < budget_s * 0.88:
        h = nm_polish(h_best, n)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 6: Try different resolutions
    if time.time() - t0 < budget_s * 0.95:
        for n2 in [50, 60, 80, 100, 140]:
            if time.time() - t0 > budget_s * 0.94:
                break
            h2 = try_resample(h_best, n, n2, maxiter=1500)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2
                # Polish at new resolution
                h3 = slsqp_opt(h_best, n, maxiter=3000, ftol=1e-15)
                c3 = compute_c5(h3)
                if c3 < c5_best:
                    h_best, c5_best = h3, c3

    # Phase 7: Final ultra-tight SLSQP
    if time.time() - t0 < budget_s * 0.99:
        h = slsqp_opt(h_best, n, maxiter=10000, ftol=1e-16)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
