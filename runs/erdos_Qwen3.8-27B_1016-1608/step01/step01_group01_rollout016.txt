```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def sum_constraint_jac(h):
    return np.ones(len(h))

def soft_max_obj(h, temp):
    c = correlate_full(h)
    m = c.max()
    if temp < 1e-15:
        return m
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def subgrad_max_corr(h):
    c = correlate_full(h)
    k_star = np.argmax(c)
    n = len(h)
    shift = k_star - (n - 1)
    m = np.arange(n)
    j_plus = m + shift
    j_minus = m - shift
    grad = np.zeros(n)
    vp = (j_plus >= 0) & (j_plus < n)
    vm = (j_minus >= 0) & (j_minus < n)
    grad[vp] += (1.0 - h[j_plus[vp]])
    grad[vm] -= h[j_minus[vm]]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        # Adjust only free variables (not at bounds)
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_slsqp(h0, n, temps, maxiter, t_end):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    constraints = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    for temp in temps:
        if time.time() > t_end:
            break
        res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def optimize_subgrad_momentum(h0, n, t_end, max_iter=20000):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    v = np.zeros(n)
    beta = 0.85
    alpha = 0.5
    patience = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        grad = subgrad_max_corr(best_h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Gradient projection: remove component in direction of sum constraint
        # Feasible directions are orthogonal to ones vector
        grad_feas = grad - np.mean(grad)  # project onto sum-preserving subspace
        v = beta * v - alpha * grad_feas
        h_trial = project_feasible(best_h + v, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_trial.copy()
            patience = 0
            alpha = min(alpha * 1.02, 5.0)
        else:
            patience += 1
            if patience % 20 == 0:
                alpha *= 0.8
                v *= 0.5
            if patience > 500:
                v = np.zeros(n)
                patience = 0
    return best_h, best_c5

def coordinate_descent(h0, n, t_end, max_rounds=100):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        for i in range(n):
            if time.time() > t_end:
                break
            # Try small perturbations
            for d in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_trial = best_h.copy()
                h_trial[i] = np.clip(h_trial[i] + d, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
                    break
    return best_h, best_c5

def binary_local_search(h0, n, t_end, max_iters=50000):
    # Convert to binary-ish and do local search
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Pick a random pair to swap
        i, j = np.random.choice(n, 2, replace=False)
        h_trial = best_h.copy()
        h_trial[i], h_trial[j] = best_h[j], best_h[i]
        # Try flipping or swapping
        # Method 1: swap
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_trial.copy()
        # Method 2: random perturbation
        elif np.random.random() < 0.3:
            h_trial = best_h + np.random.randn(n) * 0.1
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_trial.copy()
    return best_h, best_c5
