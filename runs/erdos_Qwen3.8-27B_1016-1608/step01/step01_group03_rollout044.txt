import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def _repair(h, target):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _coord_descent_pass(h, n, max_iter=200, time_limit=None, t_start=None):
    tgt = n / 2.0
    dx = 2.0 / n
    best_c5 = _compute_c5(h, dx)
    for it in range(max_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break

            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = h_save

            ts2 = np.linspace(t_lo, t_hi, 61)
            vals2 = np.max(A * ts2[:, None] + B[None, :], axis=1)

def _slsqp_polish(h, n, maxiter=200, ftol=1e-15):
    tgt = n / 2.0
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n

        return _compute_c5(np.clip(x, 0.0, 1.0), dx)

    def _gc(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        idx = int(np.argmax(corr))
        m = idx - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            gp = j + m
            gm = j - m
            if 0 <= gp < n:
                grad[j] += (1.0 - hc[gp])
            if 0 <= gm < n:
                grad[j] -= hc[gm]
        return grad

    def _sc(x):
        return np.sum(x) - tgt

    cons = [{'type': 'eq', 'fun': _sc}]
    try:
        r = minimize(_fc, h, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        h_new = _repair(r.x, tgt)
        c_new = _compute_c5(h_new, dx)
        if c_new < _compute_c5(h, dx):
            return h_new
    except Exception:
        pass
    return h

def _slsqp_soft(h, n, beta=200.0, maxiter=300):

    dx = 2.0 / n

    def _fs(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full") * dx
        M = np.max(corr)
        w = np.exp(np.clip(beta * (corr - M), -500.0, 0.0))
        return M + np.log(np.sum(w)) / beta

    def _sc2(x):

        c_old = _compute_c5(h, dx)
        if c_hard < c_old:

    except Exception:
