    # First term: j from 0 to n-1, j+offset from 0 to n-1
    # j+offset in [0,n-1] means offset <= j <= n-1-offset (if offset>=0) or n-1-|offset| <= j (if offset<0)
    if offset >= 0:
        for j in range(n - offset):  # j = 0, ..., n-1-offset
            grad[j] += (1.0 - h[j + offset])
    else:
        for j in range(-offset, n):  # j = -offset, ..., n-1
            grad[j] += (1.0 - h[j + offset])
    
    # Second term: j-offset in [0,n-1]
    if offset >= 0:
        for j in range(offset, n):  # j-offset = 0, ..., n-1-offset
            grad[j] += (-h[j - offset])
    else:
        for j in range(0, n + offset):  # j-offset = 0, ..., n-1+offset
            grad[j] += (-h[j - offset])
    
    return grad * (2.0 / n)

def soft_max_val(c, tau):
    """Soft max of array c with temperature tau."""
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def soft_max_grad(c, tau):
    """Gradient of soft max w.r.t. c."""
    cm = np.max(c)
    w = np.exp(tau * (c - cm)) / np.sum(np.exp(tau * (c - cm)))
    return w

    """Gradient of correlation at shift s w.r.t. h. Returns unnormalized (multiply by dx)."""
