    """Returns (c5_value, corr_array)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

    lo, hi = -2.0, 2.0

        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:

        else:
            hi = mid

    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

import numpy as np
import time

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def subgrad_opt(h0, n, n_iters=3000, lr0=0.05, topk=3):
    target = n/2
    h = project_to_set(h0.astype(float), target)
    best_h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0/n
    
    for it in range(n_iters):
        lr = lr0 / (1 + 0.005*it)
        corr = compute_corr(h, n)
        c5 = float(np.max(corr))
        
        # Top-k shifts
        top = np.argsort(corr)[::-1][:topk]
        w = np.zeros(2*n-1)
        for idx in top:
            w[idx] = corr[idx]
        w_sum = w[top].sum()
        if w_sum < 1e-10: continue
        w /= w_sum
        
        # Gradient
        grad = np.zeros(n)
        for idx in top:
            s = idx - (n-1)
            wi = w[idx]
            if s >= 0:
                grad[:n-s] += wi * (1.0 - h[s:])
                if n-s > 0:
                    grad[s:] -= wi * h[:n-s]
            else:
                a = -s
                grad[a:] += wi * (1.0 - h[:n-a])
                if n-a > 0:
                    grad[:n-a] -= wi * h[a:]
        
        h = project_to_set(h - lr*grad, target)
        c = compute_c5(h, n)
        if c < best:
            best = c
            best_h = h.copy()
    
    return best_h, best

def binary_ls(h, n, rng, n_passes=30, max_candidates=15):
    """Binary local search with exactly n//2 ones"""
    target = n//2
    h = np.zeros(n)
    idx = np.argsort(-h)[:target]
    h[idx] = 1.0
    best = compute_c5(h, n)
    corr = compute_corr(h, n)
    best_corr = corr.copy()
    
    for _ in range(n_passes):
        improved = False
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros[:max_candidates]:
                # Swap: h[i] 1->0, h[j] 0->1
                # Delta for each shift
                dcorr = np.zeros(2*n-1)
                # h[i] changes by -1
                s = 0
                # h[i] as first factor at shift k: position i+k
                for k in range(-(i), n-i):
                    pos = k - (n-1)
                    if 0 <= i+k < n:
                        dcorr[pos] += -1.0 * (1-h[i+k])
                    if 0 <= i-k < n:
                        dcorr[pos] += -1.0 * h[i-k]
                
                # This is getting too complex for O(n²) per swap
                # Just recompute
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                if c < best - 1e-10:
                    best = c
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h, best

Actually, let me use the incremental update for the binary LS. The delta formula:

When h[i] changes by δ (δ = -1 for 1→0, +1 for 0→1):
At correlate index m (shift s = m - (n-1)):
- h[i] as first factor: contributes δ*(1-h[i+s]) at shift s, valid if 0<=i+s<n
- h[i] in (1-h) part: contributes -δ*h[i-s] at shift s, valid if 0<=i-s<n

So for swap (i: 1→0, j: 0→1):
δ_i = -1, δ_j = +1

dcorr[m] = -1*(1-h[i+s]) [if 0<=i+s<n] + h[i-s] [if 0<=i-s<n]
          + 1*(1-h[j+s]) [if 0<=j+s<n] - h[j-s] [if 0<=j-s<n]

where s = m-(n-1).

I can compute this in O(n) using vectorized ops:

```python
def swap_delta(h, i, j, n):
    """Compute delta in corr when h[i]:1->0, h[j]:0->1"""
    d = np.zeros(2*n-1)
    s_range = np.arange(-(n-1), n)  # all shifts
    # h[i] change (-1):
    # First factor: shift s where 0<=i+s<n → s in [-i, n-1-i]
    valid1 = (i + s_range >= 0) & (i + s_range < n)
    d[valid1] += -1.0 * (1.0 - h[i + s_range[valid1]])
    # Second factor: shift s where 0<=i-s<n → s in [i-(n-1), i]
    valid2 = (i - s_range >= 0) & (i - s_range < n)
    d[valid2] += 1.0 * h[i - s_range[valid2]]  # -(-1)*h[i-s] = h[i-s]
    
    # h[j] change (+1):
    valid3 = (j + s_range >= 0) & (j + s_range < n)
    d[valid3] += 1.0 * (1.0 - h[j + s_range[valid3]])
    valid4 = (j - s_range >= 0) & (j - s_range < n)
    d[valid4] += -1.0 * h[j - s_range[valid4]]
    
    return d
```

This is O(n) per swap (vectorized). Then updating corr is just corr += d, O(n). Finding max is O(n). So each swap is O(n) total.

For n=90, each swap is ~360 ops (4 passes of 90). With 675 swaps per pass, that's 243K ops per pass. Very fast.
