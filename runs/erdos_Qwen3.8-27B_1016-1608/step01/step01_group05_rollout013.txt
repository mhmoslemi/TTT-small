def binary_2swap(h, n, rng, rounds=200):
    """Full 2-swap local search on binarized h"""
    t = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:t]] = 1.0
    best = compute_c5(hb, n)
    for _ in range(rounds):
        imp = False
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        for oi in ones:
            for oz in zeros:
                hb[oi] = 0.0
                hb[oz] = 1.0
                v = compute_c5(hb, n)
                if v < best - 1e-12:
                    best = v
                    imp = True
                    zeros = np.where(hb == 0.0)[0]
                else:
                    hb[oi] = 1.0
                    hb[oz] = 0.0
        if not imp:

def binary_sa(h, n, rng, t_init=0.1, t_final=0.001, steps=50000):
    """Simulated annealing on binary h"""
    t = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:t]] = 1.0
    best = compute_c5(hb, n)
    cur = best
    ones = list(np.where(hb == 1.0)[0])
    zeros = list(np.where(hb == 0.0)[0])
    for s in range(steps):
        T = t_init * (t_final / t_init) ** (s / steps)
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oi, oz = ones[i], zeros[j]
        hb[oi] = 0.0; hb[oz] = 1.0
        v = compute_c5(hb, n)
        d = v - cur
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-9)):
            cur = v
            ones[i] = oz; zeros[j] = oi
            if v < best:
                best = v
        else:
            hb[oi] = 1.0; hb[oz] = 0.0
    return hb, best

def coord_descent(h, n, rng, sweeps=50):
    """Adaptive coordinate descent"""
    t = n / 2.0
    h = project_to_set(h.astype(float), t)
    best = compute_c5(h, n)
    for _ in range(sweeps):
        imp = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            cands = np.array([0.0, 1.0, max(0, old-0.2), min(1, old+0.2),
                           max(0, old-0.1), min(1, old+0.1),
                           max(0, old-0.05), min(1, old+0.05),
                           max(0, old-0.01), min(1, old+0.01)])
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_to_set(h, t)
                v = compute_c5(h, n)
                if v < bv - 1e-12:
                    bv = v
                    bc = c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, t)
                best = bv
                imp = True
            else:
                h[j] = old
                h = project_to_set(h, t)
        if not imp:
            break
    return h, best
