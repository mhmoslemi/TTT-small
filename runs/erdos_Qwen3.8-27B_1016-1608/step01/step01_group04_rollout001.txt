from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(50):
        lo, hi = -1.0, 2.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2.0
        h_new = np.clip(h - t, 0.0, 1.0)
        if np.abs(np.sum(h_new) - target_sum) < 1e-12:
            break
        h = h_new
    return h

def smax_obj(h, n, eps):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    if eps < 1e-14:
        return float(m)
    s = np.clip((c - m) / eps, -60.0, 60.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

    nN = n
    def obj(x):
        return smax_obj(x, nN, eps)
    def con_fun(x):
        return float(np.sum(x) - target_sum)
    constraints = [{'type': 'eq', 'fun': con_fun}]
    bounds = [(0.0, 1.0)] * n
    r = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints,
                 options={'maxiter': maxiter, 'ftol': 1e-12})

    except Exception:
        pass
    
    for p in [4, 6, 8, 10, 12, 16, 20, 24, 32]:
        hb = make_block(n, p)
        hb = repair(hb, n, ts)
        candidates.append(hb)
    
    for k in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0]:
            hs = make_sine(n, k, 0.45, ph)
            hs = repair(hs, n, ts)
            candidates.append(hs)
    
    for _ in range(8):
        hr = np.random.uniform(0.0, 1.0, n)
        hr = repair(hr, n, ts)
        candidates.append(hr)
    
    # Evaluate all candidates
    best_h = candidates[0]
    best_c = compute_c5(best_h)
    for c in candidates[1:]:
        cv = compute_c5(c)
        if cv < best_c:
            best_c = cv
            best_h = c
    
    # Phase 1: DE from top diverse starts
    # Sort candidates by quality, pick diverse subset
    cand_fits = [(compute_c5(c), i) for i, c in enumerate(candidates)]
    cand_fits.sort()
    used = [cand_fits[0][1]]
    for f, idx in cand_fits[1:]:
        if len(used) >= 8:
            break
        diverse = True
        for u in used:
            if np.max(np.abs(candidates[idx] - candidates[u])) < 0.1:
                diverse = False
                break
        if diverse:
            used.append(idx)
    
    for ci in used:
        if time.time() - t0 > budget_s - 80:
            break
        st = candidates[ci].copy()
        hn, fn = de_optimize(st, n, ts, 40, 200, 0.7, 0.85, t0, budget_s, 60)
        if fn < best_c:
            best_c = fn
            best_h = hn
        # Also do a second DE with different params
        if time.time() - t0 > budget_s - 80:
            break
        hn2, fn2 = de_optimize(best_h, n, ts, 30, 150, 0.5, 0.7, t0, budget_s, 60)
        if fn2 < best_c:
            best_c = fn2
            best_h = hn2
    
    # Phase 2: SLSQP multi-scale polish
    for eps in [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]:
        if time.time() - t0 > budget_s - 40:
            break
        rem = budget_s - 40 - (time.time() - t0)
        mi = int(min(400, max(50, rem * 20)))
        hn = slsqp_polish(best_h, n, ts, eps, mi, t0, budget_s, 30)
        cn = compute_c5(hn)
        if cn < best_c:
            best_c = cn
            best_h = hn
    
    # Phase 3: Pair local search
    rem = budget_s - (time.time() - t0) - 10
    n_ls = int(min(50000, rem * 8000))
    best_h, best_c = pair_local_search(best_h, n, ts, n_ls, t0, budget_s, 5)
    
    # Phase 4: Final SLSQP micro-polish
    if time.time() - t0 < budget_s - 5:
        hn = slsqp_polish(best_h, n, ts, 0.00005, 200, t0, budget_s, 3)
        cn = compute_c5(hn)
        if cn < best_c:
            best_c = cn
            best_h = hn
    
    final = repair(best_h, n, ts)
    fc = compute_c5(final)
    return (final.tolist(), float(fc), n)
