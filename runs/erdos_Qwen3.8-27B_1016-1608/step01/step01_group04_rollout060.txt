import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += d / np.sum(mask)
        else:
            h += d / n
    return np.clip(h, 0.0, 1.0)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def binary_local_search(h_bin, n, n_iters, rng, t0=None, deadline=None):
    h = h_bin.copy().astype(float)
    c = corr_full(h)
    cm = float(np.max(c))
    improved = 0
    for it in range(n_iters):
        if t0 is not None and deadline is not None and time.time() > deadline:
            break
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        n_try = min(20, len(zeros) * len(ones))
        for _ in range(n_try):
            i = rng.choice(zeros)
            j = rng.choice(ones)
            h[i] = 1.0
            h[j] = 0.0
            c_new = corr_full(h)
            cm_new = float(np.max(c_new))
            if cm_new < cm - 1e-15:
                cm = cm_new
                c = c_new
                improved += 1
            else:
                h[i] = 0.0
                h[j] = 1.0
        if improved > 50 and (it + 1) % 3 == 0:
            pass
    return h, cm

        return np.ones(n)

    r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-13})
    return repair(r.x, n)


def make_thue_morse(n):
    idx = np.arange(n)
    vals = np.zeros(n, dtype=int)
    for k in range(int(np.log2(max(n, 2))) + 1):
        period = 1 << (k + 1)
        half = 1 << k
        vals = ((idx % period) // half).astype(int)
    vals[idx >= n] = 0
    return vals.astype(float)


def upsample(h, np_old, np_new):
    x_old = np.linspace(0, 2, np_old, endpoint=False)
    x_new = np.linspace(0, 2, np_new, endpoint=False)
    return np.interp(x_new, x_old, np.append(h, h[0]),
                     period=2.0)
