import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_shift(corr_full, n):
    """Get the shift (in index space) of the max correlation."""
    k = np.argmax(corr_full)
    return k - (n - 1)

def subgradient_at_shift(h, s, n):
    """Compute subgradient of c_s with respect to h.
    c_s = sum_i h[i] * (1 - h[i+s]) for valid i, i+s.
    dc_s/dh[j] = 1 - h[j+s]*[valid] - h[j-s]*[valid]
    """
    g = np.ones(n)
    if s >= 0:
        # h[j+s] valid when j+s < n, i.e., j < n-s
        g[:n-s] -= h[s:]
        # h[j-s] valid when j-s >= 0, i.e., j >= s
        g[s:] -= h[:n-s]
    else:
        ss = -s
        # h[j+s] = h[j-ss] valid when j-ss >= 0, i.e., j >= ss
        g[ss:] -= h[:n-ss]
        # h[j-s] = h[j+ss] valid when j+ss < n, i.e., j < n-ss
        g[:n-ss] -= h[ss:]
    return g

def project_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2}."""
    target = n * 0.5
    lo, hi = x.min() - 1.0, x.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_optimize(h0, n, n_iter, t0, budget, lr0=0.01, rng=None):
    """Run subgradient method for n iterations."""
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c = compute_c5(h0)
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_val = np.max(corr)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        # Find argmax (with some randomization among near-max)
        thresh = c_val - 1e-4
        candidates = np.where(corr >= thresh)[0]
        if len(candidates) > 1 and rng is not None:
            k = candidates[rng.randint(len(candidates))]

        s = k - (n - 1)

        g = g - np.mean(g)  # Actually should be more careful
        norm_g = np.linalg.norm(g)

        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.01 * it)
        
        # Gradient step

        # Project to feasible set
        h_new = project_feasible(h_new, n)
        
        # Check progress

            h_new2 = project_feasible(h_new2, n)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _fix_sum(h, n):
    target = n * 0.5
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(mask.sum())
        if k == 0:
            break
        h = np.clip(h + (d / k) * mask, 0.0, 1.0)
    return h

def _project(x, n):
    target = n * 0.5
    lo = x.min() - target / n - 0.5
    hi = x.max() - target / n + 0.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def _subgrad(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    else:
        ss = -s
        if ss < n:
            g[ss:] += (1.0 - h[:n - ss])
            g[:n - ss] -= h[ss:]
    return g

def _sr_opt(h0, n, max_iter, deadline, rng, lr0=0.015):
    dx = 2.0 / n
    h = _project(h0.copy(), n)
    best_h = h.copy()
    best_c = _c5(best_h)
    full_len = 2 * n - 1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_val = float(corr.max())
        if c_val < best_c - 1e-13:
            best_c = c_val
            best_h = h.copy()
        thresh = c_val - 0.003
        near = np.where(corr >= thresh)[0]
        if len(near) == 0:
            near = np.array([int(np.argmax(corr))])
        k_sel = min(7, len(near))
        sel = rng.choice(len(near), size=k_sel, replace=False)
        shifts = [int(near[i] - (n - 1)) for i in sel]
        g = np.zeros(n)
        for s in shifts:
            g += _subgrad(h, s, n)
        g /= k_sel
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        lr = lr0 / (1.0 + 0.003 * it)
        h_new = _project(h - lr * g / gn, n)
        c_new = _c5(h_new)
        if c_new <= c_val + 0.002:
            h = h_new
        else:
            h = _project(h - 0.1 * lr * g / gn, n)
    return best_h, best_c

def _slsqp_smooth(h0, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def f_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        sc = alpha * c
        sm = sc.max()
        e = np.exp(sc - sm)
        sw = e.sum()
        w = e / sw
        fval = (sm + np.log(sw)) / alpha
        g = np.zeros(n)
        for si in range(2 * n - 1):
            if w[si] < 1e-10:
                continue
            s = si - (n - 1)
            g += w[si] * _subgrad(x, s, n)
        g *= dx
        return fval, g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f_grad, h0, method='SLSQP', jac=True,
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _interp_to(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return _fix_sum(np.clip(np.interp(xn, xo, h), 0.0, 1.0), new_n)

def _local_perturb(h, n, rng, t_deadline):

    best_c = _c5(best_h)
    for _ in range(50000):

        delta = rng.uniform(-0.08, 0.08, size=len(idx))
        h2 = best_h.copy()
        h2[idx] = np.clip(h2[idx] + delta, 0.0, 1.0)
        h2 = _fix_sum(h2, n)
        c2 = _c5(h2)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
