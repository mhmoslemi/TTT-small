def compute_grad(h, n, w):
    """w: weights for each lag (length 2n-1), corresponding to lags -(n-1) to n-1"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for l in range(2*n - 1):
        wl = w[l]
        if wl < 1e-12:
            continue
        k = l - (n - 1)
        if k >= 0:
            # First factor: h[j] * (1-h[j+k]), j in [0, n-1-k]
            if n - 1 - k >= 0:
                grad[0:n-k] += wl * dx * (1.0 - h[k:n])
            # Second factor: -h[j-k], j in [k, n-1]
            if k > 0:
                grad[k:n] -= wl * dx * h[0:n-k]
        else:
            kk = -k
            # First factor: h[j] * (1-h[j-kk]), j in [kk, n-1]
            if n - 1 - kk >= 0:
                grad[kk:n] += wl * dx * (1.0 - h[0:n-kk])
            # Second factor: -h[j+kk], j in [0, n-1-kk]
            if kk > 0:
                grad[0:n-kk] -= wl * dx * h[kk:n]
    return grad

            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def compute_grad(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        wl = w[l]
        if wl < 1e-12:
            continue
        k = l - (n - 1)
        if k >= 0:
            if n - 1 - k >= 0:
                grad[0:n - k] += wl * dx * (1.0 - h[k:n])
            if k > 0:
                grad[k:n] -= wl * dx * h[0:n - k]
        else:
            kk = -k
            if n - 1 - kk >= 0:
                grad[kk:n] += wl * dx * (1.0 - h[0:n - kk])
            if kk > 0:
                grad[0:n - kk] -= wl * dx * h[kk:n]
    return grad

def subgradient_opt(h, n, n_iter=800, step0=0.04, tau=300.0):
    h = h.copy()
    best_h = h.copy()
    best_s = compute_score(h)
    for it in range(n_iter):
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cm = np.max(c)
        shifted = np.clip(c - cm, -10.0 / tau, 0.0)
        w = np.exp(tau * shifted)
        wsum = w.sum()
        if wsum < 1e-15:
            continue
        w /= wsum
        grad = compute_grad(h, n, w)
        step = step0 / (1.0 + it * 0.005)
        h = project(h - step * grad, n)
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
    return best_h, best_s

def coord_descent(h, n, n_iter=150, rng=None):
    h = h.copy()
    best_s = compute_score(h)
    if rng is None:
        rng = np.random.RandomState(0)
    deltas = np.array([0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1])
    for it in range(n_iter):
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                h_new = h.copy()
                h_new[i] += d
                h_new = project(h_new, n)
                s = compute_score(h_new)
                if s < best_s - 1e-13:
                    h = h_new
                    best_s = s
                    improved = True
                    break
        if not improved:
            break
    return h, best_s

def time_remaining(t0, budget_s):
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_n
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    # Phase 1: Optimize at initial resolution
    _g_n = n_init
    n = n_init
    h0 = project(h_init, n)
    s0 = compute_score(h0)
    if s0 < best_score:
        best_score, best_h = s0, h0.copy()

    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n

    for r in range(12):
        if time_remaining(t0, budget_s) < 60:
            break
        if r == 0:
            hs = h0.copy()
        elif r == 1:
            hs = project(h0 + rng.randn(n) * 0.02, n)
        elif r == 2:
            hs = project(h0 + rng.randn(n) * 0.06, n)
        elif r == 3:
            x = np.linspace(0, 2, n, endpoint=False)
            freq = rng.randint(2, 6)
            hs = 0.5 + 0.2 * np.sin(2 * np.pi * x * freq / 2.0 + rng.uniform(0, 6.28))
            hs = project(hs, n)
        elif r == 4:
            x = np.linspace(0, 2, n, endpoint=False)
            hs = 0.5 + 0.25 * (2.0 * (x % 1.0) - 1.0)
            hs = project(hs, n)
        elif r == 5:
            hs = project(rng.uniform(0.3, 0.7, n), n)
        elif r == 6:
            # Binary-ish random
            hs = (rng.rand(n) < 0.5).astype(float)
            hs = project(hs, n)
        else:
            hs = project(h0 + rng.randn(n) * 0.12, n)

        try:
            res = minimize(soft_score, hs, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 500, 'ftol': 1e-12})
            hc = project(res['x'], n)
            sc = compute_score(hc)
            if sc < best_score:
                best_score, best_h = sc, hc.copy()
            if sc < s0:
                s0, h0 = sc, hc.copy()
        except Exception:
            pass

    # Subgradient optimization rounds
    for sg in range(5):
        if time_remaining(t0, budget_s) < 60:
            break
        start_tau = 200.0 + sg * 100.0
        hc, sc = subgradient_opt(h0, n, n_iter=800, step0=0.03, tau=start_tau)
        if sc < best_score:
            best_score, best_h = sc, hc.copy()
        if sc < s0:
            s0, h0 = sc, hc.copy()

    # Coordinate descent
    if time_remaining(t0, budget_s) > 30:
        hc, sc = coord_descent(best_h, len(best_h), n_iter=200, rng=rng)
        if sc < best_score:
            best_score, best_h = sc, hc.copy()

    # SLSQP with exact max on best
    if time_remaining(t0, budget_s) > 30:
        _g_n = len(best_h)
        nf = len(best_h)
        bounds_f = [(0.0, 1.0)] * nf
        constraints_f = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        try:
            res = minimize(compute_score, best_h, method='SLSQP', bounds=bounds_f,
                           constraints=constraints_f, options={'maxiter': 1500, 'ftol': 1e-14})
            hc = project(res['x'], nf)
            sc = compute_score(hc)
            if sc < best_score:
                best_score, best_h = sc, hc.copy()
        except Exception:
            pass

    # Phase 2: Explore other resolutions
    n_candidates = [40, 48, 56, 64, 72, 80]
    for n_alt in n_candidates:
        if time_remaining(t0, budget_s) < 120:
            break
        _g_n = n_alt
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = project(h_alt, n_alt)
        s_alt = compute_score(h_alt)
        if s_alt < best_score:
            best_score, best_h = s_alt, h_alt.copy()

        b_alt = [(0.0, 1.0)] * n_alt
        c_alt = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        for r in range(5):
            if time_remaining(t0, budget_s) < 60:
                break
            if r == 0:
                hs = h_alt.copy()
            elif r == 1:
                hs = project(h_alt + rng.randn(n_alt) * 0.03, n_alt)
            else:
                hs = project(h_alt + rng.randn(n_alt) * 0.1, n_alt)
            try:
                res = minimize(soft_score, hs, method='SLSQP', bounds=b_alt,
                               constraints=c_alt, options={'maxiter': 400, 'ftol': 1e-12})
                hc = project(res['x'], n_alt)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score, best_h = sc, hc.copy()
                    h_alt = hc.copy()
            except Exception:
                pass

        if time_remaining(t0, budget_s) > 60:
            hc, sc = subgradient_opt(h_alt, n_alt, n_iter=500, step0=0.03, tau=300.0)
            if sc < best_score:
                best_score, best_h = sc, hc.copy()

    # Phase 3: Final intensive polish
    if time_remaining(t0, budget_s) > 20:
        nf = len(best_h)
        _g_n = nf
        bounds_f = [(0.0, 1.0)] * nf
        constraints_f = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        try:
            res = minimize(compute_score, best_h, method='SLSQP', bounds=bounds_f,
                           constraints=constraints_f, options={'maxiter': 3000, 'ftol': 1e-15})
            hc = project(res['x'], nf)
            sc = compute_score(hc)
            if sc < best_score:
                best_score, best_h = sc, hc.copy()
        except Exception:
            pass

        if time_remaining(t0, budget_s) > 10:
            hc, sc = coord_descent(best_h, len(best_h), n_iter=300, rng=rng)
            if sc < best_score:
                best_score, best_h = sc, hc.copy()

    return best_h, best_score, len(best_h)
