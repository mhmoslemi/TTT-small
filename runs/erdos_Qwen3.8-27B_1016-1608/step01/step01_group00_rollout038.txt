```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def eval_max_count(h):
    corr = np.correlate(h.astype(np.int32), 1 - h.astype(np.int32), mode='full')
    return int(np.max(corr))

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones using thresholding"""
    s = np.sort(h)[::-1]
    threshold = s[n//2 - 1]
    h_bin = (h >= threshold).astype(np.int32)
    # Handle ties
    while np.sum(h_bin) > n//2:
        # Flip a 1 that's closest to threshold from above
        ones = np.where(h_bin == 1)[0]
        idx = np.argmin(h[ones])
        h_bin[ones[idx]] = 0
    while np.sum(h_bin) < n//2:
        zeros = np.where(h_bin == 0)[0]
        idx = np.argmax(h[zeros])
        h_bin[zeros[idx]] = 1
    return h_bin

def steepest_descent(h, time_limit):
    """Greedy: at each step, try all swaps, pick the one that reduces max most"""
    t0 = time.time()
    n = len(h)
    h = h.astype(np.int32).copy()
    curr_max = eval_max_count(h)
    best = h.copy()
    best_max = curr_max
    stall_count = 0
    
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_new_max = curr_max
        best_pair = None
        
        # Try all swaps
        for a in ones:
            for b in zeros:
                h[a], h[b] = h[b], h[a]
                new_max = eval_max_count(h)
                h[a], h[b] = h[b], h[a]
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_pair = (a, b)
                    if best_new_max <= best_max - 2:
                        break
            if best_pair is not None and best_new_max <= best_max - 2:
                break
        
        if best_pair is None:
            break
        
        h[best_pair[0]], h[best_pair[1]] = h[best_pair[1]], h[best_pair[0]]
        curr_max = best_new_max
        
        if curr_max < best_max:
            best_max = curr_max
            best = h.copy()
            stall_count = 0
        else:
            stall_count += 1
    
    return best, best_max
```

```python
def slsqp_polish(h0, n, maxiter=1000):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return eval_c5(x)
    def con(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h0.astype(np.float64), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < obj(h0) + 1e-14:
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h0.astype(np.float64).copy()

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = len(h0)
    except Exception:
        h0 = np.full(70, 0.5)
        n0 = 70

    best_c5 = eval_c5(h0)
    best_h = h0.copy()

    # Phase 1: Binary optimization for multiple n
    n_list = [40, 48, 56, 64, 70, 80, 100]
    if n0 in n_list:
        n_list.remove(n0)
        n_list.insert(0, n0)

    bin_budget = budget_s * 0.65
    per_n_est = bin_budget / len(n_list)

    for n in n_list:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.72:
            break
        tl = min(per_n_est * 1.3, budget_s * 0.72 - elapsed)
        if tl < 8:
            break

        # Multiple initializations
        h_inits = []
        if n == n0:
            h_inits.append(to_binary(h0, n))
        h_inits.append(greedy_construct(n, rng))
        h_inits.append(np.zeros(n, dtype=np.int32))
        h_inits[-1][rng.choice(n, n // 2, replace=False)] = 1
        # Spaced init
        hs = np.zeros(n, dtype=np.int32)
        for i in range(n // 2):
            hs[i * (n // (n // 2))] = 1
        if hs.sum() == n // 2:
            h_inits.append(hs)

        for h_init in h_inits:
            if time.time() - t_start > budget_s * 0.72:
                break
            rem = budget_s * 0.72 - (time.time() - t_start)
            tl_sub = min(tl / max(1, 3), rem)
            if tl_sub < 4:
                break
            h_b, mc = binary_optimize(h_init, tl_sub, rng)
            c5 = mc * 2.0 / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_b.astype(np.float64)

    # Phase 2: Additional refinement on best binary with random subset steepest
    if best_h is not None and np.all((best_h == 0) | (best_h == 1)):
        n_cur = len(best_h)
        h_b = best_h.astype(np.int32)
        if time.time() - t_start < budget_s * 0.82:
            h_b, mc = random_subset_steepest(h_b, 30, rng)
            c5 = mc * 2.0 / n_cur
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_b.astype(np.float64)

    # Phase 3: SLSQP polish (allow non-binary)
    if time.time() - t_start < budget_s * 0.93:
        n_cur = len(best_h)
        h_pol = slsqp_polish(best_h, n_cur, maxiter=2000)
        c5 = eval_c5(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol

    # Phase 4: Try one more SLSQP from a slightly perturbed best
    if time.time() - t_start < budget_s * 0.97:
        n_cur = len(best_h)
        h_pert = repair(best_h + rng.randn(n_cur) * 0.02, n_cur)
        h_pol2 = slsqp_polish(h_pert, n_cur, maxiter=1000)
        c5 = eval_c5(h_pol2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol2

    # Final validation
    best_h = repair(best_h, len(best_h))
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
