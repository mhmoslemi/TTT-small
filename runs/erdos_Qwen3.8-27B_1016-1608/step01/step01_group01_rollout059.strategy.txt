def harmonic_project(h, num_modes=10):
    h_padded = np.pad(h, (0, 50-len(h)), 'constant')
    coeffs = np.fft.rfft(h_padded, n=50)
    h_projected = np.real(np.fft.irfft(0.95 * coeffs[:num_modes], n=50))
    return np.clip(~logistic(-h_projected), 0, 1)  # σ-shaped waveform
