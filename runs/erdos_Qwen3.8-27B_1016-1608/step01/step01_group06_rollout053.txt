```python
                s_try = compute_score(h_try)[0]
                if s_try < current_max - 1e-10:
                    h = h_try
                    current_max = s_try
                    improved = True
                    break
        
        if not improved:
            # Try smaller steps
            for j in range(n):
                grad_j = 0.0
                c_raw = np.correlate(h, 1.0 - h, mode='full')
                c = c_raw * dx
                current_max = np.max(c)
                active = np.where(c > current_max - 1e-8)[0]
                for k_idx in active:
                    lag = k_idx - (n - 1)
                    if lag >= 0:
                        if lag <= j:
                            grad_j += (1.0 - h[j - lag])
                        if j + lag < n:
                            grad_j += -h[j + lag]
                    else:
                        m = -lag
                        if j + m < n:
                            grad_j += (1.0 - h[j + m])
                        if m <= j:
                            grad_j += -h[j - m]
                
                if grad_j > 0:
                    direction = -1.0
                elif grad_j < 0:
                    direction = 1.0
                else:
                    continue
                
                for step in [0.005, 0.001, 0.0005]:
                    h_try = h.copy()
                    h_try[j] = np.clip(h_try[j] + direction * step, 0.0, 1.0)
                    # ... feasibility projection
                    s_try = compute_score(h_try)[0]
                    if s_try < current_max - 1e-12:
                        h = h_try
                        current_max = s_try
                        improved = True
                        break
                
                if improved:
                    break
        
        if not improved:
            break
    
    return h
```

Actually, one key insight I missed: the evaluation function uses `np.correlate(h, 1-h, mode="full") * dx` and takes the max. The "full" correlation has 2n-1 elements, corresponding to lags from -(n-1) to +(n-1). 

For the Erdős problem, the integral ∫h(x)(1-h(x+k))dx is over x where both h(x) and h(x+k) are defined on [0,2]. For a given shift k (in continuous terms), the integration is over the overlap of [0,2] and [-k, 2-k] (or [k, 2+k] depending on sign).

In the discrete setting with dx = 2/n:
- The shift by k grid points corresponds to a continuous shift of k*dx = 2k/n
- The integration is over the valid range of x where both h(x) and h(x+shift) exist

The np.correlate handles this correctly (it only sums over valid indices).

Let me also note: the max over ALL lags (from -(n-1) to n-1) is what matters. For the Erdős problem, we care about all shifts k, and in the discrete setting, all lags from -(n-1) to n-1 correspond to shifts from -(n-1)*dx to (n-1)*dx, i.e., from about -2 to +2.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    if tau < 50:
        return cm
    return cm + (1.0 / tau) * np.log(np.sum(np.exp(tau * (c - cm))))

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h = np.clip(h + (diff / cnt) * mask, 0.0, 1.0)
    return h

def grad_at_lags(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag >= 0:
            if lag < n:
                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] += -h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] += -h[:n - m]
    return grad

def subgradient_optimize(h, n_iters=8000, n_active=3, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    best_h = h.copy()
    best_s = compute_score(h)
    step = 0.05
    patience = 0

    for it in range(n_iters):
        if time_limit and (time.time() - t0) > time_limit:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c = c_raw * dx
        cur_max = np.max(c)

        if cur_max < best_s - 1e-12:
            best_s = cur_max
            best_h = h.copy()
            patience = 0
            step = min(step * 1.02, 0.3)
        else:
            patience += 1
            if patience > 200:
                step *= 0.95

        top_idx = np.argsort(c)[-n_active:]
        lags = (top_idx - (n - 1)).astype(int)
        grad = grad_at_lags(h, lags) / n_active

        h_new = h - step * (grad * dx)
        h_new = project_feasible(h_new, n)

        s_new = compute_score(h_new)
        if s_new < best_s + 0.005:
            h = h_new
        else:
            h = best_h.copy()

    if compute_score(best_h) < best_s:
        best_s = compute_score(best_h)
    return best_h, best_s

def coord_descent(h, max_passes=3000, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    t0 = time.time()
    best_s = compute_score(h)
    best_h = h.copy()

    for p in range(max_passes):
        if time_limit and (time.time() - t0) > time_limit:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c = c_raw * dx
        cur_max = np.max(c)
        active = np.where(c >= cur_max - 1e-9)[0]
        any_improve = False

        for j in range(n):
            if time_limit and (time.time() - t0) > time_limit:
                break
            g = 0.0
            for k_idx in active:
                lag = int(k_idx) - (n - 1)
                if lag >= 0:
                    if lag <= j <= n - 1:
                        g += (1.0 - h[j - lag])
                    if 0 <= j and (j + lag) < n:
                        g += -h[j + lag]
                else:
                    m = -lag
                    if (j + m) < n:
                        g += (1.0 - h[j + m])
                    if m <= j < n:
                        g += -h[j - m]

            if abs(g) < 1e-12:
                continue
            direction = -1.0 if g > 0 else 1.0

            for step in [0.03, 0.015, 0.008, 0.004]:
                h_try = h.copy()
                h_try[j] = np.clip(h_try[j] + direction * step, 0.0, 1.0)
                diff = target - np.sum(h_try)
                if abs(diff) > 1e-12:
                    m2 = (h_try < 1.0 - 1e-12) if diff > 0 else (h_try > 1e-12)
                    m2[j] = False
                    cnt = np.count_nonzero(m2)
                    if cnt > 0:
                        h_try[m2] = np.clip(h_try[m2] + diff / cnt, 0.0, 1.0)
                s_try = compute_score(h_try)
                if s_try < cur_max - 1e-11:
                    h = h_try
                    cur_max = s_try
                    any_improve = True
                    break
        if not any_improve:
            if abs(compute_score(h) - best_s) < 1e-10:
                break
            best_h = h.copy()
            best_s = compute_score(h)

    best_s = compute_score(best_h)
    return best_h, best_s
```
