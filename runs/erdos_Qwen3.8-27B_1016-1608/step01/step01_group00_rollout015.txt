import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    m = k_star - (n - 1)
    
    g = np.zeros(n)
    for i in range(n):
        # Term 1: (1-h[i-m]) if 0 <= i-m < n
        im = i - m
        if 0 <= im < n:
            g[i] += 1.0 - h[im]
        # Term 2: -h[i+m] if 0 <= i+m < n
        ip = i + m
        if 0 <= ip < n:
            g[i] -= h[ip]
    g *= dx
    
    h_new = h - lr * g
    h_new = repair(h_new, n)
    return h_new, k_star, corr[k_star] * dx

def slsqp_optimize(h0, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < compute_c5(h0) + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:

        else:
            mask = h > 1e-10

            break
        h[mask] += diff / float(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)

def corr_grad(h, n, k):
    """Gradient of corr[k] w.r.t. h, where m = k-(n-1) is the physical lag."""
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    # (1-h[i-m]) for valid i-m
    lo = max(0, m)
    hi = min(n, n + m)
    g[lo:hi] += 1.0 - h[lo - m:hi - m]
    # -h[i+m] for valid i+m
    lo2 = max(0, -m)
    hi2 = min(n, n - m)
    g[lo2:hi2] -= h[lo2 + m:hi2 + m]
    return g * dx

    dx = 2.0 / n

        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 7: Coordinate descent polish
    if time.time() - t0 < budget_s * 0.88:
        h = coord_descent(h_best, n, n_sweeps=8)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 8: Final SLSQP high-iteration polish
    if time.time() - t0 < budget_s * 0.95:
        h = slsqp_optimize(h_best, n, maxiter=5000, ftol=1e-16)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 9: Try alternative n values
    for n2 in [50, 35, 100, 140]:
        if time.time() - t0 > budget_s * 0.97:
            break
        if n2 == n:
            continue
        xp = np.linspace(0.0, 2.0, n)
        xn = np.linspace(0.0, 2.0, n2)
        h_interp = np.interp(xn, xp, h_best)
        h_interp = repair(h_interp, n2)
        h2 = slsqp_optimize(h_interp, n2, maxiter=600)
        c2 = compute_c5(h2)
        if c2 < c5_best:
            h_best, c5_best, n = h2, c2, n2

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
