        h_kick, c5_kick = coord_sweep(h_kick, n, max_sweeps=40)
        if c5_kick < best_c5:
            best_c5 = c5_kick
            best_h = h_kick.copy()

    # Phase 5: Try different n values (6%)
    t5 = t_start + budget_s * 0.97
    for n_trial in [44, 48, 50, 55, 60, 70, 80, 100]:
        if time.time() > t5:
            break
        if n_trial == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_trial)
        h_trial = np.interp(x_new, x_old, best_h)
        h_trial = project_feasible(h_trial, n_trial)
        h_trial, c5_trial = coord_sweep(h_trial, n_trial, max_sweeps=80)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()

    # Final polish
    if time.time() < t_start + budget_s * 0.995:
        final_n = len(best_h)
        best_h, c5_fin = coord_sweep(best_h, final_n, max_sweeps=40)
        if c5_fin < best_c5:
            best_c5 = c5_fin

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
