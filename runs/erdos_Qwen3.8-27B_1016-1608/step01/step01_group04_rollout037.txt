    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        order = np.random.permutation(n) if randomize else np.arange(n)
        for i in order:
            Di = coord_D(h, i, n)
            A = corr - h[i] * Di
            vn = solve_1d(A, Di, h[i])
            if abs(vn - h[i]) > 1e-13:
                h[i] = vn
                corr = A + vn * Di
        h = repair(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return h

def do_slsqp(h, n, eps, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c_arr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c_arr)
        s = np.clip((c_arr - m) / eps, -40.0, 40.0)
        return m + eps * np.log(np.sum(np.exp(s)))
    def obj_jac(x):
        c_arr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c_arr)
        s = np.clip((c_arr - m) / eps, -40.0, 40.0)
        w = np.exp(s) / np.sum(np.exp(s))
        diff = np.correlate(x, 1.0 - x, mode='full') * dx
        # numerical gradient
        g = np.zeros(n)
        for i in range(n):
            xp = x.copy()
            xp[i] += 1e-7
            c2 = np.correlate(xp, 1.0 - xp, mode='full') * dx
            m2 = np.max(c2)
            s2 = np.clip((c2 - m2) / eps, -40.0, 40.0)
            f2 = m2 + eps * np.log(np.sum(np.exp(s2)))
            xm = x.copy()
            xm[i] -= 1e-7
            c3 = np.correlate(xm, 1.0 - xm, mode='full') * dx
            m3 = np.max(c3)
            s3 = np.clip((c3 - m3) / eps, -40.0, 40.0)
            f3 = m3 + eps * np.log(np.sum(np.exp(s3)))
            g[i] = (f2 - f3) / (2e-7)
        return g
    def eqf(x):
        return np.sum(x) - n / 2.0
    def eqj(x):
        return np.ones(n)
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    r = spmin(obj, h, method='SLSQP', bounds=bds, constraints=cs,
              options={'maxiter': maxiter, 'ftol': 1e-14})
    if time.time() > deadline:
        return h
    return repair(r.x, n)

def snap_binary(h, n, threshold=0.05):
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1.0 - threshold] = 1.0
    return repair(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)
    n = len(h0)
    h0 = repair(h0, n)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Coordinate descent (cyclic)
    dl1 = min(t0 + budget_s * 0.25, deadline - 300)
    h = do_cd(h0.copy(), n, 500, dl1, randomize=False)
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: SLSQP multi-eps refinement
    for eps in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline - 400:
            break
        h2 = do_slsqp(best_h.copy(), n, eps, 200, deadline - 300)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: Coordinate descent (randomized order)
    dl3 = min(time.time() + budget_s * 0.2, deadline - 300)
    h = do_cd(best_h.copy(), n, 400, dl3, randomize=True)
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Binary snap exploration
    for trial in range(10):
        if time.time() > deadline - 250:
            break
        h_snap = snap_binary(best_h, n, threshold=0.03 + 0.02 * trial)
        c_snap = compute_c5(h_snap)
        if c_snap < best_c:
            best_c, best_h = c_snap, h_snap.copy()
            h = do_cd(best_h.copy(), n, 100, deadline - 250, randomize=True)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 5: Perturbation + short CD restarts
    for r in range(80):
        if time.time() > deadline - 120:
            break
        t = np.random.random()
        if t < 0.2:
            hp = best_h + np.random.randn(n) * 0.03
        elif t < 0.4:
            hp = best_h + np.random.randn(n) * 0.06
        elif t < 0.55:
            hp = best_h.copy()
            idx = np.random.randint(n)
            hp[idx] = 1.0 - hp[idx]
        elif t < 0.7:
            hp = best_h.copy()
            s = np.random.randint(0, max(1, n - 8))
            l = np.random.randint(3, min(9, n - s))
            hp[s:s + l] = 1.0 - hp[s:s + l]
        elif t < 0.85:
            hp = best_h + 0.04 * np.sin(2 * np.pi * np.random.randint(1, 6) * np.arange(n) / n + np.random.uniform(0, 2 * np.pi))
        else:
            hp = best_h.copy()
            idx = np.random.choice(n, size=max(1, n // 6), replace=False)
            hp[idx] = np.clip(hp[idx] + np.random.randn(len(idx)) * 0.1, 0, 1)
        hp = repair(hp, n)
        tlim = min(time.time() + 6, deadline - 120)
        h = do_cd(hp, n, 60, tlim, randomize=True)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 60:
        h2 = do_slsqp(best_h.copy(), n, 0.0003, 400, deadline - 60)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 7: Fine CD
    if time.time() < deadline - 20:
        h = do_cd(best_h.copy(), n, 300, deadline - 20, randomize=False)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 8: Tiny random walk
    if time.time() < deadline - 5:
        for _ in range(8000):
            if time.time() > deadline - 3:
                break
            hp = best_h + np.random.randn(n) * 0.003
            hp = repair(hp, n)
            c_try = compute_c5(hp)
            if c_try < best_c:
                best_c, best_h = c_try, hp.copy()

    final_h = repair(best_h, n)
    final_c = compute_c5(final_h)
    return (final_h.tolist(), float(final_c), n)
