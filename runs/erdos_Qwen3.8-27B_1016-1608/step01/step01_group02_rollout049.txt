import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_box_sum(h, target):
    h = np.clip(h.astype(float, copy=True), 0.0, 1.0)
    for _ in range(2000):
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 0
            na = mask.sum()
            if na == 0:
                break
            d = min(excess / na, h[mask].min())
            h[mask] -= d
        else:
            mask = h < 1
            na = mask.sum()
            if na == 0:
                break
            d = min(-excess / na, (1.0 - h[mask]).min())
            h[mask] += d
    return h

def smooth_obj(h, dx, T):
    c = get_corr(h)
    cm = c.max()
    return (cm + np.log(np.sum(np.exp(T * (c - cm))) / len(c)) / T) * dx

def smooth_grad(h, dx, T):
    n = len(h)
    c = get_corr(h)
    cm = c.max()
    s = T * (c - cm)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(n):
        ke = min(m + n, 2 * n - 1)
        grad[m] = dx * np.dot(w[m:ke], 1.0 - 2.0 * h[:ke - m])
    return grad

def sum_con_fun(x, target):
    return x.sum() - target

def sum_con_jac(x, target):
    return np.ones_like(x)

def slsqp_smooth(h0, dx, T, target, bounds, maxiter):
    cons = ({'type': 'eq', 'fun': sum_con_fun, 'args': (target,),
             'jac': sum_con_jac},)
    try:
        res = minimize(smooth_obj, h0, args=(dx, T), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_box_sum(res.x, target)
    except Exception:
        return None

def direct_obj_wrap(h, dx):
    return compute_c5(h, dx)

def slsqp_direct(h0, dx, target, bounds, maxiter):
    cons = ({'type': 'eq', 'fun': sum_con_fun, 'args': (target,),
             'jac': sum_con_jac},)
    try:
        res = minimize(direct_obj_wrap, h0, args=(dx,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return None

    best_h = np.array(initial_h_values, dtype=float).copy()

    init_c5 = best_c5

    def remaining():
        return budget_s - (time.time() - t0)

    h_warm = best_h.copy()

    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
