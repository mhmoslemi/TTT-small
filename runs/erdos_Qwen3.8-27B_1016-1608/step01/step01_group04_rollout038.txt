def coord_descent(h, n, max_passes=100, tol=1e-7):
    h = h.copy()
    for p in range(max_passes):
        improved = False
        for j in range(n):
            # Save old value
            old = h[j]
            # Compute correlation with h[j]=0
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            # Compute correlation with h[j]=1
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            # Linear interpolation: c(a) = c0 + a*(c1-c0)
            delta = c1 - c0
            # Find a in [0,1] minimizing max(c0 + a*delta)
            # This is a 1D convex minimization
            a = minimize_1d(c0, delta)
            h[j] = a
            if abs(a - old) > 1e-10:
                improved = True
        if not improved:
            break
    return h

def minimize_1d(c0, delta):
    """Minimize max_k(c0[k] + a*delta[k]) over a in [0,1]"""
    # f(a) = max(c0 + a*delta) is convex in a
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(c0 + c * delta)
    fd = np.max(c0 + d * delta)
    for _ in range(30):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(c0 + c * delta)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(c0 + d * delta)
    return (a + b) / 2

from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            h += d / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    d = n / 2.0 - np.sum(h)
    if abs(d) > 1e-14:
        for i in range(n):
            if d > 0 and h[i] < 1.0:
                adj = min(d, 1.0 - h[i])
                h[i] += adj
                d -= adj
            elif d < 0 and h[i] > 0.0:
                adj = max(d, -h[i])
                h[i] += adj
                d -= adj
            if abs(d) < 1e-14:
                break
    return h

def minimize_1d(c0, delta):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(c0 + c * delta))
    fd = float(np.max(c0 + d * delta))
    for _ in range(40):

            c = b - gr * (b - a)
            fc = float(np.max(c0 + c * delta))
        else:
            a, c, fc = c, d, fc
            d = a + gr * (b - a)
            fd = float(np.max(c0 + d * delta))
    return 0.5 * (a + b)

        for j in range(n):

            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = old
            delta = c1 - c0
            a = minimize_1d(c0, delta)
            h[j] = a
            max_change = max(max_change, abs(a - old))
        h = repair(h, n)
        if max_change < 1e-9:
            break
    return h

def obj_smooth(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(c))
    if eps < 1e-13:
        return m
    z = np.clip((c - m) / eps, -50.0, 50.0)
    return m + eps * float(np.log(np.sum(np.exp(z))))

def eqf(x):
    return float(np.sum(x) - _opt['n'] / 2.0)

def eqj(x):
    return np.ones(_opt['n'])

def do_slsqp(h, n, eps, maxiter, ftol):
    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    r = minimize(obj_smooth, h, method='SLSQP', bounds=bds, constraints=cons,
                 args=(n, eps), options={'maxiter': maxiter, 'ftol': ftol})
    return repair(r.x, n)

def do_lbfgs_pen(h, n, eps, maxiter, mu):
    _opt['n'] = n
    _opt['eps'] = eps
    def fpen(x):
        return obj_smooth(x, n, eps) + mu * (np.sum(x) - n / 2.0) ** 2
    bds = [(0.0, 1.0)] * n
    r = minimize(fpen, h, method='L-BFGS-B', bounds=bds,
                 options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-10})
    return repair(r.x, n)

def try_binary(h, n):
    best_h, best_c = h.copy(), compute_c5(h)

                hb[idx[order[i]]] = 1.0
        c = compute_c5(hb)
        if c < best_c:
            best_c, best_h = c, hb
    return best_h, best_c
