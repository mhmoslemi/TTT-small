```python
    #   (optional) step constraint: -step_max ≤ δh[j] ≤ step_max
    
    m = len(active)
    # c vector: minimize [0, ..., 0, 1] (last component is t)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag k (0-indexed in active):
    #   sum_j grads[k,j]*δh[j] - t ≤ cmax - corr[active[k]]
    A_ub = np.zeros((m, n + 1))
    for i in range(m):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
    b_ub = cmax - corr[active]
    
    # Box constraints on δh: -h[j] ≤ δh[j] ≤ 1-h[j]
    # These become bounds in linprog
    bounds = [(-h[j], 1 - h[j]) for j in range(n)] + [(0, np.inf)]
    
    # Equality: sum δh = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

This LP approach is much better than plain subgradient! It finds the optimal step direction within the linearized model.

But wait - the constraint `sum δh = 0` is too strict. We want `sum(h + δh) = target`, which is `sum(h) + sum(δh) = target`. Since we project h to have sum = target already, this becomes `sum(δh) = 0`. OK that's correct.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(100):
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 0
            if not mask.any():
                break
            s = min(excess / mask.sum(), h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            if not mask.any():
                break
            s = min(-excess / mask.sum(), (1.0 - h[mask]).min())
            h[mask] += s
    return h

def lag_gradient(d, h, n):
    grad = np.zeros(n)
    if d >= 0:
        for i in range(n - d):
            grad[i] += (1.0 - h[i + d])
            grad[i + d] -= h[i]
    else:
        dd = -d
        for i in range(dd, n):
            grad[i] += (1.0 - h[i - dd])
            grad[i - dd] -= h[i]
    return grad

def lp_step(h, target, n, tol=1e-6, step_max=0.08):
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    active = np.where(corr >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grads = np.array([lag_gradient(k - n + 1, h, n) for k in active])
    m = len(active)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    for i in range(m):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
    b_ub = cmax - corr[active]
    bounds = [(-min(h[j], step_max), min(1 - h[j], step_max)) for j in range(n)]
    bounds.append((0, np.inf))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n] - cmax
    except Exception:
        pass
    return None, None

def lp_descent(h0, target, n, max_iter=5000, tol=1e-6, step_init=0.08):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / n)
    step = step_init
    stall = 0
    for it in range(max_iter):
        step_arr, delta_t = lp_step(h, target, n, tol=tol, step_max=step)
        if step_arr is not None:
            for alpha in [1.0, 0.5, 0.25, 0.1]:
                h_new = project_h(h + alpha * step_arr, target)
                c5_new = compute_c5(h_new, 2.0 / n)
                if c5_new < best_c5 - 1e-15:
                    h = h_new
                    best_h = h.copy()
                    best_c5 = c5_new
                    stall = 0
                    break
            else:
                stall += 1
                if stall > 30:
                    step = max(0.005, step * 0.7)
                    if step < 0.006:
                        break
                h_new = project_h(h + 0.5 * step_arr, target)
                h = h_new
        else:
            break
        if it % 500 == 0 and it > 0:
            step = min(0.15, step * 1.1)
    return best_h, best_c5

def smooth_max_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    scaled = temp * (corr - m)
    return (m + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_h(np.array(initial_h_values, dtype=float), target)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    def update(h_new, c5_new):
        nonlocal best_h, best_c5
        if h_new is not None and c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 1: LP-based descent with decreasing tolerance
    for atol in [1e-3, 1e-5, 1e-7, 1e-9, 1e-10]:
        if elapsed() > budget_s - 80:
            break
        h_new, c5_new = lp_descent(best_h, target, n, max_iter=4000, tol=atol, step_init=0.06)
        update(h_new, c5_new)

    # Phase 2: Annealing with LP refinement
    for temp in [0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
        if elapsed() > budget_s - 40:
            break
        for trial in range(4):
            if elapsed() > budget_s - 40:
                break
            h0 = project_h(best_h + rng.randn(n) * temp, target)
            h_new, c5_new = lp_descent(h0, target, n, max_iter=2000, tol=1e-8, step_init=0.04)
            update(h_new, c5_new)

    # Phase 3: Smooth max SLSQP
    if elapsed() < budget_s - 40:
        bounds = [(0.0, 1.0)] * n
        for temp in [50.0, 150.0, 500.0]:
            if elapsed() > budget_s - 25:
                break
            try:
                res = minimize(smooth_max_c5, best_h, args=(dx, temp), method='SLSQP',
                              bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                              options={'maxiter': 200, 'ftol': 1e-14})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new, dx)
                update(h_new, c5_new)
            except Exception:
                pass
        # Direct SLSQP on hard max
        try:
            res = minimize(compute_c5, best_h, args=(dx,), method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                          options={'maxiter': 400, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new, dx)
            update(h_new, c5_new)
        except Exception:
            pass

    # Phase 4: Coordinate descent micro-polish
    if elapsed() < budget_s - 10:
        h_c = best_h.copy()
        c5_c = best_c5
        step_cd = 0.01
        stall_cd = 0
        while elapsed() < budget_s - 3:
            idx = rng.randint(n)
            improved = False
            for delta in [step_cd, -step_cd, step_cd * 0.3, -step_cd * 0.3, step_cd * 0.1, -step_cd * 0.1]:
                h_t = h_c.copy()
                h_t[idx] = np.clip(h_t[idx] + delta, 0, 1)
                h_t = project_h(h_t, target)
                c5_t = compute_c5(h_t, dx)
                if c5_t < c5_c - 1e-16:
                    h_c, c5_c, improved = h_t, c5_t, True
                    break
            if improved:
                stall_cd = 0
                if c5_c < best_c5 - 1e-16:
                    best_h, best_c5 = h_c.copy(), c5_c
            else:
                stall_cd += 1
                if stall_cd > 500:
                    step_cd *= 0.8
                    stall_cd = 0
                    if step_cd < 5e-6:
                        break

    # Phase 5: Final LP polish
    if elapsed() < budget_s - 5:
        h_new, c5_new = lp_descent(best_h, target, n, max_iter=2000, tol=1e-11, step_init=0.01)
        update(h_new, c5_new)

    best_h = project_h(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
```
