import numpy as np
from scipy.optimize import minimize
import time

_tb = 1100.0
_ts = 0.0
_gt = 0.0

def _tl():
    return max(0.0, _tb - (time.time() - _ts))

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if float(np.sum(np.clip(h - m, 0, 1))) > t:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _up(h, nn):
    return np.interp(np.arange(nn) / nn, np.arange(len(h)) / len(h), h)

def _bin(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _smooth(n, rng):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for _ in range(rng.randint(3, 8)):
        h += rng.uniform(0.2, 0.6) * np.sin(rng.uniform(1, 7) * np.pi * x + rng.uniform(0, 6.28))
    return _proj(h, n / 2.0)

def _sobj(x):
    return _c5(x)

def _scon(x):
    return float(np.sum(x) - _gt)

def _slsqp(h0, n, mi=300):
    global _gt
    _gt = n / 2.0
    h0 = _proj(h0.copy(), _gt)
    try:
        r = minimize(_sobj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        ho = _proj(np.array(r.x, dtype=float), _gt)
        return ho, _c5(ho)
    except Exception:
        return h0, _c5(h0)

def _coord(h, n, rng, ns=80):
    t = n / 2.0
    h = _proj(h.copy(), t)
    best = _c5(h)
    pd = [0.1, 0.05, 0.02]
    for _ in range(ns):
        imp = False
        for i in rng.permutation(n):
            for _ in range(4):
                j = rng.randint(n)
                if j == i:
                    continue
                for d in pd:
                    for s in (1.0, -1.0):
                        di = s * d
                        if min(h[i] + di, h[j] - di) < 0 or max(h[i] + di, h[j] - di) > 1:
                            continue
                        h[i] += di; h[j] -= di
                        v = _c5(h)
                        if v < best:
                            best = v; imp = True
                        else:
                            h[i] -= di; h[j] += di
            for c in (0.0, 1.0):
                if abs(c - h[i]) < 1e-9:
                    continue
                h2 = h.copy(); h2[i] = c
                hp = _proj(h2, t)
                v = _c5(hp)
                if v < best:
                    best = v; h = hp; imp = True
        if not imp:
            break
    h = _proj(h, t)
    return h, _c5(h)

def _anneal(h, n, rng, ni=30000, T0=0.0002):
    hb = (h > 0.5).astype(float)
    no = int(np.sum(hb))
    tn = n // 2
    oi = np.where(hb == 1)[0]
    zi = np.where(hb == 0)[0]
    if no > tn:
        hb[rng.choice(oi, no - tn, replace=False)] = 0
    elif no < tn:
        hb[rng.choice(zi, tn - no, replace=False)] = 1
    oi = np.where(hb == 1)[0]
    zi = np.where(hb == 0)[0]
    bh = hb.copy(); bv = _c5(hb)
    for it in range(ni):
        T = T0 * (1e-4) ** (it / max(1, ni - 1))
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        hb[i], hb[j] = 0.0, 1.0
        v = _c5(hb)
        d = v - bv
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            if v < bv:
                bv = v; bh = hb.copy()
            oi = np.where(hb == 1)[0]
            zi = np.where(hb == 0)[0]
        else:
            hb[i], hb[j] = 1.0, 0.0
    return bh, bv

    return bh, bc, bn
