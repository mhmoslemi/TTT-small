import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smoothed_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * logsumexp(beta * corr) / beta)

_beta_box = [5.0]

def _obj_smoothed(x):
    return smoothed_c5(x, _beta_box[0])

def _obj_hard(x):
    return compute_c5(x)

def _sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, obj_func, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and res.fun < obj_func(h0) + 1e-14:
            return repair(res.x, n)
    except Exception:
        pass
    return h0

def lbfgs_opt(h0, n, obj_func, maxiter=1000):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        if res.x is not None and res.fun < obj_func(h0) + 1e-14:
            return repair(res.x, n)
    except Exception:
        pass
    return h0

def binary_flip_search(h_cont, n, budget_frac=0.1):
    """Local search on near-binary version: flip small continuous adjustments."""
    h = h_cont.copy()
    target = n / 2.0
    best_c = compute_c5(h)
    step = 2.0 / n
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) > 0.5:
            idx = (np.random.rand(n) < 0.5)
            h = repair(h, n)
            continue
        i = np.random.randint(n)
        direction = np.random.choice([-1, 1])
        h[i] += direction * step * 0.5
        h = np.clip(h, 0.0, 1.0)
        h = repair(h, n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
    return h, best_c

def structured_perturb(h0, n, rng):
    """Create structured (low-frequency) perturbations."""
    mode = rng.randint(1, 5)
    h = h0.copy()
    x = np.arange(n) / n
    amp = rng.uniform(0.02, 0.15)
    phase = rng.uniform(0, 2 * np.pi)
    h += amp * np.sin(2 * np.pi * mode * x + phase)
    return repair(h, n)
