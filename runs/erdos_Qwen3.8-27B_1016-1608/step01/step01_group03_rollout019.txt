def coord_optimize(h, j, n):
    save = h[j]
    h[j] = 0.0
    C_base = np.correlate(h, 1.0 - h, mode="full")
    m = n - 1
    s_vals = np.arange(-(n - 1), n)
    jp = j + s_vals
    jm = j - s_vals
    hp = np.where((jp >= 0) & (jp < n), h[np.clip(jp, 0, n - 1)], 0.0)
    hm = np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n - 1)], 0.0)
    a = 1.0 - hp - hm
    a[m] = 0.0
    t_grid = np.linspace(0.0, 1.0, 201)
    corrs = C_base[np.newaxis, :] + t_grid[:, np.newaxis] * a[np.newaxis, :]
    corrs[:, m] = C_base[m] + t_grid - t_grid ** 2
    vals = corrs.max(axis=1)
    best_t = t_grid[vals.argmin()]
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.005)
        hi = min(1.0, best_t + 0.005)
        t_fine = np.linspace(lo, hi, 201)
        cf = C_base[np.newaxis, :] + t_fine[:, np.newaxis] * a[np.newaxis, :]
        cf[:, m] = C_base[m] + t_fine - t_fine ** 2
        vf = cf.max(axis=1)
        best_t = t_fine[vf.argmin()]
    h[j] = best_t
    return best_t

def slsqp_polish(h, n, dx, target):
    bnds = [(0.0, 1.0)] * n
    tgt = target
    def obj_func(x):
        return compute_c5(np.clip(x, 0.0, 1.0), dx)
    def sum_constr(x):
        return np.sum(x) - tgt
    cons = [{'type': 'eq', 'fun': sum_constr}]
    try:
        r = minimize(obj_func, h.copy(), method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': 400, 'ftol': 1e-16})
        h2 = repair(r.x, target)
        return h2, compute_c5(h2, dx)
    except Exception:
        return h, compute_c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = 1.0
    best_h = None
    best_n = 70

    try:
        h_orig = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_orig)
    except Exception:
        h_orig = np.ones(70) * 0.5
        n_orig = 70

    if time.time() - t0 > budget_s * 0.1:
        pass

    # Build candidate list: (n, h_init)
    candidates = []
    candidates.append((n_orig, h_orig.copy()))

    # Resampled to other n
    for n_alt in [50, 56, 60, 80, 90, 100, 64, 72]:
        if n_alt == n_orig:
            continue
        x_o = np.linspace(0.0, 2.0, n_orig)
        x_a = np.linspace(0.0, 2.0, n_alt)
        h_r = np.interp(x_a, x_o, h_orig)
        candidates.append((n_alt, h_r))

    # Random perturbations of original
    for k in range(4):
        sc = 0.05 + 0.1 * k
        h_p = h_orig + np.random.randn(n_orig) * sc
        candidates.append((n_orig, h_p))

    # Binary-ish random for n_orig
    for k in range(3):
        h_b = np.zeros(n_orig)
        idx = np.random.choice(n_orig, n_orig // 2, replace=False)
        h_b[idx] = 1.0
        h_b += np.random.rand(n_orig) * 0.3
        candidates.append((n_orig, h_b))

    # Periodic structured
    for period in [7, 10, 14, 5]:
        h_per = np.zeros(n_orig)
        for i in range(n_orig):
            h_per[i] = 1.0 if (i % (2 * period)) < period else 0.0
        h_per += np.random.randn(n_orig) * 0.15
        candidates.append((n_orig, h_per))

    # Process each candidate
    for ci, (n, h_init) in enumerate(candidates):
        if time.time() - t0 > budget_s - 20:
            break
        dx = 2.0 / n
        target = n / 2.0
        h = repair(h_init, target)
        c5 = compute_c5(h, dx)

        # Coordinate descent
        max_sweeps = 3000 if n <= 70 else 1500
        stall = 0
        rng_local = np.random.RandomState(seed + ci * 1000)

        for sweep in range(max_sweeps):
            if time.time() - t0 > budget_s - 40:
                break
            old_c5 = c5
            # Random order
            order = rng_local.permutation(n)
            for j in order:
                coord_optimize(h, j, n)
            h = repair(h, target)
            c5 = compute_c5(h, dx)
            if c5 < old_c5 - 1e-10:
                stall = 0
            else:
                stall += 1
                if stall > 0 and stall % 50 == 0:
                    # Random kick to escape
                    kick = rng_local.randn(n) * 0.02
                    h = repair(h + kick, target)
                    c5 = compute_c5(h, dx)
                    stall = 0
            if stall > 200:
                break

        # SLSQP polish
        if time.time() - t0 < budget_s - 15:
            h, c5 = slsqp_polish(h, n, dx, target)

        if c5 < best_c5:
            best_c5 = c5
            best_n = n
            best_h = h.copy()

        # If we already beat target, do extra polish on best
        if best_c5 < 0.3809 and time.time() - t0 < budget_s - 10:
            h_extra = best_h.copy()
            dn = best_n
            dx_e = 2.0 / dn
            tgt_e = dn / 2.0
            for _ in range(5):
                if time.time() - t0 > budget_s - 5:
                    break
                order2 = np.random.permutation(dn)
                for j in order2:
                    coord_optimize(h_extra, j, dn)
                h_extra = repair(h_extra, tgt_e)
                c5e = compute_c5(h_extra, dx_e)
                if c5e < best_c5:
                    best_c5 = c5e
                    best_h = h_extra.copy()
            if time.time() - t0 < budget_s - 5:
                h_extra, c5e = slsqp_polish(best_h, best_n, 2.0 / best_n, best_n / 2.0)
                if c5e < best_c5:
                    best_c5 = c5e
                    best_h = h_extra

    # Final validation
    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    # Ensure sum constraint is tight
    s = best_h.sum() * (2.0 / best_n)
    if abs(s - 1.0) > 1e-10:
        best_h = repair(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
