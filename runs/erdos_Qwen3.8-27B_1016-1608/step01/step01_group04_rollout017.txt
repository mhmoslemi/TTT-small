import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_G = {'n': 80, 'eps': 0.01}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n, iters=50):
    h = np.array(h, dtype=float).copy()
    tgt = n / 2.0
    for _ in range(iters):
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def smax(x):
    n, eps = _G['n'], _G['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    if eps < 1e-14:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.exp(s).sum()))

def cons_eq(x):
    return float(x.sum() - _G['n'] / 2.0)

def cons_jac(x):
    return np.ones(_G['n'])

def slsqp_opt(h0, n, eps, mi, ft):
    _G['n'] = n
    _G['eps'] = eps
    bds = [(0.0, 1.0)] * n
    r = minimize(smax, h0, method='SLSQP', bounds=bds,
                constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}],
                options={'maxiter': mi, 'ftol': ft})
    return repair(r.x, n)

def de_obj(x):
    n = _G['n']
    h = repair(x, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    eps = 0.004
    s = np.clip((c - m) / eps, -50.0, 50.0)
    val = m + eps * np.log(np.exp(s).sum())
    return float(val + 30.0 * (h.sum() - n / 2.0) ** 2)

def make_seeds(n, parent, seed_val):
    rng = np.random.RandomState(seed_val)
    out = []
    if parent is not None:
        if len(parent) == n:
            out.append(parent.copy())
        else:
            xo = np.linspace(0, 2, len(parent), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            out.append(repair(np.interp(xn, xo, parent, period=2.0), n))
    for p in [2, 3, 4, 5, 6, 7, 8, 10, 12, 16]:
        hb = ((np.arange(n) % p) < (p + 1) // 2).astype(float)
        out.append(repair(hb, n))
    for k in [1, 2, 3, 4, 5, 7]:
        ph = rng.uniform(0, 6.2832)
        hs = 0.5 + 0.45 * np.sin(2 * np.pi * k * np.arange(n) / n + ph)
        out.append(repair(hs, n))
    for _ in range(5):
        out.append(repair(rng.uniform(0, 1, n), n))
    return out

    t0 = time.time()

                xn = np.linspace(0, 2, n, endpoint=False)
                hbest = np.interp(xn, xo, hbest, period=2.0)
            for sh in range(1, max(2, n // 8)):

        for s in seeds:

                c = compute_c5(hnew)

            except Exception:
                pass
        # Tiny random perturbations
        for _ in range(5000):
            if time.time() - t0 > budget_s - 5:
                break
            htry = bh + np.random.randn(nb) * 0.003
            htry = repair(htry, nb)
            c = compute_c5(htry)
            if c < bc:

                htry[idx] = 1.0 - htry[idx]
            htry = repair(htry, nb)
            c = compute_c5(htry)
            if c < bc:
                bc, bh = c, htry

    return (bh.tolist(), float(bc), len(bh))
