import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_obj(x, dx, beta, target, pw):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    s = np.clip(beta * (corr - mc), -700, 700)
    val = (mc + np.log(np.sum(np.exp(s)))) / beta * dx
    val += pw * (np.sum(h) - target) ** 2
    return val

def block_opt(h, dx, target, n_blocks, blk_sz, seed_val):
    rng = np.random.RandomState(seed_val)
    n = len(h)
    h = h.copy()
    cur_best = compute_c5(h, dx)
    for bi in range(n_blocks):
        idx = rng.choice(n, blk_sz, replace=False)
        h_fix = h.copy()
        fixed_sum = np.sum(h_fix) - np.sum(h_fix[idx])
        t_blk = target - fixed_sum
        h_blk = h[idx].copy()
        def f_blk(xb):
            xf = h_fix.copy()
            xf[idx] = np.clip(xb, 0, 1)
            corr = np.correlate(xf, 1.0 - xf, mode="full")
            mc = np.max(corr)
            pen = 50.0 * (np.sum(xf) - target) ** 2
            return mc * dx + pen
        bnds_b = [(0.0, 1.0)] * blk_sz
        try:
            r = minimize(f_blk, h_blk, method='L-BFGS-B', bounds=bnds_b,
                        options={'maxiter': 80, 'ftol': 1e-14, 'maxfun': 2000})
            ht = h_fix.copy()
            ht[idx] = np.clip(r.x, 0, 1)
            ht = repair(ht, target)
            ct = compute_c5(ht, dx)
            if ct < cur_best - 1e-13:
                h = ht
                cur_best = ct
        except Exception:
            pass
    return h, cur_best

def binary_local(h, dx, target, n_sweeps=20, seed_val=0):
    rng = np.random.RandomState(seed_val)
    n = len(h)
    hr = np.round(h).astype(float)
    hr = repair(hr, target)
    cur = compute_c5(hr, dx)
    for _ in range(n_sweeps):
        improved = False
        ones = np.where(hr > 0.5)[0]
        zeros = np.where(hr < 0.5)[0]
        for _sw in range(min(len(ones) * len(zeros), 5000)):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            hr[i], hr[j] = hr[j], hr[i]
            c2 = compute_c5(hr, dx)
            if c2 < cur - 1e-12:
                cur = c2
                improved = True
            else:
                hr[i], hr[j] = hr[j], hr[i]
        if not improved:
            break
    return hr, cur

def coord_desc(h, dx, target, sweeps=30):
    n = len(h)
    h = h.copy()
    for _ in range(sweeps):
        imp = False
        corr = np.correlate(h, 1.0 - h, mode="full")
        cmax = np.max(corr)
        for i in range(n):
            old = h[i]
            for t in [0.0, 0.25, 0.5, 0.75, 1.0, max(0, old-0.1), min(1, old+0.1)]:
                if abs(t - old) < 0.02:
                    continue
                h[i] = t
                c1 = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if c1 < cmax - 1e-12:
                    cmax = c1
                    imp = True
            h[i] = old if not imp else h[i]
            if imp:
                h[i] = t
        if not imp:
            break
    return repair(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_c5, best_h, best_n = float('inf'), None, 70

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5
    n = len(h0)
    tgt, dx = n / 2.0, 2.0 / n
    h0 = repair(h0, tgt)
    best_c5 = compute_c5(h0, dx)
    best_h, best_n = h0.copy(), n
    bnds = [(0.0, 1.0)] * n

    def el():
        return time.time() - t0

    # Phase 1: LSE annealing with L-BFGS-B
    for beta in [3, 5, 10, 20, 50, 100, 300, 1000, 5000]:
        if el() > budget_s - 200:
            break
        def f1(x):
            return lse_obj(x, dx, beta, tgt, 30.0)
        try:
            r = minimize(f1, best_h, method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': 400, 'ftol': 1e-15, 'maxfun': 20000})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass

    # Phase 2: Block coordinate optimization (multiple block sizes)
    for bs in [5, 10, 15, 20]:
        if el() > budget_s - 180:
            break
        nb = max(50, int((budget_s - 180 - el()) / 4.0))
        hb, cb = block_opt(best_h, dx, tgt, nb, bs, seed + bs)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 3: Basin hopping
    for i in range(80):
        if el() > budget_s - 100:
            break
        sc = 0.03 + 0.12 * (1.0 - i / 80.0)
        hp = repair(best_h + np.random.randn(n) * sc, tgt)
        def f2(x):
            return lse_obj(x, dx, 1000, tgt, 50.0)
        try:
            r = minimize(f2, hp, method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': 200, 'ftol': 1e-14, 'maxfun': 8000})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass

    # Phase 4: Binary local search
    if el() < budget_s - 40:
        hb, cb = binary_local(best_h, dx, tgt, n_sweeps=15, seed_val=seed)
        if cb < best_c5:
            best_c5, best_h = cb, hb
        # Also try binary search on perturbed version
        hp2 = repair(best_h + np.random.randn(n) * 0.3, tgt)
        hb2, cb2 = binary_local(hp2, dx, tgt, n_sweeps=10, seed_val=seed+1)
        if cb2 < best_c5:
            best_c5, best_h = cb2, hb2

    # Phase 5: Coordinate descent
    if el() < budget_s - 15:
        hc = coord_desc(best_h, dx, tgt, sweeps=40)
        cc = compute_c5(hc, dx)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 6: Final SLSQP polish
    if el() < budget_s - 5:
        def f4(x):
            return compute_c5(np.clip(x, 0, 1), dx)
        def gc(x):
            return np.sum(x) - tgt
        try:
            r = minimize(f4, best_h, method='SLSQP', bounds=bnds,
                        constraints=[{'type': 'eq', 'fun': gc}],
                        options={'maxiter': 500, 'ftol': 1e-15})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass
    # Phase 7: Quick block refine
    if el() < budget_s - 3:
        hr, cr = block_opt(best_h, dx, tgt, 40, 8, seed + 99)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
