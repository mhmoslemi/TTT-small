    return x.sum() - _eq_target

    return ({'type': 'eq', 'fun': eq_con_fun},)

import time

_eq_target = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def smooth_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = (corr - m) / temp
    return (m + temp * np.log(np.sum(np.exp(scaled)) / len(corr))) * dx

def slsqp_smooth_obj(x, dx, temp):
    return smooth_obj(x, dx, temp)
