def coordinate_descent(h, n, n_passes=100, tol=1e-10):
    """Exact coordinate descent for C5 minimization."""
    h = h.copy()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    for p in range(n_passes):
        max_change = 0
        for j in range(n):
            # Compute D_k for all k (lags)
            h_j = h[j]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c = corr * dx  # (2n-1,)
            
            D = np.zeros(n_corr)  # d(c[k])/d(h[j])
            for lag in range(-(n-1), n):
                m = lag + n - 1
                i_start = max(0, lag)
                i_end = min(n-1, n-1+lag)
                # h[j] appears as h[i] when i=j is in [i_start, i_end]
                if i_start <= j <= i_end:
                    j2 = j - lag  # this is the index for (1-h[j2])
                    if 0 <= j2 < n:
                        D[m] += (1.0 - h[j2])
                # h[j] appears as h[i-lag] when i-lag=j, i=j+lag is in [i_start, i_end]
                j_plus = j + lag
                if i_start <= j_plus <= i_end:
                    D[m] -= h[j_plus]
            
            # C_k = c[k] - h[j] * D[k]
            C = c - h_j * D
            
            # f(t) = max_k (C[k] + t*D[k]) for t in [0,1]
            # Find min via ternary search (convex function)
            best_t = h_j  # don't change
            best_val = c[np.argmax(c)]  # current C5
            
            # Evaluate at endpoints
            f0 = np.max(C)  # t=0
            f1 = np.max(C + D)  # t=1
            
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(60):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                fm1 = np.max(C + m1 * D)
                fm2 = np.max(C + m2 * D)
                if fm1 < fm2:
                    hi = m2
                else:
                    lo = m1
            
            t_opt = (lo + hi) / 2
            f_opt = np.max(C + t_opt * D)
            
            if f_opt < best_val - tol:
                h[j] = t_opt
                max_change = max(max_change, abs(t_opt - h_j))
        
        if max_change < 1e-10:
            break
        h = repair(h, n)
    
    return h

    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc_step(h, j, n):
    dx = 2.0 / n
    h_j = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    D = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        m = lag + n - 1
        i_s = max(0, lag)
        i_e = min(n - 1, n - 1 + lag)
        if i_s <= j <= i_e:
            j2 = j - lag
            D[m] += (1.0 - h[j2])
        jp = j + lag
        if i_s <= jp <= i_e:
            D[m] -= h[jp]
    C = c - h_j * D
    lo, hi = 0.0, 1.0
    for _ in range(55):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(C + m1 * D) < np.max(C + m2 * D):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coordinate_descent(h, n, n_passes=100):
    h = h.copy()
    for p in range(n_passes):
        max_chg = 0.0

        h = repair(h, n)
        if max_chg < 1e-13:
            break
    return h

def binary_local_search(h_start, n, rng, n_iters=500):
    h = (h_start > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    while len(ones) > target:
        i = rng.choice(ones)
        h[i] = 0
        ones, zeros = np.where(h == 1), np.where(h == 0)
    while len(ones) < target:
        i = rng.choice(zeros)
        h[i] = 1
        ones, zeros = np.where(h == 1), np.where(h == 0)
    c5 = compute_c5(h)
    for _ in range(n_iters):
        ones, zeros = np.where(h == 1), np.where(h == 0)
        best_c, best_mv = c5, None
        for _ in range(100):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c = compute_c5(h)
            if c < best_c - 1e-15:
                best_c, best_mv = c, (i, j)
            h[i], h[j] = 1.0, 0.0
        if best_mv is None:
            break
        h[best_mv[0]], h[best_mv[1]] = 0.0, 1.0
        c5 = best_c
    return h, c5

def smooth_max_descent(h0, n, alpha=300, lr=0.008, max_iter=2000):
    h = repair(h0, n)
    dx = 2.0 / n
    best_h, best_c = h.copy(), compute_c5(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c_max = np.max(c)
        e = np.exp(alpha * (c - c_max))
        S = np.sum(e)
        p = e / S
        grad = np.zeros(n)
        for lag in range(-(n - 1), n):
            m = lag + n - 1
            pm = p[m] * dx
            i_s = max(0, lag)
            i_e = min(n - 1, n - 1 + lag)
            for i in range(i_s, i_e + 1):
                grad[i] += pm * (1.0 - h[i - lag])
                grad[i - lag] -= pm * h[i]
        h = repair(h - lr * grad, n)
        c_new = compute_c5(h)
        if c_new < best_c:
            best_c, best_h = c_new, h.copy()
        if it % 1500 == 1499:
            lr *= 0.5
    return best_h, best_c
