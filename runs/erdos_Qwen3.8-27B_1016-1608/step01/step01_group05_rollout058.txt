        m = np.max(corr)

        grad = np.zeros(n)

                continue

            # (1-h[j+lag]) contributes to grad[j] for j in [max(0,-lag), min(n, n-lag))
            j_start = max(0, -lag)
            j_end = min(n, n - lag)
            if j_start < j_end:
                grad[j_start:j_end] += weights[idx] * (1.0 - h[j_start+lag:j_end+lag])
            # h[j-lag] valid when 0 <= j-lag < n, i.e., j in [max(0,lag), min(n, n+lag))

            j2_end = min(n, n + lag)

            best = obj
            best_h = h_new.copy()

def smooth_max_opt(h, n, n_iter=400, beta=60.0):
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iter):
        if _time_left() < 30:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        shifted = beta * (corr - m)
        w = np.exp(shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for idx in range(len(corr)):
            if w[idx] < 1e-10:
                continue
            lag = idx - (n - 1)
            js = max(0, -lag)
            je = min(n, n - lag)
            if js < je:
                grad[js:je] += w[idx] * (1.0 - h[js+lag:je+lag])
            j2s = max(0, lag)
            j2e = min(n, n + lag)
            if j2s < j2e:
                grad[j2s:j2e] -= w[idx] * h[j2s-lag:j2e-lag]
        alpha = 0.05 / (1.0 + 0.02 * it)
        h_new = project_to_set(h - alpha * grad, target)
        obj = compute_c5(h_new, n)
        if obj < best:
            best = obj
            best_h = h_new.copy()
        h = h_new
    return best_h, best

def sa_search(h, n, rng, n_sweeps=60, T0=0.005, T1=0.0001):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    for sweep in range(n_sweeps):
        if _time_left() < 30:
            break
        T = T0 * (T1 / T0) ** (sweep / max(1, n_sweeps))
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            c = min(1.0, max(0.0, old + rng.normal(0, 0.35)))
            if abs(c - old) < 1e-8:
                continue
            h_save = h.copy()
            h[j] = c
            h = project_to_set(h, target)
            obj = compute_c5(h, n)
            if obj < cur or rng.random() < np.exp(-(obj - cur) / max(T, 1e-12)):
                cur = obj
                if cur < best:
                    best = cur
                    best_h = h.copy()
                    improved = True
            else:
                h = h_save
        if not improved and sweep > n_sweeps // 2:
            break
    return best_h, best

def make_structured(n, rng):
    x = np.linspace(0.0, 2.0, n)
    shape = rng.randint(0, 5)
    if shape == 0:
        h = (x < 1.0).astype(float)
    elif shape == 1:
        f = rng.randint(2, 6)
        p = rng.uniform(0, 2 * np.pi)
        h = (np.sin(f * np.pi * x + p) > 0).astype(float)
    elif shape == 2:
        a, b = sorted(rng.uniform(0, 2, 2))
        h = ((x > a) & (x < b)).astype(float)
    elif shape == 3:
        h = (rng.random(n) < 0.5).astype(float)
    else:
        h = np.zeros(n)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return project_to_set(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t, _best_h, _best_c5, _best_n
    _budget = float(budget_s)
    _start_t = time.time()
    _best_h = None
    _best_c5 = float('inf')
    _best_n = 0
    rng = np.random.RandomState(seed)
    n1 = 90

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) != n1:
            h_init = upsample_fn(h_init, n1)
    except Exception:
        h_init = make_structured(n1, rng)
    h_init = project_to_set(h_init, n1 / 2.0)
    _update(h_init, n1)

    # Phase 1a: coordinate search from initial
    h_opt, c5 = coord_search(h_init, n1, rng, 120)
    _update(h_opt, n1)

    # Phase 1b: smooth max subgradient
    h_opt, c5 = smooth_max_opt(_best_h, n1, 300, 80.0)
    _update(h_opt, n1)

    # Phase 1c: coordinate search again
    h_opt, c5 = coord_search(_best_h, n1, rng, 150)
    _update(h_opt, n1)

    # Phase 1d: swap search (effective for near-binary)
    h_opt, c5 = swap_search(_best_h, n1, rng, 8000)
    _update(h_opt, n1)

    # Phase 1e: SA to escape local minima
    h_opt, c5 = sa_search(_best_h, n1, rng, 50, 0.008, 0.0002)
    _update(h_opt, n1)
    h_opt, c5 = coord_search(_best_h, n1, rng, 100)
    _update(h_opt, n1)

    # Phase 1f: random restarts with local optimization
    while _time_left() > _budget * 0.35:
        h_r = make_structured(n1, rng)
        h_opt, c5 = coord_search(h_r, n1, rng, 60)
        _update(h_opt, n1)
        h_opt, c5 = coord_search(_best_h, n1, rng, 100)
        _update(h_opt, n1)
        h_opt, c5 = smooth_max_opt(_best_h, n1, 150, 70.0)
        _update(h_opt, n1)

    # Phase 1g: final n=90 polish
    h_opt, c5 = coord_search(_best_h, _best_n, rng, 200)
    _update(h_opt, _best_n)
    h_opt, c5 = swap_search(_best_h, _best_n, rng, 10000)
    _update(h_opt, _best_n)
    h_opt, c5 = coord_search(_best_h, _best_n, rng, 150)
    _update(h_opt, _best_n)

    # Phase 2: n=180
    if _time_left() > _budget * 0.22:
        n2 = 180
        h_up = project_to_set(upsample_fn(_best_h, n2), n2 / 2.0)
        _update(h_up, n2)
        h_opt, c5 = coord_search(h_up, n2, rng, 80)
        _update(h_opt, n2)
        h_opt, c5 = smooth_max_opt(_best_h, n2, 200, 50.0)
        _update(h_opt, n2)
        h_opt, c5 = coord_search(_best_h, _best_n, rng, 120)
        _update(h_opt, _best_n)
        h_opt, c5 = swap_search(_best_h, _best_n, rng, 5000)
        _update(h_opt, _best_n)

    # Phase 3: n=360
    if _time_left() > _budget * 0.10:
        n3 = 360
        h_up = project_to_set(upsample_fn(_best_h, n3), n3 / 2.0)
        _update(h_up, n3)
        h_opt, c5 = coord_search(h_up, n3, rng, 40)
        _update(h_opt, n3)
        h_opt, c5 = smooth_max_opt(_best_h, n3, 100, 40.0)
        _update(h_opt, n3)

    # Final polish on best
    if _time_left() > 10:
        h_opt, c5 = coord_search(_best_h, _best_n, rng, 200)
        _update(h_opt, _best_n)
        h_opt, c5 = swap_search(_best_h, _best_n, rng, 5000)
        _update(h_opt, _best_n)

    _best_h = project_to_set(_best_h, _best_n / 2.0)
    _best_c5 = compute_c5(_best_h, _best_n)
    return _best_h, _best_c5, _best_n
