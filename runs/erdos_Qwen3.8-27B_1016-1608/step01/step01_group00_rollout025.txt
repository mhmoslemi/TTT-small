    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):

            break

        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, alpha, gamma=50):
    """Take a subgradient descent step on the smooth max."""
    n = len(h)
    dx = 2.0 / n
    c5_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c5_vals)
    if m == 0:
        return h.copy()
    weights = np.exp(gamma * (c5_vals - m))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    # Only process significant weights
    for k_idx in range(len(c5_vals)):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        shift = k_idx - (n - 1)
        if shift == 0:
            # corr_0 = sum h[i]*(1-h[i]), grad[i] = 1-2*h[i]
            grad += w * (1.0 - 2.0 * h)
        elif shift > 0:
            # corr_s = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # grad[j] = (1-h[j-s]) for j=s..n-1 (first term, i=j-s)
            #         - h[j+s] for j=0..n-1-s (second term, i=j)
            first = np.zeros(n)
            first[shift:] = 1.0 - h[:n-shift]  # (1-h[j-s]) for j >= s
            second = np.zeros(n)
            second[:n-shift] = h[shift:]  # h[j+s] for j < n-s
            grad += w * (first - second)
        else:
            s = -shift  # s > 0, corr_{-s} = sum h[i]*(1-h[i+s])
            first = np.zeros(n)
            first[:n-s] = 1.0 - h[s:]  # (1-h[j+s]) for j < n-s
            second = np.zeros(n)
            second[s:] = h[:n-s]  # h[j-s] for j >= s
            grad += w * (first - second)

            alpha *= 1.5
        else:
            alpha *= 0.5
        if alpha < 1e-10:

    return best_h, best_c5, alpha

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = 0.5
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, passes=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    shifts = np.arange(-(n - 1), n)
    for p in range(passes):
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        for j in order:
            old_j = h[j]
            # slope_k for shift s: 1 - h[j+s] - h[j-s] (out of bounds = 0)
            slope = np.ones(2 * n - 1)
            js = j + shifts
            jm = j - shifts
            mask_p = (js >= 0) & (js < n)
            mask_m = (jm >= 0) & (jm < n)
            slope[mask_p] -= h[js[mask_p]]
            slope[mask_m] -= h[jm[mask_m]]
            
            base = corr - old_j * slope
            
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(25):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(base + m1 * slope) >= np.max(base + m2 * slope):

                else:
                    b = m2
            t_best = (a + b) / 2.0
            
            if abs(t_best - old_j) > 1e-14:
                corr += (t_best - old_j) * slope
                h[j] = t_best
        
        h = repair(h, n)

def slsqp_polish(h, n, maxiter=1000):
    target = n / 2.0
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(c) * dx)
    
    def eq(x):
        return np.sum(x) - target
    
    cons = [{'type': 'eq', 'fun': eq}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < obj(h) + 1e-15:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except:
        pass
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = 0.5
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, passes=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    shifts = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for p in range(passes):
        order = rng.permutation(n)
        for j in order:
            old_j = h[j]
            slope = np.ones(2 * n - 1)
            js = j + shifts
            jm = j - shifts
            mask_p = (js >= 0) & (js < n)
            mask_m = (jm >= 0) & (jm < n)
            slope[mask_p] -= h[js[mask_p]]
            slope[mask_m] -= h[jm[mask_m]]
            base = corr - old_j * slope
            a, b = 0.0, 1.0
            for _ in range(22):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(base + m1 * slope) >= np.max(base + m2 * slope):
                    a = m1
                else:
                    b = m2
            t_best = (a + b) / 2.0
            if abs(t_best - old_j) > 1e-14:
                corr += (t_best - old_j) * slope
                h[j] = t_best
        h = repair(h, n)
    return h

def slsqp_polish(h, n, maxiter=1000):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(c) * dx)
    def eq(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': eq}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < obj(h) + 1e-15:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h

def try_resolution(h_best, c5_best, n_best, n_target, budget_s, t0, rng):
    if time.time() - t0 > budget_s * 0.95:
        return h_best, c5_best, n_best
    xp = np.linspace(0.0, 1.0, n_best)
    xn = np.linspace(0.0, 1.0, n_target)
    hi = np.interp(xn, xp, h_best)
    hi = repair(hi, n_target)
    h2 = coord_descent(hi, n_target, passes=150, rng=rng)
    c2 = compute_c5(h2)
    if c2 < c5_best:
        h_best, c5_best, n_best = h2, c2, n_target
    if time.time() - t0 < budget_s * 0.95:
        h2 = slsqp_polish(h_best, len(h_best), maxiter=1000)
        c2 = compute_c5(h2)
        if c2 < c5_best:
            h_best, c5_best, n_best = h2, c2, len(h2)
    return h_best, c5_best, n_best
