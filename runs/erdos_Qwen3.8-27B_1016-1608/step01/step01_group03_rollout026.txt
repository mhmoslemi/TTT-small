    for _ in range(200):

        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

    x_beta = beta * np.asarray(x)
    m = np.max(x_beta)
    if not np.isfinite(m):
        return float(m) / beta
    return (m + np.log(np.sum(np.exp(x_beta - m)))) / beta

        p1_e = min(2*n-2, i+n-1)

            a[i, idx] += (1.0 - h[h_idx])
        p2_s = max(0, n-1-i)

            a[i, idx2] += -h[h_idx2]

                return smooth_max(corr + ai * (t - old), beta)
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            h[i] = float(np.clip(res.x, 0.0, 1.0))

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt

    xb = beta * np.asarray(x, dtype=float)
    m = np.max(xb)

        return float(np.max(x))
    return (m + np.log(np.sum(np.exp(xb - m)))) / beta

        p2s = max(0, n - 1 - i)
