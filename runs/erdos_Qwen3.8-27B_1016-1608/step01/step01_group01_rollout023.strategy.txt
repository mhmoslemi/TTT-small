      ```python
              new_pop.append(child)
          initial_pop = new_pop
      
      # Projected Subgradient Phase
      best_h = min(initial_pop, key=compute_c5)
      best_c5 = compute_c5(best_h)
      
      subgrad_start = time.time()
      for it in range(1500):
          if time.time() - subgrad_start > budget_s * 0.78 - (time.time() - subgrad_start):
              break
          grad = subgrad_max_corr(best_h)
          if np.linalg.norm(grad) < 1e-6:
              break
          step = 0.02 / (np.linalg.norm(grad) + 1e-9)
          h_new = best_h - step * grad
          h_new = project_feasible(h_new, n_points)
          new_c5 = compute_c5(h_new)
          if new_c5 < best_c5:
              best_c5 = new_c5
              best_h = h_new.copy()
      ```
