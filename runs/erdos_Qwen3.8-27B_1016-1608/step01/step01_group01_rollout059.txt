    return c.max() * dx

    return np.correlate(h, 1.0 - h, mode='full')

    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
