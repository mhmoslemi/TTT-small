        if wl < 1e-14:
            continue
        k = l - (n - 1)  # lag
        if k >= 0:
            # out = sum_{t=0}^{n-1-k} h[t+k](1-h[t])
            # grad[j] += wl*dx * [(k<=j<=n-1)*(1-h[j-k]) - (0<=j<=n-1-k)*h[j+k]]
            # Part 1: j from k to n-1: (1-h[j-k])
            grad[k:] += wl * dx * (1.0 - h[:n-k])
            # Part 2: j from 0 to n-1-k: -h[j+k]
            grad[:n-k] -= wl * dx * h[k:]
        else:
            m = -k
            # out = sum_{t=0}^{n-1-m} h[t](1-h[t+m])

    return grad, float(max_f)  # val = max_f + log(sum(w))/beta but for optimization we just need grad direction

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    max_f = np.max(f)
    S = np.sum(np.exp(beta * (f - max_f)))
    w = np.exp(beta * (f - max_f)) / S
    val = max_f + np.log(S) / beta
    grad = np.zeros(n)
    for l in range(len(f)):
        wl = w[l]
        if wl < 1e-13:
            continue
        k = l - (n - 1)
        if k >= 0:
            grad[k:] += wl * dx * (1.0 - h[:n-k])
            grad[:n-k] -= wl * dx * h[k:]
        else:
            m = -k
            grad[:n-m] += wl * dx * (1.0 - h[m:])
            grad[m:] -= wl * dx * h[:n-m]
    penalty = lam * (np.sum(h) - n / 2.0)
    val += 0.5 * lam * penalty
    grad += penalty
    return val, grad

def lse_lbfgs(h0, n, beta_start=5.0, beta_end=5000.0, n_stages=8, lam=50.0):
    h = h0.copy()
    bnds = [(0.0, 1.0)] * n
    for stage in range(n_stages):
        frac = stage / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        cur_lam = lam * (1 + frac * 10)
        def obj(x):
            return lse_obj_and_grad(x, beta, cur_lam)[0]
        def grad(x):
            return lse_obj_and_grad(x, beta, cur_lam)[1]
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                           bounds=bnds, options={'maxiter': 200, 'gtol': 1e-10, 'ftol': 1e-14})
            if res.x is not None:
                h = repair(np.clip(res.x, 0.0, 1.0), n)
        except Exception:
            pass
    return h

def slsqp_optimize(h0, n, maxiter=1000):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def cons(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None:
            h_new = repair(np.clip(res.x, 0.0, 1.0), n)
            if compute_c5(h_new) <= compute_c5(h0) + 1e-14:
                return h_new
    except Exception:
        pass
    return h0

def nm_optimize(h0, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='Nelder-Mead', bounds=bnds,
                       options={'maxiter': 50000, 'xatol': 1e-14, 'fatol': 1e-18})
        h_new = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_new) <= compute_c5(h0) + 1e-14:
            return h_new
    except Exception:
        pass
    return h0

def gen_structured(n, rng):
    r = rng.random()
    if r < 0.3:
        m = rng.randint(1, max(2, n // 3))
        amp = rng.uniform(0.05, 0.5)
        ph = rng.uniform(0, 2 * np.pi)
        h = 0.5 + amp * np.sin(2 * np.pi * m * np.arange(n) / n + ph)
    elif r < 0.6:
        nb = rng.randint(2, min(10, n // 3))
        bs = n // nb
        vals = rng.uniform(0.05, 0.95, nb)
        h = np.repeat(vals, bs)[:n]
        if len(h) < n:
            h = np.concatenate([h, np.full(n - len(h), 0.5)])
    elif r < 0.8:
        h = rng.uniform(0, 1, n)
        for _ in range(3):
            h = 0.5 * h + 0.25 * np.roll(h, 1) + 0.25 * np.roll(h, -1)
    else:
        period = rng.randint(3, max(4, n // 5))
        pattern = rng.uniform(0, 1, period)
        h = np.tile(pattern, n // period + 1)[:n]
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Quick SLSQP
    if time.time() - t0 < budget_s - 10:
        h = slsqp_optimize(h_best, n, 2000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: LSE-LBFGS continuation (multiple passes)
    for rnd in range(4):
        if time.time() - t0 > budget_s * 0.40:
            break
        h_lse = lse_lbfgs(h_best, n, beta_start=3.0, beta_end=8000.0,
                          n_stages=10, lam=30.0)
        c = compute_c5(h_lse)
        if c < c5_best:
            h_best, c5_best = h_lse, c
        # SLSQP polish after LSE
        h = slsqp_optimize(h_best, n, 1000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 3: Diverse restarts with SLSQP
    base_scales = [0.004, 0.01, 0.02, 0.05, 0.1, 0.2]
    trial = 0
    while time.time() - t0 < budget_s * 0.78:
        r = rng.random()
        if r < 0.45:
            sc = base_scales[rng.randint(len(base_scales))]
            h_p = repair(h_best + rng.randn(n) * sc, n)
        elif r < 0.60:
            h_p = h_best.copy()
            nf = rng.randint(1, max(2, n // 3))
            idx = rng.choice(n, nf, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
            h_p = repair(h_p, n)
        elif r < 0.72:
            h_p = gen_structured(n, rng)
        elif r < 0.84:
            a = rng.uniform(0.3, 0.7)
            h_p = repair(a * h_best + (1 - a) * gen_structured(n, rng), n)
        else:
            # LSE-based perturbation
            h_lse = lse_lbfgs(h_best, n, beta_start=1.0, beta_end=50.0,
                              n_stages=3, lam=20.0)
            h_p = repair(h_lse + rng.randn(n) * 0.02, n)

        h = slsqp_optimize(h_p, n, 400)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
        trial += 1
        if trial % 25 == 0:
            h_nm = nm_optimize(h_best, n)
            c_nm = compute_c5(h_nm)
            if c_nm < c5_best:
                h_best, c5_best = h_nm, c_nm

    # Phase 4: Aggressive local polish
    if time.time() - t0 < budget_s * 0.88:
        h = slsqp_optimize(h_best, n, 5000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
    if time.time() - t0 < budget_s * 0.93:
        h = nm_optimize(h_best, n)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
    if time.time() - t0 < budget_s * 0.97:
        h = slsqp_optimize(h_best, n, 8000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 5: Try different resolutions
    for n2 in [140, 100, 80, 60, 70]:
        if time.time() - t0 > budget_s * 0.99:
            break
        if n2 == n:
            continue
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n2)
        h2 = np.interp(x_new, x_old, h_best)
        h2 = repair(h2, n2)
        h2 = slsqp_optimize(h2, n2, 800)
        c2 = compute_c5(h2)
        if c2 < c5_best:
            h_best, c5_best, n = h2, c2, n2

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
