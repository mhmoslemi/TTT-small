    h = np.array(h, dtype=float)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        e = h.sum() - target
        if abs(e) < 1e-14:
            return h
        if e > 0:
            m = h > 1e-12
            if not m.any():
                return h
            s = min(e / m.sum(), h[m].min())
            h[m] -= s
        else:
            m = h < 1.0 - 1e-12
            if not m.any():
                return h
            s = min(-e / m.sum(), (1.0 - h[m]).min())
            h[m] += s
    return h

def optimize_lbfgsb(h0, dx, target, maxiter=300, T=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(dx, T), method='L-BFGS-B',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project(r.x, target)
    except Exception:
        return None

def optimize_slsqp(h0, dx, target, maxiter=200, T=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(smooth_max_obj, h0, args=(dx, T), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(r.x, target)
    except Exception:
        return None

def fourier_init(n, K, amps, phases):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(K):
        h += amps[k] * np.cos(2.0 * np.pi * (k + 1) * i / n + phases[k])
    return project(h, n / 2.0)

def periodic_init(n, period, duty):
    h = np.zeros(n)
    on = int(round(period * duty))
    for j in range(n):
        if (j % period) < on:
            h[j] = 1.0
    return project(h, n / 2.0)

def coord_descent(h, dx, target, n_iter, rng, step0=0.04):
    n = len(h)
    c5 = compute_c5(h, dx)
    step = step0
    for it in range(n_iter):
        i = rng.randint(n)
        for d in (step, -step):
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + d, 0.0, 1.0)
            h2 = project(h2, target)
            c52 = compute_c5(h2, dx)
            if c52 < c5 - 1e-15:
                h, c5 = h2, c52
                break
        if it % 500 == 0 and it > 0:
            step = max(0.0005, step * 0.7)
    return h, c5

def pair_swap_search(h, dx, target, n_iter, rng):
    n = len(h)
    c5 = compute_c5(h, dx)
    for it in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        for frac in (0.0, 0.25, 0.5, 0.75, 1.0):
            di = frac * 0.1
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + di, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - di, 0.0, 1.0)
            h2 = project(h2, target)
            c52 = compute_c5(h2, dx)
            if c52 < c5 - 1e-15:
                h, c5 = h2, c52
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_n = len(best_h)
    best_dx = 2.0 / best_n
    best_c5 = compute_c5(best_h, best_dx)

    # Phase 1: Polish initial solution
    if tl() > 90:
        for T in (80, 200, 500):
            if tl() < 20:
                break
            h_new = optimize_slsqp(best_h, best_dx, best_n / 2.0, maxiter=200, T=T)
            if h_new is not None:
                c5 = compute_c5(h_new, best_dx)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
        if tl() > 30:
            best_h, best_c5 = coord_descent(best_h, best_dx, best_n / 2.0, 3000, rng)

    # Phase 2: Explore multiple n values with structured inits
    n_list = [60, 70, 84, 90, 98, 104, 112, 120, 126, 140]
    K = 4
    for n_t in n_list:
        if tl() < 50:
            break
        dx_t = 2.0 / n_t
        tgt_t = n_t / 2.0
        candidates = []
        # Fourier inits
        for trial in range(3):
            amps = np.array([0.35 / (k + 1) for k in range(K)])
            if trial > 0:
                amps *= rng.uniform(0.6, 1.4, K)
            ph = np.zeros(K) if trial == 0 else rng.uniform(0, 2 * np.pi, K)
            candidates.append(fourier_init(n_t, K, amps, ph))
        # Periodic inits
        for per in (7, 9, 11, 13):
            for duty in (0.42, 0.5):
                candidates.append(periodic_init(n_t, per, duty))
        # Random binary
        for _ in range(2):
            hb = (rng.random(n_t) < 0.5).astype(float)
            candidates.append(project(hb, tgt_t))

        for h0 in candidates:
            if tl() < 15:
                break
            h_new = optimize_lbfgsb(h0, dx_t, tgt_t, maxiter=200, T=150)
            if h_new is None:
                continue
            c5_t = compute_c5(h_new, dx_t)
            if c5_t < best_c5:
                best_h, best_n, best_dx, best_c5 = h_new.copy(), n_t, dx_t, c5_t

    # Phase 3: Refine current best with multiple techniques
    if tl() > 100:
        for T in (100, 300, 800):
            if tl() < 20:
                break
            h_new = optimize_lbfgsb(best_h, best_dx, best_n / 2.0, maxiter=400, T=T)
            if h_new is not None:
                c5 = compute_c5(h_new, best_dx)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
        if tl() > 30:
            best_h, best_c5 = coord_descent(best_h, best_dx, best_n / 2.0, 5000, rng)
        if tl() > 20:
            best_h, best_c5 = pair_swap_search(best_h, best_dx, best_n / 2.0, 3000, rng)

    # Phase 4: Subspace global search on best n
    if tl() > 80:
        n_s = best_n
        dx_s = best_dx
        tgt_s = n_s / 2.0
        K_s = 5
        for _ in range(6):
            if tl() < 15:
                break
            amps = np.array([rng.uniform(0.1, 0.45) / (k + 1) for k in range(K_s)])
            ph = rng.uniform(0, 2 * np.pi, K_s)
            h0 = fourier_init(n_s, K_s, amps, ph)
            h_new = optimize_lbfgsb(h0, dx_s, tgt_s, maxiter=300, T=200)
            if h_new is not None:
                c5 = compute_c5(h_new, dx_s)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5

    # Phase 5: Final coordinate + SLSQP polish
    if tl() > 20:
        best_h, best_c5 = coord_descent(best_h, best_dx, best_n / 2.0, 4000, rng, step0=0.01)
    if tl() > 10:
        h_new = optimize_slsqp(best_h, best_dx, best_n / 2.0, maxiter=400, T=500)
        if h_new is not None:
            c5 = compute_c5(h_new, best_dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    best_h = project(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, best_dx)
    return (best_h, final_c5, best_n)
