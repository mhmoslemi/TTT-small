            j = i + shift
            if 0 <= j < n:
                val = h[i] * (1.0 - h[j])
                if val > 0.01:
                    contrib.append((i, j, val))
        contrib.sort(key=lambda x: -x[2])
        improved = False
        for (i, j, _) in contrib[:5]:
            # Try to reduce: set h[i] toward 0 or h[j] toward 1
            for (pi, ci) in [(i, max(0.0, h[i] - 0.3)), (j, min(1.0, h[j] + 0.3))]:
                old_pi, old_pj = h[pi], h[j]
                h[pi] = ci
                if pi != j:
                    h[j] = min(1.0, h[j] + 0.3) if pi == i else max(0.0, h[i] - 0.3)
                hp = project_to_set(h, target)
                v = compute_c5(hp, n)
                if v < best_c5:
                    best_c5 = v
                    h = hp
                    improved = True
                else:
                    h[pi] = old_pi
                    if pi != j:
                        h[j] = old_pj
                    h = project_to_set(h, target)
        if not improved:
            # Small random kick
            h[rng.randint(n)] = rng.random()
            h = project_to_set(h, target)
            best_c5 = compute_c5(h, n)
    return h, best_c5

def _anneal_cd(h, n, rng, time_limit, t0):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    temp = 0.05
    decay = 0.9995
    for sweep in range(100000):
        if time.time() - t0 > time_limit:
            break
        temp = max(0.001, temp * decay)
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            c = rng.choice([0.0, 1.0, max(0, old - 0.2), min(1, old + 0.2)])
            if abs(c - old) < 1e-9:
                continue
            h[j] = c
            hp = project_to_set(h, target)
            v = compute_c5(hp, n)
            delta = v - compute_c5(h, n)
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                h = hp
                if v < best_c5:
                    best_c5 = v
                    best_h = h.copy()
            else:
                h[j] = old
                h = project_to_set(h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    # Get initial
    n = 90
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) == 0:
            raise ValueError
        n = len(h_init)
    except Exception:
        h_init = np.zeros(90)
        h_init[rng.choice(90, 45, replace=False)] = 1.0

    best_h = project_to_set(h_init.copy(), n / 2.0)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Fast coordinate descent from parent (30% budget)
    t_alloc = budget * 0.30
    h1, c1 = _fast_cd(best_h, n, rng, t_alloc, t0)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Pair swap optimization (15% budget)
    if _tl(budget, t0) > budget * 0.15:
        t_alloc = budget * 0.15
        h2, c2 = _pair_swap_cd(best_h, n, rng, t_alloc, t0)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Subgradient targeted (15% budget)
    if _tl(budget, t0) > budget * 0.15:
        t_alloc = budget * 0.15
        h3, c3 = _subgrad_targeted(best_h, n, rng, t_alloc, t0)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: More coordinate descent (15% budget)
    if _tl(budget, t0) > budget * 0.15:
        t_alloc = budget * 0.15
        h4, c4 = _fast_cd(best_h, n, rng, t_alloc, t0)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Try n=180
    n2 = 180
    if _tl(budget, t0) > budget * 0.15:
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        c_up = compute_c5(h_up, n2)
        if c_up < best_c5:
            best_h, best_c5 = h_up, c_up
        t_alloc = budget * 0.12
        h5, c5 = _fast_cd(h_up, n2, rng, t_alloc, t0)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5
        if _tl(budget, t0) > budget * 0.06:
            t_alloc2 = budget * 0.06
            h5b, c5b = _pair_swap_cd(h5, n2, rng, t_alloc2, t0)
            if c5b < best_c5:
                best_h, best_c5 = h5b, c5b

    # Phase 6: Final polish on best (remaining time)
    nf = len(best_h)
    if _tl(budget, t0) > 5.0:
        t_alloc = _tl(budget, t0) * 0.7
        hf, cf = _fast_cd(best_h, nf, rng, t_alloc, t0)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
    if _tl(budget, t0) > 3.0:
        t_alloc = _tl(budget, t0) * 0.5
        hf2, cf2 = _pair_swap_cd(best_h, nf, rng, t_alloc, t0)
        if cf2 < best_c5:
            best_h, best_c5 = hf2, cf2

    best_h = project_to_set(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
