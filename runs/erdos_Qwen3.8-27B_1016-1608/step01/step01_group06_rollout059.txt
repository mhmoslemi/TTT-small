import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def get_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def grad_soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = np.max(c)
    e = np.exp(tau * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-11:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            for i in range(n - l):
                j = i + l
                grad[j] += wk * (1.0 - h[i])
                grad[i] -= wk * h[j]
        else:
            s = -l
            for i in range(n - s):
                j = i + s
                grad[i] += wk * (1.0 - h[j])
                grad[j] -= wk * h[i]
    return grad

def soft_max_val(h, tau):
    c = get_correlations(h)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(tau * (c - m)))) / tau

def _cons_eq(x, n_val):
    return np.sum(x) - n_val / 2.0

def _penalty_obj(x, n_val):

    p = (np.sum(x) - n_val / 2.0) ** 2
    v = np.sum(np.maximum(0.0, -x) ** 2) + np.sum(np.maximum(0.0, x - 1.0) ** 2)
    return s + 1000.0 * (p + v)

    dx = 2.0 / n
    m = np.max(c)
    active = np.where(c > m - threshold)[0]
    n_act = max(1, len(active))
    grad = np.zeros(n)
    for k_idx in active:

        l = k_idx - (n - 1)
        if l >= 0:

                grad[j] += wk * (1.0 - h[i])
                grad[i] -= wk * h[j]

            s = -l
            for i in range(n - s):
                j = i + s
                grad[i] += wk * (1.0 - h[j])
                grad[j] -= wk * h[i]
    return grad
