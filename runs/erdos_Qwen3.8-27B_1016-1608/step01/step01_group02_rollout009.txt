            if stall % 300 == 0:
                step = max(0.003, step * 0.9)
                if step <= 0.003:
                    break
    return h, c5_cur

def active_set_slsqp(h0, dx, target, bounds, n, max_rounds=30):
    h = h0.copy()
    c5_cur = compute_c5(h, dx)
    _dt = 0  # placeholder for closure avoidance
    for rd in range(max_rounds):
        corr = get_corr(h)
        mx = np.max(corr)
        active = np.where(corr > 0.94 * mx)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        def obj_active(x, _act=active):
            c = get_corr(x)
            return np.sum(c[_act]) * dx / len(_act)
        def obj_act(x):
            c = get_corr(x)
            m = np.max(c)
            return np.sum(c[c > 0.9*m]) * dx
        try:
            res = minimize(obj_act, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x: constrain_sum(x, target)}],
                           options={'maxiter': 80, 'ftol': 1e-12})
            h_new = project_box_sum(res.x, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5_cur:
                h = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = 70
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    def tleft():
        return budget_s - (time.time() - t0)

    def update(h_new):
        nonlocal best_h, best_c5
        if h_new is None:
            return
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-16:
            best_h = h_new.copy()
            best_c5 = c5

    # Phase 1: Multi-temperature smooth SLSQP
    for temp in [40.0, 15.0, 6.0, 2.5, 1.0, 0.4]:
        if tleft() < 40:
            break
        h_s = slsqp_smooth(best_h, dx, target, bounds, temp, maxit=250)
        update(h_s)
        h_s2 = slsqp_smooth(best_h, dx, target, bounds, temp * 0.5, maxit=150)
        update(h_s2)

    # Phase 2: Direct SLSQP polish
    if tleft() > 60:
        h_d = slsqp_direct(best_h, dx, target, bounds, maxit=400)
        update(h_d)

    # Phase 3: Active-set guided optimization
    if tleft() > 80:
        h_a, c5_a = active_set_slsqp(best_h, dx, target, bounds, n, max_rounds=25)
        update(h_a)

    # Phase 4: Single coordinate descent (fine)
    if tleft() > 40:
        h_cd, c5_cd = coord_descent(best_h, dx, target, n, tleft, max_iter=40000)
        update(h_cd)

    # Phase 5: Pair coordinate descent
    if tleft() > 30:
        h_pc, c5_pc = pair_coord_descent(best_h, dx, target, n, tleft, max_iter=50000)
        update(h_pc)

    # Phase 6: One more smooth + direct SLSQP from best
    if tleft() > 30:
        h_s = slsqp_smooth(best_h, dx, target, bounds, 0.8, maxit=200)
        update(h_s)
        h_d = slsqp_direct(best_h, dx, target, bounds, maxit=300)
        update(h_d)

    # Phase 7: Final ultra-fine coordinate descent
    if tleft() > 15:
        h_cd2, c5_cd2 = coord_descent(best_h, dx, target, n, tleft, max_iter=30000)
        update(h_cd2)

    # Phase 8: Very last SLSQP
    if tleft() > 10:
        h_f = slsqp_direct(best_h, dx, target, bounds, maxit=200)
        update(h_f)

    best_h = project_box_sum(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
