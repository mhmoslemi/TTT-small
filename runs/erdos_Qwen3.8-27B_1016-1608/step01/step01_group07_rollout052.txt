def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(64):
        lam = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - lam, 0.0, 1.0))
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _minimax_1d(a, b):

    best_c = _c5(h)
    target = n * 0.5

        if time.time() - t0 > budget - 3:
            break
        C = _corr(h)
        idx = int(np.argmax(C))
        s = idx - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            cnt = n - s
            g[:cnt] += 1.0 - h[s:s+cnt]
            g[s:s+cnt] -= h[:cnt]
        else:
            ss = -s
            cnt = n - ss
            g[ss:ss+cnt] += 1.0 - h[:cnt]
            g[:cnt] -= h[ss:ss+cnt]
        gn = np.linalg.norm(g)

def _coord_descent(h, n_passes, t0, budget, seed_val=0):

    h = h.copy()

        for i in perm:
            if time.time() - t0 > budget - 5:

            q_idx = i - s_vals
            vp = (p_idx >= 0) & (p_idx < n)
            vq = (q_idx >= 0) & (q_idx < n)
            hp = np.where(vp, h[np.clip(p_idx, 0, n-1)], 0.0)
            hq = np.where(vq, h[np.clip(q_idx, 0, n-1)], 0.0)

            a = C - b * h[i]

def _interp(h, n_new):
    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n_new) * (2.0 / n_new)
    hi = np.interp(xn, xo, h)
    return _proj_sum(np.clip(hi, 0.0, 1.0), n_new * 0.5)

def _slsqp_obj(x, alpha, n):

    c = np.correlate(x, 1.0 - x, mode='full')
    v = c * dx
    m = float(np.max(v))
    e = np.exp(np.clip(v - m, -60.0, 60.0))
    return m + np.log(float(np.sum(e))) / alpha
