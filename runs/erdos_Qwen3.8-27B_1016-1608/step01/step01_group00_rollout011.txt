def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            h[:] = diff / n
            h = np.clip(h, 0, 1)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc(h, n, sweeps=50, grid=100):
    h = h.copy()
    for sweep in range(sweeps):
        order = np.random.permutation(n)
        for i in order:
            h[i] = 0.0
            base = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            one = np.correlate(h, 1.0 - h, mode='full')
            # overlap(t) = (1-t)*base + t*one, minimize max over shifts
            ts = np.linspace(0.0, 1.0, grid)
            # Vectorized: shape (grid, 2n-1)
            vals = np.max((1.0 - ts)[:, None] * base[None, :] + ts[:, None] * one[None, :], axis=1)
            h[i] = ts[np.argmin(vals)]
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc(h, n, sweeps=50, grid=200):
    h = h.copy()
    for sweep in range(sweeps):
        order = np.random.permutation(n)
        for i in order:
            h[i] = 0.0
            base = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            one = np.correlate(h, 1.0 - h, mode='full')
            ts = np.linspace(0.0, 1.0, grid)
            vals = np.max((1.0 - ts)[:, None] * base[None, :] + ts[:, None] * one[None, :], axis=1)
            h[i] = ts[np.argmin(vals)]
        h = repair(h, n)
    return h
