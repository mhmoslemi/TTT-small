import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr / temp
    c = c - np.max(c)
    return (np.max(corr) + temp * np.log(np.sum(np.exp(c)) / len(corr))) * dx

def active_smooth_c5(h, dx, na, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(-corr)[:na]
    c = corr[idx] / temp
    c = c - np.max(c)
    return (np.max(corr[idx]) + temp * np.log(np.sum(np.exp(c)) / na)) * dx

def weighted_c5(h, dx, weights):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.sum(weights * corr) * dx)

def sum_constraint(x, target):
    return x.sum() - target

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(1000):
        np.clip(h, 0.0, 1.0, out=h)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def make_bounds(n):
    return [(0.0, 1.0)] * n

def slsqp_solve(h0, obj, obj_args, n, target, maxiter=300, ftol=1e-14):
    bounds = make_bounds(n)
    cons = ({'type': 'eq', 'fun': sum_constraint, 'args': (target,)},)
    try:
        res = minimize(obj, h0, args=obj_args, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return projection(res.x, target)
    except Exception:
        return None

def try_improve(best_h, best_c5, obj, obj_args, dx, n, target, maxiter, ftol):
    h_new = slsqp_solve(best_h, obj, obj_args, n, target, maxiter, ftol)
    if h_new is not None:
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-15:
            return h_new, c5
    return best_h, best_c5
