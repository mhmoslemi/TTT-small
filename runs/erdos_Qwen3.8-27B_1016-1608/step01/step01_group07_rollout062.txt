        if k == 0:
            break

        h = np.clip(h, 0.0, 1.0)

def _grad_top(h, n, top_k):
    dx = 2.0 / n
    one_m = 1.0 - h
    c = np.correlate(h, one_m, mode='full')
    v = c * dx
    m = float(np.max(v))
    idx = np.argpartition(v, -top_k)[-top_k:]

        hi2 = min(n - 1, n - 1 - L)
        if lo2 <= hi2:
