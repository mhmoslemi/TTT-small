import numpy as np
from scipy.optimize import minimize
import time

_G_N = 70
_G_TGT = 35.0
_G_DX = 2.0 / 70.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def active_lag_gradient(h, n):
    corr = get_corr(h)
    k_star = int(np.argmax(corr))
    m = k_star - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        for j in range(n - m):
            grad[j] += (1.0 - h[j + m])
            grad[j + m] -= h[j]
    else:
        p = -m
        for j in range(p, n):
            grad[j] += (1.0 - h[j - p])
            grad[j - p] -= h[j]
    return grad, k_star

def _cons_fun(x):
    return np.sum(x) - _G_TGT

def _cons_jac(x):
    return np.ones(_G_N)

def _obj_fun(x):
    return compute_c5(np.clip(x, 0.0, 1.0), _G_DX)

def _obj_grad(x):
    hc = np.clip(x, 0.0, 1.0)
    g, _ = active_lag_gradient(hc, _G_N)
    return g

def slsqp_optimize(h, n, dx, target, maxiter, ftol):
    global _G_N, _G_TGT, _G_DX
    _G_N, _G_TGT, _G_DX = n, target, dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    r = minimize(_obj_fun, h, jac=_obj_grad, method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    return repair(r.x, target)

        dc = cur_c5 - prev_c5

            if 0.001 < bb < 0.2:

        hc_new = repair(hc - step * g, target)

        prev_c5 = cur_c5

        c5_new = compute_c5(hc, dx)

            best_c5 = c5_new
            best = hc.copy()
        if it % 200 == 199:
            step = max(0.001, step * 0.95)
    return best, best_c5
