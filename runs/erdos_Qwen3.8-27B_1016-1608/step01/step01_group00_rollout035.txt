def soft_c5_and_grad(h, mu):
    """Compute softmax-smoothed C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    max_c = corr.max()
    exp_c = np.exp((corr - max_c) / mu)
    weights = exp_c / exp_c.sum()
    val = max_c + mu * np.log(exp_c.sum())
    
    # Gradient: weighted sum of per-shift gradients
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        s = m - (n - 1)
        # h[j+s] and h[j-s]
        hp = np.zeros(n)
        hm = np.zeros(n)
        if s >= 0:
            if n - s > 0:
                hp[:n-s] = h[s:]
            if s < n:
                hm[s:] = h[:n-s]
        else:
            ss = -s
            if n - ss > 0:
                hm[:n-ss] = h[ss:]
            if ss < n:
                hp[ss:] = h[:n-ss]
        g += w * (1.0 - hp - hm)
    return val, g

    v = params['v']
    b1 = params['beta1']
    b2 = params['beta2']
    lr = params['lr'] / np.sqrt(1 + t * params['decay'])
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad ** 2
    mhat = m / (1 - b1 ** (t + 1))
    vhat = v / (1 - b2 ** (t + 1))
    step = lr * mhat / (np.sqrt(vhat) + 1e-8)
    return step
