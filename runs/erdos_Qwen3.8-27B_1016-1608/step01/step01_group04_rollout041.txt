import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            return h
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _smax(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-13:
        return float(m)
    s = np.clip((c - m) / eps, -60.0, 60.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _eqf(x):
    return float(np.sum(x) - _opt['n'] / 2.0)

def _eqj(x):
    return np.ones(_opt['n'])

def _obf(x):
    return _smax(x, _opt['n'], _opt['eps'])

def do_slsqp(h, n, eps, maxiter, ftol):
    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _eqf, 'jac': _eqj}]
    r = minimize(_obf, h, method='SLSQP', bounds=bds, constraints=cs,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return repair(r.x, n)

def compute_effect(h, n):
    p_idx = np.arange(n, dtype=float)[:, None]
    shifts = np.arange(-(n - 1), n, dtype=float)[None, :]
    ps = p_idx - shifts
    pp = p_idx + shifts
    v_ps = np.clip(ps, 0, n - 1).astype(int)
    v_pp = np.clip(pp, 0, n - 1).astype(int)
    m_ps = ((ps >= 0) & (ps < n)).astype(float)
    m_pp = ((pp >= 0) & (pp < n)).astype(float)
    return (1.0 - h[v_ps]) * m_ps - h[v_pp] * m_pp

        return 0.0

        return lo

    fine = np.linspace(a, b, 30)
    fvals = np.max(c[None, :] + fine[:, None] * coeff[None, :], axis=1)
    fi = int(np.argmin(fvals))
    return float(fine[fi])
