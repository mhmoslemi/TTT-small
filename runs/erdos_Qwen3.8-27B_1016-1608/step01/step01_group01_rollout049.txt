        # First term: w[k] * (1-h[i-shift]) for valid i
        # i-shift ranges from 0 to n-1, so i ranges from shift to shift+n-1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, temp, n):
    c = corr_full(h) * (2.0 / n)
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def smooth_max_grad(h, temp, n):
    c = corr_full(h) * (2.0 / n)
    m = c.max()
    w = np.exp((c - m) / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        shift = k - (n - 1)
        i1 = max(0, shift)
        i2 = min(n, shift + n)
        j1 = i1 - shift
        j2 = i2 - shift
        grad[i1:i2] += w[k] * (1.0 - h[j1:j2])
        i3 = max(0, -shift)
        i4 = min(n, n - shift)
        j3 = i3 + shift
        j4 = i4 + shift
        grad[i3:i4] -= w[k] * h[j3:j4]
    return grad

def fourier_init(n, rng, base_freq=1.0):
    h = 0.5 + 0.3 * np.sin(2 * np.pi * base_freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    h += 0.15 * np.sin(4 * np.pi * rng.uniform(1, 3) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def optimize_starts(h_s, n, budget_frac, t0, budget_s, max_iter=2000):
    h_cur = h_s.copy()
    c5_cur = compute_c5(h_cur)
    constraints = [{'type': 'eq', 'fun': lambda x, nn=n: np.sum(x) - nn / 2.0,
                    'jac': lambda x, nn=n: np.ones(nn)}]
    bounds = [(0.0, 1.0)] * n
    temps = [0.08, 0.03, 0.01, 0.004, 0.001, 0.0004]
    for temp in temps:
        if time.time() - t0 > budget_s * budget_frac:
            break
        res = minimize(smooth_max_obj, h_cur, args=(temp, n),
                       jac=lambda x, t=temp, nn=n: smooth_max_grad(x, t, nn),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_cur = h_new.copy()
    return h_cur, c5_cur

def coord_descent(h, n, t0, budget_s, frac):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.04, -0.04]
    improved = True
    while improved and time.time() - t0 < budget_s * frac:

        for i in range(n):

                break
            for d in deltas:
                h_t = best_h.copy()
                h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    improved = True
                    break
    return best_h, best_c5

def shift_targeted_polish(h, n, t0, budget_s, frac):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    constraints = [{'type': 'eq', 'fun': lambda x, nn=n: np.sum(x) - nn / 2.0,
                    'jac': lambda x, nn=n: np.ones(nn)}]
    bounds = [(0.0, 1.0)] * n
    for _ in range(20):
        if time.time() - t0 > budget_s * frac:
            break
        c = corr_full(best_h) * (2.0 / n)
        k_star = np.argmax(c)
        shift = k_star - (n - 1)
        w = np.zeros(2 * n - 1)
        w[k_star] = 0.7
        top_k = np.argsort(c)[-5:]
        for k in top_k:
            if k != k_star:
                w[k] = 0.1
        w_sum = w.sum()
        w /= w_sum
        obj_fn = lambda x: np.dot(w, corr_full(x) * (2.0 / n))
        jac_fn = lambda x: np.zeros(n)
        gr = np.zeros(n)
        i = np.arange(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12: continue
            s = k - (n - 1)
            i1 = max(0, s); i2 = min(n, s + n)
            j1 = i1 - s; j2 = i2 - s

            i3 = max(0, -s); i4 = min(n, n - s)
            j3 = i3 + s; j4 = i4 + s
            gr[i3:i4] -= w[k] * x[j3:j4]
        res = minimize(obj_fn, best_h, jac=jac_fn, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 1000, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5
