import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, temp, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def to_near_binary(h, n):
    h = h.copy()
    if n % 2 == 0:
        num_ones = n // 2
        idx = np.argsort(h)
        h[:] = 0.0
        h[idx[-num_ones:]] = 1.0
    else:
        num_ones = (n - 1) // 2
        idx = np.argsort(h)
        h[:] = 0.0
        h[idx[num_ones + 1:]] = 1.0
        h[idx[num_ones]] = 0.5
    return h

def binary_swap_search(h, n, max_rounds=500, deadline=None):
    h = to_near_binary(h, n).copy()
    best_c5 = compute_c5(h)
    ones = np.where(h >= 0.9)[0]
    zeros = np.where(h < 0.1)[0]
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        best_imp = 1e-14
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                ct = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                imp = best_c5 - ct
                if imp > best_imp:
                    best_imp = imp
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_c5 = compute_c5(h)
        ones = np.where(h >= 0.9)[0]
        zeros = np.where(h < 0.1)[0]
    return h, best_c5

def make_random_binary(n, rng):
    target = n // 2
    h = np.zeros(n)
    pos = rng.choice(n, size=target, replace=False)
    h[pos] = 1.0
    if n % 2 == 1:
        h[rng.choice(n)] += 0.5
    return project_feasible(h, n)

def make_golden_seq(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    off = rng.uniform(0, 1)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (((i + 1) * phi + off) % 1.0) > 0.5 else 0.0
    return project_feasible(h, n)

def make_alternating(n, rng):
    h = np.zeros(n)
    blk = max(2, n // (10 + rng.randint(0, 5)))
    on_off = 1
    for i in range(n):
        if on_off:
            h[i] = 1.0
            if i % blk == blk - 1:
                on_off = 0
        else:
            if i % blk == blk - 1:
                on_off = 1
    return project_feasible(h, n)

            best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
