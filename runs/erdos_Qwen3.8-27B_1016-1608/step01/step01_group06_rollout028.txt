                x_ref = np.linspace(0, 2, n_pts, endpoint=False)
                h_start = h_cur.copy()

        # Random restart SLSQP
        if n_pts <= 64 and check_time(0.78):
            for r in range(3):
                h_rand = project_feasible(h_start + rng.randn(n_pts) * 0.07, n_pts)
                h_cur, s_cur = slsqp_pass(h_rand, n_pts, 100.0)
                if s_cur < best_score:
                    best_score = s_cur
                    best_h = h_cur.copy()
                    x_ref = np.linspace(0, 2, n_pts, endpoint=False)
                    h_start = h_cur.copy()

        # Projected gradient
        if check_time(0.72):
            h_cur, s_cur = proj_gradient(h_start, n_pts, rng, n_iter=400, tau_init=60.0)
            if s_cur < best_score:
                best_score = s_cur
                best_h = h_cur.copy()
                x_ref = np.linspace(0, 2, n_pts, endpoint=False)

        # Local perturbation
        if check_time(0.68):
            h_cur, s_cur = local_perturb(best_h, len(best_h), rng, 4000, amp=0.05)
            if s_cur < best_score:
                best_score = s_cur
                best_h = h_cur.copy()
                x_ref = np.linspace(0, 2, len(best_h), endpoint=False)

    # Final intensive optimization on best solution
    n_final = len(best_h)
    if check_time(0.55):
        for tau in [80.0, 150.0, 300.0, 500.0]:
            if not check_time(0.5):
                break
            best_h, best_score = slsqp_pass(best_h, n_final, tau)

    if check_time(0.4):
        best_h, best_score = proj_gradient(best_h, n_final, rng, n_iter=600, tau_init=80.0)

    if check_time(0.25):
        best_h, best_score = local_perturb(best_h, n_final, rng, 8000, amp=0.04)

    if check_time(0.15):
        best_h, best_score = slsqp_pass(best_h, n_final, 400.0)

    if check_time(0.05):
        best_h, best_score = local_perturb(best_h, n_final, rng, 5000, amp=0.03)

    # Final projection to ensure feasibility
    best_h = project_feasible(best_h, n_final)
    best_score = compute_score(best_h)[0]

    return best_h, best_score, n_final
