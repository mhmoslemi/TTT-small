But actually, I can vectorize this better. Let me think...

For each m (shift), the contribution to g is:
- For j where 0<=j-s<n: subtract dx*w[m]*h[j-s]
- For j where 0<=j+s<n: subtract dx*w[m]*h[j+s]

This is: for shift s = m-(n-1):
- If s >= 0: j-s valid means j in [s, n-1], contributes -dx*w[m]*h[j-s] = -dx*w[m]*h[0..n-1-s] to g[s..n-1]
  j+s valid means j in [0, n-1-s], contributes -dx*w[m]*h[j+s] = -dx*w[m]*h[s..n-1] to g[0..n-1-s]
- If s < 0 (d=-s): j-s = j+d valid means j in [0, n-1-d], contributes -dx*w[m]*h[j+d] to g[0..n-1-d]
  j+s = j-d valid means j in [d, n-1], contributes -dx*w[m]*h[j-d] to g[d..n-1]

```python
def soft_max_grad(h, tau=100.0):

    dx = 2.0 / n
    c = compute_all_c(h)
    cm = np.max(c)

    w /= np.sum(w)

    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        s = m - (n - 1)
        wt = dx * w[m]
        if s >= 0:
            g[s:] -= wt * h[:n - s]
            g[:n - s] -= wt * h[s:]
        else:
            d = -s
            g[:n - d] -= wt * h[d:]
            g[d:] -= wt * h[:n - d]
    return g
```

This is O(n^2) with small constant, very fast for n=50.

import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_obj(h, tau=100.0):
    c = compute_all_c(h)
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def soft_max_grad(h, tau=100.0):
    n = len(h)
    dx = 2.0 / n
    c = compute_all_c(h)
    cm = np.max(c)
    w = np.exp(tau * (c - cm))
    w /= np.sum(w)
    g = np.full(n, dx)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        s = m - (n - 1)
        wt = dx * w[m]
        if s >= 0:
            g[s:] -= wt * h[:n - s]
            g[:n - s] -= wt * h[s:]
        else:
            d = -s
            g[:n - d] -= wt * h[d:]
            g[d:] -= wt * h[:n - d]
    return g

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(600):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h = np.clip(h + step * mask.astype(float), 0.0, 1.0)
    return h

def _cons_fun(h):
    return np.sum(h) - _gn / 2.0

_gn = 0

def proj_grad_loop(h, n, t_end, rng, lr0=0.02, tau0=50.0):
    h = h.copy()
    lr = lr0
    tau = tau0
    best_s = compute_score(h)
    best_h = h.copy()
    nogo = 0
    while time.time() < t_end:
        g = soft_max_grad(h, tau)
        h_new = project_feasible(h - lr * g, n)
        s_new = compute_score(h_new)
        if s_new < best_s - 1e-11:
            best_s = s_new
            best_h = h_new.copy()
            h = h_new
            nogo = 0
            lr = min(lr * 1.03, 0.08)
            if nogo % 200 == 0:
                tau = max(tau0 * 0.5, tau * 0.97)
        else:
            nogo += 1
            lr *= 0.9
            if nogo > 80:
                h = project_feasible(best_h + rng.randn(n) * 0.015, n)
                lr = lr0 * rng.uniform(0.6, 1.4)
                tau = tau0 * rng.uniform(0.7, 1.5)
                nogo = 0
    return best_h, best_s

def slsqp_refine(h, n, tau, maxiter=400):
    global _gn
    _gn = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    try:
        res = minimize(soft_max_obj, h.copy(), args=(tau,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h2 = project_feasible(res['x'], n)
        return h2, compute_score(h2)
    except Exception:
        return h.copy(), compute_score(h)

def nelder_refine(h, n, maxiter=2000):
    def penalty(x):
        return compute_score(x) + 500.0 * (np.sum(x) - n / 2.0) ** 2
    try:
        res = minimize(penalty, h.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-7, 'fatol': 1e-11})
        h2 = project_feasible(res['x'], n)
        return h2, compute_score(h2)
    except Exception:
        return h.copy(), compute_score(h)

def perturbed_restart(h, n, rng, strategy):
    if strategy == 0:
        return project_feasible(h + rng.randn(n) * 0.04, n)
    elif strategy == 1:
        h2 = h.copy()
        k = max(2, n // 10)
        idx = rng.choice(n, size=k, replace=False)
        h2[idx] += rng.uniform(-0.4, 0.4, size=k)
        return project_feasible(h2, n)
    elif strategy == 2:
        h2 = h.copy()
        blen = max(3, n // 6)
        si = rng.randint(0, n - blen)
        h2[si:si + blen] = 1.0 - h2[si:si + blen]
        return project_feasible(h2, n)
    elif strategy == 3:
        h2 = h.copy()
        freq = rng.randint(2, 6)
        amp = rng.uniform(0.05, 0.15)
        xx = np.linspace(0, 2 * freq * np.pi, n)
        h2 += amp * np.sin(xx + rng.uniform(0, 2 * np.pi))
        return project_feasible(h2, n)
    else:
        h2 = np.zeros(n)
        nb = rng.randint(3, min(8, n // 4 + 1))
        for b in range(nb):
            s = int(b * n / nb)
            e = int((b + 1) * n / nb)
            h2[s:e] = rng.uniform(0.0, 1.0)
        return project_feasible(h2, n)

    t0 = time.time()

            if s_c < best_score:

                best_score = s_c

                best_score = s_pg2

    # Phase 6: Final intensive polish on best
    if time.time() < t0 + budget_s * 0.97:
        nf = len(best_h)
        for tau in [150, 400]:
            if time.time() > t0 + budget_s * 0.99:

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)

    return best_h, best_score, len(best_h)
