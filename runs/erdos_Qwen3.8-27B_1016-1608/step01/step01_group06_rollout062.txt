```python
from scipy.optimize import minimize

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_score(h):
    return np.max(compute_correlations(h))

def proj_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def grad_corr_at_k(h, k):
    """Analytical gradient of c_k w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # k is the lag (0 to n-1)
    # c_k = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
    # ∂c_k/∂h[i] = (1-h[i+k])*1_{0≤i≤n-1-k} - h[i-k]*1_{k≤i≤n-1}
    idx1 = np.arange(0, n - k)
    grad[idx1] += (1 - h[idx1 + k])
    idx2 = np.arange(k, n)
    grad[idx2] -= h[idx2 - k]
    return grad

def soft_max_score(h, tau):
    n = len(h)
    c = compute_correlations(h)
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau
```

```python
def pair_descent(h, n, max_sweeps=100):
    c = compute_correlations(h)
    best_score = np.max(c)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Transfer from j to i: h[i] += dt, h[j] -= dt
                # Or h[i] -= dt, h[j] += dt
                # Find best dt in [-h[j], h[i]-... wait
                # h[i] + dt ∈ [0,1] => dt ∈ [-h[i], 1-h[i]]
                # h[j] - dt ∈ [0,1] => dt ∈ [h[j]-1, h[j]]
                # Combined: dt ∈ [-h[i], h[j]] ∩ [-h[i], 1-h[i]] ∩ [h[j]-1, h[j]]
                lo = max(-h[i], h[j]-1)
                hi = min(h[j], 1-h[i])
                if hi - lo < 1e-10:
                    continue
                # Ternary search for best dt
                a, b = lo, hi
                for _ in range(30):
                    m1 = a + (b-a)*0.382
                    m2 = a + (b-a)*0.618
                    h1 = h.copy()
                    h1[i] += m1; h1[j] -= m1
                    h2 = h.copy()
                    h2[i] += m2; h2[j] -= m2
                    f1 = np.max(compute_correlations(h1))
                    f2 = np.max(compute_correlations(h2))
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                dt = (a+b)/2
                h_new = h.copy()
                h_new[i] += dt; h_new[j] -= dt
                f_new = np.max(compute_correlations(h_new))
                if f_new < best_score - 1e-12:
                    h = h_new
                    best_score = f_new
                    improved = True
        if not improved:
            break
    return h, best_score
```

```python
    n = len(h)
    C = compute_correlations(h)
    # Get indices of top_k largest C values
    idx = np.argsort(C)[-top_k:]
    weights = np.exp(C[idx] - C[idx[-1]])  # softmax-like
    weights /= weights.sum()

    eps = 1e-8

            hp[i] = min(1.0, hp[i] + eps)
```

```python
    n = len(h)
    k = i - (n - 1)
    n_full = 2 * n - 1
    idx = int(np.clip(i, 0, n_full - 1))
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # (1-h[m+k]) for m in [0, n-1-k]
        m1 = np.arange(0, n - k)
        grad[m1] += (1 - h[m1 + k])

        m2 = np.arange(k, n)

        m1 = np.arange(l, n)
        grad[m1] += (1 - h[m1 - l])
        # -h[m+l] for m in [0, n-1-l]
        m2 = np.arange(0, n - l)
        grad[m2] -= h[m2 + l]
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_tau_global = 100.0
_n_global = 50

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_score(h):
    return np.max(compute_correlations(h))

def proj_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask.astype(float), 0.0, 1.0)
    return h

def grad_C_at(h, i):
    n = len(h)
    n_full = 2 * n - 1
    idx = int(np.clip(i, 0, n_full - 1))
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        m1 = np.arange(0, n - k)
        grad[m1] += (1.0 - h[m1 + k])
        m2 = np.arange(max(0, k), n)
        if k > 0:
            grad[m2] -= h[m2 - k]
        else:
            grad[m2] -= h[m2]
    else:
        l = -k
        m1 = np.arange(l, n)
        grad[m1] += (1.0 - h[m1 - l])
        m2 = np.arange(0, n - l)
        grad[m2] -= h[m2 + l]
    return grad

def subgrad_descent(h, n, steps, t_end, lr0=0.04, beta=0.85, top_k=5):
    best_h = h.copy()
    best_s = compute_score(h)
    v = np.zeros(n)
    for it in range(steps):
        if time.time() > t_end:
            break
        C = compute_correlations(h)
        top_idx = np.argsort(C)[-top_k:]
        w = np.exp(C[top_idx] - C[top_idx[-1]])
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            g += wi * grad_C_at(h, ki)
        lr = lr0 / (1.0 + 0.02 * it)
        v = beta * v + (1.0 - beta) * g
        h = proj_feasible(h - lr * v, n)
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
    return best_h, best_s

def soft_max_obj(h, tau):
    C = compute_correlations(h)
    cm = np.max(C)
    return cm + np.log(np.sum(np.exp(tau * (C - cm)))) / tau
```
