    n = len(h)

    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def soft_max_score(h, tau=200.0):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    s = np.clip(tau * (c - cm), -50, 50)
    return cm + np.log(np.sum(np.exp(s))) / tau

def project(h, n):
    h = np.asarray(h, dtype=float).copy()
