def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx, c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[:n-shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        if s < n:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    return grad

    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[-k_lags:][::-1]
    grad = np.zeros(n)
    for k_idx in top:
        grad += _lag_grad(h, k_idx - (n - 1))
    return grad / len(top)

def obj_and_grad_smooth(h, temp):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    # log-sum-exp
    shifted = (c - m) / temp
    LOGSE = m + temp * np.log(np.sum(np.exp(shifted)))
    # Gradient: sum_k c_k * exp((c_k - m)/temp) / sum_k exp((c_k-m)/temp)
    weights = np.exp(shifted)

    grad = np.zeros(n)

        w = weights[k]
        if w > 1e-10:
            grad += w * _lag_grad(h, k - (n - 1))
    return LOGSE * (2.0 / n), grad * (2.0 / n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

        else:

            h[idx] = max(0.0, h[idx] + diff)
    return h

def _lag_grad(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    return grad

def multi_subgrad(h, k_lags=3):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[-k_lags:][::-1]
    grad = np.zeros(n)
    for k_idx in top:
        grad += _lag_grad(h, k_idx - (n - 1))
    return grad / len(top)

    return h.sum() - len(h) / 2.0

def sum_eq_jac(h):

def coord_descent(h, n, t_end, max_sweeps=200):

        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            lo = max(0.0, h[i] - 0.3)
            hi = min(1.0, h[i] + 0.3)
            vals = np.linspace(lo, hi, 15)
            vals = np.unique(np.concatenate([vals, [h[i], 0.0, 1.0]]))
            vals = vals[(vals >= 0.0) & (vals <= 1.0)]
            best_v = h[i]
            best_c5_i = best_c5

                h = project_feasible(h, n)
                best_c5 = best_c5_i
                improved = True
        if not improved:
            break
    return h, best_c5

def block_move_search(h, n, t_end, block_size=3, n_trials=50):
    best_c5 = compute_c5(h)
    rng_local = np.random.RandomState(123)
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        start = rng_local.randint(0, max(1, n - block_size))

        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            h = h_trial.copy()
    return h, best_c5
