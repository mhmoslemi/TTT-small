```python
    """Get lag indices that are near the maximum."""
    max_val = corr.max()
    return np.where(corr >= max_val - tol)[0]

def subgradient_c5(h, dx, tol=1e-10):
    """Compute C5 and its subgradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(corr.max() * dx)
    
    active = (corr >= corr.max() - tol)
    n_active = active.sum()
    if n_active == 0:
        return c5, np.zeros(n)
    
    g = np.zeros(n)
    active_indices = np.where(active)[0]
    center = n - 1
    
    for k_idx in active_indices:
        L = k_idx - center  # actual lag
        # d/dh[i] of corr[k_idx] = (1-h[i+L]) * [0≤i+L<n] - h[i-L] * [0≤i-L<n]

        g[valid_first_mask] += (1.0 - h[valid_first[valid_first_mask]])
        g[valid_second_mask] -= h[valid_second[valid_second_mask]]

    return c5, g

            cnt = mask.sum()
            if cnt == 0:

            s = min(-excess / cnt, (1.0 - h[mask]).min())
```

```python
def project_feasible(h, target):
```
```python
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(excess / cnt, h[mask].min())
            if s <= 0:
                break
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            if s <= 0:
                break
            h[mask] += s
    return h

def subgradient_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    center = n - 1

        return c5, np.zeros(n)
    g = np.zeros(n)
    idx = np.arange(n)
    for k_idx in active:
        L = k_idx - center
        i1 = idx + L
        m1 = (i1 >= 0) & (i1 < n)

            g[m1] += (1.0 - h[i1[m1]])
        i2 = idx - L
        m2 = (i2 >= 0) & (i2 < n)
        if m2.any():
            g[m2] -= h[i2[m2]]
    g *= dx / n_act
```

```python
m1 = (i1 >= 0) & (i1 < n)  # [0 ≤ j+L < n]
g[m1] += (1.0 - h[i1[m1]])  # g[j] += (1-h[j+L])

i2 = idx - L  # i2[j] = j - L
m2 = (i2 >= 0) & (i2 < n)  # [0 ≤ j-L < n]
g[m2] -= h[i2[m2]]  # g[j] -= h[j-L]
```

Yes, this is correct!

Now, the subgradient of max_k corr[k] * dx is:
(dx / |A|) * sum_{k in A} d/dh corr[k]

where A is the set of active (maximizing) lags. This is what I compute.

```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = temp * (corr - m)
    scaled = np.clip(scaled, -500, 500)
    weights = np.exp(scaled) / np.sum(np.exp(scaled))
    
    # Gradient of log-sum-exp at h: sum_k weights[k] * d/dh corr[k]
    center = n - 1
    g = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        L = k - center
        i1 = idx + L
        m1 = (i1 >= 0) & (i1 < n)

        i2 = idx - L
        m2 = (i2 >= 0) & (i2 < n)
        if m2.any():
            g[m2] -= weights[k] * h[i2[m2]]
    return g * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_global_target = 0.0
_global_dx = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(excess / cnt, h[mask].min())
            if s <= 0:
                break
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            if s <= 0:
                break
            h[mask] += s
    return h

def subgradient_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    center = n - 1
    max_val = np.max(corr)
    tol = max(1e-9, 1e-4 * max_val)
    active = np.where(corr >= max_val - tol)[0]
    n_act = len(active)
    if n_act == 0:
        return c5, np.zeros(n)
    g = np.zeros(n)
    idx = np.arange(n)
    for k_idx in active:
        L = k_idx - center
        i1 = idx + L
        m1 = (i1 >= 0) & (i1 < n)
        if m1.any():
            g[m1] += (1.0 - h[i1[m1]])
        i2 = idx - L
        m2 = (i2 >= 0) & (i2 < n)
        if m2.any():
            g[m2] -= h[i2[m2]]
    g *= dx / n_act
    return c5, g

def smooth_c5_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = np.clip(temp * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def smooth_c5_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = np.clip(temp * (corr - m), -500, 500)
    w = np.exp(scaled)
    w /= w.sum()
    center = n - 1
    g = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        L = k - center
        i1 = idx + L
        m1 = (i1 >= 0) & (i1 < n)
        if m1.any():
            g[m1] += wk * (1.0 - h[i1[m1]])
        i2 = idx - L
        m2 = (i2 >= 0) & (i2 < n)
        if m2.any():
            g[m2] -= wk * h[i2[m2]]
    return g * dx

def slsqp_obj(x):
    return compute_c5(x, _global_dx)

def slsqp_jac(x):
    _, g = subgradient_c5(x, _global_dx)
    return g

def con_fun(x):
    return x.sum() - _global_target

def con_jac(x):
    return np.ones(len(x))

def run_slsqp(h0, dx, target, maxiter=400):
    global _global_dx, _global_target
    _global_dx = dx
    _global_target = target
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': con_fun, 'jac': con_jac},)
    try:
        res = minimize(slsqp_obj, h0, jac=slsqp_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return None

def subgradient_loop(h, dx, target, n_iter, alpha_init, alpha_min, alpha_max,
                     alpha_inc, alpha_dec, t0, budget_s, reserve):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha_init
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < reserve:
            break
        c5, g = subgradient_c5(h, dx)
        h_new = project_feasible(h - alpha * g, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5 - 1e-14:
            h = h_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
            alpha = min(alpha * alpha_inc, alpha_max)
        else:
            alpha = max(alpha * alpha_dec, alpha_min)
    return best_h, best_c5

def coord_descent(h, dx, target, t0, budget_s, reserve, rng):
    h = h.copy()
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    n = len(h)
    step = 0.02
    no_improve = 0
    for it in range(50000):
        if budget_s - (time.time() - t0) < reserve:
            break
        i = rng.randint(n)
        improved = False
        for delta in (step, -step, 0.5 * step, -0.5 * step):
            h_t = h.copy()
            h_t[i] = np.clip(h_t[i] + delta, 0.0, 1.0)
            h_t = project_feasible(h_t, target)
            c5_t = compute_c5(h_t, dx)
            if c5_t < c5 - 1e-13:
                h, c5 = h_t, c5_t
                improved = True
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5
                break
        if not improved:
            no_improve += 1
            if no_improve > 3000:
                step *= 0.6
                no_improve = 0
                if step < 0.0005:
                    break
        else:
            no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float).copy()
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Coarse subgradient
    b1, c1 = subgradient_loop(best_h, dx, target, 8000, 0.03, 0.002, 0.2,
                              1.03, 0.8, t0, budget_s, 250)
    if c1 < best_c5:
        best_h, best_c5 = b1, c1

    # Phase 2: SLSQP polish
    if budget_s - (time.time() - t0) > 30:
        h_new = run_slsqp(best_h, dx, target, 500)
        if h_new is not None:
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 3: Fine subgradient
    b3, c3 = subgradient_loop(best_h, dx, target, 12000, 0.006, 0.0005, 0.06,
                              1.01, 0.85, t0, budget_s, 120)
    if c3 < best_c5:
        best_h, best_c5 = b3, c3

    # Phase 4: SLSQP from perturbed starts
    for i in range(12):
        if budget_s - (time.time() - t0) < 30:
            break
        scale = 0.02 + 0.025 * i
        h0 = project_feasible(best_h + rng.randn(n) * scale, target)
        h_new = run_slsqp(h0, dx, target, 300)
        if h_new is not None:
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 5: Coordinate descent
    if budget_s - (time.time() - t0) > 50:
        b5, c5_cd = coord_descent(best_h, dx, target, t0, budget_s, 30, rng)
        if c5_cd < best_c5:
            best_h, best_c5 = b5, c5_cd

    # Phase 6: Another fine subgradient round
    if budget_s - (time.time() - t0) > 60:
        b6, c6 = subgradient_loop(best_h, dx, target, 10000, 0.004, 0.0003, 0.04,
                                  1.005, 0.9, t0, budget_s, 40)
        if c6 < best_c5:
            best_h, best_c5 = b6, c6

    # Phase 7: Final SLSQP
    if budget_s - (time.time() - t0) > 15:
        h_new = run_slsqp(best_h, dx, target, 500)
        if h_new is not None:
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 8: Try alternate n values
    if budget_s - (time.time() - t0) > 120:
        for n_try in (n - 10, n - 5, n + 5, n + 10, n + 20, n - 20):
            if budget_s - (time.time() - t0) < 45:
                break
            if n_try < 30 or n_try > 500:
                continue
            dx_t = 2.0 / n_try
            target_t = n_try / 2.0
            x_old = np.linspace(0.0, 2.0, n)
            x_new_pts = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new_pts, x_old, best_h)
            h_interp = project_feasible(h_interp, target_t)
            b_t, c_t = subgradient_loop(h_interp, dx_t, target_t, 3000,
                                        0.02, 0.001, 0.1, 1.02, 0.8,
                                        t0, budget_s, 25)
            if c_t < best_c5:
                h_new = run_slsqp(b_t, dx_t, target_t, 300)
                if h_new is not None:
                    c5_t2 = compute_c5(h_new, dx_t)
                    if c5_t2 < best_c5:
                        best_h, best_c5, n, dx, target = h_new, c5_t2, n_try, dx_t, target_t
                else:
                    best_h, best_c5, n, dx, target = b_t, c_t, n_try, dx_t, target_t

    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
```
