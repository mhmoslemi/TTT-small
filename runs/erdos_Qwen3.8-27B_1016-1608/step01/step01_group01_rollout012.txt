    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def swap_objective(c_full, a, d):
    return np.max(c_full + d * a)

def best_swap_d(c_full, a, d_min, d_max, n_iter=50):
    # Ternary search for min of max(c_full + d*a) over d in [d_min, d_max]
    lo, hi = d_min, d_max
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = swap_objective(c_full, a, m1)
        f2 = swap_objective(c_full, a, m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d_opt = (lo + hi) / 2.0
    return d_opt

def swap_coefficients(h, i, j, n):
    # Compute a[k] for swap (i, +d) (j, -d)
    shifts = np.arange(-(n-1), n)  # length 2n-1
    a = np.zeros(2*n-1)
    # h[i] as first factor: (1-h[i+s]) if valid
    idx = i + shifts
    mask = (idx >= 0) & (idx < n)
    a[mask] += (1.0 - h[idx[mask]])
    # h[i] as second factor: -h[i-s] if valid
    idx2 = i - shifts
    mask2 = (idx2 >= 0) & (idx2 < n)
    a[mask2] -= h[idx2[mask2]]
    # h[j] as first factor: -(1-h[j+s]) if valid
    idx3 = j + shifts
    mask3 = (idx3 >= 0) & (idx3 < n)
    a[mask3] -= (1.0 - h[idx3[mask3]])
    # h[j] as second factor: +h[j-s] if valid
    idx4 = j - shifts
    mask4 = (idx4 >= 0) & (idx4 < n)
    a[mask4] += h[idx4[mask4]]
    return a

def exact_swap_descent(h, n, max_sweeps=500, tol=1e-10):

    for sweep in range(max_sweeps):
        improved = False
        c_full = np.correlate(h, 1.0 - h, mode='full')

                a = swap_coefficients(h, i, j, n)
                d_min = max(-h[i], h[j] - 1.0)

                if d_max - d_min < 1e-15:

                d_opt = best_swap_d(c_full, a, d_min, d_max)
                new_max = swap_objective(c_full, a, d_opt)
                if new_max < c_full.max() - tol:

                    h[j] -= d_opt
                    h[i] = np.clip(h[i], 0, 1)
                    h[j] = np.clip(h[j], 0, 1)
                    c_full = c_full + d_opt * a
                    improved = True
        if not improved:
            break
    return h, compute_c5(h)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        # Only adjust non-saturated coordinates
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def swap_coeffs(h, i, j, n):
    shifts = np.arange(-(n-1), n)
    a = np.zeros(2*n-1)
    idx = i + shifts
    m = (idx >= 0) & (idx < n)
    a[m] += (1.0 - h[idx[m]])
    idx2 = i - shifts
    m2 = (idx2 >= 0) & (idx2 < n)
    a[m2] -= h[idx2[m2]]
    idx3 = j + shifts
    m3 = (idx3 >= 0) & (idx3 < n)
    a[m3] -= (1.0 - h[idx3[m3]])
    idx4 = j - shifts
    m4 = (idx4 >= 0) & (idx4 < n)
    a[m4] += h[idx4[m4]]
    return a

def ternary_swap(c_full, a, d_lo, d_hi, n_iter=40):
    lo, hi = d_lo, d_hi
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c_full + m1 * a) < np.max(c_full + m2 * a):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def swap_descent(h, max_sweeps=200, tol=1e-12):
    n = len(h)
    for _ in range(max_sweeps):
        improved = False
        c_full = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            for j in range(i + 1, n):
                a = swap_coeffs(h, i, j, n)
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                d_opt = ternary_swap(c_full, a, d_lo, d_hi)
                f_opt = np.max(c_full + d_opt * a)
                if f_opt < c_full.max() - tol:
                    h[i] = np.clip(h[i] + d_opt, 0, 1)
                    h[j] = np.clip(h[j] - d_opt, 0, 1)
                    c_full = np.correlate(h, 1.0 - h, mode='full')
                    improved = True
        if not improved:
            break
    return h

def simulated_annealing(h, n, t0=0.3, t1=0.001, n_steps=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    for step in range(n_steps):
        if step % 1000 == 0:
            cur_c = compute_c5(cur)
        # Random swap perturbation
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = 0.05 + 0.1 * (1 - step / n_steps)
        d = rng.uniform(-scale, scale)
        trial = cur.copy()
        trial[i] += d
        trial[j] -= d
        trial = project_feasible(trial, n)
        if np.allclose(trial, cur):

        new_c = compute_c5(trial)
        T = t0 * (t1 / t0) ** (step / n_steps)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur = trial
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
    return best, best_c

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def swap_coeffs(h, i, j, n):
    shifts = np.arange(-(n - 1), n)
    a = np.zeros(2 * n - 1)
    idx = i + shifts
    m = (idx >= 0) & (idx < n)
    a[m] += (1.0 - h[idx[m]])
    idx2 = i - shifts
    m2 = (idx2 >= 0) & (idx2 < n)
    a[m2] -= h[idx2[m2]]
    idx3 = j + shifts
    m3 = (idx3 >= 0) & (idx3 < n)
    a[m3] -= (1.0 - h[idx3[m3]])
    idx4 = j - shifts
    m4 = (idx4 >= 0) & (idx4 < n)
    a[m4] += h[idx4[m4]]
    return a

def ternary_swap(c_full, a, d_lo, d_hi):
    lo, hi = d_lo, d_hi
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c_full + m1 * a) < np.max(c_full + m2 * a):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def swap_descent(h, max_sweeps=200, tol=1e-12):
    n = len(h)
    for _ in range(max_sweeps):
        improved = False
        c_full = np.correlate(h, 1.0 - h, mode='full')
        cm = c_full.max()
        for i in range(n):
            for j in range(i + 1, n):
                a = swap_coeffs(h, i, j, n)
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                d_opt = ternary_swap(c_full, a, d_lo, d_hi)
                f_opt = np.max(c_full + d_opt * a)
                if f_opt < cm - tol:
                    h[i] = np.clip(h[i] + d_opt, 0.0, 1.0)
                    h[j] = np.clip(h[j] - d_opt, 0.0, 1.0)
                    c_full = np.correlate(h, 1.0 - h, mode='full')
                    cm = c_full.max()
                    improved = True
        if not improved:
            break
    return h

def ils_step(h, n, rng, scale=0.15):
    c_full = np.correlate(h, 1.0 - h, mode='full')
    cm = c_full.max()

        i = rng.randint(n)
        j = rng.randint(n)

            continue

        h[i] += d
        h[j] -= d

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

    return x.sum() - n / 2.0
