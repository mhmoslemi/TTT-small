    target = n / 2.0
    h = project_to_set(h0.copy(), target)
    best = compute_c5(h, n)
    for it in range(max_iter):
        ca = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        idx = np.argsort(ca)[-top_k:]
        g = np.zeros(n)
        for ix in idx:
            g += grad_shift(h, n, ix - (n - 1))
        g /= float(top_k)
        step = step_init / np.sqrt(1.0 + it)
        h = project_to_set(h - step * g, target)
        c = compute_c5(h, n)
        if c < best:
            best = c
    return h, best

def coord_descent(h0, n, sweeps, rng):
    target = n / 2.0
    h = project_to_set(h0.copy(), target)
    best = compute_c5(h, n)
    for _ in range(sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            c0, cv = best, old
            for frac in (0.3, 0.1, 0.5, 0.05, 0.7, 0.9, 1.0):
                for c in (min(1.0, old + frac), max(0.0, old - frac)):
                    if abs(c - old) < 1e-9:
                        continue
                    h[j] = c
                    hp = project_to_set(h, target)
                    v = compute_c5(hp, n)
                    if v < c0:
                        c0, cv, h = v, c, hp
            if cv != old:
                h[j] = cv
                h = project_to_set(h, target)
                best = c0
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(h - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid

def grad_shift(h, n, s):
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s]) * dx
            g[i + s] -= h[i] * dx
    else:
        ss = -s
        for i in range(ss, n):
            g[i] += (1.0 - h[i - ss]) * dx
            g[i - ss] -= h[i] * dx
