import numpy as np
from scipy.optimize import minimize
import time

_OPT = {'n': 80, 'eps': 0.01}
best_solution = {'h': None, 'c5': 1.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] += d / np.sum(free)
        else:
            h += d / n
    return np.clip(h, 0.0, 1.0)

def _lag_grad(x, n, idx, dx):
    l = idx - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        g[l:] += (1.0 - x[:n - l])
        if n - l > 0 and l > 0:
            g[:n - l] -= x[l:]
    else:
        l2 = -l
        g[:n - l2] += (1.0 - x[l2:])
        if l2 > 0:
            g[l2:] -= x[:n - l2]
    return g * dx

def smax_obj(x):
    n, eps = _OPT['n'], _OPT['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-14:
        return m
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def smax_grad(x):
    n, eps = _OPT['n'], _OPT['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-14:
        return _lag_grad(x, n, int(np.argmax(c)), dx)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] > 1e-10:
            g += w[idx] * _lag_grad(x, n, idx, dx)
    return g

def eq_fun(x):
    return float(np.sum(x) - _OPT['n'] / 2.0)

def eq_jac(x):
    return np.ones(_OPT['n'])

def slsqp_opt(h, n, eps, maxiter):
    _OPT['n'] = n
    _OPT['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(smax_obj, h, method='SLSQP', bounds=bds, jac=smax_grad,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    return repair(res.x, n)

def project_to_simplex(v, target, n):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def proj_grad_descent(h, n, n_iter, lr_init):
    _OPT['n'] = n
    _OPT['eps'] = 1e-8
    h = h.copy().astype(float)
    nr = 1.0
    prev_h = h.copy()
    for it in range(n_iter):
        g = smax_grad(h)
        y = h + (it / (it + 3.0)) * (h - prev_h)
        gy = smax_grad(y)
        lr = lr_init
        for _ in range(30):
            candidate = y - lr * gy
            candidate = project_to_simplex(candidate, n / 2.0, n)
            c_new = smax_obj(candidate)
            c_cur = smax_obj(h)
            if c_new < c_cur - 1e-14 or lr < 1e-10:
                h = candidate
                break
            lr *= 0.5
        else:
            break
        prev_h = h
    return h

def try_solution(h, n, best_c5):
