import cvxpy as cp
import numpy as np

def sqp_step(h, n, target):
    dx = 2.0 / n
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [delta >= -h, delta <= 1-h, cp.sum(h + delta) == target, t >= 0]
    for k in range(-n+1, n):
        # Compute R_k(h) = sum h[i]*h[i+k]
        S_k = np.zeros((n,n))
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                S_k[i,j] = 1
        R_k = h @ S_k @ h  # = sum h[i]*S_k[i,j]*h[j]
        grad = 2 * S_k @ h  # gradient of h^T S_k h w.r.t. h (S_k symmetric part)
        # Linearize: R_k(h+δ) ≈ R_k + grad^T δ
        constraints.append((n/2 - R_k - grad @ delta) * dx <= t)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or CBC, CLP, etc.
    return h + delta.value

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(200):
        e = h.sum() - target
        if abs(e) < 1e-13:
            break
        if e > 0:
            m = h > 0
            if not m.any(): break
            s = min(e / m.sum(), h[m].min())
            h[m] -= s
        else:
            m = h < 1
            if not m.any(): break
            s = min(-e / m.sum(), (1.0 - h[m]).min())
            h[m] += s
    return h

def eq_target(x, target):
    return x.sum() - target

def smooth_max_fn(x, dx, beta):
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    return m + np.log(np.sum(np.exp((c - m) / beta)) / len(c))

    dx = 2.0 / n
    d = cp.Variable(n)
    t = cp.Variable()
    cons = [d >= -h - 1e-10, d <= 1.0 - h + 1e-10,
            cp.sum(d) == target - h.sum(), t >= 0]
