import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def fix_integral(h, n):
    target = 0.5 * n
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = np.sum(mask)
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def make_objective(n, alpha, penalty=1e8):
    dx = 2.0 / n
    target = 0.5 * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, 'full')
        v = c * dx
        m = np.max(v)
        e = np.exp(np.clip(alpha * (v - m), -700, 700))
        s = np.sum(e)
        val = m + np.log(s) / alpha
        val += penalty * (np.sum(x) - target) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1.0 - x, 'full')
        v = c * dx
        m = np.max(v)
        e = np.exp(np.clip(alpha * (v - m), -700, 700))
        p = e / np.sum(e)
        A = 1.0 - x
        T1 = np.convolve(p, A, 'full')[n-1 : 2*n-1]
        T2 = np.correlate(x, p, 'full')[n-1 : 2*n-1]
        g = dx * (T1 - T2)
        g += 2 * penalty * (np.sum(x) - target) * np.ones(n)
        return g
    return obj, grad

def interpolate(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return fix_integral(h_new, n_new)
