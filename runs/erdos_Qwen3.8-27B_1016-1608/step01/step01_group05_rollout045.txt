import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0
_best_h = None
_best_c5 = float('inf')
_best_n = 0
_opt_n = 0
_opt_target = 0.0
_opt_beta = 30.0

def _tl():
    return max(0.0, _budget - (time.time() - _start_t))

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def update_best(h, n):
    global _best_h, _best_c5, _best_n
    h = np.asarray(h, dtype=float)
    h = project_to_set(h, n / 2.0)
    c = compute_c5(h, n)
    if c < _best_c5:
        _best_h = h.copy()
        _best_c5 = c
        _best_n = n

def _slsqp_obj(x):
    return compute_c5(project_to_set(x, _opt_target), _opt_n)

def _slsqp_con(x):
    return float(np.sum(x) - _opt_target)

def _smooth_obj(x):
    dx = 2.0 / _opt_n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    c = np.max(corr)
    if c < 1e-30:
        return 0.0
    sh = np.clip(_opt_beta * (corr - c), -60, 60)
    return float(c + np.log(np.sum(np.exp(sh))) / _opt_beta)

def slsqp(h0, n, maxiter=600):
    global _opt_n, _opt_target
    _opt_n = n
    _opt_target = n / 2.0
    h0 = project_to_set(np.asarray(h0, dtype=float).copy(), _opt_target)
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_to_set(res.x.copy(), _opt_target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def lbgfs_smooth(h0, n, betas=None, maxiter=500):
    global _opt_n, _opt_target, _opt_beta
    if betas is None:
        betas = [20, 50, 120]
    _opt_n = n
    _opt_target = n / 2.0
    h = project_to_set(np.asarray(h0, dtype=float).copy(), _opt_target)
    best_c = compute_c5(h, n)
    for b in betas:
        _opt_beta = b
        try:
            res = minimize(_smooth_obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
            h_out = project_to_set(res.x.copy(), _opt_target)
            c = compute_c5(h_out, n)
            if c < best_c:
                best_c = c
                h = h_out.copy()
        except Exception:
            pass
    return h, best_c

def subgradient(h0, n, n_iters=3000, lr0=0.005):
    target = n / 2.0
    h = project_to_set(np.asarray(h0, dtype=float).copy(), target)
    best_h, best_c = h.copy(), compute_c5(h, n)
    v_mom = np.zeros(n)
    dx = 2.0 / n
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_active = np.argmax(corr)
        offset = k_active - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            j = i - offset
            if 0 <= j < n:
                grad[i] = (1.0 - h[j])
            j2 = i + offset
            if 0 <= j2 < n:
                grad[i] -= (1.0 - h[j2])
        grad = grad * dx
        lr = lr0 / (1.0 + 0.5 * it / 200.0)
        v_mom = 0.9 * v_mom + lr * grad
        h_new = h - v_mom
        h_new = project_to_set(h_new, target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def coord_search(h, n, rng, n_sweeps=30):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.0, 1.0, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            bl, bcv = old, best
            for s in steps:
                for sg in (-1, 1):
                    c = max(0.0, min(1.0, old + sg * s))
                    if abs(c - old) < 1e-9:
                        continue
                    h[j] = c
                    h = project_to_set(h, target)
                    v = compute_c5(h, n)
                    if v < bcv:
                        bcv, bl = v, c
            if bl != old:
                h[j] = bl
                h = project_to_set(h, target)
                best = bcv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def make_smooth(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = np.zeros(n)
    for _ in range(rng.randint(2, 7)):
        h += rng.uniform(0.3, 0.7) * np.sin(rng.randint(1, 10) * np.pi * x + rng.uniform(0, 2*np.pi))
    return project_to_set(0.5 + 0.3*h, n/2.0)

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h
