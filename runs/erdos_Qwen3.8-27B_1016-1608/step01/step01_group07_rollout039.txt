import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _obj_jac(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    e = np.exp(alpha * (c - cmax))
    sw = e.sum()
    p = e / sw
    obj = cmax + np.log(sw) / alpha
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for m in active:
        lag = m - (n - 1)
        w = p[m] * dx
        if lag >= 0:
            grad[:n-lag] += w * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:n] -= w * h[0:n-lag]
        else:
            lag2 = -lag
            grad[lag2:n] += w * (1.0 - h[0:n-lag2])
            grad[0:n-lag2] -= w * h[lag2:n]
    return obj, grad

def _pgd_block(h, n, alpha, eta, n_iter, t0, budget):
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break

            h_cand = _project(h - eta * grad, n)
            obj_cand = _c5(h_cand)
            if obj_cand <= obj0 + 1e-14:
                h = h_cand
                break
            eta *= 0.6
        if it % 100 == 99:
            alpha *= 1.5
    return h

def _slsqp(h0, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 20:
        return None
    def _f(x):
        return _obj_jac(x, alpha, n)[0]
    def _g(x):
        return _obj_jac(x, alpha, n)[1]
    def _cf(x):
        return np.sum(x) - 0.5 * n
    def _cg(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cg}]
    try:
        res = minimize(_f, h0, jac=_g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None:
            return _project(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return None

    xn = np.arange(n) * (2.0 / n)
