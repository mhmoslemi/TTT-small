    dx = 2.0 / n

    h0 = np.array(initial_h_values, dtype=float)
    h0 = project_feasible(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Optimize initial construction
    t1_end = t_start + budget_s * 0.25
    best_h, best_c5 = optimize_from(h0, n, dx, t_start, t1_end, rng)

    # Phase 2: Multiple structured restarts
    t2_end = t_start + budget_s * 0.65
    n_restarts = 12
    for r in range(n_restarts):
        if time.time() > t2_end:
            break
        t_re = t_start + min(budget_s * 0.65, time.time() + (t2_end - time.time()) / max(1, n_restarts - r))
        # Different perturbation strategies
        pert_scale = 0.03 + 0.04 * (r % 4)
        if r % 3 == 0:
            h_start = project_feasible(best_h + rng.normal(0, pert_scale, n))
        elif r % 3 == 1:
            h_start = project_feasible(np.roll(best_h, rng.randint(1, n - 1)))
            h_start = project_feasible(h_start + rng.normal(0, 0.15, n))
        else:
            h_start = project_feasible(h0 + rng.normal(0, pert_scale * 2, n))
        h_cand, c5_cand = optimize_from(h_start, n, dx, time.time(), t_re, rng)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 3: Subgradient polishing
    t3_end = t_start + budget_s * 0.75
    h3 = best_h.copy()
    for _ in range(3000):
        if time.time() > t3_end:
            break
        h3, c5_3 = subgrad_step(h3, n)
        if c5_3 < best_c5 - 1e-15:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: More pairwise from best
    t4_end = t_start + budget_s * 0.88
    best_h, best_c5 = optimize_from(best_h, n, dx, time.time(), t4_end, rng)

    # Phase 5: SLSQP polishing
    if time.time() < t_start + budget_s * 0.97:
        constraints = [{'type': 'eq', 'fun': sum_obj, 'jac': sum_jac}]
        bounds = [(0.0, 1.0)] * n
        for temp in [0.05, 0.01, 0.003, 0.001]:
            if time.time() > t_start + budget_s * 0.97:
                break
            res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                         method='SLSQP', bounds=bounds, constraints=constraints,
                         options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Final pairwise refinement
    t6_end = t_start + budget_s * 0.99
    best_h, best_c5 = optimize_from(best_h, n, dx, time.time(), t6_end, rng)

    # Final projection and evaluation
    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
