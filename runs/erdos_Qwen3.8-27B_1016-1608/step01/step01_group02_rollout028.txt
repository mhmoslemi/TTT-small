    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_fn(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = T * (corr - m)
    se = np.exp(scaled)
    return (m + np.log(se.sum() / len(corr)) / T) * dx

def euclid_project(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0
            if not mask.any():
                break
            s = min(excess / mask.sum(), h[mask].min() - 1e-15)
            if s <= 0:
                break
            h[mask] -= s
        else:
            mask = h < 1
            if not mask.any():
                break
            s = min(-excess / mask.sum(), 1.0 - h[mask].min() - 1e-15)
            if s <= 0:
                break
            h[mask] += s
    return h

def slsqp_smooth(h0, dx, target, T, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(smooth_c5_fn, h0, args=(dx, T), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return euclid_project(res.x, target)
    except:
        return None

    return x.sum() - _con_target

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_fn(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = T * (corr - m)
    se = np.exp(scaled)
    return (m + np.log(se.sum() / len(corr)) / T) * dx

def euclid_project(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 1e-14
            if not mask.any():
                break
            h[mask] -= excess / mask.sum()
        else:
            mask = h < 1.0 - 1e-14
            if not mask.any():
                break
            h[mask] += (-excess) / mask.sum()
    return np.clip(h, 0.0, 1.0)

    _con_target = target

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
