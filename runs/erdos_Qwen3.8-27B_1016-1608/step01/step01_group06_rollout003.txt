Let me think about the coordinate descent. For a fixed h with h[j] variable:
c_s = sum_i h[i]*(1-h[i+s])

The parts involving h[j]:
- Terms where j is the first index: h[j]*(1-h[j+s]) for valid j+s
- Terms where j is the second index: h[j-s]*(1-h[j]) for valid j-s

So c_s as a function of h[j] (fixing all other h):
= h[j]*(1-h[j+s]) + h[j-s]*(1-h[j]) + constants
= h[j] - h[j]*h[j+s] + h[j-s] - h[j-s]*h[j] + constants
= h[j]*(1 - h[j+s] - h[j-s]) + h[j-s] + constants

This is LINEAR in h[j]! (Because h[j] appears at most once in each product term.)

Wait, that's for a single c_s. For the sum over active shifts:

This means coordinate descent will push values to 0 or 1! That's a nice property.

For the max objective: we can't do coordinate descent directly (the max of linear functions is convex, so we'd be minimizing a convex function along each coordinate... wait no, we're MINIMIZING, and the max of linear functions is convex, so the minimum along a coordinate is at the boundary). 

Actually wait: we want to MINIMIZE max_s c_s. Along coordinate j, max_s c_s is a function of h[j]. Each c_s is linear in h[j]. The max of linear functions is convex. So minimizing a convex function over [0,1] gives the minimum at the point where the derivative changes sign (if it exists) or at the boundary.

The derivative of c_s with respect to h[j] is:
- (1-h[j+s]) - h[j-s] (when both terms are valid)
- (1-h[j+s]) (when only first term valid)
- -h[j-s] (when only second term valid)
- -2h[j] + 1 (when s=0)

This is constant (not depending on h[j])! So c_s is linear in h[j] with a fixed slope. The max of these is a convex piecewise-linear function of h[j], and its minimum is at the "kink" point or at the boundary.

For the sum of active c_s: the sum is linear in h[j], so the minimum is trivially at h[j]=0 or h[j]=1.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def grad_active_sum(h, active_shifts, n):
    grad = np.zeros(n)
    for s in active_shifts:
        if s == 0:
            grad += 1.0 - 2.0 * h
        elif s > 0:
            hi = n - s
            if hi > 0:
                grad[0:hi] += (1.0 - h[s:s+hi])
                grad[s:n] -= h[0:n-s]
        else:
            sp = -s
            hi = n - sp
            if hi > 0:
                grad[sp:n] += (1.0 - h[0:hi])
                grad[0:hi] -= h[sp:n]
    return grad

def active_set_method(h, n, max_iter=20000, tol=1e-12):
    best_h = h.copy()
    best_s = compute_score(h)
    h_c = h.copy()
    momentum = np.zeros(n)
    beta = 0.9
    alpha = 0.05
    
    for it in range(max_iter):
        c_all = compute_all_c(h_c)
        c_max = np.max(c_all)
        # Active set: shifts within 5% of max or in top 25%
        n_shifts = len(c_all)
        k_active = max(5, n_shifts // 4)
        top_idx = np.argsort(c_all)[-k_active:]
        active_shifts = top_idx - (n - 1)
        
        g = grad_active_sum(h_c, active_shifts, n)
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        momentum = beta * momentum - alpha * g
        h_new = project_feasible(h_c + momentum, n)
        s_new = compute_score(h_new)
        
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
            alpha = min(alpha * 1.1, 0.4)
        else:
            h_new = project_feasible(h_c - alpha * 0.5 * g, n)
            s_new = compute_score(h_new)
            if s_new < best_s:
                best_s = s_new
                best_h = h_new.copy()
                h_c = h_new.copy()
                momentum = momentum * 0.5
            else:
                alpha *= 0.7
                momentum = momentum * 0.3
        
        if it % 1000 == 999 and alpha < 1e-6:
            break
    
    return best_h, best_s

def slsqp_active(h, n, active_shifts, maxiter=500):
    def obj(x):
        g = 0.0
        for s in active_shifts:
            if s == 0:
                g += np.sum(x * (1.0 - x))
            elif s > 0:
                g += np.sum(x[0:n-s] * (1.0 - x[s:n]))
            else:
                sp = -s
                g += np.sum(x[sp:n] * (1.0 - x[0:n-sp]))
        return g
    
    def obj_grad(x):
        return grad_active_sum(x, active_shifts, n)
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint_fun}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_feasible(res['x'], n)
        return h_opt, compute_score(h_opt)
    except Exception:
        return h, compute_score(h)

    for sharp_iter in range(10):
        h_sharp = h_sharp ** 1.5  # push toward extremes (for values in [0,1])
        h_sharp = np.clip(h_sharp, 0.0, 1.0)
        h_sharp = project_feasible(h_sharp, n)
    s_sharp = compute_score(h_sharp)
    
    # Binary flip search on best of both
    if s_sharp < best_score:
        best_h = h_sharp.copy()
        best_score = s_sharp
    
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    h_bin = project_feasible(h_bin, n)
    # Adjust to exact n/2 sum
    if h_bin.sum() != n / 2.0:
        h_bin = project_feasible(best_h, n)
    
    s_bin = compute_score(h_bin)
    if s_bin < best_score:
        best_h = h_bin.copy()
        best_score = s_bin
    
    # Binary flip search from both fractional and binary
    for base in [best_h, h_bin]:
        if time.time() > deadline:
            break
        h_flipped, s_flipped = binary_flip_search(base, n, max_rounds=30)
        if s_flipped < best_score:
            best_score = s_flipped
            best_h = h_flipped.copy()
    
    # Phase 4: Polishing with active-set gradient again
    deadline = t0 + budget_s * 0.9
    for attempt in range(20):
        if time.time() > deadline:
            break
        h_opt, s_opt = active_set_method(best_h, n, max_iter=8000)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()
        else:
            # Try smaller perturbation
            h_try = project_feasible(best_h + rng_global.randn(n) * 0.01, n)
            h_opt, s_opt = active_set_method(h_try, n, max_iter=3000)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_opt.copy()
    
    # Phase 5: Try a different n (interpolate)
    if time.time() < t0 + budget_s * 0.95:
        for n_try in [min(n * 2, 100), n // 2 if n > 20 else 20]:
            if n_try < 10 or time.time() > t0 + budget_s * 0.95:
                continue
            x_old = np.linspace(0, 2.0, n, endpoint=False)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            h_interp = project_feasible(h_interp, n_try)
            s_interp = compute_score(h_interp)
            if s_interp < best_score:
                best_score = s_interp
                best_h = h_interp.copy()
            # Quick active-set refinement
            h_opt, s_opt = active_set_method(h_interp, n_try, max_iter=3000)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_opt.copy()
    
    # Final: verify and return
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
```
