    # a[m] = coefficient of h[j] in corr[m]
    a = np.zeros(2*n-1)
    idx = np.arange(max(0, j), min(2*n-1, j+n))

        m2 = hi - (hi - lo) / 3
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 > f2:
            lo = m1
        else:
            hi = m2

    if best_h is None:
        best_h = np.ones(70) * 0.5
        best_n = 70
        best_c5 = compute_c5(best_h, 2.0 / 70)

    # Final validation
    best_h = repair(best_h, best_n / 2.0)
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
