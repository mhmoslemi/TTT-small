import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def logsumexp(x):
    m = np.max(x)
    if m > 700:  # overflow protection
        return m
    return m + np.log(np.sum(np.exp(x - m)))

def smooth_max(corr, T):
    if T < 1e-12:
        return np.max(corr)
    return logsumexp(T * corr) / T

def objective(h, dx, T, pen, target):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = smooth_max(corr, T) * dx
    val += 0.5 * pen * (h.sum() - target)**2
    return val

def projection(h, target):
    h = h.copy()
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            m = h > 1e-14
            c = m.sum()
            if c == 0: break
            step = min(excess/c, h[m].min() - 1e-14)
            h[m] -= step
        else:
            m = h < 1 - 1e-14
            c = m.sum()
            if c == 0: break
            step = min(-excess/c, (1-h[m]).min() - 1e-14)
            h[m] += step
    return h

def subgradient_step(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    # Active set (within tolerance)
    active = corr >= mx - 1e-6
    if not active.any():
        active = np.array(corr >= mx - 0.01)
    # Compute subgradient numerically
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-7
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] += eps
        h_minus = h.copy()
        h_minus[j] -= eps
        c_plus = np.correlate(h_plus, 1-h_plus, mode="full")
        c_minus = np.correlate(h_minus, 1-h_minus, mode="full")
        # Subgradient at active set (averaged)
        g = (np.max(c_plus[active]) - np.max(c_minus[active])) / (2*eps)
        grad[j] = g
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    best_h = projection(np.array(initial_h_values, dtype=float).copy(), target)
    best_c5 = compute_c5(best_h, dx)
    bounds = [(0.0, 1.0)] * n
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    def try_update(h_new):
        nonlocal best_h, best_c5
        if h_new is None:
            return
        h_new = projection(h_new, target)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5
    
    # Phase 1: L-BFGS-B with smooth max, temperature annealing
    temps = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    for T in temps:

            break
        pen = 1000.0
        h0 = best_h.copy()
        try:
            res = minimize(objective, h0, args=(dx, T, pen, target),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-8})
            try_update(res.x)
        except Exception:
            pass
    
    # Phase 2: SLSQP on smooth max (handles equality constraint)
    for T in [1.0, 0.3, 0.1, 0.05]:
        if time_left() < 30:
            break
        try:
            res = minimize(lambda h: smooth_max(np.correlate(h, 1-h, mode="full"), T) * dx,
                          best_h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda h: h.sum() - target,
                                       'jac': lambda h: np.ones(n)}],
                          options={'maxiter': 300, 'ftol': 1e-14})

        hp = h.copy()
        hp[j] = min(1.0, hp[j] + eps)
        hm = h.copy()
        hm[j] = max(0.0, hm[j] - eps)
        denom = hp[j] - hm[j]
        if denom < 1e-12:
            grad[j] = 0.0
        else:
            cp = np.correlate(hp, 1.0 - hp, mode="full")[active].mean()
            cm = np.correlate(hm, 1.0 - hm, mode="full")[active].mean()
            grad[j] = (cp - cm) / denom
    return grad

def make_slsqp_constraints():
    return [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    slsqp_cons = make_slsqp_constraints()
    
    best_h = projection(np.array(initial_h_values, dtype=float).copy(), target)
    best_c5 = compute_c5(best_h, dx)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: L-BFGS-B with smooth max, temperature annealing
    temps = [15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    for T in temps:
        if time_left() < 25:
            break
        for attempt in range(2):
            if time_left() < 10:
                break
            if attempt == 0:
                h0 = best_h.copy()
            else:
                h0 = projection(best_h + rng.randn(n) * 0.05, target)
            try:
                res = minimize(obj_lbfgsb, h0, args=(dx, T, 500.0, target),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-10})
                h_new = projection(res.x, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass
    
    # Phase 2: SLSQP with smooth max
    for T in [2.0, 1.0, 0.5, 0.2, 0.1, 0.05]:
        if time_left() < 20:
            break
        try:
            res = minimize(obj_slsqp_smooth, best_h, args=(dx, T),
                          method='SLSQP', bounds=bounds, constraints=slsqp_cons,
                          options={'maxiter': 300, 'ftol': 1e-14})
            h_new = projection(res.x, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    # Phase 3: Projected subgradient
    if time_left() > 40:
        h_sg = best_h.copy()
        c5_sg = best_c5
        alpha = 0.05
        for it in range(30000):
            if time_left() < 5:
                break
            g = subgrad_of_max(h_sg)
            h_t = projection(h_sg - alpha * g, target)
            c5_t = compute_c5(h_t, dx)
            if c5_t < c5_sg - 1e-14:
                h_sg = h_t
                c5_sg = c5_t
                if c5_t < best_c5:
                    best_h = h_t.copy()
                    best_c5 = c5_t
            if it % 500 == 0:
                alpha = max(0.001, alpha * 0.92)
    
    # Phase 4: Block coordinate descent
    if time_left() > 30:
        h_bc = best_h.copy()
        c5_bc = best_c5
        block_size = 4
        step = 0.04
        for it in range(20000):
            if time_left() < 5:
                break
            j0 = rng.randint(0, max(1, n - block_size + 1))
            idx = range(j0, min(n, j0 + block_size))
            improved = False
            for d in [step, -step]:
                h_t = h_bc.copy()
                for j in idx:
                    h_t[j] = np.clip(h_t[j] + d, 0.0, 1.0)
