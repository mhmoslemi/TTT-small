import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def all_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def find_opt_coord(h, i, n):
    h0 = h.copy()
    h0[i] = 0.0
    c0 = all_corrs(h0)
    ns = 2 * n - 1
    a = np.zeros(ns)
    for k in range(ns):
        s = k - (n - 1)
        coef = 0.0
        if 0 <= i + s < n:
            coef += (1.0 - h[i + s])
        if 0 <= i - s < n:
            coef -= h[i - s]
        a[k] = coef
    b = c0
    best_x = 0.0
    best_val = b.max()
    v1 = (a + b).max()
    if v1 < best_val:
        best_val = v1
        best_x = 1.0
    # Find mutually-active intersections
    for i1 in range(ns):
        for i2 in range(i1 + 1, ns):
            da = a[i1] - a[i2]
            if abs(da) < 1e-14:
                continue
            x_int = (b[i2] - b[i1]) / da
            if x_int <= 0.0 or x_int >= 1.0:
                continue
            vals = a * x_int + b
            mx = vals.max()
            if mx < best_val - 1e-13:
                if abs(vals[i1] - mx) < 1e-7 and abs(vals[i2] - mx) < 1e-7:
                    best_val = mx
                    best_x = x_int
    return best_x

def exact_coord_descent(h, n, max_sweeps=60, random_order=False, rng=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for i in order:
            new_x = find_opt_coord(h, i, n)
            if abs(new_x - h[i]) > 1e-12:
                h[i] = new_x
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return h, best_c5

def soft_max_obj(h, temp):
    c = all_corrs(h)
    m = c.max()
    return (m + temp * np.log(np.sum(np.exp((c - m) / temp)))) * (2.0 / len(h))

def sum_con_fun(x):
    return x.sum() - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))
