from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = np.clip(h + (d/k) * mask, 0.0, 1.0)
    return h

def _grad_single(h, k_star):
    """Gradient of v[k_star] = dx * correlate(h, 1-h)[k_star]"""
    n = len(h)
    dx = 2.0 / n
    s = k_star - (n - 1)
    
    # A[j] = 1 if 0 <= j+s < n
    A = np.zeros(n)
    lo = max(0, -s)
    hi = min(n, n - s)
    A[lo:hi] = 1.0
    
    # h_sp[j] = h[j+s] if 0<=j+s<n else 0
    h_sp = np.zeros(n)
    if s >= 0:
        h_sp[:n-s] = h[s:]
    else:
        h_sp[-s:] = h[:n+s]
    
    # h_sm[j] = h[j-s] if 0<=j-s<n else 0
    h_sm = np.zeros(n)
    if s >= 0:
        h_sm[s:] = h[:n-s]
    else:
        h_sm[:n+s] = h[-s:]
    
    return dx * (A - h_sp - h_sm)

def _project_feasible(h, n):
    """Project to {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    return _fix(h, n)

def _proj_grad_descent(h0, n, n_iter, lr_init, t0, budget, seed=42):
    """Projected gradient with subgradient of max overlap"""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h)
    
    v = 0.0  # velocity for momentum
    gamma = 0.9  # momentum
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        
        # Compute overlaps
        c = np.correlate(h, 1.0 - h, mode='full')
        v_arr = c * (2.0/n)
        k_star = int(np.argmax(v_arr))
        cur_max = v_arr[k_star]
        
        # Subgradient
        g = _grad_single(h, k_star)
        
        # Momentum step
        v = gamma * v - lr_init * g
        h_new = h + v
        
        # Project
        h_new = _project_feasible(h_new, n)
        
        h = h_new
        cur_c = _c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Decrease learning rate
        lr_init *= 0.999
    
    return best_h, best_c

        return None

        c = np.correlate(x, 1.0 - x, mode='full')
        v = c * dx
        m = np.max(v)
        e = np.exp(np.clip(v - m, -60.0, 60.0))
        return m + np.log(np.sum(e)) / alpha

        c = np.correlate(x, 1.0 - x, mode='full')
        v = c * dx
        m = np.max(v)
        e = np.exp(np.clip(v - m, -60.0, 60.0))
        w = e / np.sum(e)

        # Actually d/dx [m + log(sum(e))/alpha] where m = max(v)
        # = w (the softmax) * dx * dc/dx (approximately, for large alpha → e_k* for argmax)
        # For the smoothed version:

        # w[k] * dv[k]/dx (weighted average of gradients)
        # For now, use the argmax gradient as approximation
        return g

        ht = _fix(np.clip(res.x, 0.0, 1.0), n)

2. When i+k-(n-1) = j, i.e., i = j-k+(n-1): h[j-k+(n-1)] * (1-h[j])

- Quadratic concave: v[k] = c_k + d_k * h[j] - 2*dx*h[j]^2 (for k = n-1)

The objective max_k v[k] as a function of h[j] is the max of these piecewise linear/quadratic functions.

For the linear terms, v[k] is decreasing in h[j] if d_k < 0, which means (1 - h[j+s_k] - h[j-s_k]) < 0, i.e., h[j+s_k] + h[j-s_k] > 1.

For the quadratic term (k=n-1): 2h[j] - 2h[j]^2 is a concave parabola with maximum at h[j] = 0.5.

So the overall picture: max_k v[k] as a function of h[j] is a piecewise function that might have a minimum in the interior of [0,1].

For coordinate descent: at each coordinate j, I can:
1. Compute v[k] for all k at the current h
2. Determine which k's are "active" (near the max)
3. For the active set, compute the v[k](h[j]) functions
4. Find h[j] that minimizes the max

This is essentially a 1D optimization! I can use `minimize_scalar` on [0,1] for each coordinate.

The cost per coordinate is O(K * n) where K is the number of active constraints and n is the number of function evaluations for minimize_scalar. With K ~ 10 and n ~ 50, that's 500 ops per coordinate, 400 coordinates = 200K per sweep. Very fast!

Actually, the key insight is: when I change only h[j], the correlation `correlate(h, 1-h)` changes only in a specific pattern. Let me derive it:

correlate(h, 1-h, 'full')[k] = sum_i h[i] * (1-h[i+k-(n-1)])

If I change h[j] from old to new (delta = new - old):
- The first factor h[i] changes for i=j: delta*(1-h[j+k-(n-1)])
- The second factor (1-h[i+k-(n-1)]) changes for i+k-(n-1)=j, i.e., i=j-k+(n-1): 
  old contribution: h[j-k+(n-1)]*(1-old)
  new contribution: h[j-k+(n-1)]*(1-new)
  change: -h[j-k+(n-1)]*delta

So the change in correlate[k] is:
delta_correlate[k] = delta*(1-h[j+k-(n-1)]) - h[j-k+(n-1)]*delta
= delta * (1 - h[j+k-(n-1)] - h[j-k+(n-1)])

where the terms are 0 if indices are out of range.

This is O(n) per k, or O(n) total (since only O(n) k's are affected).

So I can update the v array in O(n) time when h[j] changes! Then finding the max is O(n) (or O(1) if I keep a max-heap).

This makes coordinate descent O(n^2) per sweep (n coordinates, O(n) update each). For n=400, that's 160K per sweep. I can do thousands of sweeps quickly!

def _coord_descent_v2(h0, n, max_sweeps, t0, budget):
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h)
    dx = 2.0 / n
    
    # Precompute initial correlation
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    cur_max = np.max(v)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget - 5:
            break
        
        for j in range(n):
            old_hj = h[j]
            old_v = v.copy()  # save for rollback
            
            # Try candidate values of h[j]
            candidates = [0.0, 1.0, 0.5]
            if 0.01 < old_hj < 0.99:
                candidates.extend([old_hj - 0.05, old_hj + 0.05, old_hj])
            
            best_hj = old_hj
            best_max = cur_max
            
            for hj in candidates:
                delta = hj - old_hj
                if abs(delta) < 1e-10:
                    val = cur_max
                else:
                    # Update v for affected k's
                    # k affected: n-1-j <= k <= n-1+j (clipped to [0, 2n-2])
                    k_lo = max(0, n - 1 - j)
                    k_hi = min(2*n - 2, n - 1 + j)
                    
                    # delta_v[k] = dx * delta * (1 - h[j+k-(n-1)]_{valid} - h[j-k+(n-1)]_{valid})
                    # But wait, this formula assumes we change h[j] by delta
                    # and all other h[i] stay the same.
                    # The term 1-h[j+k-(n-1)] uses the CURRENT h (before change)
                    # for the "other" index, which is correct since we're only changing h[j].
                    
                    s_pos = j + np.arange(k_lo, k_hi+1) - (n-1)  # = j+k-(n-1) for each k
                    s_neg = j - np.arange(k_lo, k_hi+1) + (n-1)  # = j-k+(n-1) for each k

                    h_pos = np.zeros(k_hi - k_lo + 1)
                    h_neg = np.zeros(k_hi - k_lo + 1)
                    h_pos[mask_pos] = h[s_pos[mask_pos]]
                    h_neg[mask_neg] = h[s_neg[mask_neg]]

                    v_new = v.copy()
                    v_new[k_lo:k_hi+1] += dv

                    best_max = val
                    best_hj = hj

                h[j] = best_hj

                k_hi = min(2*n - 2, n - 1 + j)

                s_neg = j - np.arange(k_lo, k_hi+1) + (n-1)
                mask_pos = (s_pos >= 0) & (s_pos < n)
