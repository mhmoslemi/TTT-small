    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-12:
            break
        # Distribute adjustment to non-bound elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += d / np.sum(mask)
        else:
            h += d / n
    h = np.clip(h, 0.0, 1.0)

    d = n / 2.0 - np.sum(h)
    if abs(d) > 1e-12:

        h = np.clip(h, 0.0, 1.0)

    return float(np.max(corr) * dx)

def repair(h, n):

        else:

    if abs(d) > 1e-14:
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d(h, m, n):
    dx = 2.0 / n
    hm = h[m]
    h[m] = 0.0
    A = np.correlate(h, 1.0 - h, mode='full') * dx
    h[m] = 1.0

        if np.max(A + B * t1) < np.max(A + B * t2):
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0
