    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, n, maxiter=500):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x, n)
    def con(x):
        return float(np.sum(x)) - target
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)

        return h0, compute_c5(h0, n)

            best_c = best
            best_hj = old

            else:

    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, n, maxiter=600):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x, n)
    def con(x):
        return float(np.sum(x)) - target
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def coord_descent(h, n, rng, n_sweeps=80, fine=False):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            best_c = best
            best_val = old
            if fine:
                deltas = [0.0, 1.0-0.0, -0.1, 0.1, -0.05, 0.05, -0.02, 0.02, -0.01, 0.01, -0.005, 0.005]
            else:
                deltas = [0.0, 1.0, -0.5, 0.5, -0.2, 0.2, -0.1, 0.1, -0.05, 0.05, -0.02, 0.02, -0.01, 0.01]
            for d in deltas:
                c = 0.0 if d == 0.0 else (1.0 if d == 1.0 else old + d)
                if c < 0.0 or c > 1.0 or abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h_proj = project_to_set(h, target)
                cv = compute_c5(h_proj, n)
                if cv < best_c:
                    best_c = cv
                    best_val = c
            h[j] = old
            if best_val != old:
                h[j] = best_val
                h = project_to_set(h, target)
                best = best_c
                improved = True
        if not improved:
            break
    return h, best

def swap_search(h, n, rng, n_tries=20000):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_tries):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        cv = compute_c5(h, n)
        if cv < best:
            best = cv
        else:
            h[i], h[j] = h[j], h[i]
    return h, best

def pair_swap_descent(h, n, rng, n_sweeps=30):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for i in order:
            for j in range(i+1, n):
                if (j - i) > 15:
                    break
                old_i, old_j = h[i], h[j]
                h[i], h[j] = h[j], h[i]
                h_p = project_to_set(h, target)
                cv = compute_c5(h_p, n)
                if cv < best:
                    best = cv
                    improved = True
                    h = h_p
                else:
                    h[i], h[j] = old_i, old_j
        if not improved:
            break
    return h, best

    return max(0.0, budget - (time.time() - t0))
