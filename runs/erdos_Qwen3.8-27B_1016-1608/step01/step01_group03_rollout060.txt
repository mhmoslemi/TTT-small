= (1-h[j-k]) [if j-k is valid] - h[j+k] [if j+k is valid]

def linearized_step(h, dx, target, n_lags=200):
    """One linearized min-max step."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = np.max(corr)
    
    # Get top lags
    top_idx = np.argsort(corr)[-min(n_lags, 2*n-1):]
    
    # Build LP: min t s.t. corr_k(h) ≤ t for top lags, sum(h)=target, 0≤h≤1
    # Linearize: corr_k(h) ≈ corr_k(h0) + grad_k·(h-h0)
    # => grad_k·h ≤ t - corr_k(h0) + grad_k·h0 = t - (corr_k(h0) - grad_k·h0)
    
    # Actually: corr_k(h) ≈ corr_k0 + grad_k · (h - h0)
    # We want: corr_k0 + grad_k · (h - h0) ≤ t
    # => grad_k · h ≤ t - corr_k0 + grad_k · h0
    
    from scipy.optimize import linprog
    
    # Variables: [h (n), t (1)]
    # Minimize: t => c = [0,...,0, 1]
    # Constraints: for each top lag k: grad_k · h - t ≤ -(corr_k0 - grad_k · h0)
    #              sum(h) = target
    #              0 ≤ h ≤ 1
    #              t ≥ 0
    
    A_ub = []
    b_ub = []
    for k in top_idx:
        # grad_k for this lag
        gk = np.zeros(n)
        # k is index in full correlation (0 to 2n-2), center at n-1
        lag = k - (n - 1)
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            for i in range(n - lag):
                gk[i+lag] += (1 - h[i])
                gk[i] -= h[i+lag]
        # constraint: grad_k · h - t ≤ -(corr_k0 - grad_k·h0)
        rhs = -(corr[k] - np.dot(gk, h))
        row = np.zeros(n + 1)
        row[:n] = gk
        row[n] = -1
        A_ub.append(row)
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 ≤ h ≤ 1, t ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        h_new = res.x[:n]
        h_new = np.clip(h_new, 0, 1)
        h_new = repair(h_new, target)
        return h_new
    return h

    # Phase 2: Linearized min-max (LP-based)
    if elapsed() < budget_s - 250:
        h_lm, c_lm = linmax_step(best_h, dx, tgt, max_iters=40)
        if c_lm < best_c5 - 1e-7:
            best_c5, best_h = c_lm, h_lm

    # Phase 3: Nelder-Mead polish
    if elapsed() < budget_s - 200:
        try:
            r = nm_solve(best_h, dx, tgt, n)
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5 - 1e-7:
                best_c5, best_h = cn, hn.copy()
        except Exception:
            pass

    # Phase 4: Coordinate descent
    if elapsed() < budget_s - 180:
        h_cd, c_cd = coord_desc(best_h, dx, tgt, 4000, 0.02, rng)
        if c_cd < best_c5 - 1e-7:
            best_c5, best_h = c_cd, h_cd
        h_cd, c_cd = coord_desc(best_h, dx, tgt, 4000, 0.005, rng)
        if c_cd < best_c5 - 1e-7:
            best_c5, best_h = c_cd, h_cd

    # Phase 5: More linearized steps after CD
    if elapsed() < budget_s - 150:
        h_lm, c_lm = linmax_step(best_h, dx, tgt, max_iters=30)
        if c_lm < best_c5 - 1e-7:
            best_c5, best_h = c_lm, h_lm

    # Phase 6: Multi-start SLSQP (small perturbations)
    if elapsed() < budget_s - 100:
        for i in range(25):
            if elapsed() > budget_s - 80:
                break
            sc = 0.01 + 0.03 * (1.0 - i / 25.0)
            hp = repair(best_h + rng.randn(n) * sc, tgt)
            try:
                r = slsqp_solve(hp, dx, tgt, 100.0, n)
                ho = repair(r.x, tgt)
                co = compute_c5(ho, dx)
                if co < best_c5 - 1e-7:
                    best_c5, best_h = co, ho.copy()
            except Exception:
                pass

    # Phase 7: Try other n values
    for n_try in [56, 60, 66, 72, 48, 50]:
        if elapsed() > budget_s - 30:
            break
        tgt_t = n_try / 2.0
        dx_t = 2.0 / n_try
        for s in range(5):
            if elapsed() > budget_s - 20:
                break
            h_rand = random_binary(n_try, rng)
            # Quick SLSQP
            try:
                r = slsqp_solve(h_rand, dx_t, tgt_t, 50.0, n_try)
                ho = repair(r.x, tgt_t)
                co = compute_c5(ho, dx_t)
                if co < best_c5 - 1e-7:
                    best_c5, best_h, best_n = co, ho, n_try
                    # Follow up with linmax
                    if elapsed() < budget_s - 15:
                        ho2, co2 = linmax_step(ho, dx_t, tgt_t, max_iters=15)
                        if co2 < best_c5 - 1e-7:
                            best_c5, best_h, best_n = co2, ho2, n_try
            except Exception:
                pass
            # Also try with slight randomization
            h_rand2 = repair(random_binary(n_try, rng) + rng.randn(n_try) * 0.05, tgt_t)
            try:
                r = slsqp_solve(h_rand2, dx_t, tgt_t, 100.0, n_try)
                ho = repair(r.x, tgt_t)
                co = compute_c5(ho, dx_t)
                if co < best_c5 - 1e-7:
                    best_c5, best_h, best_n = co, ho, n_try
            except Exception:
                pass

    # Phase 8: Final deep polish on best
    if elapsed() < budget_s - 10:
        tgt_f = best_n / 2.0
        dx_f = 2.0 / best_n
        # LSE SLSQP with high temp
        try:
            r = slsqp_solve(best_h, dx_f, tgt_f, 500.0, best_n)
            hf = repair(r.x, tgt_f)
            cf = compute_c5(hf, dx_f)
            if cf < best_c5 - 1e-9:
                best_c5, best_h = cf, hf
        except Exception:
            pass
        # One more linmax
        if elapsed() < budget_s - 5:
            h_lm, c_lm = linmax_step(best_h, dx_f, tgt_f, max_iters=20)
            if c_lm < best_c5 - 1e-9:
                best_c5, best_h = c_lm, h_lm
        # Tiny CD
        if elapsed() < budget_s - 3:
            h_cd, c_cd = coord_desc(best_h, dx_f, tgt_f, 1000, 0.003, rng)
            if c_cd < best_c5 - 1e-9:
                best_c5, best_h = c_cd, h_cd

    # Final validation
    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
