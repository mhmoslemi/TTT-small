    b = beta * f

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    b = beta * f
    bm = np.max(b)
    val = bm + np.log(np.sum(np.exp(b - bm)))
    viol = (np.sum(h) - n / 2.0) ** 2
    return val + lam * viol

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def l_hook_opt(h0, n, beta, lam, maxiter):
    h0 = project_to_set(h0.copy(), n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0, args=(n, beta, lam), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_to_set(res.x.copy(), n / 2.0)
    except Exception:
        return h0

def coord_descent(h, n, rng, sweeps, steps):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            bc, bv = old, best
            for s in steps:
                c = old + s
                if c < 0 or c > 1 or abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv, bc = v, c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h, best

def make_structured(n, rng):
    h = np.zeros(n)
    nb = rng.randint(3, 9)
    bl = n // nb
    rem = n - bl * nb
    sz = [bl + (1 if i < rem else 0) for i in range(nb)]
    pos, on = 0, 0
    for s in sz:
        if on % 2 == 0:
            h[pos:pos + s] = 1.0
        pos += s
        on += 1
    h += rng.uniform(-0.12, 0.12, n)
    return project_to_set(h, n / 2.0)

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h
