```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_obj_grad(h, n, top_k=3):
    """Compute objective and subgradient for max_j corr_j."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Top-k indices
    top_idx = np.argsort(corr)[-top_k:]
    n_half = n - 1
    g = np.zeros(n)
    for ti in top_idx:
        j = ti - n_half  # lag
        if j >= 0:
            # f_j = sum_{m=0}^{n-1-j} h[m+j]*(1-h[m])
            # grad[x] += (1-h[x-j]) for j<=x<=n-1
            # grad[x] += -h[x+j] for 0<=x<=n-1-j
            if j > 0:
                g[j:] += (1.0 - h[:n-j])
            g[:n-j] += -h[j:]
        else:
            p = -j
            if p > 0:
                g[:n-p] += (1.0 - h[p:])
                g[p:] += -h[:n-p]
    g /= top_k
    return float(np.max(corr) * 2.0/n), g, corr * (2.0/n)

def coord_descent(h, n, n_passes=10, rng=None):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            old = h[j]
            # Try setting to 0, 1, or old ± step
            cands = [0.0, 1.0, max(0.0, old - 0.1), min(1.0, old + 0.1)]
            best_c, best_v = old, best
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = project_to_constraint(h, target)
                v = compute_c5(h, n)
                if v < best_v:
                    best_v, best_c = v, c
                h[j] = old
            if best_c != old:
                h[j] = best_c
                h = project_to_constraint(h, target)
                best = best_v
                improved = True
        if not improved:
            break
    return h, best

def swap_moves(h, n, n_iter=2000, rng=None):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iter):
        i, j = rng.choice(n, 2, replace=False)
        if abs(h[i] - h[j]) < 1e-9:
            continue
        h[i], h[j] = h[j], h[i]
        v = compute_c5(h, n)
        if v < best:
            best = v
        else:
            h[i], h[j] = h[j], h[i]
    return h, best
```

```python
def smooth_max_grad(h, n, T=50.0):
    """Compute smoothed max objective and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx  # shape (2n-1,)
    
    # Softmax weights
    max_val = np.max(vals)
    exp_vals = np.exp(T * (vals - max_val))
    weights = exp_vals / np.sum(exp_vals)
    
    obj = max_val + np.log(np.sum(exp_vals)) / T
    
    # Gradient: sum_j weights[j] * grad(corr_j) * dx
    g = np.zeros(n)
    n_half = n - 1
    for j in range(2*n - 1):
        w = weights[j]
        if abs(w) < 1e-12:
            continue
        lag = j - n_half
        if lag >= 0:
            if lag > 0:
                g[lag:] += w * (1.0 - h[:n-lag])
            g[:n-lag] += w * (-h[lag:])
        else:
            p = -lag
            if p > 0:
                g[:n-p] += w * (1.0 - h[p:])
                g[p:] += w * (-h[:n-p])
    g *= dx
    return obj, g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smoothmax_grad(h, n, T=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    v = vals - np.max(vals)
    ew = np.exp(T * v)
    w = ew / ew.sum()
    obj = np.max(vals) + np.log(ew.sum()) / T
    
    g = np.zeros(n)
    nh = n - 1
    for j in range(2 * n - 1):
        wi = w[j]
        if wi < 1e-10:
            continue
        lag = j - nh
        if lag >= 0:
            if lag > 0:
                g[lag:] += wi * (1.0 - h[:n - lag])
            g[:n - lag] += -wi * h[lag:]
        else:
            p = -lag
            if p > 0:
                g[:n - p] += wi * (1.0 - h[p:])
                g[p:] += -wi * h[:n - p]
    g *= dx
    return obj, g

def subgradient_opt(h0, n, n_iter=3000, T=50.0, alpha0=0.5, rho=0.005, rng=None):
    target = n / 2.0
    h = project_to_constraint(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for t in range(n_iter):
        alpha = alpha0 / (1.0 + t * rho)
        obj, g = smoothmax_grad(h, n, T)
        h_new = project_to_constraint(h - alpha * g, target)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h, n, n_passes=5, steps=None):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    if steps is None:
        steps = [0.1, 0.05, 0.02, 0.01, 0.005]
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            old = h[j]
            best_c, best_v = old, best
            for s in steps:
                for d in [-1.0, 1.0]:
                    c = old + d * s
                    if c < -0.01 or c > 1.01:
                        continue
                    c = max(0.0, min(1.0, c))
                    h[j] = c
                    h = project_to_constraint(h, target)
                    v = compute_c5(h, n)
                    if v < best_v:
                        best_v, best_c = v, c
                    h[j] = old
            if best_c != old:
                h[j] = best_c
                h = project_to_constraint(h, target)
                best = best_v
                improved = True
        if not improved:
            break
    return h, best

def swap_search(h, n, n_iter=3000, rng=None):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iter):
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j or abs(h[i] - h[j]) < 1e-9:
            continue
        h[i], h[j] = h[j], h[i]
        v = compute_c5(h, n)
        if v < best:
            best = v
        else:
            h[i], h[j] = h[j], h[i]
    return h, best

def thue_morse(n):
    """Thue-Morse binary sequence of length n."""
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_block_pattern(n, rng):
    """Create alternating block pattern with random block sizes."""
    h = np.zeros(n)
    pos = 0
    val = rng.choice([0.0, 1.0])
    while pos < n:
        blen = rng.randint(max(1, n // 20), max(2, n // 5) + 1)
        blen = min(blen, n - pos)
        h[pos:pos + blen] = val
        val = 1.0 - val
        pos += blen
    return project_to_constraint(h, n / 2.0)

    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)
```

result[i] = c_{i-(N-1)} = sum_n a[n + i-(N-1)] * v[n]

Let lag = i - (N-1). Then:
result[i] = sum_n a[n+lag] * v[n] (where n+lag must be in [0, N-1])

For our case: a = h, v = 1-h.
result[i] = sum_n h[n+lag] * (1-h[n])

For lag ≥ 0: n ranges from 0 to N-1-lag.
result[N-1+lag] = sum_{n=0}^{N-1-lag} h[n+lag] * (1-h[n])

For lag < 0 (lag = -p): n ranges from p to N-1.
result[N-1-p] = sum_{n=p}^{N-1} h[n-p] * (1-h[n])

Now, gradient of result[N-1+lag] = sum_{n=0}^{N-1-lag} h[n+lag](1-h[n]) w.r.t. h[x]:
- h[x] appears as h[n+lag] when n = x-lag, valid if 0 ≤ x-lag ≤ N-1-lag, i.e., lag ≤ x ≤ N-1.
  Contribution: (1-h[x-lag])
- h[x] appears as h[n] when n = x, valid if 0 ≤ x ≤ N-1-lag.
  Contribution: -h[x+lag]

So ∂/∂h[x] of result[N-1+lag] = (1-h[x-lag]) * [lag ≤ x] - h[x+lag] * [x ≤ N-1-lag]

For gradient of result[N-1-p] = sum_{n=p}^{N-1} h[n-p](1-h[n]) w.r.t. h[x]:
- h[x] appears as h[n-p] when n = x+p, valid if p ≤ x+p ≤ N-1, i.e., 0 ≤ x ≤ N-1-p.
  Contribution: (1-h[x+p])
- h[x] appears as h[n] when n = x, valid if p ≤ x ≤ N-1.
  Contribution: -h[x-p]

So ∂/∂h[x] of result[N-1-p] = (1-h[x+p]) * [x ≤ N-1-p] - h[x-p] * [x ≥ p]

```python
import numpy as np
from scipy.optimize import minimize
import time

_start_time = 0.0
_budget = 1100.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smoothmax_grad(h, n, T=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    v = vals - np.max(vals)
    ew = np.exp(T * v)
    w = ew / ew.sum()
    obj = float(np.max(vals) + np.log(ew.sum()) / T)
    g = np.zeros(n)
    nh = n - 1
    for j in range(2 * n - 1):
        wi = w[j]
        if wi < 1e-10:
            continue
        lag = j - nh
        if lag >= 0:
            if lag > 0:
                g[lag:] += wi * (1.0 - h[:n - lag])
            g[:n - lag] += -wi * h[lag:]
        else:
            p = -lag
            if p > 0:
                g[:n - p] += wi * (1.0 - h[p:])
                g[p:] += -wi * h[:n - p]
    g *= dx
    return obj, g

def subgradient_opt(h0, n, n_iter=2000, T=50.0, alpha0=0.4, rho=0.003):
    target = n / 2.0
    h = project_to_constraint(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for t in range(n_iter):
        alpha = alpha0 / (1.0 + t * rho)
        obj, g = smoothmax_grad(h, n, T)
        h_new = project_to_constraint(h - alpha * g, target)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_descent(h, n, n_passes=5):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.1, 0.05, 0.02, 0.01, 0.005]
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            old = h[j]
            best_c, best_v = old, best
            for s in steps:
                for d in (-1.0, 1.0):
                    c = max(0.0, min(1.0, old + d * s))
                    if abs(c - old) < 1e-9:
                        continue
                    h[j] = c
                    h = project_to_constraint(h, target)
                    v = compute_c5(h, n)
                    if v < best_v:
                        best_v, best_c = v, c
                    h[j] = old
            if best_c != old:
                h[j] = best_c
                h = project_to_constraint(h, target)
                best = best_v
                improved = True
        if not improved:
            break
    return h, best

def swap_search(h, n, n_iter=3000, rng=None):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iter):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or abs(h[i] - h[j]) < 1e-9:
            continue
        h[i], h[j] = h[j], h[i]
        v = compute_c5(h, n)
        if v < best:
            best = v
        else:
            h[i], h[j] = h[j], h[i]
    return h, best

def block_perturb(h, n, rng, n_iter=500):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iter):
        L = max(2, n // (rng.randint(5, 15)))
        start = int(rng.integers(0, n - L))
        h_new = h.copy()
        h_new[start:start+L] = h_new[start:start+L].astype(bool) ^ True
        h_new = project_to_constraint(h_new, target)
        v = compute_c5(h_new, n)
        if v < best:
            best = v
            h = h_new
    return h, best

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_block_pattern(n, rng):
    h = np.zeros(n)
    pos = 0
    val = rng.choice([0.0, 1.0])
    while pos < n:
        blen = rng.randint(max(1, n // 20), max(2, n // 4) + 1)
        blen = min(blen, n - pos)
        h[pos:pos+blen] = val
        val = 1.0 - val
        pos += blen
    return project_to_constraint(h, n / 2.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)
```
