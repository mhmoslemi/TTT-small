    return float(np.sum(x) - _opt['n'] / 2.0)

def _eqjac(x):
    return np.ones(_opt['n'])

def _slsqp(h, n, eps, mi, ft, obj=None):
    _opt['n'] = n
    _opt['eps'] = eps
    if obj is None:
        obj = _smack
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _eqfun, 'jac': _eqjac}]
    r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cs,
                 options={'maxiter': mi, 'ftol': ft})
    return repair(r.x, n)

def _deobj(x):
    n = _opt['n']
    h = repair(x, n)
    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 200
    _opt['n'] = n

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        hp = None
    if hp is not None and len(hp) > 1:
        xo = np.linspace(0, 1, len(hp), endpoint=False)
        xn = np.linspace(0, 1, n, endpoint=False)
        h0 = np.interp(xn, xo, hp)
    else:
        h0 = np.full(n, 0.5)
    h0 = repair(h0, n)
    bh = h0.copy()
    bc = compute_c5(bh)

    # Stage 1: Multi-scale SLSQP
    for eps in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() - t0 > budget_s - 350:
            break
        hn = _slsqp(bh, n, eps, 400, 1e-12)
        c5v = compute_c5(hn)
        if c5v < bc:
            bc, bh = c5v, hn

    # Stage 2: Active-set refinement
    if time.time() - t0 < budget_s - 350:
        dx2 = 2.0 / n
        corr = np.correlate(bh, 1.0 - bh, mode='full') * dx2
        thr = np.max(corr) - 0.008
        active = np.where(corr >= thr)[0]
        if 0 < len(active) < 300:
            _opt['active'] = active
            hn = _slsqp(bh, n, 0.0001, 400, 1e-14, obj=_active_ob)
            c5v = compute_c5(hn)
            if c5v < bc:
                bc, bh = c5v, hn

    # Stage 3: DE global exploration
    if time.time() - t0 < budget_s - 300:
        bds = [(0.0, 1.0)] * n
        try:
            res = differential_evolution(_deobj, bds, seed=seed,
                                          maxiter=200, popsize=30,
                                          mutation=(0.3, 0.9),
                                          recombination=0.8,
                                          tol=1e-10, polish=False)
            hn = repair(res.x, n)
            c5v = compute_c5(hn)
            if c5v < bc:
                bc, bh = c5v, hn
        except Exception:
            pass
        if time.time() - t0 < budget_s - 250:
            for eps2 in [0.002, 0.001, 0.0005, 0.0002]:
                if time.time() - t0 > budget_s - 250:
                    break
                hn = _slsqp(bh, n, eps2, 300, 1e-13)
                c5v = compute_c5(hn)
                if c5v < bc:
                    bc, bh = c5v, hn

    # Stage 4: Random restarts + SLSQP
    if time.time() - t0 < budget_s - 150:
        for ri in range(25):
            if time.time() - t0 > budget_s - 150:
                break
            amp = np.random.uniform(0.03, 0.18)
            h2 = bh + np.random.randn(n) * amp
            h2 = repair(h2, n)
            hn = _slsqp(h2, n, 0.003, 200, 1e-12)
            c5v = compute_c5(hn)
            if c5v < bc:
                bc, bh = c5v, hn

    # Stage 5: Local transfer search (main workhorse)
    if time.time() - t0 < budget_s - 60:
        for it in range(3000000):
            if time.time() - t0 > budget_s - 60:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            d = np.random.uniform(0.003, 0.10)
            h2 = bh.copy()
            h2[i] += d
            h2[j] -= d
            if h2[i] < 0.0 or h2[i] > 1.0 or h2[j] < 0.0 or h2[j] > 1.0:
                continue
            c5v = compute_c5(h2)
            if c5v < bc:
                bc, bh = c5v, h2

    # Stage 6: Second SLSQP pass
    if time.time() - t0 < budget_s - 30:
        hn = _slsqp(bh, n, 0.0002, 500, 1e-14)
        c5v = compute_c5(hn)
        if c5v < bc:
            bc, bh = c5v, hn

    # Stage 7: Tiny perturbations
    if time.time() - t0 < budget_s - 10:
        for _ in range(500000):
            if time.time() - t0 > budget_s - 5:
                break
            h2 = bh + np.random.randn(n) * 0.002
            h2 = repair(h2, n)
            c5v = compute_c5(h2)
            if c5v < bc:
                bc, bh = c5v, h2

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        hn = _slsqp(bh, n, 0.0001, 300, 1e-14)
        c5v = compute_c5(hn)
        if c5v < bc:
            bc, bh = c5v, hn

    fh = repair(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
