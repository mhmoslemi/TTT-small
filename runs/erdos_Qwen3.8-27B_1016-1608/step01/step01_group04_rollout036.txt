    # Phase 8: Try n=160 (interpolated)
    if remain() > 100 and n < 160:
        n2 = n * 2
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h160 = np.interp(x_new, x_old, best_h)
        h160 = repair(h160, n2)
        c160 = compute_c5(h160, n2)
        if c160 < best_c + 0.005:  # worth trying
            for lam in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                if remain() < 30:
                    break
                _g_n = n2
                hn2 = slsqp_opt(h160, n2, lam, 2000, 1e-16)
                cn2 = compute_c5(hn2, n2)
                if cn2 < c160:
                    c160, h160 = cn2, hn2
            if c160 < best_c:
                best_c, best_h, n = c160, h160, n2

    # Final verification
    best_h = repair(best_h, n)
    best_c = compute_c5(best_h, n)
    return (best_h.tolist(), float(best_c), n)
