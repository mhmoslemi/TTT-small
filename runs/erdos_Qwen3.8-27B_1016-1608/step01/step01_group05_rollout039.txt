                a *= 0.5
        if not imp:
            break
        a = min(a * 1.04, a0 * 2)
    return h, best

def _slo(x):
    return compute_c5(x, _slq_n)

def _slc(x):
    return float(np.sum(x) - _slq_tgt)

def _slcj(x):
    return np.ones(_slq_n)

def slsqp_opt(h0, n, mi=300):
    global _slq_n, _slq_tgt
    _slq_n, _slq_tgt = n, n / 2.0
    h0 = project_to_set(h0.copy(), n / 2.0)
    try:
        r = minimize(_slo, h0, method='SLSQP',
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':_slc,'jac':_slcj}],
                     options={'maxiter':mi,'ftol':1e-14})
        ho = project_to_set(r.x.copy(), n/2.0)
        return ho, compute_c5(ho, n)
    except Exception:
        return h0, compute_c5(h0, n)

def coord_polish(h, n, rng, ns=40):
    tgt = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(ns):
        order = rng.permutation(n)
        imp = False
        for j in order:
            old = h[j]
            for c in (0., 1., .5, max(0.,old-.05), min(1.,old+.05),
                      max(0.,old-.02), min(1.,old+.02)):
                if abs(c-old) < 1e-8: continue
                h[j] = c
                h = project_to_set(h, tgt)
                v = compute_c5(h, n)
                if v < best:
                    best, imp = v, True
                else:
                    h[j] = old
                    h = project_to_set(h, tgt)
                    break
            else:
                h[j] = old
                h = project_to_set(h, tgt)
        if not imp: break
    return h, best

def tleft():
    return max(0.0, _budget - (time.time() - _start_t))

def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t
    _budget, _start_t = float(budget_s), time.time()
    rng = np.random.RandomState(seed)
    bh, bc, bn = None, np.inf, 0

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.clip(0.5 + 0.3*np.sin(np.linspace(0, 4*np.pi, 90)), 0, 1)
    n1 = len(h0) if len(h0) > 0 else 90

    # === Phase 1: n=90 active set ===
    h_a, c_a = active_opt(h0, n1, max_it=20000, na=1, a0=0.05)
    if c_a < bc: bh, bc, bn = h_a, c_a, n1
    h_a2, c_a2 = active_opt(h0, n1, max_it=20000, na=3, a0=0.03)
    if c_a2 < bc: bh, bc, bn = h_a2, c_a2, n1

    # Perturbed restarts at n=90
    while tleft() > _budget * 0.45:
        sp = 0.01 + 0.01 * rng.random()
        hp = bh + rng.normal(0, sp, bn)
        hp = project_to_set(hp, bn/2.0)
        ho, co = active_opt(hp, bn, max_it=8000, na=2, a0=0.03)
        if co < bc: bh, bc, bn = ho, co, bn

    # SLSQP + coord polish at n=90
    hs, cs = slsqp_opt(bh, bn, mi=500)
    if cs < bc: bh, bc, bn = hs, cs, bn
