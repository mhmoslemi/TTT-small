def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj(h, target):
    h = np.clip(np.array(h, float, copy=True), 0.0, 1.0)
    for _ in range(200):
        e = h.sum() - target
        if abs(e) < 1e-13:
            break
        if e > 0:
            m = h > 0
            if not m.any():
                break
            s = min(e / m.sum(), h[m].min())
            if s < 1e-15:
                break
            h[m] -= s
        else:
            m = h < 1
            if not m.any():
                break
            s = min(-e / m.sum(), (1.0 - h[m]).min())
            if s < 1e-15:
                break
            h[m] += s
    return h

def sobg(h, dx, temp):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    s = (c - m) / temp
    ex = np.exp(s)
    se = ex.sum()
    val = (m + temp * np.log(se / len(c))) * dx
    w = ex / se
    g = np.zeros(n)
    sig = np.where(w > 1e-8)[0]
    for i in sig:
        wi = w[i]
        k = i - (n - 1)
        lo, hi = max(0, -k), min(n, n - k)
        g[lo:hi] -= wi * h[lo + k:hi + k]
        j2l, j2h = max(0, lo + k), min(n, hi + k)
        if j2l < j2h:
            g[j2l:j2h] += wi * (1.0 - h[j2l - k:j2h - k])
    g *= dx
    return val, g

def pen_obj(x, dx, target, pen):

    val += pen * (x.sum() - target) ** 2
    val += pen * np.sum(np.maximum(0, -x) ** 2 + np.maximum(0, x - 1) ** 2)

def subgrad_step(h, dx, tgt, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr) - (n - 1)

    g = np.zeros(n)
    g[lo:hi] -= h[lo + k_star:hi + k_star]
    j2l, j2h = max(0, lo + k_star), min(n, hi + k_star)
    if j2l < j2h:
        g[j2l:j2h] += (1.0 - h[j2l - k_star:j2h - k_star])
    h_new = proj(h - alpha * g * dx, tgt)
    return h_new
