def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(corr)
    return (M + temp * np.log(np.sum(np.exp((corr - M) / temp)))) * dx

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            return h
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:

            s = min(-excess / cnt, (1.0 - h[mask]).min())
