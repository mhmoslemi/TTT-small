import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = (h < 1.0).astype(float)
        else:
            mask = (h > 0.0).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def gradient_at_max(c, h, n):
    """Compute subgradient of max(c) w.r.t. h, where c = correlate(h,1-h,'full')*dx."""
    dx = 2.0 / n
    max_val = np.max(c)
    # Active set: all k where c[k] >= max_val - 1e-6
    active = np.where(c >= max_val - 1e-6)[0]
    grad = np.zeros(n)
    
    for k in active:
        m = (n - 1) - k  # the shift parameter
        if m >= 0:
            # valid i: 0 to n-1-m
            # Term 1: p in [0, n-1-m], grad[p] += (1-h[p+m])
            p_range = np.arange(0, n - m)
            grad[p_range] += (1.0 - h[p_range + m])
            # Term 2: p in [m, n-1], grad[p] += -h[p-m]
            p_range2 = np.arange(m, n)
            grad[p_range2] += -h[p_range2 - m]
        else:
            s = -m
            # valid i: s to n-1
            # Term 1: p in [s, n-1], grad[p] += (1-h[p-s])
            p_range = np.arange(s, n)
            grad[p_range] += (1.0 - h[p_range - s])
            # Term 2: p in [0, n-1-s], grad[p] += -h[p+s]
            p_range2 = np.arange(0, n - s)
            grad[p_range2] += -h[p_range2 + s]
    
    grad *= dx
    grad /= len(active)  # average over active set
    return grad

    best_h = h.copy()
    best_s = compute_score(h)
    
    alpha = 0.05  # initial step size
    momentum = 0.0
    mom_coeff = 0.7

        grad = gradient_at_max(c, h, n)

        h_new = h + momentum
        h_new = project_feasible(h_new, n)

            best_h = h_new.copy()
        h = h_new

    cm = np.max(c)

    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

            s = compute_score(h_new)
            if s < compute_score(h):

        except Exception:

            s = compute_score(h_new)
            if s < compute_score(h):

        except Exception:

def local_search(h, n, n_iter=5000, rng=None):

    if rng is None:

        s = compute_score(h)
        if s < best_s:
            best_s = s
            # Keep
        else:
            # Revert
            h[idx] = h_old
            h = project_feasible(h, n)
    
    return h, best_s

import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0).astype(float) if diff > 0 else (h > 0.0).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def grad_active(c, h, n, n_active=1):
    dx = 2.0 / n
    max_val = np.max(c)
    tol = 1e-10 * max_val + 1e-12
    active = np.where(c >= max_val - tol)[0]
    if len(active) > n_active:
        active = active[np.argsort(c[active])[::-1][:n_active]]
    grad = np.zeros(n)
    for k in active:
        m = (n - 1) - k
        if m >= 0:
            p1 = np.arange(0, n - m)
            grad[p1] += (1.0 - h[p1 + m])
            p2 = np.arange(m, n)
            grad[p2] += -h[p2 - m]
        else:
            s = -m
            p1 = np.arange(s, n)
            grad[p1] += (1.0 - h[p1 - s])
            p2 = np.arange(0, n - s)
            grad[p2] += -h[p2 + s]
    grad *= dx / len(active)
    return grad

def subgradient_opt(h, n, n_iter=15000):
    h = h.copy()
    best_h = h.copy()
    best_s = compute_score(h)
    alpha = 0.04
    mom = np.zeros(n)
    mu = 0.6
    stalled = 0
    for it in range(n_iter):
        c = compute_corr(h)
        g = grad_active(c, h, n, n_active=2)
        mom = mu * mom - alpha * g
        h = project_feasible(h + mom, n)
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
        if it % 100 == 0 and it > 0:
            alpha *= 0.92
        if stalled > 3000 and it > 500:
            h = project_feasible(best_h + np.random.randn(n) * 0.02, n)
            stalled = 0
            alpha = 0.02
    return best_h, best_s

def smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(np.clip(tau * (c - cm), -500, 500)))) / tau

def polish(h, n, t_limit):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    best_h, best_s = h.copy(), compute_score(h)
    for tau in [200.0, 500.0]:
        if time.time() - t0 > t_limit * 0.6:
            break
        def obj_smooth(x):
            return smooth_obj(x, tau) + 50.0 * (np.sum(x) - tgt) ** 2
        try:
            r = minimize(obj_smooth, best_h, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 3000, 'ftol': 1e-15})
            hn = project_feasible(r['x'], n)
            if compute_score(hn) < best_s:
                best_s = compute_score(hn)
                best_h = hn
        except Exception:
            pass
    if time.time() - t0 < t_limit:
        def obj_exact(x):
            return compute_score(x) + 500.0 * (np.sum(x) - tgt) ** 2
        try:
            r = minimize(obj_exact, best_h, method='SLSQP', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-14})
            hn = project_feasible(r['x'], n)
            s = compute_score(hn)
            if s < best_s:
                best_s, best_h = s, hn
        except Exception:
            pass
    return best_h, best_s

def local_search(h, n, n_iter, rng):
    h = h.copy()
    best_s = compute_score(h)
    temp = 0.15
    for it in range(n_iter):
        idx = rng.choice(n, size=rng.randint(1, 4), replace=False)
        old = h[idx].copy()
        h[idx] += rng.uniform(-0.4, 0.4, size=len(idx))
        h = project_feasible(h, n)
        s = compute_score(h)
        if s < best_s:
            best_s = s
        else:
            if rng.random() > np.exp(-(s - best_s) / max(temp, 1e-8)):
                h[idx] = old
                h = project_feasible(h, n)
        temp *= 0.9997
    return h, best_s
