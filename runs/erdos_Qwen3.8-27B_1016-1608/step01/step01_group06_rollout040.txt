import numpy as np

from scipy.optimize import minimize, minimize_scalar

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def soft_obj(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    s = np.sum(np.exp(tau * (c - cm)))
    val = cm + np.log(s) / tau
    pen = 1e5 * (np.sum(h) - n / 2.0) ** 2
    return val + pen

def soft_grad(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    e = np.exp(tau * (c - cm))
    w = e / e.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        if w[k_idx] < 1e-16:
            continue
        coeff = w[k_idx] * dx / tau
        # First term: (1-h[i+k]) for valid i+k
        if k >= 0:
            i_s, i_e = 0, n - k
            j_s, j_e = k, n
        else:
            i_s, i_e = -k, n
            j_s, j_e = 0, n + k
        if i_e > i_s:
            grad[i_s:i_e] += coeff * (1.0 - h[j_s:j_e])
        # Second term: -h[i-k] for valid i-k
        if k >= 0:
            i_s2, i_e2 = k, n
            j_s2, j_e2 = 0, n - k
        else:
            i_s2, i_e2 = 0, n + k
            j_s2, j_e2 = -k, n
        if i_e2 > i_s2:
            grad[i_s2:i_e2] -= coeff * h[j_s2:j_e2]
    # penalty gradient
    grad += 2e5 * (np.sum(h) - n / 2.0)
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

- Taking the minimum

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def soft_obj(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    s = np.sum(np.exp(tau * (c - cm)))
    val = cm + np.log(s) / tau
    pen = 1e6 * (np.sum(h) - n / 2.0) ** 2
    return val + pen

def soft_grad(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    e = np.exp(tau * (c - cm))
    w = e / e.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        if w[k_idx] < 1e-16:
            continue
        coeff = w[k_idx] * dx / tau
        if k >= 0:
            i_s, i_e = 0, n - k
            j_s, j_e = k, n
        else:
            i_s, i_e = -k, n
            j_s, j_e = 0, n + k
        if i_e > i_s:
            grad[i_s:i_e] += coeff * (1.0 - h[j_s:j_e])
        if k >= 0:
            i_s2, i_e2 = k, n
            j_s2, j_e2 = 0, n - k
        else:
            i_s2, i_e2 = 0, n + k
            j_s2, j_e2 = -k, n
        if i_e2 > i_s2:
            grad[i_s2:i_e2] -= coeff * h[j_s2:j_e2]
    grad += 2e6 * (np.sum(h) - n / 2.0)
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def _obj_fn(h, tau, n):
    return soft_obj(h, tau, n)

def _grad_fn(h, tau, n):
    return soft_grad(h, tau, n)

def coord_polish(h, max_rounds=80):
    n = len(h)
    h = h.copy()
    for rnd in range(max_rounds):
        improved = False
        c_full = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            vi = h[i]
            a = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                k = k_idx - (n - 1)
                da = 0.0
                j1 = i + k
                if 0 <= j1 < n:
                    da += (1.0 - h[j1])
                j2 = i - k
                if 0 <= j2 < n:
                    da -= h[j2]
                a[k_idx] = da
            b = c_full - vi * a
            def f1d(v):
                return np.max(a * v + b)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            h_new = res.x
            if abs(h_new - vi) > 1e-11:
                c_full = c_full + (h_new - vi) * a
                h[i] = h_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h
