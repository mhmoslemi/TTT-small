import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_fun(hv):
    return np.sum(hv) - len(hv) / 2.0

def sum_jac(hv):
    return np.ones(len(hv))

def make_soft_obj(temp):
    pass

def soft_obj(h, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = (c - m) / temp
    s = np.where(s < -100, -100.0, s)
    return m + temp * np.log(np.sum(np.exp(s)))

def solve_slsqp(h, n, temp, maxiter):
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(soft_obj, h.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

def subgrad_step(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = c.argmax()
    s = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jp = j + s
        jm = j - s
        if 0 <= jp < n:
            g += (1.0 - h[jp])
        if 0 <= jm < n:
            g -= h[jm]
        grad[j] = g
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

def pairwise_cd(h, n, max_rounds, t_limit, rng):
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for ii in range(n):
            for jj in range(ii + 1, n):
                if time.time() > t_limit:
                    return h, c5_cur
                j = order[ii]; jp = order[jj]
                cp = h[j] + h[jp]
                lo = max(0.0, cp - 1.0)
                hi = min(1.0, cp)
                if hi - lo < 1e-14:
                    continue
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                h[j] = c1; h[jp] = cp - c1
                f1 = compute_c5(h)
                h[j] = c2; h[jp] = cp - c2
                f2 = compute_c5(h)
                for _ in range(20):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b = c2; c2 = c1; f2 = f1
                        c1 = b - gr * (b - a)
                        h[j] = c1; h[jp] = cp - c1
                        f1 = compute_c5(h)
                    else:
                        a = c1; c1 = c2; f1 = f2
                        c2 = a + gr * (b - a)
                        h[j] = c2; h[jp] = cp - c2
                        f2 = compute_c5(h)
                t_opt = (a + b) / 2.0
                h[j] = t_opt; h[jp] = cp - t_opt
                h = project_feasible(h, n)

                    c5_cur = c5_new

            break

def nesterov_cd(h, n, t_limit, rng):

    alpha = 0.05
    beta = 0.9
    h_prev = h.copy()
    return_h = h.copy()

        k_star = c.argmax()
        s = k_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            grad[j] = g
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break
        momentum = beta * momentum + (1 - beta) * grad

            return_h = h_trial.copy()
            h = h_trial

            h = h_prev.copy()
        h_prev = h.copy()
