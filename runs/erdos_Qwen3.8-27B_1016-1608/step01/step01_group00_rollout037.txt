import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_raw(h):
    """Return raw correlation (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n, active_tol=0.02):
    """Compute subgradient of C5 at h."""
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corrs)
    active = np.where(corrs >= max_c - active_tol)[0]
    
    grad = np.zeros(n)
    for p in active:
        m = p - (n - 1)
        if m >= 0:
            # d corr_m / dh[i] = (1-h[i+m])*[0<=i<=n-1-m] - h[i-m]*[m<=i<=n-1]
            for i in range(n - m):
                grad[i] += (1.0 - h[i + m])
            for i in range(m, n):
                grad[i] -= h[i - m]
        else:
            m_abs = -m
            for i in range(n - m_abs):
                grad[i] -= h[i + m_abs]
            for i in range(m_abs, n):
                grad[i] += (1.0 - h[i - m_abs])
    
    return grad / len(active)

def coord_descent(h, n, max_rounds=10, grid_size=101):
    """Coordinate descent: for each i, find optimal h[i] given others fixed."""
    h = h.copy()
    xs = np.linspace(0.0, 1.0, grid_size)

            h[i] = 0.0

            h[i] = 1.0

            intercepts = corr0

            f_vals = np.max(vals, axis=0) * dx
            best_idx = np.argmin(f_vals)
            best_x = xs[best_idx]

            step = 1.0 / grid_size

            hi = min(1.0, best_x + step)

                return float(np.max(slopes * x + intercepts) * dx)
