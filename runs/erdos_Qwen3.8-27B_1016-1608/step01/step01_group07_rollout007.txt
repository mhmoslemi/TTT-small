    return float(np.max(c) * dx)

def _project(v, n):
    target = 0.5 * n
    lo, hi = v.min() - 2.0, v.max() + 2.0

        if np.clip(v - mid, 0, 1).sum() > target:

        else:

        k = int(m.sum())
        if k > 0:
