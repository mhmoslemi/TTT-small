    return h_cur, c5_cur

def pair_swap_polish(h, c5_cur, max_iters=5000, t_limit=None, t_start=None):
    n = len(h)
    h_cur = h.copy()
    alpha = 0.05
    for it in range(max_iters):
        if t_start is not None and time.time() > t_limit:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        h_trial = h_cur.copy()
        transfer = alpha * np.random.choice([-1, 1])
        h_trial[i] = np.clip(h_trial[i] + transfer, 0.0, 1.0)
        h_trial[j] = np.clip(h_trial[j] - transfer, 0.0, 1.0)
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < c5_cur - 1e-14:
            c5_cur = c5_t
            h_cur = h_trial
        else:
            alpha *= 0.995
            if alpha < 0.001:
                break
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Use initial_h_values if available
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project_feasible(h0, n0)
        c5_0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0
        best_n = n0
    except Exception:
        best_n = 47
        best_h = project_feasible(np.full(best_n, 0.5) + 0.1 * rng.randn(best_n), best_n)
        best_c5 = compute_c5(best_h)

    n_candidates = sorted(set([best_n, 32, 47, 64, 94, 128, 160]))
    n_candidates = [n for n in n_candidates if n <= 200]

    time_limit_main = t_start + budget_s * 0.80
    time_limit_final = t_start + budget_s * 0.96

    for n in n_candidates:
        if time.time() > time_limit_main:
            break
        is_best_n = (n == best_n)

        constraints = [{'type': 'eq', 'fun': sum_obj, 'jac': sum_jac}]
        bounds = [(0.0, 1.0)] * n

        # Generate diverse starts
        starts = []
        if is_best_n:
            starts.append(best_h.copy())
        for _ in range(2):
            starts.append(random_initial(n, rng))
        starts.append(binary_initial(n, rng))
        starts.append(block_initial(n, 4, rng))
        starts.append(block_initial(n, 6, rng))
        starts.append(sinusoidal_initial(n, 2, 0))
        starts.append(sinusoidal_initial(n, 3, np.pi / 3))

        for h_start in starts:
            if time.time() > time_limit_main:
                break
            h_cur = h_start.copy()
            c5_cur = compute_c5(h_cur)

            # Phase A: p-norm minimization (increasing p)
            for p in [4, 8, 16, 32, 64]:
                if time.time() > time_limit_main:
                    break
                h_new = pnrm_optimize(h_cur, n, p, constraints, bounds, 400)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new

            # Phase B: Soft-max SLSQP
            for temp in [0.01, 0.003, 0.001, 0.0003]:
                if time.time() > time_limit_main:
                    break
                h_new = slsqp_optimize(h_cur, n, temp, constraints, bounds, 800)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new

            # Phase C: Subgradient polish
            h_cur, c5_cur = subgrad_polish(h_cur, c5_cur, 2000, time_limit_main, t_start)

            # Phase D: Pair swap
            h_cur, c5_cur = pair_swap_polish(h_cur, c5_cur, 2000, time_limit_main, t_start)

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n

    # Final intensive polish on best solution
    if best_h is not None and time.time() < time_limit_final:
        n = best_n
        constraints = [{'type': 'eq', 'fun': sum_obj, 'jac': sum_jac}]
        bounds = [(0.0, 1.0)] * n

        for temp in [0.0001, 0.00003, 0.00001]:
            if time.time() > time_limit_final:
                break
            h_new = slsqp_optimize(best_h, n, temp, constraints, bounds, 3000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

        best_h, best_c5 = subgrad_polish(best_h, best_c5, 8000, time_limit_final, t_start)
        best_h, best_c5 = pair_swap_polish(best_h, best_c5, 3000, time_limit_final, t_start)

        # One more SLSQP pass
        if time.time() < t_start + budget_s * 0.98:
            h_new = slsqp_optimize(best_h, n, 0.00001, constraints, bounds, 3000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
