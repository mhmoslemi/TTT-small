    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if temp > 1e10:
        return m * dx
    scaled = (corr - m) * temp
    # For numerical stability
    if np.max(scaled) > 500:
        return m * dx
    logsum = np.log(np.sum(np.exp(scaled) + 1e-300))
    return (m + logsum / (len(corr) * temp)) * dx

def project_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                return h
            step = min(diff / mask.sum(), (1.0 - h[mask]).min())
            if step <= 0:
                return h
            h[mask] += step
        else:

            if not mask.any():
                return h

        for i in zeros[:target-n_ones]:

            if c5_b3 < best_c5:
                best_h, best_c5 = h_b3.copy(), c5_b3

    # Phase 6: L-BFGS-B with smooth objective from current best
    if deadline - time.time() > 60:
        for temp in [200.0, 500.0, 1000.0, 3000.0]:
            if deadline - time.time() < 15:
                break
            for attempt in range(2):
                if deadline - time.time() < 10:
                    break
                h0 = project_feasible(best_h + rng.randn(n) * 0.03, target)
                try:
                    res = minimize(obj_smooth, h0, args=(dx, temp, target),
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 3000, 'ftol': 1e-20,
                                           'gtol': 1e-15, 'maxls': 80, 'iprint': -1})
                    h_new = project_feasible(res.x, target)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5:
                        best_h, best_c5 = h_new, c5_new
                except Exception:
                    pass

    # Phase 7: Final coordinate descent + SLSQP polish
    if deadline - time.time() > 20:
        h_cd, c5_cd = coord_descent(best_h, dx, target, 300, deadline)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    if deadline - time.time() > 10:
        try:
            cons = [{'type': 'eq', 'fun': con_sum, 'args': (target,)}]
            res = minimize(obj_smooth, best_h.copy(), args=(dx, 5000.0, target),
                        method='SLSQP', bounds=bounds,
                        options={'maxiter': 600, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Final cleanup
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
