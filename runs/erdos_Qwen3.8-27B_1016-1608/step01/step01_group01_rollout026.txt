                        best_h = h_trial

    # Phase 7: Pair swap search
    if time.time() - t_start < budget_s * 0.99:
        for i in range(n - 1):
            if time.time() - t_start > budget_s * 0.99:
                break
            for j in range(i + 1, n):
                h_trial = best_h.copy()
                h_trial[i], h_trial[j] = best_h[j], best_h[i]
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = h_trial

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
