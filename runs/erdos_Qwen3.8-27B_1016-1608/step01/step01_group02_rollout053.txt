For k > 0, this is the number of "0→1" transitions at distance k (reading right to left).
For k < 0, this is the number of "1→0" transitions at distance k (reading left to right).

By symmetry (reversing the array), positive and negative lags are related. So we mostly care about |k|.

        one_idx = np.where(h_bin == 1)[0]
        zero_idx = np.where(h_bin == 0)[0]
        for i in one_idx:
            for j in zero_idx:
                h_bin[i], h_bin[j] = 0, 1
                c5_new = compute_c5(h_bin, dx)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 1, 0
            if improved:
                break
        if not improved:
            break
    return h_bin, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_h = projection(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': eq_constraint, 'args': (target,)},)

    def tl():
        return budget_s - (time.time() - t0)

    def try_update(h_new):
        nonlocal best_h, best_c5
        if h_new is None:
            return
        h_new = projection(h_new, target)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5

    # Phase 1: Quick SLSQP polish
    try:
        res = minimize(c5_obj, best_h, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 300, 'ftol': 1e-15})
        try_update(res.x)
    except Exception:
        pass

    # Phase 2: Binary exhaustive local search from multiple starts
    h_bin = (best_h > 0.5).astype(float)
    h_bin = fix_binary(h_bin, n, rng)
    h_bin, c5_bin = bin_local_search(h_bin, dx, n, max_rounds=300)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    for rs in range(30):
        if tl() < 30:
            break
        h_r = rng.randint(0, 2, n).astype(float)
        h_r = fix_binary(h_r, n, rng)
        h_r, c5_r = bin_local_search(h_r, dx, n, max_rounds=100)
        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r

    # Also try structured starts
    for pat in range(10):
        if tl() < 20:
            break
        h_s = np.zeros(n)
        period = rng.randint(5, 25)
        duty = rng.randint(period // 3 + 1, period - period // 3)
        for i in range(n):
            phase = (i + rng.randint(period)) % period
            h_s[i] = 1.0 if phase < duty else 0.0
        h_s = fix_binary(h_s, n, rng)
        h_s, c5_s = bin_local_search(h_s, dx, n, max_rounds=100)
        if c5_s < best_c5:
            best_h = h_s.copy()
            best_c5 = c5_s

    # Phase 3: Simulated annealing (binary swaps + continuous transfers)
    if tl() > 200:
        h_sa = best_h.copy()
        c5_sa = best_c5
        best_sa = h_sa.copy()
        best_sa_c5 = c5_sa
        T = 0.03
        sa_mode = 'binary'

        for sa_iter in range(500000):
            if tl() < 15:
                break
            if sa_iter % 5000 == 2500:
                sa_mode = 'cont' if sa_mode == 'binary' else 'binary'

            if sa_mode == 'binary':
                one_idx = np.where(h_sa > 0.5)[0]
                zero_idx = np.where(h_sa < 0.5)[0]
                if len(one_idx) == 0 or len(zero_idx) == 0:
                    continue
                i = one_idx[rng.randint(len(one_idx))]
                j = zero_idx[rng.randint(len(zero_idx))]
                h_t = h_sa.copy()
                h_t[i] = 0.0
                h_t[j] = 1.0
                c5_new = compute_c5(h_t, dx)
                d = c5_new - c5_sa
                if d < 0 or rng.random() < np.exp(-d / T):
                    h_sa, c5_sa = h_t, c5_new
                    if c5_new < best_sa_c5:
                        best_sa_c5, best_sa = c5_new, h_sa.copy()
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                delta = rng.uniform(-0.3, 0.3)
                h_t = h_sa.copy()
                h_t[i] += delta
                h_t[j] -= delta
                if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                    h_t = projection(h_t, target)
                    c5_new = compute_c5(h_t, dx)
                    d = c5_new - c5_sa
                    if d < 0 or rng.random() < np.exp(-d / T):
                        h_sa, c5_sa = h_t, c5_new
                        if c5_new < best_sa_c5:
                            best_sa_c5, best_sa = c5_new, h_sa.copy()

            if sa_iter % 3000 == 0:
                T = max(T * 0.985, 0.0005)

        try_update(best_sa)

    # Phase 4: Subgradient on top lags + SLSQP
    if tl() > 60:
        h_tg = best_h.copy()
        alpha = 0.05
        for outer in range(500):
            if tl() < 10:
                break
            corr = np.correlate(h_tg, 1.0 - h_tg, mode="full") * dx
            top_k = np.argsort(corr)[-3:]
            grad = np.zeros(n)
            for p in top_k:
                k = int(p) - (n - 1)
                if k >= 0:
                    for i in range(k, n):
                        grad[i] += (1.0 - h_tg[i - k]) * dx
                        grad[i - k] += (-h_tg[i]) * dx
                else:
                    m = -k
                    for i in range(0, n - m):
                        grad[i] += (1.0 - h_tg[i + m]) * dx
                        grad[i + m] += (-h_tg[i]) * dx
            h_new = projection(h_tg - alpha * grad, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < compute_c5(h_tg, dx) - 1e-15:
                h_tg = h_new
            else:
                alpha *= 0.7
            try_update(h_tg)
            if outer % 50 == 49:
                alpha = 0.05

        try_update(h_tg)

    # Phase 5: Pair-coordinate continuous local search
    if tl() > 60:
        h_pc = best_h.copy()
        c5_pc = best_c5
        step = 0.08
        stuck = 0
        for it in range(300000):
            if tl() < 10:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            moved = False
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                h_t = h_pc.copy()
                h_t[i] += delta
                h_t[j] -= delta
                if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1 and abs(h_t.sum() - target) < 1e-10:
                    c5_t = compute_c5(h_t, dx)
                    if c5_t < c5_pc - 1e-15:
                        h_pc, c5_pc = h_t, c5_t
                        stuck = 0
                        moved = True
                        try_update(h_pc)
                        break
            if not moved:
                stuck += 1
            if stuck > 2000:
                stuck = 0
                step = max(0.003, step * 0.6)
                h_pc = projection(h_pc + rng.randn(n) * 0.03, target)
                c5_pc = compute_c5(h_pc, dx)

    # Phase 6: Final SLSQP polish
    if tl() > 15:
        try:
            res = minimize(c5_obj, best_h, args=(dx,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-16})
            try_update(res.x)
        except Exception:
            pass

    # Phase 7: One more binary search + continuous refinement
    if tl() > 20:
        h_b2 = (best_h > 0.5).astype(float)
        h_b2 = fix_binary(h_b2, n, rng)
        h_b2, c5_b2 = bin_local_search(h_b2, dx, n, max_rounds=300)
        if c5_b2 < best_c5:
            best_h = h_b2.copy()
            best_c5 = c5_b2
        try:
            res = minimize(c5_obj, best_h, args=(dx,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16})
            try_update(res.x)
        except Exception:
            pass

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
