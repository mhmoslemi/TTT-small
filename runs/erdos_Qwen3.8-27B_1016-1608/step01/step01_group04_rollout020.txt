from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_h(n, seed):
    idx = np.random.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def to_binary(h):
    """Convert to nearest binary with correct sum"""
    n = len(h)
    target = n // 2
    order = np.argsort(-h)  # largest values first
    idx = order[:target]
    h_b = np.zeros(n)
    h_b[idx] = 1.0
    return h_b

# Global state for SLSQP callbacks
_gs = {}

def _smax_obj(x):
    n = _gs['n']
    eps = _gs['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-12:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _eq_fun(x):
    return float(np.sum(x) - _gs['n'] / 2.0)

def _eq_jac(x):
    return np.ones(_gs['n'])

def slsqp_opt(h, n, eps, maxiter, ftol):
    _gs['n'] = n
    _gs['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cst = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    r = minimize(_smax_obj, h, method='SLSQP', bounds=bds,
                 constraints=cst, options={'maxiter': maxiter, 'ftol': ftol})
    return repair_sum(r.x, n)

def local_search_binary(h, n, max_rounds, rng):

    h = to_binary(h)
    c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        best_c5 = c5
        best_i, best_j = None, None
        # Sample pairs rather than trying all (for speed)
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        n_samples = min(n_ones * n_zeros, 2000)
        for _ in range(n_samples):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            # Swap: h[i] becomes 0, h[j] becomes 1
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_i, best_j = i, j
        if best_i is not None:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5 = best_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            improved = True
    return h, c5

def sim_anneal_binary(h, n, t_start, t_end, n_iter, rng):

    h = to_binary(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    temp = t_start
    cooling = (t_end / t_start) ** (1.0 / max(1, n_iter))
    for it in range(n_iter):
        temp *= cooling
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[j] = 1.0

        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_gs = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h):
    n = len(h)
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:n // 2]] = 1.0
    return h_b

def _smax_obj(x):
    n = _gs['n']
    eps = _gs['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-12:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _eq_fun(x):
    return float(np.sum(x) - _gs['n'] / 2.0)

def _eq_jac(x):
    return np.ones(_gs['n'])

def slsqp_opt(h, n, eps, maxiter, ftol):
    _gs['n'] = n
    _gs['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cst = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    r = minimize(_smax_obj, h, method='SLSQP', bounds=bds,
                 constraints=cst, options={'maxiter': maxiter, 'ftol': ftol})
    return repair_sum(r.x, n)

def bin_ls(h, n, rounds, rng, n_samp):
    h = to_binary(h).copy()
    c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(rounds):
        best_c5 = c5
        best_pair = None
        ns = min(len(ones) * len(zeros), n_samp)
        for _ in range(ns):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            nc = compute_c5(h)
            h[i], h[j] = 1.0, 0.0
            if nc < best_c5 - 1e-15:
                best_c5 = nc
                best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
        c5 = best_c5
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return h, c5

def ils_binary(h, n, n_rounds, rng, ls_rounds, ls_samp, perturb_size):
    """Iterated local search on binary h"""
    h = to_binary(h).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
