import numpy as np
import time
from scipy.optimize import minimize

_g_tau = 50.0
_g_n = 50

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_obj(h):
    c = compute_lags(h)
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(_g_tau * (c - cm)))) / _g_tau

def eq_constr(h):
    return np.sum(h) - _g_n / 2.0

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h = np.clip(h + diff / c * m, 0.0, 1.0)
    return h

def grad_lag(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        q = -k
        g[:n - q] += (1.0 - h[q:])
        g[q:] -= h[:n - q]
    return g

def subgrad_descent(h, n, t0, t_end, n_iter=8000):

    for it in range(n_iter):
        if time.time() - t0 > t_end:
            break

        cm = np.max(c)
        active = np.where(c >= cm - 1e-5)[0]
        g = np.zeros(n)
        for m in active:
            g += grad_lag(h, m - (n - 1), n)
        g /= len(active)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        h_new = h - step * g
        h_new = project(h_new, n)
        s_new = compute_score(h_new)
        if s_new < best - 1e-13:
            best = s_new
            best_h = h_new.copy()
            h = h_new
            stale = 0
            step = min(step * 1.02, 1.5 / n)
        else:
            h = h_new
            stale += 1
            if stale > 80:
                step *= 0.75
                stale = 0
        if step < 1e-6 / n:
            break
    return best_h, best

def coord_desc(h, n, t0, t_end):
    h = h.copy()
    best = compute_score(h)
    best_h = h.copy()

    while time.time() - t0 < t_end:
        improved = False
        for j in range(n):
            if time.time() - t0 > t_end:
                break
            for d in deltas:
                h2 = h.copy()
                h2[j] = np.clip(h2[j] + d, 0, 1)
                h2 = project(h2, n)
                s = compute_score(h2)
                if s < best - 1e-13:
                    best = s
                    best_h = h2.copy()
                    h = h2
                    improved = True
                    break
        if not improved:
            break
    return best_h, best

def pair_swap(h, n, rng, t0, t_end):
    h = h.copy()
    best = compute_score(h)
