    # Find optimal δ to minimize max_k corr_k(δ)

    b_k = np.zeros(2*n-1)

    # corr[k] = sum_{j=0}^{n-1-lag} h[j]*(1-h[j+lag])
    # d/dh[p]: if p in [0, n-1-lag]: (1-h[p+lag]); if p in [lag, n-1]: -h[p-lag]

    # corr[k] = sum_{j=0}^{n-1-lag2} h[j+lag2]*(1-h[j])
    # d/dh[p]: if p in [lag2, n-1]: (1-h[p-lag2]); if p in [0, n-1-lag2]: -h[p+lag2]

            if 0 <= i <= j_max:

            if lag <= i <= n-1:
                a_k[k] += -h[i - lag]
            if 0 <= j <= j_max:

            if lag <= j <= n-1:
                b_k[k] += -h[j - lag]

def project_fix(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(excess / cnt, h[mask].min() * 0.999)
            h[mask] -= s
        else:

            cnt = mask.sum()
            if cnt == 0:

            s = min(-excess / cnt, (1.0 - h[mask]).min() * 0.999)
            h[mask] += s
