import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:

        h += d / n

def cd_sweep(h, n, order):
    c_cur = np.correlate(h, 1.0 - h, mode='full')
    n1 = n - 1
    ks = np.arange(-n1, n1 + 1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for i in order:
        idx1 = i + ks
        idx2 = i - ks
        v1 = ((idx1 >= 0) & (idx1 < n)).astype(float)
        v2 = ((idx2 >= 0) & (idx2 < n)).astype(float)
        h_ipk = np.where(v1 > 0, h[np.clip(idx1, 0, n - 1)], 0.0)
        h_imk = np.where(v2 > 0, h[np.clip(idx2, 0, n - 1)], 0.0)
        a = (1.0 - h_ipk) * v1 - h_imk * v2
        b = c_cur - h[i] * a
        ts = np.linspace(0.0, 1.0, 128)
        vals = np.max(b + ts[:, None] * a[None, :], axis=1)
        im = int(np.argmin(vals))
        lo = ts[max(0, im - 1)]
        hi = ts[min(127, im + 1)]
        if hi - lo < 1e-8:
            t_star = (lo + hi) / 2.0
        else:
            a_gs, b_gs = lo, hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            f_c = np.max(b + c_gs * a)
            f_d = np.max(b + d_gs * a)
            for _ in range(60):
