import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-15:
            break
        h += (target - s) / n
    return h

def smax_c5(h, eps):
    c = c5_vals(h)
    m = np.max(c)
    s = np.clip((c - m) / eps, -100.0, 100.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def smax_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = np.max(c)
    s = np.clip((c - m) / eps, -100.0, 100.0)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - n + 1
        lo1 = max(0, -k)
        hi1 = min(n - 1, n - 1 - k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += dx * w[mi] * (1.0 - h[lo1 + k:hi1 + k + 1])
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= dx * w[mi] * h[lo2 - k:hi2 - k + 1]
    return grad

def pgd(h, n, eps, max_iter=800, lr0=0.01):
    h = h.copy()
    f = smax_c5(h, eps)
    lr = lr0
    for _ in range(max_iter):
        g = smax_grad(h, eps)
        gnorm = np.sqrt(np.dot(g, g))
        if gnorm < 1e-14:
            break
        for _ in range(25):
            h_t = repair(h - lr * g, n)
            f_t = smax_c5(h_t, eps)
            if f_t < f - 1e-6 * lr * gnorm:
                break
            lr *= 0.4
        else:
            break
        h = h_t
        f = f_t
        lr = min(lr * 1.3, 0.08)
    return h

def pair_search(h, n, budget_s=60.0, seed=0):
    rng = np.random.default_rng(seed)
    h = h.copy()
    bc = compute_c5(h)
    t_start = time.time()
    deltas = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    it = 0
    while time.time() - t_start < budget_s:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        for d in deltas:
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c = compute_c5(h)
                if c < bc:
                    bc = c
                    break
                h[i] -= d
                h[j] += d
        for d in deltas:
            if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                h[j] += d
                h[i] -= d
                c = compute_c5(h)
                if c < bc:
                    bc = c
                    break
                h[j] -= d
                h[i] += d
        it += 1
    return h, bc

def try_slsqp(h, n, eps, maxiter=200):
    n_save = n
    e_save = eps
    def obj(x):
        return smax_c5(x, e_save)
    def obj_grad(x):
        return smax_grad(x, e_save)
    def con_fun(x):
        return np.sum(x) - n_save / 2.0
    def con_jac(x):
        return np.ones(n_save)
    bds = [(0.0, 1.0)] * n_save
    r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                 bounds=bds, constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                 options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair(r.x, n)
