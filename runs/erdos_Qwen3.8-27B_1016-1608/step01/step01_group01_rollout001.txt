import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            if mask.any():
                h[mask] += diff2 / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, temp):
    c = corr_vec(h)
    m = c.max()
    c_clip = np.clip((c - m) / temp, -60, 60)
    return m + temp * np.log(np.sum(np.exp(c_clip)))

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def sum_constraint_jac(h):
    return np.ones(len(h))
