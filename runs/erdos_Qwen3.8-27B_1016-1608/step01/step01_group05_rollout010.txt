import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _corr_array(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def _logsumexp_max(corr, beta):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _push_binary(h, eps):
    """Push values toward 0 or 1, keeping integral."""
    out = h.copy()
    for i in range(len(out)):
        if out[i] < 0.5:
            out[i] = max(0.0, min(out[i], eps))
        else:
            out[i] = min(1.0, max(out[i], 1.0 - eps))
    return _project(out, np.sum(h))

def _powell_smooth(h0, n, target, beta, maxiter, xtol=1e-9):
    def obj(x):
        h = _project(x, target)
        cv = _corr_array(h, n)
        return _logsumexp_max(cv, beta)
    try:
        res = minimize(obj, h0, method='Powell',
                       options={'maxiter': maxiter, 'xtol': xtol, 'ftol': 1e-11})
        return _project(res.x, target)
    except Exception:
        return h0

def _nm_direct(h0, n, target, maxiter=4000):
    def obj(x):
        h = _project(x, target)
        return _compute_c5(h, n)
    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9,
                                'fatol': 1e-11, 'initial_simplex_size': 0.05})
        return _project(res.x, target)
    except Exception:
        return h0

def _coord_descent(h, n, target, rng, n_sweeps=20, small_steps=True):
    h = h.copy()
    target = float(np.sum(h))
    best = _compute_c5(h, n)
    steps = [0.005, 0.01, 0.025, 0.05, 0.1, 0.2, 0.4] if small_steps else [0.05, 0.15, 0.3]
    for _ in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            cands = set()
            cands.add(0.0)
            cands.add(1.0)
            for s in steps:
                c = old - s
                if c >= 0.0:
                    cands.add(round(c, 5))
                c = old + s
                if c <= 1.0:
                    cands.add(round(c, 5))
            bv, bc = best, old
            for c in sorted(cands):
                if abs(c - old) < 1e-7:
                    continue
                h[j] = c
                hp = _project(h, target)
                v = _compute_c5(hp, n)
                if v < bv:
                    bv, bc = v, c
            if bc != old:
                h[j] = bc
                h = _project(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = _project(h, target)
        if not improved:
            break
    return _project(h, target), best

def _upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def _make_binary_muller(n, rng):
    """Generate binary via modular arithmetic for good autocorrelation."""
    h = np.zeros(n)
    used = set()
    count = 0
    attempts = 0
    while count < n // 2 and attempts < n * 50:
        idx = (rng.randint(0, n) * 7 + rng.randint(0, n)) % n

            h[idx] = 1.0
            used.add(idx)
            count += 1
        attempts += 1
    if count < n // 2:
        for i in range(n):
            if h[i] == 0 and count < n // 2:
                h[i] = 1.0
                count += 1
    return h
