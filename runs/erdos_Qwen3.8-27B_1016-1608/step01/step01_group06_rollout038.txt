    # out: current correlation values (all lags)
    # d: direction vector for this coordinate

    lo = -h[j]
    hi = 1.0 - h[j]

def project_feasible(h, n, iters=300):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(iters):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h
