np.correlate(a, v, mode="full") computes the full cross-correlation. For arrays of length n, the output has length 2n-1. The value at index 2(n-1) + k (for k in -(n-1), ..., (n-1)) is:
sum_{i} a[i+k] * v[i] (for valid i)

So np.correlate(h, 1-h, mode="full") at index n-1+k gives:
sum_i h[i+k] * (1-h[i])

This is the number of positions where h has a 1 at i+k and 0 at i. For binary h, this counts transitions from 0 to 1 at distance k.

The max over all k of this sum, times dx, gives C₅.

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def binary_project(h, n):
    idx = np.argsort(-h)[:n//2]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def upsample_linear(h, n_new):
    x_old = np.arange(len(h))
    x_new = np.linspace(0, len(h)-1, n_new)
    return np.interp(x_new, x_old, h)

def swap_descend(h, n, time_limit=30.0):
    best = compute_c5(h, n)
    h = h.copy()
    t0 = time.time()

            best_v = old
            cands = [0.0, 1.0, 0.5, 0.25, 0.75]
