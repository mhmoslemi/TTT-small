        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    n_results.sort(key=lambda r: r[2])

    # Random restarts on top-2 n values
    for ri in range(min(2, len(n_results))):
        n_r = n_results[ri][0]
        for r in range(4):
            if time.time() - t0 > budget_s * 0.88:
                break
            scale = 0.06 * (r + 1)
            base = best_h if n_r == len(best_h) else interp_h(best_h, n_r)
            h_rand = base + rng.normal(0, scale, n_r)
            h_rand = project_feasible(h_rand, n_r)
            h_rand, c5_r = full_optimize(h_rand, n_r, t0, budget_s, 0.88, do_slp=True, n_slp=30)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_rand.copy()

    # Final polish
    n_f = len(best_h)
    if time.time() - t0 < budget_s * 0.94:
        best_h, best_c5 = slsqp_polish(best_h, n_f, t0 + budget_s * 0.94,
                                        [0.0001, 0.00003, 0.00001])
    if time.time() - t0 < budget_s * 0.94:
        best_h, _ = slp_loop(best_h, n_f, t0 + budget_s * 0.94, max_iter=25)
        c5_chk = compute_c5(best_h)
        best_c5 = min(best_c5, c5_chk)
    if time.time() - t0 < budget_s * 0.94:
        best_h, best_c5 = slsqp_polish(best_h, n_f, t0 + budget_s * 0.94,
                                        [0.0001, 0.00002])

    # Ultra-fine coordinate descent
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n_f, t0 + budget_s * 0.99,
            [0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002],
            rounds=3, tol=1e-14)
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n_f, t0 + budget_s * 0.99,
            [0.0001, -0.0001, 0.00005, -0.00005],
            rounds=2, tol=1e-14)

    best_h = project_feasible(best_h, n_f)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_f)
