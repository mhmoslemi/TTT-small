        else:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t_star = int(np.argmax(corr))
    # gradient: grad[i] = (1 - 2*h[t_star - i]) if 0 <= t_star - i < n else 0
    grad = np.zeros(n)
    # t_star - i in [0, n-1] means i in [t_star-(n-1), t_star]
    i_lo = max(0, t_star - (n-1))
    i_hi = min(n-1, t_star)
    idx = np.arange(i_lo, i_hi+1)
    src = t_star - idx
    grad[idx] = 1.0 - 2.0 * h[src]
    grad *= dx
    h_new = h - alpha * grad
    return h_new

def make_block_h(n, rng):

        remaining -= block_len
        pos += block_len + rng.randint(1, max(1, n//4))
        if pos >= n:

def subgradient_optimize(h, n, n_iters=1000, alpha_start=0.3, alpha_end=0.01):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(n_iters):
        t_frac = it / max(1, n_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** t_frac
        corr = np.correlate(h, 1.0 - h, mode="full")
        t_star = int(np.argmax(corr))
        dx = 2.0 / n
        grad = np.zeros(n)
        i_lo = max(0, t_star - (n-1))
        i_hi = min(n-1, t_star)
        if i_hi >= i_lo:
            idx = np.arange(i_lo, i_hi + 1)
            src = t_star - idx
            grad[idx] = (1.0 - 2.0 * h[src]) * dx
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_to_set(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def coordinate_descent(h, n, n_sweeps=5, rng=None):
    target = n / 2.0
    h = h.copy()
    best_c = compute_c5(h)
    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng is not None else np.arange(n)
        improved = False
        for i in order:
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = old
            slope = c1 - c0
            intercept = c0
            # Ternary search for min of max_t(intercept[t] + x*slope[t])
            lo, hi = 0.0, 1.0
            for _ in range(25):
                m1 = lo + (hi - lo) * 0.333
                m2 = hi - (hi - lo) * 0.333
                f1 = np.max(intercept + m1 * slope)
                f2 = np.max(intercept + m2 * slope)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_opt = 0.5 * (lo + hi)
            if abs(x_opt - old) > 1e-6:
                h[i] = x_opt
                h = project_to_set(h, target)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    improved = True
        if not improved:
            break
    return h, best_c

def anneal(h, n, n_iters=5000, T_start=0.01, T_end=0.001, rng=None):
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    T = T_start
    for it in range(n_iters):
        frac = it / max(1, n_iters)
        T = T_start * (T_end / T_start) ** frac
        i = rng.randint(n)
        delta = rng.uniform(-0.1, 0.1)
        h[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h = project_to_set(h, target)
        c_new = compute_c5(h)
        if c_new < best_c or rng.random() < np.exp(-(c_new - best_c) / max(T, 1e-10)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            pass  # h is already updated, but we accept the move per SA
    return best_h, best_c

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):

        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
