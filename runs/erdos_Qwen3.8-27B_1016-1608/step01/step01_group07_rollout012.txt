import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = (h < 1.0).astype(float)
        else:
            mask = (h > 0.0).astype(float)
        k = mask.sum()
        if k == 0:
            break
        h = np.clip(h + d / k * mask, 0.0, 1.0)
    return h

def _sf(x):
    return np.sum(x) - 0.5 * len(x)

def _sj(x):
    return np.ones(len(x))

def _pa(h, a):
    ai, lam = a
    v = _corr(h)
    m = np.max(v)
    e = np.exp(np.clip(v[ai] - m, -50.0, 0.0))
    return float(m + np.log(e.sum()) / lam)

def _oa(h, a):
    lam = a
    v = _corr(h)
    m = np.max(v)
    e = np.exp(np.clip(v - m, -50.0, 0.0))
    return float(m + np.log(np.sum(e)) / lam)

def _slq(h0, a, maxit, t0, bud):
    if bud - (time.time() - t0) < 10:
        return None
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj}]
    try:
        r = minimize(_pa, h0, a, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
        if r.x is not None:
            return _fix(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _slq_all(h0, lam, maxit, t0, bud):
    if bud - (time.time() - t0) < 10:
        return None
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj}]
    try:
        r = minimize(_oa, h0, lam, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
        if r.x is not None:
            return _fix(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _act(h, top):
    v = _corr(h)
    return np.argsort(v)[::-1][:top]

def _sin_pert(rng, n, amp):
    f = rng.randint(1, 6)
    ph = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    return amp * np.sin(2.0 * np.pi * f * x / n + ph)

def _coord_polish(h, n, t0, bud, rounds, rng):
    bh = h.copy()
    bc = _c5(bh)
    deltas = [0.002, -0.002, 0.0008, -0.0008, 0.005, -0.005]
    for rd in range(rounds):
        if bud - (time.time() - t0) < 3:
            break
        idx = rng.permutation(n)
        for j in idx[:n]:
            if bud - (time.time() - t0) < 1:
                break
            for d in deltas:
                ht = bh.copy()
                ht[j] = np.clip(ht[j] + d, 0.0, 1.0)
                ht = _fix(ht, n)
                ct = _c5(ht)
                if ct < bc - 1e-8:
                    bh, bc = ht, ct
                    break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.full(45, 0.5)
    n = len(bh)
    bh = _fix(bh, n)
    bc = _c5(bh)

    # Phase 1: Moderate-lam active-set SLSQP, multiple restarts
    for lam in [40.0, 120.0, 300.0]:
        if budget_s - (time.time() - t0) < 70:
            break
        for att in range(3):
            if budget_s - (time.time() - t0) < 18:
                break
            hs = bh.copy()
            if att > 0:
                amp = 0.012 if att == 1 else 0.025
                hs = _fix(np.clip(bh + _sin_pert(rng, n, amp) + rng.randn(n) * amp * 0.3, 0.0, 1.0), n)
            ai = _act(hs, 14)
            ht = _slq(hs, (ai, lam), 5000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct

    # Phase 2: High-lam for sharper max approximation
    for lam in [600.0, 1500.0]:
        if budget_s - (time.time() - t0) < 50:
            break
        for att in range(2):
            if budget_s - (time.time() - t0) < 15:
                break
            hs = bh.copy()
            if att > 0:
                hs = _fix(np.clip(bh + rng.randn(n) * 0.005, 0.0, 1.0), n)
            ai = _act(hs, 8)
            ht = _slq(hs, (ai, lam), 6000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct

    # Phase 3: Fine coordinate polish
    if budget_s - (time.time() - t0) > 40:
        crng = np.random.RandomState(seed + 7)
        bh, bc = _coord_polish(bh, n, t0, budget_s, 12, crng)

    # Phase 4: Re-identify active set, another SLSQP round
    if budget_s - (time.time() - t0) > 50:
        for lam in [200.0, 500.0]:
            if budget_s - (time.time() - t0) < 20:
                break
            ai = _act(bh, 18)
            ht = _slq(bh, (ai, lam), 6000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct
                    ai = _act(bh, 18)

    # Phase 5: Second coordinate polish
    if budget_s - (time.time() - t0) > 25:
        crng2 = np.random.RandomState(seed + 13)
        bh, bc = _coord_polish(bh, n, t0, budget_s, 8, crng2)

    # Phase 6: Final all-lag SLSQP with very high lam
    if budget_s - (time.time() - t0) > 30:
        ht = _slq_all(bh, 3000.0, 4000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Phase 7: Tiny final tweaks
    if budget_s - (time.time() - t0) > 10:
        crng3 = np.random.RandomState(seed + 23)
        bh, bc = _coord_polish(bh, n, t0, budget_s, 4, crng3)

    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
