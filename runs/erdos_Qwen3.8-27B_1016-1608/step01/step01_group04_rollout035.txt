    return float(np.sum(x) - _opt['n'] / 2.0)

    return np.ones(_opt['n'])

def obf(x):
    return smax(x, _opt['n'], _opt['eps'])

def do_slsqp(h, n, eps, mi, ft):
    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    r = minimize(obf, h, method='SLSQP', bounds=bds, constraints=cs,
                 options={'maxiter': mi, 'ftol': ft})
    return repair(r.x, n)

def coord_step(h, n, i, ngrid=400):
    old = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    slopes = c1 - c0
    v_grid = np.linspace(0.0, 1.0, ngrid)
    c_mat = c0[np.newaxis, :] + np.outer(v_grid, slopes)
    c_mat[:, n-1] = c0[n-1] + v_grid * (1.0 - v_grid)
    max_c = np.max(c_mat, axis=1)
    idx = np.argmin(max_c)
    return v_grid[idx]

def coord_descent(h, n, max_sweeps, tol=1e-12, ngrid=400):
    h = h.copy()
    bc = compute_c5(h)
    stall = 0
    for sweep in range(max_sweeps):
        for i in range(n):
            new_v = coord_step(h, n, i, ngrid)
            h[i] = new_v
            # Quick repair
            d = n/2.0 - np.sum(h)
            if abs(d) > 1e-10:
                h += d / n
                h = np.clip(h, 0.0, 1.0)
        h = repair(h, n)
        c = compute_c5(h)
        if c < bc - tol:
            bc = c
            stall = 0
        else:
            stall += 1
            if stall > 3:
                break
    return h, bc

def pair_step(h, n, i, j, ngrid=200):
    """Try to optimize h[i] and h[j] simultaneously."""
    # For a pair, we need to maintain sum constraint
    # h[i] + h[j] = h[i] + h[j] (original) → h[i]+h[j] = s
    # Parameterize: h[i] = t, h[j] = s - t
    s = h[i] + h[j]
    # We need 0 <= t <= 1 and 0 <= s-t <= 1
    # So t ∈ [max(0, s-1), min(1, s)]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-10:
        return h[i]
    
    t_grid = np.linspace(t_lo, t_hi, ngrid)
    best_t = h[i]
    best_c = np.inf
    for t in t_grid[::4]:  # Coarse pass
        h[i] = t
        h[j] = s - t
        c0 = np.correlate(h, 1.0-h, mode='full')
        best_c_impl = np.max(c0)
        if best_c_impl < best_c:
            best_c = best_c_impl
            best_t = t
    # Fine pass around best
    if best_t > t_lo and best_t < t_hi:
        fine = np.linspace(max(t_lo, best_t-0.05), min(t_hi, best_t+0.05), ngrid)
        for t in fine:
            h[i] = t
            h[j] = s - t
            c0 = np.correlate(h, 1.0-h, mode='full')
            best_c_impl = np.max(c0)
            if best_c_impl < best_c:
                best_c = best_c_impl
                best_t = t
    h[i] = best_t
    h[j] = s - best_t
    return best_t

import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-12:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def slsqp_obj(x):
    return smax_obj(x, _opt['n'], _opt['eps'])

def slsqp_eq(x):
    return float(np.sum(x) - _opt['n'] / 2.0)

def slsqp_eq_jac(x):
    return np.ones(_opt['n'])

    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}]
    res = minimize(slsqp_obj, h, method='SLSQP', bounds=bds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return repair(res.x, n)

            stall = 0
        else:

                break

Actually, I want to double-check my coordinate descent formula. Let me verify:

h has n elements. correlate(h, 1-h, mode='full') has 2n-1 elements.
Element m (0-indexed) = sum over valid j: h[j] * (1-h)[j + m - (n-1)]
= sum_j h[j] * (1 - h[j + m - n + 1])

For a fixed i, which elements of the correlation involve h[i]?
- As the first factor: h[i] * (1 - h[i + m - n + 1]) for valid i, i+m-n+1
  → m ranges over [n-1-i, n-1-i+n-1] = [n-1-i, 2n-2-i]
  → m-i+n-1 = i+m-n+1, so the second index is i+m-n+1

- As the second factor: h[j] * (1 - h[i]) where j+m-n+1 = i → j = i+m-n+1
  → m ranges over [n-1-i, 2n-2-i] (same range, just re-indexed)
  → The first index j = i+m-n+1

So for correlation element m:
- If m is in [n-1-i, 2n-2-i]: h[i] appears
- Specifically:
  - Term 1 (h[i] as first factor): h[i] * (1-h[i+m-n+1]) (valid if 0 ≤ i+m-n+1 < n)
  - Term 2 (h[i] as second factor): h[i+m-n+1] * (1-h[i]) (valid if 0 ≤ i+m-n+1 < n)

When I set h[i] = v:
- Term 1 becomes: v * (1 - h[i+m-n+1])
- Term 2 becomes: h[i+m-n+1] * (1-v)

import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x):
    dx = 2.0 / _opt['n']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if _opt['eps'] < 1e-12:
        return float(m)
    s = np.clip((c - m) / _opt['eps'], -50.0, 50.0)
    return float(m + _opt['eps'] * np.log(np.sum(np.exp(s))))

def slsqp_eq(x):
    return float(np.sum(x) - _opt['n'] / 2.0)

def slsqp_eq_jac(x):
    return np.ones(_opt['n'])

def run_slsqp(h, n, eps, maxiter, ftol):
    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}]
    res = minimize(slsqp_obj, h, method='SLSQP', bounds=bds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return repair(res.x, n)

def run_coord_descent(h, n, max_sweeps, ngrid, tol=1e-11):
    h = h.copy()
    bc = compute_c5(h)
    stall = 0
    for sweep in range(max_sweeps):
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = old
            slopes = c1 - c0
            v = np.linspace(0.0, 1.0, ngrid)
            c_mat = c0[np.newaxis, :] + np.outer(v, slopes)
            c_mat[:, n - 1] = c0[n - 1] + v * (1.0 - v)
            cmax = np.max(c_mat, axis=1)
            best_i = int(np.argmin(cmax))
            new_v = v[best_i]
            if abs(new_v - old) > 1e-12:
                h[i] = new_v
        h = repair(h, n)
        c = compute_c5(h)
        if c < bc - tol:
            bc = c
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                break
    return h, bc

def refine_coord(h, n, i, v_lo, v_hi, ngrid=200):
    """Refine coordinate i in [v_lo, v_hi] with fine grid."""
    old = h[i]
    h[i] = 0.0
