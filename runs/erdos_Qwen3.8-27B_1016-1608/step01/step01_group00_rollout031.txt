from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 - 1e-14 if diff > 0 else h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(p * (corr - m)))) / p

def numerical_grad(h, p, eps=1e-6):
    n = len(h)
    f0 = smooth_obj(h, p)
    g = np.zeros(n)
    for j in range(n):
        hp = h.copy()
        hp[j] = min(1.0, h[j] + eps)
        hm = h.copy()
        hm[j] = max(0.0, h[j] - eps)
        g[j] = (smooth_obj(hp, p) - smooth_obj(hm, p)) / (hp[j] - hm[j])
    return g

def mlsqp(h0, n, maxiter=500, ftol=1e-15):
    bnds = [(0.0, 1.0)] * n
    def sum_con(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < compute_c5(h0) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

    """One step of projected gradient descent on smooth max."""
    g = numerical_grad(h, p)
    h_new = h - lr * g
    return repair(h_new, len(h))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 - 1e-14 if diff > 0 else h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, maxiter=500, ftol=1e-15):
    bnds = [(0.0, 1.0)] * n
    def eq_con(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < compute_c5(h0):
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def slsqp_smooth(h0, n, maxiter=500, p=100):
    bnds = [(0.0, 1.0)] * n
    def eq_con(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_con}]
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        sh = p * (corr - m)
        ms = np.max(sh)
        return m + (ms + np.log(np.sum(np.exp(sh - ms)))) / p
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_new) < compute_c5(h0):
            return h_new
    except Exception:
        pass
    return h0

def binary_local_search(h0, n, max_iter=800, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h = (h0 >= 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones = np.where(h == 1)[0]
        remove = ones[np.argsort(h0[ones])[:n_ones - n // 2]]
        h[remove] = 0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        add = zeros[np.argsort(-h0[zeros])[:n // 2 - n_ones]]
        h[add] = 1
    c = compute_c5(h)
    for _ in range(max_iter):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_c = c
        best_pair = None
        npairs = min(len(ones) * len(zeros), 400)
        for _ in range(npairs):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            cn = compute_c5(h)
            if cn < best_c - 1e-14:
                best_c = cn
                best_pair = (i, j)
            h[i], h[j] = 1.0, 0.0
        if best_pair is not None:
            h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
            c = best_c
            improved = True
        if not improved:
            break
    return h, compute_c5(h)

def make_perturbation(n, scale, rng):
    mode = rng.integers(5)
    if mode == 0:
        return rng.standard_normal(n) * scale
    elif mode == 1:
        blen = max(3, int(rng.integers(3, n // 3 + 1)))
        p = np.zeros(n)
        for _ in range(int(rng.integers(1, 4))):
            start = int(rng.integers(0, max(1, n - blen)))
            p[start:start + blen] = rng.choice([-1, 1]) * scale * 2
        return p
    elif mode == 2:
        f = int(rng.integers(1, n // 4 + 1))
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n * 2.0
        return scale * np.sin(2 * np.pi * f * x + phase)
    elif mode == 3:
        p = np.zeros(n)
        k = int(rng.integers(3, max(4, n // 4)))
        idx = rng.choice(n, k, replace=False)
        p[idx] = rng.choice([-1, 1], k) * scale * 3
        return p
    else:
        x = np.arange(n)
        p = scale * (np.cos(2 * np.pi * x / n) * 0.7 + rng.standard_normal(n) * 0.3)
        return p
