import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(300):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

_N = 50

def _sum_con(x):
    return np.sum(x) - _N * 0.5

def _sum_jac(x):
    return np.ones(_N)

def _obj_alpha(x, alpha):
    n = _N
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    e = np.exp(np.clip(c - m, -50.0, 50.0))
    return m + np.log(np.sum(e)) / alpha

def _down(h, n_new):
    n_old = len(h)
    if n_new >= n_old:
        return _fix(np.interp(np.arange(n_new) * 2.0 / n_new,
                              np.arange(n_old) * 2.0 / n_old, h), n_new)
    idx = np.linspace(0, n_old - 1, n_new)
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n_old - 1)
    w = idx - i0
    return _fix(h[i0] * (1 - w) + h[i1] * w, n_new)

def _run_slsqp(h0, n, alpha, maxiter, t0, budget):
    global _N
    if budget - (time.time() - t0) < 6:
        return None
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_alpha, h0, (alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and len(res.x) == n:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _multi_opt(hi, n, t0, budget, n_starts, alpha_list, maxiter):
    global _N
    rng_l = np.random.RandomState(999 + n)
    best = hi.copy()
    best_c = _c5(best)
    for s in range(n_starts):
        if budget - (time.time() - t0) < 10:
            break
        if s == 0:
            start = hi.copy()
        elif s % 3 == 1:
            start = _fix(np.clip(best + rng_l.randn(n) * 0.06, 0, 1), n)
        else:
            start = _fix(np.clip(best + rng_l.randn(n) * 0.15, 0, 1), n)
        for alpha in alpha_list:
            if budget - (time.time() - t0) < 8:
                break
            ht = _run_slsqp(start, n, alpha, maxiter, t0, budget)
            if ht is not None:
                ct = _c5(ht)
                if ct < best_c:
                    best = ht
                    best_c = ct
                    start = ht.copy()
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(48, 0.5)

    n0 = len(h0)
    bh = _fix(h0, n0)
    bc = _c5(bh)

    # Phase 1: Downsample to low resolution for fast global search
    for n_lo in [48, 60, 72]:
        if budget_s - (time.time() - t0) < 120:
            break
        hi = _down(bh, n_lo)
        hi, hi_c = _multi_opt(hi, n_lo, t0, budget_s,
                              n_starts=8,
                              alpha_list=[30.0, 100.0, 400.0, 1500.0],
                              maxiter=1000)
        if hi_c < bc:
            bh = hi
            bc = hi_c

    # Phase 2: Progressive resolution increase with optimization
    for n_hi in [80, 100, 120, 160, 200, 250, 320, 400]:
        if budget_s - (time.time() - t0) < 80:
            break
        n_cur = len(bh)
        if n_hi <= n_cur:
            continue
        hi = _down(bh, n_hi)
        ct0 = _c5(hi)
        if ct0 > bc * 1.005:
            continue
        hi, hi_c = _multi_opt(hi, n_hi, t0, budget_s,
                              n_starts=5,
                              alpha_list=[50.0, 200.0, 800.0],
                              maxiter=700)
        if hi_c < bc:
            bh = hi
            bc = hi_c

    # Phase 3: Final polish at working resolution
    nb = len(bh)
    for alpha in [1000.0, 3000.0, 8000.0]:
        if budget_s - (time.time() - t0) < 10:
            break
        ht = _run_slsqp(bh, nb, alpha, 3000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh = ht
                bc = ct

    # Phase 4: Try going back to original high resolution
    if budget_s - (time.time() - t0) > 80 and n0 > len(bh):
        hi = _down(bh, n0)
        for alpha in [500.0, 2000.0]:
            if budget_s - (time.time() - t0) < 20:
                break
            ht = _run_slsqp(hi, n0, alpha, 1500, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh = ht
                    bc = ct

    # Phase 5: One more low-res exploration with different structure
    if budget_s - (time.time() - t0) > 60:
        # Try a staircase-like initial pattern
        for n_try in [48, 64]:
            if budget_s - (time.time() - t0) < 30:
                break
            # Staircase: gradually increasing values
            xs = np.linspace(0, 1, n_try)
            h_stair = np.clip(2.0 * xs - 0.5, 0, 1)
            h_stair = _fix(h_stair, n_try)
            # Also random 0/1
            h_rnd = np.zeros(n_try)
            h_rnd[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
            for h_init in [h_stair, h_rnd, _down(bh, n_try)]:
                if budget_s - (time.time() - t0) < 15:
                    break
                hi2, ci2 = _multi_opt(h_init, n_try, t0, budget_s,
                                      n_starts=4,
                                      alpha_list=[100.0, 500.0, 2000.0],
                                      maxiter=800)
                if ci2 < bc:
                    bh = hi2
                    bc = ci2

    # Final exact constraint fix
    nb = len(bh)
    bh = np.clip(bh, 0.0, 1.0)
    target = 0.5 * nb
    diff = target - np.sum(bh)
    if abs(diff) > 1e-12:
        mask = (bh > 1e-9) & (bh < 1.0 - 1e-9)
        nmask = np.sum(mask)
        if nmask > 0:
            bh = np.clip(bh + diff / nmask * mask, 0, 1)
        else:
            if diff > 0:
                bh[np.argmin(bh)] = min(1.0, bh[np.argmin(bh)] + diff)
            else:
                bh[np.argmax(bh)] = max(0.0, bh[np.argmax(bh)] + diff)
    # One more _fix pass
    bh = _fix(bh, nb)
    bc = _c5(bh)
    return bh, bc, nb
