def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def coord_descent(h, n, max_sweeps=200, t0=time.time(), deadline=None):
    """Gauss-Seidel coordinate descent with exact 1D minimax."""
    h = h.copy()
    dx = 2.0 / n
    s_vals = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for j in range(n):
            # Compute current correlation
            C = np.correlate(h, 1.0 - h, mode='full')
            old = h[j]
            
            # B_s = 1 - h[j-s]*v1 - h[j+s]*v2
            B = np.ones(2*n-1)
            v1 = (j - s_vals >= 0) & (j - s_vals < n)
            v2 = (j + s_vals >= 0) & (j + s_vals < n)
            B[v1] -= h[j - s_vals[v1]]
            B[v2] -= h[j + s_vals[v2]]
            
            # A_s = C_s - old * B_s
            A = C - old * B
            
            # Minimize max_s(A_s + t*B_s) over t in [0,1]
            # Coarse grid
            ts = np.linspace(0, 1, 20)
            vals = np.array([np.max(A + t*B) for t in ts])
            best_i = np.argmin(vals)
            best_f = vals[best_i]
            best_t = ts[best_i]
            
            # Refine
            lo = max(0.0, best_t - 1.0/20)
            hi = min(1.0, best_t + 1.0/20)
            ts2 = np.linspace(lo, hi, 10)
            for t in ts2:
                f = np.max(A + t * B)
                if f < best_f:
                    best_f = f
                    best_t = t
            
            h[j] = best_t
        
        # Fix sum
        h = project_feasible(h, n)
    
    return h

2. The subgradient at t is conv{B_s : A_s + B_s*t = f(t)}.

1. Sort the lines by their value at t=0.
2. The upper envelope changes at crossing points.
3. Find the t where the slope of the upper envelope changes sign.

- At t=0: compute active set. If all B_s < 0 for active s, move right. If all > 0, move left.

import numpy as np
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def find_optimal_t(A, B):
    """Find t in [0,1] minimizing max_s(A_s + B_s*t) exactly."""
    m = len(A)
    f0 = np.max(A)
    f1 = np.max(A + B)
    best_f = min(f0, f1)
    best_t = 0.0 if f0 <= f1 else 1.0
    # Find all pairwise crossing points in [0,1]
    if m < 2:
        return best_t, best_f
    i_idx, j_idx = np.triu_indices(m, 1)
    db = B[i_idx] - B[j_idx]
    valid = np.abs(db) > 1e-14
    if np.any(valid):
        t_cross = (A[j_idx[valid]] - A[i_idx[valid]]) / db[valid]
        t_cross = t_cross[(t_cross >= 0.0) & (t_cross <= 1.0)]
        if len(t_cross) > 0:
            t_candidates = np.concatenate(([best_t], t_cross))
            all_vals = A[:, None] + B[:, None] * t_candidates[None, :]
            scores = np.max(all_vals, axis=0)
            bi = np.argmin(scores)
            if scores[bi] < best_f:
                best_f = scores[bi]
                best_t = t_candidates[bi]
    return best_t, best_f

def coord_descent(h, n, max_sweeps=200, deadline=None):
    """Gauss-Seidel coordinate descent with exact 1D minimax."""
    h = h.copy().astype(float)
    s_vals = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for j in range(n):
            C = np.correlate(h, 1.0 - h, mode='full')
            old = h[j]
            B = np.ones(2 * n - 1)
            v1 = (j - s_vals >= 0) & (j - s_vals < n)
            v2 = (j + s_vals >= 0) & (j + s_vals < n)
            B[v1] -= h[j - s_vals[v1]]
            B[v2] -= h[j + s_vals[v2]]
            A = C - old * B
            t_new, _ = find_optimal_t(A, B)
            h[j] = t_new
        h = project_feasible(h, n)
    return h

def subgradient_step(h, n, eta):
    """One projected subgradient step."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(c)
    active = (c >= max_val - 1e-10)
    n_act = np.sum(active)
    if n_act == 0:
        n_act = 1
    grad = np.zeros(n)
    s_vals = np.arange(-(n - 1), n)
    for k in range(len(c)):
        if not active[k]:
            continue
        s = k - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            first = np.zeros(n)
            first[s:] = 1.0 - h[:n - s]
            second = np.zeros(n)
            second[:n - s] = -h[s:]
        else:
            t = -s
            first = np.zeros(n)
            first[:n - t] = 1.0 - h[t:]
            second = np.zeros(n)
            second[t:] = -h[:n - t]
        grad += (first + second)
    grad *= dx / n_act
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new
