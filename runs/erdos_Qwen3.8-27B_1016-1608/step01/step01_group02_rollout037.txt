import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounded_sum(v, target):
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(120):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    active = np.where(corr >= m - 1e-12)[0]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        if d >= 0:
            if n > d:
                grad[:n - d] += (1.0 - h[d:n])
                grad[d:] += -h[:n - d]
        else:
            ad = -d
            grad[ad:] += (1.0 - h[:n - ad])
            grad[:n - ad] += -h[ad:]
    return grad / len(active)

def smooth_obj_val(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + temp * np.log(np.mean(np.exp((corr - m) / temp)))) * dx

def smooth_grad_val(h, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            if n > d:
                grad[:n - d] += w[k] * (1.0 - h[d:n])
                grad[d:] += -w[k] * h[:n - d]
        else:
            ad = -d
            grad[ad:] += w[k] * (1.0 - h[:n - ad])
            grad[:n - ad] += -w[k] * h[ad:]
    return grad

def sum_eq_con(x, target):
    return x.sum() - target

def c5_wrapper(x, dx):
    return compute_c5(x, dx)

def smooth_obj_wrapper(x, dx, temp):
    return smooth_obj_val(x, dx, temp)

def smooth_jac_wrapper(x, dx, temp):
    return smooth_grad_val(x, temp)
