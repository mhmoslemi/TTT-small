    """Returns (smooth_max_value, gradient)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if eps < 1e-12:
        # Hard max
        idx = np.argmax(corr)
        w = np.zeros(2*n-1)
        w[idx] = 1.0
    else:
        s = np.clip((corr - m) / eps, -50, 50)
        exps = np.exp(s)
        w = exps / exps.sum()
    value = m + eps * np.log(np.sum(np.exp(np.clip((corr-m)/eps, -50, 50)))) if eps >= 1e-12 else m
    
    # Compute gradient
    grad = np.zeros(n)
    for p in range(2*n-1):
        k = p - (n-1)
        wk = w[p] * dx
        if abs(wk) < 1e-15:
            continue
        if k == 0:
            grad += wk * (1 - 2*h)
        elif k > 0:
            grad[:n-k] -= wk * h[k:]
            grad[k:] += wk * (1 - h[:n-k])
        else:
            mk = -k
            grad[:n-mk] += wk * (1 - h[mk:])
            grad[mk:] -= wk * h[:n-mk]
    return value, grad

def pgd_optimize(h, n, eps_start=0.05, eps_end=0.0001, max_iters=2000, tol=1e-12):
    """Projected gradient descent"""
    h = h.copy()
    lr = 0.01
    # Adaptive eps schedule
    for it in range(max_iters):
        frac = it / max_iters
        eps = eps_start * (eps_end / eps_start) ** frac
        
        val, grad = smooth_max_grad(h, n, eps)
        
        # Backtracking line search
        for trial in range(30):
            h_new = project_feasible(h - lr * grad, n)
            val_new, _ = smooth_max_grad(h_new, n, eps)
            if val_new < val - 1e-15:
                break
            lr *= 0.5
        else:
            break
        
        h = h_new
        val = val_new
        
        if trial == 0 and abs(val - val_new) < tol:
            break
    return h

def slsqp_optimize(h, n, eps=0.001, maxiter=300):
    """SLSQP with smooth max"""
    def obj(x):
        v, _ = smooth_max_grad(x, n, eps)
        return v
    def obj_grad(x):
        _, g = smooth_max_grad(x, n, eps)
        return g
    def con_fun(x):
        return np.sum(x) - n/2.0
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair(res.x, n)

    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            return h
        h += (target - s) / n
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
