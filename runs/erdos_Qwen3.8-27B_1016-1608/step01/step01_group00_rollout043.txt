```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def slsqp_opt(h0, n, maxiter=1000, ftol=1e-15):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def sum_con(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < compute_c5(h0):
            return repair(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h0

def best_improvement_swap(h, n, max_sweeps=200):
    """Try all possible 1↔0 swaps, apply the best one. Repeat."""
    c5 = compute_c5(h)
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    
    impays = 0
    for sweep in range(max_sweeps):
        best_val = c5
        best_i, best_j = -1, -1
        ones_list = sorted(ones_set)
        zeros_list = sorted(zeros_set)
        
        for i in ones_list:
            for j in zeros_list:
                h[i], h[j] = 0, 1
                val = compute_c5(h)
                h[i], h[j] = 1, 0
                if val < best_val - 1e-15:
                    best_val = val
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i], h[best_j] = 0, 1
            ones_set.remove(best_i)
            ones_set.add(best_j)
            zeros_set.remove(best_j)
            zeros_set.add(best_i)
            c5 = best_val
            impays = 0
        else:
            impays += 1
            if impays >= 3:
                break
    
    return h, c5

def sa_binary_search(h, n, t_start, budget_s):
    """Simulated annealing on binary solutions."""
    c5 = compute_c5(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = 0.005
    T_min = 0.00005
    
    while time.time() - t_start < budget_s:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i], h[j] = 0, 1
        c5_new = compute_c5(h)
        h[i], h[j] = 1, 0
        
        delta = c5_new - c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-15)):
            h[i], h[j] = 0, 1
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            c5 = c5_new
        
        T = max(T * 0.999, T_min)
    
    return h, c5
```

```python
def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - m)))) + m

def smooth_c5_grad(h, beta, eps=1e-7):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_p = h.copy()
        h_m = h.copy()
        h_p[i] += eps
        h_m[i] -= eps
        # Need to maintain 0 ≤ h ≤ 1, but for small eps this is fine
        grad[i] = (smooth_c5(h_p, beta) - smooth_c5(h_m, beta)) / (2*eps)
    return grad
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def slsqp_opt(h0, n, maxiter=2000, ftol=1e-16):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def con(x):
        return float(np.sum(x) - n / 2.0)
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = repair(np.clip(res.x, 0, 1), n)
            if compute_c5(h_new) < compute_c5(h0):
                return h_new
    except Exception:
        pass
    return h0

    except Exception:
        pass
    return h0

    betas = [5, 10, 20, 50, 100, 200, 400]

            break
        h = slsqp_smooth(h, n, beta)
        # Interleave with the true objective
        h2 = slsqp_opt(h, n, maxiter=200)
        if compute_c5(h2) < compute_c5(h):
            h = h2
    return h

def best_improvement_swap(h, n, t_start, budget_s):
    c5 = compute_c5(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    no_improve = 0
    while time.time() - t_start < budget_s and no_improve < 5:
        best_val = c5
        best_i, best_j = -1, -1
        for i in sorted(ones):
            for j in sorted(zeros):
                h[i], h[j] = 0, 1
                val = compute_c5(h)
                h[i], h[j] = 1, 0
                if val < best_val - 1e-15:
                    best_val = val
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i], h[best_j] = 0, 1
            ones.discard(best_i)
            ones.add(best_j)
            zeros.discard(best_j)
            zeros.add(best_i)
            c5 = best_val
            no_improve = 0
        else:
            no_improve += 1
    
    return h, c5

def sa_binary(h, n, t_start, budget_s):

    zeros = list(np.where(h == 0)[0])

        delta = c5_new - c5
```

Let me also think about one more optimization technique: **coordinate descent**.

For the continuous problem, coordinate descent means:
- For each i, optimize h[i] while keeping all other h[j] fixed
- The constraint sum(h) = n/2 complicates this: changing h[i] requires adjusting another h[j]
- But we can relax: optimize h[i] to minimize C5, then project back

For a single coordinate:
C5 as a function of h[i] is a piecewise-linear-max of linear functions (for fixed other h values). The max makes it non-smooth but convex (since each term is linear in h[i]). Wait, is it convex?

overlap_k = sum_j h[j] * (1-h[j+k]) * dx

For fixed h (except h[i]), this is:
overlap_k = h[i] * (1-h[i+k]) * dx + (terms not involving i) + (if k' = -k exists) h[i-k] * (1-h[i]) * dx + ...

So overlap_k is linear in h[i] (at most, since h[i] appears in at most two products: as h[i] and as (1-h[i])).

Wait: h[i] appears in overlap_k at position i (as h[i]) and at position i-k (as (1-h[i])).

overlap_k = ... + h[i] * (1-h[i+k]) * dx + ... + h[i-k] * (1-h[i]) * dx + ...

= ... + h[i] * c1 * dx - h[i] * h[i-k] * dx + ...

This is AFFINE in h[i] (the -h[i]*h[i-k]*dx term is linear in h[i] since h[i-k] is fixed).

So each overlap_k is affine in h[i], and C5 = max_k overlap_k is the max of affine functions, which is CONVEX in h[i].

This means coordinate descent will work! And for a single coordinate, the optimum is where the derivative changes sign (or at the boundary 0 or 1).

The derivative of overlap_k w.r.t. h[i] is:
= (1-h[i+k]) * dx - h[i-k] * dx (for appropriate boundary terms)

So d/dh[i] C5 = max over k where overlap_k is on the "active set" of:
(1-h[i+k]) * dx - h[i-k] * dx

The optimum h[i] is where this derivative crosses 0, or at 0/1.

This is a very efficient algorithm! For each coordinate, the update is O(n) (to find the active set and compute the derivative).
