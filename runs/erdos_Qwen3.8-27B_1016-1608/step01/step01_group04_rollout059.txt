import numpy as np
from scipy.optimize import minimize
import time

_P = {'n': 0, 'eps': 0.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _smax(x):
    n, eps, dx = _P['n'], _P['eps'], 2.0 / _P['n']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-12:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _efun(x):
    return float(np.sum(x) - _P['n'] / 2.0)

def _ejac(x):
    return np.ones(_P['n'])

def slsqp(h, n, eps, mi, ft):
    _P['n'], _P['eps'] = n, eps
    r = minimize(_smax, h.copy(), method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                 options={'maxiter': mi, 'ftol': ft})
    return repair(r.x, n)
