def project_to_simplex(h, target):
    """Project h onto {0 <= h_i <= 1, sum(h) = target}"""
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 1e-15
            if mask.sum() == 0:
                break
            s = min(excess / mask.sum(), h[mask].min() - 1e-15)
            if s <= 0:
                s = excess / mask.sum()
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-15

            if s <= 0:
                s = -excess / mask.sum()
            h[mask] += s

    best_h = h_c.copy()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_simplex(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 1e-15
            if mask.sum() == 0:
                break
            s = min(excess / mask.sum(), h[mask].min() - 1e-15)
            if s <= 0:
                s = excess / mask.sum()
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-15
            if mask.sum() == 0:
                break
            s = min(-excess / mask.sum(), 1.0 - h[mask].min() - 1e-15)
            if s <= 0:
                s = -excess / mask.sum()
            h[mask] += s
    return np.clip(h, 0.0, 1.0)


def lsmax_f(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = temp * (corr - m)
    s -= s.max()
    return (m + np.log(np.exp(s).sum() / len(corr)) / temp) * dx


def corr_grad_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k < n:
        g[k:n] += (1.0 - h[0:n - k])
        g[0:n - k] -= h[k:n]
    else:
        m = k - n + 1
        g[0:n - m] += (1.0 - h[m:n])
        g[m:n] -= h[0:n - m]
    return g


def nesterov_lsmax(h0, dx, target, temp, max_iter=3000, init_step=0.005):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    step = init_step
    best = compute_c5(x, dx)
    best_x = x.copy()
    for it in range(max_iter):
        corr = np.correlate(y, 1.0 - y, mode="full")
        m = corr.max()
        s = temp * (corr - m)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-8:
                continue
            grad += w[k] * corr_grad_single(y, k)
        grad *= dx
        t = step
        y = project_to_simplex(2 * y - x, target)
        x_new = project_to_simplex(y - t * grad, target)
        x, y = x_new, y_new
        c5 = compute_c5(x, dx)
        if c5 < best:
            best = c5
            best_x = x.copy()
        if (it + 1) % 500 == 0:
            step *= 0.85
    return best_x, best


def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = 1.0
    best_n = 0
    best_dx = 1.0

    def try_update(h_new, n_new, dx_new):
        nonlocal best_h, best_c5, best_n, best_dx
        if h_new is None:
            return
        h_new = project_to_simplex(h_new, n_new / 2.0)
        c5 = compute_c5(h_new, dx_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            best_n = n_new
            best_dx = dx_new

        best_h = h_init.copy()

        best_n = n0
        best_dx = dx0
    else:

        dx0 = 2.0 / n0
        best_h = np.clip(0.5 + 0.3 * rng.randn(n0), 0, 1)
        best_h = project_to_simplex(best_h, n0 / 2.0)
        best_c5 = compute_c5(best_h, dx0)
        best_dx = dx0

    # Phase 1: SLSQP with log-sum-exp (n=70)

    bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(lsmax_f, best_h.copy(), args=(dx, temp),
                           method='SLSQP', bounds=bounds, constraints=eq_con,

        except Exception:
            pass

    # Phase 2: Nesterov accelerated gradient on lsmax
    if remaining() > 60:

            if remaining() < 30:

            h_nr, c5_nr = nesterov_lsmax(best_h, dx, target, temp,
                                         max_iter=2000, init_step=0.003)
            if c5_nr < best_c5:
                best_h, best_c5 = h_nr, c5_nr
            # SLSQP polish from nesterov result

                eq_con = [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t}]
                try:
                    res = minimize(compute_c5, best_h.copy(), args=(dx,),
                                   method='SLSQP', bounds=bounds, constraints=eq_con,
                                   options={'maxiter': 300, 'ftol': 1e-15})
                    try_update(res.x, n, dx)
                except Exception:
                    pass

    # Phase 3: Projected subgradient with iteration
    if remaining() > 80:
        h_sg = best_h.copy()
        step = 0.008
        for it in range(30000):
            if remaining() < 15:
                break
            corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
            k_star = int(np.argmax(corr))
            g = corr_grad_single(h_sg, k_star) * dx
            h_sg = project_to_simplex(h_sg - step * g, target)
            if (it + 1) % 500 == 0:
                step *= 0.95
            c5 = compute_c5(h_sg, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_sg.copy(), c5

    # Phase 4: Coordinate descent with adaptive step
    if remaining() > 40:
        h_cd = best_h.copy()
        c5_cd = best_c5
        step_cd = 0.04
        n_no_improve = 0
        for it in range(80000):
            if remaining() < 5:
                break
            idx = it % n  # cycle through all coordinates
            h_t = h_cd.copy()
            h_t[idx] = np.clip(h_t[idx] + step_cd, 0.0, 1.0)
            h_t = project_to_simplex(h_t, target)
            c5_t = compute_c5(h_t, dx)
            if c5_t < c5_cd - 1e-14:
                h_cd, c5_cd = h_t, c5_t
                n_no_improve = 0
                if c5_cd < best_c5:
                    best_h, best_c5 = h_cd.copy(), c5_cd
            else:
                h_t2 = h_cd.copy()
                h_t2[idx] = np.clip(h_t2[idx] - step_cd, 0.0, 1.0)
                h_t2 = project_to_simplex(h_t2, target)
                c5_t2 = compute_c5(h_t2, dx)
                if c5_t2 < c5_cd - 1e-14:
                    h_cd, c5_cd = h_t2, c5_t2
                    n_no_improve = 0
                    if c5_cd < best_c5:
                        best_h, best_c5 = h_cd.copy(), c5_cd
                else:
                    n_no_improve += 1
            if n_no_improve > 500:
                step_cd = max(0.002, step_cd * 0.7)
                n_no_improve = 0

    # Phase 5: Try n=140 (finer discretization)
    if remaining() > 50:
        n2, dx2, target2 = 140, 2.0 / 140, 70.0
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_to_simplex(h2, target2)
        b2 = [(0.0, 1.0)] * n2
        eq2 = [{'type': 'eq', 'fun': lambda x, t=target2: x.sum() - t}]
        for temp in [100.0, 30.0]:
            if remaining() < 20:
                break
            try:
                res = minimize(lsmax_f, h2.copy(), args=(dx2, temp),
                               method='SLSQP', bounds=b2, constraints=eq2,
                               options={'maxiter': 250, 'ftol': 1e-13})
                try_update(res.x, n2, dx2)
            except Exception:
                pass

    # Phase 6: SLSQP from best with exact objective
    if remaining() > 20:
        n, dx, target = best_n, best_dx, best_n / 2.0
        bounds = [(0.0, 1.0)] * n
        eq_con = [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t}]

                               method='SLSQP', bounds=bounds, constraints=eq_con,
                               options={'maxiter': 500, 'ftol': 1e-15})

            except Exception:
                pass

    # Fix sum exactly

        if mask.sum() > 0:

            best_h[mask] = np.clip(best_h[mask] + u, 0, 1)
    final_c5 = compute_c5(best_h, best_dx)
    return (best_h, final_c5, best_n)
