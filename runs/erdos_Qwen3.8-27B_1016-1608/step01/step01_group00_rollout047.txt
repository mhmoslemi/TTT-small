def project_onto_constraints(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0

        diff = target - s

        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    c5 = float(np.max(corr))
    active = np.where(corr >= c5 - 1e-9)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i] += (1.0 - h[i + s])
                grad[i + s] += (-h[i])
        else:
            t = -s
            for i in range(t, n):
                grad[i] += (1.0 - h[i - t])
                grad[i - t] += (-h[i])
    grad *= dx / len(active)
    return c5, grad


def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def subgrad_step(h, n, max_iter=3000, alpha0=0.08):
    c5_cur = compute_c5(h)
    alpha = alpha0
    stall = 0
    for _ in range(max_iter):
        _, grad = compute_c5_grad(h)
        improved = False
        for _ in range(25):
            h_try = project(h - alpha * grad, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-15:
                h, c5_cur = h_try, c5_try
                alpha = min(alpha * 1.08, 0.4)
                stall = 0
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            if stall > 30:
                break
    return h, c5_cur


def sum_eq(x):
    return np.sum(x) - len(x) / 2.0


def slsqp(h, n, maxiter=1000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    best = h.copy()
    best_c = compute_c5(h)
    try:
        res = minimize(compute_c5, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < best_c + 1e-14:
            best = project(np.clip(res.x, 0.0, 1.0), n)
            best_c = compute_c5(best)
    except Exception:
        pass
    return best, best_c


def block_flip(h, n, rng):
    out = h.copy()
    length = rng.randint(3, max(4, n // 3))
    start = rng.randint(0, n - length)
    out[start:start + length] = 1.0 - out[start:start + length]
    return project(out, n)


def circ_shift(h, n, rng):
    k = rng.randint(1, n // 4 + 1)
    out = np.roll(h, k)
    return project(out, n)
