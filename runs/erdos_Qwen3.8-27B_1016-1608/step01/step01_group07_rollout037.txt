def _smax_grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    v = np.clip(v, -5.0, 5.0)
    amax = v.max()
    e = np.exp(alpha * (v - amax))
    p = e / e.sum()
    
    # q = reversed p
    q = p[::-1].copy()
    
    ones_minus_h = 1.0 - h
    grad = np.zeros(n)
    
    for j in range(n):
        # term1[j] = dot(q[n-1-j : n-1-j+n], ones_minus_h)
        t1 = np.dot(q[n-1-j : 2*n-1-j], ones_minus_h)
        # term2[j] = dot(q[j : j+n][::-1], h)
        t2 = np.dot(q[j : j+n][::-1], h)
        grad[j] = dx * (t1 - t2)
    
    # penalty
    grad += 2.0 * lam * (h.sum() - 0.5 * n)
    
    obj = amax + np.log(e.sum()) / alpha + lam * (h.sum() - 0.5 * n)**2
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _fix(h, n):
    tgt = 0.5 * n
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        k = int(m.sum())
        if k == 0:
            break
        h = np.clip(h + d / k * m, 0.0, 1.0)
    return h

def _embed(h, n):
    n0 = len(h)
    xo = np.arange(n0) * 2.0 / n0
    xn = np.arange(n) * 2.0 / n
    return _fix(np.clip(np.interp(xn, xo, h), 0.0, 1.0), n)

def _fun(h, alpha, lam):
    n = len(h)
    v = np.clip(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n), -5.0, 5.0)
    m = v.max()
    return float(m + np.log(np.sum(np.exp(alpha * (v - m)))) / alpha + lam * (h.sum() - 0.5 * n) ** 2)

def _grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    v = np.clip(np.correlate(h, 1.0 - h, mode='full') * dx, -5.0, 5.0)
    m = v.max()
    e = np.exp(alpha * (v - m))
    p = e / e.sum()
    q = p[::-1]
    omh = 1.0 - h
    g = np.empty(n)
    for j in range(n):
        g[j] = dx * (np.dot(q[n - 1 - j: 2 * n - 1 - j], omh) - np.dot(q[j: j + n][::-1], h))
    g += 2.0 * lam * (h.sum() - 0.5 * n)
    return g
