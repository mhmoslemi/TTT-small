import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_c5(h, tau):
    c = corr_scaled(h)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(tau * (c - m)))) / tau

def soft_grad(h, tau):
    n = len(h)
    c = corr_scaled(h)
    m = np.max(c)
    w = np.exp(tau * (c - m))
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n - 1, n - 1 + k)

            continue

        jj = ii + k - (n - 1)
        np.add.at(grad, jj, wk * (1.0 - h[ii]))
        np.add.at(grad, ii, -wk * h[jj])
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    scaled = tau * (corr - m)
    w = np.exp(scaled)
    sw = np.sum(w)
    if sw < 1e-300:
        # All weights underflow, use argmax
        idx = np.argmax(corr)
        w = np.zeros(len(corr))
        w[idx] = 1.0
        sw = 1.0
    w /= sw
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-10:
            continue
        s = idx - (n - 1)
        a = s if s > 0 else 0
        b = -s if s < 0 else 0
        max_m = n - 1 - a - b
        j = np.arange(n)
        m1 = j - a
        mask1 = (m1 >= 0) & (m1 <= max_m)
        if np.any(mask1):
            grad[mask1] += wk * (1.0 - h[m1[mask1] + b])
        m2 = j - b
        mask2 = (m2 >= 0) & (m2 <= max_m)
        if np.any(mask2):
            grad[mask2] += -wk * h[m2[mask2] + a]
    return grad * dx

def slsqp_local(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return np.max(c) * (2.0 / len(x))
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < obj(h0) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

    t0 = time.time()

    h = h_best.copy()
    v = np.zeros(n)
    mu = 0.85
    lr = 0.03
    tau = 30.0
    best_in_phase = c5_best
    stall = 0
    
    while time.time() - t0 < budget_s * 0.50:
        f = compute_c5(h)
        if f < c5_best - 1e-12:
            c5_best = f
            h_best = h.copy()
            best_in_phase = f
            stall = 0
        else:
            stall += 1
        
        if stall > 100:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    scaled = tau * (corr - m)
    w = np.exp(scaled)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
        sw = 1.0
    w /= sw
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-10:
            continue
        s = idx - (n - 1)
        a = s if s > 0 else 0
        b = -s if s < 0 else 0
        max_m = n - 1 - a - b
        j = np.arange(n)
        m1 = j - a
        mask1 = (m1 >= 0) & (m1 <= max_m)
        if np.any(mask1):
            grad[mask1] += wk * (1.0 - h[m1[mask1] + b])
        m2 = j - b
        mask2 = (m2 >= 0) & (m2 <= max_m)
        if np.any(mask2):
            grad[mask2] += -wk * h[m2[mask2] + a]
    return grad * dx

def slsqp_local(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return np.max(c) * (2.0 / len(x))
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < obj(h0) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def binary_push(h, n, alpha=5.0):
    """Push towards binary while maintaining smoothness"""
    h_pushed = 1.0 / (1.0 + np.exp(-alpha * (2.0 * h - 1.0)))
    return repair(h_pushed, n)
