    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def soft_max_grad(h, tau=150.0):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    cm = np.max(c)
    weights = np.exp(tau * (c - cm)) / np.sum(np.exp(tau * (c - cm)))
    # grad of soft-max = sum_k weights[k] * grad(c[k])
    grad = np.zeros(n)
    j = np.arange(n)
    for m in range(len(c_raw)):
        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)
        idx1 = j + k
        mask1 = (idx1 >= 0) & (idx1 < n)
        g = np.zeros(n)
        g[mask1] += (1.0 - h[idx1[mask1]])
        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        g[mask2] -= h[idx2[mask2]]
        grad += weights[m] * g * dx
    return grad

def grad_c_at_m(h, m):
    n = len(h)
    dx = 2.0 / n
    k = m - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    idx1 = j + k
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = j - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] -= h[idx2[mask2]]
    return grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def subgradient_optimize(h_start, n, n_iter, alpha_init, alpha_final, t0, budget_frac):
    """Projected subgradient on active constraints"""
    h = h_start.copy()
    c_all = compute_corr(h)
    for it in range(n_iter):
        if time.time() - t0 > budget_frac:
            break
        progress = it / n_iter
        alpha = alpha_init * (alpha_final / alpha_init) ** progress
        # Weight top 3 active constraints
        top_idx = np.argsort(c_all)[::-1][:3]
        top_vals = c_all[top_idx]
        w = np.exp(100 * (top_vals - np.max(top_vals)))
        w /= np.sum(w)
        grad = np.zeros(n)
        for i, m in enumerate(top_idx):
            if w[i] < 0.01:
                continue
            grad += w[i] * grad_c_at_m(h, m)
        h = h - alpha * grad
        h = project_feasible(h, n)
        c_all = compute_corr(h)
    return h, compute_score(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    def elapsed():
        return time.time() - t0

    def time_left_frac(frac=0.92):
        return elapsed() < budget_s * frac

    # Phase 1: Optimize at initial resolution and nearby
    n_candidates = [n_init]
    for dn in [-5, -3, 2, 5, 8, 12, 20, 30]:
        nc = n_init + dn
        if 30 <= nc <= 200 and nc not in n_candidates:
            n_candidates.append(nc)
    n_candidates.sort()

    x_init = np.linspace(0.0, 2.0, n_init, endpoint=False)

    for n_points in n_candidates:
        if not time_left_frac(0.50):
            break
        t_phase_start = elapsed()
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_init, h_init)
        h0 = project_feasible(h0, n_points)
        s0 = compute_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()

        phase_time = min(budget_s * 0.25, max(30, (budget_s * 0.50 - elapsed()) / max(1, len(n_candidates) - n_candidates.index(n_points))))

        # SLSQP with soft-max, multiple restarts
        n_restarts = 5 if n_points <= 70 else 3
        for r in range(n_restarts):
            if not time_left_frac(0.50):
                break
            if r == 0:
                h_start = h0.copy()
            elif r == 1:
                h_start = project_feasible(h0 + rng.randn(n_points) * 0.05, n_points)
            elif r == 2:
                h_start = h0.copy()
                k = max(2, n_points // 10)
                idx = rng.choice(n_points, size=k, replace=False)
                h_start[idx] += rng.uniform(-0.4, 0.4, size=k)
                h_start = project_feasible(h_start, n_points)
            elif r == 3:
                h_start = project_feasible(h0 + rng.randn(n_points) * 0.12, n_points)
            else:
                # Fourier perturbation
                h_start = h0.copy()
                freq = rng.randint(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                h_start += 0.08 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
                h_start = project_feasible(h_start, n_points)

            # Pass 1: soft-max with moderate tau
            try:
                res = minimize(soft_max_score, h_start, args=(120.0,), method='SLSQP',
                               bounds=[(0, 1)] * n_points,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}],
                               options={'maxiter': 400, 'ftol': 1e-11})
                h_cand = project_feasible(res['x'], n_points)
                s_cand = compute_score(h_cand)
            except Exception:
                s_cand, h_cand = np.inf, h_start

            # Pass 2: refine with higher tau
            if s_cand < best_score + 0.005:
                try:
                    res2 = minimize(soft_max_score, h_cand, args=(400.0,), method='SLSQP',
                                    bounds=[(0, 1)] * n_points,
                                    constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}],
                                    options={'maxiter': 300, 'ftol': 1e-13})
                    h_cand = project_feasible(res2['x'], n_points)
                    s_cand = compute_score(h_cand)
                except Exception:
                    pass

            # Pass 3: subgradient polish
            if s_cand < best_score + 0.003 and time_left_frac(0.48):
                h_sg, s_sg = subgradient_optimize(
                    h_cand, n_points, n_iter=800,
                    alpha_init=0.02, alpha_final=0.001,
                    t0=t0, budget_frac=budget_s * 0.48)
                if s_sg < s_cand:
                    s_cand, h_cand = s_sg, h_sg

            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand.copy()
            if s_cand < s0:
                s0 = s_cand
                h0 = h_cand.copy()

    # Phase 2: Nelder-Mead polish at best resolution
    if time_left_frac(0.15) and len(best_h) <= 120:
        n_nm = len(best_h)
        for attempt in range(3):
            if not time_left_frac(0.10):
                break
            h_start = best_h.copy()
            if attempt > 0:
                h_start = project_feasible(h_start + rng.randn(n_nm) * 0.02, n_nm)

            def nm_obj(h):
                s = compute_score(h)
                viol = (np.sum(h) - n_nm / 2.0) ** 2 * 500.0
                viol += np.sum(np.maximum(0, h - 1.0)) ** 2 * 500.0
                viol += np.sum(np.maximum(0, -h)) ** 2 * 500.0
                return s + viol

            try:
                res = minimize(nm_obj, h_start, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-12})
                h_nm = project_feasible(res['x'], n_nm)
                s_nm = compute_score(h_nm)
                if s_nm < best_score:
                    best_score = s_nm
                    best_h = h_nm.copy()
            except Exception:
                pass

    # Phase 3: Final SLSQP polish
    if time_left_frac(0.03) and len(best_h) <= 120:
        n_f = len(best_h)
        try:
            res = minimize(soft_max_score, best_h, args=(500.0,), method='SLSQP',
                           bounds=[(0, 1)] * n_f,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_f / 2.0}],
                           options={'maxiter': 200, 'ftol': 1e-14})
            h_f = project_feasible(res['x'], n_f)
            s_f = compute_score(h_f)
            if s_f < best_score:
                best_score = s_f
                best_h = h_f
        except Exception:
            pass

    # Verify
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    final_score = compute_score(best_h)
    return best_h, final_score, len(best_h)
