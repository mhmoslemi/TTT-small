    h = h.copy()

    dx = 2.0 / n
    v = np.correlate(x, 1.0 - x, 'full') * dx
    vm = np.max(v)
    a = np.clip(alpha * (v - vm), -700, 700)
    exp_a = np.exp(a)
    Z = np.sum(exp_a)
    obj = vm + np.log(Z) / alpha
    w = exp_a / Z
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        g += wk * _grad_single(k, x, n, dx)
    return obj, g

def _cons_fun(x, n):
    return np.sum(x) - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def _slsqp(h0, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 5:
        return None
    bnds = [(0.0, 1.0)] * n
    def of(x):
        o, _ = _obj_and_grad(x, n, alpha)
        return o
    def gf(x):
        _, g = _obj_and_grad(x, n, alpha)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: _cons_fun(x, n),
             'jac': lambda x: _cons_jac(x, n)}]
    try:
        res = minimize(of, h0, jac=gf, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is None:
            return None
        return _project(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return None

def _active_search(h, n, t0, time_limit, max_rounds):
    dx = 2.0 / n
    best = h.copy()
    best_c = _c5_val(best)
    step = 0.05
    for rnd in range(max_rounds):
        if time.time() - t0 > t0 + time_limit:
            break
        improved = False
        v = np.correlate(best, 1.0 - best, 'full') * dx
        top_k = np.argsort(v)[-6:]
        for kk in top_k:
            if time.time() - t0 > t0 + time_limit:
                break
            g = _grad_single(int(kk), best, n, dx)
            for lr in [step, step * 0.5, step * 0.1]:
                h_try = _project(best - lr * g, n)
                c_try = _c5_val(h_try)
                if c_try < best_c - 1e-14:
                    best = h_try
                    best_c = c_try
                    improved = True
                    break
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
    return best, best_c

def _interp_to_res(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.arange(n_new) * (2.0 / n_new)
    hh = np.clip(np.interp(xn, xo, h), 0.0, 1.0)
    return _project(hh, n_new)

def _perturb(h, rng, scale):
    return _project(np.clip(h + rng.randn(len(h)) * scale, 0, 1), len(h))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(400, 0.5)
    n = len(h)
    h = _project(h, n)
    best_h = h.copy()
    best_c = _c5_val(best_h)

    # Phase 1: SLSQP with analytical gradient at full resolution
    alphas = [50.0, 150.0, 400.0, 1000.0]
    for alpha in alphas:
        if budget_s - (time.time() - t0) < 60:
            break
        for r in range(3):
            if budget_s - (time.time() - t0) < 30:
                break
            scale = 0.02 if r == 0 else 0.04
            hi = _perturb(best_h, rng, scale)
            h_try = _slsqp(hi, n, alpha, 600, t0, budget_s)
            if h_try is not None:
                c_try = _c5_val(h_try)
                if c_try < best_c:
                    best_h = h_try
                    best_c = c_try

    # Phase 2: Active constraint local search
    if budget_s - (time.time() - t0) > 200:
        tl = min(200, budget_s - (time.time() - t0) - 100)
        best_h, best_c = _active_search(best_h, n, t0, tl, 200)

    # Phase 3: Try at reduced resolution, then interpolate back
    for n_lo in [200, 300]:
        if budget_s - (time.time() - t0) < 80:
            break
        h_lo = _interp_to_res(best_h, n_lo)
        for alpha in [200.0, 600.0]:
            if budget_s - (time.time() - t0) < 30:
                break
            h_try_lo = _slsqp(h_lo, n_lo, alpha, 500, t0, budget_s)
            if h_try_lo is not None:
                h_back = _interp_to_res(h_try_lo, n)
                c_back = _c5_val(h_back)
                if c_back < best_c:
                    best_h = h_back
                    best_c = c_back
        # Also try active search at low res
        if budget_s - (time.time() - t0) > 40:
            h_opt_lo, c_opt_lo = _active_search(h_lo, n_lo, t0,
                                                 min(60, budget_s - (time.time() - t0) - 20), 100)
            h_back = _interp_to_res(h_opt_lo, n)
            c_back = _c5_val(h_back)
            if c_back < best_c:
                best_h = h_back
                best_c = c_back

    # Phase 4: Final polish with very sharp objective
    for alpha in [2000.0, 5000.0]:
        if budget_s - (time.time() - t0) < 15:
            break
        h_try = _slsqp(best_h, n, alpha, 400, t0, budget_s)
        if h_try is not None:
            c_try = _c5_val(h_try)
            if c_try < best_c:
                best_h = h_try
                best_c = c_try

    # Final active search polish
    if budget_s - (time.time() - t0) > 10:
        best_h, best_c = _active_search(best_h, n, t0,
                                         min(30, budget_s - (time.time() - t0) - 2), 100)

    best_h = _project(best_h, n)
    best_c = _c5_val(best_h)
    return best_h, best_c, n
