import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_comp(h, full_idx):
    n = len(h)
    lag = full_idx - (n - 1)
    g = np.zeros(n)
    jj = np.arange(n)
    if lag >= 0:
        m1 = jj >= lag
        m2 = jj <= n - 1 - lag
        g[m1] += 1.0 - h[jj[m1] - lag]
        g[m2] -= h[jj[m2] + lag]
    else:
        k = -lag
        m1 = jj <= n - 1 - k
        m2 = jj >= k
        g[m1] += 1.0 - h[jj[m1] + k]
        g[m2] -= h[jj[m2] - k]
    return g

def smooth_obj_grad(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    if tau <= 1e-15:
        idx = int(np.argmax(c))
        return m, _grad_comp(h, idx) * dx
    y = np.clip((c - m) / tau, -50.0, 50.0)
    ey = np.exp(y)
    se = float(np.sum(ey))
    w = ey / se
    obj = m + tau * np.log(se)
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] < 1e-12:
            continue
        g += w[i] * _grad_comp(h, i)
    return obj, g * dx

def lfbg_step(h, dx, target, mu, tau, maxiter=200):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def fpen(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(c)
        if tau <= 1e-15:
            obj = m
        else:
            y = np.clip((c - m) / tau, -50.0, 50.0)
            ey = np.exp(y)
            se = float(np.sum(ey))
            obj = m + tau * np.log(se)
        s = np.sum(x) - target
        return obj + mu * s * s

    def fpen_grad(x):
        x = np.clip(x, 0.0, 1.0)
        obj, g = smooth_obj_grad(x, dx, tau)
        s = np.sum(x) - target
        g += 2.0 * mu * s
        return g

    r = minimize(fpen, h, jac=fpen_grad, method='L-BFGS-B',
                 bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
    return repair(r.x, target)

def coord_descent(h, dx, target, n_sweeps=5):
    n = len(h)
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_sweeps):
        for j in range(n):
            a, b = 0.0, 1.0
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            h[j] = c1
            fc = compute_c5(h, dx)
            h[j] = d1
            fd = compute_c5(h, dx)
            for _ in range(35):
                if fc < fd:
                    b = d1; d1 = c1; fd = fc
                    c1 = b - gr * (b - a)
                    h[j] = c1; fc = compute_c5(h, dx)
                else:
                    a = c1; c1 = d1; fc = fd
                    d1 = a + gr * (b - a)
                    h[j] = d1; fd = compute_c5(h, dx)
            h[j] = (a + b) * 0.5
            h = np.clip(h, 0.0, 1.0)
        h = repair(h, target)
    return h

def make_diverse_starts(n, target, base, rng):
    starts = []
    starts.append(base.copy())
    for scale in [0.05, 0.1, 0.2, 0.4]:
        starts.append(repair(base + rng.randn(n) * scale, target))
    # Binary random
    for _ in range(6):
        idx = rng.choice(n, size=int(target), replace=False)
        hb = np.zeros(n); hb[idx] = 1.0
        starts.append(hb.copy())
    # Low-freq patterns
    for freq in [1, 2, 3, 5, 7]:
        t = np.linspace(0, 2*np.pi*freq, n)
        hp = 0.5 + 0.45*np.sin(t + rng.randn()*0.5)
        starts.append(repair(hp, target))
    # Block patterns
    for bw in [2, 3, 5, 7]:
        hb = np.zeros(n)
        for i in range(0, n, bw*2):
            hb[i:i+bw] = 1.0
        starts.append(repair(hb, target))
    # Sawtooth
    for per in [4, 6, 10, 14]:
        xs = np.arange(n) / per
        hs = np.clip(1.0 - 2.0*np.abs(xs - np.round(xs)), 0, 1)
        hs = 0.5 + 0.5*(hs - 0.5)
        starts.append(repair(hs, target))
    return starts

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:

            x = np.clip(x, 0, 1)

            hn = repair(r.x, tgt)

                hn = lfbg_step(best_h, dx, tgt, 100000.0, tau, maxiter=300)
