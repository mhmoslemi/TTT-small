```python
def compute_subgrad(h):
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode='full')
    # Normalize to get actual integrals
    dx = 2.0 / n
    c_vals = c_full * dx
    c_max = np.max(c_vals)
    
    # Soft weighting: weight each shift by softmax
    tau = 50.0
    weights = np.exp(tau * (c_vals - c_max))
    weights /= np.sum(weights)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_score_grad(h, tau=50.0):
    """Returns (soft_max_value, gradient)"""
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c_full)
    
    # Soft max value
    e = np.exp(tau * (c_full - c_max))
    sum_e = np.sum(e)
    sm_val = c_max + np.log(sum_e) / tau
    weights = e / sum_e
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n - shift]
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    
    return sm_val, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def subgrad_descent(h, n, t0, time_limit):
    """Adaptive subgradient descent"""
    best_h = h.copy()
    best_s = compute_score(h)
    lr = 0.1
    tau = 20.0
    
    for it in range(20000):
        if time.time() > time_limit:
            break
        
        val, grad = compute_score_grad(h, tau)
        
        # Adaptive step size
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        s_new = compute_score(h_new)
        
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            h = h_new  # still use projected version
            lr *= 0.95
        
        # Adaptively increase tau
        tau = min(tau * 1.002, 500.0)
        
        # Early stopping
        if it > 500 and abs(s_new - best_s) < 1e-10:
            break
    
    return best_h, best_s

def slsqp_refine(h, n, t0, time_limit):
    """SLSQP with soft-max objective"""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        val, _ = compute_score_grad(x, tau=100.0)
        return val + 1000.0 * (np.sum(x) - target) ** 2
    
    def obj_grad(x):
        _, g = compute_score_grad(x, tau=100.0)
        g = g + 2000.0 * (np.sum(x) - target)
        return g
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-14})
        h_opt = project_feasible(res['x'], n)
        s_opt = compute_score(h_opt)
        if s_opt < compute_score(h):
            return h_opt, s_opt
```

```python
    n = len(h)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0
_G_TAU = 80.0

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _sm_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    e = np.exp(tau * (c - cm))
    se = np.sum(e)
    sm = cm + np.log(se) / tau
    w = e / se
    grad = np.zeros(n)
    thresh = 1.0 / (n * n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < thresh:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            grad[:n - sh] += wk * (1.0 - h[sh:])
            grad[sh:] -= wk * h[:n - sh]
        else:
            s = -sh
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
    return sm, grad

def _slsqp_obj(x):
    v, _ = _sm_grad(x, _G_TAU)
    return v + 500.0 * (np.sum(x) - _G_N / 2.0) ** 2

def _slsqp_jac(x):
    _, g = _sm_grad(x, _G_TAU)
    return g + 1000.0 * (np.sum(x) - _G_N / 2.0)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def subgrad_descent(h, n, time_limit):

    lr = 0.08
    tau = 12.0
    no_improve = 0

        if time.time() > time_limit:
            break
        _, grad = _sm_grad(h, tau)
        h = project_feasible(h - lr * grad, n)
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.4)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.92
        tau = min(tau * 1.0015, 400.0)
        if it > 2000 and no_improve > 3000 and tau > 200:
            break
    return best_h, best_s

def slsqp_refine(h, n, time_limit):
    global _G_N, _G_TAU
    _G_N = n
    _G_TAU = 100.0
    bounds = [(0.0, 1.0)] * n
    try:
        if time.time() > time_limit:
            return h, compute_score(h)
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, method='SLSQP',
                       bounds=bounds, options={'maxiter': 200, 'ftol': 1e-14})
        h_opt = project_feasible(res['x'], n)
        s_opt = compute_score(h_opt)
        if s_opt < compute_score(h):
            return h_opt, s_opt
    except Exception:
        pass
    return h, compute_score(h)

def coord_refine(h, n, time_limit):
    h = h.copy()
    best_s = compute_score(h)
    target = n / 2.0
    for it in range(50000):
        if time.time() > time_limit:
            break
        i = it % n
        old = h[i]
        for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
            h[i] = min(1.0, max(0.0, old + delta))

                h[i] = old

            s = compute_score(h)

                best_s = s

def project_feasible_inplace(h, target):

    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
```
