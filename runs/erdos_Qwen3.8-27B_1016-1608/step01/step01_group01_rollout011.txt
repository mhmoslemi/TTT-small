            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def topk_obj(h, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    s = np.sort(c)[::-1]
    top5 = s[:5].sum()
    m = c.max()
    soft = m + temp * np.log(np.sum(np.exp((c - m) / temp)))
    return 0.3 * top5 + 0.7 * soft

def sum_eq(h):
    return np.sum(h) - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

    constraints = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h, args=(temp,),
                  method='SLSQP', bounds=bounds, constraints=constraints,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    if time.time() > t_limit:
        return h, compute_c5(h)
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new)

def coord_descent(h, n, t_limit, best_c5, best_h):
    h = h.copy()
    c5_cur = compute_c5(h)
    if c5_cur > best_c5:
        h = best_h.copy()
        c5_cur = best_c5
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for ss in steps:
        while time.time() < t_limit:
            improved = False
            for i in range(n):
                for d in (ss, -ss):
                    h_t = h.copy()
                    h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
                    h_t = project_feasible(h_t, n)
                    c5_t = compute_c5(h_t)
                    if c5_t < c5_cur - 1e-13:
                        h = h_t
                        c5_cur = c5_t
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        improved = True
                        break
                if time.time() > t_limit:
                    break
            if not improved:
                break
    return best_c5, best_h
