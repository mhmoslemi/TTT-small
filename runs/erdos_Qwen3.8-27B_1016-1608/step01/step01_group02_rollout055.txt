import time

    return float(np.max(corr) * dx)

        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:

            if mask.sum() == 0:
                return h
            s = min(excess / mask.sum(), h[mask].min())
            h[mask] -= s
        else:

            if mask.sum() == 0:
                return h
            s = min(-excess / mask.sum(), (1.0 - h[mask]).min())
            h[mask] += s

def smooth_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)) / len(corr))

def gradient_smooth(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w = w / w.sum()

    n = len(h)
    grad = np.zeros(n)

        if lag >= 0:

                grad[i] += wk * (1.0 - h[i + lag])
                grad[i + lag] -= wk * h[i]
        else:
            l = -lag
            for i in range(n - l):
                grad[i + l] += wk * (1.0 - h[i])
                grad[i] -= wk * h[i + l]

import numpy as np
from scipy.optimize import minimize
import time

_TGT = 0.0

def _eq(x):
    return x.sum() - _TGT

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 0
            if not mask.any():
                return h
            s = min(excess / mask.sum(), h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            if not mask.any():
                return h
            s = min(-excess / mask.sum(), (1.0 - h[mask]).min())
            h[mask] += s
    return h

def smooth_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)) / len(corr))

def slsqp_smooth(h0, dx, temp, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    con = ({'type': 'eq', 'fun': _eq},)
    try:
        res = minimize(smooth_obj, h0, args=(dx, temp), method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return projection(res.x, _TGT)
    except Exception:
        return None

def slsqp_exact(h0, dx, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    con = ({'type': 'eq', 'fun': _eq},)
    try:
        res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return projection(res.x, _TGT)
    except Exception:
        return None

def subgrad_dir(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        l = -lag
        grad[l:] += (1.0 - h[:n - l])
        grad[:n - l] -= h[l:]
    return grad * dx

def subgrad_opt(h0, dx, target, n_iter, step0):
    h = h0.copy()
    c5_cur = compute_c5(h, dx)
    step = step0
    for it in range(n_iter):
        g = subgrad_dir(h, dx)
        h_new = projection(h - step * g, target)
        c_new = compute_c5(h_new, dx)
        if c_new < c5_cur - 1e-15:
            h, c5_cur = h_new, c_new
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
    return h, c5_cur

def pair_cd(h, dx, n_iter, rng):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h, dx)
    deltas = [0.05, -0.05, 0.03, -0.03, 0.01, -0.01, 0.08, -0.08]
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        vi0, vj0 = h[i], h[j]
        best_d, best_c = 0.0, c5_cur
        for d in deltas:
            vi, vj = vi0 + d, vj0 - d
            if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
                h[i], h[j] = vi, vj
                c = compute_c5(h, dx)
                if c < best_c - 1e-15:
                    best_c, best_d = c, d
        if best_d != 0.0:
            h[i], h[j] = vi0 + best_d, vj0 - best_d
            c5_cur = best_c
        else:
            h[i], h[j] = vi0, vj0
    return h, c5_cur

def single_cd(h, dx, target, n_iter, rng):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h, dx)
    steps = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01]
    for it in range(n_iter):
        idx = rng.randint(n)
        best_d, best_c = 0.0, c5_cur
        for d in steps:
            ht = h.copy()
            ht[idx] = np.clip(ht[idx] + d, 0.0, 1.0)
            ht = projection(ht, target)
            c = compute_c5(ht, dx)
            if c < best_c - 1e-15:
                best_c, best_d = c, d
                best_ht = ht.copy()
        if best_d != 0.0:
            h, c5_cur = best_ht, best_c
    return h, c5_cur

    temps = [30.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    for temp in temps:

        if c5_sg < best_c5:
