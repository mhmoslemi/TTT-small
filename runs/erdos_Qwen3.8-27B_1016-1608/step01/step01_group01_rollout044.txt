                    for d in steps:
                        ht = best_h.copy()
                        ht[i] = np.clip(ht[i] + d, 0, 1)
                        ht[j] = np.clip(ht[j] - d, 0, 1)
                        ht = project_feasible(ht, n)
                        c5t = compute_c5(ht)
                        if c5t < best_c5 - 1e-15:
                            best_c5 = c5t
                            best_h = ht
                            improved_any = True
                            break

    # Phase 4: Single-coordinate fine search (15% time)
    if elapsed_frac() < 0.93:
        for d in [0.005, -0.005, 0.002, -0.002, 0.01, -0.01, 0.008, -0.008]:
            if elapsed_frac() > 0.93:
                break
            improved = True
            while improved and elapsed_frac() < 0.93:
                improved = False
                for i in range(n):
                    if elapsed_frac() > 0.93:
                        break
                    ht = best_h.copy()
                    ht[i] = np.clip(ht[i] + d, 0, 1)
                    ht = project_feasible(ht, n)
                    c5t = compute_c5(ht)
                    if c5t < best_c5 - 1e-15:
                        best_c5 = c5t
                        best_h = ht
                        improved = True

    # Phase 5: Final SLSQP polish (7% time)
    if elapsed_frac() < 0.98:
        for temp in [0.0003, 0.0001]:
            if elapsed_frac() > 0.98:
                break
            try:
                res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                             method='SLSQP', bounds=bounds, constraints=constraints,
                             options={'maxiter': 5000, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Final feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
