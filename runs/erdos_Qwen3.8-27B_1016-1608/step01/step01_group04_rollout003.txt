    return float(np.max(corr) * dx)

def soft_c5(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if eps < 1e-12:
        return m
    s = np.clip((corr - m) / eps, -50, 50)

def project(h, n):

        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
        h = np.clip(h, 0, 1)

def _coord_min_1d(a, b, n_pts=200):
    """Minimize max_k(a_k * t + b_k) over t in [0,1]"""
    ts = np.linspace(0, 1, n_pts)
    vals = a[:, None] * ts[None, :] + b[:, None]  # shape (n_lags, n_pts)
    f = np.max(vals, axis=0)  # shape (n_pts,)
    idx = np.argmin(f)
    # Refine around the best point
    if idx == 0:
        return 0.0
    if idx == n_pts - 1:
        return 1.0
    lo = ts[idx-1]
    hi = ts[idx+1]
    # Golden section refinement
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(30):
        t1 = lo + (1-gr) * (hi - lo)
        t2 = lo + gr * (hi - lo)
        f1 = np.max(a * t1 + b)
        f2 = np.max(a * t2 + b)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2

- When updating h[j], evaluate corr at h[j]=0 and h[j]=1
- Choose the one with lower C₅
- This is just greedy: h[j] = argmin_{v∈{0,1}} C₅(h with h[j]=v)

This is even faster (no golden section needed) and might find good solutions.

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _soft_c5(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
