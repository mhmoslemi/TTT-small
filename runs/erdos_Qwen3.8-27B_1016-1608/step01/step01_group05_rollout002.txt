import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old)
    x_new = np.arange(n_new) * (n_old / n_new)
    return np.interp(x_new, x_old, h, period=n_old)

def coord_descent(h, n, n_sweeps=80, rng=None):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    test_vals = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n) if rng is not None else range(n)
        for j in order:
            old = h[j]
            best_v = best
            best_x = old
            for tx in test_vals:
                if abs(tx - old) < 1e-8:
                    continue
                h[j] = tx
                h_p = project_to_set(h, target)
                v = compute_c5(h_p, n)
                if v < best_v:
                    best_v = v
                    best_x = tx
                    h = h_p
            if best_x != old:
                h[j] = best_x
                h = project_to_set(h, target)
                best = best_v
                improved = True
            else:
                h = project_to_set(h, target)
        if not improved and sweep > 3:
            break
    return h, best

def swap_search(h, n, rng, n_trials=3000, temp=0.0):
    target = n / 2.0
    h = h.copy()

        h[i], h[j] = old_j, old_i
        v = compute_c5(h, n)
        if v < best or (temp > 0 and rng.rand() < np.exp(-(v - best) / temp)):
            best = min(best, v)
        else:
            h[i], h[j] = old_i, old_j
    return h, best

def subgrad_steps(h, n, n_steps=500, lr=0.08):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_steps):
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr_raw)
        active = (corr_raw - corr_raw[max_idx]) * (2.0 / n) > -1e-4
        k_vals = np.where(active)[0] - (n - 1)
        grad = np.zeros(n)
        for k in k_vals:
            ak = abs(k)
            if ak == 0:
                grad += (1.0 - 2.0 * h)
            else:
                for j in range(ak, n):
                    grad[j] += (1.0 - h[j - ak])
                for j in range(n - ak):
                    grad[j] += -h[j + ak]
        grad *= (2.0 / n)
        step = lr / (1.0 + 0.005 * it)
        h_new = project_to_set(h - step * grad, target)
        v = compute_c5(h_new, n)
        if v < best:
            best = v
            best_h = h_new.copy()
        h = h_new
    return best_h, best
