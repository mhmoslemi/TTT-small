            h_best, c5_best = h, c

    # Phase 7: Try nearby resolutions
    if time.time() - t0 < budget_s * 0.99:
        for delta in [10, -10, 20, -20]:
            n2 = n + delta
            if n2 < 30 or n2 > 250:
                continue
            if time.time() - t0 > budget_s * 0.98:
                break
            xp = np.linspace(0.0, 2.0, n, endpoint=False)
            xn = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            h2 = slsqp_opt(h_interp, n2, maxiter=800)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
