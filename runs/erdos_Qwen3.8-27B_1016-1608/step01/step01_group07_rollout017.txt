    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _project(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    ac = np.clip(alpha * (c - m), -100.0, 100.0)
    e = np.exp(ac)
    se = np.sum(e)
    w = e / se
    val = m + np.log(se) / alpha
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        shift = k - (n - 1)
        lo_a = max(0, -shift)
        hi_a = min(n, n - shift)
        if lo_a < hi_a:
            A[lo_a:hi_a] += wk * (1.0 - h[lo_a + shift:hi_a + shift])
        lo_b = max(0, shift)
        hi_b = min(n, n + shift)
        if lo_b < hi_b:
            B[lo_b:hi_b] += wk * h[lo_b - shift:hi_b - shift]
    return val, dx * (A - B)

def _slsqp_obj(x, alpha):
    v, _ = _smooth_obj_and_grad(x, alpha)
    return v

def _slsqp_jac(x, alpha):
    _, g = _smooth_obj_and_grad(x, alpha)
    return g

def _con_f(x):
    return np.sum(x) - 0.5 * len(x)

def _con_j(x):
    return np.ones(len(x))

def _try_slsqp(h0, alpha, maxiter, t0, budget):
    n = len(h0)
    if budget - (time.time() - t0) < 15:
        return None
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_f, 'jac': _con_j}]
    try:
        res = minimize(_slsqp_obj, h0, (alpha,), jac=_slsqp_jac,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _try_fista(h0, alpha, niter, lr0, t0, budget):
    n = len(h0)
    target = 0.5 * n
    x = h0.copy()
    x_prev = h0.copy()
    t_val = 1.0
    lr = lr0
    best_local = h0.copy()
    best_c = _c5(h0)
    for i in range(niter):
        if budget - (time.time() - t0) < 3:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mu = (t_val - 1.0) / t_new
        y = x + mu * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        _, grad = _smooth_obj_and_grad(y, alpha)
        x_new = _project(y - lr * grad, target)
        cx = _c5(x_new)
        if cx < best_c:
            best_c = cx
            best_local = x_new.copy()
        x_prev = x
        x = x_new
        t_val = t_new
        lr = lr0 / (1.0 + i * 0.01)
    return best_local, best_c

def _interp(h, n_new):
    n0 = len(h)
    dx0 = 2.0 / n0
    xo = (np.arange(n0) + 0.5) * dx0
    dxn = 2.0 / n_new
    xn = (np.arange(n_new) + 0.5) * dxn
    h_new = np.interp(xn, xo, h)
    return _fix(np.clip(h_new, 0.0, 1.0), n_new)

def _binary_push(h, n, strength):
    s = np.sort(h)
    mid = s[n // 2]
    h_push = np.clip(1.0 - np.exp(-strength * (h - mid)), 0.0, 1.0)
    return _fix(h_push, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(45, 0.5)
    n = len(h)
    h = _fix(h, n)
    best_h = h.copy()
    best_c = _c5(best_h)

    # Phase 1: SLSQP with analytical gradient at original resolution
    for alpha in [60.0, 150.0, 400.0, 1000.0, 2500.0]:
        if budget_s - (time.time() - t0) < 70:
            break
        for r in range(3):
            if budget_s - (time.time() - t0) < 35:
                break
            if r == 0:
                h0 = best_h.copy()
            elif r == 1:
                h0 = _fix(np.clip(best_h + rng.randn(n) * 0.06, 0, 1), n)
            else:
                h0 = _binary_push(best_h, n, 3.0)
            ht = _try_slsqp(h0, alpha, 2000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < best_c:
                    best_h, best_c = ht, ct

    # Phase 2: FISTA projected gradient
    if budget_s - (time.time() - t0) > 150:
        for alpha in [100.0, 400.0, 1000.0]:
            if budget_s - (time.time() - t0) < 50:
                break
            for lr0 in [0.5, 0.2]:
                if budget_s - (time.time() - t0) < 30:
                    break
                ht, ct = _try_fista(best_h, alpha, 3000, lr0, t0, budget_s)
                if ct < best_c:
                    best_h, best_c = ht, ct

    # Phase 3: Lower resolution exploration
    if budget_s - (time.time() - t0) > 120:
        for n_low in [50, 75, 100]:
            if budget_s - (time.time() - t0) < 50:
                break
            n_cur = len(best_h)
            h_low = _interp(best_h, n_low)
            c_low = _c5(h_low)
            for alpha in [100.0, 300.0, 800.0]:
                if budget_s - (time.time() - t0) < 25:
                    break
                ht = _try_slsqp(h_low, alpha, 1500, t0, budget_s)
                if ht is not None:
                    ct = _c5(ht)
                    if ct < c_low:
                        h_low, c_low = ht, ct
            h_back = _interp(h_low, n_cur)
            c_back = _c5(h_back)
            if c_back < best_c:
                best_h, best_c = h_back, c_back
            ht = _try_slsqp(best_h, 500.0, 1000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < best_c:
                    best_h, best_c = ht, ct

    # Phase 4: Perturbation and restart
    if budget_s - (time.time() - t0) > 60:
        for r in range(5):
            if budget_s - (time.time() - t0) < 20:
                break
            scale = 0.03 + 0.02 * r
            h_pert = _fix(np.clip(best_h + rng.randn(n) * scale, 0, 1), n)
            for alpha in [300.0, 800.0]:
                if budget_s - (time.time() - t0) < 15:
                    break
                ht = _try_slsqp(h_pert, alpha, 800, t0, budget_s)
                if ht is not None:
                    ct = _c5(ht)
                    if ct < best_c:
                        best_h, best_c = ht, ct
                        h_pert = ht.copy()

    # Phase 5: Binary push + refine
    if budget_s - (time.time() - t0) > 40:
        for st in [4.0, 6.0, 8.0]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_b = _binary_push(best_h, n, st)
            hb_c = _c5(h_b)
            if hb_c < best_c:
                best_h, best_c = h_b, hb_c
            ht = _try_slsqp(best_h, 800.0, 600, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < best_c:
                    best_h, best_c = ht, ct

    # Phase 6: Final high-alpha polish
    if budget_s - (time.time() - t0) > 15:
        ht = _try_slsqp(best_h, 5000.0, 1200, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < best_c:
                best_h, best_c = ht, ct

    best_h = _fix(best_h, len(best_h))
    best_c = _c5(best_h)
    return best_h, best_c, len(best_h)
