import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def _compute_c5_linear(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _compute_c5_circ(h, n):
    dx = 2.0 / n
    omm = 1.0 - h
    corr = np.real(ifft(fft(h) * np.conj(fft(omm))))
    return float(np.max(corr) * dx), corr

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def _make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return _project(h + rng.normal(0, 0.1, n), n / 2.0)

def _make_smooth(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 * np.ones(n)
    for _ in range(rng.randint(2, 5)):
        f = rng.randint(1, 6)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.1, 0.4)
        h += a * np.sin(f * np.pi * x + p)
    return _project(h, n / 2.0)

def _pg_opt_circ(h, n, n_iter, t_limit, beta_lo=20.0, beta_hi=300.0):
    tgt = n / 2.0
    h = _project(h.copy(), tgt)
    best_h = h.copy()
    best_v = _compute_c5_circ(h, n)[0]

    for it in range(n_iter):
        if time.time() > t_limit:
            break
        frac = it / max(n_iter - 1, 1)
        beta = beta_lo + (beta_hi - beta_lo) * frac * beta
        lr = 0.8 / (1.0 + 0.05 * it)
        omm = 1.0 - h
        corr = np.real(ifft(fft(h) * np.conj(fft(omm))))
        cm = np.max(corr)
        eb = np.exp(beta * (corr - cm))
        w = eb / eb.sum()
        grad = np.zeros(n)
        for k in range(n):
            if w[k] > 1e-9:
                grad += w[k] * ((1.0 - np.roll(h, -k)) + np.roll(h, k))
        h_new = _project(h - lr * grad, tgt)
        v, _ = _compute_c5_circ(h_new, n)
        if v < best_v:
            best_v = v
            best_h = h_new.copy()
        h = h_new
    return best_h, best_v

def _coord_descend(h, n, rng, n_sweeps=10):
    tgt = n / 2.0
    h = h.copy()
    best = _compute_c5_linear(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        imp = False
        for j in order:
            old = h[j]
            cands = np.array([0.0, 1.0,
                              max(0.0, old - 0.15), min(1.0, old + 0.15),
                              max(0.0, old - 0.05), min(1.0, old + 0.05),
                              max(0.0, old - 0.02), min(1.0, old + 0.02)])
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                hp = _project(h, tgt)
                v = _compute_c5_linear(hp, n)
                if v < bv:
                    bv, bc = v, float(c)
            if bc != old:
                h[j] = bc
                h = _project(h, tgt)
                best = bv
                imp = True
            else:
                h[j] = old
                h = _project(h, tgt)
        if not imp:
            break
    return h, best

def _pair_swap(h, n, rng, n_iters=500):
    tgt = n / 2.0
    h = h.copy()
    best = _compute_c5_linear(h, n)
    for _ in range(n_iters):
        i, j = rng.choice(n, 2, replace=False)
        if abs(h[i] - h[j]) < 1e-6:
            continue
        h[i], h[j] = h[j], h[i]
        hp = _project(h, tgt)
        v = _compute_c5_linear(hp, n)
        if v < best:
            best = v
            h = hp
        else:
            h[i], h[j] = h[j], h[i]
            h = _project(h, tgt)
    return h, best

def _slsqp_local(h, n, maxiter=200, t_limit=None):
    tgt = n / 2.0
    h0 = _project(h.copy(), tgt)
    def obj(x):
        hp = _project(x, tgt)
        return _compute_c5_linear(hp, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        ho = _project(res.x, tgt)
        return ho, _compute_c5_linear(ho, n)
    except Exception:
        return h0, _compute_c5_linear(h0, n)
