                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 8000, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Many L-BFGS-B restarts from perturbed points
    for r in range(40):
        if elapsed() > budget_s * 0.55:
            break
        scale = 0.03 + rng.random() * 0.15
        h_rand = best_h + rng.normal(0, scale, n)
        h_rand = project_feasible(h_rand, n)
        temp = 0.003 + rng.random() * 0.01
        mu = 5000 + int(rng.random() * 20000)
        res = minimize(penalized_soft_obj, h_rand, args=(temp, mu, n),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Discrete pair-swap search
    if elapsed() < budget_s * 0.72:
        h_disc, c5_disc = discrete_pair_search(best_h, n, max_iters=5000)
        if c5_disc < best_c5:
            best_c5 = c5_disc
            best_h = h_disc.copy()
        # Polish after discrete moves
        res = minimize(penalized_soft_obj, best_h.copy(), args=(0.001, 50000, n),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Try a few different structures (rotations, reflections)
    if elapsed() < budget_s * 0.82:
        candidates = [best_h.copy()]
        candidates.append(best_h[::-1].copy())
        for sh in [1, 2, 3, 5, 7, 10]:
            candidates.append(np.roll(best_h, sh).copy())
            candidates.append(np.roll(best_h, -sh).copy())
        # A few random structured inits
        for _ in range(5):
            t = np.arange(n)
            freq = rng.uniform(2, 12)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 2.0)
            h_struct = 0.5 + amp * np.sin(2 * np.pi * freq * t / n + phase)
            h_struct = project_feasible(h_struct, n)
            candidates.append(h_struct)
        for h_cand in candidates:
            if elapsed() > budget_s * 0.82:
                break
            h_cand = project_feasible(h_cand, n)
            res = minimize(penalized_soft_obj, h_cand, args=(0.002, 50000, n),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Coordinate descent polish
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = coordinate_descent(best_h, n, max_passes=30)

    # Phase 6: Final L-BFGS-B polish with very low temp
    if elapsed() < budget_s * 0.97:
        res = minimize(penalized_soft_obj, best_h.copy(), args=(0.0001, 100000, n),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 10000, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
