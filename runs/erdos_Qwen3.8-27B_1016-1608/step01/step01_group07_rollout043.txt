import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, n):
    target = 0.5 * n
    for _ in range(100):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = h + (d / cnt) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_descend(h, n_sweeps=5, n_cands=25):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, 'full') * dx
    L_vals = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        for j in range(n):
            aj = h[j]
            mask1 = (L_vals >= j-(n-1)) & (L_vals <= j)
            mask2 = (L_vals >= -j) & (L_vals <= n-1-j)
            
            coeff = np.zeros(2*n-1)
            if np.any(mask1):
                coeff[mask1] = 1.0 - h[j - L_vals[mask1]]
            if np.any(mask2):
                coeff[mask2] -= h[j + L_vals[mask2]]
            coeff[n-1] = 0.0
            
            best_v = aj
            best_obj = r.max()
            
            for v in np.linspace(0, 1, n_cands):
                d = v - aj
                if abs(d) < 1e-10:
                    continue
                new_r = r + d * coeff
                new_r[n-1] = r[n-1] + (v*(1-v) - aj*(1-aj))
                obj = new_r.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_v = v

                d = best_v - aj
                h[j] = best_v
                r = r + d * coeff
                r[n-1] += (best_v*(1-best_v) - aj*(1-aj))
        
        h = _fix_sum(h, n)
        r = np.correlate(h, 1.0 - h, 'full') * dx

    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    xn = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    return np.clip(np.interp(xn, xo, h), 0.0, 1.0)

    def _obj(x):

        m = np.max(c)
        e = np.exp(np.clip(c - m, -50, 50))
        s = np.sum(e)
        return m + np.log(s) / alpha
    
    def _obj_grad(x):
        c = np.correlate(x, 1.0 - x, 'full') * dx
        m = np.max(c)
        w = np.exp(np.clip(c - m, -50, 50))

        # dc_k/dh_j = dx * [(1-h[j-k]) (if j-k valid) - h[j+k] (if j+k valid)] for k≠0

    dx = 2.0/n

    # Softmax weights (shifted for numerical stability)

    # First part: sum_L p[n-1+L] * (1-h[j-L]) for j-L valid

        # j+L valid: 0 <= j+L < n, i.e., -j <= L <= n-1-j
        L = np.arange(-(n-1), n)

        m2 = (L >= -j) & (L <= n-1-j) & (L != 0)

        if np.any(m2):
            s -= np.dot(p[n-1+L[m2]], h[j+L[m2]])

    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, n):
    target = 0.5 * n
    for _ in range(100):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = h + (d / cnt) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_descend(h, n_sweeps=5, n_cands=30):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, 'full') * dx
    L_vals = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        for j in range(n):
            aj = h[j]
            m1 = (L_vals >= j - (n - 1)) & (L_vals <= j) & (L_vals != 0)
            m2 = (L_vals >= -j) & (L_vals <= n - 1 - j) & (L_vals != 0)
            coeff = np.zeros(2 * n - 1)
            if np.any(m1):
                coeff[m1] = 1.0 - h[j - L_vals[m1]]
            if np.any(m2):
                coeff[m2] -= h[j + L_vals[m2]]
            best_v = aj
            best_obj = r.max()
            grid = np.linspace(0, 1, n_cands)
            for v in grid:
                d = v - aj
                if abs(d) < 1e-11:
                    continue
                nr0 = r + d * coeff
                nr0[n - 1] = r[n - 1] + (v * (1 - v) - aj * (1 - aj))
                obj = nr0.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_v = v
            if best_v != aj:
                d = best_v - aj
                h[j] = best_v
                r += d * coeff
                r[n - 1] += (best_v * (1 - best_v) - aj * (1 - aj))
        h = _fix_sum(h, n)
        r = np.correlate(h, 1.0 - h, 'full') * dx
    return h

def _slsqp_obj(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, 'full') * dx
    m = c.max()
    e = np.exp(np.clip(c - m, -50.0, 50.0))
    return m + np.log(e.sum()) / alpha

def _slsqp_obj_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, 'full') * dx
    cs = c - c.max()
    p = np.exp(alpha * cs)
    p = p / p.sum()
    grad = dx * p[n - 1] * (1.0 - 2.0 * x)
    L_vals = np.arange(-(n - 1), n)
    for j in range(n):
        m1 = (L_vals >= j - (n - 1)) & (L_vals <= j) & (L_vals != 0)
        m2 = (L_vals >= -j) & (L_vals <= n - 1 - j) & (L_vals != 0)
        s = 0.0
        if np.any(m1):
            s += np.dot(p[n - 1 + L_vals[m1]], 1.0 - x[j - L_vals[m1]])
        if np.any(m2):
            s -= np.dot(p[n - 1 + L_vals[m2]], x[j + L_vals[m2]])
        grad[j] += dx * s
    return grad

def _slsqp_run(h, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 8:
        return None
    bnds = [(0.0, 1.0)] * n
    c_fun = lambda x: np.sum(x) - 0.5 * n
    c_jac = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(_slsqp_obj, h, (alpha, n), jac=lambda x, a, n2: _slsqp_obj_grad(x, a, n2),
                       args=(alpha, n), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _interp_to(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.arange(n_new) * (2.0 / n_new)
    return np.clip(np.interp(xn, xo, h), 0.0, 1.0)

def _random_struct(n, rng):
    p = rng.choice([0.25, 0.333, 0.5, 0.667])
    h = np.zeros(n)
    period = int(n * p)
    duty = max(1, period // 2)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    h += rng.randn(n) * 0.3
    return _fix_sum(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.full(400, 0.5)
    n = len(bh)
    bh = _fix_sum(bh, n)
    bc = _c5(bh)

    # Phase 1: Coordinate descent
    if budget_s - (time.time() - t0) > 10:

        h = _fix_sum(h, n)
        c = _c5(h)
        if c < bc:
            bh, bc = h, c

    for alpha in [40.0, 120.0, 400.0, 1500.0]:
        if budget_s - (time.time() - t0) < 30:

        ht = _slsqp_run(bh.copy(), n, alpha, 4000, t0, budget_s)
        if ht is not None:

            h2 = _coord_descend(bh.copy(), n_sweeps=5, n_cands=30)
            h2 = _fix_sum(h2, n)
            c2 = _c5(h2)
            if c2 < bc:
                bh, bc = h2, c2

    # Phase 3: Random perturbations + CD

        hp = _fix_sum(hp, n)

        if cp < bc:
            bh, bc = hp, cp

    # Phase 4: Try structured starts

            hs = _fix_sum(hs, n)

    if budget_s - (time.time() - t0) > 120 and n < 600:

        h2 = _interp_to(bh, n2)

        h2 = _fix_sum(h2, n2)
        c2 = _c5(h2)
        if c2 < bc:
            bh, bc = h2, c2

    bh = _coord_descend(bh, n_sweeps=3, n_cands=35)
    bh = _fix_sum(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
