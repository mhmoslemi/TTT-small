def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def make_binary_top(scores, n):
    h = np.zeros(n)
    k = n // 2
    if 0 < k < n:
        idx = np.argpartition(scores, -k)[-k:]
        h[idx] = 1.0
    return h

def swap_search(h, n, rng, n_sweeps=50, n_tries=300):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bi, bj, bv = -1, -1, best
        nt = min(n_tries, len(ones) * len(zeros))
        for _ in range(nt):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            v = compute_c5(h, n)
            h[i], h[j] = h[j], h[i]
            if v < bv:
                bv = v
                bi, bj = i, j
        if bi >= 0:
            h[bi], h[bj] = h[bj], h[bi]
            best = bv
        else:
            break
    return h, best

    h = h.copy()

    best = compute_c5(h, n)

        order = rng.permutation(n)

            old = h[j]
            cands = np.unique(np.concatenate([[old, 0.0, 1.0], np.linspace(0, 1, n_cands)]))
            bc, bv = old, best
            h_saved = h.copy()
            for c in cands:
                if abs(c - old) < 1e-10:
                    continue
                h[j] = c
                ht = project_to_set(h, target)
                v = compute_c5(ht, n)
                if v < bv:
                    bv = v
                    bc = c
                    h = ht.copy()
            if abs(bc - old) > 1e-10:
                best = bv
                improved = True
            else:
                h = h_saved
        if not improved:
            break
