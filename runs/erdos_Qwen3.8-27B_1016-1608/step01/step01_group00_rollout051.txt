    lo, hi = x.min() - 2.0, x.max() + 2.0

        else:

    if abs(diff) > 0:

        if mask.sum() > 0:

        else:

def c5_smooth(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / beta

def adam_optimize(h0, n, lr=0.01, max_steps=5000, beta_smooth=50.0, t0_check=None, budget_frac=0.5):
    """Adam optimizer with projection."""
    target_sum = n / 2.0
    h = h0.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    best_h = h.copy()
    best_c = compute_c5(h0)
    
    for step in range(max_steps):
        if t0_check is not None:
            if time.time() - t0_check > budget_frac:
                break
        # Numerical gradient of smooth C5
        eps = 1e-5
        grad = np.zeros(n)
        base = c5_smooth(h, beta_smooth)
        for i in range(n):
            hp = h.copy(); hp[i] += eps
            hm = h.copy(); hm[i] -= eps
            grad[i] = (c5_smooth(hp, beta_smooth) - c5_smooth(hm, beta_smooth)) / (2*eps)
        
        # Project gradient onto tangent space of sum constraint
        grad -= np.mean(grad)
        
        # Adam update
        step_num = step + 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        m_hat = m / (1 - 0.9**step_num)
        v_hat = v / (1 - 0.999**step_num)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h_new, target_sum)
        
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

Wait, C5 as a function of h[j] (with all other h fixed) is:
max_k [dx * (sum of linear parts) + dx * (terms involving h[j])]

The terms involving h[j] in f_k are:
- If j is in the first position (i+k-n+1 = j): h[j] * (1-h[i]) = h[j] * (1-h[j-k+n-1])
- If j is in the second position (i = j): h[i+k-n+1] * (1-h[j]) = h[j+k-n+1] * (1-h[j])

So for each k, the contribution to f_k from h[j] is:
- (1-h[j-k+n+1]) * h[j] if j-k+n+1 is valid (let's call this a_k * h[j])
- -h[j] * h[j+k-n+1] if j+k-n+1 is valid (let's call this -b_k * h[j])

Plus a constant term (independent of h[j]).

So f_k as a function of h[j] is: a_k * h[j] - b_k * h[j] + const = (a_k - b_k) * h[j] + const

Wait, that's LINEAR in h[j]! Because h[j] appears at most linearly in each f_k.

So C5(h) = max_k [(c_k) * h[j] + d_k] where c_k = (1-h[j-k+n-1]) - h[j+k-n-1] (with boundary adjustments) and d_k is the rest.

This is the max of linear functions of h[j], which is a convex piecewise-linear function of h[j]. The minimum over h[j] ∈ [0,1] (subject to the sum constraint) is at a vertex (0, 1, or the kink point).

This means coordinate descent would be exact in O(n * n_corr) per step! And it would converge in finite steps (for the discrete case).

But the sum constraint complicates things: if we change h[j], we need to adjust another h[l] to maintain the sum. So it's a 2D subproblem (adjusting h[j] and h[l] simultaneously).

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, x.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(x - t, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr * beta - beta * m))) / beta

def numerical_grad(h, beta, eps=1e-5):
    n = len(h)
    grad = np.zeros(n)
    base = c5_smooth(h, beta)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(hp[i] + eps, 1.0)
        hm = h.copy()
        hm[i] = max(hm[i] - eps, 0.0)
        if hp[i] == hm[i]:
            grad[i] = 0.0
        else:
            grad[i] = (c5_smooth(hp, beta) - c5_smooth(hm, beta)) / (hp[i] - hm[i])
    return grad

def adam_optimize(h0, n, t0, time_end, lr=0.005, beta=30.0, max_steps=100000):

    h = project_feasible(h0.copy(), target_sum)

        if time.time() > time_end:
            break
        if step % 500 == 0:
            eps = max(1e-6, 1e-4 * (0.95 ** (step // 500)))
        else:
            eps = eps if step > 0 else 1e-4
        if step == 0:
            eps = 1e-4
            
        grad = numerical_grad(h, beta, eps)
        # Project to tangent space
        grad -= np.mean(grad)

        h = project_feasible(h_new, target_sum)
        
        c = compute_c5(h)
        if c < best_c - 1e-15:

            best_h = h.copy()

    lo, hi = -3.0, x.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(x - t, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_smooth_vals(corr_arr, beta):
    m = np.max(corr_arr)
    return m + np.log(np.sum(np.exp(corr_arr * beta - beta * m))) / beta

def numerical_grad(h, beta, eps):
    n = len(h)
    dx = 2.0 / n
    base_corr = np.correlate(h, 1.0 - h, mode='full') * dx
    base = c5_smooth_vals(base_corr, beta)
    max_c = np.max(base_corr)
    weights = np.exp(base_corr * beta - beta * max_c)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        old = hp[i]
        hp[i] = min(old + eps, 1.0)
        if hp[i] == old:
            grad[i] = 0.0
            continue
        hm_i = max(old - eps, 0.0)
        hp_corr = np.correlate(hp, 1.0 - hp, mode='full') * dx
        if hm_i != old:
            hp2 = h.copy()
            hp2[i] = hm_i
            hm_corr = np.correlate(hp2, 1.0 - hp2, mode='full') * dx
            dv = (c5_smooth_vals(hp_corr, beta) - c5_smooth_vals(hm_corr, beta)) / (hp[i] - hm_i)
        else:
            dv = (c5_smooth_vals(hp_corr, beta) - base) / (hp[i] - old)
        grad[i] = dv
    return grad

        v = 0.999 * v + 0.001 * grad * grad

        c = compute_c5(h)
        if c < best_c - 1e-15:
