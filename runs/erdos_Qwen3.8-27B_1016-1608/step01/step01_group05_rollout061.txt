```python
def binary_sa(h, n, rng, n_iter=1000000, T0=0.01, T1=0.0001):
    """Simulated annealing on binary sequences using swap moves."""
    h = h.copy().astype(float)
    # Force binary
    # Sort and set top n/2 to 1
    idx = np.argsort(-h)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        T = T0 * (T1/T0) ** (it / n_iter)
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        
        oi, oz = ones[i], zeros[j]
        h[oi] = 0
        h[oz] = 1
        
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1
            h[oz] = 0
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0,2,n_new), np.linspace(0,2,n_old), h)

def grad_corr(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        g[-s:] += 1.0 - h[:n+s]
        g[:n+s] -= h[-s:]
    return g

def subgrad_optimize(h0, n, budget, rng, verbose=False):
    h = project_to_set(h0.astype(float).copy(), n/2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mu = np.zeros(n)
    beta = 0.9
    alpha = 0.08
    alpha0 = 0.08
    patience = 0
    start = time.time()
    while time.time() - start < budget:
        corr = np.correlate(h, 1-h, mode="full")
        cmax = np.max(corr)
        dx = 2.0/n
        # Top-k weighted gradient
        k = max(3, len(corr)//15)
        idx = np.argsort(corr)[-k:]
        w = np.exp((corr[idx]-cmax)*30.0)
        w /= w.sum()
        g = np.zeros(n)
        for i in range(k):
            s = int(idx[i]) - (n-1)
            g += w[i]*grad_corr(h, s, n)
        mu = beta*mu - alpha*g
        h_new = project_to_set(h + mu, n/2.0)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-10:
            best_h = h_new.copy()
            best_c = c
            h = h_new
            patience = 0
            if c < best_c * 0.999:  # big improvement
                alpha = min(alpha0, alpha*1.1)
                beta = min(0.95, beta*1.005)
        else:
            patience += 1
            if patience % 100 == 0:
                alpha *= 0.95
            if patience > 5000:
                alpha = alpha0
                beta = 0.9
                mu = np.zeros(n)
                patience = 0
    return best_h, best_c

def slsqp_optimize(h0, n, maxiter=300):
    target = n/2.0
    h0 = project_to_set(h0.copy(), target)
    
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        return float(np.max(corr) * 2.0/n)
    
    def con(x):
        return float(np.sum(x) - target)
    
    bounds = [(0.0, 1.0)]*n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type':'eq','fun':con}],
                   options={'maxiter':maxiter, 'ftol':1e-12})
    h_out = project_to_set(res.x, target)
    return h_out, compute_c5(h_out, n)

def coord_descent(h, n, rng, n_sweeps=30):
    target = n/2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            cands = [0.0, 1.0, old-0.1, old+0.1, old-0.05, old+0.05]
            cands = [max(0.0, min(1.0, c)) for c in cands]
            bc, bv = old, best
            for c in cands:
                if abs(c-old) < 1e-8: continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv = v
                    bc = c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def grad_corr(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        g[-s:] += 1.0 - h[:n + s]
        g[:n + s] -= h[-s:]
    return g

def subgrad_optimize(h0, n, budget, rng):
    h = project_to_set(h0.astype(float).copy(), n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mu = np.zeros(n)
    beta = 0.88
    alpha = 0.06
    alpha0 = 0.06
    patience = 0
    t0 = time.time()
    while time.time() - t0 < budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        cmax = np.max(corr)
        k = max(5, len(corr) // 12)
        idx = np.argsort(corr)[-k:]
        w = np.exp((corr[idx] - cmax) * 40.0)
        w /= w.sum()
        g = np.zeros(n)
        for i in range(k):
            s = int(idx[i]) - (n - 1)
            g += w[i] * grad_corr(h, s, n)
        mu = beta * mu - alpha * g
        h_new = project_to_set(h + mu, n / 2.0)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c
            h = h_new
            patience = 0
            if best_c < alpha0 * n:
                alpha = min(alpha0, alpha * 1.05)
                beta = min(0.94, beta * 1.003)
        else:
            patience += 1
            if patience % 200 == 0:
                alpha *= 0.93
            if patience > 8000:
                alpha = alpha0
                beta = 0.88
                mu = np.zeros(n)
                patience = 0
    return best_h, best_c

def slsqp_optimize(h0, n, maxiter=300):
    target = n / 2.0
    h0 = project_to_set(h0.astype(float).copy(), target)

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * 2.0 / n)

    def _con(x):
        return float(np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_out = project_to_set(res.x, target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def coord_descent(h, n, rng, n_sweeps=40):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            cands = [0.0, 1.0, old - 0.15, old + 0.15, old - 0.05, old + 0.05, old - 0.02, old + 0.02]
            cands = [max(0.0, min(1.0, c)) for c in cands]
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv = v
                    bc = c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def anneal_continuous(h, n, rng, budget):
    target = n / 2.0
    h = project_to_set(h.astype(float).copy(), target)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    cur_temp = 0.005
    t0 = time.time()
    while time.time() - t0 < budget:
        n_pert = rng.randint(2, 7)
        idx = rng.choice(n, n_pert, replace=False)
        h_save = h.copy()
        for j in idx:
            h[j] = min(1.0, max(0.0, h[j] + rng.normal(0, 0.08)))
        h = project_to_set(h, target)
        c = compute_c5(h, n)
        if c < cur_c or rng.random() < np.exp((cur_c - c) / cur_temp):
            cur_c = c
            if c < best_c:
                best_h = h.copy()
                best_c = c
                cur_temp = max(0.0005, cur_temp * 0.99)
        else:
            h = h_save
    return best_h, best_c
```
