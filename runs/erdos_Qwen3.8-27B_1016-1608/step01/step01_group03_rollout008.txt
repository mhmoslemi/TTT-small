def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    """Compute all overlap values (without dx factor)"""
    n = len(h)
    overlaps = np.zeros(2*n-1)
    for k in range(2*n-1):
        s = k - (n-1)
        i_min = max(0, -s)
        i_max = min(n, n - s)
        if i_max > i_min:
            overlaps[k] = np.sum(h[i_min+s:i_max+s] * (1-h[i_min:i_max]))

            c_new = compute_c5(h_new, dx)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
