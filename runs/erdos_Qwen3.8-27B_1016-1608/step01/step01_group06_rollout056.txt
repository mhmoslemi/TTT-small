def compute_coefs(h, j, n):
    """Compute coefficient of h[j] in each correlate output element."""
    coef = np.zeros(2*n-1)
    # h[j] as first factor h[i], i=j:
    # Shift s = m-n+1: need 0 <= j+s < n, i.e., -j <= s <= n-1-j
    # m = s+n-1: m from n-1-j to n-1+j
    for m in range(max(0, n-1-j), min(2*n-1, n+j)):
        idx2 = j + m - n + 1  # index in (1-h)
        if 0 <= idx2 < n:
            coef[m] += 1.0 - h[idx2]
    # h[j] as second factor h[i+s], i+s=j, i=j-s=j-(m-n+1)=j+m-n+1...

    for m in range(max(0, j), min(2*n-1, j+n)):
        idx1 = j - m + n - 1  # index in h (first factor)
        if 0 <= idx1 < n:
            coef[m] += -h[idx1]
    return coef

import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def soft_max_score(h, tau=800.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def coord_descent(h, n, max_passes=200, tol=1e-12, use_symmetry=True):
    h = h.copy()
    dx = 2.0 / n
    prev_obj = compute_score(h)
    for p in range(max_passes):
        improved = False
        for j in range(n):
            old = h[j]
            # Compute coefficient of h[j] in each correlate output analytically
            c_cur = np.correlate(h, 1.0 - h, mode='full')
            coef = np.zeros(2 * n - 1)
            # h[j] as first factor: m range [n-1-j, n-1+j]
            lo1 = max(0, n - 1 - j)
            hi1 = min(2 * n - 1, n + j)
            for m in range(lo1, hi1):
                idx2 = j + m - n + 1
                if 0 <= idx2 < n:
                    coef[m] += 1.0 - h[idx2]
            # h[j] as second factor h[i+s]: i = j - m + n - 1
            lo2 = max(0, j)
            hi2 = min(2 * n - 1, j + n)
            for m in range(lo2, hi2):
                idx1 = j - m + n - 1
                if 0 <= idx1 < n:
                    coef[m] += -h[idx1]
            base = c_cur - old * coef
            # Minimize max(base + coef*t) for t in [0,1] via ternary search
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                o1 = np.max(base + coef * m1)
                o2 = np.max(base + coef * m2)
                if o1 <= o2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            if abs(t_opt - old) > 1e-12:
                h[j] = t_opt
                improved = True
        if use_symmetry:
            h = (h + h[::-1]) / 2.0
            h = project_feasible(h, n)
        else:
            h = project_feasible(h, n)
        obj = compute_score(h)
        if obj < prev_obj - tol:
            prev_obj = obj
        elif not improved and obj >= prev_obj - 1e-10:
            break
        prev_obj = obj
    return h

def _constrained_obj(h, n, tau):
    target = n / 2.0
    return soft_max_score(h, tau) + 200.0 * (np.sum(h) - target) ** 2
