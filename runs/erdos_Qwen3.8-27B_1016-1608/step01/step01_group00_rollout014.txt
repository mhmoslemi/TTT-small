            h_best, c5_best = h_cd, c

    # Phase 7: Final SLSQP + CD
    if time.time() - t0 < budget_s * 0.995:
        h_sl = slsqp_opt(h_best.copy(), n, 800)
        c = compute_c5(h_sl)
        if c < c5_best:
            h_best, c5_best = h_sl, c
        h_cd = coord_descent(h_best.copy(), max_sweeps=150)
        c = compute_c5(h_cd)
        if c < c5_best:
            h_best, c5_best = h_cd, c

    h_best = repair(h_best)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
