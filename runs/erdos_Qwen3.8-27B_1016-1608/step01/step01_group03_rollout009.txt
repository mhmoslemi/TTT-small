                hf = repair(r.x, _G_TGT)
                cf = compute_c5(hf, _G_DX)
                if cf < best_c5:
                    best_h, best_c5 = hf, cf
            except Exception:
                break

    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
