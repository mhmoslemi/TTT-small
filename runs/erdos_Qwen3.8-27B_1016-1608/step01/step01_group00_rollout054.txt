    """Gradient of sum of c[p] for p in active_idx, with respect to h."""
    g = 1.0 - h
    grad = np.zeros(n)
    for p in active_idx:
        q = p - (n - 1)  # shift
        for m in range(n):
            term = 0.0
            i1 = m - q
            if 0 <= i1 < n:
                term += g[i1]  # = 1 - h[i1]
            i2 = m + q
            if 0 <= i2 < n:
                term -= h[i2]
            grad[m] += term
    return grad

    """Subgradient descent with annealing."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    T = t_end - t_start
    while time.time() < t_end:
        elapsed = time.time() - t_start
        frac = min(elapsed / T, 1.0)
        alpha = beta_start * (beta_end / beta_start) ** frac  # geometric annealing

        # Step
        h_new = h - alpha * grad
        h = project_onto_simplex_box(h_new, n)
        
        c_now = compute_c5(h)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()

        i = oi[np.random.randint(len(oi))]
        j = zi[np.random.randint(len(zi))]
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c_new = compute_c5(h_cur)
        delta = c_new - cur_c
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_cur.copy()
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
    return best_h, best_c

def slsqp_polish(h0, n, maxiter=400, ftol=1e-12):
    def obj(x):
        return compute_c5(x)
    def con(x):
        return np.sum(x) - n / 2.0
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < obj(h0) + 1e-14:
            return project(res.x, n), float(res.fun)
    except Exception:
        pass
    return h0.copy(), compute_c5(h0)

def cyb_polish(h0, n, maxiter=3000):
    def obj(x):
        return compute_c5(x)
    try:
        res = minimize(obj, h0, method='COBYLA',
                       constraints=[lambda x: np.sum(x) - n / 2.0],
                       options={'maxiter': maxiter, 'rhobeg': 0.05, 'tol': 1e-12})
        if res.fun < obj(h0) + 1e-14:
            return project(res.x, n), float(res.fun)
    except Exception:
        pass
    return h0.copy(), compute_c5(h0)

def thue_morse_init(n):
    t = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        t[i] = 1.0 if cnt % 2 == 1 else 0.0
    t = project(t, n)
    return t

def golden_init(n):
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Subgradient with analytical gradient (35% time)
    t1 = t0 + budget_s * 0.35
    if time.time() < t1:
        h, c = subgradient_phase(h_best, n, t0, t1, top_k_max=8)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: Binary simulated annealing (25% time)
    t2 = t0 + budget_s * 0.60
    if time.time() < t2:
        h, c = binary_sa(h_best, n, time.time(), t2)
        if c < c5_best:
            h_best, c5_best = h, c
        # Also try from Thue-Morse and golden ratio starts
        for init_fn in [thue_morse_init, golden_init]:
            if time.time() > t2:
                break
            h_try = init_fn(n)
            h2, c2 = binary_sa(h_try, n, time.time(), min(time.time() + 60, t2))
            if c2 < c5_best:
                h_best, c5_best = h2, c2

    # Phase 3: SLSQP polish + multi-start (20% time)
    t3 = t0 + budget_s * 0.80
    if time.time() < t3:
        h, c = slsqp_polish(h_best, n, maxiter=600)
        if c < c5_best:
            h_best, c5_best = h, c
        scales = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]
        for sc in scales:
            if time.time() > t3:
                break
            h_p = project(h_best + np.random.randn(n) * sc, n)
            h, c = slsqp_polish(h_p, n, maxiter=400)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 4: Multi-resolution (10% time)
    t4 = t0 + budget_s * 0.92
    if time.time() < t4:
        for n2 in [40, 56, 84, 100, 140]:
            if time.time() > t4:
                break
            xp = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
            xn = np.linspace(0.0, 2.0, n2, endpoint=False) + 1.0 / n2
            h_interp = np.interp(xn, xp, h_best)
            h_p = project(h_interp, n2)
            h, c = slsqp_polish(h_p, n2, maxiter=300)
            if c < c5_best:
                h_best, c5_best, n = h, c, n2

    # Phase 5: Final polish (8% time)
    t5 = t0 + budget_s * 0.97
    if time.time() < t5:
        h, c = slsqp_polish(h_best, n, maxiter=1000, ftol=1e-14)
        if c < c5_best:
            h_best, c5_best = h, c
        h, c = cyb_polish(h_best, n, maxiter=2000)
        if c < c5_best:
            h_best, c5_best = h, c
        # One more subgradient burst
        if time.time() < t5:
            h, c = subgradient_phase(h_best, n, time.time(), t5, top_k_max=4)
            if c < c5_best:
                h_best, c5_best = h, c

    h_best = project(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
