def project_to_set(h, target):
    """Project h to {x: 0≤x≤1, sum(x)=target}"""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def coord_descent(h, n, n_sweeps=100, n_grid=31, rng=None, ann_temp=0.0):
    h = h.copy()
    target = n / 2.0
    h = project_to_set(h, target)
    best = compute_c5(h, n)
    elapsed_best = 0
    for sweep in range(n_sweeps):
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        for j in order:
            old_val = h[j]
            h[j] = 0.0
            intercept = np.correlate(h, 1.0 - h, mode="full")
            slope = np.zeros(2 * n - 1)
            slope[j:j + n] = 1.0 - 2.0 * h
            ts = np.linspace(0.0, 1.0, n_grid)
            vals = np.max(intercept[None, :] + slope[:, None] * ts[None, :], axis=1)
            new_val = ts[np.argmin(vals)]
            if ann_temp > 0.01 and rng is not None:
                r = rng.exponential(ann_temp)
                if r < new_val < 1.0:
                    new_val = 1.0
                elif 0.0 < r <= new_val:
                    new_val = 0.0
            h[j] = new_val
        h = project_to_set(h, target)
        cur = compute_c5(h, n)
        if cur < best - 1e-12:
            best = cur
            improved = True
            elapsed_best = 0
        else:
            elapsed_best += 1
            if elapsed_best > max(3, n_sweeps // 10):
                break
    return h, best

def upsample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, n_old), h)

        positions.append(rng.randint(n))
    positions = np.array(positions[:n_ones])
    h[positions] = 1.0
    return project_to_set(h, n / 2.0)
