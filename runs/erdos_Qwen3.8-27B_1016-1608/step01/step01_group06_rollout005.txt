            hc2, sc2 = _coord_desc(hc.copy(), n_pts, iters=25, grid_pts=20)
            hc2 = _project(hc2, n_pts)
            sc2 = _score(hc2)
            if sc2 < sc:
                hc, sc = hc2, sc2

            if sc < local_score:
                local_score = sc
                local_best = hc.copy()
            if sc < best_score:
                best_score = sc
                best_h = hc.copy()

        # COBYLA on local best
        if _el() < budget_s * 0.55:
            try:
                res = minimize(_score, local_best, method='COBYLA',
                              bounds=bounds, options={'maxiter': 400, 'rhobnd': 1e-5})
                hc = _project(res['x'], n_pts)
                sc = _score(hc)
                hc2, sc2 = _coord_desc(hc.copy(), n_pts, iters=30, grid_pts=25)
                hc2 = _project(hc2, n_pts)
                if sc2 < sc:
                    hc, sc = hc2, sc2
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
            except Exception:
                pass

        x_init = np.linspace(0, 2, n_pts, endpoint=False)

    # Phase 2: Aggressive multi-start at small n (24-48)
    if _el() < budget_s * 0.78:
        for n_pts in [24, 30, 36, 42, 48]:
            if _el() > budget_s * 0.75:
                break
            _global_n = n_pts
            bounds = [(0.0, 1.0)] * n_pts
            con = [{'type': 'eq', 'fun': _slsqp_con}]
            n_trials = 15 if n_pts <= 36 else 10
            starts = _gen_starts(n_pts, rng, best_h, n_trials)
            for h_start in starts:
                if _el() > budget_s * 0.75:
                    break
                # Nelder-Mead
                try:
                    simplex = h_start + 0.05 * rng.randn(n_pts)
                    simplex = _project(simplex, n_pts)
                    res = minimize(_score, h_start, method='Nelder-Mead',
                                  options={'maxiter': 3000, 'xatol': 1e-8,
                                           'fatol': 1e-11, 'initial_simplex': simplex})
                    hc = _project(res['x'], n_pts)
                    sc = _score(hc)
                except Exception:
                    hc, sc = h_start, _score(h_start)
                # Coord descent
                hc2, sc2 = _coord_desc(hc.copy(), n_pts, iters=30, grid_pts=25)
                hc2 = _project(hc2, n_pts)
                if sc2 < sc:
                    hc, sc = hc2, sc2
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()

    # Phase 3: Intensive refinement of best
    if _el() < budget_s * 0.95:
        n_final = len(best_h)
        _global_n = n_final
        bounds = [(0.0, 1.0)] * n_final
        con = [{'type': 'eq', 'fun': _slsqp_con}]

        for rnd in range(6):
            if _el() > budget_s * 0.95:
                break
            # SLSQP tight
            try:
                res = minimize(_soft_score, best_h, args=(500.0,), method='SLSQP',
                              bounds=bounds, constraints=con,
                              options={'maxiter': 400, 'ftol': 1e-14})
                hc = _project(res['x'], n_final)
                sc = _score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc
            except Exception:
                pass
            # Coordinate descent (more iterations)
            hc, sc = _coord_desc(best_h.copy(), n_final, iters=60, grid_pts=40)
            hc = _project(hc, n_final)
            sc = _score(hc)
            if sc < best_score:
                best_score = sc
                best_h = hc
            # Small perturbation + coord descent (restart)
            if rnd < 3:
                hp = _project(best_h + rng.randn(n_final) * 0.02, n_final)
                hc, sc = _coord_desc(hp, n_final, iters=40, grid_pts=30)
                hc = _project(hc, n_final)
                sc = _score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc

    # Final
    best_h = _project(best_h, len(best_h))
    best_score = _score(best_h)
    return best_h, best_score, len(best_h)
